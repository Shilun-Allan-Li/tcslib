\section{Overview}

This chapter develops the Fourier--Walsh theory of Boolean functions
$f : \{0,1\}^n \to \bbr$.  The uniform probability measure on $\{0,1\}^n$
endows the space of Boolean functions with an $L^2$ inner product
$\langle f, g\rangle = \E[fg]$.  The Walsh characters
$\chi_S(x) = \prod_{i\in S}(-1)^{x_i}$, indexed by subsets $S\subseteq[n]$,
form an orthonormal basis for this space, yielding the \textbf{Walsh--Fourier
expansion} $f = \sum_S \hat f(S)\,\chi_S$.

The main results are:
\begin{itemize}
  \item Walsh expansion (Fourier inversion).
  \item Parseval's identity: $\|f\|^2 = \sum_S \hat f(S)^2$.
  \item Influence via Fourier: $\mathrm{Inf}_i[f] = \sum_{S\ni i}\hat f(S)^2$.
  \item Total influence: $I[f] = \sum_S |S|\,\hat f(S)^2$.
  \item Noise operator in the Fourier domain: $\widehat{T_\rho f}(S) = \rho^{|S|}\hat f(S)$.
\end{itemize}

\textbf{Reference:} Ryan O'Donnell, \emph{Analysis of Boolean Functions},
Cambridge University Press, 2014.

\label{BooleanAnalysis.BoolCube}
\leanok
The \emph{Boolean hypercube} of dimension $n$ is $\{0,1\}^n$,
represented as the function type $\mathrm{Fin}\,n \to \mathrm{Bool}$.

\lean{BooleanAnalysis.BooleanFunc}

\leanok

\end{definition}

\section{Uniform measure and expectation}

\begin{definition}[Uniform weight]
\lean{BooleanAnalysis.uniformWeight}
\label{BooleanAnalysis.uniformWeight}
\leanok
The uniform probability measure on $\{0,1\}^n$ assigns weight
$2^{-n}$ to each point.
\end{definition}

\begin{definition}[Expectation]
\lean{BooleanAnalysis.expect}
\label{BooleanAnalysis.expect}
\leanok
\uses{BooleanAnalysis.uniformWeight, BooleanAnalysis.BooleanFunc}
The \emph{expectation} of $f$ under the uniform measure:
\[
  \E[f] \;=\; 2^{-n}\sum_{x\in\{0,1\}^n} f(x).
\]
\end{definition}

\begin{definition}[$L^2$ inner product]
\lean{BooleanAnalysis.innerProduct}
\label{BooleanAnalysis.innerProduct}
\leanok
\statementsource{boolean-ch08-generalized-domains}{pdf-50833afe7403-p001-b004}
\uses{BooleanAnalysis.expect}
The \emph{$L^2$ inner product} of two Boolean functions with respect to the
uniform measure:
\[
  \langle f, g\rangle \;=\; \E[f\cdot g]
  \;=\; 2^{-n}\sum_{x\in\{0,1\}^n} f(x)\,g(x).
\]
\end{definition}

\begin{definition}[$L^2$ norm]
\lean{BooleanAnalysis.l2Norm}
\label{BooleanAnalysis.l2Norm}
\leanok
\uses{BooleanAnalysis.innerProduct}
The \emph{$L^2$ norm} of $f$: $\|f\| = \sqrt{\langle f,f\rangle}$.
\end{definition}

\begin{lemma}[Symmetry of the $L^2$ inner product]
\lean{BooleanAnalysis.innerProduct_comm}
\label{BooleanAnalysis.innerProduct_comm}
\leanok
\uses{BooleanAnalysis.innerProduct}
\difficulty{1}
For any two Boolean functions $f, g : \{0,1\}^n \to \bbr$, the $L^2$ inner product with
respect to the uniform measure is symmetric:
\[
  \langle f, g\rangle = \langle g, f\rangle.
\]
\end{lemma}

\begin{lemma}[Nonnegativity of the $L^2$ norm squared]
\lean{BooleanAnalysis.innerProduct_self_nonneg}
\label{BooleanAnalysis.innerProduct_self_nonneg}
\leanok
\uses{BooleanAnalysis.innerProduct}
\difficulty{2}
Let $f : \{0,1\}^n \to \bbr$ be a Boolean function of arity $n$. Then its $L^2$ inner
product with itself is nonnegative:
\[
  \langle f, f\rangle \;=\; 2^{-n}\sum_{x\in\{0,1\}^n} f(x)^2 \;\ge\; 0.
\]
\end{lemma}

\begin{lemma}[Additivity of the inner product in the first argument]
\lean{BooleanAnalysis.innerProduct_add_left}
\label{BooleanAnalysis.innerProduct_add_left}
\leanok
\uses{BooleanAnalysis.innerProduct}
\difficulty{2}
Let $n$ be a natural number and let $f, g, h : \{0,1\}^n \to \bbr$ be Boolean functions.
With respect to the $L^2$ inner product under the uniform measure, $\langle f, g\rangle
= 2^{-n}\sum_{x\in\{0,1\}^n} f(x)\,g(x)$, one has
\[
  \langle f + g,\, h\rangle \;=\; \langle f, h\rangle + \langle g, h\rangle,
\]
where $f + g$ denotes the pointwise sum.
\end{lemma}

\section{Walsh--Fourier characters}

\begin{definition}[Sign encoding]
\lean{BooleanAnalysis.boolToSign}
\label{BooleanAnalysis.boolToSign}
\leanok
The \emph{sign encoding} maps $\mathrm{Bool}$ to $\{-1,1\}\subset\bbr$:
\[
  \sigma(b) \;=\; \begin{cases} 1 & b = \texttt{false},\\
                                -1 & b = \texttt{true}. \end{cases}
\]
\end{definition}

\begin{lemma}[Sign encoding of false]
\lean{BooleanAnalysis.boolToSign_false}
\label{BooleanAnalysis.boolToSign_false}
\proofsource{boolean-ch02-social-choice-arrow}{pdf-352ab7ff3113-p002-b001}
\lean{BooleanAnalysis.boolToSign_true}
\label{BooleanAnalysis.boolToSign_true}
\proofsource{boolean-ch01-fourier-blr}{pdf-cdca27e1b5fd-p004-b004}
\lean{BooleanAnalysis.boolToSign_sq}
\label{BooleanAnalysis.boolToSign_sq}
\lean{BooleanAnalysis.boolToSign_mul_self}
\label{BooleanAnalysis.boolToSign_mul_self}
\lean{BooleanAnalysis.boolToSign_not}
\label{BooleanAnalysis.boolToSign_not}
\leanok
\uses{BooleanAnalysis.boolToSign}
\difficulty{1}
Under the sign encoding $\sigma\colon\mathrm{Bool}\to\{-1,1\}$, the Boolean value
$\mathrm{false}$ is sent to $1$; that is, $\sigma(\mathrm{false}) = 1$.

Under the sign encoding $\sigma\colon \mathrm{Bool}\to\{-1,1\}$, which sends the truth
value false to $1$ and true to $-1$, the truth value true is mapped to $-1$; that is,
$\sigma(\text{true}) = -1$.

For every Boolean value $b$, the sign encoding satisfies $\sigma(b)^2 = 1$.

For every $b \in \mathrm{Bool}$, the sign encoding $\sigma(b) \in \{-1,1\}$ satisfies
$\sigma(b)\cdot\sigma(b) = 1$; that is, $\sigma(b)^2 = 1$.

Let $\sigma\colon\mathrm{Bool}\to\{-1,1\}$ be the sign encoding, so that
$\sigma(\mathrm{false})=1$ and $\sigma(\mathrm{true})=-1$. Then for every Boolean value
$b$, negating $b$ flips the sign of its encoding:
\[
  \sigma(\lnot b) \;=\; -\,\sigma(b).
\]
\end{lemma}

\begin{definition}[Walsh character]
\lean{BooleanAnalysis.chiS}
\label{BooleanAnalysis.chiS}
\leanok
\statementsource{boolean-ch08-generalized-domains}{
  pdf-50833afe7403-p002-b007,
  pdf-50833afe7403-p003-b005
}
\uses{BooleanAnalysis.boolToSign}
The \emph{Walsh--Fourier character} associated to $S\subseteq[n]$ is
\[
  \chi_S(x) \;=\; \prod_{i\in S}(-1)^{x_i}
  \;=\; \prod_{i\in S}\sigma(x_i).
\]
The family $\{\chi_S\}_{S\subseteq[n]}$ forms an orthonormal basis for
$L^2(\{0,1\}^n,\mathrm{uniform})$.
\end{definition}

\begin{lemma}[Empty-set Walsh character]
\lean{BooleanAnalysis.chiS_empty}
\label{BooleanAnalysis.chiS_empty}
\leanok
\statementsource{boolean-ch01-fourier-blr}{pdf-cdca27e1b5fd-p004-b001}
\uses{BooleanAnalysis.chiS}
\difficulty{1}
The Walsh–Fourier character associated to the empty subset of $[n]$ is the constant
function equal to $1$; that is, $\chi_\emptyset(x) = 1$ for every point $x$ of the
Boolean hypercube $\{0,1\}^n$.
\end{lemma}

\begin{lemma}[Value of a singleton Walsh character]
\lean{BooleanAnalysis.chiS_singleton}
\label{BooleanAnalysis.chiS_singleton}
\leanok
\uses{BooleanAnalysis.chiS, BooleanAnalysis.boolToSign}
\difficulty{1}
Fix a dimension $n$, a coordinate $i \in [n]$, and a point $x$ of the Boolean hypercube
$\{0,1\}^n$. Then the Walsh--Fourier character of the singleton set $\{i\}$, evaluated
at $x$, is the sign encoding of the $i$-th coordinate of $x$:
\[
  \chi_{\{i\}}(x) \;=\; \sigma(x_i).
\]
\end{lemma}

\begin{lemma}[Walsh characters square to one]
\lean{BooleanAnalysis.chiS_sq_eq_one}
\label{BooleanAnalysis.chiS_sq_eq_one}
\leanok
\uses{BooleanAnalysis.chiS, BooleanAnalysis.boolToSign_sq}
\difficulty{4}
Let $n$ be a natural number, let $S \subseteq [n]$, and let $x \in \{0,1\}^n$ be a point
of the Boolean hypercube. Then the Walsh--Fourier character $\chi_S$ satisfies
$\chi_S(x)^2 = 1$; that is, every character takes values in $\{-1, 1\}$.
\end{lemma}

\begin{lemma}[Multiplicativity of Walsh characters on disjoint sets]
\lean{BooleanAnalysis.chiS_mul_of_disjoint}
\label{BooleanAnalysis.chiS_mul_of_disjoint}
\leanok
\proofsource{boolean-ch01-fourier-blr}{pdf-cdca27e1b5fd-p006-b004}
\uses{BooleanAnalysis.chiS}
\difficulty{1}
Let $n$ be a natural number, and let $S, T \subseteq [n]$ be disjoint subsets of
coordinates. Then for every point $x$ of the Boolean hypercube $\{0,1\}^n$, the
Walsh--Fourier character of the union factors as the product of the characters of the
two sets:
\[
  \chi_{S \cup T}(x) \;=\; \chi_S(x)\,\chi_T(x).
\]
\end{lemma}

\begin{lemma}[Walsh characters multiply by symmetric difference]
\lean{BooleanAnalysis.chiS_mul_chiS}
\label{BooleanAnalysis.chiS_mul_chiS}
\leanok
\statementsource{boolean-ch01-fourier-blr}{pdf-cdca27e1b5fd-p006-b004}
\uses{BooleanAnalysis.chiS}
\difficulty{5}
Fix $n$, and for a subset $S\subseteq[n]$ let $\chi_S\colon\{0,1\}^n\to\bbr$ denote the
associated Walsh--Fourier character. Then for all subsets $S,T\subseteq[n]$ and every
point $x\in\{0,1\}^n$,
\[
  \chi_S(x)\,\chi_T(x) \;=\; \chi_{S\,\triangle\,T}(x),
\]
where $S\,\triangle\,T$ is the symmetric difference of $S$ and $T$.
\end{lemma}

\begin{lemma}[Walsh character under global bit flip]
\lean{BooleanAnalysis.chiS_neg}
\label{BooleanAnalysis.chiS_neg}
\leanok
\uses{BooleanAnalysis.chiS, BooleanAnalysis.boolToSign_not}
\difficulty{4}
Let $n$ be a natural number, let $S \subseteq [n]$, and let $x \in \{0,1\}^n$. Writing
$\lnot x$ for the point obtained by flipping every coordinate of $x$, the Walsh--Fourier
character $\chi_S$ satisfies
\[
  \chi_S(\lnot x) \;=\; (-1)^{\abs{S}}\,\chi_S(x),
\]
where $\abs{S}$ is the number of elements of $S$.
\end{lemma}

\begin{definition}[Fourier coefficient]
\lean{BooleanAnalysis.fourierCoeff}
\label{BooleanAnalysis.fourierCoeff}
\leanok
\statementsource{boolean-ch08-generalized-domains}{pdf-50833afe7403-p005-b001}

The \emph{Fourier--Walsh coefficient} of $f$ at frequency $S$ is
\[
  \hat f(S) \;=\; \langle f, \chi_S\rangle
  \;=\; 2^{-n}\sum_{x\in\{0,1\}^n} f(x)\,\chi_S(x).
\]
\end{definition}

\lean{BooleanAnalysis.fourierCoeff_empty}

\[
  \hat f(\varnothing) \;=\; \E[f].
\]
\end{lemma}

\begin{theorem}[Walsh–Fourier expansion of a Boolean function]
\lean{BooleanAnalysis.walsh_expansion}
\label{BooleanAnalysis.walsh_expansion}
\leanok

\begin{definition}[Influence]
\lean{BooleanAnalysis.influence}
\label{BooleanAnalysis.influence}
\leanok
\statementsource{boolean-ch08-generalized-domains}{
  pdf-50833afe7403-p007-b006,
  pdf-50833afe7403-p008-b001
}
\uses{BooleanAnalysis.expect, BooleanAnalysis.flipBit}
The \emph{influence} of coordinate $i$ on $f$ measures how often flipping
bit $i$ changes the output:
\[
  \mathrm{Inf}_i[f]

\end{definition}

Let $n$ be an arity and let $i$ be one of its coordinates. For every scalar $c \in \bbr$
and every Boolean function $f : \{0,1\}^n \to \bbr$, applying the expectation operator
in direction $i$ to the scaled function $c\,f$ yields $c$ times the expectation operator
applied to $f$:
\[
  E_i(c\,f) = c\,E_i f.
\]
\end{lemma}

\begin{lemma}[Self-adjointness of the noise operator]
\lean{BooleanAnalysis.noiseOp_self_adjoint}
\statementsource{boolean-ch08-generalized-domains}{pdf-50833afe7403-p030-b001}
\label{BooleanAnalysis.noiseOp_self_adjoint}
\leanok
\uses{BooleanAnalysis.BooleanFunc, BooleanAnalysis.fourierCoeff, BooleanAnalysis.innerProduct, BooleanAnalysis.noiseOp, BooleanAnalysis.noiseOp_fourier, BooleanAnalysis.plancherel}
\difficulty{4}
Fix a dimension $n$ and let $f, g : \{0,1\}^n \to \bbr$ be Boolean functions. For any
real number $\rho$, the noise operator $T_\rho$ is self-adjoint with respect to the
$L^2$ inner product induced by the uniform measure on $\{0,1\}^n$:
\[
  \langle T_\rho f, g\rangle \;=\; \langle f, T_\rho g\rangle.
\]
\end{lemma}
