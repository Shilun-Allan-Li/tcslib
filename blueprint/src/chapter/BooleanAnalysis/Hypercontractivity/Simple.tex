\section{Overview}

This module develops the ``simple'' hypercontractivity results for real-valued
Boolean functions $f : \{0,1\}^n \to \bbr$, that is, the results obtainable by
direct induction on coordinates without the full general machinery. It builds up
the $(2,4)$, $(4/3,2)$, $(2,2k)$, and $(p,2)$ hypercontractivity inequalities for
the noise operator $T_\rho$, together with the supporting character, cardinality,
Fourier, and combinatorial lemmas used along the way.

\begin{lemma}[Walsh character is unchanged by appending an unused coordinate]
\lean{BooleanAnalysis.Hypercontractivity.chiS_snoc_castSucc}
\label{BooleanAnalysis.Hypercontractivity.chiS_snoc_castSucc}
\leanok
\uses{BooleanAnalysis.BoolCube, BooleanAnalysis.chiS}
\difficulty{3}
Fix $n\in\bbn$, and let $S\subseteq[n]$, $x\in\{0,1\}^n$, and $b\in\{0,1\}$. Form the
point $(x,b)\in\{0,1\}^{n+1}$ obtained by appending $b$ as the final coordinate, and
regard $S$ as a subset $S'\subseteq[n+1]$ via the inclusion $[n]\hookrightarrow[n+1]$
that leaves each index fixed (so the new last coordinate never lies in $S'$). Then the
Walsh--Fourier character of $S'$ evaluated at the extended point agrees with that of $S$
at the original point:
\[
  \chi_{S'}\big((x,b)\big) \;=\; \chi_S(x).
\]
\end{lemma}

\begin{lemma}[Walsh character on an extended coordinate set]
\lean{BooleanAnalysis.Hypercontractivity.chiS_snoc_with_last}
\label{BooleanAnalysis.Hypercontractivity.chiS_snoc_with_last}
\leanok
\uses{BooleanAnalysis.BoolCube, BooleanAnalysis.boolToSign, BooleanAnalysis.chiS}
\difficulty{3}
Let $S \subseteq [n]$, let $x \in \{0,1\}^n$, and let $b$ be a bit. Write $(x,b) \in
\{0,1\}^{n+1}$ for the point obtained by appending $b$ as the final coordinate, and let
$S' \subseteq [n+1]$ be the image of $S$ under the inclusion $[n] \hookrightarrow [n+1]$
of the first $n$ coordinates, together with the new last coordinate $n+1$. Then
\[
  \chi_{S'}(x,b) \;=\; \sigma(b)\,\chi_S(x),
\]
where $\sigma(0)=1$ and $\sigma(1)=-1$; that is, adjoining the last coordinate to the
lifted index set multiplies the Walsh character by the sign of $b$.
\end{lemma}

\begin{lemma}[Splitting a subset-sum by the last coordinate]
\lean{BooleanAnalysis.Hypercontractivity.finset_fin_succ_sum_partition}
\label{BooleanAnalysis.Hypercontractivity.finset_fin_succ_sum_partition}
\leanok
\difficulty{5}
Fix $n \in \bbn$ and identify a subset of $\{0, 1, \dots, n\}$ that omits the top
element $n$ with the corresponding subset of $\{0, 1, \dots, n-1\}$, via the inclusion
$\iota$ that sends $T \subseteq \{0,\dots,n-1\}$ to the same set regarded inside
$\{0,\dots,n\}$. Then for every function $\varphi$ assigning a real number to each
subset of $\{0, 1, \dots, n\}$,
\[
  \sum_{S \subseteq \{0,\dots,n\}} \varphi(S)
  = \sum_{T \subseteq \{0,\dots,n-1\}} \varphi\bigl(\iota(T)\bigr)
  + \sum_{T \subseteq \{0,\dots,n-1\}} \varphi\bigl(\iota(T) \cup \{n\}\bigr),
\]
the two terms collecting the subsets of $\{0,\dots,n\}$ that exclude and that include
the element $n$, respectively.
\end{lemma}

\begin{lemma}[Cardinality is preserved under the successor inclusion]
\lean{BooleanAnalysis.Hypercontractivity.card_image_castSucc}
\label{BooleanAnalysis.Hypercontractivity.card_image_castSucc}
\leanok
\difficulty{1}
Let $S \subseteq \{0, 1, \dots, n-1\}$ be a finite set of indices, and let $\iota \colon
\{0, 1, \dots, n-1\} \hookrightarrow \{0, 1, \dots, n\}$ be the order-preserving
inclusion that regards each index as an element of the enlarged range. Then the image
$\iota(S)$ has the same cardinality as $S$, that is, $\abs{\iota(S)} = \abs{S}$.
\end{lemma}

\begin{lemma}[Cardinality after lifting a set and adjoining the top coordinate]
\lean{BooleanAnalysis.Hypercontractivity.card_image_castSucc_union_last}
\label{BooleanAnalysis.Hypercontractivity.card_image_castSucc_union_last}
\leanok
\difficulty{2}
Let $n$ be a natural number and let $S \subseteq \{0, 1, \dots, n-1\}$. Embed $S$ into
$\{0, 1, \dots, n\}$ by the map that sends each index to itself, and adjoin the top
index $n$. Since $n$ is not the image of any element of $S$, this union satisfies
\[
\abs{\{\, i : i \in S \,\} \cup \{n\}} = \abs{S} + 1.
\]
\end{lemma}

\begin{lemma}[Noise operator splits along the last coordinate]
\lean{BooleanAnalysis.Hypercontractivity.noiseOp_snoc}
\label{BooleanAnalysis.Hypercontractivity.noiseOp_snoc}
\leanok
\uses{BooleanAnalysis.BoolCube, BooleanAnalysis.BooleanFunc, BooleanAnalysis.Hypercontractivity.avgLast, BooleanAnalysis.Hypercontractivity.card_image_castSucc, BooleanAnalysis.Hypercontractivity.card_image_castSucc_union_last, BooleanAnalysis.Hypercontractivity.chiS_snoc_castSucc, BooleanAnalysis.Hypercontractivity.chiS_snoc_with_last, BooleanAnalysis.Hypercontractivity.diffLast, BooleanAnalysis.Hypercontractivity.finset_fin_succ_sum_partition, BooleanAnalysis.Hypercontractivity.fourierCoeff_avgLast, BooleanAnalysis.Hypercontractivity.fourierCoeff_diffLast, BooleanAnalysis.boolToSign, BooleanAnalysis.chiS, BooleanAnalysis.fourierCoeff, BooleanAnalysis.noiseOp}
\difficulty{4}
Fix $\rho \in \bbr$ and a Boolean function $f \colon \{0,1\}^{n+1} \to \bbr$, and let
$Af$ and $Df$ be the average and the half-difference of $f$ over its last coordinate,
each a Boolean function on $\{0,1\}^n$, with
$Af(x)=(f(x,0)+f(x,1))/2$ and $Df(x)=(f(x,0)-f(x,1))/2$.
For every $x \in \{0,1\}^n$ and every bit $b$,
\[
  T_\rho f(x, b) \;=\; T_\rho(Af)(x) \;+\; \sigma(b)\,\rho\,T_\rho(Df)(x),
\]
where $(x,b) \in \{0,1\}^{n+1}$ is the point obtained by appending $b$ as the last
coordinate, $\sigma(0)=1$, $\sigma(1)=-1$, and $T_\rho$ is the noise operator.
\end{lemma}

\begin{lemma}[Fourth-moment decomposition under the noise operator]
\lean{BooleanAnalysis.Hypercontractivity.fourth_moment_noise_decomp}
\label{BooleanAnalysis.Hypercontractivity.fourth_moment_noise_decomp}
\leanok
\proofsource{boolean-ch09-hypercontractivity-kkl}{pdf-4ba63c9394e5-p007-b004}
\uses{BooleanAnalysis.BooleanFunc, BooleanAnalysis.Hypercontractivity.avgLast, BooleanAnalysis.Hypercontractivity.diffLast, BooleanAnalysis.Hypercontractivity.fourth_moment_decomp, BooleanAnalysis.Hypercontractivity.noiseOp_snoc, BooleanAnalysis.Hypercontractivity.restrictLast, BooleanAnalysis.expect, BooleanAnalysis.noiseOp}
\difficulty{4}
Let $f:\{0,1\}^{n+1}\to\bbr$ be a Boolean function, and let $\rho\in\bbr$. Write $g$ for
the average of $f$ over its last coordinate and $h$ for the corresponding
half-difference, so that
\[
  g(x)=\tfrac12\bigl(f(x,0)+f(x,1)\bigr),\qquad
  h(x)=\tfrac12\bigl(f(x,0)-f(x,1)\bigr)
\]
are Boolean functions of the first $n$ variables. Then the fourth moment of $T_\rho f$
under the uniform measure decomposes as
\[
  \E\bigl[(T_\rho f)^4\bigr]
  = \E\bigl[(T_\rho g)^4\bigr]
  + 6\rho^2\,\E\bigl[(T_\rho g)^2\,(T_\rho h)^2\bigr]
  + \rho^4\,\E\bigl[(T_\rho h)^4\bigr].
\]
\end{lemma}

Let $f:\{0,1\}^n\to\bbr$ be a Boolean function and let $\rho\in\bbr$ be a noise
parameter satisfying $\rho^2\le 1/3$. Then the fourth moment of the noised function is
controlled by the square of its second moment:
\[
  \E\bigl[(T_\rho f)^4\bigr]\;\le\;\bigl(\E[f^2]\bigr)^2,
\]
where $T_\rho$ is the noise operator and $\E$ denotes expectation under the uniform
measure on $\{0,1\}^n$. Equivalently, $\norm{T_\rho f}_4\le\norm{f}_2$.
\end{theorem}

\begin{lemma}[Hölder's inequality for the exponents $4/3$ and $4$]
\lean{BooleanAnalysis.Hypercontractivity.innerProduct_le_L43_L4}
\label{BooleanAnalysis.Hypercontractivity.innerProduct_le_L43_L4}
\leanok
\uses{BooleanAnalysis.BoolCube, BooleanAnalysis.BooleanFunc, BooleanAnalysis.expect, BooleanAnalysis.innerProduct, BooleanAnalysis.uniformWeight}
\difficulty{5}
Let $f, g : \{0,1\}^n \to \bbr$ be Boolean functions, and take all expectations with
respect to the uniform measure on $\{0,1\}^n$. Then their $L^2$ inner product satisfies
\[
  \langle f, g \rangle
  \;\le\;
  \bigl(\E[\,\abs{f}^{4/3}\,]\bigr)^{3/4}\,
  \bigl(\E[\,\abs{g}^{4}\,]\bigr)^{1/4}.
\]
\end{lemma}
