\chapter{Linear Codes and the Uniformity Lemma}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Overview}

This chapter develops the theory of linear codes and a key probabilistic tool:
for a uniformly random $n\times k$ generator matrix $G$ and any nonzero message
$x\in\alpha^k$, the output $Gx$ is uniformly distributed over $\alpha^n$.

\section{Distance equals minimum weight in linear codes}

\begin{theorem}[Minimum distance equals minimum nonzero weight]
\lean{ErrorCorrectingCodes.Codeword.Linear_Code_dist_eq_min_weight}
\label{ErrorCorrectingCodes.Codeword.Linear_Code_dist_eq_min_weight}
\leanok
\proofsource{ada-coding-book}{pdf-7251cb8e4bc5-p047-b004}
\uses{ErrorCorrectingCodes.Codeword.Linear_Code', ErrorCorrectingCodes.Codeword.weight, ErrorCorrectingCodes.Codeword.distance}
\difficulty{4}
Let $\alpha$ be a finite field and let $C$ be a linear code of length $n$ over $\alpha$,
meaning there is a generator matrix $G$ whose columns generate $C$ (every $G$-image of a
message word lies in $C$, and every codeword of $C$ is such an image). Suppose $d$ is
the minimum distance of $C$, that is, some two distinct codewords of $C$ lie at Hamming
distance exactly $d$ and no two distinct codewords lie closer. Then every nonzero
codeword of $C$ has Hamming weight at least $d$, and some codeword of $C$ has Hamming
weight exactly $d$, where the weight of a codeword is its Hamming distance from the
all-zero codeword.
\end{theorem}

\section{Uniform distributions and random matrices}

\begin{definition}[Uniform distribution on vectors]
\lean{ErrorCorrectingCodes.Codeword.uniform_vector_dist}
\label{ErrorCorrectingCodes.Codeword.uniform_vector_dist}
\leanok
\statementsource{ada-coding-book}{pdf-7251cb8e4bc5-p062-b003}
The uniform probability mass function on $\alpha^n$, encoded as a function
$\alpha^n\to\mathbb{R}$ assigning $1/|\alpha|^n$ to each vector.
\end{definition}

\begin{lemma}[Finiteness of the solution set of a linear matrix equation]
\lean{ErrorCorrectingCodes.Codeword.finite_matrix_dist}
\label{ErrorCorrectingCodes.Codeword.finite_matrix_dist}
\leanok
\difficulty{2}
Let $\alpha$ be a finite field, and let $n$ and $k$ be natural numbers. Fix a codeword
$v : \mathrm{Fin}\,n \to \alpha$ of length $n$ and a codeword $x : \mathrm{Fin}\,k \to
\alpha$ of length $k$. Then the set of matrices $G \in \alpha^{n \times k}$ satisfying
$Gx = v$ is finite.
\end{lemma}

\begin{definition}[Distribution induced by random matrices]
\lean{ErrorCorrectingCodes.Codeword.matrix_dist}
\label{ErrorCorrectingCodes.Codeword.matrix_dist}
\leanok
$\mu_x(v)$ is the fraction of $n\times k$ matrices $G$ satisfying $Gx=v$.
\end{definition}

\begin{definition}[Row extraction]
\lean{ErrorCorrectingCodes.Codeword.get_matrix_row}
\label{ErrorCorrectingCodes.Codeword.get_matrix_row}
\leanok
Utility returning the $i$-th row of a matrix as a $1\times k$ matrix.
\end{definition}

\difficulty{7}
Let $\alpha$ be a finite field, and fix $n \ge 0$ and $k \ge 1$. Let $x \in \alpha^k$ be
a nonzero vector. Then the distribution on $\alpha^n$ obtained by sampling an $n \times
k$ matrix $G$ with entries in $\alpha$ uniformly at random and returning the product
$Gx$ coincides with the uniform distribution on $\alpha^n$; that is, for every $v \in
\alpha^n$,
\[
\frac{\bigl|\{\,G \in \alpha^{n \times k} : Gx = v\,\}\bigr|}{\abs{\alpha}^{nk}} =
\frac{1}{\abs{\alpha}^{n}}.
\]
\end{theorem}
