\chapter{Johnson Bound}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Overview}

This chapter proves the \textbf{binary Johnson bound}: for a binary code $C$
of length $n$ with minimum distance $d$, if every codeword has weight at most
$w \le J_2(n,d)$ where
\[
J_2(n,d) = \frac{n}{2} - \sqrt{\frac{n(n-d)}{4}},
\]
then $|C| \le 2n$.

The proof reduces to a \emph{Rankin bound} from geometry: a set of unit
vectors in $\mathbb{R}^n$ with pairwise non-positive inner products has
size at most $2n$.  Codewords are embedded via a shifted $\{\pm1\}$ map so
that the distance and weight constraints force the inner-product condition.

\lean{A0}

$A_0(n,d,w)$ is the maximum size of an $(n,d,w)$-admissible code.
\end{definition}

\begin{theorem}[Johnson bound in radius form]
\lean{binary_johnson_bound_radius}
\label{binary_johnson_bound_radius}
\leanok
\uses{A0, CodingTheory.Johnson.binary_johnson_card_bound_of_admissible}
\difficulty{2}
Fix integers $n, d, w$ with $n \ge 1$, $1 \le d$, and $2d \le n$, and suppose that $w
\le J_2(n,d) = \tfrac{1}{2}\bigl(n - \sqrt{n\,(n-2d)}\bigr)$. Then the maximum size
$A_0(n,d,w)$ of an $(n,d,w)$-admissible binary code — a set of binary words of length
$n$ in which any two distinct words are at Hamming distance at least $d$ and every word
has weight at most $w$ — satisfies
\[
A_0(n,d,w) \;\le\; 2n.
\]
\end{theorem}

\begin{definition}[Hamming weight]

\end{definition}

\begin{definition}[Hamming distance]
\lean{CodingTheory.Johnson.hdist}
\label{CodingTheory.Johnson.hdist}
\leanok

For binary words $x,y$ of length $n$, the Hamming distance is the number of
coordinates on which they differ:
$\mathrm{hdist}(x,y) = \abs{\{\, i : x_i \neq y_i \,\}}$.
\end{definition}

\begin{lemma}[Coordinatewise product of $\pm 1$ embeddings]
\lean{CodingTheory.Johnson.coord_mul_pmOne}

\begin{lemma}[Inner product of two ±1 embeddings]
\lean{CodingTheory.Johnson.inner_pmOne_pmOne}
\label{CodingTheory.Johnson.inner_pmOne_pmOne}
\leanok
\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.coord_mul_pmOne, CodingTheory.Johnson.hdist, CodingTheory.Johnson.pmOne}
\difficulty{4}
For any two binary words $x, y$ of length $n$, let $\phi(x), \phi(y) \in \mathbb{R}^n$
be their $\pm 1$ embeddings, whose $i$-th coordinates are $-1$ when the corresponding
bit is true and $+1$ otherwise. Then their Euclidean inner product is
\[
\langle \phi(x),\, \phi(y)\rangle \;=\; n - 2\,d(x,y),
\]
where $d(x,y)$ is the Hamming distance, the number of coordinates on which $x$ and $y$
differ.
\end{lemma}

\begin{lemma}[Inner product of the ±1 embedding with the all-ones vector]
\lean{CodingTheory.Johnson.inner_pmOne_ones}

\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.ones, CodingTheory.Johnson.pmOne, CodingTheory.Johnson.wt}
\difficulty{4}

\langle s(x), \mathbf{1} \rangle = n - 2\,\mathrm{wt}(x).
\]
\end{lemma}

\begin{lemma}[Bilinear expansion of the shifted inner product]
\lean{CodingTheory.Johnson.inner_shifted_expand}
\label{CodingTheory.Johnson.inner_shifted_expand}
\leanok
\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.ones, CodingTheory.Johnson.pmOne, CodingTheory.Johnson.shifted}
\difficulty{3}
Let $\alpha \in \bbr$, let $x, y$ be binary words of length $n$, and write $\mathbf{1}$
for the all-ones vector in $\bbr^n$ and $s(x), s(y) \in \bbr^n$ for the $\pm 1$
embeddings of $x$ and $y$ (with $i$-th coordinate $-1$ where the corresponding bit is
true and $+1$ otherwise). Form the shifted vectors $\hat{x}^\alpha = s(x) - \alpha
\mathbf{1}$ and $\hat{y}^\alpha = s(y) - \alpha \mathbf{1}$. Then their Euclidean inner
product expands as
\[
\langle \hat{x}^\alpha,\, \hat{y}^\alpha \rangle
= \langle s(x),\, s(y) \rangle
- \alpha\,\langle s(x),\, \mathbf{1} \rangle
- \alpha\,\langle s(y),\, \mathbf{1} \rangle
+ \alpha^2\,\langle \mathbf{1},\, \mathbf{1} \rangle.
\]
\end{lemma}

\begin{lemma}[Upper bound on the shifted inner product]
\lean{CodingTheory.Johnson.inner_shifted_le_expr}
\label{CodingTheory.Johnson.inner_shifted_le_expr}
\leanok
\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.hdist, CodingTheory.Johnson.inner_ones_ones, CodingTheory.Johnson.inner_pmOne_ones, CodingTheory.Johnson.inner_pmOne_pmOne, CodingTheory.Johnson.inner_shifted_expand, CodingTheory.Johnson.shifted, CodingTheory.Johnson.wt}
\difficulty{4}
Let $x$ and $y$ be binary words of length $n$, and let $\alpha \ge 0$. Suppose $d$ and
$w$ are natural numbers for which the Hamming distance between $x$ and $y$ is at least
$d$ and the Hamming weights of $x$ and $y$ are each at most $w$. Writing
$\hat{x}^\alpha$ and $\hat{y}^\alpha$ for the corresponding shifted vectors in
$\mathbb{R}^n$ — the $\pm 1$ embedding of the word minus $\alpha$ times the all-ones

\[
\langle \hat{x}^\alpha, \hat{y}^\alpha \rangle \;\le\; (n - 2d) + \alpha^2 n +
2\alpha\,(2w - n).
\]
\end{lemma}

\begin{lemma}[Nonpositivity of inner products under projection off a unit vector]
\lean{CodingTheory.Johnson.inner_proj_le_zero}
\label{CodingTheory.Johnson.inner_proj_le_zero}
\leanok
\difficulty{3}
Let $V$ be a real inner product space, let $u \in V$ be a unit vector, and let $x, y \in
V$ satisfy $\langle x, u\rangle \le 0$, $\langle y, u\rangle \le 0$, and $\langle x,
y\rangle \le 0$. Then the components of $x$ and $y$ orthogonal to $u$ again have
nonpositive inner product:
\[
\langle\, x - \langle x, u\rangle\, u,\ y - \langle y, u\rangle\, u \,\rangle \le 0.
\]
\end{lemma}

\begin{lemma}[Shifted $\pm 1$ vectors are nonzero for $\alpha < 1$]
\lean{CodingTheory.Johnson.shifted_ne_zero_of_alpha_lt_one}
\label{CodingTheory.Johnson.shifted_ne_zero_of_alpha_lt_one}
\leanok
\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.Euc, CodingTheory.Johnson.ones, CodingTheory.Johnson.pmOne, CodingTheory.Johnson.shifted}
\difficulty{3}
Let $n \ge 1$ and let $\alpha$ be a real number with $0 \le \alpha < 1$. Then for every
binary word $x$ of length $n$, the shifted vector $\hat{x}^\alpha \in \bbr^n$ — obtained
from the $\pm 1$ embedding of $x$ by subtracting $\alpha$ times the all-ones vector — is
nonzero.
\end{lemma}

\begin{theorem}[Johnson cardinality bound, parametric form]
\lean{CodingTheory.Johnson.binary_johnson_card_bound_parametric}
\label{CodingTheory.Johnson.binary_johnson_card_bound_parametric}
\leanok
\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.Euc, CodingTheory.Johnson.hdist, CodingTheory.Johnson.inner_shifted_le_expr, CodingTheory.Johnson.normalize, CodingTheory.Johnson.rankin_finset_bound, CodingTheory.Johnson.shifted, CodingTheory.Johnson.wt}
\difficulty{5}
Let $n \ge 1$, let $d, w$ be natural numbers, and let $C$ be a finite set of binary
words of length $n$ such that any two distinct words of $C$ are at Hamming distance at
least $d$, and every word of $C$ has weight at most $w$. Suppose $\alpha \ge 0$ is a
real parameter for which the shifted $\pm 1$ vector $\hat{x}^\alpha$ is nonzero for
every $x \in C$, and for which
\[
(n - 2d) + \alpha^2 n + 2\alpha(2w - n) \le 0 .
\]
Then $\abs{C} \le 2n$.
\end{theorem}

\begin{theorem}[Binary Johnson cardinality bound]
\lean{CodingTheory.Johnson.binary_johnson_card_bound}
\label{CodingTheory.Johnson.binary_johnson_card_bound}
\leanok
\uses{CodingTheory.Johnson.BitVec, CodingTheory.Johnson.J2, CodingTheory.Johnson.alpha, CodingTheory.Johnson.alpha_lt_one_of_hd1, CodingTheory.Johnson.alpha_nonneg, CodingTheory.Johnson.binary_johnson_card_bound_parametric, CodingTheory.Johnson.hdist, CodingTheory.Johnson.johnson_arith, CodingTheory.Johnson.shifted, CodingTheory.Johnson.shifted_ne_zero_of_alpha_lt_one, CodingTheory.Johnson.wt}
\difficulty{3}
Let $n \ge 1$ and let $d$ be an integer with $1 \le d$ and $2d \le n$. Suppose $C$ is a
finite set of binary words of length $n$ such that any two distinct words of $C$ are at
Hamming distance at least $d$, and every word of $C$ has Hamming weight at most $w$. If
$w \le J_2(n,d)$, where $J_2(n,d) = \tfrac{1}{2}\bigl(n - \sqrt{n(n-2d)}\bigr)$ is the
binary Johnson radius, then $\abs{C} \le 2n$.
\end{theorem}

\begin{theorem}[Johnson cardinality bound for admissible constant-weight codes]
\lean{CodingTheory.Johnson.binary_johnson_card_bound_of_admissible}
\label{CodingTheory.Johnson.binary_johnson_card_bound_of_admissible}
\leanok
\uses{CodingTheory.Johnson.AdmissibleCode, CodingTheory.Johnson.BitVec, CodingTheory.Johnson.J2, CodingTheory.Johnson.binary_johnson_card_bound}
\difficulty{1}
Let $n \ge 1$, $d \ge 1$, and suppose $2d \le n$. If $C$ is an $(n,d,w)$-admissible code
— a finite set of binary words of length $n$ in which any two distinct words are at
Hamming distance at least $d$ and every word has weight at most $w$ — and the weight
bound satisfies $w \le J_2(n,d)$, where $J_2(n,d) = \tfrac{1}{2}\bigl(n -
\sqrt{n(n-2d)}\bigr)$, then $\abs{C} \le 2n$.
\end{theorem}

\begin{lemma}[Uniform bound transfers to the maximum admissible code size]
\lean{A0_le_of_forall_le}

\end{lemma}
