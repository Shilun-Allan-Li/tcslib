\chapter{List Decoding}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Overview}

This chapter defines list-decodable codes and proves the
\textbf{list-decoding capacity theorem}: for any $0<\rho\le 1-1/q$ and $L\ge 1$,
there exist codes of rate
\[
r \;=\; 1 - H_q(\rho) - \tfrac{1}{L}
\]
that are $(\rho,L)$-list-decodable.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{List-decodable codes}

\begin{definition}[List-decodable code]
\lean{ErrorCorrectingCodes.Codeword.list_decodable}
\label{ErrorCorrectingCodes.Codeword.list_decodable}
\leanok
A code $C$ is $(\rho,L)$-list-decodable if every Hamming ball of radius
$\lfloor\rho n\rfloor$ contains at most $L$ codewords of $C$.
\end{definition}

\section{Counting lemmas}

\begin{lemma}[Existence of list-decodable codes]
\lean{ErrorCorrectingCodes.Codeword.exists_listDecodable_code}
\label{ErrorCorrectingCodes.Codeword.exists_listDecodable_code}
\leanok
\uses{ErrorCorrectingCodes.Codeword.list_decodable, ErrorCorrectingCodes.Codeword.hamming_ball}
\difficulty{6}
Let $\alpha$ be a finite field, and fix natural numbers $n$, $L$, $M$ and a real number
$p$ with $0 \le p \le 1$, $1 \le L$, $L < M$, and $M \le \abs{\alpha}^n$. Suppose $V$ is
a natural number bounding the volume of every Hamming ball of radius $\lfloor
pn\rfloor$, in the sense that each such ball around an arbitrary word $y \in \alpha^n$
contains at most $V$ codewords, and suppose that
\[
\abs{\alpha}^n \cdot \binom{V}{L+1} \cdot \binom{\abs{\alpha}^n - (L+1)}{\,M-(L+1)\,}
\;<\; \binom{\abs{\alpha}^n}{M}.
\]
Then there exists a code $C$ of length $n$ over $\alpha$ consisting of exactly $M$
codewords that is $(p,L)$-list-decodable.
\end{lemma}

\begin{lemma}[Binomial ratio bound]
\lean{ErrorCorrectingCodes.Codeword.binom_ratio_bound}
\label{ErrorCorrectingCodes.Codeword.binom_ratio_bound}
\leanok
\difficulty{5}
Let $N$, $M$, and $k$ be natural numbers with $k \le M \le N$. Then the ratio of
binomial coefficients satisfies
\[
\frac{\binom{N-k}{M-k}}{\binom{N}{M}} \;\le\; \left(\frac{M}{N}\right)^{k},
\]
where the quantities are compared as real numbers.
\end{lemma}

\begin{lemma}[Counting inequality for list decoding]
\lean{ErrorCorrectingCodes.Codeword.listDecoding_counting_ineq}
\label{ErrorCorrectingCodes.Codeword.listDecoding_counting_ineq}
\leanok
\uses{ErrorCorrectingCodes.Codeword.binom_ratio_bound, ErrorCorrectingCodes.Codeword.qaryEntropy}
\difficulty{5}
Fix an integer alphabet size $q \ge 2$, a real number $p$, and integers $n$ and $L \ge
1$. Set the rate
\[
r \;=\; 1 - H_q(p) - \frac{1}{L},
\]
and define
\[
M \;=\; \bigl\lfloor q^{\,r n} \bigr\rfloor,
\qquad
V \;=\; \bigl\lfloor q^{\,H_q(p)\, n} \bigr\rfloor .
\]
Assume that $0 < M \le q^n$ and $L < M$. Then
\[
q^n \cdot \binom{V}{L+1} \cdot \binom{q^n - (L+1)}{\,M - (L+1)\,}
\;<\;
\binom{q^n}{M}.
\]
\end{lemma}

\section{List-decoding capacity}

\begin{theorem}[List-decoding over a q-ary alphabet]
\lean{ErrorCorrectingCodes.Codeword.list_decoding_capacity}
\label{ErrorCorrectingCodes.Codeword.list_decoding_capacity}
\leanok
\uses{ErrorCorrectingCodes.Codeword.exists_listDecodable_code, ErrorCorrectingCodes.Codeword.listDecoding_counting_ineq, ErrorCorrectingCodes.Codeword.qary_entropy_pos, ErrorCorrectingCodes.Codeword.hamming_ball_size_asymptotic_upper_bound}
\difficulty{5}
Let $\alpha$ be a finite field, write $q = \abs{\alpha}$ for the number of its elements,
and fix a real number $p$ with $0 < p \le 1 - \tfrac{1}{q}$ together with an integer $L
\ge 1$. In this regime one considers block codes over the alphabet $\alpha$ that admit
list decoding from a $p$-fraction of symbol errors with output lists of size at most
$L$, and the associated rate $r$ they can achieve.
\end{theorem}
