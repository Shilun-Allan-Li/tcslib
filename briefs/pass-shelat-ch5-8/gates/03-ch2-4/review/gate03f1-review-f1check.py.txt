from fractions import Fraction as F
from itertools import product
def isprime(x):
    return x>1 and all(x%d for d in range(2,int(x**0.5)+1))
def listToNat(w):
    a=0
    for c in w: a=2*a+(1 if c else 0)
    return a
def sampleBelowL(m,k,B,T,r):
    for t in range(T):
        blk=r[t*B:t*B+B][:k] if True else None
        blk=(r[t*B:])[:k]
        v=listToNat(blk)
        if v<m: return v
    return None
