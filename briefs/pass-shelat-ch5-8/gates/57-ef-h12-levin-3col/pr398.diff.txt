+/-- Reader for `LevinNPHard` (missing in `ZeroKnowledge/NP.lean`): `(L', R')` is NP-hard with
+Levin reductions iff every language with a witness relation has a Levin reduction to it,
+`LevinNPHard L' R' ↔ ∀ L R, WitnessRelation L R → ∃ f w p, LevinReduction L L' R R' f w p`. -/
+theorem levinNPHard_iff (L' : Language Bool) (R' : List Bool → List Bool → Prop) :
+    LevinNPHard L' R' ↔ ∀ (L : Language Bool) (R : List Bool → List Bool → Prop),
+      WitnessRelation L R → ∃ (f : List Bool → List Bool) (w : List Bool → List Bool → List Bool)
+        (p : PolyExpr), LevinReduction L L' R R' f w p := Iff.rfl

+# The circuit-to-3SAT step of the Levin reduction, with its witness map
+
+Step 2 of the route `(L, R) → (CKT-SAT, R_CKT) → (3SAT, R_3SAT) → (3COL, R_3COL)`: the
+instance map is the Tseitin map `BoolCircuit.dagCktSatToSAT3` of [AB09, Lemma 6.11, p. 111]
+(polynomial-time by `BoolCircuit.polyTimeComputable_dagCktSatToSAT3`), and the witness map
+sends a satisfying input `u` of the circuit described by `d` to the list of the values of all
+wires of the circuit on `u`, as [PS10, §4.7, Theorem 124.1, step 2] asks ("Cook's reduction
+can be applied to the witness").
+
+The wire values are computed with the circuit-value algorithm `BoolCircuit.CircuitEval.verdict`
+(no public all-wire evaluator exists: `CircuitEvalSpec.lean` and `CircuitEvalRun.lean` expose
+only the single-output verdict): vertex `v` of `C` is the output of the circuit `C` re-targeted
+at `v` (`withOutput`), whose description `retarget C.encode v` is the description of `C` with
+its last field `1^out 0` replaced by `1^v 0`. The prefix of the description before that field
+is read by a one-pass transducer on the reversed string (`descPrefix`).
+
+## Main definitions
+
+* `TCSlib.Cryptography.LevinHardness.descPrefix` — a description without its output field.
+* `TCSlib.Cryptography.LevinHardness.retarget` — a description with output field `1^v 0`.
+* `TCSlib.Cryptography.LevinHardness.withOutput` — a circuit re-targeted at vertex `v`.
+* `TCSlib.Cryptography.LevinHardness.padInput` — an input padded with `|d|` zeros.
+* `TCSlib.Cryptography.LevinHardness.wireValue`, `TCSlib.Cryptography.LevinHardness.wireValues`
+  — the value of one wire, and the list of all wire values.
+
+## Main results
+
+* `TCSlib.Cryptography.LevinHardness.wireValues_getD` — on a fan-in-two description the
+  computed list is the circuit's vertex values.
+* `TCSlib.Cryptography.LevinHardness.polyTimeComputable_wireValues` — the witness map is
+  polynomial-time.
+* `TCSlib.Cryptography.LevinHardness.R_3SAT_wireValues` — the witness map maps
+  `R_CKT` into `R_3SAT`.
+* `TCSlib.Cryptography.LevinHardness.exists_eval_of_R_3SAT` — a witness of the Tseitin image
+  of a fan-in-two circuit gives a satisfying input.
+
+**Disclosures.** The wire-value witness is the design of the EF-H12 brief. The proofs of
+`R_3SAT_wireValues` and `exists_eval_of_R_3SAT` follow the pattern of
+`BoolCircuit.DAGCircuit.satisfiable_iff_toCNF_satisfiable` (`DAGCircuitSat.lean:354`) and
+`Complexity.TQBFHard.toCNF_extension_iff` (`TQBFHardTseitin.lean:67`), with an explicit
+witness. `BoolCircuit.DAGCircuit.toCNF` has no published equation; the first theorem reads it
+through the quarantined reader `Bridge.toCNF_eq`,
+`C.toCNF = [(C.output, true)] :: tseitinGates n C.gates`.
+

+/-! ## The description without its output field -/
+
+/-- The transition of the prefix transducer on the reversed description: state `0` skips
+the last bit, state `1` skips the run of `1`s of the output field, state `2` copies. Glue. -/
+def stripNext (q : Fin 3) (b : Bool) : Fin 3 :=
+  if q = 0 then 1 else if q = 1 ∧ b = true then 1 else 2
+
+/-- Reader of `stripNext`: `stripNext q b = if q = 0 then 1 else if q = 1 ∧ b = true then 1
+else 2`. Glue. -/
+theorem stripNext_eq (q : Fin 3) (b : Bool) :
+    stripNext q b = if q = 0 then 1 else if q = 1 ∧ b = true then 1 else 2 := rfl
+
+/-- The output of the prefix transducer: nothing in states `0` and `1` on a `1`, otherwise the
+bit read. Glue. -/
+def stripOut (q : Fin 3) (b : Bool) : Option Bool :=
+  if q = 0 then none else if q = 1 ∧ b = true then none else some b
+
+/-- Reader of `stripOut`: `stripOut q b = if q = 0 then none else if q = 1 ∧ b = true then
+none else some b`. Glue. -/
+theorem stripOut_eq (q : Fin 3) (b : Bool) :
+    stripOut q b = if q = 0 then none else if q = 1 ∧ b = true then none else some b := rfl
+
+/-- **A description without its output field**: `descPrefix d` is the reverse of the
+transducer `transduce stripNext stripOut 0` run on `d.reverse`; on `C.encode` it is
+`encodeNat n ++ encodeList DAGGate.encode C.gates` (`descPrefix_encode`). Glue. -/
+def descPrefix (d : List Bool) : List Bool :=
+  (transduce stripNext stripOut 0 d.reverse).reverse
+
+/-- Reader of `descPrefix`:
+`descPrefix d = (transduce stripNext stripOut 0 d.reverse).reverse`. Glue. -/
+theorem descPrefix_eq (d : List Bool) :
+    descPrefix d = (transduce stripNext stripOut 0 d.reverse).reverse := rfl
+
+/-- A list code ends with its stop bit: `∃ X, encodeList f as = X ++ [false]`. Glue. -/
+private theorem encodeList_eq_append_false {α : Type} (f : α → List Bool) (as : List α) :
+    ∃ X, encodeList f as = X ++ [false] := by
+  induction as with
+  | nil => exact ⟨[], by rw [encodeList_nil]; rfl⟩
+  | cons a as ih =>
+    obtain ⟨X, hX⟩ := ih
+    exact ⟨true :: (f a ++ X), by rw [encodeList_cons, hX]; simp⟩
+
+/-- In state `1` the prefix transducer skips a run of `1`s. Glue. -/
+private theorem transduce_strip_ones (k : ℕ) (l : List Bool) :
+    transduce stripNext stripOut 1 (List.replicate k true ++ l) =
+      transduce stripNext stripOut 1 l := by
+  induction k with
+  | zero => rfl
+  | succ k ih =>
+    rw [List.replicate_succ, List.cons_append, transduce_cons, stripNext_eq, stripOut_eq]
+    simpa using ih
+
+/-- The prefix of a circuit description: `descPrefix C.encode = encodeNat n ++
+encodeList DAGGate.encode C.gates`. Glue.
+
+**Proof sketch.** The gate-list code ends with its stop bit `0`
+(`encodeList_eq_append_false`), so the reversed description is
+`0 · 1^out · 0 · R` with `R` the rest of the reversed prefix. The transducer drops the first
+`0` (state `0`), the run `1^out` (state `1`, `transduce_strip_ones`), emits the next `0` and
+copies `R` (state `2`, `Complexity.transduce_of_copy`); reversing gives the prefix. -/
+theorem descPrefix_encode {n : ℕ} (C : DAGCircuit n) :
+    descPrefix C.encode = encodeNat n ++ encodeList DAGGate.encode C.gates := by
+  obtain ⟨X, hX⟩ := encodeList_eq_append_false DAGGate.encode C.gates
+  set P := encodeNat n ++ encodeList DAGGate.encode C.gates with hP
+  have hrev : P.reverse = false :: (encodeNat n ++ X).reverse := by
+    rw [hP, hX, ← List.append_assoc, List.reverse_append]; rfl
+  have henc : C.encode = P ++ encodeNat C.output := by
+    rw [DAGCircuit.encode_parts, hP, List.append_assoc]
+  rw [descPrefix_eq, henc, List.reverse_append, hrev, encodeNat_def, List.reverse_append,
+    List.reverse_replicate]
+  have hcopy : transduce stripNext stripOut 2 (encodeNat n ++ X).reverse =
+      (encodeNat n ++ X).reverse :=
+    transduce_of_copy _ _ _ (fun b => by rw [stripNext_eq]; simp)
+      (fun b => by rw [stripOut_eq]; simp) _
+  simp only [List.reverse_cons, List.reverse_nil, List.nil_append, List.cons_append]
+  rw [transduce_cons, stripNext_eq, stripOut_eq]
+  simp only [if_true, Option.toList_none, List.nil_append]
+  rw [transduce_strip_ones, transduce_cons, stripNext_eq, stripOut_eq]
+  simp only [Fin.one_eq_zero_iff, OfNat.ofNat_ne_one, if_false, Bool.false_eq_true,
+    and_false, Option.toList_some, List.singleton_append, hcopy]
+  rw [← hrev, List.reverse_reverse]
+
+/-! ## Re-targeting a circuit at a vertex -/
+
+/-- **A description re-targeted at vertex `v`**: `retarget d v = descPrefix d ++ encodeNat v`,
+the description `d` with its output field replaced by `1^v 0`. Glue. -/
+def retarget (d : List Bool) (v : ℕ) : List Bool := descPrefix d ++ encodeNat v
+
+/-- Reader of `retarget`: `retarget d v = descPrefix d ++ encodeNat v`. Glue. -/
+theorem retarget_eq (d : List Bool) (v : ℕ) : retarget d v = descPrefix d ++ encodeNat v := rfl
+
+/-- On a circuit description, `retarget C.encode v = encodeNat n ++
+(encodeList DAGGate.encode C.gates ++ encodeNat v)`. Glue. -/
+theorem retarget_encode {n : ℕ} (C : DAGCircuit n) (v : ℕ) :
+    retarget C.encode v = encodeNat n ++ (encodeList DAGGate.encode C.gates ++ encodeNat v) := by
+  rw [retarget_eq, descPrefix_encode, List.append_assoc]
+
+/-- **The circuit `C` with output vertex `v`**, for `v < n + |C.gates|`: the same gates,
+`(withOutput C v hv).gates = C.gates` and `(withOutput C v hv).output = v`. Glue. -/
+def withOutput {n : ℕ} (C : DAGCircuit n) (v : ℕ) (hv : v < n + C.gates.length) :
+    DAGCircuit n :=
+  ⟨C.gates, v, C.args_lt, hv⟩
+
+/-- Reader of `withOutput`: its gates are `(withOutput C v hv).gates = C.gates`. Glue. -/
+theorem withOutput_gates {n : ℕ} (C : DAGCircuit n) (v : ℕ) (hv : v < n + C.gates.length) :
+    (withOutput C v hv).gates = C.gates := rfl
+
+/-- Reader of `withOutput`: its output is `(withOutput C v hv).output = v`. Glue. -/
+theorem withOutput_output {n : ℕ} (C : DAGCircuit n) (v : ℕ) (hv : v < n + C.gates.length) :
+    (withOutput C v hv).output = v := rfl
+
+/-- The description of the re-targeted circuit is the re-targeted description:
+`(withOutput C v hv).encode = retarget C.encode v`. Glue. -/
+theorem encode_withOutput {n : ℕ} (C : DAGCircuit n) (v : ℕ) (hv : v < n + C.gates.length) :
+    (withOutput C v hv).encode = retarget C.encode v := by
+  rw [DAGCircuit.encode_parts, retarget_encode, withOutput_gates, withOutput_output]
+
+/-- Re-targeting keeps fan-in two: `C.IsFaninTwo → (withOutput C v hv).IsFaninTwo`. Glue. -/
+theorem isFaninTwo_withOutput {n : ℕ} {C : DAGCircuit n} (hC : C.IsFaninTwo) (v : ℕ)
+    (hv : v < n + C.gates.length) : (withOutput C v hv).IsFaninTwo := by
+  rw [DAGCircuit.isFaninTwo_iff, withOutput_gates]
+  exact (DAGCircuit.isFaninTwo_iff C).mp hC
+
+/-- The re-targeted circuit computes vertex `v`:
+`(withOutput C v hv).eval x = (C.values x).getD v false`. Glue. -/
+theorem eval_withOutput {n : ℕ} (C : DAGCircuit n) (v : ℕ) (hv : v < n + C.gates.length)
+    (x : Fin n → Bool) : (withOutput C v hv).eval x = (C.values x).getD v false := by
+  rw [DAGCircuit.eval_eq, DAGCircuit.values_eq, withOutput_gates, withOutput_output,
+    ← DAGCircuit.values_eq]
+
+/-- A description of the parts `encodeNat m ++ (encodeList DAGGate.encode G ++ encodeNat o)`
+determines them: `n = m`, `C.gates = G` and `C.output = o`. Glue (the parts form of
+`BoolCircuit.DAGCircuit.eq_of_encode_eq`, for parts that need not form a circuit).
+
+**Proof sketch.** Decode the number of inputs (`decodeNat_encodeNat`), then the gate list
+(`decodeList_encodeList`), then the output from both sides of the equation, and compare. -/
+private theorem parts_of_encode_eq {n m : ℕ} {C : DAGCircuit n} {G : List DAGGate} {o : ℕ}
+    (h : C.encode = encodeNat m ++ (encodeList DAGGate.encode G ++ encodeNat o)) :
+    n = m ∧ C.gates = G ∧ C.output = o := by
+  rw [DAGCircuit.encode_parts] at h
+  have h1 := congrArg decodeNat h
+  rw [decodeNat_encodeNat, decodeNat_encodeNat] at h1
+  obtain ⟨hn, hr⟩ := Prod.mk.inj (Option.some.inj h1)
+  have hl₁ := length_encodeList_ge DAGGate.encode C.gates
+  have hl₂ := length_encodeList_ge DAGGate.encode G
+  have hlen := congrArg List.length hr
+  simp only [List.length_append] at hlen
+  have h2 := congrArg (decodeList DAGGate.decode
+    ((encodeList DAGGate.encode G ++ encodeNat o).length + 1)) hr
+  rw [decodeList_encodeList DAGGate.decode DAGGate.encode C.gates
+      (fun g _ r => DAGGate.decode_encode g r) _ (by simp only [List.length_append]; omega),
+    decodeList_encodeList DAGGate.decode DAGGate.encode G
+      (fun g _ r => DAGGate.decode_encode g r) _ (by simp only [List.length_append]; omega)]
+    at h2
+  obtain ⟨hg, ho⟩ := Prod.mk.inj (Option.some.inj h2)
+  have ho' := congrArg List.length ho
+  simp only [encodeNat_def, List.length_append, List.length_replicate,
+    List.length_singleton] at ho'
+  exact ⟨hn, hg, by omega⟩
+
+/-! ## The wire values -/
+
+/-- The input padded with `|d|` zeros: `padInput d u = u ++ List.replicate d.length false`
+(so that it has at least as many bits as the circuit `d` has inputs). Glue. -/
+def padInput (d u : List Bool) : List Bool := u ++ List.replicate d.length false
+
+/-- Reader of `padInput`: `padInput d u = u ++ List.replicate d.length false`. Glue. -/
+theorem padInput_eq (d u : List Bool) : padInput d u = u ++ List.replicate d.length false := rfl
+
+/-- Padding does not change the bits read with default `false`:
+`(padInput d u).getD i false = u.getD i false`. Glue. -/
+theorem padInput_getD (d u : List Bool) (i : ℕ) :
+    (padInput d u).getD i false = u.getD i false := by
+  rw [padInput_eq]
+  by_cases hi : i < u.length
+  · exact List.getD_append _ _ _ _ hi
+  · rw [List.getD_eq_default u false (by omega), List.getD_eq_getElem?_getD,
+      List.getElem?_append_right (by omega), List.getElem?_replicate]
+    split <;> rfl
+
+/-- **The value of wire `v`** of the circuit described by `d` on input `u`: the circuit-value
+verdict `CircuitEval.verdict false (pairEncode (retarget d v) (padInput d u))`. Glue. -/
+def wireValue (d u : List Bool) (v : ℕ) : Bool :=
+  CircuitEval.verdict false (pairEncode (retarget d v) (padInput d u))
+
+/-- Reader of `wireValue`:
+`wireValue d u v = CircuitEval.verdict false (pairEncode (retarget d v) (padInput d u))`.
+Glue. -/
+theorem wireValue_eq (d u : List Bool) (v : ℕ) :
+    wireValue d u v = CircuitEval.verdict false (pairEncode (retarget d v) (padInput d u)) :=
+  rfl
+
+/-- **The wire-value witness map** ([PS10, §4.7, Theorem 124.1, step 2]): the values of the
+vertices `0, …, |d| - 1`, `wireValues d u = (List.range d.length).map (wireValue d u)`. -/
+def wireValues (d u : List Bool) : List Bool := (List.range d.length).map (wireValue d u)
+
+/-- Reader of `wireValues`: `wireValues d u = (List.range d.length).map (wireValue d u)`. -/
+theorem wireValues_eq (d u : List Bool) :
+    wireValues d u = (List.range d.length).map (wireValue d u) := rfl
+
+/-- On a fan-in-two description, wire `v` carries vertex `v`'s value:
+`wireValue C.encode u v = (C.values (fun i => u.getD i false)).getD v false`.
+
+**Proof sketch.** By the algorithm's correctness `CircuitEval.verdict_pairEncode`, the verdict
+is `true` iff some fan-in-two circuit `C'` on `n' ≤ |padInput|` inputs is described by
+`retarget C.encode v` and outputs `1` on the padded input. That description is
+`1^n 0 · gates(C) · 1^v 0`, so `C'` has `n` inputs, the gates of `C` and output `v`
+(`parts_of_encode_eq`), and its output is vertex `v` of `C` (padding is invisible,
+`padInput_getD`). Conversely, if vertex `v` of `C` is `1` then `v` is a vertex
+(`length_values`), and `withOutput C v` is such a `C'`
+(`encode_withOutput`, `isFaninTwo_withOutput`, `eval_withOutput`); it has
+`n ≤ |C.encode| ≤ |padInput|` inputs (`DAGCircuit.size_le_length_encode`). -/
+theorem wireValue_encode {n : ℕ} (C : DAGCircuit n) (hC : C.IsFaninTwo) (u : List Bool)
+    (v : ℕ) : wireValue C.encode u v = (C.values (fun i => u.getD i false)).getD v false := by
+  have hx : (fun i : Fin n => (padInput C.encode u).getD i false) =
+      fun i : Fin n => u.getD i false := funext fun i => padInput_getD _ _ _
+  apply Bool.eq_iff_iff.mpr
+  rw [wireValue_eq, CircuitEval.verdict_pairEncode]
+  constructor
+  · rintro ⟨n', C', -, hE, -, -, hev⟩
+    rw [retarget_encode] at hE
+    obtain ⟨rfl, hg, ho⟩ := parts_of_encode_eq hE
+    rw [DAGCircuit.eval_eq, DAGCircuit.values_eq, hg, ho, ← DAGCircuit.values_eq, hx] at hev
+    exact hev
+  · intro h
+    have hv : v < n + C.gates.length := by
+      by_contra hv
+      rw [List.getD_eq_default _ _ (by rw [DAGCircuit.length_values]; omega)] at h
+      exact Bool.false_ne_true h
+    have hs := DAGCircuit.size_le_length_encode C
+    rw [DAGCircuit.size_eq] at hs
+    refine ⟨n, withOutput C v hv, isFaninTwo_withOutput hC v hv, encode_withOutput C v hv,
+      ?_, fun h => absurd h Bool.false_ne_true, ?_⟩
+    · rw [padInput_eq, List.length_append, List.length_replicate]; omega
+    · rw [hx, eval_withOutput]; exact h
+
+/-- **The wire values are the circuit's vertex values** [AB09, Def 6.1]: for a fan-in-two
+`C : DAGCircuit n` and every `v`,
+`(wireValues C.encode u).getD v false = (C.values (fun i => u.getD i false)).getD v false`.
+
+**Proof sketch.** Below `|C.encode|` this is `wireValue_encode`; from `|C.encode|` on both
+sides are `false`, since the description is at least as long as the number of vertices
+(`DAGCircuit.size_le_length_encode`, `DAGCircuit.length_values`). -/
+theorem wireValues_getD {n : ℕ} (C : DAGCircuit n) (hC : C.IsFaninTwo) (u : List Bool)
+    (v : ℕ) :
+    (wireValues C.encode u).getD v false = (C.values (fun i => u.getD i false)).getD v false := by
+  rw [wireValues_eq]
+  by_cases hv : v < C.encode.length
+  · rw [List.getD_eq_getElem?_getD, List.getElem?_map, List.getElem?_range hv]
+    exact wireValue_encode C hC u v
+  · have hs := DAGCircuit.size_le_length_encode C
+    rw [DAGCircuit.size_eq] at hs
+    rw [List.getD_eq_default _ _ (by simp only [List.length_map, List.length_range]; omega),
+      List.getD_eq_default _ _ (by rw [DAGCircuit.length_values]; omega)]
+
+/-! ## The witness map is polynomial-time -/
+
+/-- `descPrefix` is polynomial-time: a transducer between two reversals. Glue. -/
+theorem polyTimeComputable_descPrefix : PolyTimeComputable descPrefix :=
+  (BinArith.polyTimeComputable_reverse.comp
+      ((polyTimeComputable_transduce stripNext stripOut 0).comp
+        BinArith.polyTimeComputable_reverse)).congr fun d => (descPrefix_eq d).symm
+
+/-- **The witness map is polynomial-time**:
+`PolyTimeComputable fun z => wireValues (pairFstD z) (pairSndD z)`.
+
+**Proof sketch.** `wireValues d u` concatenates, over `v < |d|`, the one-bit chunks
+`[wireValue d u v]` (`List.flatMap` of singletons), so the tabulation host
+`ComposeMachines.polyTimeComputable_flatMap_range` applies with the unary count `1^|d|`
+(`polyTimeComputable_unary`). The chunk at `⟨1^v, ⟨d, u⟩⟩` is the circuit-value machine
+(`CircuitEval.evalTM_computes`, time `12 (m + 1)²`) after pairing
+`retarget d v = descPrefix d ++ 1^v 0` (`polyTimeComputable_descPrefix`, the unary map and a
+constant) with `padInput d u = u ++ 0^|d|` (`GI.zeros_poly`). -/
+theorem polyTimeComputable_wireValues :
+    PolyTimeComputable fun z => wireValues (pairFstD z) (pairSndD z) := by
+  have hV : PolyTimeComputable fun z => [CircuitEval.verdict false z] :=
+    (polyTimeComputable_iff _).mpr ⟨CircuitEval.evalTM false, 12, 2,
+      (Turing.FinTM.computesFunInTime_iff _ _ _).mpr fun z => CircuitEval.evalTM_computes false z⟩
+  have hD : PolyTimeComputable fun w => pairFstD (pairSndD w) :=
+    polyTimeComputable_pairFstD.comp polyTimeComputable_pairSndD
+  have hU : PolyTimeComputable fun w => pairSndD (pairSndD w) :=
+    polyTimeComputable_pairSndD.comp polyTimeComputable_pairSndD
+  have hR : PolyTimeComputable fun w => retarget (pairFstD (pairSndD w)) (pairFstD w).length :=
+    ((polyTimeComputable_descPrefix.comp hD).append
+      ((polyTimeComputable_unary.comp polyTimeComputable_pairFstD).append
+        (polyTimeComputable_const [false]))).congr fun w => by
+      rw [retarget_eq, encodeNat_def]; rfl
+  have hP : PolyTimeComputable fun w => padInput (pairFstD (pairSndD w)) (pairSndD (pairSndD w)) :=
+    (hU.append (GI.zeros_poly.comp hD)).congr fun w => by
+      rw [padInput_eq]; rfl
+  have hE : PolyTimeComputable fun w =>
+      [wireValue (pairFstD (pairSndD w)) (pairSndD (pairSndD w)) (pairFstD w).length] :=
+    (hV.comp (hR.pairEncode hP)).congr fun _ => rfl
+  refine (SecureComputation.ComposeMachines.polyTimeComputable_flatMap_range
+    (E := fun z i => [wireValue (pairFstD z) (pairSndD z) i]) hE
+    (polyTimeComputable_unary.comp polyTimeComputable_pairFstD)).congr fun z => ?_
+  rw [wireValues_eq, List.map_eq_flatMap]
+
+/-! ## The witness relations along the Tseitin map -/
+
+/-- **The witness map maps `R_CKT` into `R_3SAT`** ([AB09, Lemma 6.11, p. 111];
+[PS10, §4.7, Theorem 124.1, step 2]):
+`R_CKT d u → R_3SAT (dagCktSatToSAT3 d) (wireValues d u)`.
+
+**Proof sketch.** `d = C.encode` for a fan-in-two `C` outputting `1` on `u`. The image is the
+serialization of `C.toCNF` (`TQBFHard.dagCktSatToSAT3_encode`), which decodes back
+(`CNF.decode_serialize`) to a 3-CNF (`DAGCircuit.toCNF_widthAtMost`). The assignment read from
+`wireValues C.encode u` is the vertex-value assignment (`wireValues_getD`), which satisfies the
+gate clauses (`DAGCircuit.eval_tseitinGates_values`) and the output clause, the output vertex
+being `1` (`DAGCircuit.eval_eq`); the Tseitin formula is read through the quarantined reader
+`Bridge.toCNF_eq`. Pattern of `DAGCircuit.satisfiable_iff_toCNF_satisfiable`. -/
+theorem R_3SAT_wireValues {d u : List Bool} (h : R_CKT d u) :
+    R_3SAT (dagCktSatToSAT3 d) (wireValues d u) := by
+  obtain ⟨n, C, hC, rfl, hev⟩ := (R_CKT_iff d u).mp h
+  rw [R_3SAT_iff, TQBFHard.dagCktSatToSAT3_encode C hC, CNF.decode_serialize]
+  refine ⟨C.toCNF_widthAtMost, ?_⟩
+  have hW : (fun v => (wireValues C.encode u).getD v false) =
+      fun v => (C.values (fun i => u.getD i false)).getD v false :=
+    funext (wireValues_getD C hC u)
+  rw [hW, Bridge.toCNF_eq, CNF.eval_cons,
+    C.eval_tseitinGates_values ((DAGCircuit.isFaninTwo_iff C).mp hC).2, Bool.and_true,
+    CNF.Clause.eval_cons, CNF.Clause.eval_nil, Bool.or_false, ← DAGCircuit.eval_eq, hev]
+  rfl
+
+/-- **The converse along the Tseitin map** [AB09, Lemma 6.11, p. 111]: for a fan-in-two
+`C : DAGCircuit n`, `(∃ a, R_3SAT (dagCktSatToSAT3 C.encode) a) → ∃ x, C.eval x = true`.
+
+**Proof sketch.** The image decodes to `C.toCNF` (`TQBFHard.dagCktSatToSAT3_encode`,
+`CNF.decode_serialize`), so the assignment read from `a` satisfies it; by the pointwise
+Tseitin lemma `TQBFHard.toCNF_extension_iff` (taking the assignment itself as the extension)
+the circuit outputs `1` on the input it assigns. -/
+theorem exists_eval_of_R_3SAT {n : ℕ} (C : DAGCircuit n) (hC : C.IsFaninTwo)
+    (h : ∃ a, R_3SAT (dagCktSatToSAT3 C.encode) a) : ∃ x : Fin n → Bool, C.eval x = true := by
+  obtain ⟨a, ha⟩ := h
+  rw [R_3SAT_iff, TQBFHard.dagCktSatToSAT3_encode C hC, CNF.decode_serialize] at ha
+  exact ⟨_, (TQBFHard.toCNF_extension_iff C ((DAGCircuit.isFaninTwo_iff C).mp hC).2
+    (fun v => a.getD v false)).mp
+    ⟨_, fun _ _ => rfl, ha.2⟩⟩
+
+/-- **The Tseitin map preserves witness existence both ways** ([AB09, Lemma 6.11, p. 111]): for
+a fan-in-two `C : DAGCircuit n`,
+`(∃ a, R_3SAT (dagCktSatToSAT3 C.encode) a) ↔ ∃ x : Fin n → Bool, C.eval x = true`.
+Glue: `exists_eval_of_R_3SAT`, and `R_3SAT_wireValues` at `u = List.ofFn x`. -/
+theorem exists_R_3SAT_iff
+    {n : ℕ} (C : DAGCircuit n) (hC : C.IsFaninTwo) :
+    (∃ a, R_3SAT (dagCktSatToSAT3 C.encode) a) ↔ ∃ x : Fin n → Bool, C.eval x = true := by
+  refine ⟨exists_eval_of_R_3SAT C hC, fun ⟨x, hx⟩ => ⟨_, R_3SAT_wireValues
+    ((R_CKT_iff _ (List.ofFn x)).mpr ⟨n, C, hC, rfl, ?_⟩)⟩⟩
+  have : (fun i : Fin n => (List.ofFn x).getD i false) = x := funext fun i => by
+    rw [List.getD_eq_getElem?_getD, List.getElem?_ofFn]; simp
+  rw [this]; exact hx
+

+/-!
+# The string-level 3SAT → 3COL instance map
+
+The instance map of the Levin reduction `(3SAT, R_3SAT) → (3COL, R_3COL)` ([AB09, §2.3]; the
+gadget of the course reduction `Complexity/NPReductions/ThreeSATToColoring.lean`, course material
+of CS 294-268, UC Berkeley, Spring 2026), as an explicit map on binary strings.
+
+A string `s` is read as the formula `CNF.decode s`, whose variables are bounded by `K = |s|`. A
+literal `(v, b)` has code `litCode (v, b) = 2v + (if b then 0 else 1)`; a clause of one, two or
+three literals is widened to three literals by repeating its last one (`wideLit`), and has clause
+code `ℓ₁ + 2K ℓ₂ + (2K)² ℓ₃` (`clauseCode`). When every clause of `CNF.decode s` has one to three
+literals (`goodS`), the instance is the adjacency matrix of the gadget graph
+`gadgetGraph |s| (memS s)` whose clause gadgets are switched on by the clause codes of the formula
+(`memS`); otherwise it is the adjacency matrix of the complete graph `k4` on four vertices, which
+is not 3-colourable.
+
+## Main definitions
+
+* `litCode`, `wideLit`, `clauseCode` — the codes of literals and widened clauses.
+* `memS`, `goodS` — the clause-code membership test and the width test of a string.
+* `k4` — the complete loopless graph on four vertices.
+* `colorInst` — the instance map.
+* `gadgetSimpleGraph` — the gadget graph as a Mathlib simple graph.
+* `litEquiv`, `clauseEquiv`, `vertexEquiv` — the explicit identification of the codes and of the
+  vertices `Fin (gadgetN K)` with the literals, clauses and vertices of the course reduction over
+  `Fin K`.
+
+## Main results
+
+* `memS_iff`, `goodS_iff` — the specifications of the two tests.
+* `clauseLit_code`, `clauseLit_clauseCode`, `clauseSat_clauseCode` — clause codes decode to the
+  widened clause, and a code is satisfied iff the clause is.
+* `clauseSat_of_memS_of_lt` — a satisfying assignment of `CNF.decode s` satisfies every member code.
+* `not_threeColorable_k4` — `k4` is not 3-colourable.
+* `encGraph_gadgetGraph` — the gadget adjacency matrix as a tabulation over `i < N²`.
+* `polyTimeComputable_gadgetAdj`, `polyTimeComputable_colorInst_of` — the instance map is
+  polynomial-time computable once the two tests `memS`, `goodS` are.
+
+## References
+
+* CS 294-268 course material (UC Berkeley, Spring 2026, V. Guruswami), the 3SAT → 3-colouring
+  reduction as formalized in `Complexity/NPReductions/ThreeSATToColoring.lean`.

+/-! ### Literal and clause codes -/
+
+/-- **The code of a literal** `(v, b)`: `litCode (v, b) = 2v + (if b then 0 else 1)` (even codes
+are the positive literals). -/
+def litCode (ℓ : Literal ℕ) : ℕ := 2 * ℓ.1 + if ℓ.2 then 0 else 1
+
+/-- The defining equation of `litCode`. -/
+theorem litCode_eq (ℓ : Literal ℕ) : litCode ℓ = 2 * ℓ.1 + if ℓ.2 then 0 else 1 := rfl
+
+/-- A literal on a variable `v < K` has a code `litCode ℓ < 2K`. Glue. -/
+theorem litCode_lt {K : ℕ} {ℓ : Literal ℕ} (h : ℓ.1 < K) : litCode ℓ < 2 * K := by
+  rw [litCode_eq]
+  split <;> omega
+
+/-- The truth value of the code of a literal is the truth value of the literal:
+`litVal val (litCode ℓ) = (val ℓ.1 == ℓ.2)`. Glue. -/
+theorem litVal_litCode (val : ℕ → Bool) (ℓ : Literal ℕ) :
+    litVal val (litCode ℓ) = (val ℓ.1 == ℓ.2) := by
+  obtain ⟨v, b⟩ := ℓ
+  rw [litVal_eq, litCode_eq]
+  cases b
+  · have h1 : (2 * v + 1) % 2 = 1 := by omega
+    have h2 : (2 * v + 1) / 2 = v := by omega
+    simp only [Bool.false_eq_true, ↓reduceIte, h1, h2, one_ne_zero]
+    cases val v <;> rfl
+  · have h1 : (2 * v + 0) % 2 = 0 := by omega
+    have h2 : (2 * v + 0) / 2 = v := by omega
+    simp only [↓reduceIte, h1, h2]
+    cases val v <;> rfl
+
+/-- **The `j`-th literal of the widening of a clause** to three literals: the clause's literal
+number `min j (|C| - 1)` (its last literal repeated), `wideLit C j = C.getD (min j (|C| - 1))
+(0, true)`. -/
+def wideLit (C : CNF.Clause ℕ) (j : ℕ) : Literal ℕ := C.getD (min j (C.length - 1)) (0, true)
+
+/-- The defining equation of `wideLit`. -/
+theorem wideLit_eq (C : CNF.Clause ℕ) (j : ℕ) :
+    wideLit C j = C.getD (min j (C.length - 1)) (0, true) := rfl
+
+/-- Every literal of the widening of a nonempty clause is a literal of the clause. Glue. -/
+theorem wideLit_mem {C : CNF.Clause ℕ} (hC : 1 ≤ C.length) (j : ℕ) : wideLit C j ∈ C := by
+  rw [wideLit_eq, List.getD_eq_getElem _ _ (by omega)]
+  exact List.getElem_mem _
+
+/-- Every literal of a clause of at most three literals occurs in its widening:
+`ℓ ∈ C → ∃ j < 3, wideLit C j = ℓ`. Glue. -/
+theorem exists_wideLit_eq {C : CNF.Clause ℕ} (hC : C.length ≤ 3) {ℓ : Literal ℕ}
+    (hℓ : ℓ ∈ C) : ∃ j < 3, wideLit C j = ℓ := by
+  obtain ⟨i, hi, rfl⟩ := List.getElem_of_mem hℓ
+  refine ⟨i, by omega, ?_⟩
+  rw [wideLit_eq, List.getD_eq_getElem _ _ (by omega)]
+  congr 1
+  omega
+
+/-- **The code of a clause** for `K` variables: the codes `ℓ₁, ℓ₂, ℓ₃` of its widening, combined
+as `clauseCode K C = ℓ₁ + 2K ℓ₂ + (2K)² ℓ₃`. -/
+def clauseCode (K : ℕ) (C : CNF.Clause ℕ) : ℕ :=
+  litCode (wideLit C 0) + 2 * K * litCode (wideLit C 1) + (2 * K) ^ 2 * litCode (wideLit C 2)
+
+/-- The defining equation of `clauseCode`. -/
+theorem clauseCode_eq (K : ℕ) (C : CNF.Clause ℕ) :
+    clauseCode K C = litCode (wideLit C 0) + 2 * K * litCode (wideLit C 1) +
+      (2 * K) ^ 2 * litCode (wideLit C 2) := rfl
+
+/-- **The literals of a three-digit code** `a + 2K b + (2K)² d` with `a, b, d < 2K` are its
+digits: `clauseLit K (a + 2K b + (2K)² d) j = [a, b, d]_j`. Glue.
+
+**Proof sketch.** Write `m = 2K`. The code is `a + m (b + m d)`, so its remainder mod `m` is `a`
+and its quotient by `m` is `b + m d`, whose remainder is `b`; its quotient by `m²` is `d` because
+`a + m b < m²`. -/
+theorem clauseLit_code {K a b d : ℕ} (ha : a < 2 * K) (hb : b < 2 * K) (hd : d < 2 * K)
+    (j : Fin 3) :
+    clauseLit K (a + 2 * K * b + (2 * K) ^ 2 * d) j = ![a, b, d] j := by
+  set m := 2 * K with hm
+  have hm0 : 0 < m := by omega
+  have hx : a + m * b + m ^ 2 * d = a + m * (b + m * d) := by ring
+  have hdiv : (a + m * (b + m * d)) / m = b + m * d := by
+    rw [Nat.add_mul_div_left _ _ hm0, Nat.div_eq_of_lt ha, zero_add]
+  have hlow : a + m * b < m ^ 2 := by nlinarith
+  rw [clauseLit_eq, ← hm, hx]
+  fin_cases j
+  · simp only [Fin.zero_eta, Fin.isValue, Fin.reduceEq, ↓reduceIte, pow_zero, Nat.div_one,
+      Nat.add_mul_mod_self_left, Nat.mod_eq_of_lt ha, Matrix.cons_val_zero]
+  · simp only [Fin.mk_one, Fin.isValue, Fin.reduceEq, ↓reduceIte, pow_one, hdiv,
+      Nat.add_mul_mod_self_left, Nat.mod_eq_of_lt hb, Matrix.cons_val_one, Matrix.cons_val_zero]
+  · simp only [Fin.reduceFinMk, Fin.isValue, ↓reduceIte, Matrix.cons_val_two,
+      Matrix.tail_cons, Matrix.head_cons]
+    rw [← hx, show a + m * b + m ^ 2 * d = (a + m * b) + m ^ 2 * d by ring,
+      Nat.add_mul_div_left _ _ (by positivity), Nat.div_eq_of_lt hlow, zero_add]
+
+/-- **A clause code decodes to the widened clause**: if every variable of the nonempty clause `C`
+is `< K`, then `clauseLit K (clauseCode K C) j = litCode (wideLit C j)`. Glue. -/
+theorem clauseLit_clauseCode {K : ℕ} {C : CNF.Clause ℕ} (hK : ∀ ℓ ∈ C, ℓ.1 < K)
+    (hC : 1 ≤ C.length) (j : Fin 3) :
+    clauseLit K (clauseCode K C) j = litCode (wideLit C j) := by
+  have h : ∀ i, litCode (wideLit C i) < 2 * K := fun i => litCode_lt (hK _ (wideLit_mem hC i))
+  rw [clauseCode_eq, clauseLit_code (h 0) (h 1) (h 2)]
+  fin_cases j <;> rfl
+
+/-- **A clause is true iff one of its literals is**: a clause evaluates to `true` under `val`
+iff one of its literals `(v, b)` has `val v = b`,
+`C.eval val = true ↔ ∃ ℓ ∈ C, (val ℓ.1 == ℓ.2) = true` (Std `CNF.Clause.eval_cons`). Glue. -/
+theorem clause_eval_eq_true_iff (val : ℕ → Bool) (C : CNF.Clause ℕ) :
+    C.eval val = true ↔ ∃ ℓ ∈ C, (val ℓ.1 == ℓ.2) = true := by
+  induction C with
+  | nil => simp [CNF.Clause.eval_nil]
+  | cons ℓ C ih => simp [CNF.Clause.eval_cons, ih]
+
+/-- **A formula is true iff all its clauses are**: a CNF evaluates to `true` under `val` iff
+every clause does, `φ.eval val = true ↔ ∀ C ∈ φ, C.eval val = true` (Std `CNF.eval_cons`).
+Glue. -/
+theorem cnf_eval_eq_true_iff (val : ℕ → Bool) (φ : CNF ℕ) :
+    φ.eval val = true ↔ ∀ C ∈ φ, C.eval val = true := by
+  induction φ with
+  | nil => simp [CNF.eval_nil]
+  | cons C φ ih => simp [CNF.eval_cons, ih]
+
+/-- **A clause code is satisfied iff its clause is**: for a clause `C` of one to three literals
+on variables `< K`, `clauseSat K val (clauseCode K C) = C.eval val`. Glue.
+
+**Proof sketch.** The three literal codes of `clauseCode K C` are the codes of the widened
+literals (`clauseLit_clauseCode`), whose truth values are those of the literals
+(`litVal_litCode`); the widened literals are literals of `C` (`wideLit_mem`) and every literal of
+`C` is one of them (`exists_wideLit_eq`). -/
+theorem clauseSat_clauseCode {K : ℕ} (val : ℕ → Bool) {C : CNF.Clause ℕ}
+    (hK : ∀ ℓ ∈ C, ℓ.1 < K) (h1 : 1 ≤ C.length) (h3 : C.length ≤ 3) :
+    clauseSat K val (clauseCode K C) = C.eval val := by
+  apply Bool.eq_iff_iff.mpr
+  rw [clause_eval_eq_true_iff, clauseSat_eq]
+  simp only [clauseLit_clauseCode hK h1, litVal_litCode, Bool.or_eq_true]
+  constructor
+  · rintro ((h | h) | h)
+    · exact ⟨_, wideLit_mem h1 0, h⟩
+    · exact ⟨_, wideLit_mem h1 1, h⟩
+    · exact ⟨_, wideLit_mem h1 2, h⟩
+  · rintro ⟨ℓ, hℓ, h⟩
+    obtain ⟨j, hj, rfl⟩ := exists_wideLit_eq h3 hℓ
+    interval_cases j
+    · exact Or.inl (Or.inl h)
+    · exact Or.inl (Or.inr h)
+    · exact Or.inr h
+
+/-! ### The two tests of a string -/
+
+/-- **The clause-membership test of a string**: `memS s c` holds when `c` is the code
+`clauseCode |s| C` of a clause `C` of `CNF.decode s` with `1 ≤ |C| ≤ 3`. -/
+def memS (s : List Bool) (c : ℕ) : Bool :=
+  (CNF.decode s).any fun C =>
+    decide (1 ≤ C.length) && decide (C.length ≤ 3) && decide (clauseCode s.length C = c)
+
+/-- **Specification of `memS`**: `memS s c = true ↔ ∃ C ∈ CNF.decode s, 1 ≤ C.length ∧
+C.length ≤ 3 ∧ clauseCode s.length C = c`. -/
+theorem memS_iff (s : List Bool) (c : ℕ) :
+    memS s c = true ↔ ∃ C ∈ CNF.decode s, 1 ≤ C.length ∧ C.length ≤ 3 ∧
+      clauseCode s.length C = c := by
+  simp [memS, and_assoc]
+
+/-- **The width test of a string**: `goodS s` holds when every clause of `CNF.decode s` has one
+to three literals. -/
+def goodS (s : List Bool) : Bool :=
+  (CNF.decode s).all fun C => decide (1 ≤ C.length) && decide (C.length ≤ 3)
+
+/-- **Specification of `goodS`**: `goodS s = true ↔ ∀ C ∈ CNF.decode s, 1 ≤ C.length ∧
+C.length ≤ 3`. -/
+theorem goodS_iff (s : List Bool) :
+    goodS s = true ↔ ∀ C ∈ CNF.decode s, 1 ≤ C.length ∧ C.length ≤ 3 := by
+  simp [goodS]
+
+/-- **A satisfying assignment satisfies every member code**: if every variable of
+`CNF.decode s` is `< |s|` and `(CNF.decode s).eval val = true`, then `memS s c = true` implies
+`clauseSat |s| val c = true` (the bridge from 3SAT witnesses to the gadget colouring). Glue. -/
+theorem clauseSat_of_memS_of_lt {s : List Bool} {val : ℕ → Bool} {c : ℕ}
+    (hK : ∀ C ∈ CNF.decode s, ∀ ℓ ∈ C, ℓ.1 < s.length)
+    (heval : (CNF.decode s).eval val = true) (hm : memS s c = true) :
+    clauseSat s.length val c = true := by
+  obtain ⟨C, hC, h1, h3, rfl⟩ := (memS_iff s c).1 hm
+  rw [clauseSat_clauseCode val (hK C hC) h1 h3]
+  exact (cnf_eval_eq_true_iff val _).1 heval C hC
+
+/-! ### The complete graph on four vertices -/
+
+/-- **The complete loopless graph on four vertices**: `k4 u v = decide (u ≠ v)`. -/
+def k4 : Graph 4 := fun u v => decide (u ≠ v)
+
+/-- The entries of `k4`: `k4 u v = decide (u ≠ v)`. -/
+theorem k4_apply (u v : Fin 4) : k4 u v = decide (u ≠ v) := rfl
+
+/-- **`k4` is not 3-colourable**: a proper colouring `Fin 4 → Fin 3` of the complete graph would
+be injective (pigeonhole, `Fintype.card_le_of_injective`). -/
+theorem not_threeColorable_k4 : ¬ ThreeColorable k4 := by
+  rw [threeColorable_iff]
+  rintro ⟨c, hc⟩
+  have hinj : Function.Injective c := by
+    intro u v huv
+    by_contra hne
+    exact (hc.ne (by rw [k4_apply]; exact decide_eq_true hne)) huv
+  have := Fintype.card_le_of_injective c hinj
+  simp at this
+
+/-! ### The instance map -/
+
+/-- **The 3SAT → 3COL instance map** on strings: the adjacency matrix of the gadget graph
+`gadgetGraph |s| (memS s)` when every clause of `CNF.decode s` has one to three literals
+(`goodS s`), and of `k4` otherwise:
+`colorInst s = if goodS s then encGraph (gadgetGraph |s| (memS s)) else encGraph k4`. -/
+def colorInst (s : List Bool) : List Bool :=
+  if goodS s then encGraph (gadgetGraph s.length (memS s)) else encGraph k4
+
+/-- The defining equation of `colorInst`. -/
+theorem colorInst_eq (s : List Bool) :
+    colorInst s = if goodS s then encGraph (gadgetGraph s.length (memS s)) else encGraph k4 :=
+  rfl
+
+/-! ### The adjacency matrix as a tabulation -/
+
+/-- **The gadget adjacency matrix is a tabulation over `i < N²`**: with `N = gadgetN K`,
+`encGraph (gadgetGraph K mem) = (List.range (N * N)).flatMap fun i =>
+[gadgetAdj K mem (i / N) (i % N)]` (row-major order, `encGraph_getD_nat`). Glue. -/
+theorem encGraph_gadgetGraph (K : ℕ) (mem : ℕ → Bool) :
+    encGraph (gadgetGraph K mem) = (List.range (gadgetN K * gadgetN K)).flatMap fun i =>
+      [gadgetAdj K mem (i / gadgetN K) (i % gadgetN K)] := by
+  set N := gadgetN K with hN
+  have hN0 : 0 < N := by rw [hN, gadgetN_eq]; omega
+  have hmap : ((List.range (N * N)).flatMap fun i => [gadgetAdj K mem (i / N) (i % N)]) =
+      (List.range (N * N)).map fun i => gadgetAdj K mem (i / N) (i % N) :=
+    List.map_eq_flatMap.symm
+  rw [hmap]
+  apply List.ext_getElem (by rw [length_encGraph, List.length_map, List.length_range])
+  intro i h1 h2
+  have hi : i < N * N := by rwa [length_encGraph] at h1
+  have hu : i / N < N := (Nat.div_lt_iff_lt_mul hN0).2 hi
+  have hv : i % N < N := Nat.mod_lt _ hN0
+  have hget := encGraph_getD_nat (gadgetGraph K mem) ⟨i / N, hu⟩ ⟨i % N, hv⟩
+  rw [gadgetGraph_apply] at hget
+  rw [show i % N + gadgetN K * (i / N) = i from Nat.mod_add_div i N] at hget
+  rw [List.getElem_map, List.getElem_range, ← hget, List.getD_eq_getElem _ _ h1]

+/-! ### Polynomial time -/
+
+/-- Glue: the in-gadget index-pair test of `gadgetEdge` as a Boolean combination of equalities. -/
+private theorem decide_mem_gadgetPairs (a b : ℕ) :
+    decide ((a, b) ∈ [(0, 1), (0, 2), (1, 2), (2, 3), (3, 4), (3, 5), (4, 5)]) =
+      (decide (a = 0) && decide (b = 1) || decide (a = 0) && decide (b = 2) ||
+        decide (a = 1) && decide (b = 2) || decide (a = 2) && decide (b = 3) ||
+        decide (a = 3) && decide (b = 4) || decide (a = 3) && decide (b = 5) ||
+        decide (a = 4) && decide (b = 5)) := by
+  simp only [List.mem_cons, Prod.mk.injEq, List.mem_nil_iff, or_false]
+  simp only [Bool.decide_or, Bool.decide_and]
+  simp only [Bool.or_assoc]
+
+/-- **The gadget adjacency is a polynomial-time test** of unary data: if `K`, `t`, `t'` are unary
+functions and the membership test `mem x (c x)` is polynomial-time for every unary `c`, then
+`x ↦ [gadgetAdj (K x) (mem x) (t x) (t' x)]` is polynomial-time computable.
+
+**Proof sketch.** Expand `gadgetAdj_eq` and `gadgetEdge_eq` into `&&`/`||` combinations
+(`polyTimeComputable_and`, `polyTimeComputable_or`, `polyTimeComputable_not`) of comparisons of
+unary quantities (`EncodingMachines.ltTest_poly`, `leTest_poly`, `eqTest_poly`), built from `K`,
+`t`, `t'` and constants by the unary rows (`unaryAdd_poly`, `unarySub_poly`, `unaryMul_poly`,
+`divNum_poly`, `modNum_poly`), the literal codes `clauseLit` and the membership test. -/
+theorem polyTimeComputable_gadgetAdj {K t t' : List Bool → ℕ} {mem : List Bool → ℕ → Bool}
+    (hK : PolyTimeComputable fun x => List.replicate (K x) true)
+    (ht : PolyTimeComputable fun x => List.replicate (t x) true)
+    (ht' : PolyTimeComputable fun x => List.replicate (t' x) true)
+    (hmem : ∀ c : List Bool → ℕ, (PolyTimeComputable fun x => List.replicate (c x) true) →
+      PolyTimeComputable fun x => [mem x (c x)]) :
+    PolyTimeComputable fun x => [gadgetAdj (K x) (mem x) (t x) (t' x)] := by
+  open EncodingMachines in
+  have hN : PolyTimeComputable fun x => List.replicate (gadgetN (K x)) true := by
+    have hcube : PolyTimeComputable fun x => List.replicate ((2 * K x) ^ 3) true := by
+      simpa only [PolyExpr.eval_pow, PolyExpr.eval_var] using
+        unaryPolyOf_poly (.pow .var 3) (unaryMul_poly (unaryConst_poly 2) hK)
+    simp only [gadgetN_eq]
+    exact unaryAdd_poly (unaryAdd_poly (unaryConst_poly 3) (unaryMul_poly (unaryConst_poly 2) hK))
+      (unaryMul_poly (unaryConst_poly 6) hcube)
+  have hpairs : ∀ a b : List Bool → ℕ, (PolyTimeComputable fun x => List.replicate (a x) true) →
+      (PolyTimeComputable fun x => List.replicate (b x) true) →
+      PolyTimeComputable fun x =>
+        [decide ((a x, b x) ∈ [(0, 1), (0, 2), (1, 2), (2, 3), (3, 4), (3, 5), (4, 5)])] := by
+    intro a b ha hb
+    simp only [decide_mem_gadgetPairs]
+    repeat' first
+      | apply polyTimeComputable_or | apply polyTimeComputable_and
+      | exact EncodingMachines.eqTest_poly ha (EncodingMachines.unaryConst_poly _)
+      | exact EncodingMachines.eqTest_poly hb (EncodingMachines.unaryConst_poly _)
+  have h02 : ∀ a : List Bool → ℕ, (PolyTimeComputable fun x => List.replicate (a x) true) →
+      PolyTimeComputable fun x => [decide (a x = 0 ∨ a x = 2)] := by
+    intro a ha
+    simp only [Bool.decide_or]
+    exact polyTimeComputable_or (EncodingMachines.eqTest_poly ha (unaryConst_poly 0))
+      (EncodingMachines.eqTest_poly ha (unaryConst_poly 2))
+  have hE : ∀ u u' : List Bool → ℕ, (PolyTimeComputable fun x => List.replicate (u x) true) →
+      (PolyTimeComputable fun x => List.replicate (u' x) true) →
+      PolyTimeComputable fun x => [gadgetEdge (K x) (mem x) (u x) (u' x)] := by
+    intro u u' hu hu'
+    simp only [gadgetEdge_eq, ← divNum_eq, ← modNum_eq]
+    repeat' first
+      | exact hN | exact hK | exact hu | exact hu'
+      | apply divNum_poly | apply modNum_poly
+      | apply unaryConst_poly
+      | apply polyTimeComputable_or | apply polyTimeComputable_and
+      | apply hpairs | apply h02 | apply hmem
+      | apply clauseLit_poly
+      | apply ltTest_poly | apply leTest_poly | apply eqTest_poly
+      | apply unaryAdd_poly | apply unarySub_poly | apply unaryMul_poly
+  simp only [gadgetAdj_eq, decide_not]
+  exact polyTimeComputable_and (polyTimeComputable_not (eqTest_poly ht ht'))
+    (polyTimeComputable_or (hE t t' ht ht') (hE t' t ht' ht))
+
+/-- **The instance map is polynomial-time computable once its two tests are**: if
+`⟨1^c, s⟩ ↦ [memS s c]` and `s ↦ [goodS s]` are polynomial-time, then so is `colorInst`.
+
+**Proof sketch.** `colorInst` is `polyTimeComputable_ite` on `goodS` between the constant
+`encGraph k4` and the tabulation `encGraph_gadgetGraph`, which is
+`ComposeMachines.polyTimeComputable_flatMap_range` over `i < N²` (`N = gadgetN |s|` unary) of the
+one-bit chunk `polyTimeComputable_gadgetAdj` at `K = |s|`, `t = i / N`, `t' = i % N`; the
+membership test at a unary code `c` is the hypothesis composed with `⟨1^{c}, s⟩`. -/
+theorem polyTimeComputable_colorInst_of
+    (hmem : PolyTimeComputable fun w => [memS (pairSndD w) (pairFstD w).length])
+    (hgood : PolyTimeComputable fun s => [goodS s]) :
+    PolyTimeComputable colorInst := by
+  open EncodingMachines in
+  have hs : PolyTimeComputable fun w : List Bool => List.replicate (pairSndD w).length true :=
+    unaryLength_poly polyTimeComputable_pairSndD
+  have hi : PolyTimeComputable fun w : List Bool => List.replicate (pairFstD w).length true :=
+    unaryLength_poly polyTimeComputable_pairFstD
+  have hNu : ∀ {K : List Bool → ℕ}, (PolyTimeComputable fun x => List.replicate (K x) true) →
+      PolyTimeComputable fun x => List.replicate (gadgetN (K x)) true := by
+    intro K hK
+    have hcube : PolyTimeComputable fun x => List.replicate ((2 * K x) ^ 3) true := by
+      simpa only [PolyExpr.eval_pow, PolyExpr.eval_var] using
+        unaryPolyOf_poly (.pow .var 3) (unaryMul_poly (unaryConst_poly 2) hK)
+    simp only [gadgetN_eq]
+    exact unaryAdd_poly (unaryAdd_poly (unaryConst_poly 3) (unaryMul_poly (unaryConst_poly 2) hK))
+      (unaryMul_poly (unaryConst_poly 6) hcube)
+  have hchunk : PolyTimeComputable fun w =>
+      [gadgetAdj (pairSndD w).length (memS (pairSndD w))
+        ((pairFstD w).length / gadgetN (pairSndD w).length)
+        ((pairFstD w).length % gadgetN (pairSndD w).length)] := by
+    refine polyTimeComputable_gadgetAdj hs
+      ((divNum_poly hi (hNu hs)).congr fun w => by rw [divNum_eq])
+      ((modNum_poly hi (hNu hs)).congr fun w => by rw [modNum_eq]) ?_
+    intro c hc
+    refine (hmem.comp' (hc.pairEncode polyTimeComputable_pairSndD)).congr fun w => ?_
+    simp only [pairFstD_pairEncode, pairSndD_pairEncode, List.length_replicate]
+  have hG : PolyTimeComputable fun s : List Bool =>
+      (List.range (gadgetN s.length * gadgetN s.length)).flatMap fun i =>
+        [gadgetAdj s.length (memS s) (i / gadgetN s.length) (i % gadgetN s.length)] :=
+    SecureComputation.ComposeMachines.polyTimeComputable_flatMap_range hchunk
+      (unaryMul_poly (hNu polyTimeComputable_unary) (hNu polyTimeComputable_unary))
+  refine (polyTimeComputable_ite hgood hG (polyTimeComputable_const (encGraph k4))).congr
+    fun s => ?_
+  rw [colorInst_eq, encGraph_gadgetGraph]
+
+/-! ### The gadget graph as a simple graph -/
+
+/-- **The gadget graph as a Mathlib simple graph** on `Fin (gadgetN K)`: `u ~ v` iff
+`gadgetAdj K mem u v = true` (symmetric by `gadgetAdj_comm`, loopless by `gadgetAdj_self`). -/
+def gadgetSimpleGraph (K : ℕ) (mem : ℕ → Bool) : SimpleGraph (Fin (gadgetN K)) where
+  Adj u v := gadgetAdj K mem u v = true
+  symm u v h := by
+    change gadgetAdj K mem v u = true
+    rw [gadgetAdj_comm]
+    exact h
+  loopless u h := by
+    have h' : gadgetAdj K mem u u = true := h
+    rw [gadgetAdj_self] at h'
+    exact Bool.false_ne_true h'
+
+/-- The adjacency of `gadgetSimpleGraph`: `u ~ v ↔ gadgetAdj K mem u v = true`. -/
+theorem gadgetSimpleGraph_adj (K : ℕ) (mem : ℕ → Bool) (u v : Fin (gadgetN K)) :
+    (gadgetSimpleGraph K mem).Adj u v ↔ gadgetAdj K mem u v = true := Iff.rfl
+
+/-- **The gadget graph is 3-colourable iff its simple graph is**:
+`ThreeColorable (gadgetGraph K mem) ↔ (gadgetSimpleGraph K mem).Colorable 3` (a Mathlib
+`Coloring` into `Fin 3` is a proper 3-colouring). Glue. -/
+theorem threeColorable_gadgetGraph_iff (K : ℕ) (mem : ℕ → Bool) :
+    ThreeColorable (gadgetGraph K mem) ↔ (gadgetSimpleGraph K mem).Colorable 3 := by
+  rw [threeColorable_iff]
+  constructor
+  · rintro ⟨c, hc⟩
+    exact ⟨SimpleGraph.Coloring.mk c fun {u v} h =>
+      hc.ne (by rw [gadgetGraph_apply]; exact (gadgetSimpleGraph_adj K mem u v).1 h)⟩
+  · rintro ⟨C⟩
+    refine ⟨C, (ProperColoring_iff _ _).2 fun u v h => C.valid ?_⟩
+    rw [gadgetSimpleGraph_adj, ← gadgetGraph_apply]
+    exact h
+
+/-! ### The vertices of the course reduction -/
+
+/-- **Literal codes are the literals of the course reduction** over `Fin K`: the code `ℓ < 2K`
+is `pos (ℓ / 2)` for even `ℓ` and `neg (ℓ / 2)` for odd `ℓ`; inversely `pos v ↦ 2v`,
+`neg v ↦ 2v + 1`. -/
+def litEquiv (K : ℕ) : Fin (2 * K) ≃ SATtoColor.Literal (Fin K) where
+  toFun ℓ := if ℓ.val % 2 = 0 then .pos ⟨ℓ.val / 2, by omega⟩ else .neg ⟨ℓ.val / 2, by omega⟩
+  invFun
+    | .pos v => ⟨2 * v.val, by omega⟩
+    | .neg v => ⟨2 * v.val + 1, by omega⟩
+  left_inv ℓ := by
+    by_cases h : ℓ.val % 2 = 0
+    · simp only [h, ↓reduceIte]; ext; simp only; omega
+    · simp only [h, ↓reduceIte]; ext; simp only; omega
+  right_inv
+    | .pos v => by
+        simp only [Nat.mul_mod_right, ↓reduceIte, SATtoColor.Literal.pos.injEq]; ext; simp
+    | .neg v => by
+        have h : (2 * v.val + 1) % 2 ≠ 0 := by omega
+        simp only [h, ↓reduceIte, SATtoColor.Literal.neg.injEq]; ext; simp only; omega
+
+/-- The defining equation of `litEquiv`. -/
+theorem litEquiv_apply (K : ℕ) (ℓ : Fin (2 * K)) :
+    litEquiv K ℓ = if ℓ.val % 2 = 0 then .pos ⟨ℓ.val / 2, by omega⟩
+      else .neg ⟨ℓ.val / 2, by omega⟩ := rfl
+
+/-- The code of a positive literal: `(litEquiv K).symm (pos v) = 2v`. -/
+theorem litEquiv_symm_pos (K : ℕ) (v : Fin K) :
+    (((litEquiv K).symm (.pos v) : Fin (2 * K)) : ℕ) = 2 * v.val := rfl
+
+/-- The code of a negative literal: `(litEquiv K).symm (neg v) = 2v + 1`. -/
+theorem litEquiv_symm_neg (K : ℕ) (v : Fin K) :
+    (((litEquiv K).symm (.neg v) : Fin (2 * K)) : ℕ) = 2 * v.val + 1 := rfl
+
+/-- Glue: the three literal codes of `c` are `c % 2K`, `c / 2K % 2K` and `c / (2K)²`
+(`clauseLit_eq`). -/
+private theorem clauseLit_vals (K c : ℕ) :
+    clauseLit K c 0 = c % (2 * K) ∧ clauseLit K c 1 = c / (2 * K) % (2 * K) ∧
+      clauseLit K c 2 = c / (2 * K) ^ 2 := by
+  refine ⟨?_, ?_, ?_⟩ <;> rw [clauseLit_eq] <;> simp
+
+/-- **Every natural number is the three-digit code of its `clauseLit` digits**:
+`clauseLit K c 0 + 2K clauseLit K c 1 + (2K)² clauseLit K c 2 = c`. Glue. -/
+theorem clauseLit_decomp (K c : ℕ) :
+    clauseLit K c 0 + 2 * K * clauseLit K c 1 + (2 * K) ^ 2 * clauseLit K c 2 = c := by
+  obtain ⟨e0, e1, e2⟩ := clauseLit_vals K c
+  rw [e0, e1, e2]
+  set m := 2 * K
+  have h1 := Nat.mod_add_div c m
+  have h2 := Nat.mod_add_div (c / m) m
+  have h3 : c / m ^ 2 = c / m / m := by rw [pow_two, Nat.div_div_eq_div_mul]
+  rw [h3]
+  calc c % m + m * (c / m % m) + m ^ 2 * (c / m / m)
+      = c % m + m * (c / m % m + m * (c / m / m)) := by ring
+    _ = c := by rw [h2, h1]
+
+/-- The literal codes of a clause code `c < (2K)³` are literal codes: `clauseLit K c j < 2K`.
+Glue. -/
+theorem clauseLit_lt {K c : ℕ} (hc : c < (2 * K) ^ 3) (j : Fin 3) : clauseLit K c j < 2 * K := by
+  have hm : 0 < 2 * K := by
+    rcases Nat.eq_zero_or_pos (2 * K) with h | h
+    · rw [h] at hc; simp at hc
+    · exact h
+  obtain ⟨e0, e1, e2⟩ := clauseLit_vals K c
+  fin_cases j
+  · exact e0 ▸ Nat.mod_lt _ hm
+  · exact e1 ▸ Nat.mod_lt _ hm
+  · rw [show ((⟨2, by omega⟩ : Fin 3)) = 2 from rfl, e2, Nat.div_lt_iff_lt_mul (by positivity)]
+    calc c < (2 * K) ^ 3 := hc
+      _ = 2 * K * (2 * K) ^ 2 := by ring
+
+/-- **A three-digit code with digits `< m` is `< m³`**: `a + m b + m² d < m³`. Glue. -/
+theorem code_lt {m a b d : ℕ} (ha : a < m) (hb : b < m) (hd : d < m) :
+    a + m * b + m ^ 2 * d < m ^ 3 := by
+  have h1 : a + m * b < m ^ 2 := by nlinarith
+  have h2 : m ^ 2 * d + m ^ 2 ≤ m ^ 3 := by
+    calc m ^ 2 * d + m ^ 2 = m ^ 2 * (d + 1) := by ring
+      _ ≤ m ^ 2 * m := Nat.mul_le_mul_left _ hd
+      _ = m ^ 3 := by ring
+  omega
+
+/-- **Clause codes are below `(2K)³`**: if every variable of the nonempty clause `C` is `< K`,
+then `clauseCode K C < (2K)³` (`code_lt` on the three literal codes `< 2K`). Glue. -/
+theorem clauseCode_lt {K : ℕ} {C : CNF.Clause ℕ} (hK : ∀ ℓ ∈ C, ℓ.1 < K) (hC : 1 ≤ C.length) :
+    clauseCode K C < (2 * K) ^ 3 := by
+  have h : ∀ i, litCode (wideLit C i) < 2 * K := fun i => litCode_lt (hK _ (wideLit_mem hC i))
+  rw [clauseCode_eq]
+  exact code_lt (h 0) (h 1) (h 2)
+
+/-- **Clause codes are the clauses of the course reduction** over `Fin K`: the code
+`c < (2K)³` is the clause whose literals are `litEquiv` of `clauseLit K c 0, 1, 2`; inversely
+`⟨ℓ₁, ℓ₂, ℓ₃⟩ ↦ ℓ₁ + 2K ℓ₂ + (2K)² ℓ₃` on the literal codes. -/
+def clauseEquiv (K : ℕ) : Fin ((2 * K) ^ 3) ≃ SATtoColor.Clause (Fin K) where
+  toFun c := ⟨litEquiv K ⟨clauseLit K c 0, clauseLit_lt c.isLt 0⟩,
+    litEquiv K ⟨clauseLit K c 1, clauseLit_lt c.isLt 1⟩,
+    litEquiv K ⟨clauseLit K c 2, clauseLit_lt c.isLt 2⟩⟩
+  invFun C := ⟨((litEquiv K).symm C.l1 : ℕ) + 2 * K * ((litEquiv K).symm C.l2 : ℕ) +
+      (2 * K) ^ 2 * ((litEquiv K).symm C.l3 : ℕ),
+    code_lt ((litEquiv K).symm C.l1).isLt ((litEquiv K).symm C.l2).isLt
+      ((litEquiv K).symm C.l3).isLt⟩
+  left_inv c := by
+    ext
+    simp only [Equiv.symm_apply_apply]
+    exact clauseLit_decomp K c
+  right_inv C := by
+    obtain ⟨l1, l2, l3⟩ := C
+    have ha := ((litEquiv K).symm l1).isLt
+    have hb := ((litEquiv K).symm l2).isLt
+    have hd := ((litEquiv K).symm l3).isLt
+    simp only [SATtoColor.Clause.mk.injEq]
+    refine ⟨?_, ?_, ?_⟩
+    · conv_rhs => rw [← (litEquiv K).apply_symm_apply l1]
+      congr 1; ext; simpa using clauseLit_code ha hb hd 0
+    · conv_rhs => rw [← (litEquiv K).apply_symm_apply l2]
+      congr 1; ext; simpa using clauseLit_code ha hb hd 1
+    · conv_rhs => rw [← (litEquiv K).apply_symm_apply l3]
+      congr 1; ext; simpa using clauseLit_code ha hb hd 2
+
+/-- The defining equation of `clauseEquiv`: the literals of the clause of code `c` are
+`litEquiv K` of `clauseLit K c 0, 1, 2`. -/
+theorem clauseEquiv_apply (K : ℕ) (c : Fin ((2 * K) ^ 3)) :
+    clauseEquiv K c = ⟨litEquiv K ⟨clauseLit K c 0, clauseLit_lt c.isLt 0⟩,
+      litEquiv K ⟨clauseLit K c 1, clauseLit_lt c.isLt 1⟩,
+      litEquiv K ⟨clauseLit K c 2, clauseLit_lt c.isLt 2⟩⟩ := rfl
+
+/-- The code of a clause: `(clauseEquiv K).symm C = ℓ₁ + 2K ℓ₂ + (2K)² ℓ₃` for the literal codes
+`ℓᵢ = (litEquiv K).symm C.lᵢ`. -/
+theorem clauseEquiv_symm_val (K : ℕ) (C : SATtoColor.Clause (Fin K)) :
+    (((clauseEquiv K).symm C : Fin ((2 * K) ^ 3)) : ℕ) =
+      ((litEquiv K).symm C.l1 : ℕ) + 2 * K * ((litEquiv K).symm C.l2 : ℕ) +
+        (2 * K) ^ 2 * ((litEquiv K).symm C.l3 : ℕ) := rfl
+
+/-- **The literal codes of the code of a clause** are the codes of its literals:
+`clauseLit K ((clauseEquiv K).symm C) j = (litEquiv K).symm (![C.l1, C.l2, C.l3] j)`. Glue. -/
+theorem clauseLit_clauseEquiv_symm (K : ℕ) (C : SATtoColor.Clause (Fin K)) (j : Fin 3) :
+    clauseLit K ((clauseEquiv K).symm C) j = ((litEquiv K).symm (![C.l1, C.l2, C.l3] j) : ℕ) := by
+  rw [clauseEquiv_symm_val, clauseLit_code ((litEquiv K).symm C.l1).isLt
+    ((litEquiv K).symm C.l2).isLt ((litEquiv K).symm C.l3).isLt]
+  fin_cases j <;> rfl
+
+/-- Glue: the clause index of a gadget vertex is a clause code. -/
+private theorem gadgetIdx_lt {K t : ℕ} (hL : 3 + 2 * K ≤ t) (ht : t < gadgetN K) :
+    (t - (3 + 2 * K)) / 6 < (2 * K) ^ 3 := by
+  rw [gadgetN_eq] at ht
+  generalize (2 * K) ^ 3 = X at ht ⊢
+  omega
+
+/-- Glue: the vertex of a gadget node lies below `gadgetN K`. -/
+private theorem gadgetVertex_lt {K c i : ℕ} (hc : c < (2 * K) ^ 3) (hi : i < 6) :
+    3 + 2 * K + 6 * c + i < gadgetN K := by
+  rw [gadgetN_eq]
+  generalize (2 * K) ^ 3 = X at hc ⊢
+  omega
+
+/-- **The vertices of the gadget graph are the vertices of the course reduction** over `Fin K`:
+`t < 3` is the palette vertex `t`; `3 ≤ t < 3 + 2K` is the literal node of code `t - 3`; and
+`t = 3 + 2K + 6c + i` (`i < 6`) is gadget node `i` of the clause of code `c`. -/
+def vertexEquiv (K : ℕ) : Fin (gadgetN K) ≃ SATtoColor.OutputVertex (Fin K) where
+  toFun t :=
+    if h : t.val < 3 then .palette ⟨t.val, h⟩
+    else if h' : t.val < 3 + 2 * K then .literalNode (litEquiv K ⟨t.val - 3, by omega⟩)
+    else .clauseGadget (clauseEquiv K ⟨(t.val - (3 + 2 * K)) / 6, gadgetIdx_lt (by omega) t.isLt⟩)
+      ⟨(t.val - (3 + 2 * K)) % 6, by omega⟩
+  invFun
+    | .palette p => ⟨p.val, by rw [gadgetN_eq]; omega⟩
+    | .literalNode l => ⟨3 + ((litEquiv K).symm l).val, by
+        have := ((litEquiv K).symm l).isLt
+        rw [gadgetN_eq]
+        omega⟩
+    | .clauseGadget c i => ⟨3 + 2 * K + 6 * ((clauseEquiv K).symm c).val + i.val,
+        gadgetVertex_lt ((clauseEquiv K).symm c).isLt i.isLt⟩
+  left_inv t := by
+    by_cases h1 : t.val < 3
+    · simp only [h1, ↓reduceDIte]
+    · by_cases h2 : t.val < 3 + 2 * K
+      · simp only [h1, h2, ↓reduceDIte, Equiv.symm_apply_apply]
+        ext; simp only; omega
+      · simp only [h1, h2, ↓reduceDIte, Equiv.symm_apply_apply]
+        ext; simp only; omega
+  right_inv v := by
+    rcases v with p | l | ⟨c, i⟩
+    · have h1 : p.val < 3 := p.isLt
+      simp only [h1, ↓reduceDIte]
+    · have hl := ((litEquiv K).symm l).isLt
+      have h1 : ¬ 3 + ((litEquiv K).symm l).val < 3 := by omega
+      have h2 : 3 + ((litEquiv K).symm l).val < 3 + 2 * K := by omega
+      simp only [h1, h2, ↓reduceDIte, SATtoColor.OutputVertex.literalNode.injEq]
+      conv_rhs => rw [← (litEquiv K).apply_symm_apply l]
+      congr 1; ext; simp only; omega
+    · have hi := i.isLt
+      have h1 : ¬ 3 + 2 * K + 6 * ((clauseEquiv K).symm c).val + i.val < 3 := by omega
+      have h2 : ¬ 3 + 2 * K + 6 * ((clauseEquiv K).symm c).val + i.val < 3 + 2 * K := by omega
+      simp only [h1, h2, ↓reduceDIte, SATtoColor.OutputVertex.clauseGadget.injEq]
+      constructor
+      · conv_rhs => rw [← (clauseEquiv K).apply_symm_apply c]
+        congr 1; ext; simp only; omega
+      · ext; simp only; omega
+
+/-- The defining equation of `vertexEquiv` (see its docstring for the three ranges). -/
+theorem vertexEquiv_apply (K : ℕ) (t : Fin (gadgetN K)) :
+    vertexEquiv K t =
+      if h : t.val < 3 then .palette ⟨t.val, h⟩
+      else if h' : t.val < 3 + 2 * K then .literalNode (litEquiv K ⟨t.val - 3, by omega⟩)
+      else .clauseGadget (clauseEquiv K ⟨(t.val - (3 + 2 * K)) / 6, gadgetIdx_lt (by omega) t.isLt⟩)
+        ⟨(t.val - (3 + 2 * K)) % 6, by omega⟩ := rfl
+
+/-- The vertex of a palette node: `(vertexEquiv K).symm (palette p) = p`. -/
+theorem vertexEquiv_symm_palette (K : ℕ) (p : Fin 3) :
+    (((vertexEquiv K).symm (.palette p) : Fin (gadgetN K)) : ℕ) = p.val := rfl
+
+/-- The vertex of a literal node: `(vertexEquiv K).symm (literalNode l) = 3 + (litEquiv K).symm l`.
+-/
+theorem vertexEquiv_symm_literalNode (K : ℕ) (l : SATtoColor.Literal (Fin K)) :
+    (((vertexEquiv K).symm (.literalNode l) : Fin (gadgetN K)) : ℕ) =
+      3 + ((litEquiv K).symm l : ℕ) := rfl
+
+/-- The vertex of a gadget node: `(vertexEquiv K).symm (clauseGadget c i) =
+3 + 2K + 6 ((clauseEquiv K).symm c) + i`. -/
+theorem vertexEquiv_symm_clauseGadget (K : ℕ) (c : SATtoColor.Clause (Fin K)) (i : Fin 6) :
+    (((vertexEquiv K).symm (.clauseGadget c i) : Fin (gadgetN K)) : ℕ) =
+      3 + 2 * K + 6 * ((clauseEquiv K).symm c : ℕ) + i.val := rfl

+# Parsing tests of the 3SAT → 3COL instance map
+
+The two string tests of the instance map `colorInst` (`LevinHardness/ColoringGraph.lean`) of
+the Levin reduction `3SAT → 3COL` ([AB09, §2.3]; the course reduction
+`Complexity/NPReductions/ThreeSATToColoring.lean`, CS 294-268, UC Berkeley, Spring 2026) are
+polynomial-time, so the instance map is.
+
+The serialization `Std.Sat.CNF.serialize` is read only through the readers of
+`LevinHardness/Bridge.lean`. A position `p` of a well-formed string `s` is a *clause boundary*
+when the grammar automaton `Std.Sat.CNF.syntaxStep` is back in its formula state `0` after
+`s.take p`; every such prefix is the serialization of a list of clauses (`exists_recs_of_foldl`).
+
+* `memS s c` holds iff `s` is well-formed and, at some clause boundary, `s` continues with the
+  record `1 :: serializeClause C'` of one of the at most three clauses `C'` with clause code `c`
+  (`candClause`, `memS_eq`).
+* `goodS s` holds iff `s` is not well-formed or a finite automaton counting the literals of each
+  clause up to four (`widthStep`) never sees an empty or over-long clause (`goodS_eq`).
+
+## Main definitions
+
+* `litOfCode`, `candClause`, `candValid` — the clauses with a given clause code.
+* `clauseRec`, `boundary`, `memAlg` — clause records, clause boundaries, the string test of `memS`.
+* `widthStep` — the width automaton.
+
+## Main results
+
+* `var_lt_length_of_mem_decode`, `clauseSat_of_memS` — variables of `CNF.decode s` are `< |s|`;
+  the unconditional bridge to the colouring witness.
+* `polyTimeComputable_foldl_accept` — a finite automaton's verdict is polynomial-time.
+* `memS_eq`, `goodS_eq` — the string forms of the two tests.
+* `polyTimeComputable_memS`, `polyTimeComputable_goodS`, `polyTimeComputable_colorInst`.
+
+## References
+
+* CS 294-268 course material (UC Berkeley, Spring 2026, V. Guruswami), the 3SAT → 3-colouring
+  reduction as formalized in `Complexity/NPReductions/ThreeSATToColoring.lean`.
+* [AB09] S. Arora, B. Barak, *Computational Complexity: A Modern Approach*, Cambridge
+  University Press, 2009. (§2.3.)

+/-! ### The variable bound -/

+/-- Glue: a member of a list of naturals is at most its `foldr max 0`. -/
+private theorem le_foldr_max_zero {n : ℕ} {l : List ℕ} (h : n ∈ l) : n ≤ l.foldr max 0 := by
+  induction l with
+  | nil => cases h
+  | cons m l ih =>
+    rcases List.mem_cons.1 h with rfl | h
+    · exact le_max_left _ _
+    · exact (ih h).trans (le_max_right _ _)
+
+/-- **Variables of a decoded formula are below the string length**: for `C ∈ CNF.decode s` and
+`ℓ ∈ C`, `ℓ.1 < |s|` (`Bridge.numVars_eq`, `CNF.numVars_decode_le`). Glue. -/
+theorem var_lt_length_of_mem_decode {s : List Bool} {C : CNF.Clause ℕ}
+    (hC : C ∈ CNF.decode s) {ℓ : Literal ℕ} (hℓ : ℓ ∈ C) : ℓ.1 < s.length := by
+  have h1 : ℓ.1 + 1 ≤ (CNF.decode s).numVars := by
+    rw [Bridge.numVars_eq]
+    exact le_foldr_max_zero (List.mem_flatMap.2 ⟨C, hC, List.mem_map.2 ⟨ℓ, hℓ, rfl⟩⟩)
+  have := CNF.numVars_decode_le s
+  omega
+
+/-- **A satisfying assignment satisfies every member code**: if
+`(CNF.decode s).eval val = true` and `memS s c = true`, then `clauseSat |s| val c = true`
+(`clauseSat_of_memS_of_lt` with `var_lt_length_of_mem_decode`). Glue. -/
+theorem clauseSat_of_memS {s : List Bool} {val : ℕ → Bool} {c : ℕ}
+    (heval : (CNF.decode s).eval val = true) (hm : memS s c = true) :
+    clauseSat s.length val c = true :=
+  clauseSat_of_memS_of_lt (fun _ hC _ hℓ => var_lt_length_of_mem_decode hC hℓ) heval hm

+/-! ### Finite automata in polynomial time -/
+
+/-- Glue: the last output of the verdict transducer is the verdict on the whole word. -/
+private theorem getLastD_transduce_accept {σ : Type} (δ : σ → Bool → σ) (acc : σ → Bool)
+    (x : List Bool) (q : σ) (d : Bool) :
+    (transduce δ (fun q b => some (acc (δ q b))) q x).getLastD d =
+      if x = [] then d else acc (x.foldl δ q) := by
+  induction x generalizing q d with
+  | nil => rw [transduce_nil]; rfl
+  | cons b l ih =>
+    rw [transduce_cons, Option.toList_some, List.singleton_append, List.getLastD_cons, ih]
+    cases l <;> simp
+
+/-- **The verdict of a finite automaton is polynomial-time**: for a finite state type `σ`, a
+transition `δ`, a start state `q₀` and an acceptance test `acc`, the map
+`x ↦ [acc (x.foldl δ q₀)]` is polynomial-time computable.
+
+**Proof sketch.** The transducer emitting `acc (δ q b)` after every bit
+(`polyTimeComputable_transduce`) ends with the verdict on nonempty words
+(`getLastD_transduce_accept`), read by `BinArith.polyTimeComputable_lastBit`; the empty word is
+split off with `polyTimeComputable_isNil` and `polyTimeComputable_ite`. -/
+theorem polyTimeComputable_foldl_accept {σ : Type} [Fintype σ] [DecidableEq σ]
+    (δ : σ → Bool → σ) (q₀ : σ) (acc : σ → Bool) :
+    PolyTimeComputable fun x => [acc (x.foldl δ q₀)] := by
+  have ht := BinArith.polyTimeComputable_lastBit.comp'
+    (polyTimeComputable_transduce δ (fun q b => some (acc (δ q b))) q₀)
+  refine (polyTimeComputable_ite polyTimeComputable_isNil (polyTimeComputable_const [acc q₀])
+    ht).congr fun x => ?_
+  by_cases hx : x = []
+  · subst hx; simp
+  · simp only [hx, decide_false, Bool.false_eq_true, ↓reduceIte, getLastD_transduce_accept]
+
+/-- **The syntax pass is polynomial-time**: `x ↦ [CNF.syntaxOK x]` (`Bridge.syntaxOK_eq`,
+`polyTimeComputable_foldl_accept`). -/
+theorem polyTimeComputable_syntaxOK : PolyTimeComputable fun x => [CNF.syntaxOK x] :=
+  (polyTimeComputable_foldl_accept CNF.syntaxStep 0 fun q => decide (q = 4)).congr fun x => by
+    rw [Bridge.syntaxOK_eq]

+/-! ### Clause records and clause boundaries -/
+
+/-- **The record of a clause** in a serialized formula: `clauseRec C = 1 :: serializeClause C`. -/
+def clauseRec (C : CNF.Clause ℕ) : List Bool := true :: CNF.serializeClause C
+
+/-- The defining equation of `clauseRec`. -/
+theorem clauseRec_eq (C : CNF.Clause ℕ) : clauseRec C = true :: CNF.serializeClause C := rfl
+
+/-- Glue: a serialized formula is its clause records followed by `0`. -/
+theorem serialize_eq_recs (φ : CNF ℕ) : CNF.serialize φ = φ.flatMap clauseRec ++ [false] := by
+  rw [Bridge.serialize_eq]; rfl
+
+/-- Glue: a well-formed string is the serialization of its decoding. -/
+theorem eq_serialize_decode {s : List Bool} (h : CNF.syntaxOK s = true) :
+    s = CNF.serialize (CNF.decode s) := by
+  rw [CNF.syntaxOK_spec] at h
+  obtain ⟨φ, hφ⟩ := Option.isSome_iff_exists.1 h
+  rw [Bridge.decode_eq, hφ, Option.getD_some]
+  exact CNF.parse_repr hφ
+
+/-- Glue: serializations are well-formed. -/
+theorem syntaxOK_serialize (φ : CNF ℕ) : CNF.syntaxOK (CNF.serialize φ) = true := by
+  rw [CNF.syntaxOK_spec, CNF.parse_serialize]; rfl
+
+/-- Glue: the serialization is injective (`CNF.parse_serialize`). -/
+theorem serialize_injective {φ ψ : CNF ℕ} (h : CNF.serialize φ = CNF.serialize ψ) : φ = ψ := by
+  have := CNF.parse_serialize φ
+  rw [h, CNF.parse_serialize] at this
+  exact (Option.some.inj this).symm
+
+/-- Glue: the automaton accepts exactly when its run ends in the exact-end state `4`. -/
+private theorem foldl_eq_four {x : List Bool} :
+    x.foldl CNF.syntaxStep 0 = 4 ↔ CNF.syntaxOK x = true := by
+  rw [Bridge.syntaxOK_eq, decide_eq_true_iff]
+
+/-- Glue: reading `0` leads to the exact-end state only from the formula state. -/
+private theorem syntaxStep_false_eq_four {q : Fin 6} (h : CNF.syntaxStep q false = 4) : q = 0 := by
+  rw [Bridge.syntaxStep_eq] at h
+  fin_cases q <;> simp_all
+
+/-- **A clause record returns the automaton to the formula state**:
+`(clauseRec C).foldl syntaxStep 0 = 0`. Glue. -/
+theorem foldl_clauseRec (C : CNF.Clause ℕ) : (clauseRec C).foldl CNF.syntaxStep 0 = 0 := by
+  have h := foldl_eq_four.2 (syntaxOK_serialize [C])
+  rw [serialize_eq_recs, List.flatMap_singleton, List.foldl_append, List.foldl_cons,
+    List.foldl_nil] at h
+  exact syntaxStep_false_eq_four h
+
+/-- Glue: a list of clause records returns the automaton to the formula state. -/
+theorem foldl_recs (χ : CNF ℕ) : (χ.flatMap clauseRec).foldl CNF.syntaxStep 0 = 0 := by
+  induction χ with
+  | nil => rfl
+  | cons C χ ih => rw [List.flatMap_cons, List.foldl_append, foldl_clauseRec, ih]
+
+/-- **Prefixes ending in the formula state are lists of clause records**: if
+`t.foldl syntaxStep 0 = 0`, then `t = χ.flatMap clauseRec` for some clause list `χ`. Glue.
+
+**Proof sketch.** `t ++ [0]` is accepted (state `0` reads `0` into `4`), so it is the
+serialization of some `χ` (`eq_serialize_decode`), i.e. `χ.flatMap clauseRec ++ [0]`. -/
+theorem exists_recs_of_foldl {t : List Bool} (h : t.foldl CNF.syntaxStep 0 = 0) :
+    ∃ χ : CNF ℕ, t = χ.flatMap clauseRec := by
+  have hok : CNF.syntaxOK (t ++ [false]) = true := by
+    rw [← foldl_eq_four, List.foldl_append, h, List.foldl_cons, List.foldl_nil,
+      Bridge.syntaxStep_eq]
+    rfl
+  refine ⟨CNF.decode (t ++ [false]), ?_⟩
+  have := eq_serialize_decode hok
+  rw [serialize_eq_recs] at this
+  exact List.append_cancel_right this
+
+/-- **The suffix after a clause boundary**: if `s = serialize φ` and `s.take p` ends in the
+formula state, then `φ = χ ++ ψ` with `s.drop p = serialize ψ`. Glue. -/
+theorem exists_split_of_boundary {φ : CNF ℕ} {p : ℕ}
+    (h : ((CNF.serialize φ).take p).foldl CNF.syntaxStep 0 = 0) :
+    ∃ χ ψ : CNF ℕ, φ = χ ++ ψ ∧ (CNF.serialize φ).drop p = CNF.serialize ψ := by
+  obtain ⟨χ, hχ⟩ := exists_recs_of_foldl h
+  have hok : CNF.syntaxOK ((CNF.serialize φ).drop p) = true := by
+    have h4 := foldl_eq_four.2 (syntaxOK_serialize φ)
+    rw [← List.take_append_drop p (CNF.serialize φ), List.foldl_append, h] at h4
+    exact foldl_eq_four.1 h4
+  obtain ⟨ψ, hψ⟩ : ∃ ψ, (CNF.serialize φ).drop p = CNF.serialize ψ :=
+    ⟨_, eq_serialize_decode hok⟩
+  refine ⟨χ, ψ, serialize_injective ?_, hψ⟩
+  calc CNF.serialize φ = (CNF.serialize φ).take p ++ (CNF.serialize φ).drop p :=
+        (List.take_append_drop p _).symm
+    _ = χ.flatMap clauseRec ++ CNF.serialize ψ := by rw [hχ, hψ]
+    _ = CNF.serialize (χ ++ ψ) := by
+        rw [serialize_eq_recs, serialize_eq_recs, List.flatMap_append, List.append_assoc]
+
+/-! ### Prefix-freeness of the serialization -/
+
+/-- Glue: literal serializations are prefix-free. -/
+private theorem serializeLit_append_inj {a b : Literal ℕ} {x y : List Bool}
+    (h : CNF.serializeLit a ++ x = CNF.serializeLit b ++ y) : a = b ∧ x = y := by
+  obtain ⟨v, β⟩ := a
+  obtain ⟨w, γ⟩ := b
+  rw [Bridge.serializeLit_eq, Bridge.serializeLit_eq] at h
+  simp only [List.append_assoc, List.cons_append, List.nil_append] at h
+  induction v generalizing w with
+  | zero =>
+    cases w with
+    | zero => simp_all
+    | succ w => simp [List.replicate_succ] at h
+  | succ v ih =>
+    cases w with
+    | zero => simp [List.replicate_succ] at h
+    | succ w =>
+      obtain ⟨hab, hxy⟩ := ih w (by
+        simpa only [List.replicate_succ, List.cons_append, List.cons.injEq, true_and] using h)
+      simp only [Prod.mk.injEq] at hab ⊢
+      exact ⟨⟨by omega, hab.2⟩, hxy⟩
+
+/-- **Clause serializations are prefix-free**: `serializeClause C ++ x = serializeClause D ++ y`
+implies `C = D` and `x = y`. Glue.
+
+**Proof sketch.** Induction on `C`: a literal record starts with `1` and the clause terminator
+is `0`, so both sides end or continue together; two literal records followed by anything agree
+in their unary variable run, polarity bit and remainder (`serializeLit_eq`). -/
+theorem serializeClause_append_inj {C D : CNF.Clause ℕ} {x y : List Bool}
+    (h : CNF.serializeClause C ++ x = CNF.serializeClause D ++ y) : C = D ∧ x = y := by
+  rw [Bridge.serializeClause_eq, Bridge.serializeClause_eq] at h
+  simp only [List.append_assoc, List.singleton_append] at h
+  induction C generalizing D with
+  | nil =>
+    cases D with
+    | nil => simpa using h
+    | cons b D =>
+      rw [List.flatMap_cons, Bridge.serializeLit_eq] at h
+      simp [List.replicate_succ] at h
+  | cons a C ih =>
+    cases D with
+    | nil =>
+      rw [List.flatMap_cons, Bridge.serializeLit_eq] at h
+      simp [List.replicate_succ] at h
+    | cons b D =>
+      rw [List.flatMap_cons, List.flatMap_cons, List.append_assoc, List.append_assoc] at h
+      obtain ⟨rfl, h'⟩ := serializeLit_append_inj h
+      obtain ⟨rfl, rfl⟩ := ih h'
+      exact ⟨rfl, rfl⟩

+/-! ### The clauses of a clause code -/
+
+/-- **The literal of a literal code**: `litOfCode ℓ = (ℓ / 2, decide (ℓ % 2 = 0))`. -/
+def litOfCode (ℓ : ℕ) : Literal ℕ := (ℓ / 2, decide (ℓ % 2 = 0))
+
+/-- The defining equation of `litOfCode`. -/
+theorem litOfCode_eq (ℓ : ℕ) : litOfCode ℓ = (ℓ / 2, decide (ℓ % 2 = 0)) := rfl
+
+/-- `litCode (litOfCode ℓ) = ℓ`. Glue. -/
+theorem litCode_litOfCode (ℓ : ℕ) : litCode (litOfCode ℓ) = ℓ := by
+  rw [litCode_eq, litOfCode_eq]
+  by_cases h : ℓ % 2 = 0
+  · simp only [h, decide_true, ↓reduceIte, add_zero]; omega
+  · simp only [h, decide_false, Bool.false_eq_true, ↓reduceIte]; omega
+
+/-- `litOfCode (litCode a) = a`. Glue. -/
+theorem litOfCode_litCode (a : Literal ℕ) : litOfCode (litCode a) = a := by
+  obtain ⟨v, β⟩ := a
+  rw [litCode_eq, litOfCode_eq]
+  cases β
+  · simp only [Bool.false_eq_true, ↓reduceIte, Prod.mk.injEq, decide_eq_false_iff_not]; omega
+  · simp only [↓reduceIte, add_zero, Prod.mk.injEq, decide_eq_true_eq]; omega
+
+/-- **The candidate clauses of a clause code** `c` for `K` variables: with the literals
+`Lᵢ = litOfCode (clauseLit K c i)`, the clause `[L₀]` (`j = 1`), `[L₀, L₁]` (`j = 2`) and
+`[L₀, L₁, L₂]` (otherwise). -/
+def candClause (K c : ℕ) : ℕ → CNF.Clause ℕ
+  | 1 => [litOfCode (clauseLit K c 0)]
+  | 2 => [litOfCode (clauseLit K c 0), litOfCode (clauseLit K c 1)]
+  | _ => [litOfCode (clauseLit K c 0), litOfCode (clauseLit K c 1), litOfCode (clauseLit K c 2)]
+
+/-- **The validity of a candidate**: `[L₀]` needs equal digits `d₀ = d₁ = d₂`, `[L₀, L₁]` needs
+`d₁ = d₂` (the widening repeats the last literal), `[L₀, L₁, L₂]` is always valid
+(`dᵢ = clauseLit K c i`). -/
+def candValid (K c : ℕ) : ℕ → Bool
+  | 1 => decide (clauseLit K c 0 = clauseLit K c 1) && decide (clauseLit K c 1 = clauseLit K c 2)
+  | 2 => decide (clauseLit K c 1 = clauseLit K c 2)
+  | _ => true
+
+/-- The case readers of `candClause`: `candClause K c 1 = [L₀]`. -/
+theorem candClause_one (K c : ℕ) : candClause K c 1 = [litOfCode (clauseLit K c 0)] := rfl
+
+/-- See `candClause_one`. -/
+theorem candClause_two (K c : ℕ) :
+    candClause K c 2 = [litOfCode (clauseLit K c 0), litOfCode (clauseLit K c 1)] := rfl
+
+/-- See `candClause_one`. -/
+theorem candClause_three (K c : ℕ) : candClause K c 3 =
+    [litOfCode (clauseLit K c 0), litOfCode (clauseLit K c 1), litOfCode (clauseLit K c 2)] :=
+  rfl
+
+/-- The case readers of `candValid`: `candValid K c 1 = (d₀ = d₁ ∧ d₁ = d₂)`. -/
+theorem candValid_one (K c : ℕ) : candValid K c 1 =
+    (decide (clauseLit K c 0 = clauseLit K c 1) && decide (clauseLit K c 1 = clauseLit K c 2)) :=
+  rfl
+
+/-- See `candValid_one`: `candValid K c 2 = (d₁ = d₂)`. -/
+theorem candValid_two (K c : ℕ) :
+    candValid K c 2 = decide (clauseLit K c 1 = clauseLit K c 2) := rfl
+
+/-- See `candValid_one`: `candValid K c 3 = true`. -/
+theorem candValid_three (K c : ℕ) : candValid K c 3 = true := rfl
+
+/-- **The candidates are the clauses with clause code `c`**: if every variable of `φ` is `< K`,
+then `(∃ C ∈ φ, 1 ≤ |C| ≤ 3 ∧ clauseCode K C = c)` iff some valid candidate `candClause K c j`
+(`j = 1, 2, 3`) belongs to `φ`. Glue.
+
+**Proof sketch.** A valid candidate has one to three literals and, its widening repeating the
+last literal, digits `clauseLit K c i`, so its code is `c` (`clauseLit_decomp`). Conversely the
+digits of `clauseCode K C` are the codes of the widened literals (`clauseLit_clauseCode`), so
+the candidate of length `|C|` is valid and equals `C` (`litOfCode_litCode`). -/
+theorem exists_mem_clauseCode_iff {K c : ℕ} {φ : CNF ℕ} (hK : ∀ C ∈ φ, ∀ ℓ ∈ C, ℓ.1 < K) :
+    (∃ C ∈ φ, 1 ≤ C.length ∧ C.length ≤ 3 ∧ clauseCode K C = c) ↔
+      ∃ j ∈ [1, 2, 3], candValid K c j = true ∧ candClause K c j ∈ φ := by
+  have hdec := clauseLit_decomp K c
+  constructor
+  · rintro ⟨C, hC, h1, h3, rfl⟩
+    have hd := clauseLit_clauseCode (hK C hC) h1
+    match C, h1, h3 with
+    | [a], _, _ =>
+      have e : ∀ i : Fin 3, clauseLit K (clauseCode K [a]) i = litCode a := fun i => by
+        rw [hd, wideLit_eq]; simp
+      refine ⟨1, by simp, ?_, ?_⟩
+      · simp [candValid, e]
+      · rw [candClause_one, e, litOfCode_litCode]; simpa using hC
+    | [a, b], _, _ =>
+      have e0 : clauseLit K (clauseCode K [a, b]) 0 = litCode a := by rw [hd, wideLit_eq]; simp
+      have e1 : clauseLit K (clauseCode K [a, b]) 1 = litCode b := by rw [hd, wideLit_eq]; simp
+      have e2 : clauseLit K (clauseCode K [a, b]) 2 = litCode b := by rw [hd, wideLit_eq]; simp
+      refine ⟨2, by simp, ?_, ?_⟩
+      · simp [candValid, e1, e2]
+      · rw [candClause_two, e0, e1, litOfCode_litCode, litOfCode_litCode]; exact hC
+    | [a, b, d], _, _ =>
+      have e0 : clauseLit K (clauseCode K [a, b, d]) 0 = litCode a := by
+        rw [hd, wideLit_eq]; simp
+      have e1 : clauseLit K (clauseCode K [a, b, d]) 1 = litCode b := by
+        rw [hd, wideLit_eq]; simp
+      have e2 : clauseLit K (clauseCode K [a, b, d]) 2 = litCode d := by
+        rw [hd, wideLit_eq]; simp
+      refine ⟨3, by simp, rfl, ?_⟩
+      rw [candClause_three, e0, e1, e2, litOfCode_litCode, litOfCode_litCode, litOfCode_litCode]
+      exact hC
+  · rintro ⟨j, hj, hv, hmem⟩
+    refine ⟨_, hmem, ?_⟩
+    simp only [List.mem_cons, List.not_mem_nil, or_false] at hj
+    rcases hj with rfl | rfl | rfl
+    · simp only [candValid, Bool.and_eq_true, decide_eq_true_eq] at hv
+      rw [← hv.2, ← hv.1] at hdec
+      refine ⟨by simp [candClause_one], by simp [candClause_one], ?_⟩
+      simp [clauseCode_eq, candClause_one, wideLit_eq, litCode_litOfCode]
+      linarith
+    · simp only [candValid, decide_eq_true_eq] at hv
+      rw [← hv] at hdec
+      refine ⟨by simp [candClause_two], by simp [candClause_two], ?_⟩
+      simp [clauseCode_eq, candClause_two, wideLit_eq, litCode_litOfCode]
+      linarith
+    · refine ⟨by simp [candClause_three], by simp [candClause_three], ?_⟩
+      simp [clauseCode_eq, candClause_three, wideLit_eq, litCode_litOfCode]
+      linarith

+/-! ### The string form of `memS` -/
+
+/-- **A clause record at a clause boundary**: `hasRec s r` holds when, at some position
+`p ≤ |s|` where the grammar automaton is in its formula state after `s.take p`, the string `s`
+continues with `r`. -/
+def hasRec (s r : List Bool) : Bool :=
+  decide (∃ p < s.length + 1, (decide ((s.take p).foldl CNF.syntaxStep 0 = 0) &&
+    decide ((s.drop p).take r.length = r)) = true)
+
+/-- The defining equation of `hasRec`. -/
+theorem hasRec_eq (s r : List Bool) :
+    hasRec s r = decide (∃ p < s.length + 1, (decide ((s.take p).foldl CNF.syntaxStep 0 = 0) &&
+      decide ((s.drop p).take r.length = r)) = true) := rfl
+
+/-- **Clause records at clause boundaries are the clauses**: for a serialized formula,
+`hasRec (serialize φ) (clauseRec C) = true ↔ C ∈ φ`. Glue.
+
+**Proof sketch.** At a clause boundary `p` the formula splits as `χ ++ ψ` with the suffix
+`serialize ψ` (`exists_split_of_boundary`); a record prefix of it is the record of the head
+clause of `ψ` by prefix-freeness (`serializeClause_append_inj`). Conversely, for
+`φ = χ ++ C :: ψ` the position `|χ.flatMap clauseRec|` is a boundary (`foldl_recs`) followed by
+the record of `C`. -/
+theorem hasRec_serialize_iff (φ : CNF ℕ) (C : CNF.Clause ℕ) :
+    hasRec (CNF.serialize φ) (clauseRec C) = true ↔ C ∈ φ := by
+  rw [hasRec_eq, decide_eq_true_iff]
+  constructor
+  · rintro ⟨p, -, hp⟩
+    simp only [Bool.and_eq_true, decide_eq_true_eq] at hp
+    obtain ⟨χ, ψ, rfl, hψ⟩ := exists_split_of_boundary hp.1
+    have hpre : clauseRec C <+: CNF.serialize ψ := by
+      rw [List.prefix_iff_eq_take, ← hψ]; exact hp.2.symm
+    obtain ⟨t, ht⟩ := hpre
+    cases ψ with
+    | nil =>
+      rw [serialize_eq_recs, clauseRec_eq] at ht
+      simp at ht
+    | cons D ψ =>
+      rw [serialize_eq_recs, List.flatMap_cons, clauseRec_eq, clauseRec_eq] at ht
+      simp only [List.cons_append, List.cons.injEq, true_and, List.append_assoc] at ht
+      obtain ⟨rfl, -⟩ := serializeClause_append_inj ht
+      exact List.mem_append_right _ List.mem_cons_self
+  · intro hC
+    obtain ⟨χ, ψ, rfl⟩ := List.append_of_mem hC
+    refine ⟨(χ.flatMap clauseRec).length, ?_, ?_⟩
+    · rw [serialize_eq_recs, List.flatMap_append, List.length_append, List.length_append]
+      omega
+    · have hs : CNF.serialize (χ ++ C :: ψ) =
+          χ.flatMap clauseRec ++ (clauseRec C ++ (ψ.flatMap clauseRec ++ [false])) := by
+        rw [serialize_eq_recs, List.flatMap_append, List.flatMap_cons, List.append_assoc,
+          List.append_assoc]
+      rw [hs, List.take_left' rfl, List.drop_left' rfl, List.take_left' rfl, foldl_recs]
+      simp
+
+/-- **The string test of `memS`**: `s` is well-formed and the record of a valid candidate clause
+of `c` (`candClause`, `candValid`) sits at a clause boundary of `s`. -/
+def memAlg (s : List Bool) (c : ℕ) : Bool :=
+  CNF.syntaxOK s &&
+    (candValid s.length c 1 && hasRec s (clauseRec (candClause s.length c 1)) ||
+     candValid s.length c 2 && hasRec s (clauseRec (candClause s.length c 2)) ||
+     candValid s.length c 3 && hasRec s (clauseRec (candClause s.length c 3)))
+
+/-- The defining equation of `memAlg`. -/
+theorem memAlg_eq (s : List Bool) (c : ℕ) :
+    memAlg s c = (CNF.syntaxOK s &&
+      (candValid s.length c 1 && hasRec s (clauseRec (candClause s.length c 1)) ||
+       candValid s.length c 2 && hasRec s (clauseRec (candClause s.length c 2)) ||
+       candValid s.length c 3 && hasRec s (clauseRec (candClause s.length c 3)))) := rfl
+
+/-- Glue: an ill-formed string decodes to the empty formula. -/
+theorem decode_eq_nil_of_syntaxOK {s : List Bool} (h : CNF.syntaxOK s = false) :
+    CNF.decode s = [] := by
+  rw [CNF.syntaxOK_spec, Option.isSome_eq_false_iff, Option.isNone_iff_eq_none] at h
+  rw [Bridge.decode_eq, h, Option.getD_none, Bridge.fallback_eq]
+
+/-- **`memS` is its string test**: `memS s c = memAlg s c`.
+
+**Proof sketch.** Ill-formed strings decode to `[]` (`decode_eq_nil_of_syntaxOK`), where both
+sides fail. A well-formed `s` is `serialize (decode s)` (`eq_serialize_decode`); `memS_iff`,
+the variable bound `var_lt_length_of_mem_decode`, `exists_mem_clauseCode_iff` and
+`hasRec_serialize_iff` identify the two sides. -/
+theorem memS_eq (s : List Bool) (c : ℕ) : memS s c = memAlg s c := by
+  rw [memAlg_eq]
+  cases hok : CNF.syntaxOK s with
+  | false =>
+    simp only [Bool.false_and]
+    rw [Bool.eq_false_iff, ne_eq, memS_iff, decode_eq_nil_of_syntaxOK hok]
+    simp
+  | true =>
+    simp only [Bool.true_and]
+    apply Bool.eq_iff_iff.mpr
+    rw [memS_iff, exists_mem_clauseCode_iff
+      (fun C hC ℓ hℓ => var_lt_length_of_mem_decode hC hℓ)]
+    have hs := eq_serialize_decode hok
+    have hR : ∀ j, hasRec s (clauseRec (candClause s.length c j)) = true ↔
+        candClause s.length c j ∈ CNF.decode s := fun j => by
+      have h := hasRec_serialize_iff (CNF.decode s) (candClause s.length c j)
+      rwa [← hs] at h
+    simp only [List.mem_cons, List.not_mem_nil, or_false, exists_eq_or_imp, exists_eq_left,
+      Bool.or_eq_true, Bool.and_eq_true, hR, or_assoc]
+
+/-! ### `memS` in polynomial time -/
+
+/-- **The boundary-record test is polynomial-time**: for polynomial-time `S`, `R`, the map
+`w ↦ [hasRec (S w) (R w)]` is polynomial-time.
+
+**Proof sketch.** A bounded search (`EncodingMachines.polyTimeComputable_firstIdx`, bound
+`|S w| + 1`) for a position `p` where the automaton verdict on `(S w).take p`
+(`polyTimeComputable_foldl_accept` after `polyTimeComputable_takePrefixByLength`) is the formula
+state and `((S w).drop p).take |R w| = R w` (`polyTimeComputable_pairDropPoly`,
+`BinArith.polyTimeComputable_eqW`); success is `firstIdx < |S w| + 1`
+(`EncodingMachines.firstIdx_lt_iff`). -/
+theorem polyTimeComputable_hasRec {S R : List Bool → List Bool} (hS : PolyTimeComputable S)
+    (hR : PolyTimeComputable R) : PolyTimeComputable fun w => [hasRec (S w) (R w)] := by
+  open EncodingMachines in
+  have hSv : PolyTimeComputable fun v => S (pairSndD v) := hS.comp' polyTimeComputable_pairSndD
+  have hRv : PolyTimeComputable fun v => R (pairSndD v) := hR.comp' polyTimeComputable_pairSndD
+  have hpair := polyTimeComputable_pairFstD.pairEncode hSv
+  have htake : PolyTimeComputable fun v => (S (pairSndD v)).take (pairFstD v).length :=
+    (polyTimeComputable_takePrefixByLength.comp' hpair).congr fun v => by
+      rw [PrefixByLength.take_def, pairFstD_pairEncode, pairSndD_pairEncode]
+  have hdrop : PolyTimeComputable fun v => (S (pairSndD v)).drop (pairFstD v).length :=
+    ((polyTimeComputable_pairDropPoly .var).comp' hpair).congr fun v => by
+      simp only [pairFstD_pairEncode, pairSndD_pairEncode, PolyExpr.eval_var]
+  have hbd : PolyTimeComputable fun v =>
+      [decide (((S (pairSndD v)).take (pairFstD v).length).foldl CNF.syntaxStep 0 = 0)] :=
+    (polyTimeComputable_foldl_accept CNF.syntaxStep 0 fun q => decide (q = 0)).comp' htake
+  have htake2 : PolyTimeComputable fun v =>
+      ((S (pairSndD v)).drop (pairFstD v).length).take (R (pairSndD v)).length :=
+    (polyTimeComputable_takePrefixByLength.comp' (hRv.pairEncode hdrop)).congr fun v => by
+      rw [PrefixByLength.take_def, pairFstD_pairEncode, pairSndD_pairEncode]
+  have heq : PolyTimeComputable fun v =>
+      [decide (((S (pairSndD v)).drop (pairFstD v).length).take (R (pairSndD v)).length =
+        R (pairSndD v))] :=
+    (BinArith.polyTimeComputable_eqW.comp' (htake2.pairEncode hRv)).congr fun v => by
+      rw [BinArith.eqW_pairEncode]
+  have hlen : PolyTimeComputable fun w => List.replicate ((S w).length + 1) true :=
+    unaryAdd_poly (unaryLength_poly hS) (unaryConst_poly 1)
+  have hf := polyTimeComputable_firstIdx
+    (p := fun w p => decide (((S w).take p).foldl CNF.syntaxStep 0 = 0) &&
+      decide (((S w).drop p).take (R w).length = R w)) (polyTimeComputable_and hbd heq) hlen
+  refine (ltTest_poly hf hlen).congr fun w => ?_
+  rw [hasRec_eq]
+  simp only [firstIdx_lt_iff]
+
+/-- Glue: the serialization of the literal of a unary code is polynomial-time,
+`x ↦ serializeLit (litOfCode (ℓ x)) = 1^(ℓ x / 2 + 1) ++ [0, ℓ x even]`
+(`Bridge.serializeLit_eq`). -/
+private theorem polyTimeComputable_serializeLit_litOfCode {ℓ : List Bool → ℕ}
+    (hℓ : PolyTimeComputable fun x => List.replicate (ℓ x) true) :
+    PolyTimeComputable fun x => CNF.serializeLit (litOfCode (ℓ x)) := by
+  open EncodingMachines in
+  refine ((unaryAdd_poly (unaryDivConst_poly hℓ 2) (unaryConst_poly 1)).append
+    ((polyTimeComputable_const [false]).append
+      (eqTest_poly (unaryModConst_poly hℓ 2) (unaryConst_poly 0)))).congr fun x => ?_
+  rw [Bridge.serializeLit_eq, litOfCode_eq]
+  rfl
+
+/-- **`memS` is polynomial-time**: `⟨1^c, s⟩ ↦ [memS s c]`.
+
+**Proof sketch.** `memS_eq`: the syntax pass (`polyTimeComputable_syntaxOK`), the validity
+tests (`EncodingMachines.eqTest_poly` on `clauseLit_poly`) and the record tests
+(`polyTimeComputable_hasRec` on records assembled by `PolyTimeComputable.append` from the
+literal serializations of `clauseLit_poly` codes), combined by `polyTimeComputable_and/or`. -/
+theorem polyTimeComputable_memS :
+    PolyTimeComputable fun w => [memS (pairSndD w) (pairFstD w).length] := by
+  open EncodingMachines in
+  have hS := polyTimeComputable_pairSndD
+  have hK : PolyTimeComputable fun w => List.replicate (pairSndD w).length true :=
+    unaryLength_poly hS
+  have hc : PolyTimeComputable fun w => List.replicate (pairFstD w).length true :=
+    unaryLength_poly polyTimeComputable_pairFstD
+  have hd := fun i => clauseLit_poly hK hc i
+  have hl := fun i => polyTimeComputable_serializeLit_litOfCode (hd i)
+  have hT := polyTimeComputable_const [true]
+  have hF := polyTimeComputable_const [false]
+  have hr1 : PolyTimeComputable fun w =>
+      clauseRec (candClause (pairSndD w).length (pairFstD w).length 1) :=
+    (hT.append ((hl 0).append hF)).congr fun w => by
+      rw [candClause_one, clauseRec_eq, Bridge.serializeClause_eq]; simp
+  have hr2 : PolyTimeComputable fun w =>
+      clauseRec (candClause (pairSndD w).length (pairFstD w).length 2) :=
+    (hT.append ((hl 0).append ((hl 1).append hF))).congr fun w => by
+      rw [candClause_two, clauseRec_eq, Bridge.serializeClause_eq]; simp
+  have hr3 : PolyTimeComputable fun w =>
+      clauseRec (candClause (pairSndD w).length (pairFstD w).length 3) :=
+    (hT.append ((hl 0).append ((hl 1).append ((hl 2).append hF)))).congr fun w => by
+      rw [candClause_three, clauseRec_eq, Bridge.serializeClause_eq]; simp
+  have hv1 : PolyTimeComputable fun w => [candValid (pairSndD w).length (pairFstD w).length 1] :=
+    polyTimeComputable_and (eqTest_poly (hd 0) (hd 1)) (eqTest_poly (hd 1) (hd 2))
+  have hv2 : PolyTimeComputable fun w => [candValid (pairSndD w).length (pairFstD w).length 2] :=
+    eqTest_poly (hd 1) (hd 2)
+  have hv3 : PolyTimeComputable fun w => [candValid (pairSndD w).length (pairFstD w).length 3] :=
+    polyTimeComputable_const [true]
+  refine (polyTimeComputable_and (polyTimeComputable_syntaxOK.comp' hS)
+    (polyTimeComputable_or (polyTimeComputable_or
+      (polyTimeComputable_and hv1 (polyTimeComputable_hasRec hS hr1))
+      (polyTimeComputable_and hv2 (polyTimeComputable_hasRec hS hr2)))
+      (polyTimeComputable_and hv3 (polyTimeComputable_hasRec hS hr3)))).congr fun w => ?_
+  rw [memS_eq, memAlg_eq]
+
+/-! ### The width automaton and `goodS` -/
+
+/-- **The width automaton**: on the state `(q, n, bad)` it runs the grammar automaton `q`,
+counts the literals `n` of the current clause up to `4` (reset at formula level, incremented
+when a literal starts at clause level), and raises `bad` when a clause ends with `0` or `4`
+counted literals. -/
+def widthStep (st : Fin 6 × Fin 5 × Bool) (b : Bool) : Fin 6 × Fin 5 × Bool :=
+  (CNF.syntaxStep st.1 b,
+    if st.1 = 0 then 0
+    else if st.1 = 1 ∧ b = true then ⟨min (st.2.1.val + 1) 4, by omega⟩ else st.2.1,
+    st.2.2 || (decide (st.1 = 1) && !b && decide (st.2.1.val = 0 ∨ st.2.1.val = 4)))
+
+/-- The defining equation of `widthStep`. -/
+theorem widthStep_eq (st : Fin 6 × Fin 5 × Bool) (b : Bool) :
+    widthStep st b = (CNF.syntaxStep st.1 b,
+      if st.1 = 0 then 0
+      else if st.1 = 1 ∧ b = true then ⟨min (st.2.1.val + 1) 4, by omega⟩ else st.2.1,
+      st.2.2 || (decide (st.1 = 1) && !b && decide (st.2.1.val = 0 ∨ st.2.1.val = 4))) := rfl
+
+/-- Glue: the unary run of a literal stays in the index state. -/
+private theorem foldl_widthStep_replicate (n : Fin 5) (β : Bool) (v : ℕ) :
+    (List.replicate v true).foldl widthStep (2, n, β) = (2, n, β) := by
+  induction v with
+  | zero => rfl
+  | succ v ih =>
+    rw [List.replicate_succ, List.foldl_cons, widthStep_eq, Bridge.syntaxStep_eq]
+    simpa using ih
+
+/-- Glue: a literal increments the (capped) literal count at clause level. -/
+private theorem foldl_widthStep_lit (n : Fin 5) (β : Bool) (ℓ : Literal ℕ) :
+    (CNF.serializeLit ℓ).foldl widthStep (1, n, β) = (1, ⟨min (n.val + 1) 4, by omega⟩, β) := by
+  rw [Bridge.serializeLit_eq, List.replicate_succ, List.cons_append, List.foldl_cons,
+    List.foldl_append]
+  have h1 : widthStep (1, n, β) true = (2, ⟨min (n.val + 1) 4, by omega⟩, β) := by
+    rw [widthStep_eq, Bridge.syntaxStep_eq]; simp
+  rw [h1, foldl_widthStep_replicate]
+  simp [widthStep_eq, Bridge.syntaxStep_eq]
+
+/-- Glue: the literals of a clause add their number to the capped count. -/
+private theorem foldl_widthStep_lits (C : CNF.Clause ℕ) (n : Fin 5) (β : Bool) :
+    (C.flatMap CNF.serializeLit).foldl widthStep (1, n, β) =
+      (1, ⟨min (n.val + C.length) 4, by omega⟩, β) := by
+  induction C generalizing n with
+  | nil =>
+    simp only [List.flatMap_nil, List.foldl_nil, List.length_nil, add_zero, Prod.mk.injEq,
+      true_and, and_true]
+    rw [Fin.ext_iff]
+    show (n : ℕ) = min (n : ℕ) 4
+    omega
+  | cons ℓ C ih =>
+    rw [List.flatMap_cons, List.foldl_append, foldl_widthStep_lit, ih]
+    simp only [List.length_cons, Prod.mk.injEq, true_and, and_true]
+    ext; simp only; omega
+
+/-- Glue: the capped count `min k 4` flags `k = 0` or `k ≥ 4`. -/
+private theorem decide_min_bad (k : ℕ) :
+    decide (min k 4 = 0 ∨ min k 4 = 4) = !(decide (1 ≤ k) && decide (k ≤ 3)) := by
+  rcases Nat.lt_or_ge k 4 with h | h
+  · interval_cases k <;> rfl
+  · rw [min_eq_right h]
+    have : ¬ k ≤ 3 := by omega
+    simp [this]
+
+/-- Glue: a clause record returns to formula level and flags a clause with `0` or `≥ 4`
+literals. -/
+private theorem foldl_widthStep_rec (C : CNF.Clause ℕ) (n : Fin 5) (β : Bool) :
+    ∃ m : Fin 5, (clauseRec C).foldl widthStep (0, n, β) =
+      (0, m, β || !(decide (1 ≤ C.length) && decide (C.length ≤ 3))) := by
+  rw [clauseRec_eq, Bridge.serializeClause_eq, List.foldl_cons, List.foldl_append]
+  have h1 : widthStep (0, n, β) true = (1, 0, β) := by
+    rw [widthStep_eq, Bridge.syntaxStep_eq]; simp
+  rw [h1, foldl_widthStep_lits, List.foldl_cons, List.foldl_nil]
+  refine ⟨⟨min (0 + C.length) 4, by omega⟩, ?_⟩
+  rw [widthStep_eq, Bridge.syntaxStep_eq]
+  simp only [Prod.mk.injEq]
+  refine ⟨by simp, by simp, ?_⟩
+  dsimp only
+  rw [decide_min_bad (((0 : Fin 5) : ℕ) + C.length)]
+  simp
+
+/-- Glue: a list of clause records returns to formula level and flags its bad clauses. -/
+private theorem foldl_widthStep_recs (χ : CNF ℕ) (n : Fin 5) (β : Bool) :
+    ∃ m : Fin 5, (χ.flatMap clauseRec).foldl widthStep (0, n, β) =
+      (0, m, β || χ.any fun C => !(decide (1 ≤ C.length) && decide (C.length ≤ 3))) := by
+  induction χ generalizing n β with
+  | nil => exact ⟨n, by simp⟩
+  | cons C χ ih =>
+    obtain ⟨m, hm⟩ := foldl_widthStep_rec C n β
+    rw [List.flatMap_cons, List.foldl_append, hm]
+    obtain ⟨m', hm'⟩ := ih m (β || !(decide (1 ≤ C.length) && decide (C.length ≤ 3)))
+    rw [hm']
+    exact ⟨m', by simp [Bool.or_assoc]⟩
+
+/-- **`goodS` is its string test**: `goodS s = (!syntaxOK s || !(s.foldl widthStep (0, 0,
+false)).2.2)`.
+
+**Proof sketch.** Ill-formed strings decode to `[]`, where `goodS` holds. A well-formed `s` is
+`serialize φ` (`eq_serialize_decode`), the records of `φ` followed by `0`, and the width
+automaton flags exactly the clauses with `0` or `≥ 4` literals (`foldl_widthStep_recs`). -/
+theorem goodS_eq (s : List Bool) :
+    goodS s = (!CNF.syntaxOK s || !(s.foldl widthStep (0, 0, false)).2.2) := by
+  cases hok : CNF.syntaxOK s with
+  | false =>
+    simp only [Bool.not_false, Bool.true_or]
+    rw [goodS_iff, decode_eq_nil_of_syntaxOK hok]
+    simp
+  | true =>
+    simp only [Bool.not_true, Bool.false_or]
+    have hs := eq_serialize_decode hok
+    obtain ⟨m, hm⟩ := foldl_widthStep_recs (CNF.decode s) 0 false
+    have hrun : (s.foldl widthStep (0, 0, false)).2.2 =
+        (CNF.decode s).any fun C => !(decide (1 ≤ C.length) && decide (C.length ≤ 3)) := by
+      conv_lhs => rw [hs, serialize_eq_recs, List.foldl_append, hm]
+      rw [List.foldl_cons, List.foldl_nil, widthStep_eq]
+      simp
+    rw [hrun]
+    apply Bool.eq_iff_iff.mpr
+    rw [goodS_iff]
+    simp [Nat.one_le_iff_ne_zero]
+
+/-- **`goodS` is polynomial-time**: `s ↦ [goodS s]` (`goodS_eq`, `polyTimeComputable_syntaxOK`,
+`polyTimeComputable_foldl_accept` on `widthStep`). -/
+theorem polyTimeComputable_goodS : PolyTimeComputable fun s => [goodS s] :=
+  (polyTimeComputable_or (polyTimeComputable_not polyTimeComputable_syntaxOK)
+    (polyTimeComputable_not (polyTimeComputable_foldl_accept widthStep (0, 0, false)
+      fun st => st.2.2))).congr fun s => by rw [goodS_eq]
+
+/-! ### The instance map -/
+
+/-- **The 3SAT → 3COL instance map is polynomial-time computable**:
+`PolyTimeComputable colorInst` (`polyTimeComputable_colorInst_of` with
+`polyTimeComputable_memS` and `polyTimeComputable_goodS`). -/
+theorem polyTimeComputable_colorInst : PolyTimeComputable colorInst :=
+  polyTimeComputable_colorInst_of polyTimeComputable_memS polyTimeComputable_goodS
+

+# The 3-colouring witness of the gadget graph
+
+A satisfying assignment of a 3-CNF gives an explicit proper 3-colouring of the gadget graph
+`gadgetGraph K mem` (`LevinHardness/Gadget.lean`), and the encoded colouring is a
+polynomial-time function of the pair `⟨s, a⟩` of the formula word and the assignment word.
+
+**Overlap (disclosed, brief EF-H12 item (b)).** The colouring `gadgetColor` re-derives, for the
+`Fin N` layout of `Gadget.lean`, the completeness colouring of the course reduction
+(`SATtoColor.SATtoColorCompleteness`, `Complexity/NPReductions/ThreeSATToColoring.lean:193`,
+whose colourings `sat3Coloring` and `clauseGadgetColor` are private there): the palette vertex
+`p` gets colour `p`, a literal node gets `1` if the literal is true and `2` otherwise, and the six
+nodes of a clause gadget get the colours of the course table `clauseGadgetColor`
+(`gadgetNodeColor` here). The proof that it is proper is redone here by a case analysis over the
+edge families of `gadgetEdge`.
+
+This is the completeness colouring of [St73]'s reduction, which the paper leaves as "the converse
+is similar" [St73, p. 21], made explicit (disclosed re-derivation of the course's `sat3Coloring`).
+
+## Main definitions
+
+* `TCSlib.Cryptography.LevinHardness.gadgetNodeColor` — the colours of the six gadget nodes.
+* `TCSlib.Cryptography.LevinHardness.gadgetColor` — the colouring of the gadget graph.
+* `TCSlib.Cryptography.LevinHardness.colorWord` — the encoded colouring of `⟨s, a⟩`.
+
+## Main results
+
+* `TCSlib.Cryptography.LevinHardness.properColoring_gadgetColor` — the colouring is proper
+  when every clause of the formula is satisfied.
+* `TCSlib.Cryptography.LevinHardness.threeColorable_gadgetGraph` — hence the gadget graph is
+  3-colourable.

+  `⟨s, a⟩ ↦ colorWord s a` is polynomial-time.

+/-! ### The colouring -/
+
+/-- **The colours of the six nodes of a clause gadget** whose literals have the truth values
+`a, b, c`, at index `i` (re-derived from the course table `clauseGadgetColor`,
+`ThreeSATToColoring.lean`): node `5` gets `1`; if `a = b = c = false` every other node gets `0`;
+if `a = b = false` the nodes `4, 3, 2, 1, 0` get `2, 0, 2, 1, 0`; otherwise they get
+`0, 2, 1, (b ? 2 : 0), (a ∧ ¬b ? 2 : 0)`. -/
+def gadgetNodeColor (a b c : Bool) (i : ℕ) : Fin 3 :=
+  if i = 5 then 1
+  else if (!a && !b && !c) = true then 0
+  else if (!a && !b) = true then
+    (if i = 4 then 2 else if i = 3 then 0 else if i = 2 then 2 else if i = 1 then 1 else 0)
+  else
+    (if i = 4 then 0 else if i = 3 then 2 else if i = 2 then 1
+     else if i = 1 then (if (!b) = true then 0 else 2)
+     else (if (!a) = true then 0 else if (!b) = true then 2 else 0))
+
+/-- The defining equation of `gadgetNodeColor` (see its docstring for the table). -/
+theorem gadgetNodeColor_eq (a b c : Bool) (i : ℕ) :
+    gadgetNodeColor a b c i =
+      if i = 5 then 1
+      else if (!a && !b && !c) = true then 0
+      else if (!a && !b) = true then
+        (if i = 4 then 2 else if i = 3 then 0 else if i = 2 then 2 else if i = 1 then 1 else 0)
+      else
+        (if i = 4 then 0 else if i = 3 then 2 else if i = 2 then 1
+         else if i = 1 then (if (!b) = true then 0 else 2)
+         else (if (!a) = true then 0 else if (!b) = true then 2 else 0)) := rfl
+
+/-- **The 3-colouring of the gadget graph** from the assignment `val` (re-derived, for the
+`Fin N` layout, from the completeness colouring `sat3Coloring` of the course reduction): the
+palette `t < 3` gets colour `t`; the literal node `3 + ℓ` gets `1` if `litVal val ℓ` and `2`
+otherwise; gadget node `i` of clause code `c` (vertex `3 + 2K + 6c + i`) gets
+`gadgetNodeColor` of the truth values of the clause's three literals at `i`. -/
+def gadgetColor (K : ℕ) (val : ℕ → Bool) (t : ℕ) : Fin 3 :=
+  if t = 0 then 0 else if t = 1 then 1 else if t = 2 then 2
+  else if t < 3 + 2 * K then (if litVal val (t - 3) = true then 1 else 2)
+  else gadgetNodeColor (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 0))
+    (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 1))
+    (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 2)) ((t - (3 + 2 * K)) % 6)
+
+/-- The defining equation of `gadgetColor` (see its docstring for the cases). -/
+theorem gadgetColor_eq (K : ℕ) (val : ℕ → Bool) (t : ℕ) :
+    gadgetColor K val t =
+      if t = 0 then 0 else if t = 1 then 1 else if t = 2 then 2
+      else if t < 3 + 2 * K then (if litVal val (t - 3) = true then 1 else 2)
+      else gadgetNodeColor (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 0))
+        (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 1))
+        (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 2)) ((t - (3 + 2 * K)) % 6) := rfl
+
+/-- The palette vertex `t < 3` has colour `t`: `(gadgetColor K val t : ℕ) = t`. -/
+theorem gadgetColor_val_of_lt_three (K : ℕ) (val : ℕ → Bool) {t : ℕ} (ht : t < 3) :
+    (gadgetColor K val t : ℕ) = t := by
+  rw [gadgetColor_eq]
+  interval_cases t <;> rfl
+
+/-- The literal node `t` (`3 ≤ t < 3 + 2K`) has colour `1` if `litVal val (t - 3)` and `2`
+otherwise. -/
+theorem gadgetColor_of_lit (K : ℕ) (val : ℕ → Bool) {t : ℕ} (h3 : 3 ≤ t)
+    (hL : t < 3 + 2 * K) :
+    gadgetColor K val t = if litVal val (t - 3) = true then 1 else 2 := by
+  rw [gadgetColor_eq, if_neg (by omega), if_neg (by omega), if_neg (by omega), if_pos hL]
+
+/-- A gadget vertex `t ≥ 3 + 2K` has colour `gadgetNodeColor` of the truth values of the three
+literals of its clause code `(t - (3 + 2K)) / 6`, at index `(t - (3 + 2K)) % 6`. -/
+theorem gadgetColor_of_gadget (K : ℕ) (val : ℕ → Bool) {t : ℕ} (hL : 3 + 2 * K ≤ t) :
+    gadgetColor K val t =
+      gadgetNodeColor (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 0))
+        (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 1))
+        (litVal val (clauseLit K ((t - (3 + 2 * K)) / 6) 2)) ((t - (3 + 2 * K)) % 6) := by
+  rw [gadgetColor_eq, if_neg (by omega), if_neg (by omega), if_neg (by omega),
+    if_neg (by omega)]
+
+/-! ### The colouring is proper, edge family by edge family -/
+
+/-- Glue. Inside the gadget of a satisfied clause, the seven internal edges get distinct
+colours: `(i, j)` in the internal pair list and `a ∨ b ∨ c` give
+`gadgetNodeColor a b c i ≠ gadgetNodeColor a b c j`. -/
+private theorem gadgetNodeColor_ne_internal {a b c : Bool} (habc : (a || b || c) = true)
+    {i j : ℕ} (hij : (i, j) ∈ [(0, 1), (0, 2), (1, 2), (2, 3), (3, 4), (3, 5), (4, 5)]) :
+    gadgetNodeColor a b c i ≠ gadgetNodeColor a b c j := by
+  simp only [List.mem_cons, Prod.mk.injEq, List.not_mem_nil, or_false] at hij
+  rcases hij with ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩ |
+    ⟨rfl, rfl⟩ | ⟨rfl, rfl⟩ <;> revert habc <;> revert a b c <;> decide
+
+/-- Glue. The three literal–gadget edges get distinct colours, for any truth values:
+the first literal against node `0`, the second against node `1`, the third against node `4`. -/
+private theorem litColor_ne_gadgetNodeColor (a b c : Bool) :
+    ((if a = true then 1 else 2 : Fin 3) ≠ gadgetNodeColor a b c 0) ∧
+    ((if b = true then 1 else 2 : Fin 3) ≠ gadgetNodeColor a b c 1) ∧
+    ((if c = true then 1 else 2 : Fin 3) ≠ gadgetNodeColor a b c 4) := by
+  revert a b c; decide
+
+/-- Glue. One orientation of each edge of the gadget graph joins two vertices of distinct
+colours, when every clause of the formula is satisfied: `t ≠ t'` and
+`gadgetEdge K mem t t' = true` give `gadgetColor K val t ≠ gadgetColor K val t'`.
+
+**Proof sketch.** Split `gadgetEdge_eq` into its six edge families. Palette edges: the colours
+are the vertices themselves. Base–literal: `0` against `1` or `2`. Positive–negative literal:
+the two literal codes `2v, 2v + 1` have opposite `litVal`. Internal gadget edges: both ends lie
+in the gadget of the same clause code `c`, which is in the formula and hence satisfied, so
+`gadgetNodeColor_ne_internal`. Literal–gadget edges: the literal's colour is the colour of the
+matching truth value, `litColor_ne_gadgetNodeColor`. Node `5` (colour `1`) against the base
+(`0`) and the false node (`2`). -/
+private theorem gadgetColor_ne_of_edge {K : ℕ} {mem : ℕ → Bool} {val : ℕ → Bool}
+    (hsat : ∀ c, mem c = true → clauseSat K val c = true) {t t' : ℕ} (hne : t ≠ t')
+    (h : gadgetEdge K mem t t' = true) : gadgetColor K val t ≠ gadgetColor K val t' := by
+  rw [gadgetEdge_eq] at h
+  simp only [Bool.or_eq_true, Bool.and_eq_true, decide_eq_true_eq] at h
+  rcases h with (((((⟨ht, ht'⟩ | ⟨⟨rfl, h3⟩, hL⟩) | ⟨⟨⟨h3, hL⟩, hev⟩, rfl⟩) | h4) | h5) | h6)
+  · intro heq
+    have := congrArg Fin.val heq
+    rw [gadgetColor_val_of_lt_three K val ht, gadgetColor_val_of_lt_three K val ht'] at this
+    exact hne this
+  · rw [gadgetColor_of_lit K val h3 hL]
+    intro heq
+    have := congrArg Fin.val heq
+    rw [gadgetColor_val_of_lt_three K val (by omega)] at this
+    split_ifs at this <;> simp at this
+  · rw [gadgetColor_of_lit K val h3 hL, gadgetColor_of_lit K val (by omega) (by omega)]
+    have hneg : litVal val (t + 1 - 3) = !litVal val (t - 3) := by
+      rw [litVal_eq, litVal_eq, if_neg (by omega), if_pos hev,
+        show (t + 1 - 3) / 2 = (t - 3) / 2 by omega]
+    rw [hneg]
+    cases litVal val (t - 3) <;> decide
+  · obtain ⟨⟨⟨⟨⟨⟨hLt, -⟩, hLt'⟩, -⟩, hq⟩, hm⟩, hij⟩ := h4
+    rw [gadgetColor_of_gadget K val hLt, gadgetColor_of_gadget K val hLt', ← hq]
+    have hc := hsat _ hm
+    rw [clauseSat_eq] at hc
+    exact gadgetNodeColor_ne_internal hc hij
+  · obtain ⟨⟨⟨⟨h3, hL⟩, hLt'⟩, -⟩, hj⟩ := h5
+    rw [gadgetColor_of_lit K val h3 hL, gadgetColor_of_gadget K val hLt']
+    rcases hj with (⟨he, hi⟩ | ⟨he, hi⟩) | ⟨he, hi⟩ <;> rw [he, hi]
+    · exact (litColor_ne_gadgetNodeColor _ _ _).1
+    · exact (litColor_ne_gadgetNodeColor _ _ _).2.1
+    · exact (litColor_ne_gadgetNodeColor _ _ _).2.2
+  · obtain ⟨⟨⟨hLt, -⟩, h5⟩, h02⟩ := h6
+    rw [gadgetColor_of_gadget K val hLt, h5, gadgetNodeColor_eq, if_pos rfl]
+    intro heq
+    have := congrArg Fin.val heq
+    rcases h02 with rfl | rfl <;>
+      rw [gadgetColor_val_of_lt_three K val (by omega)] at this <;> simp at this
+
+/-- **The colouring `gadgetColor` is proper** (completeness colouring of the 3SAT → 3COL
+reduction, re-derived for the `Fin N` layout from `SATtoColor.SATtoColorCompleteness`,
+`ThreeSATToColoring.lean`): if every clause code in the formula (`mem c = true`) is satisfied by
+`val` (`clauseSat K val c = true`), then
+`ProperColoring (gadgetGraph K mem) (fun u => gadgetColor K val u)`. Gadgets of clause codes
+outside the formula carry no internal edges, and their edges to literals, base and false node
+are properly coloured for any truth values. -/
+theorem properColoring_gadgetColor (K : ℕ) (mem : ℕ → Bool) (val : ℕ → Bool)
+    (hsat : ∀ c, mem c = true → clauseSat K val c = true) :
+    ProperColoring (gadgetGraph K mem) (fun u => gadgetColor K val u) := by
+  rw [ProperColoring_iff]
+  intro u v huv
+  rw [gadgetGraph_apply, gadgetAdj_eq] at huv
+  simp only [Bool.and_eq_true, Bool.or_eq_true, decide_eq_true_eq] at huv
+  obtain ⟨hne, he | he⟩ := huv
+  · exact gadgetColor_ne_of_edge hsat hne he
+  · exact fun heq => gadgetColor_ne_of_edge hsat (Ne.symm hne) he heq.symm
+
+/-- **The gadget graph of a satisfied formula is 3-colourable**: if
+`∀ c, mem c = true → clauseSat K val c = true`, then `ThreeColorable (gadgetGraph K mem)`,
+witnessed by `gadgetColor K val` (`properColoring_gadgetColor`). -/
+theorem threeColorable_gadgetGraph (K : ℕ) (mem : ℕ → Bool) (val : ℕ → Bool)
+    (hsat : ∀ c, mem c = true → clauseSat K val c = true) :
+    ThreeColorable (gadgetGraph K mem) :=
+  (threeColorable_iff _).2 ⟨_, properColoring_gadgetColor K mem val hsat⟩
+
+/-! ### The encoded colouring -/
+
+/-- **The witness string of the gadget graph**: for the formula word `s` (with `K = |s|`
+variables) and the assignment word `a` (read with default `false`),
+`colorWord s a = encColoring (fun u : Fin (gadgetN |s|) => gadgetColor |s| (a.getD · false) u)`,
+the encoding of the colouring `gadgetColor` (re-derived from the course completeness
+colouring). -/
+def colorWord (s a : List Bool) : List Bool :=
+  encColoring (fun u : Fin (gadgetN s.length) =>
+    gadgetColor s.length (fun v => a.getD v false) u)
+
+/-- The defining equation of `colorWord`. -/
+theorem colorWord_eq (s a : List Bool) :
+    colorWord s a = encColoring (fun u : Fin (gadgetN s.length) =>
+      gadgetColor s.length (fun v => a.getD v false) u) := rfl
+
+/-- Glue. The two bits `[x / 2 = 1, x % 2 = 1]` of a colour in the layout of `encColoring`. -/
+private def colorBits (x : Fin 3) : List Bool := [decide (x.val / 2 = 1), decide (x.val % 2 = 1)]
+
+/-- Glue. Concatenating the two bits of the colours `g 0, …, g (n - 1)` gives
+`encColoring (fun u : Fin n => g u)`.
+
+**Proof sketch.** By induction on `n` (`List.range_succ`, `List.getD_append`,
+`List.getD_append_right`) the concatenation has length `2n` and bits `2i`, `2i + 1` equal to the
+two bits of `g i`; these are the bits of `encColoring` (`getD_encColoring_two_mul`,
+`getD_encColoring_two_mul_add_one`, `length_encColoring`), so the lists agree
+(`List.ext_getElem`). -/
+private theorem flatMap_colorBits (g : ℕ → Fin 3) (n : ℕ) :
+    (List.range n).flatMap (fun t => colorBits (g t)) = encColoring (fun u : Fin n => g u) := by
+  have hlen : ∀ n, ((List.range n).flatMap (fun t => colorBits (g t))).length = 2 * n := by
+    intro n
+    induction n with
+    | zero => simp
+    | succ n ih =>
+      rw [List.range_succ, List.flatMap_append, List.flatMap_singleton, List.length_append, ih]
+      simp [colorBits]
+      ring
+  have hget : ∀ n i, i < n →
+      ((List.range n).flatMap (fun t => colorBits (g t))).getD (2 * i) false =
+        decide ((g i).val / 2 = 1) ∧
+      ((List.range n).flatMap (fun t => colorBits (g t))).getD (2 * i + 1) false =
+        decide ((g i).val % 2 = 1) := by
+    intro n
+    induction n with
+    | zero => intro i hi; omega
+    | succ n ih =>
+      intro i hi
+      rw [List.range_succ, List.flatMap_append, List.flatMap_singleton]
+      rcases Nat.lt_or_ge i n with h | h
+      · rw [List.getD_append _ _ _ _ (by rw [hlen]; omega),
+          List.getD_append _ _ _ _ (by rw [hlen]; omega)]
+        exact ih i h
+      · obtain rfl : i = n := by omega
+        rw [List.getD_append_right _ _ _ _ (by simp [hlen]),
+          List.getD_append_right _ _ _ _ (by simp [hlen]), hlen]
+        simp [colorBits]
+  have hgetD : ∀ (l : List Bool) (m : ℕ) (h : m < l.length), l[m] = l.getD m false :=
+    fun l m h => by rw [List.getD_eq_getElem?_getD, List.getElem?_eq_getElem h, Option.getD_some]
+  refine List.ext_getElem (by rw [hlen, length_encColoring]; ring) fun m h1 h2 => ?_
+  rw [hgetD _ _ h1, hgetD _ _ h2]
+  rw [hlen] at h1
+  rcases Nat.even_or_odd' m with ⟨i, rfl | rfl⟩
+  · have := getD_encColoring_two_mul (fun u : Fin n => g u) ⟨i, by omega⟩
+    rw [(hget n i (by omega)).1, this]
+  · have := getD_encColoring_two_mul_add_one (fun u : Fin n => g u) ⟨i, by omega⟩
+    rw [(hget n i (by omega)).2, this]
+
+/-! ### The witness map is polynomial-time -/
+
+/-- Glue. The two bits of the gadget-node colour are a polynomial-time function of the three
+truth values (one-bit tests) and the index (unary):
+`x ↦ colorBits (gadgetNodeColor (a x) (b x) (c x) (i x))`, an `ite` tree of constants
+(`polyTimeComputable_ite`, `polyTimeComputable_and`, `polyTimeComputable_not`,
+`EncodingMachines.eqTest_poly`).
+
+**Proof sketch.** The colour table `gadgetNodeColor_eq` is a finite case split on the three
+truth values and the index; each branch is a constant chunk, so the chunk is an `ite` tree of
+constants guarded by polynomial-time one-bit tests, and both sides agree by `split_ifs`. -/
+private theorem gadgetNodeColorBits_poly {a b c : List Bool → Bool} {i : List Bool → ℕ}
+    (ha : PolyTimeComputable fun x => [a x]) (hb : PolyTimeComputable fun x => [b x])
+    (hc : PolyTimeComputable fun x => [c x])
+    (hi : PolyTimeComputable fun x => List.replicate (i x) true) :
+    PolyTimeComputable fun x => colorBits (gadgetNodeColor (a x) (b x) (c x) (i x)) := by
+  have hk : ∀ k : ℕ, PolyTimeComputable fun x => [decide (i x = k)] :=
+    fun k => eqTest_poly hi (unaryConst_poly k)
+  have na := polyTimeComputable_not ha
+  have nb := polyTimeComputable_not hb
+  have h2 := polyTimeComputable_and na nb
+  have h3 := polyTimeComputable_and h2 (polyTimeComputable_not hc)
+  have K : ∀ v : Fin 3, PolyTimeComputable fun _ : List Bool => colorBits v :=
+    fun v => polyTimeComputable_const _
+  refine (polyTimeComputable_ite (hk 5) (K 1) (polyTimeComputable_ite h3 (K 0)
+    (polyTimeComputable_ite h2
+      (polyTimeComputable_ite (hk 4) (K 2) (polyTimeComputable_ite (hk 3) (K 0)
+        (polyTimeComputable_ite (hk 2) (K 2) (polyTimeComputable_ite (hk 1) (K 1) (K 0)))))
+      (polyTimeComputable_ite (hk 4) (K 0) (polyTimeComputable_ite (hk 3) (K 2)
+        (polyTimeComputable_ite (hk 2) (K 1) (polyTimeComputable_ite (hk 1)
+          (polyTimeComputable_ite nb (K 0) (K 2))
+          (polyTimeComputable_ite na (K 0) (polyTimeComputable_ite nb (K 2) (K 0)))))))))).congr
+    fun x => ?_
+  rw [gadgetNodeColor_eq]
+  simp only [decide_eq_true_eq]
+  split_ifs <;> rfl
+
+/-- Glue. The two bits of the colour `gadgetColor (K x) (a x).getD (t x)` are a polynomial-time
+function of the unary `K x`, `t x` and the word `a x` (an `ite` tree over the vertex ranges,
+`litVal_poly`, `clauseLit_poly`, `gadgetNodeColorBits_poly`).
+
+**Proof sketch.** Split on the vertex range (`gadgetColor_eq`): the palette colour is the vertex
+itself, a literal colour is a test on `litVal_poly`, and a gadget colour is
+`gadgetNodeColorBits_poly` of the three literal values of the clause code `(t - L) / 6`
+(`unaryDivConst_poly`, `clauseLit_poly`) at index `(t - L) % 6`. -/
+private theorem gadgetColorBits_poly {K t : List Bool → ℕ} {a : List Bool → List Bool}
+    (hK : PolyTimeComputable fun x => List.replicate (K x) true)
+    (ht : PolyTimeComputable fun x => List.replicate (t x) true) (ha : PolyTimeComputable a) :
+    PolyTimeComputable fun x =>
+      colorBits (gadgetColor (K x) (fun v => (a x).getD v false) (t x)) := by
+  have hk : ∀ k : ℕ, PolyTimeComputable fun x => [decide (t x = k)] :=
+    fun k => eqTest_poly ht (unaryConst_poly k)
+  have hL : PolyTimeComputable fun x => List.replicate (3 + 2 * K x) true :=
+    unaryAdd_poly (unaryConst_poly 3) (unaryMul_poly (unaryConst_poly 2) hK)
+  have hd := unarySub_poly ht hL
+  have hlit := litVal_poly (unarySub_poly ht (unaryConst_poly 3)) ha
+  have hj : ∀ j : Fin 3, PolyTimeComputable fun x =>
+      [litVal (fun v => (a x).getD v false) (clauseLit (K x) ((t x - (3 + 2 * K x)) / 6) j)] :=
+    fun j => litVal_poly (clauseLit_poly hK (unaryDivConst_poly hd 6) j) ha
+  have hnode := gadgetNodeColorBits_poly (hj 0) (hj 1) (hj 2) (unaryModConst_poly hd 6)
+  have C : ∀ v : Fin 3, PolyTimeComputable fun _ : List Bool => colorBits v :=
+    fun v => polyTimeComputable_const _
+  refine (polyTimeComputable_ite (hk 0) (C 0) (polyTimeComputable_ite (hk 1) (C 1)
+    (polyTimeComputable_ite (hk 2) (C 2) (polyTimeComputable_ite (ltTest_poly ht hL)
+      (polyTimeComputable_ite hlit (C 1) (C 2)) hnode)))).congr fun x => ?_
+  rw [gadgetColor_eq]
+  simp only [decide_eq_true_eq]
+  split_ifs <;> rfl
+
+/-- **The witness map of the gadget graph is polynomial-time**:
+`PolyTimeComputable fun z => colorWord (pairFstD z) (pairSndD z)`.
+
+**Proof sketch.** `colorWord s a` is the concatenation over `t < gadgetN |s|` of the two bits
+`[c / 2 = 1, c % 2 = 1]` of the colour `c = gadgetColor |s| (a.getD · false) t` (the layout of
+`encColoring`: `getD_encColoring_two_mul`, `getD_encColoring_two_mul_add_one`,
+`length_encColoring`). Emit it with the tabulation host
+`ComposeMachines.polyTimeComputable_flatMap_range`: the count `gadgetN |s| = 3 + 2K + 6(2K)³` is
+unary arithmetic (`EncodingMachines.unaryAdd_poly`, `EncodingMachines.unaryMul_poly`), and each
+chunk is an `ite` tree of constants over unary comparisons of `t`, divisions and remainders by
+constants and by `2K`, and bits of `a` at unary indices (`EncodingMachines.getBit_poly`). -/
+theorem polyTimeComputable_colorWord :
+    PolyTimeComputable fun z => colorWord (pairFstD z) (pairSndD z) := by
+  have hK : PolyTimeComputable fun z => List.replicate (pairFstD z).length true :=
+    unaryLength_poly polyTimeComputable_pairFstD

+# The 3SAT → 3COL gadget graph on `Fin N`
+
+The graph of the course reduction from 3SAT to 3-colouring (`SATtoColor.ReductionGraph`,
+`Complexity/NPReductions/ThreeSATToColoring.lean`, course material of CS 294-268, UC Berkeley,
+Spring 2026) laid out on the vertex set `{0, …, N - 1}` so that a Turing machine can print its
+adjacency matrix. For `K` variables, literal codes are `ℓ < 2K` (variable `ℓ / 2`, positive iff
+`ℓ` is even), clause codes are `c < (2K)³` (the three literals `c % 2K`, `(c / 2K) % 2K`,
+`c / (2K)²`), and the vertices are
+
+* the palette `0, 1, 2` (base, true, false);
+* the literal nodes `3 + ℓ` for `ℓ < 2K`;
+* the six gadget nodes `3 + 2K + 6c + i` (`i < 6`) of **every** clause code `c < (2K)³`,
+
+`N = 3 + 2K + 6(2K)³` in all, one vertex for each vertex of `ReductionGraph` over `Fin K` (whose
+vertex type has a gadget for every clause, present in the formula or not). The edges are those of
+the course reduction: the palette triangle; the base to every literal; each positive literal to
+its negation; the gadget triangles `{0,1,2}`, `{3,4,5}` and the edge `{2,3}` inside the gadget of
+each clause **of the formula** (the test `mem`); gadget nodes `0, 1, 4` to the clause's first,
+second and third literal; gadget node `5` to the base and to the false node.
+
+This is the graph of Stockmeyer's reduction from 3SAT [St73, proof of the Theorem, part 1),
+p. 22, Fig. 1 p. 24] in the course material's variant: gadget nodes `0, …, 5` are [St73]'s
+`v₁, v₂, v₄, v₅, v₃, v₆`, with the literals attached at `0, 1, 4` as at `v₁, v₂, v₃`; the
+palette is a triangle (base, true, false) instead of [St73]'s arc `{t₁, t₂}` (base, false), and
+gadget node `5` is joined to the base as well as to the false node, which [St73] does not do.
+
+## Main definitions

+* `TCSlib.Cryptography.LevinHardness.gadgetN` — the number of vertices `N = 3 + 2K + 6(2K)³`.
+* `TCSlib.Cryptography.LevinHardness.clauseLit` — the `j`-th literal code of a clause code.
+* `TCSlib.Cryptography.LevinHardness.litVal`, `TCSlib.Cryptography.LevinHardness.clauseSat` — the
+  truth value of a literal code and of a clause code under an assignment.
+* `TCSlib.Cryptography.LevinHardness.gadgetEdge` — the directed edge relation (one orientation).
+* `TCSlib.Cryptography.LevinHardness.gadgetAdj` — the symmetric loopless adjacency.
+* `TCSlib.Cryptography.LevinHardness.gadgetGraph` — the graph on `Fin (gadgetN K)`.
+
+## Main results

+  `gadgetAdj_eq`, `gadgetGraph_apply`;

+/-- **The number of vertices of the gadget graph** for `K` variables:
+`gadgetN K = 3 + 2K + 6 (2K)³` (palette, literal nodes, six nodes per clause code). -/
+def gadgetN (K : ℕ) : ℕ := 3 + 2 * K + 6 * (2 * K) ^ 3
+
+/-- The defining equation of `gadgetN`: `gadgetN K = 3 + 2K + 6 (2K)³`. -/
+theorem gadgetN_eq (K : ℕ) : gadgetN K = 3 + 2 * K + 6 * (2 * K) ^ 3 := rfl
+
+/-- **The `j`-th literal code of the clause code `c`** for `K` variables (`j = 0, 1, 2`):
+`clauseLit K c j = (c / (2K)^j) % 2K` for `j < 2` and `c / (2K)²` for `j = 2`. -/
+def clauseLit (K c : ℕ) (j : Fin 3) : ℕ :=
+  if j = 2 then c / (2 * K) ^ 2 else c / (2 * K) ^ (j : ℕ) % (2 * K)
+
+/-- The defining equation of `clauseLit`. -/
+theorem clauseLit_eq (K c : ℕ) (j : Fin 3) :
+    clauseLit K c j = if j = 2 then c / (2 * K) ^ 2 else c / (2 * K) ^ (j : ℕ) % (2 * K) := rfl
+
+/-- **The truth value of a literal code** under the assignment `val`: the code `ℓ` is the variable
+`ℓ / 2`, positive iff `ℓ` is even, so `litVal val ℓ = val (ℓ / 2)` for even `ℓ` and
+`!val (ℓ / 2)` for odd `ℓ`. -/
+def litVal (val : ℕ → Bool) (ℓ : ℕ) : Bool :=
+  if ℓ % 2 = 0 then val (ℓ / 2) else !val (ℓ / 2)
+
+/-- The defining equation of `litVal`. -/
+theorem litVal_eq (val : ℕ → Bool) (ℓ : ℕ) :
+    litVal val ℓ = if ℓ % 2 = 0 then val (ℓ / 2) else !val (ℓ / 2) := rfl
+
+/-- **A clause code is satisfied** by `val` when one of its three literal codes is true:
+`clauseSat K val c = litVal val (clauseLit K c 0) || litVal val (clauseLit K c 1) ||
+litVal val (clauseLit K c 2)`. -/
+def clauseSat (K : ℕ) (val : ℕ → Bool) (c : ℕ) : Bool :=
+  litVal val (clauseLit K c 0) || litVal val (clauseLit K c 1) || litVal val (clauseLit K c 2)
+
+/-- The defining equation of `clauseSat`. -/
+theorem clauseSat_eq (K : ℕ) (val : ℕ → Bool) (c : ℕ) :
+    clauseSat K val c = (litVal val (clauseLit K c 0) || litVal val (clauseLit K c 1) ||
+      litVal val (clauseLit K c 2)) := rfl
+
+/-- **One orientation of the edges of the gadget graph** on vertices `t, t'` (natural numbers),
+for `K` variables and the clause-membership test `mem` on clause codes, following the edge relation
+of the course reduction (`SATtoColor.EdgeRelation`):
+* palette–palette (`t, t' < 3`);
+* base–literal (`t = 0`, `3 ≤ t' < 3 + 2K`);
+* positive literal `3 + 2v` – negative literal `3 + 2v + 1`;
+* inside the gadget of a clause code `c` with `mem c`: index pairs `(0,1), (0,2), (1,2), (2,3),
+  (3,4), (3,5), (4,5)`;
+* literal `3 + ℓ` – gadget node `i` of clause `c` when `ℓ` is the clause's literal `0, 1, 2` and
+  `i = 0, 1, 4` respectively;
+* gadget node `5` – base and gadget node `5` – false node (`t' = 0, 2`). -/
+def gadgetEdge (K : ℕ) (mem : ℕ → Bool) (t t' : ℕ) : Bool :=
+  let L := 3 + 2 * K
+  let N := gadgetN K
+  (decide (t < 3) && decide (t' < 3)) ||
+  (decide (t = 0) && decide (3 ≤ t') && decide (t' < L)) ||
+  (decide (3 ≤ t) && decide (t < L) && decide ((t - 3) % 2 = 0) && decide (t' = t + 1)) ||
+  (decide (L ≤ t) && decide (t < N) && decide (L ≤ t') && decide (t' < N) &&
+    decide ((t - L) / 6 = (t' - L) / 6) && mem ((t - L) / 6) &&
+    decide (((t - L) % 6, (t' - L) % 6) ∈
+      [(0, 1), (0, 2), (1, 2), (2, 3), (3, 4), (3, 5), (4, 5)])) ||
+  (decide (3 ≤ t) && decide (t < L) && decide (L ≤ t') && decide (t' < N) &&
+    ((decide (t - 3 = clauseLit K ((t' - L) / 6) 0) && decide ((t' - L) % 6 = 0)) ||
+     (decide (t - 3 = clauseLit K ((t' - L) / 6) 1) && decide ((t' - L) % 6 = 1)) ||
+     (decide (t - 3 = clauseLit K ((t' - L) / 6) 2) && decide ((t' - L) % 6 = 4)))) ||
+  (decide (L ≤ t) && decide (t < N) && decide ((t - L) % 6 = 5) && decide (t' = 0 ∨ t' = 2))
+
+/-- The defining equation of `gadgetEdge` (see its docstring for the cases).
+
+**Proof sketch.** `rfl`: the right-hand side is the body of `gadgetEdge` with its two `let`s
+inlined. -/
+theorem gadgetEdge_eq (K : ℕ) (mem : ℕ → Bool) (t t' : ℕ) :
+    gadgetEdge K mem t t' =
+      ((decide (t < 3) && decide (t' < 3)) ||
+      (decide (t = 0) && decide (3 ≤ t') && decide (t' < 3 + 2 * K)) ||
+      (decide (3 ≤ t) && decide (t < 3 + 2 * K) && decide ((t - 3) % 2 = 0) &&
+        decide (t' = t + 1)) ||
+      (decide (3 + 2 * K ≤ t) && decide (t < gadgetN K) && decide (3 + 2 * K ≤ t') &&
+        decide (t' < gadgetN K) &&
+        decide ((t - (3 + 2 * K)) / 6 = (t' - (3 + 2 * K)) / 6) &&
+        mem ((t - (3 + 2 * K)) / 6) &&
+        decide (((t - (3 + 2 * K)) % 6, (t' - (3 + 2 * K)) % 6) ∈
+          [(0, 1), (0, 2), (1, 2), (2, 3), (3, 4), (3, 5), (4, 5)])) ||
+      (decide (3 ≤ t) && decide (t < 3 + 2 * K) && decide (3 + 2 * K ≤ t') &&
+        decide (t' < gadgetN K) &&
+        ((decide (t - 3 = clauseLit K ((t' - (3 + 2 * K)) / 6) 0) &&
+            decide ((t' - (3 + 2 * K)) % 6 = 0)) ||
+         (decide (t - 3 = clauseLit K ((t' - (3 + 2 * K)) / 6) 1) &&
+            decide ((t' - (3 + 2 * K)) % 6 = 1)) ||
+         (decide (t - 3 = clauseLit K ((t' - (3 + 2 * K)) / 6) 2) &&
+            decide ((t' - (3 + 2 * K)) % 6 = 4)))) ||
+      (decide (3 + 2 * K ≤ t) && decide (t < gadgetN K) &&
+        decide ((t - (3 + 2 * K)) % 6 = 5) && decide (t' = 0 ∨ t' = 2))) := rfl
+
+/-- **The adjacency of the gadget graph**: `t` and `t'` are adjacent iff they are distinct and an
+edge joins them in either orientation, `gadgetAdj K mem t t' = (t ≠ t') ∧ (gadgetEdge K mem t t' ∨
+gadgetEdge K mem t' t)` (the symmetrization of `SATtoColor.ReductionGraph`). -/
+def gadgetAdj (K : ℕ) (mem : ℕ → Bool) (t t' : ℕ) : Bool :=
+  decide (t ≠ t') && (gadgetEdge K mem t t' || gadgetEdge K mem t' t)
+
+/-- The defining equation of `gadgetAdj`. -/
+theorem gadgetAdj_eq (K : ℕ) (mem : ℕ → Bool) (t t' : ℕ) :
+    gadgetAdj K mem t t' =
+      (decide (t ≠ t') && (gadgetEdge K mem t t' || gadgetEdge K mem t' t)) := rfl
+
+/-- The gadget adjacency is symmetric: `gadgetAdj K mem t t' = gadgetAdj K mem t' t`. -/
+theorem gadgetAdj_comm (K : ℕ) (mem : ℕ → Bool) (t t' : ℕ) :
+    gadgetAdj K mem t t' = gadgetAdj K mem t' t := by
+  unfold gadgetAdj
+  rw [Bool.or_comm]
+  by_cases h : t = t' <;> simp [h, eq_comm]
+
+/-- The gadget adjacency is loopless: `gadgetAdj K mem t t = false`. -/
+theorem gadgetAdj_self (K : ℕ) (mem : ℕ → Bool) (t : ℕ) : gadgetAdj K mem t t = false := by
+  simp [gadgetAdj]
+
+/-- **The gadget graph** on `Fin (gadgetN K)` for `K` variables and the clause-membership test
+`mem`: `gadgetGraph K mem u v = gadgetAdj K mem u v`. -/
+def gadgetGraph (K : ℕ) (mem : ℕ → Bool) : Graph (gadgetN K) :=
+  fun u v => gadgetAdj K mem u v
+
+/-- The entries of the gadget graph: `gadgetGraph K mem u v = gadgetAdj K mem u v`. -/
+theorem gadgetGraph_apply (K : ℕ) (mem : ℕ → Bool) (u v : Fin (gadgetN K)) :
+    gadgetGraph K mem u v = gadgetAdj K mem u v := rfl
+
+end TCSlib.Cryptography.LevinHardness

+/-- **A witness of `R_3SAT` passes the width test and gives a proper colouring**: if
+`R_3SAT s a`, then `goodS s = true` and the colouring `gadgetColor |s| (a.getD · false)` of
+`ColoringWitness.lean` is proper on `gadgetGraph |s| (memS s)`.

+**Proof sketch.** Every clause of `CNF.decode s` is satisfied, hence nonempty
+(`clause_eval_eq_true_iff`), and has at most three literals (`R_3SAT_iff`, read through
+`Bridge.widthAtMost_iff`), so `goodS s` (`goodS_iff`); every clause code accepted by `memS s` is
+satisfied (`clauseSat_of_memS`), so the colouring is proper (`properColoring_gadgetColor`). -/
+theorem goodS_and_properColoring_of_R_3SAT {s a : List Bool} (h : R_3SAT s a) :
+    goodS s = true ∧ ProperColoring (gadgetGraph s.length (memS s))
+      (fun u => gadgetColor s.length (fun v => a.getD v false) u) := by
+  obtain ⟨hw, he⟩ := (R_3SAT_iff s a).1 h
+  have hall := (cnf_eval_eq_true_iff _ _).1 he
+  refine ⟨(goodS_iff s).2 fun C hC => ⟨?_, (Bridge.widthAtMost_iff _ _).1 hw C hC⟩,
+    properColoring_gadgetColor _ _ _ fun c hc => clauseSat_of_memS he hc⟩
+  obtain ⟨ℓ, hℓ, -⟩ := (clause_eval_eq_true_iff _ _).1 (hall C hC)
+  exact List.length_pos_of_mem hℓ

+/-- **A 3-colourable gadget graph of a good string has a witness of `R_3SAT`**: if
+`goodS s = true` and `ThreeColorable (gadgetGraph |s| (memS s))`, then `∃ a, R_3SAT s a`.
+
+**Proof sketch.** Soundness transported from the course theorem
+(`exists_clauseSat_of_colorable`) gives `val`, `false` beyond `|s|`, satisfying every accepted
+clause code. Every clause `C` of `CNF.decode s` has one to three literals (`goodS_iff`) over
+variables `< |s|` (`var_lt_length_of_mem_decode`), so its code is accepted (`memS_iff`) and below
+`(2|s|)³` (`clauseCode_lt`), and `clauseSat` of its code is `C.eval val`
+(`clauseSat_clauseCode`). The word `a = [val 0, …, val (|s| - 1)]` reads back `val`. -/
+theorem exists_R_3SAT_of_colorable {s : List Bool} (hg : goodS s = true)
+    (hc : ThreeColorable (gadgetGraph s.length (memS s))) : ∃ a, R_3SAT s a := by
+  obtain ⟨val, hval0, hsat⟩ :=
+    exists_clauseSat_of_colorable ((threeColorable_gadgetGraph_iff _ _).1 hc)
+  refine ⟨(List.range s.length).map val, (R_3SAT_iff _ _).2
+    ⟨(Bridge.widthAtMost_iff _ _).2 fun C hC => ((goodS_iff s).1 hg C hC).2, ?_⟩⟩
+  have hget : (fun v => ((List.range s.length).map val).getD v false) = val := by
+    funext v
+    by_cases hv : v < s.length
+    · simp [List.getD_eq_getElem?_getD, hv]
+    · rw [hval0 v (Nat.not_lt.1 hv)]
+      simp [List.getD_eq_getElem?_getD, hv]
+  rw [hget, cnf_eval_eq_true_iff]
+  intro C hC
+  obtain ⟨h1, h3⟩ := (goodS_iff s).1 hg C hC
+  have hK : ∀ ℓ ∈ C, ℓ.1 < s.length := fun ℓ hℓ => var_lt_length_of_mem_decode hC hℓ
+  rw [← clauseSat_clauseCode val hK h1 h3]
+  exact hsat _ (clauseCode_lt hK h1) ((memS_iff s _).2 ⟨C, hC, h1, h3, rfl⟩)
+
+/-- **Membership of the padded instance** (glue): for `|colorInst s| ≤ B`,
+`padWord (colorInst s) B ∈ L3COL ↔ goodS s = true ∧ ThreeColorable (gadgetGraph |s| (memS s))`
+(`colorInst_eq`, `padWord_encGraph`, `encGraph_mem_L3COL_iff`, `threeColorable_padGraph_iff`, and
+`not_threeColorable_k4` for a string failing the width test). -/
+theorem padWord_colorInst_mem_iff {s : List Bool} {B : ℕ} (hB : (colorInst s).length ≤ B) :
+    padWord (colorInst s) B ∈ L3COL ↔
+      goodS s = true ∧ ThreeColorable (gadgetGraph s.length (memS s)) := by
+  by_cases hg : goodS s = true
+  · have hc : colorInst s = encGraph (gadgetGraph s.length (memS s)) := by
+      rw [colorInst_eq, if_pos hg]
+    rw [hc] at hB ⊢
+    rw [length_encGraph] at hB
+    rw [padWord_encGraph, encGraph_mem_L3COL_iff,
+      threeColorable_padGraph_iff _ ((Nat.le_mul_self _).trans hB)]
+    exact ⟨fun h => ⟨hg, h⟩, fun h => h.2⟩
+  · have hc : colorInst s = encGraph k4 := by rw [colorInst_eq, if_neg hg]
+    rw [hc] at hB ⊢
+    rw [length_encGraph] at hB
+    rw [padWord_encGraph, encGraph_mem_L3COL_iff,
+      threeColorable_padGraph_iff _ ((Nat.le_mul_self _).trans hB)]
+    exact iff_of_false not_threeColorable_k4 fun h => hg h.1
+
+/-- **Step 3 with padding, membership**: for `|colorInst s| ≤ B`,
+`(∃ a, R_3SAT s a) ↔ padWord (colorInst s) B ∈ L3COL`
+(`goodS_and_properColoring_of_R_3SAT`, `exists_R_3SAT_of_colorable`). -/
+theorem exists_R_3SAT_iff_padWord {s : List Bool} {B : ℕ} (hB : (colorInst s).length ≤ B) :
+    (∃ a, R_3SAT s a) ↔ padWord (colorInst s) B ∈ L3COL := by
+  rw [padWord_colorInst_mem_iff hB]
+  constructor
+  · rintro ⟨a, ha⟩
+    obtain ⟨hg, hp⟩ := goodS_and_properColoring_of_R_3SAT ha
+    exact ⟨hg, (threeColorable_iff _).2 ⟨_, hp⟩⟩
+  · rintro ⟨hg, hc⟩
+    exact exists_R_3SAT_of_colorable hg hc
+
+/-- **Step 3 with padding, witness**: for `|colorInst s| ≤ B`, `R_3SAT s a` implies
+`R3COL (padWord (colorInst s) B) (padColWord (colorWord s a) B)`: the padded colouring of
+`gadgetColor` (`padColWord_encColoring`) is proper on the padded graph
+(`properColoring_padGraph`). -/
+theorem R3COL_padWord {s a : List Bool} {B : ℕ} (ha : R_3SAT s a)
+    (hB : (colorInst s).length ≤ B) :
+    R3COL (padWord (colorInst s) B) (padColWord (colorWord s a) B) := by
+  obtain ⟨hg, hp⟩ := goodS_and_properColoring_of_R_3SAT ha
+  have hc : colorInst s = encGraph (gadgetGraph s.length (memS s)) := by
+    rw [colorInst_eq, if_pos hg]
+  rw [hc] at hB ⊢
+  rw [length_encGraph] at hB
+  have hN : gadgetN s.length ≤ B := (Nat.le_mul_self _).trans hB
+  refine (R3COL_iff _ _).2 ⟨B, padGraph (gadgetGraph s.length (memS s)) B,
+    padColoring (fun u => gadgetColor s.length (fun v => a.getD v false) u) B,
+    padWord_encGraph _ _, ?_, properColoring_padGraph B hp⟩
+  rw [colorWord_eq, padColWord_encColoring _ hN]
+
+/-- The padded word has length `B²`: `|padWord w B| = B * B` (`padWord_eq`). Glue. -/
+theorem length_padWord (w : List Bool) (B : ℕ) : (padWord w B).length = B * B := by
+  rw [padWord_eq, List.length_map, List.length_range]
+

+/-- **`3COL` is NP-hard with Levin reductions** [PS10, §4.7, Theorem 124.1, step 2 (the
+hypothesis of the "every language in NP" form)]: every language `L` with a witness relation `R`
+has a Levin reduction `(f, w)` to `(3COL, R_3COL)` with length polynomial `p ≥ id`,
+`LevinNPHard L3COL R3COL`. In Karp form this is the polynomial completeness of 3-COLORABILITY
+[St73, Theorem, p. 22]. *Deviation:* [St73] gives an instance map only; this theorem adds a
+polynomial-time witness map and an exact output length (padding), reaches 3SAT along the circuit
+route [AB09, Lemmas 6.10 and 6.11, p. 111], and ends with [St73]'s reduction from 3SAT
+[St73, proof of the Theorem, part 1), pp. 21–22, Fig. 1] in the variant of the course material
+(a palette triangle in place of [St73]'s arc `{t₁, t₂}`, and each gadget's output node joined to
+the base as well as to the false node). The witness colouring, which [St73] leaves as "the
+converse is similar", is explicit here.
+
+**Proof sketch.** Let `q, M, C, c` and the tableau instance map `f₁` come from
+`exists_tabReduction`; `f₂ = dagCktSatToSAT3`, `f₃ = colorInst`. The composite `f₃ ∘ f₂ ∘ f₁` is
+polynomial-time, so its output length is at most `A (n + 1)^a`; put
+`B(n) = A (n + 1)^a + n + 1` and `f x = padWord (f₃ (f₂ (f₁ x))) B(|x|)`, of length exactly
+`B(|x|)²` (`length_padWord`), with `p = B² ≥ id`. Membership:
+`x ∈ L ↔` the tableau circuit is satisfiable `↔ ∃ a, R_3SAT (f₂ (f₁ x)) a` (`exists_R_3SAT_iff`,
+the instance being the circuit's description) `↔ f x ∈ L3COL` (`exists_R_3SAT_iff_padWord`).
+Witnesses: `R x y` gives `R_CKT (f₁ x) (certOf q x y)`, then `R_3SAT` of the wire values
+(`R_3SAT_wireValues`), then `R3COL` of the padded colouring (`R3COL_padWord`). The witness map
+`⟨x, y⟩ ↦ padColWord (colorWord (f₂ (f₁ x)) (wireValues (f₁ x) (certOf q x y))) B(|x|)` is a
+composition of polynomial-time maps on pairs (`polyTimeComputable_certOf`,
+`polyTimeComputable_wireValues`, `polyTimeComputable_colorWord`,
+`polyTimeComputable_padColWord`). -/
+theorem levinNPHard_L3COL : LevinNPHard L3COL R3COL := by
+  refine (Bridge.levinNPHard_iff _ _).2 fun L R hR => ?_
+  obtain ⟨q, M, C, c, hf1, henc, hfan, hmem, hwit⟩ := exists_tabReduction hR
+  generalize tabInst M (q.add (.const 1)) (verTime C c (q.add (.const 1))) = f₁ at *
+  -- the composite instance map and its length bound
+  have hs : PolyTimeComputable fun x => dagCktSatToSAT3 (f₁ x) :=
+    polyTimeComputable_dagCktSatToSAT3.comp hf1
+  have hg : PolyTimeComputable fun x => colorInst (dagCktSatToSAT3 (f₁ x)) :=
+    polyTimeComputable_colorInst.comp hs
+  obtain ⟨A, a, hA⟩ := hg.output_length_le
+  let Bp : PolyExpr := (PolyExpr.normal A a).add (PolyExpr.var.add (.const 1))
+  have hBp : ∀ n, Bp.eval n = A * (n + 1) ^ a + (n + 1) := fun n => by
+    simp only [Bp, PolyExpr.eval_add, PolyExpr.eval_normal, PolyExpr.eval_var,
+      PolyExpr.eval_const]
+  have hB : ∀ x : List Bool, (colorInst (dagCktSatToSAT3 (f₁ x))).length ≤ Bp.eval x.length :=
+    fun x => by rw [hBp]; have := hA x; omega
+  refine ⟨fun x => padWord (colorInst (dagCktSatToSAT3 (f₁ x))) (Bp.eval x.length),
+    fun x y => padColWord (colorWord (dagCktSatToSAT3 (f₁ x))
+      (wireValues (f₁ x) (certOf q x y))) (Bp.eval x.length),
+    Bp.mul Bp, ⟨?_, ?_, fun x => ?_, fun x y hxy => ?_, fun x => ?_, fun N => ?_⟩⟩
+  · -- the instance map is polynomial-time
+    exact (polyTimeComputable_padWord.comp (hg.pairEncode (polyTimeComputable_unaryPoly Bp))).congr
+      fun x => by
+        simp only [Function.comp_apply, pairFstD_pairEncode, pairSndD_pairEncode,
+          List.length_replicate]
+  · -- the witness map is polynomial-time on ⟨x, y⟩
+    have hwv : PolyTimeComputable fun z =>
+        wireValues (f₁ (pairFstD z)) (certOf q (pairFstD z) (pairSndD z)) :=
+      (polyTimeComputable_wireValues.comp
+        ((hf1.comp polyTimeComputable_pairFstD).pairEncode (polyTimeComputable_certOf q))).congr
+        fun z => by simp only [Function.comp_apply, pairFstD_pairEncode, pairSndD_pairEncode]
+    have hcw : PolyTimeComputable fun z => colorWord (dagCktSatToSAT3 (f₁ (pairFstD z)))
+        (wireValues (f₁ (pairFstD z)) (certOf q (pairFstD z) (pairSndD z))) :=
+      (polyTimeComputable_colorWord.comp
+        ((hs.comp polyTimeComputable_pairFstD).pairEncode hwv)).congr
+        fun z => by simp only [Function.comp_apply, pairFstD_pairEncode, pairSndD_pairEncode]
+    exact (polyTimeComputable_padColWord.comp (hcw.pairEncode
+      ((polyTimeComputable_unaryPoly Bp).comp polyTimeComputable_pairFstD))).congr
+      fun z => by
+        simp only [Function.comp_apply, pairFstD_pairEncode, pairSndD_pairEncode,
+          List.length_replicate]
+  · -- membership
+    rw [hmem x, ← exists_R_3SAT_iff _ (hfan x), ← henc x]
+    exact exists_R_3SAT_iff_padWord (hB x)
+  · -- witnesses
+    exact R3COL_padWord (R_3SAT_wireValues (hwit x y hxy)) (hB x)
+  · -- exact length `B(|x|)²`
+    rw [length_padWord, PolyExpr.eval_mul]
+  · -- `B² ≥ id`
+    rw [PolyExpr.eval_mul, hBp]
+    exact (Nat.le_add_left (N + 1) _).trans' (Nat.le_succ N) |>.trans (Nat.le_mul_self _)
+
+/-- **[PS10, Theorem 124.1], "if one-way permutations exist then every language in NP has a
+zero-knowledge proof", for every witness relation, unconditionally**: the conclusion of
+`zk_proof_of_witnessRelation_of_owp` without its hypothesis `LevinNPHard L3COL R3COL`, which is
+`levinNPHard_L3COL`. -/
+theorem zk_proof_of_witnessRelation_of_owp_unconditional
+    (how : ∃ P : PermFamily, OneWayPermutation P) {L : Language Bool}
+    {R : List Bool → List Bool → Prop} (hR : WitnessRelation L R) :
+    ∃ (Com : List Bool → List Bool) (ℓ : PolyExpr), Commitment Com ℓ ∧
+      ∃ (f : List Bool → List Bool) (w : List Bool → List Bool → List Bool) (p : PolyExpr),
+        LevinReduction L L3COL R R3COL f w p ∧ (∀ N, N ≤ p.eval N) ∧
+        EfficientProverProof ((colProtocolRep Com ℓ).pullback f w p) L R ∧
+        ZK ((colProtocol Com ℓ).pullback f w p) R ∧
+        EfficientProverProofErr ((colProtocol Com ℓ).pullback f w p) L R
+          (fun N => 1 - 1 / (p.eval N : ℝ) + (1 / 2 : ℝ) ^ p.eval N) :=
+  zk_proof_of_witnessRelation_of_owp levinNPHard_L3COL how hR
+
+/-- **[PS10, Theorem 124.1], "if one-way permutations exist then every language in NP has a
+zero-knowledge proof", one-iteration form, unconditionally**: the conclusion of
+`zk_proof_of_mem_NP_of_owp` without its hypothesis `LevinNPHard L3COL R3COL`, which is
+`levinNPHard_L3COL`. -/
+theorem zk_proof_of_mem_NP_of_owp_unconditional
+    (how : ∃ (P : PermFamily) (h : ∀ n, (Fin n → Bool) → Bool),
+      OneWayPermutation P ∧ P.HardCore h)
+    {L : Language Bool} (hL : L ∈ NP) :
+    ∃ R : List Bool → List Bool → Prop, WitnessRelation L R ∧
+      (∃ π : Protocol, EfficientProverProof π L R) ∧
+      ∃ (π : Protocol) (p : PolyExpr), (∀ N, N ≤ p.eval N) ∧ ZK π R ∧
+        EfficientProverProofErr π L R
+          (fun N => 1 - 1 / (p.eval N : ℝ) + (1 / 2 : ℝ) ^ p.eval N) :=
+  zk_proof_of_mem_NP_of_owp levinNPHard_L3COL how hL

+# Padding graphs and colourings to an exact vertex count
+
+The Levin reduction to `3COL` ([AB09, §2.3]; [PS10, §4.7, Theorem 124.1, step 2]) is assembled
+from maps whose output graphs should have an exact, length-determined number of vertices `B`.
+This file pads a graph `G : Graph n` with isolated vertices to `padGraph G B : Graph B` and a
+colouring with colour `0` to `padColoring c B`, together with the string maps on encodings:
+`padWord` on adjacency matrices (`encGraph`) and `padColWord` on colourings (`encColoring`),
+both polynomial-time with `B` given in unary. Framework glue, no textbook counterpart.
+
+## Main definitions
+
+* `TCSlib.Cryptography.LevinHardness.padGraph` — `G` padded with isolated vertices to `B`.
+* `TCSlib.Cryptography.LevinHardness.padColoring` — a colouring extended by colour `0`.
+* `TCSlib.Cryptography.LevinHardness.padWord` — the padding on adjacency-matrix words.
+* `TCSlib.Cryptography.LevinHardness.padColWord` — the padding on colouring words.
+
+## Main results
+
+* `TCSlib.Cryptography.LevinHardness.properColoring_padGraph`,
+  `TCSlib.Cryptography.LevinHardness.threeColorable_padGraph_iff` — padding preserves proper
+  colourings and (for `n ≤ B`) 3-colourability both ways.
+* `TCSlib.Cryptography.LevinHardness.padWord_encGraph`,
+  `TCSlib.Cryptography.LevinHardness.padColWord_encColoring` — the string maps encode the
+  paddings.
+* `TCSlib.Cryptography.LevinHardness.polyTimeComputable_padWord`,
+  `TCSlib.Cryptography.LevinHardness.polyTimeComputable_padColWord` — they are polynomial-time.
+

+/-! ### Padding graphs and colourings -/
+
+/-- **A graph padded with isolated vertices**: for `G : Graph n` and a vertex count `B`,
+`padGraph G B u v = G u v` if `u, v < n` and `false` otherwise (for `B < n` the graph is
+truncated to its first `B` vertices). -/
+def padGraph {n : ℕ} (G : Graph n) (B : ℕ) : Graph B := fun u v =>
+  if h : (u : ℕ) < n ∧ (v : ℕ) < n then G ⟨u, h.1⟩ ⟨v, h.2⟩ else false
+
+/-- The entries of the padded graph:
+`padGraph G B u v = if h : u < n ∧ v < n then G ⟨u, h.1⟩ ⟨v, h.2⟩ else false`. -/
+theorem padGraph_apply {n : ℕ} (G : Graph n) (B : ℕ) (u v : Fin B) :
+    padGraph G B u v =
+      if h : (u : ℕ) < n ∧ (v : ℕ) < n then G ⟨u, h.1⟩ ⟨v, h.2⟩ else false := rfl
+
+/-- **A colouring extended by colour `0`**: for `c : Fin n → Fin 3`,
+`padColoring c B u = c u` if `u < n` and `0` otherwise. -/
+def padColoring {n : ℕ} (c : Fin n → Fin 3) (B : ℕ) : Fin B → Fin 3 := fun u =>
+  if h : (u : ℕ) < n then c ⟨u, h⟩ else 0
+
+/-- The values of the padded colouring: `padColoring c B u = if h : u < n then c ⟨u, h⟩ else 0`. -/
+theorem padColoring_apply {n : ℕ} (c : Fin n → Fin 3) (B : ℕ) (u : Fin B) :
+    padColoring c B u = if h : (u : ℕ) < n then c ⟨u, h⟩ else 0 := rfl
+
+/-- **Padding preserves proper colourings**: if `ProperColoring G c`, then
+`ProperColoring (padGraph G B) (padColoring c B)` (every edge of the padded graph is an edge of
+`G` between vertices `< n`, coloured by `c`; no hypothesis on `B` is needed). -/
+theorem properColoring_padGraph {n : ℕ} {G : Graph n} {c : Fin n → Fin 3} (B : ℕ)
+    (hc : ProperColoring G c) : ProperColoring (padGraph G B) (padColoring c B) := by
+  rw [ProperColoring_iff] at hc ⊢
+  intro u v huv
+  rw [padGraph_apply] at huv
+  by_cases h : (u : ℕ) < n ∧ (v : ℕ) < n
+  · rw [dif_pos h] at huv
+    rw [padColoring_apply, padColoring_apply, dif_pos h.1, dif_pos h.2]
+    exact hc _ _ huv
+  · rw [dif_neg h] at huv
+    exact absurd huv Bool.false_ne_true
+
+/-- **Padding preserves 3-colourability both ways**: for `n ≤ B`,
+`ThreeColorable (padGraph G B) ↔ ThreeColorable G`. Forward: restrict a colouring of the padded
+graph to the first `n` vertices; backward: `properColoring_padGraph`. -/
+theorem threeColorable_padGraph_iff {n : ℕ} (G : Graph n) {B : ℕ} (hnB : n ≤ B) :
+    ThreeColorable (padGraph G B) ↔ ThreeColorable G := by
+  rw [threeColorable_iff, threeColorable_iff]
+  constructor
+  · rintro ⟨c', hc'⟩
+    refine ⟨fun u => c' ⟨u, by omega⟩, ?_⟩
+    rw [ProperColoring_iff] at hc' ⊢
+    intro u v huv
+    refine hc' ⟨u, by omega⟩ ⟨v, by omega⟩ ?_
+    rw [padGraph_apply, dif_pos ⟨u.2, v.2⟩]
+    exact huv
+  · rintro ⟨c, hc⟩
+    exact ⟨_, properColoring_padGraph B hc⟩
+
+/-! ### Padding adjacency-matrix words -/
+
+/-- **The graph padding on adjacency-matrix words**: with `n = ⌊√|w|⌋` (the vertex count read
+by `decGraph`), `padWord w B` is the `B × B` word whose bit `k` (row `k / B`, column `k % B`) is
+`[k / B < n] ∧ [k % B < n] ∧ w.getD (k % B + n (k / B)) false`. -/
+def padWord (w : List Bool) (B : ℕ) : List Bool :=
+  (List.range (B * B)).map fun k =>
+    decide (k / B < Nat.sqrt w.length) && decide (k % B < Nat.sqrt w.length) &&
+      w.getD (k % B + Nat.sqrt w.length * (k / B)) false
+
+/-- The defining equation of `padWord`. -/
+theorem padWord_eq (w : List Bool) (B : ℕ) :
+    padWord w B = (List.range (B * B)).map fun k =>
+      decide (k / B < Nat.sqrt w.length) && decide (k % B < Nat.sqrt w.length) &&
+        w.getD (k % B + Nat.sqrt w.length * (k / B)) false := rfl
+
+/-- **`padWord` encodes the padded graph**: `padWord (encGraph G) B = encGraph (padGraph G B)`
+(for every `B`; for `B < n` both sides truncate).
+
+**Proof sketch.** Both words have `B²` bits. Bit `m = (m % B) + B (m / B)` of the right side is
+`padGraph G B (m / B) (m % B)` (`encGraph_getD_nat`, `Nat.mod_add_div`); on the left
+`⌊√(n²)⌋ = n` (`length_encGraph`, `Nat.sqrt_eq`), and below `n` the bit is
+`(encGraph G).getD (m % B + n (m / B)) false = G (m / B) (m % B)` (`encGraph_getD_nat`). -/
+theorem padWord_encGraph {n : ℕ} (G : Graph n) (B : ℕ) :
+    padWord (encGraph G) B = encGraph (padGraph G B) := by
+  have hn : Nat.sqrt (encGraph G).length = n := by rw [length_encGraph, Nat.sqrt_eq]
+  refine List.ext_getElem (by rw [padWord_eq, List.length_map, List.length_range,
+    length_encGraph]) fun m h1 h2 => ?_
+  have hm : m < B * B := by rwa [padWord_eq, List.length_map, List.length_range] at h1
+  have hB : 0 < B := Nat.pos_of_ne_zero fun h => by simp [h] at hm
+  have hu : m / B < B := (Nat.div_lt_iff_lt_mul hB).2 hm
+  have hkey := encGraph_getD_nat (padGraph G B) ⟨m / B, hu⟩ ⟨m % B, Nat.mod_lt _ hB⟩
+  rw [show m % B + B * (m / B) = m from Nat.mod_add_div m B, List.getD_eq_getElem?_getD,
+    List.getElem?_eq_getElem h2, Option.getD_some] at hkey
+  rw [hkey, padGraph_apply]
+  simp only [padWord_eq, List.getElem_map, List.getElem_range, hn]
+  by_cases h : m / B < n ∧ m % B < n
+  · rw [dif_pos h, ← encGraph_getD_nat G ⟨m / B, h.1⟩ ⟨m % B, h.2⟩]
+    simp [h.1, h.2]
+  · rw [dif_neg h]
+    by_cases h' : m / B < n
+    · have h'' : ¬ m % B < n := fun h'' => h ⟨h', h''⟩
+      simp [h'']
+    · simp [h']
+
+/-- **The graph padding is polynomial-time** with the vertex count in unary:
+`PolyTimeComputable fun z => padWord (pairFstD z) (pairSndD z).length`.
+
+**Proof sketch.** Emit the bits `k < B²` with the tabulation host
+`ComposeMachines.polyTimeComputable_flatMap_range` (`List.map_eq_flatMap`); bit `k` is a
+conjunction (`polyTimeComputable_and`) of the unary comparisons `k / B < n`, `k % B < n`
+(`EncodingMachines.ltTest_poly`, `EncodingMachines.divNum_poly`, `EncodingMachines.modNum_poly`
+with their readers, `n = ⌊√|w|⌋` by `EncodingMachines.unarySqrt_poly`) and the bit of `w` at the
+unary index `k % B + n (k / B)` (`EncodingMachines.getBit_poly`). The count `B²` is
+`EncodingMachines.unaryMul_poly`. -/
+theorem polyTimeComputable_padWord :
+    PolyTimeComputable fun z => padWord (pairFstD z) (pairSndD z).length := by
+  have hk : PolyTimeComputable fun w => List.replicate (pairFstD w).length true :=
+    unaryLength_poly polyTimeComputable_pairFstD
+  have hW : PolyTimeComputable fun w => pairFstD (pairSndD w) :=
+    polyTimeComputable_pairFstD.comp polyTimeComputable_pairSndD
+  have hB : PolyTimeComputable fun w => List.replicate (pairSndD (pairSndD w)).length true :=
+    unaryLength_poly (polyTimeComputable_pairSndD.comp polyTimeComputable_pairSndD)
+  have hn := unarySqrt_poly (unaryLength_poly hW)
+  have hu : PolyTimeComputable fun w =>
+      List.replicate ((pairFstD w).length / (pairSndD (pairSndD w)).length) true :=
+    (divNum_poly hk hB).congr fun w => by rw [divNum_eq]
+  have hv : PolyTimeComputable fun w =>
+      List.replicate ((pairFstD w).length % (pairSndD (pairSndD w)).length) true :=
+    (modNum_poly hk hB).congr fun w => by rw [modNum_eq]
+  have hE := polyTimeComputable_and (polyTimeComputable_and (ltTest_poly hu hn)
+    (ltTest_poly hv hn)) (getBit_poly hW (unaryAdd_poly hv (unaryMul_poly hn hu)))
+  have hB0 : PolyTimeComputable fun z => List.replicate (pairSndD z).length true :=
+    unaryLength_poly polyTimeComputable_pairSndD
+  refine (SecureComputation.ComposeMachines.polyTimeComputable_flatMap_range
+    (E := fun z k => [decide (k / (pairSndD z).length < Nat.sqrt (pairFstD z).length) &&
+      decide (k % (pairSndD z).length < Nat.sqrt (pairFstD z).length) &&
+      (pairFstD z).getD (k % (pairSndD z).length +
+        Nat.sqrt (pairFstD z).length * (k / (pairSndD z).length)) false])
+    hE (unaryMul_poly hB0 hB0)).congr fun z => ?_
+  rw [padWord_eq, List.map_eq_flatMap]

+/-! ### Padding colouring words -/

+/-- **The colouring padding on words**: `padColWord w B = w ++ 0^(2B - |w|)` (two zero bits,
+colour `0`, for each added vertex). Near-relative: `EncodingMachines.padTo (2B) w`, which agrees
+with it when `|w| ≤ 2B`. -/
+def padColWord (w : List Bool) (B : ℕ) : List Bool :=
+  w ++ List.replicate (2 * B - w.length) false
+
+/-- The defining equation of `padColWord`: `padColWord w B = w ++ 0^(2B - |w|)`. -/
+theorem padColWord_eq (w : List Bool) (B : ℕ) :
+    padColWord w B = w ++ List.replicate (2 * B - w.length) false := rfl
+
+/-- **`padColWord` encodes the padded colouring**: for `n ≤ B`,
+`padColWord (encColoring c) B = encColoring (padColoring c B)`.
+
+**Proof sketch.** Both words have `2B` bits (`length_encColoring`). For a vertex `i < n` the
+bits `2i, 2i + 1` come from `encColoring c` on both sides (`List.getD_append`,
+`getD_encColoring_two_mul`, `getD_encColoring_two_mul_add_one`); for `n ≤ i < B` they are the
+padding zeros on the left (`List.getD_append_right`, `List.getD_replicate`) and the two bits of
+colour `0` on the right. -/
+theorem padColWord_encColoring {n : ℕ} (c : Fin n → Fin 3) {B : ℕ} (hnB : n ≤ B) :
+    padColWord (encColoring c) B = encColoring (padColoring c B) := by
+  have hlen : (padColWord (encColoring c) B).length = B * 2 := by
+    rw [padColWord_eq, List.length_append, List.length_replicate, length_encColoring]
+    omega
+  have hgetD : ∀ (l : List Bool) (m : ℕ) (h : m < l.length), l[m] = l.getD m false :=
+    fun l m h => by rw [List.getD_eq_getElem?_getD, List.getElem?_eq_getElem h, Option.getD_some]
+  refine List.ext_getElem (by rw [hlen, length_encColoring]) fun m h1 h2 => ?_
+  rw [hgetD _ _ h1, hgetD _ _ h2, padColWord_eq]
+  rw [hlen] at h1
+  rcases Nat.even_or_odd' m with ⟨i, rfl | rfl⟩
+  · have hr := getD_encColoring_two_mul (padColoring c B) ⟨i, by omega⟩
+    rw [hr, padColoring_apply]
+    by_cases hi : i < n
+    · rw [List.getD_append _ _ _ _ (by rw [length_encColoring]; omega), dif_pos hi]
+      exact getD_encColoring_two_mul c ⟨i, hi⟩
+    · rw [List.getD_append_right _ _ _ _ (by rw [length_encColoring]; omega),
+        List.getD_replicate _ (by rw [length_encColoring]; omega), dif_neg hi]
+      decide
+  · have hr := getD_encColoring_two_mul_add_one (padColoring c B) ⟨i, by omega⟩
+    rw [hr, padColoring_apply]
+    by_cases hi : i < n
+    · rw [List.getD_append _ _ _ _ (by rw [length_encColoring]; omega), dif_pos hi]
+      exact getD_encColoring_two_mul_add_one c ⟨i, hi⟩
+    · rw [List.getD_append_right _ _ _ _ (by rw [length_encColoring]; omega),
+        List.getD_replicate _ (by rw [length_encColoring]; omega), dif_neg hi]
+      decide
+
+/-- **The colouring padding is polynomial-time** with the vertex count in unary:
+`PolyTimeComputable fun z => padColWord (pairFstD z) (pairSndD z).length` (append the zeros
+`EncodingMachines.zeros_poly` of the unary length `2B - |w|`, `EncodingMachines.unarySub_poly`,
+`EncodingMachines.unaryMul_poly`). -/
+theorem polyTimeComputable_padColWord :
+    PolyTimeComputable fun z => padColWord (pairFstD z) (pairSndD z).length :=
+  (polyTimeComputable_pairFstD.append (EncodingMachines.zeros_poly
+    (unarySub_poly (unaryMul_poly (unaryConst_poly 2)
+      (unaryLength_poly polyTimeComputable_pairSndD))
+      (unaryLength_poly polyTimeComputable_pairFstD)))).congr fun z => by rw [padColWord_eq]

+# The witness relations of the circuit route to `3COL`
+
+The two intermediate witness relations of the Levin reduction
+`(L, R) → (CKT-SAT, R_CKT) → (3SAT, R_3SAT) → (3COL, R_3COL)` ([AB09, §6.1.2, Lemmas 6.10 and
+6.11, p. 111]; [PS10, §4.7, Theorem 124.1, step 2]), stated once and publicly so that the four
+files of `ZeroKnowledge/LevinHardness/` read them through their readers.
+
+* `R_CKT d u`: the string `d` describes (`BoolCircuit.DAGCircuit.encode`) a fan-in-two circuit
+  that outputs `1` on the input read from `u` with default `false` ([AB09, Def 6.9] takes
+  `u ∈ {0,1}ⁿ`; reading `u` with default `false` is TCSlib's certificate convention, that of
+  `BoolCircuit.CVALPrefix`).
+* `R_3SAT s a`: the formula `CNF.decode s` is a 3-CNF satisfied by the assignment read from the
+  word `a` with default `false`.

+convention `a.getD v false` of the private `satAssignment` of `Complexity/ClassNP/SAT.lean`.

+## Main results

+/-- **The witness relation of CKT-SAT** [AB09, Def 6.9]: `R_CKT d u` holds when `d` is the
+description `C.encode` of a fan-in-two circuit `C : DAGCircuit n` with
+`C.eval (fun i => u.getD i false) = true`. -/
+def R_CKT (d u : List Bool) : Prop :=
+  ∃ (n : ℕ) (C : DAGCircuit n), C.IsFaninTwo ∧ C.encode = d ∧

+theorem R_CKT_iff (d u : List Bool) :
+    R_CKT d u ↔ ∃ (n : ℕ) (C : DAGCircuit n), C.IsFaninTwo ∧ C.encode = d ∧
+      C.eval (fun i => u.getD i false) = true := Iff.rfl

+`CNF.decode s` has at most three literals per clause and
+`(CNF.decode s).eval (fun v => a.getD v false) = true`. -/
+def R_3SAT (s a : List Bool) : Prop :=
+  (CNF.decode s).WidthAtMost 3 ∧ (CNF.decode s).eval (fun v => a.getD v false) = true
+
+/-- The defining property of `R_3SAT`: `R_3SAT s a` iff `(CNF.decode s).WidthAtMost 3` and
+`(CNF.decode s).eval (fun v => a.getD v false) = true`. -/
+theorem R_3SAT_iff (s a : List Bool) :
+    R_3SAT s a ↔ (CNF.decode s).WidthAtMost 3 ∧
+      (CNF.decode s).eval (fun v => a.getD v false) = true := Iff.rfl

+`c ∈ gadgetFormula K mem → mem ((clauseEquiv K).symm c) = true`. -/
+theorem mem_of_mem_gadgetFormula {K : ℕ} {mem : ℕ → Bool} {c : SATtoColor.Clause (Fin K)}
+    (hc : c ∈ gadgetFormula K mem) : mem ((clauseEquiv K).symm c) = true := by
+  rw [gadgetFormula_eq, List.mem_map] at hc
+  obtain ⟨x, hx, rfl⟩ := hc
+  rw [Equiv.symm_apply_apply]
+  exact (List.mem_filter.1 hx).2
+
+/-- **The homomorphism from the course graph into the gadget graph**:
+`ReductionGraph (gadgetFormula K mem) →g gadgetSimpleGraph K mem`, the map `(vertexEquiv K).symm`
+(`gadgetEdge_or_of_edgeRelation`, distinct vertices having distinct numbers). -/
+def reductionGraphHom (K : ℕ) (mem : ℕ → Bool) :
+    ReductionGraph (gadgetFormula K mem) →g gadgetSimpleGraph K mem where
+  toFun := (vertexEquiv K).symm
+  map_rel' {a b} h := by
+    obtain ⟨hne, hab⟩ := (Bridge.reductionGraph_adj _ a b).1 h
+    rw [gadgetSimpleGraph_adj, gadgetAdj_eq, Bool.and_eq_true, decide_eq_true_eq]
+    refine ⟨fun he => hne ((vertexEquiv K).symm.injective (Fin.ext he)), ?_⟩
+    rcases hab with hab | hab
+    · exact gadgetEdge_or_of_edgeRelation K mem _ (fun c => mem_of_mem_gadgetFormula) a b hab
+    · rw [Bool.or_comm]
+      exact gadgetEdge_or_of_edgeRelation K mem _ (fun c => mem_of_mem_gadgetFormula) b a hab

+/-- The underlying map of `reductionGraphHom` is `(vertexEquiv K).symm`. -/
+theorem reductionGraphHom_apply (K : ℕ) (mem : ℕ → Bool) (a : OutputVertex (Fin K)) :
+    reductionGraphHom K mem a = (vertexEquiv K).symm a := rfl

+/-- **Soundness, transported from the course theorem**: if the gadget graph is 3-colourable then
+the course formula of its codes is satisfiable,
+`(gadgetSimpleGraph K mem).Colorable 3 → IsSatisfiable (gadgetFormula K mem)`
+(`SATtoColor.SATtoColorReduction` after pulling the colouring back along
+`reductionGraphHom`). -/
+theorem isSatisfiable_of_colorable {K : ℕ} {mem : ℕ → Bool}
+    (h : (gadgetSimpleGraph K mem).Colorable 3) : IsSatisfiable (gadgetFormula K mem) := by
+  obtain ⟨C⟩ := h
+  exact (SATtoColorReduction _).2 ((Bridge.is3Colorable_iff _).2
+    ⟨C.comp (reductionGraphHom K mem)⟩)
+
+/-- Glue: the course truth value of a literal of code `ℓ` is `litVal` of the code, for the
+assignment `val v = a v` on `v < K` (`false` beyond):
+`SatisfiesLiteral a (litEquiv K ℓ) = litVal val ℓ`. -/
+theorem satisfiesLiteral_litEquiv {K : ℕ} (a : Fin K → Bool) (ℓ : Fin (2 * K)) :
+    SatisfiesLiteral a (litEquiv K ℓ) =
+      litVal (fun v => if h : v < K then a ⟨v, h⟩ else false) ℓ := by
+  have hl := ℓ.isLt
+  have hK : (ℓ : ℕ) / 2 < K := by omega
+  rw [litEquiv_apply, litVal_eq]
+  by_cases h : (ℓ : ℕ) % 2 = 0
+  · rw [if_pos h, if_pos h, Bridge.satisfiesLiteral_pos, dif_pos hK]
+  · rw [if_neg h, if_neg h, Bridge.satisfiesLiteral_neg, dif_pos hK]
+
+/-- **A 3-colourable gadget graph has a satisfying assignment of its clause codes**: there is
+`val : ℕ → Bool`, `false` on `v ≥ K`, with `clauseSat K val c = true` for every code `c < (2K)³`
+with `mem c = true`.
+
+**Proof sketch.** By `isSatisfiable_of_colorable` some `a : Fin K → Bool` satisfies every clause
+of `gadgetFormula K mem`; for a code `c` with `mem c`, the clause `clauseEquiv K c` is in the
+formula, its three literals are `litEquiv K` of the codes `clauseLit K c j`
+(`clauseEquiv_apply`), and their course values are `litVal val` of those codes
+(`satisfiesLiteral_litEquiv`), which is `clauseSat K val c` (`clauseSat_eq`). -/
+theorem exists_clauseSat_of_colorable {K : ℕ} {mem : ℕ → Bool}
+    (h : (gadgetSimpleGraph K mem).Colorable 3) :
+    ∃ val : ℕ → Bool, (∀ v, K ≤ v → val v = false) ∧
+      ∀ c, c < (2 * K) ^ 3 → mem c = true → clauseSat K val c = true := by
+  obtain ⟨a, ha⟩ := (Bridge.isSatisfiable_iff _).1 (isSatisfiable_of_colorable h)
+  refine ⟨fun v => if h : v < K then a ⟨v, h⟩ else false,
+    fun v hv => dif_neg (Nat.not_lt.2 hv), fun c hc hm => ?_⟩
+  rw [Bridge.satisfiesSat3_eq, List.all_eq_true] at ha
+  have hmem : clauseEquiv K ⟨c, hc⟩ ∈ gadgetFormula K mem := by
+    rw [gadgetFormula_eq]
+    exact List.mem_map_of_mem (List.mem_filter.2 ⟨List.mem_finRange _, hm⟩)
+  have hs := ha _ hmem
+  rw [Bridge.satisfiesClause_eq, clauseEquiv_apply, satisfiesLiteral_litEquiv,
+    satisfiesLiteral_litEquiv, satisfiesLiteral_litEquiv] at hs
+  rw [clauseSat_eq]
+  exact hs

+    PolyTimeComputable fun x => List.replicate (f x % k) true :=

+/-- **The literal codes of a clause code are unary polynomial-time**: for unary polynomial-time
+`K`, `c`, the map `x ↦ 1^(clauseLit (K x) (c x) j)` is polynomial-time (`clauseLit_eq`:
+`c % 2K`, `c / 2K % 2K`, `c / (2K)²`, by `EncodingMachines.divNum_poly`/`modNum_poly` read
+through `divNum_eq`/`modNum_eq`, and `(2K)²` by `EncodingMachines.unaryPolyOf_poly`). Glue. -/
+theorem clauseLit_poly {K c : List Bool → ℕ}
+    (hK : PolyTimeComputable fun x => List.replicate (K x) true)
+    (hc : PolyTimeComputable fun x => List.replicate (c x) true) (j : Fin 3) :

+  have h2K := unaryMul_poly (unaryConst_poly 2) hK
+  simp only [clauseLit_eq]
+  fin_cases j
+  · simpa [modNum_eq] using modNum_poly hc h2K
+  · simpa [divNum_eq, modNum_eq] using modNum_poly (divNum_poly hc h2K) h2K
+  · simpa [divNum_eq] using divNum_poly hc (unaryPolyOf_poly (.pow .var 2) h2K)

+polynomial-time code `ℓ` and a polynomial-time assignment word `a`, the map
+`x ↦ [litVal (fun v => (a x).getD v false) (ℓ x)]` is polynomial-time (the parity of `ℓ x` by
+`unaryModConst_poly`, the variable bit by `EncodingMachines.getBit_poly`, `litVal_eq`). Glue. -/
+theorem litVal_poly {ℓ : List Bool → ℕ} {a : List Bool → List Bool}
+    (hℓ : PolyTimeComputable fun x => List.replicate (ℓ x) true) (ha : PolyTimeComputable a) :
+    PolyTimeComputable fun x => [litVal (fun v => (a x).getD v false) (ℓ x)] := by

+  have hbit := getBit_poly ha (unaryDivConst_poly hℓ 2)
+  refine (polyTimeComputable_ite hpar hbit (polyTimeComputable_not hbit)).congr fun x => ?_
+  rw [litVal_eq]
+  by_cases h : ℓ x % 2 = 0 <;> simp [h]

+# Step 1 of the circuit route: from a witness relation to CKT-SAT, with the witness
+
+The first step of the Levin reduction `(L, R) → (CKT-SAT, R_CKT)` [AB09, Lemma 6.10]: for a
+witness relation `R` of `L` (`WitnessRelation`, [PS10, Definition 191.11]) with length bound `q`
+and polynomial-time decider `b` on `⟨x, y⟩`,
+
+* the **certificate** of a witness `y` of `x` is `certOf q x y = y ++ 1 0^{q(|x|) - |y|}`, of
+  length exactly `q(|x|) + 1`, from which `y` is recovered by stripping at the last `1`
+  (`Turing.splitAtLastTrue`, `Complexity.PolyHierarchy.splitAtLastTrue_marker`);
+* the **verifier test** `verTest b ⟨x, u⟩` strips the certificate `u` with the catalog's marker
+  strip (`Turing.FinTM.computesFunInTime_stripLast`) and runs `b` on `⟨x, y⟩`, rejecting a
+  certificate without marker; it is polynomial-time;
+* the **instance map** prints, with the polynomial-time emitter of [AB09, Remark 6.7]
+  (`Complexity.UTab.tabEmit`), the uniform configuration tableau of a machine `M` computing the
+  test, run for `t(|x|)` steps on the virtual input `⟨x, []⟩ ++ u = ⟨x, u⟩`
+  (`Complexity.BinArith.pairEncode_nil_append`): `⟨x, []⟩` is hard-wired and the certificate `u`
+  is the circuit's input (`Complexity.cfgTab`);
+* the **witness map** is the certificate itself: the tableau circuit outputs `1` on
+  `certOf q x y` whenever `R x y`.
+
+The circuit outputs `1` on a certificate `u` iff `u` strips to a witness, so it is satisfiable
+iff `x ∈ L`.
+
+**Overlap (disclosed, brief EF-H12 item (a)).** The instance map re-instantiates the core of the
+proof of `BoolCircuit.dagCktSatLang_NPHard` (`CircuitSatNPHard.lean:93-127`, [AB09, Lemma 6.10])
+for this explicit verifier: that theorem is existential and provides no witness map. The
+difference: the hard-wired prefix is `⟨x, []⟩` (so the verifier reads a pair and needs no
+length-split search) and the time bound is an explicit `PolyExpr`.

+/-! ### The certificate -/
+
+/-- **The certificate of a witness** `y` of `x` for the length bound `q`: the witness followed
+by the marker `1` and `q(|x|) - |y|` zeros, `certOf q x y = y ++ 1 0^{q(|x|) - |y|}`
+[PS10, §4.7, Theorem 124.1, step 2]. Glue. -/
+def certOf (q : PolyExpr) (x y : List Bool) : List Bool :=
+  y ++ true :: List.replicate (q.eval x.length - y.length) false
+
+/-- The defining equation of `certOf`: `certOf q x y = y ++ 1 0^{q(|x|) - |y|}`. -/
+theorem certOf_eq (q : PolyExpr) (x y : List Bool) :
+    certOf q x y = y ++ true :: List.replicate (q.eval x.length - y.length) false := rfl
+
+/-- A certificate of a witness within the bound has length exactly `q(|x|) + 1`:
+`|y| ≤ q(|x|) → |certOf q x y| = q(|x|) + 1`. -/
+theorem length_certOf (q : PolyExpr) (x y : List Bool) (hy : y.length ≤ q.eval x.length) :
+    (certOf q x y).length = q.eval x.length + 1 := by
+  rw [certOf_eq]
+  simp only [List.length_append, List.length_cons, List.length_replicate]
+  omega
+
+/-- Stripping a certificate at its last `1` recovers the witness:
+`splitAtLastTrue (certOf q x y) = some y` (`splitAtLastTrue_marker`). -/
+theorem splitAtLastTrue_certOf (q : PolyExpr) (x y : List Bool) :
+    splitAtLastTrue (certOf q x y) = some y := by
+  rw [certOf_eq]
+  exact PolyHierarchy.splitAtLastTrue_marker y _

+/-! ### The verifier -/
+
+/-- **The catalog's marker strip on a pair** (the function of
+`Turing.FinTM.computesFunInTime_stripLast`): `⟨a, v⟩ ↦ ⟨a, u⟩` when `v` strips to `u` at its last
+`1`, and `[]` when `v` has no `1` or the input is not a pair. -/
+def stripCert (z : List Bool) : List Bool :=
+  match pairDecode z with
+  | some (a, v) =>
+    match splitAtLastTrue v with
+    | some u => pairEncode a u
+    | none => []
+  | none => []
+
+/-- The defining equation of `stripCert` (the catalog function of
+`Turing.FinTM.computesFunInTime_stripLast`). -/
+theorem stripCert_eq (z : List Bool) :
+    stripCert z = match pairDecode z with
+      | some (a, v) =>
+        match splitAtLastTrue v with
+        | some u => pairEncode a u
+        | none => []
+      | none => [] := rfl
+
+/-- **The marker strip is polynomial-time** (`Turing.FinTM.computesFunInTime_stripLast`).
+
+**Proof sketch.** The catalog row computes, in time `c (n + 1)²`, a function stated as a nested
+anonymous `match`. It equals `stripCert` pointwise: split the outer `match` on `pairDecode z`,
+case on `splitAtLastTrue v`, and rewrite the inner `match` by the equation lemmas `eq_1`/`eq_2` of
+its compiled matcher `Turing.FinTM.stripLastTM_computes.match_1`, never by unfolding it; the
+`stripCert` side is `stripCert_eq`. Transport `computesFunInTime_stripLast` along this equality and
+conclude by `polyTimeComputable_iff`. -/
+theorem polyTimeComputable_stripCert : PolyTimeComputable stripCert := by
+  obtain ⟨M, c, hM⟩ := FinTM.computesFunInTime_stripLast
+  refine (polyTimeComputable_iff _).2 ⟨M, c, 2, ?_⟩
+  refine cast (congrArg (fun f => M.ComputesFunInTime f fun n => c * (n + 1) ^ 2)
+    (funext fun z => ?_)) hM
+  beta_reduce
+  split
+  · rename_i a v hz
+    cases hv : splitAtLastTrue v with
+    | none =>
+      refine (FinTM.stripLastTM_computes.match_1.eq_2 (fun _ => List Bool)
+        (fun u => pairEncode a u) (fun _ => [])).trans ?_
+      simp only [stripCert_eq, hz, hv]
+    | some u =>
+      refine (FinTM.stripLastTM_computes.match_1.eq_1 (fun _ => List Bool) u
+        (fun u => pairEncode a u) (fun _ => [])).trans ?_
+      simp only [stripCert_eq, hz, hv]
+  · rename_i hz
+    simp only [stripCert_eq, hz]
+
+/-- **The verifier test** for the decider `b` of a witness relation: strip the certificate of
+`⟨x, u⟩` and run `b` on the resulting pair `⟨x, y⟩`; reject when the strip fails,
+`verTest b z = (pairDecode (stripCert z)).isSome && b (stripCert z)`. Glue. -/
+def verTest (b : List Bool → Bool) (z : List Bool) : Bool :=
+  (pairDecode (stripCert z)).isSome && b (stripCert z)
+
+/-- The defining equation of `verTest`. -/
+theorem verTest_eq (b : List Bool → Bool) (z : List Bool) :
+    verTest b z = ((pairDecode (stripCert z)).isSome && b (stripCert z)) := rfl
+
+/-- **The verifier on a pair**: `verTest b ⟨x, u⟩` runs `b` on `⟨x, y⟩` when `u` strips to `y`
+at its last `1`, and rejects when `u` has no `1`. -/
+theorem verTest_pairEncode (b : List Bool → Bool) (x u : List Bool) :
+    verTest b (pairEncode x u) =
+      match splitAtLastTrue u with
+      | some y => b (pairEncode x y)
+      | none => false := by
+  rw [verTest_eq, stripCert_eq, pairDecode_pairEncode]
+  cases hs : splitAtLastTrue u with
+  | none => simp only [hs, pairDecode_nil, Option.isSome_none, Bool.false_and]
+  | some y => simp only [hs, pairDecode_pairEncode, Option.isSome_some, Bool.true_and]
+
+/-- The verifier accepts the pair of `x` and a certificate of `y` iff `b` accepts `⟨x, y⟩`:
+`verTest b ⟨x, certOf q x y⟩ = b ⟨x, y⟩`. -/
+theorem verTest_certOf (b : List Bool → Bool) (q : PolyExpr) (x y : List Bool) :
+    verTest b (pairEncode x (certOf q x y)) = b (pairEncode x y) := by
+  rw [verTest_pairEncode, splitAtLastTrue_certOf]
+
+/-- An accepted pair `⟨x, u⟩` has a stripped witness accepted by `b`:
+`verTest b ⟨x, u⟩ = true → ∃ y, b ⟨x, y⟩ = true`. -/
+theorem exists_of_verTest (b : List Bool → Bool) (x u : List Bool)
+    (h : verTest b (pairEncode x u) = true) : ∃ y, b (pairEncode x y) = true := by
+  rw [verTest_pairEncode] at h
+  cases hs : splitAtLastTrue u with
+  | none => rw [hs] at h; exact absurd h Bool.false_ne_true
+  | some y => rw [hs] at h; exact ⟨y, h⟩
+
+/-- **The verifier test is polynomial-time** when the decider `b` is:
+`PolyTimeComputable (fun z => [b z]) → PolyTimeComputable (fun z => [verTest b z])`
+(`polyTimeComputable_and` of the pair-validity row and `b`, both after the marker strip). -/
+theorem polyTimeComputable_verTest {b : List Bool → Bool}
+    (hb : PolyTimeComputable fun z => [b z]) :
+    PolyTimeComputable fun z => [verTest b z] := by
+  have hvalid : PolyTimeComputable fun z => [(pairDecode z).isSome] :=
+    polyTimeComputable_of_linear FinTM.computesFunInTime_pairValid
+  exact polyTimeComputable_and
+    (BinArith.polyTimeComputable_test_app hvalid polyTimeComputable_stripCert)
+    (BinArith.polyTimeComputable_test_app hb polyTimeComputable_stripCert)
+
+/-- **The verifier language is in `P`** [AB09, Lemma 6.10, the verifier of `L ∈ NP`]:
+`{z | verTest b z = true} ∈ P` for a polynomial-time decider `b`. -/
+theorem verLang_mem_P {b : List Bool → Bool} (hb : PolyTimeComputable fun z => [b z]) :
+    {z | verTest b z = true} ∈ P :=
+  mem_P_of_test (polyTimeComputable_verTest hb)

+/-! ### The tableau instance map -/
+
+/-- **The tableau's time bound** for a test machine running within `C (N + 1)^c` steps and
+certificates of length `m(n)`: `verTime C c m n = C (2n + 3 + m(n))^c + 1`, which bounds the
+machine's time on `⟨x, u⟩` (length `2|x| + 2 + m(|x|)`) and is at least `1`. Glue. -/
+def verTime (C c : ℕ) (m : PolyExpr) : PolyExpr :=
+  .add (.mul (.const C) (.pow (.add (.add (.mul (.const 2) .var) (.const 3)) m) c)) (.const 1)
+
+/-- The defining equation of `verTime`: `verTime C c m = C · (2 var + 3 + m)^c + 1` as a
+`PolyExpr`. -/
+theorem verTime_eq (C c : ℕ) (m : PolyExpr) :
+    verTime C c m =
+      .add (.mul (.const C) (.pow (.add (.add (.mul (.const 2) .var) (.const 3)) m) c))
+        (.const 1) := rfl
+
+/-- The value of `verTime`: `verTime C c m` at `n` is `C (2n + 3 + m(n))^c + 1`. -/
+theorem eval_verTime (C c : ℕ) (m : PolyExpr) (n : ℕ) :
+    (verTime C c m).eval n = C * (2 * n + 3 + m.eval n) ^ c + 1 := by
+  simp [verTime_eq]
+
+/-- **The tableau circuit of `x`** for a machine `M`, certificate length `m` and time `t`: the
+uniform configuration tableau (`Complexity.cfgTab`, [AB09, Remark 6.7]) of `M` on the virtual
+input `⟨x, []⟩ ++ u` for `t(|x|)` steps, with `m(|x|)` inputs `u`. -/
+noncomputable def tabCkt (M : FinTM Bool) (m t : PolyExpr) (x : List Bool) :
+    DAGCircuit (m.eval x.length) :=
+  cfgTab M (m.eval x.length) (pairEncode x []) (t.eval x.length)
+
+/-- The defining equation of `tabCkt`: `tabCkt M m t x = cfgTab M m(|x|) ⟨x, []⟩ t(|x|)`. -/
+theorem tabCkt_eq (M : FinTM Bool) (m t : PolyExpr) (x : List Bool) :
+    tabCkt M m t x = cfgTab M (m.eval x.length) (pairEncode x []) (t.eval x.length) := rfl
+
+/-- The tableau circuit has fan-in two (`Complexity.cfgTab_isFaninTwo`). -/
+theorem tabCkt_isFaninTwo (M : FinTM Bool) (m t : PolyExpr) (x : List Bool) :
+    (tabCkt M m t x).IsFaninTwo := by
+  rw [tabCkt_eq]
+  exact cfgTab_isFaninTwo M _ _ _
+
+/-- **The instance map** [AB09, Lemma 6.10]: the emitter of [AB09, Remark 6.7] on
+`1^{m(|x|)} 0 1^{t(|x|)} 0 ⟨x, []⟩`, as in the proof of `BoolCircuit.dagCktSatLang_NPHard`. -/
+noncomputable def tabInst (M : FinTM Bool) (m t : PolyExpr) (x : List Bool) : List Bool :=
+  UTab.tabEmit M (List.replicate (m.eval x.length) true ++ false ::
+    (List.replicate (t.eval x.length) true ++ false :: pairEncode x []))
+
+/-- The defining equation of `tabInst`. -/
+theorem tabInst_def (M : FinTM Bool) (m t : PolyExpr) (x : List Bool) :
+    tabInst M m t x = UTab.tabEmit M (List.replicate (m.eval x.length) true ++ false ::
+      (List.replicate (t.eval x.length) true ++ false :: pairEncode x [])) := rfl
+
+/-- **The instance map prints the tableau circuit**: for `t(|x|) ≥ 1`,
+`tabInst M m t x = (tabCkt M m t x).encode` (`Complexity.UTab.tabEmit_eq`). -/
+theorem tabInst_eq (M : FinTM Bool) (m t : PolyExpr) (x : List Bool)
+    (ht : 1 ≤ t.eval x.length) : tabInst M m t x = (tabCkt M m t x).encode := by
+  rw [tabInst_def, tabCkt_eq]
+  exact UTab.tabEmit_eq M _ _ _ ht
+
+/-- **The instance map is polynomial-time** (`Complexity.UTab.polyTimeComputable_tabEmit` after
+the front end `Complexity.polyTimeComputable_tabInput`). -/
+theorem polyTimeComputable_tabInst (M : FinTM Bool) (m t : PolyExpr) :
+    PolyTimeComputable (tabInst M m t) := by
+  have hpair : PolyTimeComputable fun x : List Bool => pairEncode x [] :=
+    polyTimeComputable_id.pairEncode (polyTimeComputable_const [])
+  exact (UTab.polyTimeComputable_tabEmit M).comp (polyTimeComputable_tabInput
+    (polyTimeComputable_unaryPoly m) (polyTimeComputable_unaryPoly t) hpair)
+
+/-- **The tableau circuit computes the test on `⟨x, u⟩`**: if `M` computes `z ↦ [β z]` within
+`C (N + 1)^c` steps, then on every input `v` the circuit `tabCkt M m (verTime C c m) x` outputs
+`β ⟨x, v⟩` (`Complexity.cfgTab_eval`, `Complexity.decide_emits_eq_of_computesInTime`).
+
+**Proof sketch.** The tableau outputs `1` iff some step before `t(|x|)` of `M` on
+`⟨x, []⟩ ++ v = ⟨x, v⟩` emits `1`; the input has length `2|x| + 2 + m(|x|)`, so `M` halts
+with output `[β ⟨x, v⟩]` within `C(2|x| + 3 + m(|x|))^c ≤ t(|x|)` steps, and the emissions are
+exactly that output. -/
+theorem tabCkt_eval {M : FinTM Bool} {β : List Bool → Bool} {C c : ℕ}
+    (hM : M.ComputesFunInTime (fun z => [β z]) fun n => C * (n + 1) ^ c) (m : PolyExpr)
+    (x : List Bool) (v : Fin (m.eval x.length) → Bool) :
+    (tabCkt M m (verTime C c m) x).eval v = β (pairEncode x (List.ofFn v)) := by
+  rw [tabCkt_eq, cfgTab_eval, BinArith.pairEncode_nil_append]
+  apply decide_emits_eq_of_computesInTime
+  refine ((FinTM.computesFunInTime_iff M _ _).1 hM _).mono ?_
+  rw [eval_verTime, length_pairEncode, List.length_ofFn]
+  have h : 2 * x.length + 2 + m.eval x.length + 1 = 2 * x.length + 3 + m.eval x.length := by
+    omega
+  rw [h]
+  exact Nat.le_succ _
+
+/-! ### Step 1 for a witness relation -/
+
+/-- **Step 1 of the Levin reduction to `3COL`** [AB09, Lemma 6.10] with the witness map of
+[PS10, §4.7, Theorem 124.1, step 2]: for a witness relation `R` of `L` there are a length bound
+`q`, a machine `M` and constants `C, c` such that, for `m = q + 1` and `t = verTime C c m`,
+the polynomial-time instance map `f₁ = tabInst M m t` prints the fan-in-two tableau circuit
+`tabCkt M m t x`,
+* `x ∈ L ↔ ∃ v, (tabCkt M m t x).eval v = true`, and
+* `R x y → R_CKT (f₁ x) (certOf q x y)` (the certificate is the witness of `R_CKT`).
+
+**Proof sketch.** Take `q` from `WitnessRelation.bounded` and the decider `b` from
+`WitnessRelation.decidable`; `M` computes the verifier test `verTest b`
+(`polyTimeComputable_verTest`) within `C (N + 1)^c`. By `tabCkt_eval` the circuit outputs
+`verTest b ⟨x, v⟩` on `v`. If `R x y`, then `|y| ≤ q(|x|)`, the certificate has the circuit's
+input length (`length_certOf`) and `verTest b ⟨x, certOf q x y⟩ = b ⟨x, y⟩ = true`
+(`verTest_certOf`), which is the witness of `R_CKT` (`ofFn_getD_eq_take` reads the input back).
+Conversely an accepting input strips to some `y` with `b ⟨x, y⟩ = true`, i.e. `R x y`
+(`exists_of_verTest`), so `x ∈ L`. -/
+theorem exists_tabReduction {L : Language Bool} {R : List Bool → List Bool → Prop}
+    (hR : WitnessRelation L R) :
+    ∃ (q : PolyExpr) (M : FinTM Bool) (C c : ℕ),
+      PolyTimeComputable (tabInst M (q.add (.const 1)) (verTime C c (q.add (.const 1)))) ∧
+      (∀ x, tabInst M (q.add (.const 1)) (verTime C c (q.add (.const 1))) x =
+        (tabCkt M (q.add (.const 1)) (verTime C c (q.add (.const 1))) x).encode) ∧
+      (∀ x, (tabCkt M (q.add (.const 1)) (verTime C c (q.add (.const 1))) x).IsFaninTwo) ∧
+      (∀ x, x ∈ L ↔
+        ∃ v, (tabCkt M (q.add (.const 1)) (verTime C c (q.add (.const 1))) x).eval v = true) ∧
+      ∀ x y, R x y →
+        R_CKT (tabInst M (q.add (.const 1)) (verTime C c (q.add (.const 1))) x)
+          (certOf q x y) := by
+  obtain ⟨q, hq⟩ := hR.bounded
+  obtain ⟨b, hb, hbR⟩ := hR.decidable
+  obtain ⟨M, C, c, hM⟩ := (polyTimeComputable_iff _).1 (polyTimeComputable_verTest hb)
+  set m : PolyExpr := q.add (.const 1) with hm
+  set t : PolyExpr := verTime C c m with ht
+  have ht1 : ∀ x : List Bool, 1 ≤ t.eval x.length := fun x => by
+    rw [ht, eval_verTime]; exact Nat.le_add_left _ _
+  have heval := tabCkt_eval hM m
+  -- the certificate of a witness is accepted
+  have hcert : ∀ x y, R x y →
+      (tabCkt M m t x).eval (fun i => (certOf q x y).getD i false) = true := by
+    intro x y hxy
+    have hlen : (certOf q x y).length = m.eval x.length := by
+      rw [length_certOf q x y (hq x y hxy), hm, PolyExpr.eval_add, PolyExpr.eval_const]
+    rw [heval, CircuitEval.ofFn_getD_eq_take _ _ hlen.ge, List.take_of_length_le hlen.le,
+      verTest_certOf]
+    exact (hbR x y).2 hxy
+  refine ⟨q, M, C, c, polyTimeComputable_tabInst M m t,
+    fun x => tabInst_eq M m t x (ht1 x), fun x => tabCkt_isFaninTwo M m t x,
+    fun x => ⟨?_, ?_⟩, fun x y hxy => ?_⟩
+  · intro hx
+    obtain ⟨y, hxy⟩ := (hR.mem_iff x).1 hx
+    exact ⟨_, hcert x y hxy⟩
+  · rintro ⟨v, hv⟩
+    rw [heval] at hv
+    obtain ⟨y, hy⟩ := exists_of_verTest b x _ hv
+    exact (hR.mem_iff x).2 ⟨y, (hbR x y).1 hy⟩
+  · exact (R_CKT_iff _ _).2 ⟨_, tabCkt M m t x, tabCkt_isFaninTwo M m t x,
+      (tabInst_eq M m t x (ht1 x)).symm, hcert x y hxy⟩
+
+end TCSlib.Cryptography.LevinHardness
