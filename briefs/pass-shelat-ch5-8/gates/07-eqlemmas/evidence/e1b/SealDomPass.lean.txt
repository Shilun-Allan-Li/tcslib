/-! Seal probe, published domain equivalences: elements `x : C.D i` of each concrete collection
read through `C_domEquiv`, with the collection sealed. -/

set_option autoImplicit false
open Complexity Turing TCSlib.Cryptography TCSlib.Cryptography.Candidates TCSlib.Cryptography.NumberTheory

attribute [local irreducible] expCollection dlCollection fullDLCollection factoringCollection rsaTrapdoor rsaCollection rabinCollection rabinBlumTrapdoor

-- expCollection: decoding inverts encoding, through the equivalence
example {V : List Bool → Prop} (H : ArithmeticFP) (G : ExpGroupGen V) (hV : ∀ i, V i → ExpValid i)
    (i : List Bool) (x : (expCollection H G hV).D i) :
    (expCollection H G hV).decD i ((expCollection H G hV).encD i x) = some x := by
  rw [expCollection_encD, expCollection_decD, expDec_expEnc, Option.map_some,
    Equiv.symm_apply_apply]
-- expCollection: `f` is injective at a valid index, through the equivalence
example {V : List Bool → Prop} (H : ArithmeticFP) (G : ExpGroupGen V) (hV : ∀ i, V i → ExpValid i)
    {i : List Bool} (hi : ExpValid i) (x y : (expCollection H G hV).D i)
    (h : (expCollection H G hV).f i x = (expCollection H G hV).f i y) : x = y := by
  rw [expCollection_f, expCollection_f] at h
  exact (expCollection_domEquiv H G hV i).injective (expF_injective hi h)
-- expCollection: a uniform statement over `D i` transports to `Z_m`
example {V : List Bool → Prop} (H : ArithmeticFP) (G : ExpGroupGen V) (hV : ∀ i, V i → ExpValid i)
    (i : List Bool) (P : ZMod (expOrd i) → Prop) :
    uprob (fun x : (expCollection H G hV).D i => P (expCollection_domEquiv H G hV i x)) =
      uprob P :=
  uprob_comp_equiv _ P

-- the published uniformity transports
example {V : List Bool → Prop} (H : ArithmeticFP) (G : ExpGroupGen V) (hV : ∀ i, V i → ExpValid i)
    (i : List Bool) (P : ZMod (expOrd i) → Prop) :
    uprob (fun x : (expCollection H G hV).D i => P (expCollection_domEquiv H G hV i x)) =
      uprob P := expCollection_domEquiv_uprob H G hV i P
example (H : ArithmeticFP) (n : ℕ) (P : DistinctPrimes (max (List.replicate n true).length 2) → Prop) :
    uprob (fun x : (factoringCollection H).D (List.replicate n true) =>
      P (factoringCollection_domEquiv H _ x)) = uprob P :=
  factoringCollection_domEquiv_uprob H _ P

      uprob P := rabinBlumTrapdoor_domEquiv_uprob H i P
