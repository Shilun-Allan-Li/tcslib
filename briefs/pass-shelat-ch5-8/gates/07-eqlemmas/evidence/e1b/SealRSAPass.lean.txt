import TCSlib.Cryptography.Candidates.Factoring
import TCSlib.Cryptography.NumberTheory.Units
import Mathlib.Data.Nat.Size

/-! Seal probe, RSA route (the imports of `Candidates/RSA`, consumer of unit rsa): `Factoring`
(P10, S1, S2) and `Arithmetic` (P9, `sizeStr_eq`) readings. -/

set_option autoImplicit false
open Complexity Turing TCSlib.Cryptography TCSlib.Cryptography.Candidates TCSlib.Cryptography.NumberTheory

attribute [local irreducible] factorProb modulusBits samplePrimeL xgcdStr divModStr sizeStr natToList

example (A : NUPPT) (n : ℕ) : factorProb A n ≤ 1 := by
  rw [factorProb_eq]; exact uprob_le_one _
example (n : ℕ) (z : DistinctPrimes n) :
    bitsToNat (modulusBits n z) = z.p * z.q := by
  rw [modulusBits_eq, bitsToNat_natToBits z.mul_lt]
example (a b : List Bool) :
    (pairSndD (xgcdStr (pairEncode a b))).length = b.length := by
  rw [xgcdStr_pairEncode, pairSndD_pairEncode, length_natToList]
example (a b : List Bool) :
    (pairFstD (divModStr (pairEncode a b))).length = a.length := by
  rw [divModStr_pairEncode, pairFstD_pairEncode, length_natToList]
example (w : List Bool) : sizeStr w = List.replicate (Nat.size (listToNat w)) true := sizeStr_eq w
example (H : ArithmeticFP) :
    PolyTimeComputable fun z => samplePrimeL (pairFstD z).length (pairSndD z) :=
  polyTimeComputable_samplePrimeL H.prime
example (n : ℕ) (hn : 2 ≤ n) :
    StatClose (fun s : Fin (2 * (2 * n ^ 2 * n)) → Bool =>
        (listToNat (samplePrimeL n ((List.ofFn s).take (2 * n ^ 2 * n))),
          listToNat (samplePrimeL n ((List.ofFn s).drop (2 * n ^ 2 * n)))))
      (fun z : DistinctPrimes n => (z.p, z.q)) (2 * (1 / 2 : ℝ) ^ n + 2 * n / 2 ^ n) :=
  statClose_samplePrimeL_pair n hn
