/-! Seal control, DL route: the same readings by definitional unfolding (`rfl`, `show`,

namespace Unsealed

example (i : List Bool) : dlP i = listToNat (pairFstD i) := rfl
example (i : List Bool) : dlM i = listToNat (pairFstD (pairSndD i)) := rfl
example (i : List Bool) : dlG i = listToNat (pairSndD (pairSndD i)) := rfl

example {n : ℕ} {i : List Bool} (h : WellFormedIndex n i) :
    ∃ a b c : List Bool, i = pairEncode a (pairEncode b c) := by
  obtain ⟨a, b, c, hi, -⟩ := h
  exact ⟨a, b, c, hi⟩
example {i : List Bool} (h : SafePrimeValid i) : dlP i = 2 * dlM i + 1 := h.2.2

    expF i x = natToList (dlW i) (dlG i ^ x.val % dlP i) := rfl

    (n : ℕ) (hn : G.n₀ ≤ n) :
    uprob (fun r : G.gen.Coins n => dlP (G.gen.runAt n [] r) < 2 ^ (n - 1)) ≤ δ n := h n hn
example {V : List Bool → Prop} (G : ExpGroupGen V) (h : G.DLHard) (A : NUPPT) :
    Negligible fun n => G.dlProb A n := h A
example {V : List Bool → Prop} (G : ExpGroupGen V) (A : NUPPT) (n : ℕ) : G.dlProb A n ≤ 1 := by
  haveI : Nonempty (G.gen.Coins n) := ⟨fun _ => false⟩
  show uexp _ ≤ 1
  exact (OneWay.uexp_mono fun r => uprob_le_one _).trans (OneWay.uexp_const 1).le
example {V : List Bool → Prop} (H : ArithmeticFP) (G : ExpGroupGen V) (hV : ∀ i, V i → ExpValid i)
    (i w : List Bool) : (expCollection H G hV).decD i w = expDec i w := rfl
example {V : List Bool → Prop} (H : ArithmeticFP) (G : ExpGroupGen V) (hV : ∀ i, V i → ExpValid i) :
    (expCollection H G hV).gen.coins = (expGen G).coins := rfl
example (k N : ℕ) : natToList k N = List.ofFn (natToBits k N) := rfl
example (w : List Bool) : sizeStr w = List.replicate (Nat.size (listToNat w)) true := rfl
example (a b : List Bool) : divModStr (pairEncode a b) = pairEncode
    (natToList (pairFstD (pairEncode a b)).length (listToNat (pairFstD (pairEncode a b)) /
      listToNat (pairSndD (pairEncode a b))))
    (natToList (pairSndD (pairEncode a b)).length (listToNat (pairFstD (pairEncode a b)) %
      listToNat (pairSndD (pairEncode a b)))) := rfl
example (a : List Bool) : xgcdStr a = pairEncode
    (natToList (pairSndD a).length (Nat.gcd (listToNat (pairFstD a)) (listToNat (pairSndD a))))
    (natToList (pairSndD a).length
      (ZMod.val ((listToNat (pairFstD a) : ZMod (listToNat (pairSndD a)))⁻¹))) := rfl

end Unsealed
