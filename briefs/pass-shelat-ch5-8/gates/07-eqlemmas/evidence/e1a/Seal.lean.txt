section Collections

example (F : Collection) (δ : ℕ → ℝ) (h : F.SampleClose δ) (n : ℕ) (r : F.gen.Coins n) :
    StatClose (fun s : F.sample.Coins n => F.decD (F.index n r) (F.sample.runAt n (F.index n r) s))
      (fun x : F.D (F.index n r) => some x) (δ n) :=
  (F.sampleClose_iff δ).1 h n r

    F.index n r = pairFstD (F.gen.runAt n [] r) := by rw [Collection.index_eq]

    F.invProbAt A n = uexp fun r : F.gen.Coins n =>
      uprob fun z : F.D (F.index n r) × A.Coins n =>
        ∃ x', F.decD (F.index n r)
            (A.runAt n (pairEncode (F.index n r) (F.f (F.index n r) z.1)) z.2) = some x' ∧
          F.f (F.index n r) x' = F.f (F.index n r) z.1 := by
  rw [Collection.invProbAt_eq]

        (samplePerm3 (encGraph G).length (r.take (3 * (encGraph G).length)) (c i)))

    (colProver Com ℓ).coinLen x.length = 3 * x.length + x.length * ℓ.eval x.length := by
