def belowAcc (x : List Bool) : Bool :=
  !decide (listToNat (pairFstD x) ≤ listToNat (pairSndD x))

lemma belowAcc_poly (H : ArithmeticFP) : PolyTimeComputable fun x => [belowAcc x] :=
  polyTimeComputable_not (H.le_poly polyTimeComputable_pairFstD polyTimeComputable_pairSndD)

lemma belowAcc_spec (w b : List Bool) :
    belowAcc (pairEncode w b) = decide (listToNat b < listToNat w) := by
  simp only [belowAcc, pairFstD_pairEncode, pairSndD_pairEncode, ← decide_not, Nat.not_le]

lemma belowRoute (H : ArithmeticFP) {par kw bw tw c : List Bool → List Bool}
    (hpar : PolyTimeComputable par) (hkw : PolyTimeComputable kw)
    (hbw : PolyTimeComputable bw) (htw : PolyTimeComputable tw) (hc : PolyTimeComputable c) :
    ∃ g : List Bool → List Bool, PolyTimeComputable g ∧ ∀ z,
      (g z).tail.length ≤ (kw z).length ∧
      sampleBelowL (listToNat (par z)) (kw z).length (bw z).length (tw z).length (c z) =
        if g z = [] then none else some (listToNat (g z).tail) := by
  obtain ⟨g, hg, hs⟩ := polyTimeComputable_rejectBlocksL (fun a v => v < a)
    (belowAcc_poly H) belowAcc_spec hpar hkw hbw htw hc
  refine ⟨g, hg, fun z => ⟨(hs z).1, ?_⟩⟩
  rw [sampleBelowL_eq_rejectBlocksL]
  exact (hs z).2

lemma belowZeroDefault (H : ArithmeticFP) {par kw bw tw c : List Bool → List Bool}
    (hpar : PolyTimeComputable par) (hkw : PolyTimeComputable kw)
    (hbw : PolyTimeComputable bw) (htw : PolyTimeComputable tw) (hc : PolyTimeComputable c) :
    ∃ payload : List Bool → List Bool, PolyTimeComputable payload ∧ ∀ z,
      (payload z).length ≤ (kw z).length ∧
      (sampleBelowL (listToNat (par z)) (kw z).length (bw z).length (tw z).length (c z)).getD 0 =
        listToNat (payload z) := by
  obtain ⟨g, hg, hs⟩ := belowRoute H hpar hkw hbw htw hc
  refine ⟨(fun z => (g z).tail), ((polyTimeComputable_tail.comp' hg).congr fun z => (List.drop_one (l := g z))), fun z => ⟨(hs z).1, ?_⟩⟩
  change _ = listToNat (g z).tail
  rw [(hs z).2]
  split_ifs with he
  · rw [he, List.tail_nil, listToNat_nil]
    rfl
  · rfl

lemma unitRoute {acc : List Bool → Bool}
    (hacc : PolyTimeComputable fun x => [acc x])
    (hspec : ∀ w b : List Bool, acc (pairEncode w b) =
      decide (listToNat b < listToNat w ∧ Nat.Coprime (listToNat b) (listToNat w)))
    {par kw bw tw c : List Bool → List Bool} (hpar : PolyTimeComputable par)
    (hkw : PolyTimeComputable kw) (hbw : PolyTimeComputable bw)
    (htw : PolyTimeComputable tw) (hc : PolyTimeComputable c) :
    ∃ g : List Bool → List Bool, PolyTimeComputable g ∧ ∀ z,
      (g z).tail.length ≤ (kw z).length ∧
      sampleUnitL (listToNat (par z)) (kw z).length (bw z).length (tw z).length (c z) =
        if g z = [] then none else some (listToNat (g z).tail) := by
  obtain ⟨g, hg, hs⟩ := polyTimeComputable_rejectBlocksL
    (fun a v => v < a ∧ Nat.Coprime v a) hacc hspec hpar hkw hbw htw hc
  refine ⟨g, hg, fun z => ⟨(hs z).1, ?_⟩⟩
  rw [sampleUnitL_eq]
  exact (hs z).2

lemma noTrials (E : ℕ → Prop) [DecidablePred E] (k B : ℕ) (r : List Bool) :
    rejectBlocksL E k B 0 r = none := by
  rw [rejectBlocksL_eq_find?]
  rfl
