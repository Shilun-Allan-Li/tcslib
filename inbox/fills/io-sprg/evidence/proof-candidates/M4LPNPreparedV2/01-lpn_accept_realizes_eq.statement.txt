theorem lpn_accept_realizes_eq (P : Params) (π : PrimeSchedule) (A B : NUPPT)
    (ω : ∀ L, P.LPNAdvice π L)
    (hB : B.RealizesOn (P.lpnEnc π) (A.coins.subst .var)
      (fun L x r => P.lpnRed π A L (ω L) x r))
    (L : ℕ) {σ : Type} [Fintype σ] (X : σ → P.LPNChal π L) :
    acceptProb X (fun x (r : B.Coins L) => B.distinguisherL L (P.lpnEnc π L x) r) =
      acceptProb X (fun x (r : A.Coins (P.kappaOf L)) =>
        decide (A.runAtL (P.kappaOf L) (P.lpnPre π L (ω L) x)
          (List.ofFn r) = [true]))
