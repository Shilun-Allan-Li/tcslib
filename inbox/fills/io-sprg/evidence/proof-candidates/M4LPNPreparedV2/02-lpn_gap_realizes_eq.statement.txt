theorem lpn_gap_realizes_eq (P : Params) (π : PrimeSchedule) (A B : NUPPT)
    (ω : ∀ L, P.LPNAdvice π L)
    (hB : B.RealizesOn (P.lpnEnc π) (A.coins.subst .var)
      (fun L x r => P.lpnRed π A L (ω L) x r)) (L : ℕ) :
    LWE.decisionAdvantage ((P.lpnFamily_raw π).toLWE.parameters L)
      ((P.lpnFamily_raw π).toLWE.noise L)
      (fun x (r : A.Coins (P.kappaOf L)) =>
        decide (A.runAtL (P.kappaOf L) (P.lpnPre π L (ω L) x)
          (List.ofFn r) = [true])) = (P.lpnFamily_raw π).adv B L
