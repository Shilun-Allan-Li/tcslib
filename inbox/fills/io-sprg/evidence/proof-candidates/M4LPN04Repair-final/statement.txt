    (h : L = P.ℓ (P.kappaOf L)) :
    let κ := P.kappaOf L
    let pp := π κ
    let Q := (P.lpnFamily_raw π).toLWE.parameters L
    let χ := (P.lpnFamily_raw π).toLWE.noise L
    ∃ e : (Matrix (Fin Q.samples) (Fin Q.dimension) (ZMod Q.modulus) ×
        ((Fin Q.dimension → ZMod Q.modulus) × (Fin Q.samples → χ.Coins))) ≃
        (Matrix (Fin (P.n κ)) (Fin (P.ℓ κ)) (ZMod pp.p) ×
          ((Fin (P.ℓ κ) → ZMod pp.p) ×
            (Fin (P.n κ) → Fin (P.W κ) × ZMod pp.p))),
