/-- A canonical real challenge replaces precisely the selected repetition in the chain word.
Preprocessing glue for [JLS26, Lemma 4.2, p. 19].

**Proof sketch.** Read preprocessing and its schedule guard, then substitute the supplied
matrix and noisy-value coordinate equations. Extensionality reduces the public vectors to
the selected and unselected coordinates; the chain and vector readers and finite index order
identify the appropriate updated repetition in each case. -/
theorem lpn_pre_real_update (P : Params) (π : PrimeSchedule) (L : ℕ)
    (h : L = P.ℓ (P.kappaOf L)) (w : P.LPNAdvice π L)
    (k : Fin (P.kappaOf L)) (hk : k.val = w.hop.val)
    (M : Matrix (Fin (P.n (P.kappaOf L))) (Fin (P.ℓ (P.kappaOf L)))
      (ZMod (π (P.kappaOf L)).p))
    (rep : (Fin (P.ℓ (P.kappaOf L)) → ZMod (π (P.kappaOf L)).p) ×
      (Fin (P.n (P.kappaOf L)) → Fin (P.W (P.kappaOf L)) ×
        ZMod (π (P.kappaOf L)).p))
    (x : P.LPNChal π L) (hM : P.lpnMatrixAt π L h x = M)
    (hb : P.lpnValuesAt π L h x = fun i =>
      (∑ a, M i a * rep.1 a) + P.noiseValue (P.kappaOf L) (π (P.kappaOf L)) (rep.2 i)) :
    P.lpnPre π L w x =
      P.challengeStr (P.kappaOf L) (π (P.kappaOf L)) (w.I, w.buckets, M)
        (P.publicSeed (P.kappaOf L) (π (P.kappaOf L))
          (P.chainPublic (P.kappaOf L) (π (P.kappaOf L)) (w.I, w.buckets, M)
            (w.draw.1, Function.update w.draw.2 k rep) w.uniform w.hop.val) false)
        w.tail := by
  classical
  rw [P.lpnPre_eq, dif_pos h, hM, hb]
  dsimp only
  congr 1
  apply congrArg (fun bs => P.publicSeed (P.kappaOf L) (π (P.kappaOf L)) bs false)
  funext q i
  by_cases hq : q = k
  · subst q
    simp only [P.chainPublic_eq, P.lpnVector_eq, Function.update_same, hk,
      lt_self_iff_false, if_false, ite_true]
  · have hval : q.val ≠ w.hop.val := fun heq => hq (Fin.ext (heq.trans hk.symm))
    simp only [P.chainPublic_eq, P.lpnVector_eq, Function.update_of_ne hq, if_neg hval]

end Internal

end TCSlib.Cryptography.Obfuscation.StructuredSeed
