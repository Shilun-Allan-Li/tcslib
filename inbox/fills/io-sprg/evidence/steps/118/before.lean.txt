/-!
# Failure events and locality bounds for Claim 4.2

## Main definitions

* `Params.Cond1`, `Params.Cond2`, `Params.Cond3` are the three failure events.
* `Params.jointLaw` and `Params.errProb` retain the actual index and full core law.

## Main results

* `Params.errProb_nonneg`, `Params.badSet_expect_le`, `Params.cond1_prob_le`.

This module relocation preserves the existing declarations and their documentation.

## References

* [JLS26] A. Jain, H. Lin, A. Sahai, *Indistinguishability Obfuscation from
  Well-Founded Assumptions*, J. ACM 73(1), Article 1, February 2026,
  proof of Claim 4.2, pp. 19-21.
* [PS10] R. Pass, A. Shelat, *A Course in Cryptography*, 2010, Definition 26.8.
-/

namespace TCSlib.Cryptography.Obfuscation.StructuredSeed

/-- The first failure condition is too many bad outputs in repetition k
[JLS26, proof of Claim 4.2, pp. 19-20]. Dv3 uses the rounded threshold Γ. -/
noncomputable def Cond1 (P : Params) (κ : ℕ) (pp : PrimeParam κ)
    (I : P.Idx' κ pp) (d : P.Draw κ pp) (k : Fin κ) : Prop :=
  P.Γ κ < (P.badSet κ pp I (fun i => P.noiseValue κ pp ((d.2 k).2 i))).card

/-- The first condition compares the bad-output count with Γ; glue. -/
theorem cond1_iff (P : Params) (κ : ℕ) (pp : PrimeParam κ)
    (I : P.Idx' κ pp) (d : P.Draw κ pp) (k : Fin κ) :
    P.Cond1 κ pp I d k ↔
      P.Γ κ < (P.badSet κ pp I (fun i => P.noiseValue κ pp ((d.2 k).2 i))).card :=
  Iff.rfl

/-- The second failure condition is more than t bad outputs in some bucket
[JLS26, proof of Claim 4.2, pp. 19-20]. Dv3 uses an integer threshold. -/
noncomputable def Cond2 (P : Params) (κ : ℕ) (pp : PrimeParam κ)
    (I : P.Idx' κ pp) (d : P.Draw κ pp) (k : Fin κ) : Prop :=
  ∃ b : Fin (P.B κ), P.t κ <
    (Buckets.bucket I.2.1 b ∩
      P.badSet κ pp I (fun i => P.noiseValue κ pp ((d.2 k).2 i))).card

/-- The second condition is the finite bucket-overload event for bad outputs; glue. -/
theorem cond2_iff (P : Params) (κ : ℕ) (pp : PrimeParam κ)
    (I : P.Idx' κ pp) (d : P.Draw κ pp) (k : Fin κ) :
    P.Cond2 κ pp I d k ↔ ∃ b : Fin (P.B κ), P.t κ <
      (Buckets.bucket I.2.1 b ∩
        P.badSet κ pp I (fun i => P.noiseValue κ pp ((d.2 k).2 i))).card := Iff.rfl

/-- The third failure condition is a bucket beyond capacity
[JLS26, proof of Claim 4.2, pp. 19-20]. This event is shared by all repetitions. -/
noncomputable def Cond3 (P : Params) (κ : ℕ) (pp : PrimeParam κ)
    (I : P.Idx' κ pp) : Prop := Buckets.overfull I.2.1 (P.cap κ)

/-- The third condition is the overfull-placement predicate; glue. -/
theorem cond3_iff (P : Params) (κ : ℕ) (pp : PrimeParam κ) (I : P.Idx' κ pp) :
    P.Cond3 κ pp I ↔ Buckets.overfull I.2.1 (P.cap κ) := Iff.rfl

/-- The proof-facing joint law retains index and full seed-core output
[JLS26, Claim 4.2, p. 19]. Dv17 uses exact finite draws, without a coin sampler. -/
noncomputable def jointLaw (P : Params) (κ : ℕ) (pp : PrimeParam κ) :
    ProbComp (P.Idx' κ pp × P.SdCoreOut κ pp) :=
  (P.idSamp' κ pp).bind fun I => (fun x => (I, x)) <$> P.sdSampFull κ pp I

/-- The joint law samples index and the corresponding core output; glue. -/
theorem jointLaw_eq (P : Params) (κ : ℕ) (pp : PrimeParam κ) :
    P.jointLaw κ pp = (P.idSamp' κ pp).bind fun I =>
      (fun x => (I, x)) <$> P.sdSampFull κ pp I := rfl

/-- Error probability is the exact chance that SdSamp' finds no successful repetition
[JLS26, Claim 4.2, p. 19]. Dv7 leaves its value explicit at every κ and pp. -/
noncomputable def errProb (P : Params) (κ : ℕ) (pp : PrimeParam κ) : ℝ :=
  ProbComp.prob ((P.idSamp' κ pp).bind fun I => P.sdSampFull κ pp I)
    (fun x => x.err = true)

/-- Error probability is the stated exact-law event; glue. -/
theorem errProb_eq (P : Params) (κ : ℕ) (pp : PrimeParam κ) :
    P.errProb κ pp = ProbComp.prob
      ((P.idSamp' κ pp).bind fun I => P.sdSampFull κ pp I) (fun x => x.err = true) := rfl

/-- Error probabilities are nonnegative; glue.

**Proof sketch.** Use the existing nonnegativity of ProbComp.prob. -/
theorem errProb_nonneg (P : Params) (κ : ℕ) (pp : PrimeParam κ) :
    0 ≤ P.errProb κ pp := by
  rw [P.errProb_eq]
  exact Games.ProbComp.prob_nonneg _ _

/-- Locality bounds the expected number of bad output coordinates by cLoc*m/W
[JLS26, proof of Claim 4.2, p. 20]. Dv3 uses exact rounded noise, and the bound
is uniform in every public prime and index.

**Proof sketch.** Union-bound the error events in each output's at most cLoc variables,
then sum their indicator expectations using the existing probability readers. -/
theorem badSet_expect_le (P : Params) {cLoc : ℕ} (hLoc : P.R.HasLocality cLoc)
    (κ : ℕ) (pp : PrimeParam κ) (I : P.Idx' κ pp) :
    ProbComp.expect (ProbComp.sample (Fin (P.n κ) → Fin (P.W κ) × ZMod pp.p))
      (fun r => ((P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))).card : ℝ)) ≤
        (cLoc : ℝ) * P.m κ / P.W κ := by
  classical
  let Noise := Fin (P.W κ) × ZMod pp.p
  let NoiseVec := Fin (P.n κ) → Noise
  letI : Nonempty (Fin (P.W κ)) := ⟨0⟩
  have hscalar : ProbComp.prob (ProbComp.sample Noise)
      (fun c => P.noiseValue κ pp c ≠ 0) ≤ 1 / (P.W κ : ℝ) := ?_
  swap
  · have hzero : uprob (fun c : Noise => P.noiseValue κ pp c = 0) =
        1 - 1 / (P.W κ : ℝ) + 1 / ((P.W κ : ℝ) * pp.p) := ?_
    swap
    · simpa only [P.noiseValue_eq, cast_eq, if_pos rfl] using
        (lpnNoise_law pp.p (P.W κ) (0 : ZMod pp.p))
    rw [Bridge.prob_sample_eq_uprob (fun c : Noise => P.noiseValue κ pp c ≠ 0),
      uprob_not (fun c : Noise => P.noiseValue κ pp c = 0), hzero]
    have hnonneg : 0 ≤ 1 / ((P.W κ : ℝ) * pp.p) :=
      div_nonneg zero_le_one (mul_nonneg (Nat.cast_nonneg _) (Nat.cast_nonneg _))
    linarith
  have hcoord : ∀ i : Fin (P.n κ),
      ProbComp.prob (ProbComp.sample NoiseVec)
        (fun r => P.noiseValue κ pp (r i) ≠ 0) ≤ 1 / (P.W κ : ℝ) := ?_
  swap
  · intro i
    rw [ProbComp.prob_eq_expect]
    have h := ProbComp.expect_sample_coord (β := Noise) i
      (fun c (_ : NoiseVec) => if P.noiseValue κ pp c ≠ 0 then (1 : ℝ) else 0)
      (fun _ _ _ => rfl)
    calc
      ProbComp.expect (ProbComp.sample NoiseVec)
          (fun r => if P.noiseValue κ pp (r i) ≠ 0 then (1 : ℝ) else 0) =
        ProbComp.prob (ProbComp.sample Noise)
          (fun c => P.noiseValue κ pp c ≠ 0) := ?_
      _ ≤ 1 / (P.W κ : ℝ) := hscalar
    rw [ProbComp.prob_eq_expect]
    simpa only [ProbComp.expect_const] using h
  have hbad : ∀ j : Fin (P.m κ),
      ProbComp.prob (ProbComp.sample NoiseVec)
        (fun r => j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))) ≤
          (cLoc : ℝ) / P.W κ := ?_
  swap
  · intro j
    let s := (P.R.poly (P.n κ) I.1 (P.outputIndex κ j)).vars
    have hevent : ∀ r : NoiseVec,
        j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i)) ↔
          ∃ i ∈ s, P.noiseValue κ pp (r i) ≠ 0 := ?_
    swap
    · intro r
      rw [P.badSet_eq]
      constructor
      · intro hj
        obtain ⟨i, hi⟩ := (Finset.mem_filter.mp hj).2
        have hparts := Finset.mem_inter.mp hi
        have herr := hparts.2
        rw [P.errSet_eq] at herr
        exact ⟨i, hparts.1, (Finset.mem_filter.mp herr).2⟩
      · rintro ⟨i, hi, hnoise⟩
        apply Finset.mem_filter.mpr
        refine ⟨Finset.mem_univ _, i, Finset.mem_inter.mpr ⟨hi, ?_⟩⟩
        rw [P.errSet_eq]
        exact Finset.mem_filter.mpr ⟨Finset.mem_univ _, hnoise⟩
    have hlocal : s.card ≤ cLoc :=
      (BoolPRG.MultilinearRep.hasLocality_iff P.R cLoc).mp hLoc
        (P.n κ) I.1 (P.outputIndex κ j)
    have hlocalR : (s.card : ℝ) ≤ cLoc := ?_
    swap
    · exact_mod_cast hlocal
    rw [Bridge.prob_sample_eq_uprob]
    calc
      uprob (fun r : NoiseVec =>
          j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))) =
        uprob (fun r : NoiseVec => ∃ i ∈ s, P.noiseValue κ pp (r i) ≠ 0) :=
          OneWay.uprob_congr (α := NoiseVec) hevent
      _ ≤ ∑ i ∈ s, uprob (fun r : NoiseVec => P.noiseValue κ pp (r i) ≠ 0) :=
        uprob_exists_le (α := NoiseVec) s
          (fun (i : Fin (P.n κ)) (r : NoiseVec) => P.noiseValue κ pp (r i) ≠ 0)
      _ ≤ ∑ _i ∈ s, 1 / (P.W κ : ℝ) := Finset.sum_le_sum (fun i _ => ?_)
      _ = (s.card : ℝ) / P.W κ := ?_
      _ ≤ (cLoc : ℝ) / P.W κ :=
        div_le_div_of_nonneg_right hlocalR (Nat.cast_nonneg _)
    · simpa only [Bridge.prob_sample_eq_uprob] using hcoord i
    · rw [Finset.sum_const, nsmul_eq_mul]
      ring
  have hcard : ∀ r : NoiseVec,
      ((P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))).card : ℝ) =
        ∑ j : Fin (P.m κ),
          if j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))
          then (1 : ℝ) else 0 := ?_
  swap
  · intro r
    simpa only [Finset.filter_mem_eq_inter, Finset.univ_inter] using
      (Finset.sum_boole
        (fun (j : Fin (P.m κ)) => j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i)))
        Finset.univ).symm
  calc
    ProbComp.expect (ProbComp.sample NoiseVec)
        (fun r => ((P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))).card : ℝ)) =
      ProbComp.expect (ProbComp.sample NoiseVec)
        (fun r => ∑ j : Fin (P.m κ),
          if j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))
          then (1 : ℝ) else 0) :=
        congrArg (ProbComp.expect (ProbComp.sample NoiseVec)) (funext hcard)
    _ = ∑ j : Fin (P.m κ), ProbComp.expect (ProbComp.sample NoiseVec)
        (fun r => if j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))
          then (1 : ℝ) else 0) :=
      ProbComp.expect_sum (ProbComp.sample NoiseVec) Finset.univ
        (fun j r => if j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))
          then (1 : ℝ) else 0)
    _ = ∑ j : Fin (P.m κ), ProbComp.prob (ProbComp.sample NoiseVec)
        (fun r => j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))) :=
      Finset.sum_congr rfl (fun j _ =>
        (ProbComp.prob_eq_expect (ProbComp.sample NoiseVec)
          (fun r => j ∈ P.badSet κ pp I (fun i => P.noiseValue κ pp (r i)))).symm)
    _ ≤ ∑ _j : Fin (P.m κ), (cLoc : ℝ) / P.W κ :=
      Finset.sum_le_sum (fun j _ => hbad j)
    _ = (cLoc : ℝ) * P.m κ / P.W κ := ?_
  rw [Finset.sum_const, Finset.card_univ, Fintype.card_fin, nsmul_eq_mul]
  ring

/-- Markov's inequality bounds the too-many-bad event using Γ+1
[JLS26, proof of Claim 4.2, p. 20]. Dv3 makes the strict integer threshold
explicit, avoiding any division by Γ=0.

**Proof sketch.** On the event, the integer bad count is at least Γ+1. Apply the published
Markov inequality and the locality expectation estimate, using W>0. -/
theorem cond1_prob_le (P : Params) {cLoc : ℕ} (hLoc : P.R.HasLocality cLoc)
    (κ : ℕ) (pp : PrimeParam κ) (I : P.Idx' κ pp) :
    ProbComp.prob (ProbComp.sample (Fin (P.n κ) → Fin (P.W κ) × ZMod pp.p))
      (fun r => P.Γ κ < (P.badSet κ pp I (fun i => P.noiseValue κ pp (r i))).card) ≤
        (cLoc : ℝ) * P.m κ / ((P.W κ : ℝ) * (P.Γ κ + 1)) := sorry


end Params
