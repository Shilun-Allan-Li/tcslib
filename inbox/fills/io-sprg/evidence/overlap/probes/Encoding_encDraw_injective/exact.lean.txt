import TCSlib.Complexity.CircuitComplexity.Uniform
import TCSlib.Cryptography.Encoding.Modular
import TCSlib.Complexity.TuringMachine.Encoding
import Lean.Elab.Tactic.LibrarySearch

set_option maxHeartbeats 100000
set_option relaxedAutoImplicit false
set_option autoImplicit false

open Turing
open TCSlib.Cryptography

namespace IoSprgEncodingOverlap

example (κ ell n W p width : ℕ) [NeZero p]
    (hW : W ≤ 2 ^ (ell + 1)) (hp : p ≤ 2 ^ width) :
    Function.Injective (fun r : (Fin n → Bool) ×
        (Fin κ → (Fin ell → ZMod p) × (Fin n → Fin W × ZMod p)) =>
      pairEncode (List.ofFn r.1)
        (BoolCircuit.encodeList (fun one : (Fin ell → ZMod p) ×
            (Fin n → Fin W × ZMod p) =>
          pairEncode (Encoding.Modular.encodeVector width one.1)
            (BoolCircuit.encodeList (fun c : Fin W × ZMod p =>
              pairEncode (Candidates.natToList (ell + 1) c.1.val)
                (Candidates.natToList width c.2.val)) (List.ofFn one.2)))
          (List.ofFn r.2))) := by
  exact?

end IoSprgEncodingOverlap
