example {m B : ℕ} {R : Type*} [AddCommMonoid R] [DecidableEq R]
    (φ : Fin m → Fin B) (rc : ℕ) (corr : Fin m → R) (bad : Finset (Fin m))
    (hcorr : ∀ j, j ∉ bad → corr j = 0) (b : Fin B) :
    (Finset.univ.filter (fun c : Fin rc × Fin rc =>
      (∑ i : Fin m, if φ i = b ∧ (if h : ¬ (∃ b : Fin B, rc * rc < (Finset.univ.filter (fun q :
        Fin m => φ q = b)).card) ∧
          (Finset.univ.filter (fun q : Fin m => q < i ∧ φ q = φ i)).card < rc * rc then
        some ((finProdFinEquiv : (Fin rc × Fin rc) ≃ Fin (rc * rc)).symm
          ⟨(Finset.univ.filter (fun q : Fin m => q < i ∧ φ q = φ i)).card, h.2⟩)
      else none) = some (c.1, c.2)
        then corr i else 0) ≠ 0)).card ≤
      ((Finset.univ.filter (fun j : Fin m => φ j = b)) ∩ bad).card := by
  exact?

end IoSprgBucketsSparseOverlap
