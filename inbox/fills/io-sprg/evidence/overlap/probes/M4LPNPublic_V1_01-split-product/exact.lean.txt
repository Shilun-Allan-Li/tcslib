import TCSlib.Cryptography.Games.Bridges.Experiments
import Lean.Elab.Tactic.LibrarySearch

set_option maxHeartbeats 100000
set_option relaxedAutoImplicit false
set_option autoImplicit false

open TCSlib.Cryptography

example {G Buck Seed Table Post : Type}
    [Fintype G] [Nonempty G] [Fintype Buck] [Nonempty Buck]
    [Fintype Seed] [Nonempty Seed] [Fintype Table] [Nonempty Table]
    [Fintype Post] [Nonempty Post]
    (R : G × (Buck × (Seed × (Table × Post))) → ℝ) :
    Games.ProbComp.expect (Games.ProbComp.sample (G × (Buck × (Seed × (Table × Post))))) R =
      Games.ProbComp.expect (Games.ProbComp.sample G) (fun g =>
        Games.ProbComp.expect (Games.ProbComp.sample Buck) (fun b =>
          Games.ProbComp.expect (Games.ProbComp.sample Seed) (fun σ =>
            Games.ProbComp.expect (Games.ProbComp.sample Table) (fun t =>
              Games.ProbComp.expect (Games.ProbComp.sample Post) (fun p => R (g, b, σ, t, p))))))
  := by
    exact?
