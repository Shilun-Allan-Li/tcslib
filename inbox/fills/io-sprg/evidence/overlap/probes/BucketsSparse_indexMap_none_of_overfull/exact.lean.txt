namespace IoSprgBucketsSparseOverlap

example {m B : ℕ} (φ : Fin m → Fin B) (rc : ℕ)
    (h : (∃ b : Fin B, rc * rc < (Finset.univ.filter (fun q : Fin m => φ q = b)).card)) (j :
        Fin m) :
    (if h : ¬ (∃ b : Fin B, rc * rc < (Finset.univ.filter (fun q : Fin m => φ q = b)).card) ∧
          (Finset.univ.filter (fun q : Fin m => q < j ∧ φ q = φ j)).card < rc * rc then
        some ((finProdFinEquiv : (Fin rc × Fin rc) ≃ Fin (rc * rc)).symm
          ⟨(Finset.univ.filter (fun q : Fin m => q < j ∧ φ q = φ j)).card, h.2⟩)
      else none) = none := by
  exact?

end IoSprgBucketsSparseOverlap
