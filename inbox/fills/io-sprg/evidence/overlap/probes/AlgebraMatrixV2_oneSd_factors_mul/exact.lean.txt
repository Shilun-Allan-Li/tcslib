namespace IoSprgAlgebraOverlap

example {rows cols t : ℕ} {R : Type*} [Semiring R] [DecidableEq R]
    (M : Matrix (Fin rows) (Fin cols) R) (L : List (Fin rows × Fin cols))
    (hL : L.Nodup) (hlen : L.length ≤ t) (hmem : ∀ c, c ∈ L ↔ M c.1 c.2 ≠ 0) :
    (Matrix.of (fun (i : Fin rows) (k : Fin t) => match L[k.val]? with
      | none => 0
      | some c => if i = c.1 then M c.1 c.2 else 0) : Matrix (Fin rows) (Fin t) R) *
    (Matrix.of (fun (k : Fin t) (j : Fin cols) => match L[k.val]? with
      | none => 0
      | some c => if j = c.2 then 1 else 0) : Matrix (Fin t) (Fin cols) R) = M := by
  exact?

end IoSprgAlgebraOverlap
