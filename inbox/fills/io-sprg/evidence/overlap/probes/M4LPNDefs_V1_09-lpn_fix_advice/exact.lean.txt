  let areaPrefix := "TCSlib.Cryptography.Obfuscation.StructuredSeed."

    "Internal.lpn_accept_realizes_eq",
    "Internal.lpn_gap_realizes_eq",
    "Internal.lpn_uniform_challenge_reindex",
    "Internal.lpn_real_seed_reindex",

    "Internal.lpn_real_section",
    "Internal.lpn_uniform_section",
    "Internal.lpn_fix_advice",
    "Params.lpn_hop"
  ]
  for suffix in suffixes do
    let name := ((areaPrefix ++ suffix).splitOn ".").foldl
      (fun n p => Lean.Name.str n p) Lean.Name.anonymous
    if env.contains name (skipRealize := false) then
      Lean.throwError "LPN_DEFS_TARGET_PRESENT"

    (h : L = P.ℓ (P.kappaOf L)) (tailReal : Bool)
    (hop : Fin (P.kappaOf L + 1)) (hhop : hop.val < P.kappaOf L) :
    ∃ w : P.LPNAdvice π L,
      Games.advantage (P.chainGame A tailReal hop.val (P.kappaOf L) (π (P.kappaOf L)))
        (P.chainGame A tailReal (hop.val + 1) (P.kappaOf L) (π (P.kappaOf L))) ≤
      LWE.decisionAdvantage ((P.lpnFamily_raw π).toLWE.parameters L)
        ((P.lpnFamily_raw π).toLWE.noise L)
        (fun x (r : A.Coins (P.kappaOf L)) =>
          decide (A.runAtL (P.kappaOf L) (P.lpnPre π L w x) (List.ofFn r) = [true]))
  := by
    exact?
