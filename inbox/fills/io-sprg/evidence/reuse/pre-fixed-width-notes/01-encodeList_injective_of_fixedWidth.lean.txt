/-- Fixed-width injective blocks make the existing tagged list code injective; encoding glue.
The private BoolCircuit prefix-free theorem overlaps conceptually but is not called or copied.

**Proof sketch.** Compare the leading nil/cons marker. In the cons case, equal block widths
let List.append_inj split the head code from the rest. Apply block injection and induction. -/
theorem encodeList_injective_of_fixedWidth {α : Type} {f : α → List Bool}
    (hf : Function.Injective f) (width : ℕ) (hwidth : ∀ a, (f a).length = width) :
    Function.Injective (BoolCircuit.encodeList f) := by
  intro xs
  induction xs with
  | nil =>
    intro ys h
    cases ys with
    | nil => rfl
    | cons y ys =>
      rw [BoolCircuit.encodeList_nil, BoolCircuit.encodeList_cons] at h
      cases (List.cons.inj h).1
  | cons x xs ih =>
    intro ys h
    cases ys with
    | nil =>
      rw [BoolCircuit.encodeList_cons, BoolCircuit.encodeList_nil] at h
      cases (List.cons.inj h).1
    | cons y ys =>
      rw [BoolCircuit.encodeList_cons, BoolCircuit.encodeList_cons] at h
      have hcodes := (List.cons.inj h).2
      have hparts := List.append_inj hcodes ((hwidth x).trans (hwidth y).symm)
      exact congrArg₂ List.cons (hf hparts.1) (ih ys hparts.2)

end TCSlib.Cryptography.Obfuscation.StructuredSeed.Internal
