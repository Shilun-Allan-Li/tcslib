/-!
Failed alternative (consumer): P-1's `Init` case through the whole-function reader and through
its pointwise form, then `oracleImpl_zero_onSpec`. Under the development seals the second

open TCSlib.Cryptography.AdvancedEncryption.IdentityBased

theorem altWhole_init_none (S : IBEScheme) (b : Bool) (κ : ℕ) (pp : S.PP κ) (msk : S.MSK κ)
    (x : (S.sidSpec κ).dom 0) (c : Option (Fin (S.ctLen.eval κ) → Bool))
    (h0 : (S.sidSpec κ).dom 0 = (Fin (S.idLen.eval κ) → Bool))
    (h0' : (S.sidSpec κ).rng 0 = (Fin (S.ppLen.eval κ) → Bool)) :
    S.sidImpl b κ pp msk 0 x (none, c) =
      .pure (cast h0'.symm (S.encPP κ pp), (some (cast h0 x), c)) := by
  rw [IBEScheme.alt_sidImpl_eq_oracleImpl, IBEScheme.oracleImpl_zero_onSpec,
    IBEScheme.sidImpl_init_none, OracleComp.pure_bind]

theorem altPointwise_init_none (S : IBEScheme) (b : Bool) (κ : ℕ) (pp : S.PP κ) (msk : S.MSK κ)

    (h0 : (S.sidSpec κ).dom 0 = (Fin (S.idLen.eval κ) → Bool))
    (h0' : (S.sidSpec κ).rng 0 = (Fin (S.ppLen.eval κ) → Bool)) :

/-- The explicit instance of `oracleImpl_zero_onSpec` at the game's own handlers. -/
