/-! Mathlib-first search for the Q10 helpers (bb1-sim-statements, series 2): each statement in a
generic form (a finite group `G` with `x ^ n = 1` or `|G| = p`), closed by `exact?`/`apply?`. -/

-- (1) the additive exponent law
example (G : Type) [Group G] (n : ℕ) [NeZero n] (x : G) (hx : x ^ n = 1) (a b : ZMod n) :
    x ^ (a + b).val = x ^ a.val * x ^ b.val := by
  exact?

-- (2) the negation exponent law
example (G : Type) [Group G] (n : ℕ) [NeZero n] (x : G) (hx : x ^ n = 1) (a : ZMod n) :
    x ^ (-a).val = (x ^ a.val)⁻¹ := by
  exact?

-- (3) Z_p ≃ G for g ≠ 1
example (G : Type) [Group G] [Fintype G] (p : ℕ) [Fact p.Prime] (hc : Fintype.card G = p)
    (g : G) (hg : g ≠ 1) : Function.Bijective fun a : ZMod p => g ^ a.val := by
  exact?

-- (4) injectivity of natural casts below the order
example (p : ℕ) (m n : ℕ) (hm : m < p) (hn : n < p) : (m : ZMod p) = n ↔ m = n := by
  exact?
