open Turing Complexity

-- N1 ordinary conversion: a `rfl` theorem through a foreign data definition
theorem cp2_N1_rfl : pairDecode [] = none := rfl -- CP2-SITE N1

-- N2 ordinary conversion through `show`
theorem cp2_N2_show (z : List Bool) : pairDecode (false :: true :: z) = some ([], z) := by
  show some ([], z) = some ([], z) -- CP2-SITE N2
  rfl

-- N3 anonymous command
example : pairDecode [] = none := rfl -- CP2-SITE N3

-- N4 generated matcher: the unfolding equation is used, the compiler-generated matcher reduction
-- is what the seal must block
theorem cp2_N4_matcher (z : List Bool) : pairDecode (false :: true :: z) = some ([], z) := by
  rw [pairDecode.eq_def]
  rfl -- CP2-SITE N4

-- N5 definition-body sensitivity: a data definition whose body needs the foreign body
def cp2_N5_body : {l : List Bool // pairDecode l = none} := ⟨[], rfl⟩ -- CP2-SITE N5

-- N6 field of a structure-valued foreign definition by conversion
theorem cp2_N6_field {k : ℕ} {S H : Type} {input : List Bool} (emb : S → H) (ret : H)
    (pre : List Bool) (c : Cfg k Bool S input) :
    (emitCfg emb ret pre c).output = pre ++ c.output := rfl -- CP2-SITE N6

-- N7 destructuring a foreign predicate definition
theorem cp2_N7_pred (f : List Bool → List Bool) (h : PolyTimeComputable f) :
    ∃ M : FinTM Bool, M = M := by
  obtain ⟨M, -⟩ := h -- CP2-SITE N7
  exact ⟨M, rfl⟩

-- P1 the published case reader
theorem cp2_P1_reader : pairDecode [] = none := pairDecode_nil

-- P2 the published field reader
theorem cp2_P2_field {k : ℕ} {S H : Type} {input : List Bool} (emb : S → H) (ret : H)
    (pre : List Bool) (c : Cfg k Bool S input) :
    (emitCfg emb ret pre c).output = pre ++ c.output := emitCfg_output emb ret pre c

-- P3 the published predicate reader
theorem cp2_P3_iff (f : List Bool → List Bool) (h : PolyTimeComputable f) :
    ∃ M : FinTM Bool, M = M := by
  obtain ⟨M, -⟩ := (polyTimeComputable_iff f).1 h
  exact ⟨M, rfl⟩

-- P4 standing-rule notation: `PolyExpr.eval` is never sealed
theorem cp2_P4_notation : (PolyExpr.var.add (PolyExpr.const 1)).eval 3 = 4 := rfl
