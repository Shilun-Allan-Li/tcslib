-- N4 generated matcher: after the unfolding equation the goal compares the foreign definition's
-- compiler-generated matcher with this file's own matcher on a free discriminant; closing it needs
-- the foreign matcher's body (delta), which is what the seal must block. (A matcher applied to a
-- constructor discriminant is reduced by `whnfCore` without delta, so it is no seal control;
-- the first attempt used that form and is retained as `Controls-attempt1.source.txt`.)
theorem cp2_N4_matcher (z : List Bool) :
    pairDecode z = (match z with
      | false :: false :: rest => (pairDecode rest).map fun p => (false :: p.1, p.2)
      | true :: true :: rest => (pairDecode rest).map fun p => (true :: p.1, p.2)
      | false :: true :: rest => some ([], rest)
      | _ => none) := by
  rw [pairDecode.eq_def]
  rfl -- CP2-SITE N4
