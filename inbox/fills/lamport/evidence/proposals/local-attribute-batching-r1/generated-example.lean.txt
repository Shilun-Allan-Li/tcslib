
set_option maxHeartbeats 0
run_cmd do
  let attrStx ← `(attr| irreducible)
  Lean.Elab.Command.liftCoreM do
    let label : String := "BATCH_PROPOSAL_INSTALL"
    let payload : String := "BoolCircuit.CompileListSpec.mk._flat_ctor\tTCSlib.Complexity.CircuitComplexity.TreeDAG\nBoolCircuit.CompileSpec.mk._flat_ctor\tTCSlib.Complexity.CircuitComplexity.TreeDAG\nBoolCircuit.DAGCircuit.IsFaninTwo\tTCSlib.Complexity.CircuitComplexity.DAGCircuit"
    let lines := if payload.isEmpty then [] else payload.splitOn "
"
    unless lines.length == 3 do throwError "LAMP-HARNESS bad serialized count"
    let env ← Lean.getEnv
    let mut catalog : Std.HashMap String (Array Lean.Name) := {}
    for (n, _) in env.constants.map₁.toList do
      let some mi := env.getModuleIdxFor? n | continue
      unless (`TCSlib).isPrefixOf env.header.moduleNames[mi.toNat]! do continue
      catalog := catalog.insert n.toString (((catalog[n.toString]?).getD #[]).push n)
    let attr ← Lean.ofExcept <| Lean.getAttributeImpl (← Lean.getEnv) `irreducible
    let mut count : Nat := 0
    for row in lines do
      let [spelling, home] := row.splitOn "	" | throwError "LAMP-HARNESS malformed row"
      let ns := (catalog[spelling]?).getD #[]
      unless ns.size == 1 do throwError "LAMP-HARNESS absent/ambiguous {spelling}"
      let n := ns[0]!
      let some mi := env.getModuleIdxFor? n | throwError "LAMP-HARNESS no home {spelling}"
      unless env.header.moduleNames[mi.toNat]!.toString == home do
        throwError "LAMP-HARNESS wrong home {spelling}"
      let some ci := env.find? n | throwError "LAMP-HARNESS no constant {spelling}"
      unless ci.isDefinition do throwError "LAMP-HARNESS not definition {spelling}"
      let before ← Lean.getReducibilityStatus n
      unless before == .irreducible do
        Lean.withExporting (isExporting := false) do
          attr.add n attrStx.raw .local
      let after ← Lean.getReducibilityStatus n
      unless after == .irreducible do throwError "LAMP-HARNESS unsealed {spelling}"
      Lean.logInfo m!"LAMP-SEAL-NAME	{label}	INSTALL	{spelling}	{home}"
      count := count + 1
    unless count == 3 do throwError "LAMP-HARNESS wrong verified count"
    Lean.logInfo m!"LAMP-SEAL-INSTALL	{label}	{count}"
