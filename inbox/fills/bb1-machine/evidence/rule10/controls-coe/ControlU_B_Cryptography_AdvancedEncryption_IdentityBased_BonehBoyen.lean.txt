        S.MsgSpace κ x.1 m → S.dec κ x.2.1 x.2.2 = some m := Iff.rfl

      (S.honestRun κ id m).AllOutputs fun x =>
        S.MsgSpace κ x.1 m → S.dec κ x.2.1 x.2.2 = some m :=
  S.correct_iff
