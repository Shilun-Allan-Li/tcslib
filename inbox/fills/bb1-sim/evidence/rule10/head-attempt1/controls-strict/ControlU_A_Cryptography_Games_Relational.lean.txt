import Lean
import TCSlib.Cryptography.Games.Relational

set_option maxHeartbeats 0
set_option autoImplicit false
set_option relaxedAutoImplicit false

namespace PairingControls

open TCSlib.Cryptography.Games in
theorem ctlAllOutputs {α : Type} (P : α → Prop) (a : α) :
    (OracleComp.pure (spec := unifSpec) a).AllOutputs P ↔ P a := Iff.rfl

open TCSlib.Cryptography.Games in
theorem ctlAllOutputsReader {α : Type} (P : α → Prop) (a : α) :
    (OracleComp.pure (spec := unifSpec) a).AllOutputs P ↔ P a :=
  OracleComp.allOutputs_pure

open TCSlib.Cryptography.Games in
theorem ctlExpect {α : Type} (a : α) (g : α → ℝ) :
    ProbComp.expect (OracleComp.pure (spec := unifSpec) a) g = g a := rfl

open TCSlib.Cryptography.Games in
theorem ctlExpectReader {α : Type} (a : α) (g : α → ℝ) :
    ProbComp.expect (OracleComp.pure (spec := unifSpec) a) g = g a :=
  ProbComp.expect_pure a g

end PairingControls
