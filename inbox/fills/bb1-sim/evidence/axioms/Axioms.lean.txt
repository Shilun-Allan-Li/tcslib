open Lean Elab Command

elab "#sorry_trace " n:ident : command => do
  let env ← getEnv
  let root := n.getId
  let mut seen : NameSet := {}
  let mut stack : Array Name := #[root]
  let mut hits : Array Name := #[]
  let mut fuel := 1000000
  while fuel > 0 && stack.size > 0 do
    fuel := fuel - 1
    let c := stack.back!
    stack := stack.pop
    if seen.contains c then continue
    seen := seen.insert c
    match env.find? c with
    | none => pure ()
    | some ci =>
      let refs := ci.getUsedConstantsAsSet
      if refs.contains ``sorryAx then hits := hits.push c
      for r in refs do
        unless seen.contains r do stack := stack.push r
  logInfo m!"TRACE {root}: closure {seen.size}; direct sorryAx users: {hits.qsort Name.lt}"
