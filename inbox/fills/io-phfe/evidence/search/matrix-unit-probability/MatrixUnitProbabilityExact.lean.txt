open scoped Matrix

example {K : Type} [Field K] [Fintype K] (n : ℕ) :
    ProbComp.prob (ProbComp.sample (Matrix (Fin n) (Fin n) K)) (fun M => IsUnit M) =
      (((∏ i : Fin n,
          Fintype.card K ^ n - Fintype.card K ^ (i : ℕ)) : ℕ) : ℝ) /
        (Fintype.card K : ℝ) ^ (n * n) := exact?%

end TCSlib.Cryptography.Pairing
