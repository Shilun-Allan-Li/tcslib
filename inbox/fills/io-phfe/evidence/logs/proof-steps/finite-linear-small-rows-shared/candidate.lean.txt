/-!
# Uniform matrix images over finite fields

## Main definitions

The results concern exact finite laws and introduce no sampling implementation.

## Main results

* `expect_sample_matrix_mul_of_mulVec_surjective`: uniform images under surjective matrix maps.
* `prob_sample_matrix_isUnit_eq`: the published general-linear count as an exact probability.
* `smallRows_surjectivity_failure_le`: the two-column rank bound for at most two rows.

## References

* [Wee20] H. Wee, *Functional Encryption for Quadratic Functions from k-Lin, Revisited*,
  TCC 2020; full version, Cryptology ePrint Archive, Report 2020/1420, Section 2.1.
* [GJLS21] R. Gay, A. Jain, H. Lin, A. Sahai, *Indistinguishability Obfuscation from
  Simple-to-State Hard Problems: New Assumptions, New Techniques, and Simplification*,
  EUROCRYPT 2021; ePrint 2020/764, revision of 3 January 2025, Definition 3.3.
-/

/-- A surjective scalar column map sends a uniform coefficient matrix to a uniform output
matrix. This is finite-field distribution glue; no sampler or machine is constructed.

**Proof sketch.** Regard multiplication of the whole coefficient matrix as a surjective
additive homomorphism, using a right inverse of the column map. Reuse the equivalences
from each fiber to its kernel to split a uniform input into output and kernel coordinates.
The existing uniform-average and product laws then remove the unused kernel coordinate. -/
theorem expect_sample_matrix_mul_of_mulVec_surjective
    {K : Type} [Field K] [Fintype K] {r m c : ℕ}
    (M : Matrix (Fin r) (Fin m) K) (hM : Function.Surjective (Matrix.mulVec M))
    (F : Matrix (Fin r) (Fin c) K → ℝ) :
    ProbComp.expect (ProbComp.sample (Matrix (Fin m) (Fin c) K)) (fun S => F (M * S)) =
      ProbComp.expect (ProbComp.sample (Matrix (Fin r) (Fin c) K)) F := by
  classical
  obtain ⟨R, hR⟩ :=
    (Matrix.mulVec_surjective_iff_exists_right_inverse (A := M)).mp hM
  let f : Matrix (Fin m) (Fin c) K →+ Matrix (Fin r) (Fin c) K :=
    Matrix.addMonoidHomMulLeft M
  have hf : Function.Surjective f := ?_
  swap
  · intro U
    refine ⟨R * U, ?_⟩
    change M * (R * U) = U
    rw [← Matrix.mul_assoc, hR, Matrix.one_mul]
  let e : Matrix (Fin m) (Fin c) K ≃ (Matrix (Fin r) (Fin c) K × f.ker) :=
    (Equiv.sigmaFiberEquiv f).symm.trans
      (Equiv.sigmaEquivProdOfEquiv
        (fun U => AddMonoidHom.fiberEquivKerOfSurjective (f := f) hf U))
  have hFst (S : Matrix (Fin m) (Fin c) K) : (e S).1 = M * S := rfl
  have h := ProbComp.expect_sample_equiv e (fun p => F p.1)
  simp only [hFst] at h
  rw [h, ← Bridge.expect_sample_pair]
  simp only [ProbComp.expect_bind, ProbComp.expect_pure, ProbComp.expect_const]

/-- The probability that a uniform square matrix over a finite field is a unit is the
published general-linear-group cardinality divided by the cardinality of all matrices.
The numerator uses natural subtraction, and the formula includes dimension zero.

**Proof sketch.** Read prob_sample_eq_uprob and uprob_def. Identify the unit predicate
subtype with the range of Units.val and use card_range_of_injective val_injective.
Apply Matrix.card_GL_field and the existing function-cardinality/power laws. -/
theorem prob_sample_matrix_isUnit_eq {K : Type} [Field K] [Fintype K] (n : ℕ) :
    ProbComp.prob (ProbComp.sample (Matrix (Fin n) (Fin n) K)) (fun M => IsUnit M) =
      (((∏ i : Fin n,
          (Fintype.card K ^ n - Fintype.card K ^ (i : ℕ))) : ℕ) : ℝ) /
        (Fintype.card K : ℝ) ^ (n * n) := by
  classical
  rw [Bridge.prob_sample_eq_uprob, uprob_def]
  change (Nat.card (Set.range (Units.val :
      (Matrix (Fin n) (Fin n) K)ˣ → Matrix (Fin n) (Fin n) K)) : ℝ) /
      (Fintype.card (Matrix (Fin n) (Fin n) K) : ℝ) = _
  rw [Nat.card_range_of_injective Units.val_injective,
    Matrix.card_GL_field (𝔽 := K) n]
  congr 1
  change (Fintype.card (Fin n → Fin n → K) : ℝ) =
    (Fintype.card K : ℝ) ^ (n * n)
  simp only [Fintype.card_fun, Fintype.card_fin, Nat.cast_pow]
  exact (pow_mul (Fintype.card K : ℝ) n n).symm

/-- A uniform matrix with two columns and at most two rows fails surjectivity with
probability at most `2/|K|`. This is the exact rank-failure obligation for small-row MDDH.

**Proof sketch.** Split the row count into zero, one and two. Zero rows are automatically
surjective. One row fails only at the zero row, with mass `|K|⁻²`. For two rows use the
existing finite-field general-linear cardinality formula to obtain singular mass
`|K|⁻¹+|K|⁻²-|K|⁻³ ≤ 2/|K|`. -/
theorem smallRows_surjectivity_failure_le {K : Type} [Field K] [Fintype K]
    (r : ℕ) (hr : r ≤ 2) :
    ProbComp.prob (ProbComp.sample (Matrix (Fin r) (Fin 2) K))
        (fun M => ¬ Function.Surjective (Matrix.mulVec M)) ≤
      2 / (Fintype.card K : ℝ) := by
  classical
  let q : ℕ := Fintype.card K
  have hq2 : 2 ≤ q := Fintype.one_lt_card_iff_nontrivial.mpr inferInstance
  have hq1 : 1 ≤ q := (Nat.le_add_right 1 1).trans hq2
  have hq : (1 : ℝ) ≤ (q : ℝ) := Nat.one_le_cast.mpr hq1
  have hqpos : (0 : ℝ) < (q : ℝ) := lt_of_lt_of_le zero_lt_one hq
  have hrecip : (1 : ℝ) / (q : ℝ) ^ 2 ≤ 1 / (q : ℝ) := ?_
  swap
  · simpa only [pow_one] using
      one_div_pow_le_one_div_pow_of_le hq (Nat.le_add_right 1 1)
  have htwo : (2 : ℝ) / (q : ℝ) = 2 * (1 / (q : ℝ)) := by ring
  cases r with
  | zero =>
      have hs : ∀ M : Matrix (Fin 0) (Fin 2) K,
          Function.Surjective (Matrix.mulVec M) := ?_
      swap
      · intro M y
        exact ⟨0, Subsingleton.elim _ _⟩
      simp only [ProbComp.prob_eq_expect, hs, not_true_eq_false, if_false,
        ProbComp.expect_const]
      positivity
  | succ r =>
      cases r with
      | zero =>
          have hbad : ∀ M : Matrix (Fin 1) (Fin 2) K,
              ¬ Function.Surjective (Matrix.mulVec M) → M = 0 := ?_
          swap
          · intro M hM
            let e : (Fin 1 → K) ≃ₗ[K] K := LinearEquiv.funUnique (Fin 1) K K
            let L : Module.Dual K (Fin 2 → K) :=
              e.toLinearMap.comp (Matrix.mulVecLin M)
            have hL : L = 0 := ?_
            swap
            · by_contra hL
              apply hM
              have hsurj : Function.Surjective L :=
                LinearMap.range_eq_top.mp (Module.Dual.range_eq_top_of_ne_zero hL)
              exact (Equiv.comp_surjective (Matrix.mulVec M) e.toEquiv).mp hsurj
            ext i j
            change M i j = 0
            have he : e (M *ᵥ Pi.single j 1) = 0 :=
              congrArg (fun F : Module.Dual K (Fin 2 → K) => F (Pi.single j 1)) hL
            rw [Matrix.mulVec_single_one] at he
            change (LinearEquiv.funUnique (Fin 1) K K) (M.col j) = 0 at he
            rw [LinearEquiv.funUnique_apply] at he
            change M default j = 0 at he
            have hi : i = (default : Fin 1) := Subsingleton.elim _ _
            simpa only [hi] using he
          have hzero : uprob (fun M : Matrix (Fin 1) (Fin 2) K => M = 0) =
              1 / (q : ℝ) ^ 2 := ?_
          swap
          · rw [TCSlib.Cryptography.OneWay.uprob_eq_one_div_card]
            change (1 : ℝ) / (Fintype.card (Fin 1 → Fin 2 → K) : ℝ) = _
            simp only [Fintype.card_fun, Fintype.card_fin, pow_one, Nat.cast_pow]
            rfl
          rw [Bridge.prob_sample_eq_uprob]
          refine le_trans (uprob_mono hbad) ?_
          rw [hzero]
          have hx0 : (0 : ℝ) ≤ 1 / (q : ℝ) := one_div_nonneg.mpr hqpos.le
          change (1 : ℝ) / (q : ℝ) ^ 2 ≤ 2 / (q : ℝ)
          rw [htwo]
          nlinarith only [hrecip, hx0]
      | succ r =>
          have hr0 : r = 0 := ?_
          swap
          · omega
          subst r
          have hsq1 : 1 ≤ q ^ 2 := one_le_pow₀ hq1
          have hsqq : q ≤ q ^ 2 := le_self_pow₀ hq1 (Nat.succ_ne_zero 1)
          rw [ProbComp.prob_not]
          simp only [Matrix.mulVec_surjective_iff_isUnit]
          rw [prob_sample_matrix_isUnit_eq]
          simp only [Fin.prod_univ_two, Fin.val_zero, Fin.val_one, pow_zero, pow_one,
            Nat.cast_mul, Nat.cast_sub hsq1, Nat.cast_sub hsqq, Nat.cast_pow, Nat.cast_one]
          change 1 - (((q : ℝ) ^ 2 - 1) * ((q : ℝ) ^ 2 - q)) /
              (q : ℝ) ^ 4 ≤ 2 / (q : ℝ)
          have hformula : 1 - (((q : ℝ) ^ 2 - 1) * ((q : ℝ) ^ 2 - q)) /
              (q : ℝ) ^ 4 =
              1 / (q : ℝ) + 1 / (q : ℝ) ^ 2 - 1 / (q : ℝ) ^ 3 := ?_
          swap
          · field_simp [ne_of_gt hqpos] <;> ring
          rw [hformula]
          have hz0 : (0 : ℝ) ≤ 1 / (q : ℝ) ^ 3 :=
            one_div_nonneg.mpr (pow_nonneg hqpos.le 3)
          rw [htwo]
          linarith only [hrecip, hz0]
