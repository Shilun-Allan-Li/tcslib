/-!
# Fixed-width PHFE encodings

Public and master-secret keys use their exact word widths. Ciphertexts concatenate their
source-group entries without framing overhead. Function keys are padded to the garbling's
uniform polynomial bound and retain the original function word; it determines the actual
coordinate lengths before padding. All data, including simulator state, are represented
without extracting an exponent from a group element.

## Main definitions

* `Wee20.PHFE.encMPK`, `encMSK`, `encSK`, `encCT`: fixed-width data encodings.
* `Wee20.PHFE.keyWord`: a function word and its typed quadratic key.

## Main results

Exact public/secret/ciphertext word lengths, a bounded function-key length, and injectivity
of every data and simulator-state encoding.

## References

* [Wee20] H. Wee, *Functional Encryption for Quadratic Functions from k-Lin, Revisited*,
  TCC 2020; full version, Cryptology ePrint Archive 2020/1420, Sections 3.1-3.2 and 4.2.
* [GJLS21] R. Gay, A. Jain, H. Lin, A. Sahai, *Indistinguishability Obfuscation from
  Simple-to-State Hard Problems: New Assumptions, New Techniques, and Simplification*,
  EUROCRYPT 2021; full version, Cryptology ePrint Archive, Report 2020/764,
  revision of 3 January 2025, Definitions 4.1-4.2 and 4.6.

These encoding lemmas are technical representation glue for the cited algorithms.
-/

/-- The public key has its exact polynomial wire width.

**Proof sketch.** Read the group-vector, matrix and public-key word equations. Use
`length_pairEncode` and `List.length_ofFn` for each matrix row, then evaluate `mpkLen`.
No padding is included in this count. -/
theorem quadraticMPK_length (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) (pk : Quadratic.MPK G ℓ n n κ pp) :
    (Wire.quadraticMPK pp pk).length = (mpkLen G ℓ n).eval κ := by
  simp only [Wire.quadraticMPK_eq, Wire.linearMPK_eq, length_pairEncode,
    Wire.groupMatrix_length, mpkLen_eval, Quadratic.piDim_eq,
    PolyExpr.eval_add, PolyExpr.eval_mul, PolyExpr.eval_const]
  ring

/-- The master secret has its exact polynomial wire width.

**Proof sketch.** Count the public matrix entries and the three scalar matrix blocks using
shared modular encoding length readers. Expand only the published wire equations and the
numeric width reader; collect polynomial terms. -/
theorem quadraticMSK_length (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Quadratic.MSK G ℓ n n κ pp) :
    (Wire.quadraticMSK pp sk).length = (mskLen G ℓ n).eval κ := by
  simp only [Wire.quadraticMSK_eq, Wire.linearMSK_eq, length_pairEncode,
    Wire.groupMatrix_length, Encoding.Modular.encodeMatrix_length, mskLen_eval,
    Quadratic.piDim_eq, PolyExpr.eval_add, PolyExpr.eval_mul, PolyExpr.eval_const]
  ring

/-- The ciphertext has exactly `6n+5` source-group entries.

**Proof sketch.** Count the two outer vectors and the three linear ciphertext blocks;
use concatenation lengths and the polynomial evaluation reader. -/
theorem quadraticCT_length (G : BilinearGen) (n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) (ct : Quadratic.CT G n n κ pp) :
    (Wire.quadraticCT pp ct).length = (ctLen G n).eval κ := by
  simp only [Wire.quadraticCT_eq, Wire.linearCT_eq, List.length_append,
    Wire.groupVector_length, ctLen_eval, Quadratic.piDim_eq, PolyExpr.eval_add,
    PolyExpr.eval_mul, PolyExpr.eval_const]
  ring

/-- A function key's word retains the original function bits before the typed key. -/
def keyWord (G : BilinearGen) {ℓ n : PolyExpr} {d : ℕ}
    (PG : PartialGarbling.Raw ℓ (n.mul n) (TensorFn ℓ n d)) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Key G PG κ pp) : List Bool :=
  pairEncode (List.ofFn sk.function) (Wire.quadraticKey pp PG sk.quadratic)

/-- The function-key word is exactly the original function and its quadratic-key word. -/
theorem keyWord_eq (G : BilinearGen) {ℓ n : PolyExpr} {d : ℕ}
    (PG : PartialGarbling.Raw ℓ (n.mul n) (TensorFn ℓ n d)) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Key G PG κ pp) :
    keyWord G PG κ pp sk =
      pairEncode (List.ofFn sk.function) (Wire.quadraticKey pp PG sk.quadratic) := rfl

/-- Every function-key word fits the garbling's uniform fixed width.

**Proof sketch.** Count the original function bits and group entries in each key block.
Apply `mCount_le_mBound` to its actual function, then the explicit `skLen_eval` reader. -/
theorem keyWord_length_le (G : BilinearGen) {ℓ n : PolyExpr} {d : ℕ}
    (PG : PartialGarbling.Raw ℓ (n.mul n) (TensorFn ℓ n d)) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Key G PG κ pp) :
    (keyWord G PG κ pp sk).length ≤ (skLen G PG).eval κ := by
  have hLen :
      (keyWord G PG κ pp sk).length =
        2 * (tensorFnLen ℓ (n.mul n) G.wG d).eval κ +
          (8 * n.eval κ ^ 3 + 6 * n.eval κ ^ 2 +
            (6 * ℓ.eval κ + 8) * PG.mCount κ pp.orderParam
              (decodeFunction G ℓ n d κ pp sk.function)) * G.wG.eval κ + 10 := ?_
  swap
  · simp only [keyWord_eq, Wire.quadraticKey_eq, Wire.linearKey_eq,
      length_pairEncode, List.length_ofFn, Wire.groupMatrix_length,
      phfeClass_fnLen, Quadratic.piDim_eq, PolyExpr.eval_add, PolyExpr.eval_mul,
      PolyExpr.eval_const]
    ring
  have hNode :
      (6 * ℓ.eval κ + 8) * PG.mCount κ pp.orderParam
          (decodeFunction G ℓ n d κ pp sk.function) ≤
        (6 * ℓ.eval κ + 8) * PG.mBound.eval κ :=
    mul_le_mul_of_nonneg_left
      (PG.mCount_le_mBound κ pp.orderParam (decodeFunction G ℓ n d κ pp sk.function))
      (Nat.zero_le _)
  rw [hLen, skLen_eval]
  exact add_le_add_right
    (add_le_add_left
      (mul_le_mul_of_nonneg_right
        (add_le_add_left hNode (8 * n.eval κ ^ 3 + 6 * n.eval κ ^ 2))
        (Nat.zero_le _))
      (2 * (tensorFnLen ℓ (n.mul n) G.wG d).eval κ)) 10

/-- Public-key encoding reads the exact word at its fixed polynomial width. -/
def encMPK (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ) (pp : PublicParams G κ)
    (pk : Quadratic.MPK G ℓ n n κ pp) : Fin ((mpkLen G ℓ n).eval κ) → Bool :=
  padFin ((mpkLen G ℓ n).eval κ) (Wire.quadraticMPK pp pk)

/-- The public-key encoder is the shared exact-width word adapter. -/
theorem encMPK_eq (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) (pk : Quadratic.MPK G ℓ n n κ pp) :
    encMPK G ℓ n κ pp pk = padFin ((mpkLen G ℓ n).eval κ) (Wire.quadraticMPK pp pk) := rfl

/-- Master-secret encoding uses the exact word at its fixed polynomial width. -/
def encMSK (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ) (pp : PublicParams G κ)
    (sk : Quadratic.MSK G ℓ n n κ pp) : Fin ((mskLen G ℓ n).eval κ) → Bool :=
  padFin ((mskLen G ℓ n).eval κ) (Wire.quadraticMSK pp sk)

/-- The master-secret encoder is the shared exact-width word adapter. -/
theorem encMSK_eq (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Quadratic.MSK G ℓ n n κ pp) :
    encMSK G ℓ n κ pp sk = padFin ((mskLen G ℓ n).eval κ) (Wire.quadraticMSK pp sk) := rfl

/-- Function-key encoding pads the bounded word to its fixed polynomial width. -/
def encSK (G : BilinearGen) {ℓ n : PolyExpr} {d : ℕ}
    (PG : PartialGarbling.Raw ℓ (n.mul n) (TensorFn ℓ n d)) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Key G PG κ pp) : Fin ((skLen G PG).eval κ) → Bool :=
  padFin ((skLen G PG).eval κ) (keyWord G PG κ pp sk)

/-- The function-key encoder is the shared padding adapter at its uniform width. -/
theorem encSK_eq (G : BilinearGen) {ℓ n : PolyExpr} {d : ℕ}
    (PG : PartialGarbling.Raw ℓ (n.mul n) (TensorFn ℓ n d)) (κ : ℕ)
    (pp : PublicParams G κ) (sk : Key G PG κ pp) :
    encSK G PG κ pp sk = padFin ((skLen G PG).eval κ) (keyWord G PG κ pp sk) := rfl

/-- Ciphertext encoding reads the concatenated group word at its exact width. -/
def encCT (G : BilinearGen) (n : PolyExpr) (κ : ℕ) (pp : PublicParams G κ)
    (ct : Quadratic.CT G n n κ pp) : Fin ((ctLen G n).eval κ) → Bool :=
  padFin ((ctLen G n).eval κ) (Wire.quadraticCT pp ct)

/-- The ciphertext encoder is the shared exact-width word adapter. -/
theorem encCT_eq (G : BilinearGen) (n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) (ct : Quadratic.CT G n n κ pp) :
    encCT G n κ pp ct = padFin ((ctLen G n).eval κ) (Wire.quadraticCT pp ct) := rfl

/-- Public-key encoding is injective on all typed public keys.

**Proof sketch.** Use exact width to recover the complete word from equal encodings.
Separate its pair blocks, then each fixed-width matrix entry. Apply encG_injective. -/
theorem encMPK_injective (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) : Function.Injective (encMPK G ℓ n κ pp) := by
  intro a b h
  have hw := congrArg List.ofFn h
  simp only [encMPK_eq] at hw
  rw [ofFn_padFin_of_length (quadraticMPK_length G ℓ n κ pp a),
    ofFn_padFin_of_length (quadraticMPK_length G ℓ n κ pp b)] at hw
  simp only [Wire.quadraticMPK_eq, Wire.linearMPK_eq] at hw
  let aTail0 := pairEncode
    (Wire.groupMatrix pp (fun i j => a.linear.AU i (finProdFinEquiv.symm j)))
    (Wire.groupMatrix pp a.linear.AV)
  let aTail1 := pairEncode (Wire.groupMatrix pp a.linear.AW) aTail0
  let aTail2 := pairEncode (Wire.groupMatrix pp a.linear.A) aTail1
  let aTail3 := pairEncode (Wire.groupMatrix pp a.A2) aTail2
  let bTail0 := pairEncode
    (Wire.groupMatrix pp (fun i j => b.linear.AU i (finProdFinEquiv.symm j)))
    (Wire.groupMatrix pp b.linear.AV)
  let bTail1 := pairEncode (Wire.groupMatrix pp b.linear.AW) bTail0
  let bTail2 := pairEncode (Wire.groupMatrix pp b.linear.A) bTail1
  let bTail3 := pairEncode (Wire.groupMatrix pp b.A2) bTail2
  change pairEncode (Wire.groupMatrix pp a.A1) aTail3 =
    pairEncode (Wire.groupMatrix pp b.A1) bTail3 at hw
  obtain ⟨h1, ht⟩ := Prod.mk.inj
    (@pairEncode_injective (Wire.groupMatrix pp a.A1, aTail3)
      (Wire.groupMatrix pp b.A1, bTail3) hw)
  obtain ⟨h2, hl⟩ := Prod.mk.inj
    (@pairEncode_injective (Wire.groupMatrix pp a.A2, aTail2)
      (Wire.groupMatrix pp b.A2, bTail2) ht)
  obtain ⟨hA, hl⟩ := Prod.mk.inj
    (@pairEncode_injective (Wire.groupMatrix pp a.linear.A, aTail1)
      (Wire.groupMatrix pp b.linear.A, bTail1) hl)
  obtain ⟨hW, hl⟩ := Prod.mk.inj
    (@pairEncode_injective (Wire.groupMatrix pp a.linear.AW, aTail0)
      (Wire.groupMatrix pp b.linear.AW, bTail0) hl)
  obtain ⟨hU, hV⟩ := Prod.mk.inj
    (@pairEncode_injective
      (Wire.groupMatrix pp (fun i j => a.linear.AU i (finProdFinEquiv.symm j)),
        Wire.groupMatrix pp a.linear.AV)
      (Wire.groupMatrix pp (fun i j => b.linear.AU i (finProdFinEquiv.symm j)),
        Wire.groupMatrix pp b.linear.AV) hl)
  have hu : a.linear.AU = b.linear.AU := ?_
  swap
  · funext i j
    have hh := congrFun (congrFun (Wire.groupMatrix_injective pp hU) i)
      (finProdFinEquiv j)
    simpa only [Equiv.symm_apply_apply] using hh
  have h1' := Wire.groupMatrix_injective pp h1
  have h2' := Wire.groupMatrix_injective pp h2
  have hA' := Wire.groupMatrix_injective pp hA
  have hW' := Wire.groupMatrix_injective pp hW
  have hV' := Wire.groupMatrix_injective pp hV
  rcases a with ⟨a1, a2, ⟨aA, aW, aU, aV⟩⟩
  rcases b with ⟨b1, b2, ⟨bA, bW, bU, bV⟩⟩
  cases h1'
  cases h2'
  cases hA'
  cases hW'
  cases hu
  cases hV'
  rfl

/-- Master-secret encoding is injective on all typed master secrets.

**Proof sketch.** Recover the exact word and its pair blocks. Use encG_injective for public
matrix metadata and the shared modular-matrix injectivity under the group's order-width
bound for all scalar blocks. -/
theorem encMSK_injective (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) : Function.Injective (encMSK G ℓ n κ pp) := by
  intro a b h
  have hw := congrArg List.ofFn h
  simp only [encMSK_eq] at hw
  rw [ofFn_padFin_of_length (quadraticMSK_length G ℓ n κ pp a),
    ofFn_padFin_of_length (quadraticMSK_length G ℓ n κ pp b)] at hw
  rcases a with ⟨M₁, ⟨W₁, U₁, V₁⟩⟩
  rcases b with ⟨M₂, ⟨W₂, U₂, V₂⟩⟩
  simp only [Wire.quadraticMSK_eq, Wire.linearMSK_eq] at hw
  have hwidth : pp.group.order ≤ 2 ^ G.wG.eval κ := ?_
  swap
  · simpa only [PublicParams.group_eq] using
      (IdentityBased.BB1Machine.order_le_two_pow G κ pp.description.val)
  let tail1 := pairEncode (Encoding.Modular.encodeMatrix (G.wG.eval κ) W₁)
    (pairEncode (Encoding.Modular.encodeMatrix (G.wG.eval κ)
      (fun i j => U₁ i (finProdFinEquiv.symm j)))
      (Encoding.Modular.encodeMatrix (G.wG.eval κ) V₁))
  let tail2 := pairEncode (Encoding.Modular.encodeMatrix (G.wG.eval κ) W₂)
    (pairEncode (Encoding.Modular.encodeMatrix (G.wG.eval κ)
      (fun i j => U₂ i (finProdFinEquiv.symm j)))
      (Encoding.Modular.encodeMatrix (G.wG.eval κ) V₂))
  have hp := @pairEncode_injective (Wire.groupMatrix pp M₁, tail1)
    (Wire.groupMatrix pp M₂, tail2) hw
  have hM := Wire.groupMatrix_injective pp (congrArg Prod.fst hp)
  have hL := @pairEncode_injective
    (Encoding.Modular.encodeMatrix (G.wG.eval κ) W₁,
      pairEncode (Encoding.Modular.encodeMatrix (G.wG.eval κ)
        (fun i j => U₁ i (finProdFinEquiv.symm j)))
        (Encoding.Modular.encodeMatrix (G.wG.eval κ) V₁))
    (Encoding.Modular.encodeMatrix (G.wG.eval κ) W₂,
      pairEncode (Encoding.Modular.encodeMatrix (G.wG.eval κ)
        (fun i j => U₂ i (finProdFinEquiv.symm j)))
        (Encoding.Modular.encodeMatrix (G.wG.eval κ) V₂)) (congrArg Prod.snd hp)
  have hW := Encoding.Modular.encodeMatrix_injective (q := pp.group.order)
    (G.wG.eval κ) hwidth (congrArg Prod.fst hL)
  have hUV := @pairEncode_injective
    (Encoding.Modular.encodeMatrix (G.wG.eval κ) (fun i j => U₁ i (finProdFinEquiv.symm j)),
      Encoding.Modular.encodeMatrix (G.wG.eval κ) V₁)
    (Encoding.Modular.encodeMatrix (G.wG.eval κ) (fun i j => U₂ i (finProdFinEquiv.symm j)),
      Encoding.Modular.encodeMatrix (G.wG.eval κ) V₂) (congrArg Prod.snd hL)
  have hUflat := Encoding.Modular.encodeMatrix_injective (q := pp.group.order)
    (G.wG.eval κ) hwidth (congrArg Prod.fst hUV)
  have hV := Encoding.Modular.encodeMatrix_injective (q := pp.group.order)
    (G.wG.eval κ) hwidth (congrArg Prod.snd hUV)
  have hU : U₁ = U₂ := ?_
  swap
  · funext i j
    simpa only [Equiv.symm_apply_apply] using
      (congrFun (congrFun hUflat i) (finProdFinEquiv j))
  cases hM
  cases hW
  cases hU
  cases hV
  rfl

/-- Function-key encoding is injective, including across different original function words.

**Proof sketch.** Equal padded words first have equal fixed-length original function blocks.
That fixes their coordinate dimensions, hence their unpadded lengths. Recover the complete
key words and use pair and element-encoding injectivity in each block. -/
theorem encSK_injective (G : BilinearGen) {ℓ n : PolyExpr} {d : ℕ}
    (PG : PartialGarbling.Raw ℓ (n.mul n) (TensorFn ℓ n d)) (κ : ℕ)
    (pp : PublicParams G κ) : Function.Injective (encSK G PG κ pp) := sorry

/-- Ciphertext encoding is injective on all typed ciphertexts.

**Proof sketch.** Recover the exact word. Split the fixed-width vector blocks and read their
individual group encodings; encG_injective and record extensionality finish. -/
theorem encCT_injective (G : BilinearGen) (n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) : Function.Injective (encCT G n κ pp) := by
  intro c d h
  have hw := congrArg List.ofFn h
  simp only [encCT_eq] at hw
  rw [ofFn_padFin_of_length (quadraticCT_length G n κ pp c),
    ofFn_padFin_of_length (quadraticCT_length G n κ pp d)] at hw
  simp only [Wire.quadraticCT_eq, Wire.linearCT_eq, List.append_assoc] at hw
  rcases c with ⟨y1, y2, ⟨c0, c1, c2⟩⟩
  rcases d with ⟨z1, z2, ⟨d0, d1, d2⟩⟩
  rcases List.append_inj hw (by simp only [Wire.groupVector_length]) with ⟨h1, ht⟩
  rcases List.append_inj ht (by simp only [Wire.groupVector_length]) with ⟨h2, hl⟩
  rcases List.append_inj hl (by simp only [Wire.groupVector_length]) with ⟨h0, hl⟩
  rcases List.append_inj hl (by simp only [Wire.groupVector_length]) with ⟨hc1, hc2⟩
  cases Wire.groupVector_injective pp h1
  cases Wire.groupVector_injective pp h2
  cases Wire.groupVector_injective pp h0
  cases Wire.groupVector_injective pp hc1
  cases Wire.groupVector_injective pp hc2
  rfl

/-- The complete private simulator-state word is injective on all typed states.

**Proof sketch.** Separate the fixed number of nested blocks. Read the public group matrix
and all scalar matrices/vectors with their public encoding injectivity theorems. The scalar
modulus fits the source-group width by the existing order-width theorem. -/
theorem quadraticSimState_injective (G : BilinearGen) (ℓ n : PolyExpr) (κ : ℕ)
    (pp : PublicParams G κ) :
    Function.Injective (Wire.quadraticSimState pp (ℓ := ℓ) (n1 := n) (n2 := n)) := sorry
