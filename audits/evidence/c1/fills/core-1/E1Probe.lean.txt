namespace Turing

/-- The maintainer's #122 equation, declared locally for the probe (identical statement). -/
theorem counterValue_eq (w : List Bool) :
    counterValue w = Nat.ofDigits 2 (w.map Bool.toNat) :=
  rfl

/-- #122's second equation, likewise. -/
theorem codeBitsNat_eq (xs : List Bool) : codeBitsNat xs = xs.foldr Nat.bit 0 := rfl

-- (1) The loop's `ht` for the counter `bits t`, through the published equation only.
example (t : ℕ) : Nat.ofDigits 2 ((Nat.bits t).map Bool.toNat) = t := by
  rw [← counterValue_eq]
  exact counterValue_bits t

-- (1') the same as a term
example (t : ℕ) : Nat.ofDigits 2 ((Nat.bits t).map Bool.toNat) = t :=
  (counterValue_eq _).symm.trans (counterValue_bits t)

-- (2) the window copier's `y` for the counter `bits t` (its statement also reads `Nat.ofDigits`)
example (t p : ℕ) (z : List Bool) :
    List.take (Nat.ofDigits 2 ((Nat.bits t).map Bool.toNat)) (List.drop p z) =
      List.take t (List.drop p z) := by
  rw [← counterValue_eq, counterValue_bits]

-- (3) the alternative route through `codeBitsNat_eq` (no `counterValue` at all)
example (t : ℕ) : Nat.ofDigits 2 ((Nat.bits t).map Bool.toNat) = t := by
  rw [← foldr_bit_eq_ofDigits, ← codeBitsNat_eq]
  exact codeBitsNat_bits t

end Turing

-- (4) At the base, without the local equation, the defeq route is what H1 already uses
-- (CodeSimulator.lean:5103 passes `counterValue_bits t` to `ht`): it typechecks only by unfolding
-- the `def` `counterValue` during unification.
example (t : ℕ) : Nat.ofDigits 2 ((Nat.bits t).map Bool.toNat) = t := Turing.counterValue_bits t
