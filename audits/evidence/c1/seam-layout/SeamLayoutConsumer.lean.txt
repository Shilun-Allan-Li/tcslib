/-- Stands for #122's `Turing.FinTM.bufferTape_of_neg` (not in this probe's snapshot). -/
theorem FinTM.bufferTape_of_neg' {Symbol : Type*} (w : List Symbol) {z : ℤ} (hz : z < 0) :
    FinTM.bufferTape w z = none := by
  simp [FinTM.bufferTape, show ¬ (0 ≤ z) by omega]

theorem exists_seamLayoutTM (k : ℕ) (src tbl sm : Fin k) (hst : src ≠ tbl) (hss : src ≠ sm)
    (hts : tbl ≠ sm) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM k Bool Q)
      (entry done : Q) (C : ℕ),
      ∀ {z : List Bool} (d : Cfg k Bool Q z) (a v : List Bool) (q : ℕ),
        let s := pairEncode a ((List.replicate q true ++ [false]) ++ v)
        d.state = some entry →
        d.workTapes src = FinTM.bufferTape s →
        0 ≤ d.workTapePos src → d.workTapePos src ≤ (s.length : ℤ) →
        d.workTapes tbl = (fun _ => none) → d.workTapePos tbl = 0 →
        d.workTapes sm = (fun _ => none) → d.workTapePos sm = 0 →
        ∃ T ≤ C * (s.length + 1),
          (L.runFrom d T).state = some done ∧
          (L.runFrom d T).inputPos = d.inputPos ∧
          (L.runFrom d T).workTapes =
            Function.update (Function.update (Function.update d.workTapes src (fun _ => none))
              tbl (FinTM.bufferTape v))
              sm (fun p => FinTM.bufferTape (List.replicate q true ++ [false]) (p - 2)) ∧
          (L.runFrom d T).workTapePos =
            Function.update (Function.update (Function.update d.workTapePos src 0) tbl 0) sm 0 ∧
          (∀ u ≤ T, (L.runFrom d u).output = d.output) ∧
          ∀ u < T, (L.runFrom d u).state ≠ some done := by
  sorry


/-- The start word is the serialization. -/
example {k n : ℕ} (M : MultiTapeTM k Bool (Fin (n + 1))) :
    serializeK M = pairEncode n.bits ((List.replicate (M.q₀ : ℕ) true ++ [false]) ++
      tableBitsK M.tr) := by
  rw [serializeK_eq, unaryFin_eq]

/-- The summary tape at the end is the empty-output summary of the initial state. -/
example {n : ℕ} (q₀ : Fin (n + 1)) :
    (fun p => FinTM.bufferTape (List.replicate (q₀ : ℕ) true ++ [false]) (p - 2)) =
      StepBlock.summaryCells (StepBlock.outputSummary []) (unaryFin q₀) := by
  funext p
  rw [StepBlock.summaryCells_eq, StepBlock.outputSummary_nil, unaryFin_eq]
  split_ifs with hp
  · subst hp
    exact FinTM.bufferTape_of_neg' _ (by norm_num)
  · rfl

end Turing
