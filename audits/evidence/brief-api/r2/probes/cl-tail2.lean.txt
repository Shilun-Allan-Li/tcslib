/-! Review probe (seed-inclusive inventory): can a consumer reach `roundOrbit_zero`/`_succ`
through the abbreviation `counterLoopOrbit` without unfolding it? -/
section OrbitProbe
open Turing
variable {k : ℕ} {Γ S : Type} {x : List Γ}

example (M : MultiTapeTM k Γ S) (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) (r : ℕ) :
    counterLoopOrbit M τ c (r + 1) =
      M.runFrom (counterLoopOrbit M τ c r) (τ (counterLoopOrbit M τ c r)) := by
  fail_if_success rw [roundOrbit_succ]
  exact roundOrbit_succ M τ c r

example (M : MultiTapeTM k Γ S) (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) (r : ℕ) :
    counterLoopOrbit M τ c (r + 1) =
      M.runFrom (counterLoopOrbit M τ c r) (τ (counterLoopOrbit M τ c r)) := by
  simp only [roundOrbit_succ]

example (M : MultiTapeTM k Γ S) (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) :
    counterLoopOrbit M τ c 0 = c :=
  roundOrbit_zero M τ c

end OrbitProbe
