
open Lean Elab Command in
/-- Maintainer pre-ship inventory, per constant (read-only): for each named constant, the
definitions of `TCSlib` modules occurring in its type, one line `INV <constant> -> <def> @ <module>`. -/
elab "#def_inventory_each " ids:ident* : command => do
  let env ← getEnv
  let mut out : Array String := #[]
  for id in ids do
    let some ci := env.find? id.getId | throwError "unknown constant {id.getId}"
    for c in ci.type.getUsedConstants do
      match env.find? c with
      | some (.defnInfo _) =>
        let modName := match env.getModuleIdxFor? c with
          | some idx => env.header.moduleNames[idx.toNat]!.toString
          | none => "(this file)"
        if (modName.startsWith "TCSlib" || modName == "(this file)") &&
            !(Lean.Meta.isInstanceCore env c) && !c.isInternal then
          out := out.push s!"INV {id.getId} -> {c} @ {modName}"
      | _ => pure ()
  logInfo m!"{String.intercalate "\n" out.toList}"
