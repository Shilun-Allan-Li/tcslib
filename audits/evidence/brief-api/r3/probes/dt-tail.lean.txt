section R3Consumer
open Turing
variable {k : ℕ} {Γ S : Type} {x : List Γ}
attribute [local irreducible] Turing.MultiTapeTM.step Turing.MultiTapeTM.outputSymbol in
example (M : MultiTapeTM k Γ S) (c : Cfg k Γ S x) (q : S) (hq : c.state = some q) :
    M.step c = (M.tr q c.inputSymbol c.workTapeSymbols).apply c ∧
      M.outputSymbol c = (M.tr q c.inputSymbol c.workTapeSymbols).output ∧
      (M.runFrom c 1) = (M.tr q c.inputSymbol c.workTapeSymbols).apply c :=
  ⟨MultiTapeTM.step_of_some hq, MultiTapeTM.outputSymbol_of_some hq, by
    rw [MultiTapeTM.runFrom_succ_eq_step', MultiTapeTM.runFrom_zero, MultiTapeTM.step_of_some hq]⟩
end R3Consumer
