/-- **Gate evidence, not landing: the retired existential, recovered up to its exact-length
clause.** The draft's corollary `exists_adjacency_codec_cnf_of_closedForm`, restated over the
landing names: the theorems of `ConfigCodec` give, for every machine, a constant `C` and for all
`(n, s)` a codec of exact width `ℓ ≤ C · (s + n + 1)` with every other clause of
`Complexity.exists_adjacency_codec_cnf` verbatim. The only clause not recovered is the old
"length exactly `C · (s + n + 1)`", which needed padding. -/
theorem rd2Evidence_oldShape (M : FinTM Bool) :
    ∃ C : ℕ, 0 < C ∧ ∀ (n s : ℕ),
      ∃ (ℓ : ℕ) (code : (x : List Bool) → Cfg M.k Bool M.State x → List Bool)
        (φv φa φacc : CNF ℕ),
        ℓ ≤ C * (s + n + 1) ∧
        (∀ (x : List Bool) (c : Cfg M.k Bool M.State x), (code x c).length = ℓ) ∧
        (∀ (x x' : List Bool), x.length = n → x'.length = n →
          ∀ (c : Cfg M.k Bool M.State x) (c' : Cfg M.k Bool M.State x'),
            c.InWindow s → c'.InWindow s → code x c = code x' c' → x = x') ∧
        (∀ (x : List Bool), x.length = n →
          ∀ c d : Cfg M.k Bool M.State x, c.InWindow s → d.InWindow s →
            (code x c = code x d ↔ NDTM.coreSum c = NDTM.coreSum d)) ∧
        (∀ w : List Bool, w.length = ℓ →
          (φv.eval (fun v => w.getD v false) = true ↔
            ∃ (x : List Bool), x.length = n ∧
              ∃ c : Cfg M.k Bool M.State x, c.InWindow s ∧ w = code x c)) ∧
        (∀ (x : List Bool), x.length = n →
          ∀ c d : Cfg M.k Bool M.State x, c.InWindow s → d.InWindow s →
            (φa.eval (fun v => (code x c ++ code x d).getD v false) = true ↔
              NDTM.coreSum (M.tm.step c) = NDTM.coreSum d)) ∧
        (∀ (x x' : List Bool), x.length = n → x'.length = n → x ≠ x' →
          ∀ (c : Cfg M.k Bool M.State x) (c' : Cfg M.k Bool M.State x'),
            c.InWindow s → c'.InWindow s →
            φa.eval (fun v => (code x c ++ code x' c').getD v false) = false) ∧
        (∀ (x : List Bool), x.length = n →
          ∀ c : Cfg M.k Bool M.State x, c.InWindow s →
            (φacc.eval (fun v => (code x c).getD v false) = true ↔
              (c.state = none ∧ outSummary c.output = OutSummary.accept))) ∧
        φv.numVars ≤ ℓ ∧ φacc.numVars ≤ ℓ ∧ φa.numVars ≤ 2 * ℓ ∧
        (CNF.serialize φv).length ≤ C * (s + n + 2) ^ C ∧
        (CNF.serialize φa).length ≤ C * (s + n + 2) ^ C ∧
        (CNF.serialize φacc).length ≤ C * (s + n + 2) ^ C := by
  obtain ⟨K, hK⟩ := ConfigCodec.exists_serialize_length_le M
  let C := max (Fintype.card M.State + 6 * M.k + 5) K
  have hC : 0 < C := lt_of_lt_of_le (by omega) (le_max_left _ _)
  have hpow : ∀ n s : ℕ, K * (s + n + 2) ^ K ≤ C * (s + n + 2) ^ C := fun n s =>
    Nat.mul_le_mul (le_max_right _ _) (Nat.pow_le_pow_right (by omega) (le_max_right _ _))
  refine ⟨C, hC, fun n s => ⟨ConfigCodec.width M n s, ConfigCodec.code M n s,
    ConfigCodec.validity M n s, ConfigCodec.adjacency M n s, ConfigCodec.acceptance M n s,
    (ConfigCodec.width_le M n s).trans (Nat.mul_le_mul_right _ (le_max_left _ _)),
    ConfigCodec.code_length M n s,
    fun x x' hx hx' c c' _ _ h => ConfigCodec.code_input_eq M n s x x' hx hx' c c' h,
    fun x hx c d hc hd => ConfigCodec.code_eq_iff M n s x hx c d hc hd,
    ConfigCodec.validity_eval_iff M n s,
    fun x hx c d hc hd => ConfigCodec.adjacency_eval_iff M n s x hx c d hc hd,
    fun x x' hx hx' hne c c' _ _ => ConfigCodec.adjacency_eval_of_ne M n s x x' hx hx' hne c c',
    fun x _ c _ => ConfigCodec.acceptance_eval_iff M n s x c,
    ConfigCodec.validity_numVars_le M n s, ConfigCodec.acceptance_numVars_le M n s,
    ConfigCodec.adjacency_numVars_le M n s,
    (hK n s).1.trans (hpow n s), (hK n s).2.1.trans (hpow n s),
    (hK n s).2.2.trans (hpow n s)⟩⟩

/-- **Gate evidence, not landing: the old-length uniformity instance.** The uniformity
theorems at input length `x.length` (the draft's form) are the instance `n := List.length`. -/
theorem rd2Evidence_validity_at_length (M : FinTM Bool) (s o : List Bool → ℕ)
    (hs : PolyTimeComputable fun x => List.replicate (s x) true)
    (ho : PolyTimeComputable fun x => List.replicate (o x) true) :
    PolyTimeComputable fun x =>
      CNF.serialize ((ConfigCodec.validity M x.length (s x)).relabel (o x + ·)) :=
  ConfigCodec.polyTimeComputable_serialize_validity M List.length s o polyTimeComputable_unary hs ho

/-- **Gate evidence, not landing: the codec's gate kinds are a finite, decidable type**, as
`Complexity.FormulaKit.polyTimeComputable_fragment_gate_select` requires. -/
example (M : FinTM Bool) : Fintype (ConfigCodec.GateKind M) := inferInstance

/-- Gate evidence, not landing: decidable equality of the gate kinds. -/
example (M : FinTM Bool) : DecidableEq (ConfigCodec.GateKind M) := inferInstance

/-- **Gate evidence, not landing: a transition gate is literally a snapshot-keyed kit gate**,
the predicate family `κ ↦ kindPred M κ ∘ readBit` at the kind `(q, interior p, expectOf …)`,
wired by `portWire ∘ portOf`. -/
example (M : FinTM Bool) (n s p h : ℕ) (q : ConfigCodec.Case M) (v : ℕ) :
    ConfigCodec.transitionGate M n s p h q v =
      FormulaKit.gate (ConfigCodec.portCount M.k)
        ((fun κ a => ConfigCodec.kindPred M κ
            (fun port => FormulaKit.readBit a (ConfigCodec.portIndex port)))
          (q, decide (0 < p ∧ p ≤ n), ConfigCodec.expectOf M n s p h q v))
        (fun j => ConfigCodec.portWire M n s p h q v (ConfigCodec.portOf M.k j)) := rfl

end Complexity
