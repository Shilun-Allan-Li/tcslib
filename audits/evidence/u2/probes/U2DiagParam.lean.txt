/-! ## 1. The generic layer -/

/-- The diagonal language of an arbitrary simulator `W`. -/
def diagLangOf (W : FinTM Bool) (g : ℕ → ℕ) : Language Bool :=
  {x | ¬W.ComputesInTime x [true] (g x.length)}

/-- The received diagonal language is the instance at `diagSim`, definitionally. -/
theorem diagLang_eq_diagLangOf (g : ℕ → ℕ) : diagLang g = diagLangOf diagSim g := rfl

/-- The upper bound, for every simulator (the received proof with `diagSim` replaced by `W`). -/
theorem diagLangOf_mem_DTIME (W : FinTM Bool) {g : ℕ → ℕ} (hg : TimeConstructible g) :
    diagLangOf W g ∈ DTIME (fun n => g n + 1) := by
  classical
  obtain ⟨hgn, cK, _, K, hK⟩ := hg
  refine ⟨cK + 15, clockTM K W, fun x => ?_⟩
  have h := clockTM_spec K W x (g x.length).bits _ (hK x)
  rw [ctrVal_bits] at h
  have hout : [!(decide (W.ComputesInTime x [true] (g x.length)))] =
      [MultiTapeTM.indicator (diagLangOf W g : Set (List Bool)) x] := by
    by_cases hx : W.ComputesInTime x [true] (g x.length)
    · have : x ∉ (diagLangOf W g : Set (List Bool)) := fun h' => h' hx
      simp [MultiTapeTM.indicator, hx, this]
    · have : x ∈ (diagLangOf W g : Set (List Bool)) := hx
      simp [MultiTapeTM.indicator, hx, this]
  rw [hout] at h
  apply h.mono
  have h1 := length_bits_le_self (g x.length)
  have h2 := hgn x.length
  nlinarith

/-- **A diagonal simulator with overhead `Φ`**: one machine `W` that reproduces, on some
input of every sufficiently large length `n`, the verdict of every `DTIME T` language within
`A · Φ (T n) n` steps, `A` depending on the language. The clock is not part of it: the
clocked runner `clockTM` supplies it for every `W`. -/
structure DiagSimulator (Φ : ℕ → ℕ → ℕ) where
  /-- the simulating machine -/
  W : FinTM Bool
  /-- padded simulation of every `DTIME T` language -/
  simulates : ∀ (T : ℕ → ℕ) (L : Language Bool), L ∈ DTIME T →
    ∃ A N₀ : ℕ, ∀ n ≥ N₀, ∃ x : List Bool, x.length = n ∧
      W.ComputesInTime x [MultiTapeTM.indicator (L : Set (List Bool)) x] (A * Φ (T n) n)

namespace DiagSimulator

variable {Φ : ℕ → ℕ → ℕ}

/-- Weakening the overhead up to a constant factor. -/
def mono (S : DiagSimulator Φ) {Φ' : ℕ → ℕ → ℕ} (K : ℕ) (h : ∀ t n, Φ t n ≤ K * Φ' t n) :
    DiagSimulator Φ' where
  W := S.W
  simulates T L hL := by
    obtain ⟨A, N₀, hA⟩ := S.simulates T L hL
    refine ⟨A * K, N₀, fun n hn => ?_⟩
    obtain ⟨x, hx, hW⟩ := hA n hn
    exact ⟨x, hx, hW.mono (by rw [Nat.mul_assoc]; exact Nat.mul_le_mul_left A (h _ _))⟩

/-- The lower bound, for every diagonal simulator (the received contradiction step). -/
theorem diagLang_not_mem_DTIME (S : DiagSimulator Φ) {g T : ℕ → ℕ}
    (hT : ∀ A N : ℕ, ∃ n ≥ N, A * Φ (T n) n ≤ g n) :
    diagLangOf S.W g ∉ DTIME T := by
  classical
  intro hL
  obtain ⟨A, N₀, hA⟩ := S.simulates T _ hL
  obtain ⟨n, hn, hgn⟩ := hT A N₀
  obtain ⟨x, hx, hW⟩ := hA n hn
  let χ := MultiTapeTM.indicator (diagLangOf S.W g : Set (List Bool))
  have hD : S.W.ComputesInTime x [χ x] (g x.length) := hW.mono (by rw [hx]; exact hgn)
  by_cases hmem : x ∈ (diagLangOf S.W g : Set (List Bool))
  · have hχ : χ x = true := by simp [χ, MultiTapeTM.indicator, hmem]
    rw [hχ] at hD
    exact hmem hD
  · have hχ : χ x = false := by simp [χ, MultiTapeTM.indicator, hmem]
    rw [hχ] at hD
    have htrue : S.W.ComputesInTime x [true] (g x.length) := by
      by_contra hc
      exact hmem hc
    have := htrue.output_unique hD
    simp at this

/-- **The hierarchy theorem of a diagonal simulator** (the received `time_hierarchy`, with
`(t + n + 1)²` replaced by `Φ t n`). -/
theorem time_hierarchy (S : DiagSimulator Φ) (hΦ : ∀ t n, t ≤ Φ t n)
    {f g : ℕ → ℕ} (hg : TimeConstructible g)
    (hfg : ∀ A : ℕ, ∃ N, ∀ n ≥ N, A * Φ (f n) n ≤ g n) :
    DTIME f ⊂ DTIME (fun n => g n + 1) := by
  refine ⟨?_, fun hsub => S.diagLang_not_mem_DTIME (g := g) (T := f) ?_
    (hsub (diagLangOf_mem_DTIME S.W hg))⟩
  · obtain ⟨N, hN⟩ := hfg 1
    let F := ∑ i ∈ Finset.range N, f i
    have hle : ∀ n, f n ≤ (F + 1) * (g n + 1) := by
      intro n
      by_cases hn : n < N
      · have : f n ≤ F := Finset.single_le_sum (f := f) (fun _ _ => Nat.zero_le _)
          (Finset.mem_range.mpr hn)
        nlinarith
      · have h := hN n (by omega)
        have h' := hΦ (f n) n
        have : f n ≤ g n := by omega
        nlinarith
    rintro L ⟨c, M, hM⟩
    exact ⟨c * (F + 1), M, fun x => (hM x).mono (by
      simp only [Nat.mul_assoc]; exact Nat.mul_le_mul_left c (hle x.length))⟩
  · intro A N₀
    obtain ⟨N, hN⟩ := hfg A
    exact ⟨max N N₀, le_max_right _ _, hN _ (le_max_left _ _)⟩

/-- The positive-bound form (the received `time_hierarchy_of_pos`, generically). -/
theorem time_hierarchy_of_pos (S : DiagSimulator Φ) (hΦ : ∀ t n, t ≤ Φ t n)
    {f g : ℕ → ℕ} (hg : TimeConstructible g) (hpos : ∀ n, 0 < g n)
    (hfg : ∀ A : ℕ, ∃ N, ∀ n ≥ N, A * Φ (f n) n ≤ g n) :
    DTIME f ⊂ DTIME g := by
  have heq : DTIME (fun n => g n + 1) = DTIME g := by
    apply Set.Subset.antisymm
    · rintro L ⟨c, M, hM⟩
      refine ⟨2 * c, M, fun x => (hM x).mono ?_⟩
      have := hpos x.length
      dsimp only
      nlinarith
    · exact DTIME.mono (fun n => Nat.le_succ _)
  rw [← heq]
  exact S.time_hierarchy hΦ hg hfg

end DiagSimulator

/-- **A self-simulator with normal-form overhead `Ψ`**: the universal-machine-level
hypothesis. A machine `W` and a code family `decode` such that
* `W` reproduces the one-bit verdict of the machine coded by `α` on `pairEncode α w`
  **itself** (self-application) within `C_α · (t + |input| + 1)`, the constant depending on
  the code only (per-code linear overhead; no uniformity, no clock, no output forwarding
  beyond the verdict);
* every `DTIME T` language has a code whose machine decides it within `c · (Ψ (T n) + 1)`
  (the normal form: `Ψ = (t + 1)²` for one work tape, `(t + 1)(log₂(t + 1) + 1)` for two). -/
structure SelfSimulator (Ψ : ℕ → ℕ) where
  /-- the self-application simulator -/
  W : FinTM Bool
  /-- the code family it simulates -/
  decode : List Bool → FinTM Bool
  /-- per-code linear self-simulation of one-bit verdicts -/
  simulates : ∀ α : List Bool, ∃ C : ℕ, ∀ (w : List Bool) (b : Bool) (t : ℕ),
    (decode α).ComputesInTime (pairEncode α w) [b] t →
    W.ComputesInTime (pairEncode α w) [b] (C * (t + (pairEncode α w).length + 1))
  /-- the normal form: every `DTIME T` language has a coded decider within `c · (Ψ T + 1)` -/
  normalForm : ∀ (T : ℕ → ℕ) (L : Language Bool), L ∈ DTIME T →
    ∃ (α : List Bool) (c : ℕ), ∀ x : List Bool,
      (decode α).ComputesInTime x [MultiTapeTM.indicator (L : Set (List Bool)) x]
        (c * (Ψ (T x.length) + 1))

/-- **Input padding**: a self-simulator is a diagonal simulator with overhead `Ψ t + n + 1`.
Fix the code `α` of a decider and run on `pairEncode α 1^m`, of every length `n ≥ 2|α| + 2`. -/
def SelfSimulator.toDiagSimulator {Ψ : ℕ → ℕ} (S : SelfSimulator Ψ) :
    DiagSimulator (fun t n => Ψ t + n + 1) where
  W := S.W
  simulates T L hL := by
    obtain ⟨α, c, hc⟩ := S.normalForm T L hL
    obtain ⟨C, hC⟩ := S.simulates α
    refine ⟨C * (c + 1), 2 * α.length + 2, fun n hn => ?_⟩
    have hx : (pairEncode α (List.replicate (n - (2 * α.length + 2)) true)).length = n := by
      simp only [length_pairEncode, List.length_replicate]; omega
    refine ⟨pairEncode α (List.replicate (n - (2 * α.length + 2)) true), hx, ?_⟩
    have h := hC _ _ _ (hc (pairEncode α (List.replicate (n - (2 * α.length + 2)) true)))
    rw [hx] at h
    apply h.mono
    have hle : c * (Ψ (T n) + 1) + n + 1 ≤ (c + 1) * (Ψ (T n) + n + 1) := by nlinarith
    calc C * (c * (Ψ (T n) + 1) + n + 1) ≤ C * ((c + 1) * (Ψ (T n) + n + 1)) :=
          Nat.mul_le_mul_left C hle
      _ = C * (c + 1) * (Ψ (T n) + n + 1) := by ring

/-! ## 2. The received `f²` theorem as an instance -/

/-- **The one-tape instance**: `diagSim` (= `preTM` then `univTM`) self-simulates the codes of
`code`, and the one-work-tape normal form costs a square. -/
noncomputable def oneTapeSelfSimulator : SelfSimulator (fun t => (t + 1) ^ 2) where
  W := diagSim
  decode α := (code.decode α).toFinTM
  simulates α := by
    obtain ⟨C, hC⟩ := univTM_spec α
    refine ⟨C + 7, fun w b t h => ?_⟩
    have hU := (hC (pairEncode α w)).1 [b] t h
    have hP := preTM_computes (pairEncode α w)
    rw [scanPre_pairEncode_append α w] at hP
    have hD := bufferedCompTM_computesInTime preTM univTM hP hU
    apply hD.mono
    have h1 : (pairEncode α (pairEncode α w)).length ≤ 2 * (pairEncode α w).length := by
      simp only [length_pairEncode]; omega
    nlinarith
  normalForm T L := by
    rintro ⟨c₀, M, hM⟩
    obtain ⟨M₁, c₁, hk, h₁⟩ := one_work_tape_binary M
      (fun x => [MultiTapeTM.indicator (L : Set (List Bool)) x]) (fun n => c₀ * T n) hM
    obtain ⟨N, hN⟩ := exists_codeTM M₁ hk
    refine ⟨code.encode N, c₁ * (c₀ + 1) ^ 2, fun x => ?_⟩
    rw [show code.decode (code.encode N) = N from code.toMachineCode.decode_encode N]
    apply ((hN x _ _).2 (h₁ x)).mono
    have hlin : c₀ * T x.length + 1 ≤ (c₀ + 1) * (T x.length + 1) := by nlinarith
    have hsq : (c₀ * T x.length + 1) ^ 2 ≤ (c₀ + 1) ^ 2 * (T x.length + 1) ^ 2 := by
      rw [← mul_pow]; exact Nat.pow_le_pow_left hlin 2
    have := Nat.mul_le_mul_left c₁ hsq
    nlinarith

/-- The received overhead `(t + n + 1)²`, from padding, up to the factor `2`. -/
noncomputable def sqDiagSimulator : DiagSimulator (fun t n => (t + n + 1) ^ 2) :=
  oneTapeSelfSimulator.toDiagSimulator.mono 2 (fun t n => by
    show (t + 1) ^ 2 + n + 1 ≤ 2 * (t + n + 1) ^ 2
    have h1 : (t + 1) ^ 2 ≤ (t + n + 1) ^ 2 := Nat.pow_le_pow_left (by omega) 2
    have h2 : t + n + 1 ≤ (t + n + 1) ^ 2 := Nat.le_self_pow (by norm_num) _
    omega)

/-- The received overhead dominates the time: `t ≤ (t + n + 1)²`. -/
theorem le_sq_overhead (t n : ℕ) : t ≤ (t + n + 1) ^ 2 := by
  have h : t + n + 1 ≤ (t + n + 1) ^ 2 := Nat.le_self_pow (by norm_num) _
  omega

/-- The simulator of the instance is the received `diagSim`, definitionally. -/
theorem sqDiagSimulator_W : sqDiagSimulator.W = diagSim := rfl

/-! ## 4. The `f log f` layer, from the U2 + G5 deliverable as a hypothesis -/

/-- The Hennie–Stearns normal-form overhead `(t + 1)(log₂(t + 1) + 1)`. -/
def logOverhead (t : ℕ) : ℕ := (t + 1) * (Nat.log 2 (t + 1) + 1)

/-- The book-strength overhead in the received shape: `(t + n + 1)(log₂(t + n + 1) + 1)`,
the `f log f` analogue of `(t + n + 1)²`. -/
def logDiagOverhead (t n : ℕ) : ℕ := (t + n + 1) * (Nat.log 2 (t + n + 1) + 1)

theorem le_logDiagOverhead (t n : ℕ) : t ≤ logDiagOverhead t n := by
  unfold logDiagOverhead; nlinarith

theorem logOverhead_add_le (t n : ℕ) : logOverhead t + n + 1 ≤ 2 * logDiagOverhead t n := by
  unfold logOverhead logDiagOverhead
  have hlog : Nat.log 2 (t + 1) ≤ Nat.log 2 (t + n + 1) := Nat.log_mono_right (by omega)
  have h1 : (t + 1) * (Nat.log 2 (t + 1) + 1) ≤ (t + n + 1) * (Nat.log 2 (t + n + 1) + 1) :=
    Nat.mul_le_mul (by omega) (by omega)
  nlinarith

/-- **The time hierarchy at `f log f`** (the corollary the parameterization exists for), from a
two-tape self-simulator with the Hennie–Stearns normal form. -/
theorem time_hierarchy_log (S : SelfSimulator logOverhead) {f g : ℕ → ℕ}
    (hg : TimeConstructible g)
    (hfg : ∀ A : ℕ, ∃ N, ∀ n ≥ N,
      A * ((f n + n + 1) * (Nat.log 2 (f n + n + 1) + 1)) ≤ g n) :
    DTIME f ⊂ DTIME (fun n => g n + 1) :=
  (S.toDiagSimulator.mono 2 logOverhead_add_le).time_hierarchy le_logDiagOverhead hg hfg

/-- The arithmetic of the book form: for `2 ≤ n ≤ f`,
`(f + n + 1)(log₂(f + n + 1) + 1) ≤ 12 · f · log₂ f`. -/
theorem logDiagOverhead_le_book {f n : ℕ} (hn : 2 ≤ n) (hf : n ≤ f) :
    (f + n + 1) * (Nat.log 2 (f + n + 1) + 1) ≤ 12 * (f * Nat.log 2 f) := by
  have hf2 : 2 ≤ f := le_trans hn hf
  have hlog1 : 1 ≤ Nat.log 2 f := Nat.le_log_of_pow_le (by norm_num) (by simpa using hf2)
  have h3 : f + n + 1 ≤ f * 2 * 2 := by omega
  have hlog : Nat.log 2 (f + n + 1) ≤ Nat.log 2 f + 2 := by
    calc Nat.log 2 (f + n + 1) ≤ Nat.log 2 (f * 2 * 2) := Nat.log_mono_right h3
      _ = Nat.log 2 f + 2 := by
        rw [Nat.log_mul_base (by norm_num) (by omega),
          Nat.log_mul_base (by norm_num) (by omega)]
  have hA : f + n + 1 ≤ 3 * f := by omega
  have hB : Nat.log 2 (f + n + 1) + 1 ≤ 4 * Nat.log 2 f := by omega
  calc (f + n + 1) * (Nat.log 2 (f + n + 1) + 1) ≤ (3 * f) * (4 * Nat.log 2 f) :=
        Nat.mul_le_mul hA hB
    _ = 12 * (f * Nat.log 2 f) := by ring

/-- **The book form** [AB09, Theorem 3.1]: if `n ≤ f(n)` (true for time-constructible `f`) and
`f(n) log₂ f(n) = o(g(n))`, then `DTIME(f) ⊊ DTIME(g + 1)`. -/
theorem time_hierarchy_book (S : SelfSimulator logOverhead) {f g : ℕ → ℕ}
    (hg : TimeConstructible g) (hf : ∀ n, n ≤ f n)
    (hfg : ∀ A : ℕ, ∃ N, ∀ n ≥ N, A * (f n * Nat.log 2 (f n)) ≤ g n) :
    DTIME f ⊂ DTIME (fun n => g n + 1) := by
  refine time_hierarchy_log S hg fun A => ?_
  obtain ⟨N, hN⟩ := hfg (12 * A)
  refine ⟨max N 2, fun n hn => ?_⟩
  have h := hN n (le_of_max_le_left hn)
  have hb := logDiagOverhead_le_book (le_of_max_le_right hn) (hf n)
  calc A * ((f n + n + 1) * (Nat.log 2 (f n + n + 1) + 1)) ≤ A * (12 * (f n * Nat.log 2 (f n))) :=
        Nat.mul_le_mul_left A hb
    _ = 12 * A * (f n * Nat.log 2 (f n)) := by ring
    _ ≤ g n := h

/-- Logarithms are eventually below every linear function: `A (log₂ n + 1) ≤ n` eventually. -/
theorem eventually_mul_log_le (A : ℕ) : ∃ N, ∀ n ≥ N, A * (Nat.log 2 n + 1) ≤ n := by
  obtain ⟨M, hM⟩ := eventually_poly_le_two_pow A 1
  refine ⟨2 ^ M, fun n hn => ?_⟩
  have hn0 : n ≠ 0 := by have := Nat.one_le_two_pow (n := M); omega
  have hlogM : M ≤ Nat.log 2 n := Nat.le_log_of_pow_le (by norm_num) hn
  have h := hM (Nat.log 2 n) hlogM
  simp only [pow_one] at h
  exact h.trans (Nat.pow_log_le_self 2 hn0)

/-- **Below the quadratic threshold** (not derivable from the received `f²` theorem):
`DTIME(n + 1) ⊊ DTIME((n + 1)²)`. -/
theorem dtime_linear_ssubset_dtime_sq (S : SelfSimulator logOverhead) :
    DTIME (fun n => n + 1) ⊂ DTIME (fun n => (n + 1) ^ 2) := by
  have hg : TimeConstructible fun n => (n + 1) ^ 2 := by
    simpa using timeConstructible_poly 1 1 Nat.one_pos
  refine (S.toDiagSimulator.mono 2 logOverhead_add_le).time_hierarchy_of_pos
    le_logDiagOverhead hg (fun n => by positivity) fun A => ?_
  obtain ⟨N, hN⟩ := eventually_mul_log_le (4 * A)
  refine ⟨N, fun n hn => ?_⟩
  show A * ((n + 1 + n + 1) * (Nat.log 2 (n + 1 + n + 1) + 1)) ≤ (n + 1) ^ 2
  have h := hN (n + 1 + n + 1) (by omega)
  have h' : 2 * A * (Nat.log 2 (n + 1 + n + 1) + 1) ≤ n + 1 := by
    have : 4 * A * (Nat.log 2 (n + 1 + n + 1) + 1) =
        2 * (2 * A * (Nat.log 2 (n + 1 + n + 1) + 1)) := by ring
    omega
  calc A * ((n + 1 + n + 1) * (Nat.log 2 (n + 1 + n + 1) + 1))
      = (2 * A * (Nat.log 2 (n + 1 + n + 1) + 1)) * (n + 1) := by ring
    _ ≤ (n + 1) * (n + 1) := Nat.mul_le_mul_right _ h'
    _ = (n + 1) ^ 2 := by ring

end Complexity.TimeHierarchy.U2Probe

/-! ## 5. The chain: `k`-tape schemes, G5, U2 (statements sorried), and the assembly -/

namespace Turing.U2Probe

open Turing Turing.FinTM

/-- **An effective scheme for `k`-work-tape codes** (proposed U2-S declaration): a total decoding
into `k`-work-tape binary machines and a machine computing the fixed `k`-record serialization
`serializeK` of the decoded machine, in arbitrary time. -/
structure EffectiveCodeK (k : ℕ) where
  /-- one less than the decoded machine's state count -/
  numStates : List Bool → ℕ
  /-- the decoded machine -/
  machine : (α : List Bool) → MultiTapeTM k Bool (Fin (numStates α + 1))
  /-- the canonizer -/
  canonizer : FinTM Bool
  /-- its arbitrary time bound -/
  canonizerTime : ℕ → ℕ
  /-- it computes the fixed serialization of the decoded machine -/
  canonizer_computes :
    canonizer.ComputesFunInTime (fun α => serializeK (machine α)) canonizerTime

/-- The bundled decoded machine. -/
def EffectiveCodeK.toFinTM {k : ℕ} (E : EffectiveCodeK k) (α : List Bool) : FinTM Bool where
  k := k
  State := Fin (E.numStates α + 1)
  tm := E.machine α

/-- The one-tape serialization is `serializeK` (public twin of `CodeSimulator`'s private
`oneTape_serialize_eq`). -/

/-- **The two-tape coding step** (proposed additive export of `Codes2Tape`; the mirror of
`Turing.exists_codeTM`, proved here by its proof). -/
theorem exists_code2TM (M : FinTM Bool) (hk : M.k = 2) :
    ∃ M' : Code2TM, ∀ (x output : List Bool) (t : ℕ),
      M'.toFinTM.ComputesInTime x output t ↔ M.ComputesInTime x output t := by
  classical
  rcases M with @⟨k, Q, hQ, dQ, tm⟩
  dsimp only at hk
  subst k
  letI : Fintype Q := hQ
  letI : DecidableEq Q := dQ
  have hcard : Fintype.card Q = (Fintype.card Q - 1) + 1 := by
    have : 0 < Fintype.card Q := Fintype.card_pos_iff.mpr ⟨tm.q₀⟩
    omega
  let e := Fintype.equivFinOfCardEq hcard
  refine ⟨⟨Fintype.card Q - 1, tm.relabelState e⟩, ?_⟩
  intro x output t
  simp only [Code2TM.toFinTM, FinTM.ComputesInTime, MultiTapeTM.ComputesInTimeAndSpace,
    MultiTapeTM.relabelState_runFrom_init, Cfg.mapState, Option.map_eq_none_iff]
  constructor
  · rintro ⟨s, hhalt, hout, -⟩
    exact ⟨_, hhalt, hout, rfl⟩
  · rintro ⟨s, hhalt, hout, -⟩
    exact ⟨_, hhalt, hout, rfl⟩

/-- **G5's interface, per run** (sorried: G5's statement phase). Hennie–Stearns `k → 2`:
every binary machine has a two-work-tape binary equivalent that reproduces each halting run of
`t` steps within `c · (t + 1)(log₂(t + 1) + 1)` steps, and whose completed outputs are the
source's. -/
theorem two_work_tapes_run (M : FinTM Bool) :
    ∃ (M' : FinTM Bool) (c : ℕ), M'.k = 2 ∧
      (∀ (x output : List Bool) (t : ℕ), M.ComputesInTime x output t →
        M'.ComputesInTime x output (c * ((t + 1) * (Nat.log 2 (t + 1) + 1)))) ∧
      ∀ (x output : List Bool), (∃ t, M'.ComputesInTime x output t) →
        ∃ t, M.ComputesInTime x output t := by
  sorry

/-- G5's function-level form, the `one_work_tape_binary` mirror (from the per-run form). -/
theorem two_work_tapes_binary (M : FinTM Bool) (f : List Bool → List Bool) (T : ℕ → ℕ)
    (hM : M.ComputesFunInTime f T) :
    ∃ (M' : FinTM Bool) (c : ℕ), M'.k = 2 ∧
      M'.ComputesFunInTime f fun n => c * ((T n + 1) * (Nat.log 2 (T n + 1) + 1)) := by
  obtain ⟨M', c, hk, h, -⟩ := two_work_tapes_run M
  exact ⟨M', c, hk, fun x => h x (f x) _ (hM x)⟩

/-- **H2-F's encoding**: the emitting round loop's configuration for a source configuration `c`
of `M`: the mirrored bank and input head are `c`'s, tape `k` holds the table, tape `k + 1` the
blank summary cell and the unary state (empty when halted), both auxiliary heads at `0`, the
output is `c`'s **real** output, and the control is `entry` while `c` is live, halted otherwise.
It is `Turing.StepBlock.Rep` of `{c with output := []}` with the physical output restored
(B6's reset invariant, H2-S finding 4). -/
def emitLoopCfg {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x) :
    Cfg (k + 2) Bool Q x where
  state := c.state.map fun _ => entry
  inputPos := c.inputPos
  workTapes := Fin.addCases (motive := fun _ => ℤ → Option Bool) c.workTapes fun j =>
    if j = 0 then FinTM.bufferTape (tableBitsK M.tr)
    else StepBlock.summaryCells none ((c.state.map unaryFin).getD [])
  workTapePos := Fin.addCases (motive := fun _ => ℤ) c.workTapePos fun _ => 0
  output := c.output

/-- **H2-F, the emitting round loop** (sorried: U2's statement phase; the host that B6's `U₀`
fill, U2-U, U2-E and U-1 share). One finite machine per work-tape count `k`: the step block with
a forwarding leaf (blank the summary cell before each round, emit the round's exact emission
after it, halt physically after a simulated halt). It is a **macro-step lockstep**
(`Turing.Lockstep`) of every `k`-tape table through `emitLoopCfg`, each simulated step costing
at most `50 · (|tableBitsK| + 1) + C`, with the step block's space ledger. -/
theorem exists_emitLoopTM (k : ℕ) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM (k + 2) Bool Q)
      (entry : Q) (C : ℕ),
      ∀ {n : ℕ} {x : List Bool} (M : MultiTapeTM k Bool (Fin (n + 1))),
        ∃ cost : Cfg k Bool (Fin (n + 1)) x → ℕ,
          Lockstep M L (emitLoopCfg entry M) cost {none} {none} ∧
          (∀ c, c.state ≠ none → 0 < cost c ∧
            cost c ≤ 50 * ((tableBitsK M.tr).length + 1) + C) ∧
          ∀ (c : Cfg k Bool (Fin (n + 1)) x) (t : ℕ), (∀ u < t, (M.runFrom c u).state ≠ none) →
            (∀ i : Fin k, L.visitedByTapeHead (emitLoopCfg entry M c)
                (∑ u ∈ Finset.range t, cost (M.runFrom c u)) (Fin.castAdd 2 i) ⊆
              M.visitedByTapeHead c t i) ∧
            ∀ j : Fin (k + 2), k ≤ j.val →
              L.visitedByTapeHead (emitLoopCfg entry M c)
                  (∑ u ∈ Finset.range t, cost (M.runFrom c u)) j ⊆
                Finset.Icc (-((50 * ((tableBitsK M.tr).length + 1) + C : ℕ) : ℤ))
                  ((50 * ((tableBitsK M.tr).length + 1) + C : ℕ) : ℤ) := by
  sorry

/-- The encoding at an initial configuration is exactly U2-S's endpoint layout on the step
block's `k + 2` tapes (the seam the startup must reach). -/
example {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q) (M : MultiTapeTM k Bool (Fin (n + 1))) :
    (emitLoopCfg entry M (M.initCfg x)).output = [] ∧
      (emitLoopCfg entry M (M.initCfg x)).state = some entry := by
  simp [emitLoopCfg, MultiTapeTM.initCfg]

/-- **U2-V, the verdict self-simulator** (sorried: U2's statement phase). For a `k`-tape
scheme, one machine `W` such that for every code `α` there is `C` with: whenever the coded
machine halts on `pairEncode α w` within `t` steps with `output`, `W` halts on the same input
within `C · (t + 1)` steps with the one-bit verdict `[output = [true]]`. -/
theorem exists_selfVerdictTMK (k : ℕ) (E : EffectiveCodeK k) :
    ∃ W : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, ∀ (w output : List Bool) (t : ℕ),
      (E.toFinTM α).ComputesInTime (pairEncode α w) output t →
      W.ComputesInTime (pairEncode α w) [decide (output = [true])] (C * (t + 1)) := by
  sorry

/-- **U2-S, the `k`-generic startup** (sorried: U2's statement phase; B6's gated
`Turing.exists_selfStartupTM` is its `k = 1` instance). It ends at the step block's seam for the
decoded machine's initial configuration: mirrored tapes `0, …, k - 1` blank, the table on tape
`k`, the unary initial state on tape `k + 1` from cell `2`, the scratch tapes blank, the input
head at `1`, the output empty. -/
theorem exists_selfStartupTMK (k : ℕ) (E : EffectiveCodeK k) :
    ∃ (K : ℕ) (P : Type) (_ : Fintype P) (_ : DecidableEq P)
      (R : MultiTapeTM (K + (k + 2)) Bool P) (exit : P),
      ∀ α : List Bool, ∃ T B : ℕ, ∀ w : List Bool,
        R.runFrom (R.initCfg (pairEncode α w)) T =
          ⟨some exit, 1,
            fun j => if j.val = k then FinTM.bufferTape (tableBitsK (E.machine α).tr)
              else if j.val = k + 1 then
                fun p => FinTM.bufferTape (unaryFin (E.machine α).q₀) (p - 2)
              else fun _ => none,
            fun _ => 0, []⟩ ∧
        (∀ u < T, (R.runFrom (R.initCfg (pairEncode α w)) u).state ≠ some exit) ∧
        (∀ u ≤ T, (R.runFrom (R.initCfg (pairEncode α w)) u).output = []) ∧
        R.spaceUsed (R.initCfg (pairEncode α w)) T ≤ B := by
  sorry

/-- B6's gated startup statement (copied verbatim from the tip) is the `k = 1` instance. -/
example (c : EffectiveMachineCode) :
    ∃ (K : ℕ) (P : Type) (_ : Fintype P) (_ : DecidableEq P)
      (R : MultiTapeTM (K + 3) Bool P) (exit : P),
      ∀ α : List Bool, ∃ T B : ℕ, ∀ w : List Bool,
        R.runFrom (R.initCfg (pairEncode α w)) T =
          ⟨some exit, 1,
            fun j => if j = 1 then FinTM.bufferTape (tableBitsK (c.decode α).tm.tr)
              else if j = 2 then fun p => FinTM.bufferTape (unaryFin (c.decode α).tm.q₀) (p - 2)
              else fun _ => none,
            fun _ => 0, []⟩ ∧
        (∀ u < T, (R.runFrom (R.initCfg (pairEncode α w)) u).state ≠ some exit) ∧
        (∀ u ≤ T, (R.runFrom (R.initCfg (pairEncode α w)) u).output = []) ∧
        R.spaceUsed (R.initCfg (pairEncode α w)) T ≤ B := by
  obtain ⟨K, P, hP, dP, R, exit, h⟩ := exists_selfStartupTMK 1 (toK₁ c)
  refine ⟨K, P, hP, dP, R, exit, fun α => ?_⟩
  obtain ⟨T, B, hT⟩ := h α
  refine ⟨T, B, fun w => ?_⟩
  obtain ⟨hrun, hcut, hout, hspace⟩ := hT w
  refine ⟨?_, hcut, hout, hspace⟩
  rw [hrun]
  congr 1
  funext j
  have m1 : 1 % (K + 3) = 1 := Nat.mod_eq_of_lt (by omega)
  have m2 : 2 % (K + 3) = 2 := Nat.mod_eq_of_lt (by omega)
  have h1 : (j = 1) ↔ j.val = 1 := by
    rw [Fin.ext_iff]; simp [m1]
  have h2 : (j = 2) ↔ j.val = 2 := by
    rw [Fin.ext_iff]; simp [m2]
  by_cases a : j.val = 1
  · simp [h1.mpr a, toK₁]
  · by_cases b : j.val = 2
    · have hne : (2 : Fin (K + (1 + 2))) ≠ 1 := by
        intro h
        have := congrArg Fin.val h
        simp [m1, m2] at this
      simp [h2.mpr b, hne, m2, toK₁]
    · simp [a, b, h1, h2]

/-- **U2-U, the self-application universal machine with time and space ledgers, both
directions** (sorried: U2's statement phase; B6's gated `U₀` is its `k = 1` instance). -/
theorem exists_selfUniversalK (k : ℕ) (E : EffectiveCodeK k) :
    ∃ U : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, 0 < C ∧ ∀ w : List Bool,
      (∀ (output : List Bool) (t : ℕ),
        (E.toFinTM α).ComputesInTime (pairEncode α w) output t →
        U.ComputesInTime (pairEncode α w) output (C * (t + 1)) ∧
        U.tm.spaceUsed (U.tm.initCfg (pairEncode α w)) (C * (t + 1)) ≤
          C * ((E.toFinTM α).tm.spaceUsed ((E.toFinTM α).tm.initCfg (pairEncode α w)) t + 1)) ∧
      ∀ output : List Bool, (∃ t', U.ComputesInTime (pairEncode α w) output t') →
        ∃ t, (E.toFinTM α).ComputesInTime (pairEncode α w) output t := by
  sorry

/-- **B6's gated `U₀` statement is the `k = 1` instance** (the statement of
`Turing.exists_selfUniversal`, copied verbatim from the tip, derived from U2-U). -/
example (c : EffectiveMachineCode) :
    ∃ U₀ : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, 0 < C ∧
      ∀ (w output : List Bool) (t s : ℕ),
        (c.decode α).toFinTM.ComputesInTime (pairEncode α w) output t →
        (c.decode α).toFinTM.tm.spaceUsed
            ((c.decode α).toFinTM.tm.initCfg (pairEncode α w)) t ≤ s →
          ∃ t' : ℕ, U₀.ComputesInTime (pairEncode α w) output t' ∧
            U₀.tm.spaceUsed (U₀.tm.initCfg (pairEncode α w)) t' ≤ C * (s + 1) := by
  obtain ⟨U, hU⟩ := exists_selfUniversalK 1 (toK₁ c)
  refine ⟨U, fun α => ?_⟩
  obtain ⟨C, hC, h⟩ := hU α
  refine ⟨C, hC, fun w output t s hrun hs => ⟨C * (t + 1), (h w).1 output t hrun |>.1, ?_⟩⟩
  have := ((h w).1 output t hrun).2
  rw [toK₁_toFinTM] at this
  exact this.trans (Nat.mul_le_mul_left C (by omega))

/-- **U2-E, the code-first evaluator** (sorried: U2's statement phase; Thm 1.9's core; at
`k = 1` it is `Turing.universal`'s statement, the U-1 re-derivation). -/
theorem universalK (k : ℕ) (E : EffectiveCodeK k) :
    ∃ U : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, ∀ x : List Bool,
      (∀ (output : List Bool) (t : ℕ), (E.toFinTM α).ComputesInTime x output t →

      ∀ output : List Bool, (∃ t, U.ComputesInTime (pairEncode α x) output t) →
        ∃ t, (E.toFinTM α).ComputesInTime x output t := by
  sorry

/-- At `k = 1`, U2-E is `Turing.universal`'s statement verbatim (the U-1 instance). -/
example (c : EffectiveMachineCode) :
    ∃ U : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, ∀ x : List Bool,
      (∀ (output : List Bool) (t : ℕ),
        (c.decode α).toFinTM.ComputesInTime x output t →
        U.ComputesInTime (pairEncode α x) output (C * (t + 1))) ∧
      (∀ output : List Bool,
        (∃ t, U.ComputesInTime (pairEncode α x) output t) →
        ∃ t, (c.decode α).toFinTM.ComputesInTime x output t) :=

/-- **Theorem 1.9 at `T log T`** [AB09, Theorem 1.9, relaxed code quantifier as in
`Turing.universal_quadratic`, but per run and with the converse]: one machine `U` such that every
binary machine `M` has a code `α` with which `U` reproduces each halting run of `M` of `t` steps
within `C · (t + 1)(log₂(t + 1) + 1)` steps, and every completed output of `U` is one of `M`'s. -/
theorem universal_TlogT (c : EffectiveMachineCode2) :
    ∃ U : FinTM Bool, ∀ M : FinTM Bool, ∃ (α : List Bool) (C : ℕ),
      (∀ (x output : List Bool) (t : ℕ), M.ComputesInTime x output t →
        U.ComputesInTime (pairEncode α x) output (C * ((t + 1) * (Nat.log 2 (t + 1) + 1)))) ∧
      ∀ (x output : List Bool), (∃ t, U.ComputesInTime (pairEncode α x) output t) →
        ∃ t, M.ComputesInTime x output t := by
  obtain ⟨U, hU⟩ := universalK 2 (toK₂ c)
  refine ⟨U, fun M => ?_⟩
  obtain ⟨M', c₁, hk, hfwd, hback⟩ := two_work_tapes_run M
  obtain ⟨N, hN⟩ := exists_code2TM M' hk
  have hdec : (toK₂ c).toFinTM (c.encode N) = N.toFinTM := by
    rw [toK₂_toFinTM, c.toMachineCode2.decode_encode N]
  obtain ⟨C, hC⟩ := hU (c.encode N)
  refine ⟨c.encode N, C * (c₁ + 1), fun x output t h => ?_, fun x output h => ?_⟩
  · have h₁ := (hN x output _).2 (hfwd x output t h)
    rw [← hdec] at h₁
    apply ((hC x).1 output _ h₁).mono
    have hpos : 1 ≤ (t + 1) * (Nat.log 2 (t + 1) + 1) := Nat.one_le_iff_ne_zero.mpr (by positivity)
    calc C * (c₁ * ((t + 1) * (Nat.log 2 (t + 1) + 1)) + 1)
        ≤ C * ((c₁ + 1) * ((t + 1) * (Nat.log 2 (t + 1) + 1))) := by
          apply Nat.mul_le_mul_left; nlinarith
      _ = C * (c₁ + 1) * ((t + 1) * (Nat.log 2 (t + 1) + 1)) := by ring
  · obtain ⟨t, ht⟩ := (hC x).2 output h
    rw [hdec] at ht
    exact hback x output ⟨t, (hN x output t).1 ht⟩

/-! ### The two-tape `SelfSimulator` and the book-strength hierarchy at it -/

/-- `log₂ (n · 2^m) = log₂ n + m` for `n ≠ 0`. -/
theorem log_mul_two_pow {n : ℕ} (hn : n ≠ 0) (m : ℕ) :
    Nat.log 2 (n * 2 ^ m) = Nat.log 2 n + m := by
  induction m with
  | zero => simp
  | succ m ih =>
    rw [pow_succ, ← Nat.mul_assoc, Nat.log_mul_base (by norm_num) (by positivity), ih]
    ring

/-- **Constant-factor absorption** of the `T log T` overhead:
`logOverhead (c₀ T) ≤ (c₀ + 1)(log₂(c₀ + 1) + 2) · logOverhead T`. -/
theorem logOverhead_mul_le (c₀ T : ℕ) :
    Complexity.TimeHierarchy.U2Probe.logOverhead (c₀ * T) ≤
      (c₀ + 1) * (Nat.log 2 (c₀ + 1) + 2) * Complexity.TimeHierarchy.U2Probe.logOverhead T := by
  unfold Complexity.TimeHierarchy.U2Probe.logOverhead
  set m := Nat.log 2 (c₀ + 1) + 1 with hm
  have hc : c₀ + 1 < 2 ^ m := Nat.lt_pow_succ_log_self (by norm_num) _
  have h1 : c₀ * T + 1 ≤ (c₀ + 1) * (T + 1) := by nlinarith
  have h2 : c₀ * T + 1 ≤ (T + 1) * 2 ^ m := by nlinarith
  have hlog : Nat.log 2 (c₀ * T + 1) ≤ Nat.log 2 (T + 1) + m := by
    calc Nat.log 2 (c₀ * T + 1) ≤ Nat.log 2 ((T + 1) * 2 ^ m) := Nat.log_mono_right h2
      _ = Nat.log 2 (T + 1) + m := log_mul_two_pow (by omega) m
  have h3 : Nat.log 2 (c₀ * T + 1) + 1 ≤ (m + 1) * (Nat.log 2 (T + 1) + 1) := by nlinarith
  calc (c₀ * T + 1) * (Nat.log 2 (c₀ * T + 1) + 1)
      ≤ ((c₀ + 1) * (T + 1)) * ((m + 1) * (Nat.log 2 (T + 1) + 1)) := Nat.mul_le_mul h1 h3
    _ = (c₀ + 1) * (m + 1) * ((T + 1) * (Nat.log 2 (T + 1) + 1)) := by ring

/-- The two-tape scheme of the hierarchy (proved: `Turing.exists_effectiveMachineCode2` is
sorry-free; no uniformity is needed). -/
noncomputable def code₂ : EffectiveMachineCode2 := Classical.choice exists_effectiveMachineCode2

/-- **The two-tape self-simulator** (the `f log f` instance of the parameterization): U2-V over
`code₂`, with G5 + the coding step as its normal form. -/
noncomputable def twoTapeSelfSimulator :
    Complexity.TimeHierarchy.U2Probe.SelfSimulator
      Complexity.TimeHierarchy.U2Probe.logOverhead where
  W := Classical.choose (exists_selfVerdictTMK 2 (toK₂ code₂))
  decode α := (code₂.decode α).toFinTM
  simulates α := by
    obtain ⟨C, hC⟩ := Classical.choose_spec (exists_selfVerdictTMK 2 (toK₂ code₂)) α
    refine ⟨C, fun w b t h => ?_⟩
    have h' := hC w [b] t h
    have hb : decide ([b] = [true]) = b := by cases b <;> simp
    rw [hb] at h'
    exact h'.mono (Nat.mul_le_mul_left C (by omega))
  normalForm T L := by
    rintro ⟨c₀, M, hM⟩
    obtain ⟨M₁, c₁, hk, h₁⟩ := two_work_tapes_binary M
      (fun x => [MultiTapeTM.indicator (L : Set (List Bool)) x]) (fun n => c₀ * T n) hM
    obtain ⟨N, hN⟩ := exists_code2TM M₁ hk
    refine ⟨code₂.encode N, c₁ * ((c₀ + 1) * (Nat.log 2 (c₀ + 1) + 2)), fun x => ?_⟩
    rw [show code₂.decode (code₂.encode N) = N from code₂.toMachineCode2.decode_encode N]
    apply ((hN x _ _).2 (h₁ x)).mono
    have := Nat.mul_le_mul_left c₁ (logOverhead_mul_le c₀ (T x.length))
    show c₁ * ((c₀ * T x.length + 1) * (Nat.log 2 (c₀ * T x.length + 1) + 1)) ≤ _
    refine this.trans ?_
    rw [Nat.mul_assoc c₁]
    exact Nat.mul_le_mul_left c₁ (Nat.mul_le_mul_left _ (Nat.le_succ _))

/-- **Thm 3.1 at book strength, end to end** (modulo the sorried G5 and U2-V statements only). -/
theorem time_hierarchy_book {f g : ℕ → ℕ} (hg : Complexity.TimeConstructible g)
    (hf : ∀ n, n ≤ f n)
    (hfg : ∀ A : ℕ, ∃ N, ∀ n ≥ N, A * (f n * Nat.log 2 (f n)) ≤ g n) :
    Complexity.DTIME f ⊂ Complexity.DTIME (fun n => g n + 1) :=
  Complexity.TimeHierarchy.U2Probe.time_hierarchy_book twoTapeSelfSimulator hg hf hfg

/-- **`DTIME(n + 1) ⊊ DTIME((n + 1)²)`, end to end** (modulo the same two statements). -/
theorem dtime_linear_ssubset_dtime_sq :
    Complexity.DTIME (fun n => n + 1) ⊂ Complexity.DTIME (fun n => (n + 1) ^ 2) :=
  Complexity.TimeHierarchy.U2Probe.dtime_linear_ssubset_dtime_sq twoTapeSelfSimulator
