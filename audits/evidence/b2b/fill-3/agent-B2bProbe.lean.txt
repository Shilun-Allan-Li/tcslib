/-- Route 2: the SAT3 corollary, verbatim (modulo route 1). -/
example : SAT3 ∈ PSPACE := NP_subset_PSPACE SAT3_mem_NP

/-- G-0 (no new machinery): the verifier language `V ∈ P` has a decider whose
cleaned machine, on every word `y`, halts with the indicator, blank work tapes and
origin heads, every head within a radius polynomial in `|y|`. Space comes from
`P_subset_PSPACE` (that is, `DTIME_subset_SPACE` at the verifier's time bound),
and the clean-up is `Complexity.LogProg.cleanTM_run`. -/
theorem clean_verifier (V : Language Bool) (hV : V ∈ P) :
    ∃ (e cV : ℕ) (MV : FinTM Bool), ∀ y : List Bool, ∃ T : ℕ,
      ((LogProg.cleanTM MV.tm MV.tm.q₀).runFrom
          (Cfg.init (LogProg.CleanSt.init MV.tm.q₀) y) T).state = none ∧
      ((LogProg.cleanTM MV.tm MV.tm.q₀).runFrom
          (Cfg.init (LogProg.CleanSt.init MV.tm.q₀) y) T).output =
        [MultiTapeTM.indicator (V : Set (List Bool)) y] ∧
      ((LogProg.cleanTM MV.tm MV.tm.q₀).runFrom
          (Cfg.init (LogProg.CleanSt.init MV.tm.q₀) y) T).workTapes = (fun _ _ => none) ∧
      ((LogProg.cleanTM MV.tm MV.tm.q₀).runFrom
          (Cfg.init (LogProg.CleanSt.init MV.tm.q₀) y) T).workTapePos = (fun _ => 0) ∧
      ∀ t ≤ T, ∀ j, |((LogProg.cleanTM MV.tm MV.tm.q₀).runFrom
          (Cfg.init (LogProg.CleanSt.init MV.tm.q₀) y) t).workTapePos j| ≤
            (cV * (y.length ^ e + 1) : ℕ) := by
  obtain ⟨e, he⟩ := Set.mem_iUnion.mp (P_subset_PSPACE hV)
  obtain ⟨cV, MV, hMV⟩ := he
  refine ⟨e, cV, MV, fun y => ?_⟩
  obtain ⟨t, ht, hsp⟩ := hMV y
  obtain ⟨hh, ho⟩ := (FinTM.computesInTime_iff _ _ _ _).mp ht
  have hs : ∀ i, (MV.tm.visitedByTapeHead (Cfg.init MV.tm.q₀ y) t i).card ≤
      cV * (y.length ^ e + 1) := fun i =>
    (MV.tm.spaceUsedByTape_le_spaceUsed _ t i).trans hsp
  obtain ⟨T, h1, h2, h3, h4, h5⟩ :=
    LogProg.cleanTM_run MV.tm MV.tm.q₀ MV.tm.q₀ y t hh _ hs
  exact ⟨T, h1, h2.trans ho, h3, h4, h5⟩

/-- G-2: the input rewind glue. Its table is the `timed_rewind` shape, so its
contract is an instance of the public lemma with no new induction. Under
`vhostEmitTM` it rewinds the buffer head. -/
inductive RwPh | start | scan | done
  deriving DecidableEq, Fintype

/-- `halt = false`: end in the live anchor `done` (a seam exit);
`halt = true`: halt, so an enclosing R1′ return can fire. -/
def rewindDest (halt : Bool) : Option RwPh := if halt then none else some .done

def rewindTM (k : ℕ) (halt : Bool) : MultiTapeTM k Bool RwPh where
  q₀ := .start
  tr := fun q inp _ =>
    match q with
    | .start => FinTM.controlAction .neg (some .scan)
    | .scan =>
      match inp with
      | some _ => FinTM.controlAction .neg (some .scan)
      | none => FinTM.controlAction .pos (rewindDest halt)
    | .done => FinTM.controlAction 0 (some .done)

theorem rewindTM_run {k : ℕ} {x : List Bool} (halt : Bool) (c : Cfg k Bool RwPh x)
    (hs : c.state = some .start) :
    ∃ r ≤ c.inputPos.val + 2,
      (rewindTM k halt).runFrom c r = { c with state := rewindDest halt, inputPos := 1 } :=
  FinTM.timed_rewind (rewindTM k halt) .start .scan (rewindDest halt) (fun _ _ => rfl)
    (fun inp _ => by cases inp <;> rfl) c hs

/-- The same contract with the work-head trajectory, from the newly public
`FinTM.rewind_heads` (pre-issue 2). -/
theorem rewindTM_run_heads {k : ℕ} {x : List Bool} (halt : Bool) (c : Cfg k Bool RwPh x)
    (hs : c.state = some .start) :
    ∃ r ≤ c.inputPos.val + 2,
      (rewindTM k halt).runFrom c r = { c with state := rewindDest halt, inputPos := 1 } ∧
      ∀ u ≤ r, ((rewindTM k halt).runFrom c u).workTapePos = c.workTapePos :=
  FinTM.rewind_heads (rewindTM k halt) .start .scan (rewindDest halt) (fun _ _ => rfl)
    (fun inp _ => by cases inp <;> rfl) c hs

/-- **Work-tape origin finding by virtual input** (revision 2, the setup's key step).
Host the halting rewind on a buffer tape holding a word `w` with its head anywhere in
`[-1, |w|]` (the capture leaves it at `|w|`): the buffer head returns to `0` and the buffer
is unchanged. Only public rows: `vhostEmitTM_runFrom` and the rewind instance. -/
theorem vhost_rewind {x y : List Bool} (c : Cfg 0 Bool RwPh y) (hs : c.state = some .start)
    (b : Bool) (hb : FinTM.VirtualTag c.inputPos b) (p : Fin (x.length + 2)) (pre : List Bool) :
    ∃ u ≤ c.inputPos.val + 2, ∃ b',
      (vhostEmitTM (rewindTM 0 true)).runFrom (vhostCfg c b p pre) u =
        vhostCfg ({ c with state := none, inputPos := 1 } : Cfg 0 Bool RwPh y) b' p pre ∧
      ((vhostEmitTM (rewindTM 0 true)).runFrom (vhostCfg c b p pre) u).workTapePos
        (vhostBuffer 0) = 0 ∧
      ((vhostEmitTM (rewindTM 0 true)).runFrom (vhostCfg c b p pre) u).workTapes
        (vhostBuffer 0) = FinTM.bufferTape y := by
  obtain ⟨u, hu, hrun⟩ := rewindTM_run true c hs
  obtain ⟨b', _, he⟩ := vhostEmitTM_runFrom (rewindTM 0 true) c b hb p pre u
  rw [hrun] at he
  refine ⟨u, hu, b', he, ?_, ?_⟩
  · rw [he]; simp only [vhostCfg, vhostBuffer, Fin.addCases_left]; simp
  · rw [he]; simp only [vhostCfg, vhostBuffer, Fin.addCases_left]

/-- G-3: the verdict reader on a capture tape `i` holding `[b]` with its head on
cell `1` (the R1′ capture convention): one left step, then read, erase, and enter
the live anchor `done b`. Two steps; the tape ends blank with its head at `0`. -/
inductive VrPh | left | read | done (b : Bool)
  deriving DecidableEq, Fintype

def verdictTM (k : ℕ) (i : Fin k) : MultiTapeTM k Bool VrPh where
  q₀ := .left
  tr := fun q _ w =>
    match q with
    | .left => tapeAction i 0 none .neg .read
    | .read =>
      match w i with
      | some b => tapeAction i 0 (some none) 0 (.done b)
      | none => tapeAction i 0 none 0 (.done false)
    | .done b => inputMoveAction 0 (.done b)

theorem verdictTM_run {k : ℕ} {x : List Bool} (i : Fin k) (d : Cfg k Bool VrPh x) (b : Bool)
    (hs : d.state = some .left) (hb : d.workTapes i (d.workTapePos i - 1) = some b) :
    ((verdictTM k i).runFrom d 2).state = some (.done b) ∧
      ((verdictTM k i).runFrom d 2).workTapePos i = d.workTapePos i - 1 ∧
      ((verdictTM k i).runFrom d 2).workTapes i (d.workTapePos i - 1) = none ∧
      (∀ j, j ≠ i → ((verdictTM k i).runFrom d 2).workTapes j = d.workTapes j ∧
        ((verdictTM k i).runFrom d 2).workTapePos j = d.workTapePos j) ∧
      ((verdictTM k i).runFrom d 2).inputPos = d.inputPos ∧
      ((verdictTM k i).runFrom d 2).output = d.output := by
  have h1 : (verdictTM k i).runFrom d 1 =
      { d with state := some .read,
               workTapePos := fun j => if j = i then d.workTapePos j - 1 else d.workTapePos j } := by
    apply Cfg.ext <;>
      simp [MultiTapeTM.runFrom, MultiTapeTM.step, hs, verdictTM, tapeAction, Action.apply,
        moveInputPos_zero]
    · funext j; by_cases hj : j = i <;> simp [hj]
    · funext j; by_cases hj : j = i <;> simp [hj, sub_eq_add_neg]
  have hr : (verdictTM k i).runFrom d 2 = (verdictTM k i).step ((verdictTM k i).runFrom d 1) :=
    MultiTapeTM.runFrom_succ_eq_step'
  rw [hr, h1]
  simp [MultiTapeTM.step, verdictTM, Cfg.workTapeSymbols, hb, tapeAction, Action.apply,
    moveInputPos_zero]
  refine ⟨by rw [if_pos rfl]; exact Function.update_self _ _ _, fun j hj => ?_⟩
  simp [hj]

/-- The counter word `1^Q` has value `2^Q - 1`: `d = 2^Q - 1` rounds, and the
done-continuation tests the last certificate. -/
theorem ofDigits_replicate_true (Q : ℕ) :
    Nat.ofDigits 2 ((List.replicate Q true).map Bool.toNat) = 2 ^ Q - 1 := by
  induction Q with
  | zero => rfl
  | succ Q ih =>
    rw [List.replicate_succ, List.map_cons, Nat.ofDigits_cons, ih]
    have : 1 ≤ 2 ^ Q := Nat.one_le_two_pow
    simp only [Bool.toNat_true, pow_succ]
    omega

/-- The certificate orbit: from `0^Q`, the `r`-th certificate is `enumWord Q r`
for every `r < 2^Q`, through the public increment arithmetic. -/
theorem orbit_enumWord (Q r : ℕ) (hr : r < 2 ^ Q) :
    (fun u => (incFixed u).getD u)^[r] (List.replicate Q false) = enumWord Q r := by
  induction r with
  | zero => exact (Meyer.enumWord_zero Q).symm
  | succ r ih =>
    rw [Function.iterate_succ_apply', ih (by omega),
      Meyer.incFixed_enumWord Q r hr, Option.getD_some]

/-- Completeness: every width-`Q` certificate is some `enumWord Q r`, `r < 2^Q`. -/
theorem enumWord_onto (u : List Bool) :
    ∃ r < 2 ^ u.length, enumWord u.length r = u :=
  ⟨TimeHierarchy.ctrVal u, Meyer.ctrVal_lt u, Meyer.enumWord_ctrVal u⟩

/-- Final packaging (P4.1 Q5 normalization): a decider with a halting time and a
ledger `D·(n+1)^f` puts the language in `PSPACE`. -/
theorem mem_PSPACE_of_ledger (L : Language Bool) (M : FinTM Bool) (D f : ℕ)
    (h : ∀ x : List Bool, ∃ t,
      M.ComputesInTime x [MultiTapeTM.indicator (L : Set (List Bool)) x] t ∧
        M.tm.spaceUsed (M.tm.initCfg x) t ≤ D * (x.length + 1) ^ f) :
    L ∈ PSPACE := by
  have hS : L ∈ SPACE (fun n => D * (n + 1) ^ f) := ⟨1, M, fun x => by
    obtain ⟨t, ht, hs⟩ := h x
    exact ⟨t, ht, by simpa using hs⟩⟩
  refine space_poly_subset_PSPACE f (SPACE.mono_const (a := D * 2 ^ f) (fun n => ?_) hS)
  calc D * (n + 1) ^ f ≤ D * (2 ^ f * (n ^ f + 1)) := Nat.mul_le_mul_left D (succ_pow_le n f)
    _ = D * 2 ^ f * (n ^ f + 1) := (Nat.mul_assoc _ _ _).symm

/-- Type-level assembly of the hosted source `D` (rewind; cleaned decider with a
live exit; rewind) and its R1′ call on `1 + 2k` bank tapes, so the tape counts of
the seams and the virtual-input host match. -/
noncomputable def hostedSource (MV : FinTM Bool) :
    MultiTapeTM (MV.k + MV.k) Bool
      ((RwPh ⊕ Unit) ⊕ ((LogProg.CleanSt MV.State MV.k ⊕ Unit) ⊕ RwPh)) :=
  seamCompTM (embedEmitRetTM (Function.Embedding.refl _) (rewindTM _ true)) (Sum.inr ())
    (seamCompTM (embedEmitRetTM (Function.Embedding.refl _)
        (LogProg.cleanTM MV.tm MV.tm.q₀)) (Sum.inr ())
      (rewindTM _ true) .start)
    (Sum.inl (Sum.inl (LogProg.CleanSt.init MV.tm.q₀)))

/-- The anchor phase: one stationary state, always redirected by the phase host, so the
loop anchor is a state the round never revisits. -/
def anchorTM (k : ℕ) : MultiTapeTM k Bool Unit where
  q₀ := ()
  tr := fun _ _ _ => inputMoveAction 0 ()

/-- Revision 2 assembly check: a phase-host body (any finite phase family) is a valid
`counterLoopTM` body, and the hosts' state types are finite and decidable, so the whole
host packs into a `FinTM` with no seam composite and no R8.8. -/
noncomputable def loopOverPhases {k : ℕ} {I : Type} [DecidableEq I] {Ph : I → Type}
    [∀ i, DecidableEq (Ph i)] (M : (i : I) → MultiTapeTM k Bool (Ph i)) (q0 : Σ i, Ph i)
    (red : (i : I) → Ph i → Option (Σ i, Ph i)) (a e : Σ i, Ph i) :
    MultiTapeTM (k + 1) Bool (CounterLoopState (Σ i, Ph i)) :=
  counterLoopTM (phaseHostTM M q0 red) a (some e)

/-- The phase host's dispatch lands on the next phase's start in the form its own lemma
consumes: retagging commutes with control replacement. -/
theorem dispatch_retag {k : ℕ} {I : Type} {Ph : I → Type} {S : Type} {x : List Bool}
    (c : Cfg k Bool S x) (j : I) (q : Ph j) :
    c.mapState (fun _ => (⟨j, q⟩ : Σ i, Ph i)) =
      (c.mapState (fun _ => q)).mapState (Sigma.mk j) := by
  cases c; simp [Cfg.mapState]; cases ‹Option S› <;> rfl

/-- The halting rewind's first halt, as `exists_first_halt` gives it, feeds the
R1′ return (`embedEmitRetTM_run`) for a live, first-visited exit. -/
theorem rewind_first_halt {k : ℕ} {x : List Bool} (c : Cfg k Bool RwPh x)
    (hs : c.state = some .start) :
    ∃ u ≤ c.inputPos.val + 2, (∀ v < u, ¬((rewindTM k true).runFrom c v).Halted) ∧
      (rewindTM k true).runFrom c u = { c with state := none, inputPos := 1 } := by
  obtain ⟨r, hr, hrun⟩ := rewindTM_run true c hs
  have hh : ((rewindTM k true).runFrom c r).state = none := by rw [hrun]; rfl
  obtain ⟨u, hu, hlive, heq⟩ := MultiTapeTM.exists_first_halt _ c hh
  exact ⟨u, hu.trans hr, hlive, heq.trans hrun⟩

noncomputable def hostedCall (MV : FinTM Bool) (K : ℕ) (ι : Fin (1 + (MV.k + MV.k)) ↪ Fin K)
    (cap : Fin K) :=
  embedSilentRetTM ι cap (vhostEmitTM (hostedSource MV))
