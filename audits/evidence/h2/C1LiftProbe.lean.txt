/-!
# Probe: the nondeterministic table is a deterministic `tableBitsK` table

Design evidence for `h2/H2-design.md` (consumer C1), not a landing file. The table part of
`Turing.CodeNDTM.serialize` (choice bit outermost, then states, then reads) is, bit for bit,
`Turing.tableBitsK` of the *choice-lifted* deterministic two-work-tape table on
`2·(numStates + 1)` states: state `p < numStates + 1` runs the `false` table at `p`, state
`numStates + 1 + q` the `true` table at `q`, and successors stay in the lower half. So the
step block of host H2, unchanged, performs one nondeterministic step under choice `b` from the
lifted state `b·(numStates + 1) + q`.

## References

* [AB09] S. Arora, B. Barak, *Computational Complexity: A Modern Approach*,
  Cambridge University Press, 2009. (§1.4, §3.2.)
-/

namespace Turing.Draft.C1Probe

/-- The choice-lifted table of a coded NDTM with `n + 1` states: on `2·(n + 1)` states, the
lower half runs the `false` table and the upper half the `true` table, with every successor
in the lower half. [AB09, §3.2] -/
def liftTr (M : CodeNDTM) :
    Fin (M.numStates + 1 + M.numStates + 1) → Option Bool → (Fin 2 → Option Bool) →
      Action 2 Bool (Fin (M.numStates + 1 + M.numStates + 1)) :=
  fun p inp w =>
    if h : p.val < M.numStates + 1 then
      (M.tm.tr false ⟨p.val, h⟩ inp w).mapState (Fin.castLE (by omega))
    else
      (M.tm.tr true ⟨p.val - (M.numStates + 1), by have := p.isLt; omega⟩ inp w).mapState
        (Fin.castLE (by omega))

/-- Renaming the successor along an order embedding of `Fin`s keeps a record's bits.

**Proof sketch.** The successor is serialized in unary by its value, which `Fin.castLE`
keeps. -/
lemma actionBitsK_mapState_castLE {k n m : ℕ} (h : n + 1 ≤ m + 1)
    (a : Action k Bool (Fin (n + 1))) :
    actionBitsK (a.mapState (Fin.castLE h)) = actionBitsK a := by
  cases hs : a.state with
  | none => simp [actionBitsK, Action.mapState, hs, optStateBits]
  | some s => simp [actionBitsK, Action.mapState, hs, optStateBits, unaryFin]

/-- **The nondeterministic table is a lifted deterministic table**: the table part of
`Turing.CodeNDTM.serialize` equals `Turing.tableBitsK (liftTr M)`. [AB09, §1.4, §3.2]

**Proof sketch.** Split the lifted state range into its two halves (`List.ofFn_add`); on the
lower half the lifted table is the `false` table, on the upper half the `true` table, each
with successors renamed by `Fin.castLE`, which keeps the bits; the work reads of
`Turing.workReadsK 2` are the `workPair`s in the same order (`Turing.workReadsK_two`), and
`actionBits₂` is `actionBitsK` at two tapes. -/
theorem ndTable_eq_tableBitsK (M : CodeNDTM) :
    (([false, true] : List Bool).flatMap fun b =>
        (List.finRange (M.numStates + 1)).flatMap fun q =>
          ([none, some false, some true] : List (Option Bool)).flatMap fun inp =>
            ([none, some false, some true] : List (Option Bool)).flatMap fun w₀ =>
              ([none, some false, some true] : List (Option Bool)).flatMap fun w₁ =>
                actionBits₂ (M.tm.tr b q inp (workPair w₀ w₁))) =
      tableBitsK (liftTr M) := by
  unfold tableBitsK
  rw [workReadsK_two]
  have split : List.finRange (M.numStates + 1 + M.numStates + 1) =
      List.ofFn (fun i : Fin ((M.numStates + 1) + (M.numStates + 1)) => i) := rfl
  rw [split, List.ofFn_add, List.flatMap_append, List.ofFn_eq_map, List.ofFn_eq_map,
    List.flatMap_map, List.flatMap_map]
  simp only [List.flatMap_cons, List.flatMap_nil, List.append_nil, actionBits₂_eq_actionBitsK]
  congr 1
  · congr 1
    funext q
    simp [liftTr, actionBitsK_mapState_castLE]
  · congr 1
    funext q
    simp [liftTr, actionBitsK_mapState_castLE]

end Turing.Draft.C1Probe
