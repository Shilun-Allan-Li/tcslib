"""Summarize one Lean output transcript: diagnostics by severity and category, and the
`#print axioms` lines (which ones use `sorryAx`, which go beyond the standard triple).

Usage:  python3 -I elabsummary.py OUTPUT.txt [--expect-sorry-decls N] [--expect-errors 0]
                                  [--expect-other-warnings N] [--expect-axiom-lines N]
                                  [--expect-sorryax NAME,NAME,...]
Exit status 0 iff every given expectation holds.
"""
import collections
import re
import sys

STD = {'propext', 'Classical.choice', 'Quot.sound'}


def main():
    path = sys.argv[1]
    opts = dict(zip(sys.argv[2::2], sys.argv[3::2]))
    text = open(path, encoding='utf-8').read()
    diag = re.compile(r'^\S+?:(\d+):(\d+): (error|warning)(?:\([^)]*\))?: (.*)$', re.M)
    ds = [(int(m.group(1)), m.group(3), m.group(4)) for m in diag.finditer(text)]
    errors = [d for d in ds if d[1] == 'error']
    sorry = [d for d in ds if d[1] == 'warning' and "declaration uses 'sorry'" in d[2]]
    other = [d for d in ds if d[1] == 'warning' and d not in sorry]
    cat = collections.Counter()
    for _, _, msg in other:
        m = msg.strip()
        if m.startswith('This simp argument is unused'):
            cat['unused simp argument'] += 1
        elif 'unreachable' in m or m.startswith('this tactic is never executed'):
            cat['unreachable tactic'] += 1
        elif '<;>' in m or 'tactic sequencing' in m.lower() or m.startswith("'<;>'"):
            cat['<;> sequencing lint'] += 1
        elif 'omega' in m:
            cat['no-op omega'] += 1
        elif 'simpa' in m:
            cat['simpa suggestion'] += 1
        elif 'congr' in m:
            cat['no-op congr'] += 1
        elif 'unused variable' in m:
            cat['unused variable'] += 1
        else:
            cat['other: ' + m[:70]] += 1
    ax = re.findall(r"^'([^']+)' (depends on axioms: \[([^\]]*)\]|does not depend on any axioms)",
                    text, re.M)
    axrows = []
    for name, _, lst in ax:
        s = {x.strip() for x in lst.split(',') if x.strip()}
        axrows.append((name, s))
    with_sorry = [n for n, s in axrows if 'sorryAx' in s]
    beyond = [(n, sorted(s - STD - {'sorryAx'})) for n, s in axrows if s - STD - {'sorryAx'}]
    print(f'# {path}')
    print(f'errors: {len(errors)}')
    for ln, _, msg in errors:
        print(f'  ERROR line {ln}: {msg[:120]}')
    print(f"sorry warnings: {len(sorry)} (lines {', '.join(str(d[0]) for d in sorry)})")
    print(f'other warnings: {len(other)}')
    for k, v in sorted(cat.items(), key=lambda kv: -kv[1]):
        print(f'  {v:3d}  {k}')
    print(f'#print axioms lines: {len(axrows)}')
    print(f'  within the standard triple (or none): {len(axrows) - len(with_sorry) - len(beyond)}')
    print(f'  with sorryAx: {len(with_sorry)}')
    for n in with_sorry:
        print(f'    {n}')
    print(f'  beyond the standard triple and sorryAx: {len(beyond)}')
    for n, s in beyond:
        print(f'    {n}: {s}')
    ok = True
    def check(key, val):
        nonlocal ok
        if key in opts:
            good = str(val) == opts[key]
            print(f'expect {key[len("--expect-"):]} = {opts[key]}: '
                  f'{"PASS" if good else "FAIL (got " + str(val) + ")"}')
            ok &= good
    check('--expect-errors', len(errors))
    check('--expect-sorry-decls', len(sorry))
    check('--expect-other-warnings', len(other))
    check('--expect-axiom-lines', len(axrows))
    if '--expect-sorryax' in opts:
        want = sorted(x for x in opts['--expect-sorryax'].split(',') if x)
        got = sorted(with_sorry)
        good = want == got
        print(f'expect sorryAx exactly in {len(want)} names: {"PASS" if good else "FAIL"}')
        if not good:
            print('  missing:', sorted(set(want) - set(got)))
            print('  extra  :', sorted(set(got) - set(want)))
        ok &= good
    if beyond:
        ok = False
        print('FAIL: an axiom beyond the standard triple and sorryAx')
    print('RESULT:', 'PASS' if ok else 'FAIL')
    sys.exit(0 if ok else 1)


if __name__ == '__main__':
    main()
