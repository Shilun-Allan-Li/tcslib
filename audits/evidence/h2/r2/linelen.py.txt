"""Line-length census (finding 8): lines longer than 100, counted two ways.

Usage:  python3 -I linelen.py FILE[:FROM-TO[,FROM-TO...]] ...
For each file (optionally restricted to 1-based inclusive line ranges) print the number of
lines with more than 100 Unicode characters (code points; the policy's metric) and, for
comparison, more than 100 UTF-8 bytes (round 1's mistaken metric). Lines are counted without
their newline.
"""
import sys


def main():
    for arg in sys.argv[1:]:
        path, _, rng = arg.partition(':')
        if not rng and ':' in arg[2:]:
            path, _, rng = arg.rpartition(':')
        lines = open(path, encoding='utf-8').read().split('\n')
        idx = range(len(lines))
        if rng:
            sel = set()
            for part in rng.split(','):
                a, b = part.split('-')
                sel.update(range(int(a) - 1, int(b)))
            idx = sorted(i for i in sel if i < len(lines))
        chars = sum(1 for i in idx if len(lines[i]) > 100)
        byts = sum(1 for i in idx if len(lines[i].encode('utf-8')) > 100)
        longest = max((len(lines[i]) for i in idx), default=0)
        print(f'{arg}: {len(idx)} lines; >100 Unicode characters: {chars}; '
              f'>100 UTF-8 bytes: {byts}; longest: {longest} characters')


if __name__ == '__main__':
    main()
