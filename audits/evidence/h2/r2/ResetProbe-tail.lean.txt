/-! ## Probe, not part of the draft: consumer obligation CO-RF, the data-level steps

These three facts are the first, purely data-level steps of the reset/forward wrapper that B6's
`U₀` and U-1 must build (design note r2, CO-RF). They are consumer-side and are not proposed
for the H2 surface. -/

/-- CO-RF step 1: blanking the summary cell of a representing configuration gives a
configuration representing the same source configuration with its output erased. -/
theorem probe_rep_reset {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (d : Cfg (k + 2) Bool (Σ key, Ph k key) x)
    (rel : Rep tr c d) :
    Rep tr { c with output := [] }
      { d with workTapes := (Function.update d.workTapes (summaryTape k)
          (Function.update (d.workTapes (summaryTape k)) 0 none)) } := by
  obtain ⟨hin, hbank, htab, hsum, htp, hsp⟩ := rel
  have hmir : ∀ i : Fin k, Fin.castAdd 2 i ≠ summaryTape k := by
    intro i h
    have := congrArg Fin.val h
    simp [summaryTape] at this
    omega
  have hts : tableTape k ≠ summaryTape k := by
    intro h
    have := congrArg Fin.val h
    simp [tableTape, summaryTape] at this
  refine ⟨hin, fun i => ⟨?_, (hbank i).2⟩, ?_, ?_, htp, hsp⟩
  · simp only [Function.update_of_ne (hmir i)]
    exact (hbank i).1
  · simp only [Function.update_of_ne hts]
    exact htab
  · simp only [Function.update_self, hsum]
    funext z
    by_cases hz : z = 0
    · subst hz
      simp [summaryCells, outputSummary]
    · simp [summaryCells, hz]

/-- CO-RF step 2: erasing the output changes neither the reads nor the state, so the source
selects the same action; and applying it to the erased configuration gives the successor with
only that action's emission as output. -/
theorem probe_reset_action {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1))) :
    ({ c with output := [] } : Cfg k Bool (Fin (n + 1)) x).inputSymbol = c.inputSymbol ∧
      ({ c with output := [] } : Cfg k Bool (Fin (n + 1)) x).workTapeSymbols =
        c.workTapeSymbols ∧
      ({ c with output := [] } : Cfg k Bool (Fin (n + 1)) x).state = c.state ∧
      a.apply { c with output := [] } = { a.apply c with output := a.output.toList } := by
  refine ⟨rfl, rfl, rfl, ?_⟩
  simp [Action.apply]

/-- CO-RF step 3: after a round started on the blanked cell, the cell holds exactly the
round's emission (the contract's summary clause and `summaryEmit_none`), whatever the earlier
output was. -/
theorem probe_cell_after_reset (b : Option Bool) (s : Option Bool) (hs : s = none) :
    summaryEmit s b = b := by
  subst hs
  exact summaryEmit_none b

end Turing.Draft.StepBlock
