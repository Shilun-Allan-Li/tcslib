"""H2-S round 2: build the two axiom-probe files (probe copies, not drafts).

Usage:  python3 -I mkprobes.py DRAFT.lean MOVE.lean AXPROBE_OUT.lean MOVEPLUS_OUT.lean

* AXPROBE_OUT: the statement draft verbatim, followed by `#print axioms` for every one of its
  declarations (the draft itself carries no `#print`).
* MOVEPLUS_OUT: the generated move file (MOVE, from movemanifest.py) with the draft's six
  additive declarations appended inside `namespace Turing.StepBlock`, the two generic
  visited-set lemmas placed at their proposed landing names
  `Turing.MultiTapeTM.visitedByTapeHead_add_subset` / `_step_subset` (design note O9), and
  `#print axioms` for every declaration. This checks the additive proofs against the *proved*
  contract rather than the draft's admitted one.
"""
import re
import sys

ADDITIVE = ['summaryEmit_none', 'visitedByTapeHead_add_subset', 'visitedByTapeHead_step_subset',
            'roundTM_visited', 'roundTM_iterate', 'roundTM_firstExit_unique']
GENERIC = ['visitedByTapeHead_add_subset', 'visitedByTapeHead_step_subset']
DECL = re.compile(r'^(?:@\[[^\]]*\]\s*)?(?:(?:private|protected|noncomputable)\s+)*'
                  r'(def|theorem|lemma|abbrev|inductive|structure|instance|class)\s+(\S+)')
STOP = re.compile(r'^(?:/-|end\b|namespace\b|section\b|#|set_option\b|open\b|variable\b)')


def in_comment_at_start(lines):
    """For each line: is its first character inside a block comment opened earlier?"""
    out, depth = [], 0
    for l in lines:
        out.append(depth > 0)
        i = 0
        while i < len(l):
            if depth == 0 and l.startswith('--', i):
                break
            if l.startswith('/-', i):
                depth += 1
                i += 2
            elif depth > 0 and l.startswith('-/', i):
                depth -= 1
                i += 2
            else:
                i += 1
    return out


def blocks(lines):
    """(name, start, end) for each declaration, start at its docstring, end exclusive."""
    inc = in_comment_at_start(lines)
    starts = [(i, DECL.match(l).group(2)) for i, l in enumerate(lines)
              if not inc[i] and DECL.match(l)]
    out = []
    for i, name in starts:
        a = i
        if a > 0 and lines[a - 1].rstrip().endswith('-/'):
            a -= 1
            while not lines[a].startswith('/--'):
                a -= 1
        b = i + 1
        while b < len(lines) and (inc[b] or (not DECL.match(lines[b])
                                             and not STOP.match(lines[b]))):
            b += 1
        while b > i + 1 and not lines[b - 1].strip():
            b -= 1
        out.append((name, a, b))
    return out


def qualify(name, ns):
    name = name.replace('_root_.', '')
    return name if name.startswith('Turing.') else ns + '.' + name


def main():
    draft, move, axout, mpout = sys.argv[1:5]
    dl = open(draft, encoding='utf-8').read().split('\n')
    db = blocks(dl)
    names = [n for n, _, _ in db]
    assert len(names) == len(set(names)), 'duplicate names in draft'
    for n in ADDITIVE:
        assert n in names, n
    with open(axout, 'w', encoding='utf-8') as f:
        f.write('\n'.join(dl).rstrip('\n') + '\n\n')
        f.write('-- ===== axiom probe (not part of the draft) =====\n')
        for n in names:
            f.write(f'#print axioms {qualify(n, "Turing.Draft.StepBlock")}\n')
    print(f'{axout}: draft + {len(names)} #print axioms')

    ml = open(move, encoding='utf-8').read().rstrip('\n').split('\n')
    assert ml[-3:] == ['end StepBlock', '', 'end Turing'], ml[-3:]
    body = ml[:-3]
    add = []
    for n, a, b in db:
        if n in ADDITIVE:
            add.append('\n'.join(dl[a:b]))
    txt = '\n\n'.join(add)
    for g in GENERIC:
        k = len(re.findall(r'^lemma ' + g + r'\b', txt, re.M))
        assert k == 1, (g, k)
        txt = re.sub(r'^lemma ' + g + r'\b', 'lemma _root_.Turing.MultiTapeTM.' + g, txt,
                     flags=re.M)
        refs = len(re.findall(r'(?<![.\w])' + g + r'\b', txt))
        txt = re.sub(r'(?<![.\w])' + g + r'\b', 'MultiTapeTM.' + g, txt)
        print(f'  {g}: declared at Turing.MultiTapeTM.{g}; {refs} references qualified')
    mb = blocks(body)
    nsline = body.index('namespace StepBlock')
    allnames = [qualify(n, 'Turing' if a < nsline else 'Turing.StepBlock') for n, a, _ in mb]
    for n, _, _ in blocks(txt.split('\n')):
        allnames.append(qualify(n, 'Turing.StepBlock'))
    with open(mpout, 'w', encoding='utf-8') as f:
        f.write('\n'.join(body).rstrip('\n') + '\n\n')
        f.write('/-! ## Additive results (round-2 draft, appended for the axiom probe) -/\n\n')
        f.write(txt + '\n\nend StepBlock\n\nend Turing\n\n')
        f.write('-- ===== axiom probe =====\n')
        for n in allnames:
            f.write(f'#print axioms {n}\n')
    print(f'{mpout}: move ({len(mb)} declarations) + {len(ADDITIVE)} additive; '
          f'{len(allnames)} #print axioms')


if __name__ == '__main__':
    main()
