/-! ### Relabelling preserves space -/

/-- **State relabelling preserves every head trajectory.** For any machine `tm` and any state
equivalence `e`, head `i` of the relabelled machine visits, in its first `t` steps from its
initial configuration, exactly the cells that head `i` of `tm` visits. This is the space
clause that the coding step `Turing.exists_codeTM` does not state. [AB09, §1.4: the coding
step of the normal form]

**Proof sketch.** The relabelled run is the original run with its states renamed
(`Turing.MultiTapeTM.relabelState_runFrom_init`), and renaming changes no head position. -/
theorem MultiTapeTM.relabelState_visitedByTapeHead {k : ℕ} {Symbol : Type*}
    {State State' : Type*} (tm : MultiTapeTM k Symbol State) (e : State ≃ State')
    (input : List Symbol) (t : ℕ) (i : Fin k) :
    (tm.relabelState e).visitedByTapeHead ((tm.relabelState e).initCfg input) t i =
      tm.visitedByTapeHead (tm.initCfg input) t i := by
  simp only [MultiTapeTM.visitedByTapeHead, MultiTapeTM.relabelState_runFrom_init,
    Cfg.mapState]

/-- **State relabelling preserves space at every time.** The relabelled machine uses exactly
the space of the original up to every time `t`. [AB09, §1.4]

**Proof sketch.** Sum `Turing.MultiTapeTM.relabelState_visitedByTapeHead` over the tapes. -/
theorem MultiTapeTM.relabelState_spaceUsed {k : ℕ} {Symbol : Type*} {State State' : Type*}
    (tm : MultiTapeTM k Symbol State) (e : State ≃ State') (input : List Symbol) (t : ℕ) :
    (tm.relabelState e).spaceUsed ((tm.relabelState e).initCfg input) t =
      tm.spaceUsed (tm.initCfg input) t := by
  simp only [MultiTapeTM.spaceUsed, MultiTapeTM.spaceUsedByTape,
    MultiTapeTM.relabelState_visitedByTapeHead]

end Turing

/-- **A coded run visits at least one cell.** Every run of a coded machine, at every time,
has visited at least one work cell: its one work head starts somewhere. So every qualifying
budget `s` of `Complexity.space_universal` is positive, which is what lets its addend-free bound
`C · s` absorb a ruler of the header's length (the RD-3 statement gate's row-1 calculation).

**Proof sketch.** The visited set of tape `0` contains the head's initial position, and it
is one summand of the space. -/
theorem CodeTM.one_le_spaceUsed (M : CodeTM) (x : List Bool) (t : ℕ) :
    1 ≤ M.toFinTM.tm.spaceUsed (M.toFinTM.tm.initCfg x) t := by
  let i₀ : Fin M.toFinTM.k := ⟨0, by simp⟩
  have hne : (M.toFinTM.tm.visitedByTapeHead (M.toFinTM.tm.initCfg x) t i₀).Nonempty :=
    ⟨_, Finset.mem_image.mpr ⟨0, Finset.mem_range.mpr (Nat.succ_pos t), rfl⟩⟩
  exact (Finset.card_pos.mpr hne).trans_le
    (M.toFinTM.tm.spaceUsedByTape_le_spaceUsed _ t i₀)

/-- **The space-faithful coding step.** Every one-work-tape binary machine `M` is equivalent
to a coded machine `M'` that halts with the same outputs at the same times
(`Turing.exists_codeTM`'s clause) **and uses the same space at every time**. This is the space
clause of the coding step named by `Complexity.space_hierarchy`'s sketch. [AB09, §1.4]

**Proof sketch.** `State` carries `Fintype`/`DecidableEq` instances and is inhabited by `q₀`,
so `Fintype.equivFinOfCardEq` gives an equivalence with `Fin (|Q| - 1 + 1)`. Relabel the states
along it (`Turing.MultiTapeTM.relabelState`): by `Turing.MultiTapeTM.relabelState_runFrom_init`
the relabelled run is the original run with renamed states, so halting and outputs correspond
at every time, and the space clause is `Turing.MultiTapeTM.relabelState_spaceUsed`. The
tape-count cast uses `hk : M.k = 1`, eliminated after destructuring the bundle. -/
theorem exists_codeTM_spaceUsed (M : FinTM Bool) (hk : M.k = 1) :
    ∃ M' : CodeTM,
      (∀ (x output : List Bool) (t : ℕ),
        M'.toFinTM.ComputesInTime x output t ↔ M.ComputesInTime x output t) ∧
      ∀ (x : List Bool) (t : ℕ),
        M'.toFinTM.tm.spaceUsed (M'.toFinTM.tm.initCfg x) t =
          M.tm.spaceUsed (M.tm.initCfg x) t := by
  classical
  rcases M with @⟨k, Q, hQ, dQ, tm⟩
  dsimp only at hk
  subst k
  letI : Fintype Q := hQ
  letI : DecidableEq Q := dQ
  have hcard : Fintype.card Q = (Fintype.card Q - 1) + 1 := by
    have : 0 < Fintype.card Q := Fintype.card_pos_iff.mpr ⟨tm.q₀⟩
    omega
  let e := Fintype.equivFinOfCardEq hcard
  refine ⟨⟨Fintype.card Q - 1, tm.relabelState e⟩, ?_, ?_⟩
  · intro x output t
    simp only [CodeTM.toFinTM, FinTM.ComputesInTime, MultiTapeTM.ComputesInTimeAndSpace,
      MultiTapeTM.relabelState_runFrom_init, Cfg.mapState, Option.map_eq_none_iff]
    constructor
    · rintro ⟨s, hhalt, hout, -⟩
      exact ⟨_, hhalt, hout, rfl⟩
    · rintro ⟨s, hhalt, hout, -⟩
      exact ⟨_, hhalt, hout, rfl⟩
  · intro x t
    exact MultiTapeTM.relabelState_spaceUsed tm e x t

end Turing

/-! ### Space at every time, and deciding through the alphabet reduction -/

/-- **The halt-time space bound holds at every time.** If `M` has halted on `x` with output
`w` by time `t`, having visited at most `s` cells by then, it has visited at most `s` cells
by **every** time `t'`: before `t` by monotonicity, after `t` because the halted
configuration is frozen. This is the all-times bridge named by `Complexity.space_hierarchy`'s
sketch (`Turing.FinTM.DecidesInSpace` bounds the space at the halt; the normal form needs every
time). [AB09, §4.1, Definition 4.1]

**Proof sketch.** Every head position visited by time `t'` was visited by time `t`: a time
`u ≤ t` is in range, and at a time `u > t` the configuration is the one at `t`
(`Turing.MultiTapeTM.runFrom_of_halt`). So each visited set at `t'` is contained in the one at
`t`. -/
theorem ComputesInTime.spaceUsed_le_of_halt {Symbol : Type} {M : FinTM Symbol}
    {x w : List Symbol} {t s : ℕ} (h : M.ComputesInTime x w t)
    (hs : M.tm.spaceUsed (M.tm.initCfg x) t ≤ s) :
    ∀ t', M.tm.spaceUsed (M.tm.initCfg x) t' ≤ s := by
  have hhalt := ((computesInTime_iff M x w t).mp h).1
  have key : ∀ t' i, M.tm.visitedByTapeHead (M.tm.initCfg x) t' i ⊆
      M.tm.visitedByTapeHead (M.tm.initCfg x) t i := by
    intro t' i z hz
    simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_image, Finset.mem_range] at hz ⊢
    obtain ⟨u, hu, rfl⟩ := hz
    by_cases hut : u ≤ t
    · exact ⟨u, by omega, rfl⟩
    · refine ⟨t, by omega, ?_⟩
      rw [show u = t + (u - t) by omega, MultiTapeTM.runFrom_add,
        MultiTapeTM.runFrom_of_halt _ hhalt]
  intro t'
  refine le_trans ?_ hs
  exact Finset.sum_le_sum fun i _ => Finset.card_le_card (key t' i)

/-- **A space-bounded computation in the normal form's hypotheses.** If `M` computes `f`
within space `s`, then it computes `f` within the length-dependent time
`n ↦ configBound M n (s n)`, and its space is at most `s |x|` at every time on every `x`.
These are exactly the hypotheses `hM` and `hS` of
`Turing.FinTM.one_work_tape_binary_spaceUsed`. [AB09, §4.1.1]

**Proof sketch.** Time: `Turing.FinTM.ComputesInTime.of_spaceUsed_le`. Space at every time:
`Turing.FinTM.ComputesInTime.spaceUsed_le_of_halt`. -/
theorem ComputesInSpace.computesFunInTime_and_forall_spaceUsed_le {M : FinTM Bool}
    {f : List Bool → List Bool} {s : ℕ → ℕ} (h : M.ComputesInSpace f s) :
    M.ComputesFunInTime f (fun n => M.configBound n (s n)) ∧
      ∀ (x : List Bool) (t : ℕ), M.tm.spaceUsed (M.tm.initCfg x) t ≤ s x.length := by
  refine ⟨fun x => ?_, fun x t => ?_⟩
  · obtain ⟨t, ht, hsp⟩ := h x
    exact ht.of_spaceUsed_le hsp
  · obtain ⟨t₀, ht, hsp⟩ := h x
    exact ht.spaceUsed_le_of_halt hsp t

/-- **Deciding through the alphabet reduction.** Let `D` be a machine over a finite alphabet
`Γ` with an embedding `e` of `Bool`. Suppose that on every `Bool` input `x` it halts with the
one-symbol output `[x ∈ L]` (embedded), and that on **every** input over `Γ` (foreign symbols
included) and at every time it has visited at most `S |x|` cells. Then some binary machine
decides `L` within space `c · (S n + 1)`. No time bound on `D` is assumed: the reduction's
time function is the maximum of the first halting times over the finitely many inputs of each
length, which need not be computable (the RD-3 gate's check 7). [AB09, §1.3.1, Claim 1.5, with
the space clause of `Turing.FinTM.alphabet_reduction_spaceUsed`]

**Proof sketch.** Choose a halting time `g x` for each `Bool` input, and let `T n` be the
maximum of `g` over the inputs of length `n` (a supremum over `Fin n → Bool`). Then `D`
computes the indicator via `e` within `T`, by monotonicity of halted runs. Apply
`Turing.FinTM.alphabet_reduction_spaceUsed`; its machine decides `L` at the halting time
`c · (T n + 1)`, within `c · (S n + 1)` cells. -/
theorem decidesInSpace_of_alphabetReduction {Γ : Type} [Fintype Γ] [DecidableEq Γ]
    (e : Bool ↪ Γ) (D : FinTM Γ) (L : Language Bool) (S : ℕ → ℕ)
    (hD : ∀ x : List Bool, ∃ t, D.ComputesInTime (x.map e)
      ([MultiTapeTM.indicator (L : Set (List Bool)) x].map e) t)
    (hS : ∀ (x : List Γ) (t : ℕ), D.tm.spaceUsed (D.tm.initCfg x) t ≤ S x.length) :
    ∃ (c : ℕ) (M : FinTM Bool), M.DecidesInSpace L fun n => c * (S n + 1) := by
  classical
  choose g hg using hD
  let T : ℕ → ℕ := fun n => (Finset.univ : Finset (Fin n → Bool)).sup fun v => g (List.ofFn v)
  have hM : D.ComputesFunInTimeVia e
      (fun x => [MultiTapeTM.indicator (L : Set (List Bool)) x]) T := by
    intro x
    have hle := Finset.le_sup (f := fun v : Fin x.length → Bool => g (List.ofFn v))
      (Finset.mem_univ x.get)
    dsimp only at hle
    rw [List.ofFn_get] at hle
    exact (hg x).mono hle
  obtain ⟨c, M', -, hM', hS'⟩ := alphabet_reduction_spaceUsed e D _ T S hM hS
  exact ⟨c, M', fun x => ⟨c * (T x.length + 1), hM' x, hS' x _⟩⟩

end FinTM

/-! ### The output summary's `Fin 3` code and its tag update -/

/-- **The output summary's `Fin 3` code** (`machine-library-design.md` §12.8a, decision O4):
`empty ↦ 0`, `accept ↦ 1`, `dead ↦ 2`. It is the record tag of the windowed runner's instance
for `Complexity.space_hierarchy` (`Turing.summaryTag`), and the summary code of the
configuration-graph vertex records below. [AB09, §4.1.1] -/
def outSummaryCode : OutSummary → Fin 3
  | .empty => 0
  | .accept => 1
  | .dead => 2

/-- The `Fin 3` code is injective.

**Proof sketch.** Compare the three cases. -/
theorem outSummaryCode_injective : Function.Injective outSummaryCode := by
  intro a b h
  cases a <;> cases b <;> simp_all [outSummaryCode]

/-- **A representative output word for each tag**: `[]` for `0`, `[true]` for `1` and
`[false]` for `2`. -/
def summaryWord (τ : Fin 3) : List Bool :=
  if τ = 0 then [] else if τ = 1 then [true] else [false]

/-- Each representative word has the tag it represents.

**Proof sketch.** Evaluate the three cases. -/
theorem outSummaryCode_summaryWord (τ : Fin 3) :
    outSummaryCode (outSummary (summaryWord τ)) = τ := by
  fin_cases τ <;> rfl

/-- **The summary tag update**: the tag after emitting `b` from an output with tag `τ`,
computed on the representative word. It is the windowed runner's tag update `upd` for the
output summary. -/
def summaryTag (τ : Fin 3) (b : Option Bool) : Fin 3 :=
  outSummaryCode (outSummary (summaryWord τ ++ b.toList))

/-- A silent step leaves the tag unchanged (the record step's `hupd` hypothesis).

**Proof sketch.** Appending nothing keeps the representative word. -/
theorem summaryTag_none (τ : Fin 3) : summaryTag τ none = τ := by
  simpa [summaryTag] using outSummaryCode_summaryWord τ

/-- **The tag of an appended output is the tag update** of the tag of the output: the
compatibility hypothesis `hσ` of the windowed run for `σ = outSummaryCode ∘ outSummary`.
[AB09, §4.1.1]

**Proof sketch.** The representative and the actual word have equal summaries, and summaries
are append-compatible (`Turing.outSummary_append_congr`). -/
theorem summaryTag_append (u : List Bool) (b : Option Bool) :
    outSummaryCode (outSummary (u ++ b.toList)) =
      summaryTag (outSummaryCode (outSummary u)) b :=
  (congrArg outSummaryCode (outSummary_append_congr
    (outSummaryCode_injective (outSummaryCode_summaryWord _)) b.toList)).symm

/-- The tag `1` (`accept`) recognizes exactly the output `[true]`, not merely an output that
begins with `true` (the RD-3 gate's check 5).

**Proof sketch.** Injectivity of the code and `Turing.outSummary_eq_accept_iff`. -/
theorem outSummaryCode_eq_one_iff (u : List Bool) :
    outSummaryCode (outSummary u) = 1 ↔ u = [true] := by
  rw [← outSummary_eq_accept_iff]
  constructor
  · intro h
    exact outSummaryCode_injective (h.trans rfl)
  · intro h
    rw [h]
    rfl

/-- The empty output has tag `0`. -/
theorem outSummaryCode_nil : outSummaryCode (outSummary []) = 0 := rfl

end Turing

/-!
# The space hierarchy theorem

[AB09, §4.1.3, Theorem 4.8] with the space-bounded universal machine of
[AB09, Exercise 4.1] (its forward clause, as the book states it), the
strictness corollary `L ⊊ PSPACE` on the p. 92 chain, and `SPACE(n+1) ≠ NP`
([AB09, Exercise 3.2], restated with the positive normalization per the P0
convention — `AroraBarakChapters3-4Plan.md` §2.4). Phase P4.3; on 2026-10-12
`space_universal` was restated forward-only and `space_hierarchy`'s sketch
moved to the window route (`machine-library-design.md` §16b item 3,
`backlog.md` RD-3; statement gate passed 2026-10-13), and on 2026-10-13
`space_universal` was strengthened to the addend-free `C · s`, pending the
statement gate of its hosts (H3, H5 and `U₀`).

## Design

* **Constant-factor overhead is the point** ([AB09]: "one can have a
  universal TM using only a constant factor of space overhead, and hence we
  don't need the logarithmic term of Theorem 3.1"): the hierarchy hypothesis
  below is eventual domination of every constant multiple — no square, no
  log — in contrast to the received `f²` time hierarchy.
* **The space-universal machine** reuses the fixed code scheme
  `Complexity.TimeHierarchy.code` (one-work-tape binary codes) and the layout
  `⟨bits s, ⟨α, x⟩⟩`, and keeps only Exercise 4.1's forward clause: a run of
  the coded machine within `s` work cells is reproduced, output included,
  within constant-factor space. Overruns and divergence are unconstrained, so
  the machine tests no budget, runs no clock and replays nothing. Its space
  bound is Exercise 4.1's `C_α · t`, with no addend: the simulated input, a
  suffix of its input, is read through H5's bounded-prefix ruler device
  (`Turing.exists_rulerHostTM`), a unary ruler of the header's length, which
  the bound absorbs because every qualifying run visits a cell (strengthened
  2026-10-13 from a `+ logSpace n` addend).
* **The hierarchy takes the window route** (design §16b item 3): the
  diagonalizer runs one fixed self-application universal machine `U₀` in a
  window of radius `a · g(n)` on the configuration-record rows
  (`Build/CfgRecord.lean`, `Build/RecordTable.lean`), stopped by a
  configuration-count clock, and flips its verdict. The window, not `U₀`'s
  code-dependent constant, bounds the diagonalizer's space. The chapter-1
  interpreter (`Turing.universalInterpreter`) serves neither machine: it
  tracks the simulated input head in unary on its fourth tape, which uses one
  plus the largest simulated input position reached, `Θ(n + 1)` cells on a
  full scan, so it cannot guarantee `O(S(n))` space uniformly for sublinear
  `S` (Finding B, recorded in the sketches).
* **Both bounds space-constructible**, as in the book; `g`'s witness drives
  the window radius and the clock width, and of `f` the proof uses only the
  bundled floor `1 ≤ logSpace n ≤ f n` (round-1 confirmed, recorded in the
  sketch).
* Facade wiring: root-wired while the P4.1 gate was live; the
  `SpaceComplexity.lean` facade has carried this module since that gate
  closed.

## Main results (all sorried; phase-P4.3 statements)

* `Complexity.space_universal` — [AB09, Exercise 4.1], forward clause
  (restated 2026-10-12; addend-free `C · s` since 2026-10-13).

/-- **The space-bounded universal machine** ([AB09, Exercise 4.1], p. 93,
forward clause; spec, fill pending — phase P4.3, restated forward-only
2026-10-12 by design §16b item 3, RD-3 statement gate passed 2026-10-13;
strengthened to the addend-free `C · s` 2026-10-13, statement gate pending):
for the fixed scheme there is a machine `SU` such that for every code `α`
there is a constant `C > 0` with: whenever the coded machine halts on `x`
with `output`, having visited at most `s` work cells by then, `SU` on input
`⟨bits s, ⟨α, x⟩⟩` halts with the same `output`, having visited at most
`C · s` work cells: the book's `C · t`. Nothing is claimed when the coded
machine overruns `s` or does not halt. The time is existential, as
`Turing.FinTM.ComputesInTime`'s halting demands, with no stated bound. Every
qualifying run has `s ≥ 1` (`Turing.CodeTM.one_le_spaceUsed`), so the bound
is never `0`.

Deviations from the book's `SU(α, t, x) = M_α(x)` within `C · t` cells:
* **`C` depends on the representation `α`**, not only on `M_α`, as in
  `Turing.universal`, whose module docstring records why an arbitrary
  effective scheme forces this.
* "Halts before using `t` cells" is read as "at most `s` visited cells"; the
  budget is an argument of `SU`, as in the book, but is never read.
* The output is the simulated output itself: with no failure clause there is
  no verdict to tag. The former `[false]` clause, and with it the
  visited-interval counters, the core-count clock and the probe/replay of the
  previous statement, went with design §16b item 3.

There is no input-device addend: the gated `+ logSpace |x| + 1` (a binary
position for the simulated input head) is replaced by a unary ruler of the
header's length, whose cells the bound absorbs (the RD-3 gate's minor 1).

**Proof sketch.** A code-first interpreter in three parts. (i) **Startup.**
Parse the header `bits s`, `α` and the two separators by a finite-state pair
scan, writing one ruler cell per header cell, so that the ruler is `true^ℓ`
with `ℓ = 2|bits s| + 2|α| + 4` (`Turing.pairEncode_pairEncode_length`,
`Turing.pairEncode_pairEncode_drop`) and the physical input head ends at
`ℓ + 1`, the virtual position `1`; the budget is never read. Canonize `α`
into the table and state tapes as `U₀`'s startup does
(`Turing.exists_selfStartupTM`'s phases, reading the code where the header
holds it). Its cells are a constant depending only on `α`
(`Turing.MultiTapeTM.spaceUsed_le_mul_succ` at the canonizer's halting time);
the effective scheme supplies no bound linear in `|α|`, and none is claimed.
(ii) **Input device and step block** (H5's bounded-prefix ruler device and
H2). Run `U₀`'s wrapped round loop (`Turing.exists_selfUniversal`'s sketch:
H2's step block on the output-erased source, with the leaf that forwards each
emission before the next round and before the halt) through the ruler host
`Turing.exists_rulerHostTM` on the virtual input `x`. The host is a positive
finite block simulation: it reads a blank and suppresses left moves at the
virtual left boundary, which it tests by restoring sweeps of `ℓ` cells
(`Turing.exists_rulerTestTM`), so `SU` halts exactly when the round loop does,
with its output, the coded machine's. The chapter-1 interpreter's device
cannot serve: its fourth tape holds a unary boundary marker whose head
`applyRecord` moves with the input head (`Turing.universalInterpreter`), so
that tape uses one plus the largest simulated input position reached,
`Θ(|x| + 1)` cells on a full scan, which a budget `s = o(|x|)` does not
cover (Finding B). (iii) **Space.** The host's bank visits exactly the round
loop's cells: the mirrored tape only the coded machine's, at most `s`, and the
table and summary tapes `O(|table| + 1)`, a constant for fixed `α`. Its ruler
visits `ℓ + 2` cells. Every qualifying run has `s ≥ 1`
(`Turing.CodeTM.one_le_spaceUsed`) and `|bits s| ≤ s`
(`Turing.length_bits_le_self`), so `ℓ ≤ 2s + 2|α| + 4 ≤ (2|α| + 6) · s`, and
the total `C_α + s + ℓ + 2` is at most `C · s`. Fill obligations, named: the
header-ruler startup with the buffered canonization; the H5 ruler host;
`U₀`'s wrapped round loop (H2 at one coded work tape, with its head
envelope); the per-code constant ledger. -/
theorem space_universal :
    ∃ SU : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, 0 < C ∧

        ((TimeHierarchy.code).decode α).toFinTM.tm.spaceUsed
            (((TimeHierarchy.code).decode α).toFinTM.tm.initCfg x) t ≤ s →

            SU.ComputesInTime (pairEncode (Nat.bits s) (pairEncode α x)) output t' ∧

/-- **The space hierarchy theorem** ([AB09, Theorem 4.8]; [SHL65] through
[AB09]; spec, fill pending — phase P4.3, sketch moved to the window route
2026-10-12 by design §16b item 3, RD-3 statement gate passed 2026-10-13): for
space-constructible `f` and `g`, if every constant multiple of `f` is
eventually below `g` (that is, `f = o(g)`), then `SPACE f ⊊ SPACE g`.
Constant-factor hypothesis — no square and no logarithmic term, by a
universal simulation with constant-factor space overhead (the
self-application universal `U₀` of the sketch, the sibling of
`Complexity.space_universal`); both constructibility hypotheses are the
book's, and the proof uses `g`'s witness and floor but, of `f`'s, only the
bundled floor (recorded here, seeded to the audit). Positivity of both bounds
is automatic from the bundled `logSpace` floor, so the P0 zero-bound
convention needs no side condition.

**Proof sketch.** *Inclusion.* `hfg` at `A = 1` gives `f n ≤ g n` from some
`N` on; below `N` each `f n` is at most `F := max_{n<N} f n` times
`g n ≥ 1` (`g`'s floor), so `f ≤ max F 1 · g` everywhere and
`Complexity.SPACE.mono_const` applies.

*The universal machine `U₀`.* One fixed machine for the scheme
`Complexity.TimeHierarchy.code`: on an input `x = pairEncode α w` it canonizes
`α` from the doubled prefix (the chapter-1 startup, `Turing.universalCanonTM`),
returns its input head to the origin, and simulates the coded machine on `x`
itself — self-application, so no `preTM` copy is needed — with its own input
head as the simulated one (H5's native device: the simulated input is all of
`U₀`'s input, so its left boundary is a real blank and no position is stored)
and the H2 step block on the canonical table. Its forward ledger: for each
`α` a constant `C_α` such that whenever the coded machine halts on `x` with
some output, having visited at most `s` cells, `U₀` halts on `x` with the
same output, having visited at most `C_α · (s + 1)` cells
(`Turing.exists_selfUniversal`). This is `space_universal`'s forward clause
with the native device in place of the bounded-prefix ruler, so no header
ruler is needed; only this direction is used.

*The diagonalizer `D`.* On input `x` with `n = |x|`, over the record alphabet:
(i) validate the input (`Turing.inputFoldTM`, R8.5d; decision O6), so that
`D`'s space is bounded on foreign rich inputs, as
`Turing.FinTM.alphabet_reduction_spaceUsed` requires, and compute
`bits (g n)` by `g`'s witness, captured on a work tape, in `O(g n)` cells;
(ii) form the radius `r = a · g n` for a fixed `a ≥ 1`
(`a = 1` suffices, the code-dependent constant being paid by the hypothesis
below; a power of two costs only low zero digits) and write `U₀`'s initial
record on `x` at radius `r`, tagged with the empty-output summary
(`Turing.exists_recordInitTM`); (iii) run the H3 windowed runner: each round
tests the record for a halted state (`Turing.exists_recordTestTM`), else
applies one step of `U₀` (`Turing.exists_recordStepTM`, whose `no` verdict
is the window exit), the tag updated by the three-valued output summary
`Turing.outSummary` (append-compatible, `Turing.outSummary_append_congr`;
coded in `Fin 3` by design §12.8a decision O4), so the emitted word is
never buffered (W1). The rounds are driven by the configuration-count clock,
the rich counter loop (`Turing.counterLoopTMVia`, §12.7 and R8.0b) on an
all-`true` word of width `W = |Q₀| + posWidth n + 3 · k₀ · (2r + 1) + 1`,
where `Q₀` and `k₀` are `U₀`'s states and tape count. Its value `2^W - 1`
exceeds `Turing.FinTM.configBound U₀ n r`, which is at most
`2^(|Q₀| + posWidth n + 3 · k₀ · (2r + 1))`, each of its four factors being
at most a power of two. (iv) Answer `false` if the runner stopped at a halted
record with summary `accept` (output exactly `[true]`,
`Turing.outSummary_eq_accept_iff`), and `true` on every other exit: halted
with another output, window exit, or clock exhaustion. `D` halts on every
input. Its cells are the witness's `O(g n)`, the record's
`k₀ · (2r + 1) + posWidth n + k₀ + 4` (`Turing.recordLength`), the record
rows' `O(record length)` scratch and the clock's `W`: all `O(g n)` with
constants depending only on `U₀` and `g`'s witness, **uniformly in the code
part of `x`**, because the window, not `U₀`'s constant, bounds the
simulation (round-1 finding 5); the `posWidth n = O(logSpace n)` terms are
absorbed by `g`'s floor. After the alphabet reduction, the language `L_D`
that `D` decides lies in `SPACE g`.

*`L_D ∉ SPACE f`.* Suppose `M` decides `L_D` within `c₀ · f n` cells. Then
`M` halts within `configBound M n (c₀ · f n)` steps
(`Turing.FinTM.ComputesInTime.of_spaceUsed_le`), and its space bound holds at
every time (`Turing.MultiTapeTM.spaceUsed_mono` up to the halt, the halted
configuration frozen after it). The space-faithful normal form
`Turing.FinTM.one_work_tape_binary_spaceUsed` gives a one-work-tape binary
`M'` within `c · (c₀ · f n + 1)` cells. Relabel it to a coded machine `N`
(`Turing.exists_codeTM`, plus the space clause it does not state:
relabelling states leaves tapes and heads untouched,
`Turing.MultiTapeTM.relabelState_runFrom_init`) and set
`α_M := code.encode N`, so that `code.decode α_M = N`. Pad the **payload**,
never the code: on `x = pairEncode α_M w`, `U₀` outputs `N`'s verdict
`[x ∈ L_D]` within `C_{α_M} · (c · (c₀ · f n + 1) + 1) ≤ A · f n` cells,
where `A := C_{α_M} · (c · c₀ + c + 1)`, using `1 ≤ f n`. The hypothesis
`hfg` at this one `A` gives `N₀` with `A · f n ≤ g n ≤ r` for `n ≥ N₀`;
take `w := List.replicate N₀ true`. Then every configuration of `U₀`'s run
on `x` up to its halt is in the radius-`r` window, since each
tape's visited cells form an interval around `0` of size at most `r`
(`Turing.MultiTapeTM.abs_pos_lt_card_visited`) and every nonblank cell was
visited (`Turing.MultiTapeTM.mem_visited_of_ne_none`); and the halt comes by
step `configBound U₀ n r` (`ComputesInTime.of_spaceUsed_le` at the bound
`r`), before the clock runs out. So the runner stops at `U₀`'s halted record,
whose summary is `accept` exactly when `x ∈ L_D`, and `D` answers the
opposite: `x ∈ L_D ↔ x ∉ L_D`, a contradiction. Fill obligations, named in
the host vocabulary of design §14: `U₀` with its forward ledger (the H2 step
block at one coded work tape, H5's native input device, the chapter-1
startup), a public row as H2's named consumer, not a private host of this
proof; the H3 windowed runner at `U₀` with the output-summary tag, over the
record rows `exists_recordInitTM` (record-table row 18; row 17,
`exists_recordTemplateTM`, beneath it), `exists_recordTestTM` and
`exists_recordStepTM`, clocked by `counterLoopTMVia`; the radius and
clock-width phases (`g`'s witness captured, radius digits, width expansion);
the window-containment lemma from the visited-interval facts; the space
clause of `exists_codeTM` and the all-times space bound of a decider; the
contradiction arithmetic; `D`'s `DecidesInSpace` packaging through the
alphabet reduction. -/

namespace Complexity

open Turing

/-- **The strengthened `space_universal` implies the gated one** with the same machine and
constant: `C · s ≤ C · (s + logSpace |x| + 1)`. The statement is the gated `space_universal`
of the worktree, extracted verbatim and renamed. -/
theorem space_universal_gated_form :
    ∃ SU : FinTM Bool, ∀ α : List Bool, ∃ C : ℕ, 0 < C ∧
      ∀ (s : ℕ) (x output : List Bool) (t : ℕ),
        ((TimeHierarchy.code).decode α).toFinTM.ComputesInTime x output t →
        ((TimeHierarchy.code).decode α).toFinTM.tm.spaceUsed
            (((TimeHierarchy.code).decode α).toFinTM.tm.initCfg x) t ≤ s →
          ∃ t' : ℕ,
            SU.ComputesInTime (pairEncode (Nat.bits s) (pairEncode α x)) output t' ∧
            SU.tm.spaceUsed
                (SU.tm.initCfg (pairEncode (Nat.bits s) (pairEncode α x))) t'
              ≤ C * (s + logSpace x.length + 1) := by
  obtain ⟨SU, hSU⟩ := space_universal
  refine ⟨SU, fun α => ?_⟩
  obtain ⟨C, hC, h⟩ := hSU α
  refine ⟨C, hC, fun s x output t h₁ h₂ => ?_⟩
  obtain ⟨t', h₃, h₄⟩ := h s x output t h₁ h₂
  exact ⟨t', h₃, h₄.trans (Nat.mul_le_mul_left C (by omega))⟩

end Complexity
