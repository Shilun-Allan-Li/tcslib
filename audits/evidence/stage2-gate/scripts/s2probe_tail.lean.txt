
/-! ## Sanity checks of the definitions (probe only; not part of any landing module) -/

namespace WordRAM.Probe

open WordRAM

/-- Read off the emitted bits after one step. -/
def outAfter {W : ℕ} : XCfg W ⊕ Bool → List Bool
  | .inl c => c.out
  | .inr _ => []

/-- Read off a cell after one step. -/
def cellAfter {W : ℕ} (a : ℤ) : XCfg W ⊕ Bool → BitVec W
  | .inl c => c.mem a
  | .inr _ => 0

/-- Read off the position after one step. -/
def pcAfter {W : ℕ} : XCfg W ⊕ Bool → ℕ
  | .inl c => c.pc
  | .inr _ => 0

/-- Read off the coin count after one step. -/
def rdAfter {W : ℕ} : XCfg W ⊕ Bool → ℕ
  | .inl c => c.rd
  | .inr _ => 0

-- accept, reject, and reject past the end
example : xstep (W := 4) [.base .accept] [] [] (XCfg.init 4 0) = .inr true := rfl
example : xstep (W := 4) [.base .reject] [] [] (XCfg.init 4 0) = .inr false := rfl
example : xstep (W := 4) [] [] [] (XCfg.init 4 0) = .inr false := rfl
-- initMem puts |x| = 3 into cell -1; emit appends "nonzero"
example : outAfter (xstep (W := 4) [.emit (-1)] [true, false, true] [] (XCfg.init 4 3)) = [true] := by
  decide
example : outAfter (xstep (W := 4) [.emit 5] [true, false, true] [] (XCfg.init 4 3)) = [false] := by
  decide
-- rdIn: [0] := x[[-1]] = x[3]; on a 4-bit input that is the last bit
example : cellAfter 0 (xstep (W := 4) [.rdIn 0 (-1)] [false, false, false, true] []
    (XCfg.init 4 3)) = 1 := by decide
-- off the input: position 3 of a 3-bit input reads 0
example : cellAfter 0 (xstep (W := 4) [.rdIn 0 (-1)] [true, true, true] [] (XCfg.init 4 3))
    = 0 := by decide
-- a negative position reads 0: [-1] holds 15 = -1 as a signed 4-bit word
example : cellAfter 0 (xstep (W := 4) [.rdIn 0 (-1)] [true] [] (XCfg.init 4 15)) = 0 := by
  decide
-- coin reads the stream at the count, false past its end, and counts
example : cellAfter 7 (xstep (W := 4) [.coin 7] [] [true] (XCfg.init 4 0)) = 1 := by decide
example : cellAfter 7 (xstep (W := 4) [.coin 7] [] [] (XCfg.init 4 0)) = 0 := by decide
example : rdAfter (xstep (W := 4) [.coin 7] [] [] (XCfg.init 4 0)) = 1 := by decide
-- half: arithmetic shift, -3 ↦ -2 (floor) and 5 ↦ 2
example : cellAfter 0 (xstep (W := 4) [.half 0 (-1)] [] []
    { pc := 0, mem := fun a => if a = -1 then BitVec.ofInt 4 (-3) else 0, out := [], rd := 0 })
    = BitVec.ofInt 4 (-2) := by decide
example : cellAfter 0 (xstep (W := 4) [.half 0 (-1)] [] [] (XCfg.init 4 5))
    = BitVec.ofInt 4 2 := by decide
-- bltz on a negative cell jumps
example : pcAfter (xstep (W := 4) [.base (.bltz (-1) 9)] [] []
    { pc := 0, mem := fun a => if a = -1 then BitVec.ofInt 4 (-3) else 0, out := [], rd := 0 })
    = 9 := by decide
-- a straight-line base instruction goes through WordRAM.effect
example : cellAfter 2 (xstep (W := 4) [.base (.add 2 (-1) (-1))] [] [] (XCfg.init 4 3))
    = BitVec.ofInt 4 6 := by decide
-- the one-instruction accepting program halts within 1 step touching no cell
example (x ρ : List Bool) (W : ℕ) : HaltsWithin [.base .accept] x ρ W 1 true [] 0 :=
  ⟨0, XCfg.init W x.length, ∅, Nat.zero_lt_one, le_rfl, rfl, rfl, rfl⟩
-- adversarial: within one step, the rejecting program does not accept
example (x ρ : List Bool) (W f : ℕ) : ¬ HaltsWithin [.base .reject] x ρ W 1 true [] f := by
  rintro ⟨n, c, K, hn, -, hrun, hv, -⟩
  obtain rfl : n = 0 := by omega
  cases hrun
  simp [xstep] at hv
-- adversarial: nothing halts within zero steps
example (P : XProgram) (x ρ : List Bool) (W f : ℕ) (v : Bool) (o : List Bool) :
    ¬ HaltsWithin P x ρ W 0 v o f := by
  rintro ⟨n, c, K, hn, -⟩
  omega
-- the projection replaces the new instructions by reject
example : XProgram.proj [.base (.one 0), .coin 1, .base .accept] = [.one 0, .reject, .accept] :=
  rfl
-- touches: load counts the address it reads from memory
example : XInstr.touches (.base (.load 0 (-1))) (initMem 4 3) = {0, -1, 3} := by decide
-- the output modes
example : outputOf false true [false, false] = [true] := rfl
example : outputOf true true [false, false] = [false, false] := rfl
-- the bounds at a sample point: TW = 0, A = 1, n = 2, W = 3, t = 4, f = 5
example : simTime 1 (fun _ => 0) 2 3 4 5 = 5 * (5 * 4 + 9 + 2 + 1) := rfl
example : simSpace 1 (fun _ => 0) 2 3 5 = 5 * 4 + 3 + Complexity.logSpace 2 + 1 := by
  simp [simSpace]

end WordRAM.Probe
