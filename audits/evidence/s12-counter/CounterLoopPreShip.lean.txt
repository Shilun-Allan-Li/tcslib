/-! ## Shared fixtures -/

def frameCfg {S : Type} (q : S) (w : List Bool) (h0 h1 : ℤ) : Cfg 2 Bool S [true, false] :=
  { state := some q, inputPos := 2, output := [false]
    workTapes := fun j p =>
      if j.val = 0 then
        (if h0 - 1 ≤ p ∧ p ≤ h0 + w.length then FinTM.bufferTape w (p - h0) else some true)
      else (if p % 3 = 0 then some false else if p % 3 = 1 then some true else none)
    workTapePos := fun j => if j.val = 0 then h0 else h1 }

def cells2 {S : Type} (c : Cfg 2 Bool S [true, false]) : List (List (Option Bool)) :=
  [ (List.range 40).map (fun n => c.workTapes 0 ((n : ℤ) - 15)),
    (List.range 40).map (fun n => c.workTapes 1 ((n : ℤ) - 15)) ]

def eq2 {S : Type} [DecidableEq S] (l r : Cfg 2 Bool S [true, false]) : Bool :=
  decide (l.state = r.state) && decide (cells2 l = cells2 r) &&
    decide (l.workTapePos 0 = r.workTapePos 0) && decide (l.workTapePos 1 = r.workTapePos 1) &&
    decide (l.inputPos = r.inputPos) && decide (l.output = r.output)

def allWords : ℕ → List (List Bool)
  | 0 => [[]]
  | n + 1 => (allWords n).flatMap fun w => [false :: w, true :: w]

def wordsUpTo (n : ℕ) : List (List Bool) := (List.range (n + 1)).flatMap allWords

def val (w : List Bool) : ℕ := Nat.ofDigits 2 (w.map Bool.toNat)

/-! ## 1. Symbol transport (`MultiTapeTM.mapWorkSymbols_runFrom`) -/

/-- Reads the input, writes it on tape 0, emits it, and moves both heads right. -/
def echoTM : MultiTapeTM 2 Bool (Fin 2) where
  q₀ := 0
  tr := fun _ inp _ =>
    match inp with
    | some b => ⟨SignType.pos, fun j => if j.val = 0 then (some (some b), SignType.pos)
        else (none, SignType.neg), some b, some 0⟩
    | none => ⟨0, fun _ => (none, 0), none, none⟩

def transportOK {S : Type} [DecidableEq S] (M : MultiTapeTM 2 Bool S)
    (c : Cfg 2 Bool S [true, false]) (T : ℕ) : Bool :=
  (List.range (T + 1)).all fun t =>
    eq2 ((M.mapWorkSymbols Equiv.boolNot).runFrom (c.mapWorkSymbols Equiv.boolNot) t)
      ((M.runFrom c t).mapWorkSymbols Equiv.boolNot)

#eval transportOK (incrementTM 2 0) (frameCfg FlagPhase.run [true, true, false, true] 3 (-2)) 14
#eval transportOK (incrementTM 2 1) (frameCfg FlagPhase.run [true] (-1) 4) 14
#eval transportOK (copyTM 2 0 1) (frameCfg SweepPhase.sweep [true, false, false] (-4) 6) 14
#eval transportOK (transferTM 2 1 0) (frameCfg SweepPhase.sweep [false, true] 2 0) 14
#eval transportOK (compareTM 2 0 1) (frameCfg FlagPhase.run [true, false] 0 0) 14
#eval transportOK echoTM (frameCfg (0 : Fin 2) [true, false] 1 (-3)) 8
-- negative control: without renaming the start configuration the identity fails
#eval (List.range 6).all fun t =>
  eq2 ((copyTM 2 0 1).mapWorkSymbols Equiv.boolNot |>.runFrom
      (frameCfg SweepPhase.sweep [true, false] 0 3) t)
    (((copyTM 2 0 1).runFrom (frameCfg SweepPhase.sweep [true, false] 0 3) t).mapWorkSymbols
      Equiv.boolNot)

/-! ## 2. Framed decrement (`decrementTM_run_succ_ofCfg`, `_underflow_ofCfg`) -/

def finDec (d : Cfg 2 Bool FlagPhase [true, false]) (flag : Bool) (len : ℕ) (v : List Bool) :
    Cfg 2 Bool FlagPhase [true, false] :=
  { d with state := some (FlagPhase.done flag),
           workTapes := fun j q =>
            if j = 0 ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (len : ℤ)
              then FinTM.bufferTape v (q - d.workTapePos j) else d.workTapes j q }

def doneAny (s : Option FlagPhase) : Bool :=
  decide (s = some (FlagPhase.done true)) || decide (s = some (FlagPhase.done false))

def framedOK (M : MultiTapeTM 2 Bool FlagPhase) (d f : Cfg 2 Bool FlagPhase [true, false])
    (T : ℕ) (lo hi : ℤ) : Bool :=
  eq2 (M.runFrom d T) f &&
    (List.range T).all (fun t => !doneAny (M.runFrom d t).state) &&
    (List.range (T + 1)).all (fun t => let c := M.runFrom d t
      decide (lo ≤ c.workTapePos 0 ∧ c.workTapePos 0 ≤ hi ∧ c.workTapePos 1 = d.workTapePos 1))

def checkDec (w : List Bool) (h0 h1 : ℤ) : Bool :=
  let d := frameCfg FlagPhase.run w h0 h1
  match decFixed w with
  | some v => let q := (w.takeWhile not).length
      framedOK (decrementTM 2 0) d (finDec d true w.length v) (2 * q + 2) (h0 - 1) (h0 + q)
  | none => framedOK (decrementTM 2 0) d (finDec d false w.length (List.replicate w.length true))
      (2 * w.length + 2) (h0 - 1) (h0 + w.length)

#eval (wordsUpTo 5).all fun w => checkDec w 3 (-2) && checkDec w (-4) 5 && checkDec w 0 0
-- the borrow cost really is carry-sensitive: q leading falses
#eval ([[true], [false, true], [false, false, true], [false, false, false]].map
  fun w => (w.takeWhile not).length)
-- negative control: one step early must fail
#eval (let w := [false, false, true]; let d := frameCfg FlagPhase.run w 3 (-2)
  framedOK (decrementTM 2 0) d (finDec d true 3 [true, true, false]) 5 2 5)
-- decFixed is incFixed conjugated by complement, and lowers the value by one
#eval (wordsUpTo 7).all fun w =>
  decide (decFixed w = (incFixed (w.map not)).map (List.map not)) &&
  (match decFixed w with
   | some v => decide (val v + 1 = val w ∧ v.length = w.length)
   | none => decide (val w = 0))

/-! ## 4. The host (`counterLoopTM_run_done`, `_escape`, `counterLoop_time_le`) -/

/-- A body with configuration-dependent rounds: from the anchor it moves right,
scans a block of `true`s, extends it by one, returns to the blank left of the
block, and steps back onto the anchor cell. It also moves the input head and
emits a `true` each round. Rounds cost 3, 4, 6, 8, … -/
def growTM : MultiTapeTM 2 Bool (Fin 3) where
  q₀ := 0
  tr := fun q _ w =>
    if q = 0 then
      ⟨SignType.pos, fun j => if j.val = 0 then (none, SignType.pos) else (none, 0), some true,
        some 1⟩
    else if q = 1 then
      match w 0 with
      | some true => ⟨0, fun j => if j.val = 0 then (none, SignType.pos) else (none, 0), none, some 1⟩
      | _ => ⟨0, fun j => if j.val = 0 then (some (some true), SignType.neg) else (none, 0),
          none, some 2⟩
    else
      match w 0 with
      | some true => ⟨0, fun j => if j.val = 0 then (none, SignType.neg) else (none, 0), none, some 2⟩
      | _ => ⟨0, fun j => if j.val = 0 then (none, SignType.pos) else (none, 0), none, some 0⟩

/-- A body with one-step rounds that marks its cell blank → false → true and
then reaches the exit `2`; round 1 also moves tape 1 right. -/
def exitTM : MultiTapeTM 2 Bool (Fin 3) where
  q₀ := 0
  tr := fun q _ w =>
    if q = 0 then
      match w 0 with
      | none => ⟨0, fun j => if j.val = 0 then (some (some false), 0) else (none, 0), none, some 0⟩
      | some false => ⟨0, fun j => if j.val = 0 then (some (some true), 0) else (none, SignType.pos),
          none, some 0⟩
      | some true => ⟨0, fun _ => (none, 0), none, some 2⟩
    else ⟨0, fun _ => (none, 0), none, some q⟩

def bodyCfg (h0 h1 : ℤ) : Cfg 2 Bool (Fin 3) [true, false] :=
  { state := some 0, inputPos := 1, output := [false]
    workTapes := fun j p =>
      if j.val = 0 then (if h0 - 2 ≤ p ∧ p ≤ h0 + 30 then none else some (decide (p % 2 = 0)))
      else (if p % 2 = 0 then some true else none)
    workTapePos := fun j => if j.val = 0 then h0 else h1 }

def counterTape (w : List Bool) (cp : ℤ) : ℤ → Option Bool := fun p =>
  if cp - 1 ≤ p ∧ p ≤ cp + w.length then FinTM.bufferTape w (p - cp)
  else if p % 2 = 0 then some true else some false

/-- First return to the anchor or the exit (or a halt): how a consumer defines `τ`. -/
def tauOf {S : Type} [DecidableEq S] (M : MultiTapeTM 2 Bool S) (a : S) (exit : Option S)
    (c : Cfg 2 Bool S [true, false]) : ℕ :=
  (((List.range 200).drop 1).find? fun u =>
    match (M.runFrom c u).state with
    | some s => decide (s = a) || decide (exit = some s)
    | none => true).getD 0

def cells3 {S : Type} (c : Cfg 3 Bool S [true, false]) : List (List (Option Bool)) :=
  (List.finRange 3).map fun j => (List.range 50).map fun n => c.workTapes j ((n : ℤ) - 20)

def eq3 {S : Type} [DecidableEq S] (l r : Cfg 3 Bool S [true, false]) : Bool :=
  decide (l.state = r.state) && decide (cells3 l = cells3 r) &&
    (List.finRange 3).all (fun j => decide (l.workTapePos j = r.workTapePos j)) &&
    decide (l.inputPos = r.inputPos) && decide (l.output = r.output)

/-- The statement's round hypotheses for rounds `r < n`, with the return clause
on rounds below `ret`. -/
def roundsOK {S : Type} [DecidableEq S] (M : MultiTapeTM 2 Bool S) (a : S) (exit : Option S)
    (c : Cfg 2 Bool S [true, false]) (n ret : ℕ) : Bool :=
  let τ := tauOf M a exit
  (List.range n).all fun r =>
    let o := counterLoopOrbit M τ c r
    decide (o.state = some a) && decide (0 < τ o) &&
      ((List.range (τ o)).drop 1).all (fun u =>
        match (M.runFrom o u).state with
        | some s => decide (s ≠ a) && decide (exit ≠ some s)
        | none => false) &&
      (decide (ret ≤ r) || decide ((counterLoopOrbit M τ c (r + 1)).state = some a))

/-- The conclusion clauses: exact final configuration at `T`, no exit earlier,
the counter head's range, and the body-head trajectory over `pairs`. -/
def hostOK {S : Type} [DecidableEq S] (M : MultiTapeTM 2 Bool S) (a : S) (exit : Option S)
    (c : Cfg 2 Bool S [true, false]) (w : List Bool) (cp : ℤ) (T : ℕ)
    (fin : Cfg 3 Bool (CounterLoopState S) [true, false]) (pairs : List (ℕ × ℕ)) : Bool :=
  let τ := tauOf M a exit
  let start := counterLoopCfg (some (.dec .run)) c (counterTape w cp) cp
  let H := counterLoopTM M a exit
  eq3 (H.runFrom start T) fin &&
    (List.range T).all (fun t => let s := (H.runFrom start t).state
      decide (s ≠ some .done) && decide (s ≠ some .escape)) &&
    (List.range (T + 1)).all (fun t => let p := (H.runFrom start t).workTapePos (Fin.last 2)
      decide (cp - 1 ≤ p ∧ p ≤ cp + w.length)) &&
    (List.range (T + 1)).all (fun t => let h := H.runFrom start t
      pairs.any fun ru => (List.finRange 2).all fun j =>
        decide (h.workTapePos j.castSucc =
          (M.runFrom (counterLoopOrbit M τ c ru.1) ru.2).workTapePos j))

def doneCheck {S : Type} [DecidableEq S] (M : MultiTapeTM 2 Bool S) (a : S) (exit : Option S)
    (c : Cfg 2 Bool S [true, false]) (w : List Bool) (cp : ℤ) (slack : ℤ := 0) : Bool × Bool :=
  let τ := tauOf M a exit
  let d := val w
  let T := ∑ r ∈ Finset.range d, τ (counterLoopOrbit M τ c r) + counterOverhead w (d + 1)
  let fin := counterLoopCfg (some .done) (counterLoopOrbit M τ c d)
    (fun q => if cp ≤ q ∧ q < cp + (w.length : ℤ) then some true else counterTape w cp q) cp
  let pairs := ((List.range d).flatMap fun r =>
      (List.range (τ (counterLoopOrbit M τ c r) + 1)).map fun u => (r, u)) ++ [(d, 0)]
  (roundsOK M a exit c d d, hostOK M a exit c w cp ((T : ℤ) + slack).toNat fin pairs)

def escapeCheck {S : Type} [DecidableEq S] (M : MultiTapeTM 2 Bool S) (a e : S)
    (c : Cfg 2 Bool S [true, false]) (w : List Bool) (cp : ℤ) (r₀ : ℕ) (shift : ℕ := 1) :
    Bool × Bool :=
  let τ := tauOf M a (some e)
  let T := ∑ r ∈ Finset.range (r₀ + 1), τ (counterLoopOrbit M τ c r) + counterOverhead w (r₀ + 1)
  let fin := counterLoopCfg (some .escape) (counterLoopOrbit M τ c (r₀ + 1))
    (fun q => if cp ≤ q ∧ q < cp + (w.length : ℤ)
      then FinTM.bufferTape (counterWord w (r₀ + shift)) (q - cp) else counterTape w cp q) cp
  let pairs := ((List.range (r₀ + 1)).flatMap fun r =>
      (List.range (τ (counterLoopOrbit M τ c r) + 1)).map fun u => (r, u)) ++ [(r₀ + 1, 0)]
  (decide (r₀ < val w) && decide (e ≠ a) && roundsOK M a (some e) c (r₀ + 1) r₀ &&
      decide ((counterLoopOrbit M τ c (r₀ + 1)).state = some e),
    hostOK M a (some e) c w cp T fin pairs)
