/-! ## The §13d additions to the pure layer (G3/G4)

The mirror, the left cascade derived from it, the scanned index, the total guarded rebalance,
the extension by one level, and the virtual tape a track represents. Nothing here is a machine;
the multi-track carrier is `Build/ZoneTape` and the machine rows are `Build/ZoneRebalance`. -/

/-- The zone words of one side: `side = true` is the right stack, `false` the left. -/
def ZoneContents.words {ℓ : ℕ} (side : Bool) (z : ZoneContents ℓ) : Fin ℓ → List (Option Bool) :=
  if side then z.right else z.left

/-- Each side's words fit their zones. -/
theorem ZoneContents.words_le {ℓ : ℕ} (side : Bool) (z : ZoneContents ℓ) (i : Fin ℓ) :
    (z.words side i).length ≤ zoneCapacity i.val := by
  cases side
  · exact z.left_le i
  · exact z.right_le i

/-- **The mirror** of zone contents: the left and right stacks exchanged, the home kept. It
reverses the represented virtual tape about the head (`Turing.zoneTrackTape_swap`). -/
def ZoneContents.swap {ℓ : ℕ} (z : ZoneContents ℓ) : ZoneContents ℓ where
  home := z.home
  left := z.right
  right := z.left
  left_le := z.right_le
  right_le := z.left_le

/-- The mirror is an involution. -/
@[simp] theorem ZoneContents.swap_swap {ℓ : ℕ} (z : ZoneContents ℓ) : z.swap.swap = z := by
  cases z
  rfl

/-- **The classical left-move rebalance at index `j`**: the mirror of `Turing.zoneCascadeRight`,
defined through `Turing.ZoneContents.swap` (design §13d, G4: "its lemmas derived by symmetry,
never copied"). [AB09, §1.7], the shift performed for a left head move. -/
def zoneCascadeLeft {ℓ : ℕ} (j : ℕ) (z : ZoneContents ℓ) : ZoneContents ℓ :=
  (zoneCascadeRight j z.swap).swap

/-- **The left cascade realizes one virtual left move** under the mirrored classical
pre-state: left zones below `j` empty, right zones below `j` full, the left donor `j`
nonempty, and the top-right room for one pass. The old home joins the right word's inner end
and the left word loses its inner cell. [AB09, §1.7]

**Proof sketch.** `Turing.zoneSide_cascadeRight` applied to the mirror; the mirror's stacks
are the original's exchanged, so its hypotheses and conclusions are these, read through
`Turing.ZoneContents.swap`. -/
theorem zoneSide_cascadeLeft {ℓ : ℕ} (j : ℕ) (hj : j < ℓ) (z : ZoneContents ℓ)
    (hl : ∀ k (hk : k < j), z.left ⟨k, by omega⟩ = [])
    (hr : ∀ k (hk : k < j), (z.right ⟨k, by omega⟩).length = zoneCapacity k)
    (hdonor : z.left ⟨j, hj⟩ ≠ [])
    (hroom : (z.right ⟨j, hj⟩).length + 2 ^ (j - 1) ≤ zoneCapacity j) :
    zoneSide (zoneCascadeLeft j z).right = z.home :: zoneSide z.right ∧
    zoneSide (zoneCascadeLeft j z).left = (zoneSide z.left).tail :=
  zoneSide_cascadeRight j hj z.swap hl hr hdonor hroom

/-- **The left cascade restores the half-full discipline below its index**: the mirror of
`Turing.zoneCascadeRight_lengths`. [AB09, §1.7]

**Proof sketch.** `Turing.zoneCascadeRight_lengths` applied to the mirror. -/
theorem zoneCascadeLeft_lengths {ℓ : ℕ} (j : ℕ) (hj : j < ℓ) (z : ZoneContents ℓ)
    (hl : ∀ k (hk : k < j), z.left ⟨k, by omega⟩ = [])
    (hr : ∀ k (hk : k < j), (z.right ⟨k, by omega⟩).length = zoneCapacity k)
    (hdonor : 2 ^ j ≤ (z.left ⟨j, hj⟩).length)
    (hroom : (z.right ⟨j, hj⟩).length + 2 ^ j ≤ zoneCapacity j) :
    (∀ k (hk : k < j),
      ((zoneCascadeLeft j z).left ⟨k, by omega⟩).length = 2 ^ k ∧
      ((zoneCascadeLeft j z).right ⟨k, by omega⟩).length = 2 ^ k) ∧
    ((zoneCascadeLeft j z).left ⟨j, hj⟩).length = (z.left ⟨j, hj⟩).length - 2 ^ j ∧
    ((zoneCascadeLeft j z).right ⟨j, hj⟩).length = (z.right ⟨j, hj⟩).length + 2 ^ j :=
  zoneCascadeRight_lengths j hj z.swap hl hr hdonor hroom

/-- The cascade in direction `dir` (`true` is a right head move). -/
def zoneCascade {ℓ : ℕ} (dir : Bool) (j : ℕ) (z : ZoneContents ℓ) : ZoneContents ℓ :=
  if dir then zoneCascadeRight j z else zoneCascadeLeft j z

/-- **The scanned index of a rebalance** (design §13d, G4 step 1): for a head move in
direction `dir`, the least level whose zone on the *feeding* side (the right stack for a right
move, the left for a left move) is nonempty; `ℓ` when every feeding zone is empty. [AB09,
§1.7]: "the smallest `i₀` such that `R_{i₀}` is not empty", the *index* of the shift. -/
def zoneShiftIndex {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) : ℕ :=
  (List.finRange ℓ).findIdx fun i => !(z.words dir i).isEmpty

/-- The scanned index is at most the number of laid-out levels. -/
theorem zoneShiftIndex_le {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) :
    zoneShiftIndex dir z ≤ ℓ := by
  have h := List.findIdx_le_length (p := fun i => !(z.words dir i).isEmpty)
    (xs := List.finRange ℓ)
  simpa [zoneShiftIndex] using h

/-- Every feeding zone below the scanned index is empty.

**Proof sketch.** `List.not_of_lt_findIdx` at the position of the level in `List.finRange`. -/
theorem zoneShiftIndex_empty {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) (i : Fin ℓ)
    (hi : i.val < zoneShiftIndex dir z) : z.words dir i = [] := by
  have h := List.not_of_lt_findIdx (p := fun i => !(z.words dir i).isEmpty)
    (xs := List.finRange ℓ) hi
  simp only [List.getElem_finRange, Bool.not_eq_eq_eq_not, Bool.not_false,
    List.isEmpty_iff] at h
  simpa using h

/-- Below `ℓ`, the feeding zone at the scanned index is nonempty.

**Proof sketch.** `List.findIdx_getElem` at the found position. -/
theorem zoneShiftIndex_nonempty {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ)
    (h : zoneShiftIndex dir z < ℓ) : z.words dir ⟨zoneShiftIndex dir z, h⟩ ≠ [] := by
  have hl : (List.finRange ℓ).findIdx (fun i => !(z.words dir i).isEmpty) <
      (List.finRange ℓ).length := by simpa [zoneShiftIndex] using h
  have h' := List.findIdx_getElem (p := fun i => !(z.words dir i).isEmpty)
    (xs := List.finRange ℓ) (w := hl)
  simp only [List.getElem_finRange, Bool.not_eq_eq_eq_not, Bool.not_true,
    List.isEmpty_eq_false_iff] at h'
  simpa [zoneShiftIndex] using h'

/-- The scanned index is `j` when every feeding zone below `j` is empty and the one at `j` is
not; the characterization the extension lemma uses.

**Proof sketch.** `List.findIdx_eq` at position `j` of `List.finRange ℓ`. -/
theorem zoneShiftIndex_eq {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) (j : ℕ) (hj : j < ℓ)
    (hbelow : ∀ i : Fin ℓ, i.val < j → z.words dir i = [])
    (hat : z.words dir ⟨j, hj⟩ ≠ []) : zoneShiftIndex dir z = j := by
  unfold zoneShiftIndex
  rw [List.findIdx_eq (by simpa using hj)]
  constructor
  · simpa [List.getElem_finRange] using hat
  · intro i hi
    simpa [List.getElem_finRange] using hbelow ⟨i, by omega⟩ hi

/-- The scanned index equals `ℓ` exactly when every feeding zone is empty.

**Proof sketch.** `List.findIdx_lt_length` and the bound `Turing.zoneShiftIndex_le`. -/
theorem zoneShiftIndex_eq_length_iff {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) :
    zoneShiftIndex dir z = ℓ ↔ ∀ i, z.words dir i = [] := by
  constructor
  · intro h i
    exact zoneShiftIndex_empty dir z i (by omega)
  · intro h
    have hle := zoneShiftIndex_le dir z
    by_contra hne
    exact zoneShiftIndex_nonempty dir z (by omega) (h _)

/-- **The total guarded rebalance** for a head move in direction `dir` (design §13d, decision
13d.3, in the A-S2 totality form): the cascade at the scanned index when that index is below
`ℓ`, the identity when every feeding zone is empty (the case the extension row serves). It is
defined for every contents; the classical fullness discipline is not assumed. [AB09, §1.7] -/
def zoneRebalance {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) : ZoneContents ℓ :=
  if zoneShiftIndex dir z < ℓ then zoneCascade dir (zoneShiftIndex dir z) z else z

/-- With no nonempty feeding zone the rebalance is the identity. -/
theorem zoneRebalance_of_le {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ)
    (h : ℓ ≤ zoneShiftIndex dir z) : zoneRebalance dir z = z := by
  simp [zoneRebalance, show ¬ zoneShiftIndex dir z < ℓ by omega]

/-- **A right rebalance realizes one virtual right move** under the classical left state at
its index (left zones below the index full, room at the index for one pass); the right-side
hypotheses of `Turing.zoneSide_cascadeRight` are supplied by the scan. [AB09, §1.7]

**Proof sketch.** Unfold the guard; the scan gives the empty lower right zones
(`Turing.zoneShiftIndex_empty`) and the nonempty donor (`Turing.zoneShiftIndex_nonempty`), and
`Turing.zoneSide_cascadeRight` finishes. -/
theorem zoneSide_rebalance_right {ℓ : ℕ} (z : ZoneContents ℓ) (h : zoneShiftIndex true z < ℓ)
    (hl : ∀ k (hk : k < zoneShiftIndex true z),
      (z.left ⟨k, by omega⟩).length = zoneCapacity k)
    (hroom : (z.left ⟨zoneShiftIndex true z, h⟩).length + 2 ^ (zoneShiftIndex true z - 1) ≤
      zoneCapacity (zoneShiftIndex true z)) :
    zoneSide (zoneRebalance true z).left = z.home :: zoneSide z.left ∧
    zoneSide (zoneRebalance true z).right = (zoneSide z.right).tail := by
  have hr : ∀ k (hk : k < zoneShiftIndex true z), z.right ⟨k, by omega⟩ = [] :=
    fun k hk => zoneShiftIndex_empty true z ⟨k, by omega⟩ hk
  have hd : z.right ⟨zoneShiftIndex true z, h⟩ ≠ [] := zoneShiftIndex_nonempty true z h
  simp only [zoneRebalance, if_pos h, zoneCascade, if_true]
  exact zoneSide_cascadeRight _ h z hr hl hd hroom

/-- The rebalance in direction `false` is the mirror of the rebalance in direction `true`. -/
theorem zoneRebalance_false {ℓ : ℕ} (z : ZoneContents ℓ) :
    zoneRebalance false z = (zoneRebalance true z.swap).swap := by
  have hi : zoneShiftIndex false z = zoneShiftIndex true z.swap := rfl
  by_cases h : zoneShiftIndex true z.swap < ℓ <;>
    simp [zoneRebalance, zoneCascade, zoneCascadeLeft, hi, h]

/-- **The cascade keeps every level above its index** (a promotion: the private
`zoneCascadeRight_frame` of `Build/Zone.lean`, made public at landing with its existing proof;
in this unit its statement is sorried). Levels strictly above `j` are untouched on both sides.

**Proof sketch.** Induction on `j`: the base move touches level `0` only, and each outer pair
`zoneStepPair (j + 1)` touches levels `j` and `j + 1` only. -/
theorem zoneCascadeRight_frame {ℓ : ℕ} (j : ℕ) (z : ZoneContents ℓ)
    (k : Fin ℓ) (hk : j < k.val) :
    (zoneCascadeRight j z).left k = z.left k ∧
      (zoneCascadeRight j z).right k = z.right k := by
  sorry

/-- **The rebalance keeps every level above its scanned index**, on both sides and in both
directions: the zones a rebalance may change are those of levels `0` to the index.

**Proof sketch.** A guarded identity changes nothing; otherwise this is
`Turing.zoneCascadeRight_frame` at the index, read through the mirror for `dir = false`. -/
theorem zoneRebalance_frame {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ) (k : Fin ℓ)
    (hk : zoneShiftIndex dir z < k.val) :
    (zoneRebalance dir z).left k = z.left k ∧ (zoneRebalance dir z).right k = z.right k := by
  cases dir
  · have hk' : zoneShiftIndex true z.swap < k.val := hk
    rw [zoneRebalance_false]
    unfold zoneRebalance
    split_ifs with h
    · simp only [zoneCascade, if_true]
      exact (zoneCascadeRight_frame _ z.swap k hk').symm
    · exact ⟨rfl, rfl⟩
  · unfold zoneRebalance
    split_ifs with h
    · simp only [zoneCascade, if_true]
      exact zoneCascadeRight_frame _ z k hk
    · exact ⟨rfl, rfl⟩

/-- **The extension by one level** (design §13d, decision 13d.2): the contents with level `ℓ`
laid out on both sides, each new zone **half full of virtual blanks** (`2^ℓ` stored blank
cells, the inner half of capacity `2 · 2^ℓ`), every other field unchanged. [AB09, §1.7]:
zones are made half full "the first time we encounter it". -/
def ZoneContents.extend {ℓ : ℕ} (z : ZoneContents ℓ) : ZoneContents (ℓ + 1) where
  home := z.home
  left := Fin.lastCases (List.replicate (2 ^ ℓ) none) z.left
  right := Fin.lastCases (List.replicate (2 ^ ℓ) none) z.right
  left_le := by
    intro i
    refine Fin.lastCases ?_ (fun j => ?_) i
    · simp only [Fin.lastCases_last, List.length_replicate, Fin.val_last, zoneCapacity]
      omega
    · simpa only [Fin.lastCases_castSucc, Fin.coe_castSucc] using z.left_le j
  right_le := by
    intro i
    refine Fin.lastCases ?_ (fun j => ?_) i
    · simp only [Fin.lastCases_last, List.length_replicate, Fin.val_last, zoneCapacity]
      omega
    · simpa only [Fin.lastCases_castSucc, Fin.coe_castSucc] using z.right_le j

/-- The extension keeps the old levels' words. -/
theorem ZoneContents.extend_words_castSucc {ℓ : ℕ} (side : Bool) (z : ZoneContents ℓ)
    (i : Fin ℓ) : z.extend.words side i.castSucc = z.words side i := by
  cases side <;> simp [ZoneContents.words, ZoneContents.extend]

/-- The extension's new level holds `2^ℓ` stored blank cells on each side. -/
theorem ZoneContents.extend_words_last {ℓ : ℕ} (side : Bool) (z : ZoneContents ℓ) :
    z.extend.words side (Fin.last ℓ) = List.replicate (2 ^ ℓ) none := by
  cases side <;> simp [ZoneContents.words, ZoneContents.extend]

/-- The extension pads each represented half-word with `2^ℓ` virtual blanks at its outer end.

**Proof sketch.** `List.finRange_succ_last` splits the levels into the old ones, embedded by
`Fin.castSucc`, and the new last one; the concatenation splits accordingly. -/
theorem zoneSide_extend {ℓ : ℕ} (side : Bool) (z : ZoneContents ℓ) :
    zoneSide (z.extend.words side) = zoneSide (z.words side) ++ List.replicate (2 ^ ℓ) none := by
  unfold zoneSide
  rw [List.finRange_succ_last, List.flatMap_append, List.flatMap_map]
  simp [ZoneContents.extend_words_castSucc, ZoneContents.extend_words_last]

/-- **After a short scan, the extended contents scan to the new level**: when every feeding
zone of `z` is empty (scanned index `ℓ`), the extension's scanned index is `ℓ`, now below the
level count `ℓ + 1`, since the new feeding zone holds `2^ℓ ≥ 1` blank cells.

**Proof sketch.** `Turing.zoneShiftIndex_eq` at `ℓ`: the old levels are the old zones, all
empty by `Turing.zoneShiftIndex_eq_length_iff`, and the new one is a nonempty replicate. -/
theorem zoneShiftIndex_extend {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ)
    (h : zoneShiftIndex dir z = ℓ) : zoneShiftIndex dir z.extend = ℓ := by
  have hall := (zoneShiftIndex_eq_length_iff dir z).mp h
  apply zoneShiftIndex_eq dir z.extend ℓ (by omega)
  · intro i hi
    have : i = (⟨i.val, hi⟩ : Fin ℓ).castSucc := Fin.ext rfl
    rw [this, ZoneContents.extend_words_castSucc]
    exact hall _
  · have : (⟨ℓ, by omega⟩ : Fin (ℓ + 1)) = Fin.last ℓ := rfl
    rw [this, ZoneContents.extend_words_last]
    exact List.ne_nil_of_length_pos (by rw [List.length_replicate]; exact Nat.two_pow_pos ℓ)

/-- The extension does not move a scanned index that is already below `ℓ`.

**Proof sketch.** `Turing.zoneShiftIndex_eq` at the old index: the zones up to it are the old
ones. -/
theorem zoneShiftIndex_extend_of_lt {ℓ : ℕ} (dir : Bool) (z : ZoneContents ℓ)
    (h : zoneShiftIndex dir z < ℓ) : zoneShiftIndex dir z.extend = zoneShiftIndex dir z := by
  apply zoneShiftIndex_eq dir z.extend _ (by omega)
  · intro i hi
    have : i = (⟨i.val, by omega⟩ : Fin ℓ).castSucc := Fin.ext rfl
    rw [this, ZoneContents.extend_words_castSucc]
    exact zoneShiftIndex_empty dir z _ hi
  · have : (⟨zoneShiftIndex dir z, by omega⟩ : Fin (ℓ + 1)) = (⟨_, h⟩ : Fin ℓ).castSucc :=
      Fin.ext rfl
    rw [this, ZoneContents.extend_words_castSucc]
    exact zoneShiftIndex_nonempty dir z h

/-- **The virtual tape a track represents** (design §13d, G3), relative to its head: position
`0` is the home cell, positive positions read the right half-word inner end first, negative
positions the left one; a stored virtual blank (`none` in a zone word) and a position beyond the
stored words both read as blank. [AB09, §1.7]: the simulated tape, read off the zones. -/
def zoneTrackTape {ℓ : ℕ} (z : ZoneContents ℓ) (p : ℤ) : Option Bool :=
  if p = 0 then z.home
  else if 0 < p then ((zoneSide z.right)[(p - 1).toNat]?).join
  else ((zoneSide z.left)[(-p - 1).toNat]?).join

/-- Empty contents represent the blank tape. -/
theorem zoneTrackTape_empty (ℓ : ℕ) : zoneTrackTape (ZoneContents.empty ℓ) = fun _ => none := by
  funext p
  simp [zoneTrackTape, ZoneContents.empty, zoneSide]

/-- The mirror reverses the represented tape about the head.

**Proof sketch.** Unfold; the two half-words exchange and `-p` exchanges the sign cases. -/
theorem zoneTrackTape_swap {ℓ : ℕ} (z : ZoneContents ℓ) (p : ℤ) :
    zoneTrackTape z.swap p = zoneTrackTape z (-p) := by
  unfold zoneTrackTape
  by_cases h0 : p = 0
  · subst h0
    simp [ZoneContents.swap]
  · by_cases hp : 0 < p
    · simp only [h0, hp, if_false, if_true, neg_eq_zero, show ¬ 0 < -p by omega]
      rw [neg_neg]
      rfl
    · simp only [h0, hp, if_false, neg_eq_zero, show 0 < -p by omega, if_true]
      rw [show -p - 1 = -p - 1 from rfl]
      rfl

/-- **The extension does not change the represented tape**: the padding cells are virtual
blanks at the outer ends.

**Proof sketch.** `Turing.zoneSide_extend`; a lookup in the padded word is the old lookup
inside the old word and a stored blank beyond it, and both read as blank there. -/
theorem zoneTrackTape_extend {ℓ : ℕ} (z : ZoneContents ℓ) :
    zoneTrackTape z.extend = zoneTrackTape z := by
  have hr := zoneSide_extend true z
  have hl := zoneSide_extend false z
  simp only [ZoneContents.words, if_true, Bool.false_eq_true, if_false] at hr hl
  have key : ∀ (w : List (Option Bool)) (n : ℕ),
      ((w ++ List.replicate (2 ^ ℓ) none)[n]?).join = (w[n]?).join := by
    intro w n
    by_cases hn : n < w.length
    · rw [List.getElem?_append_left hn]
    · rw [List.getElem?_append_right (by omega), List.getElem?_replicate,
        List.getElem?_eq_none (show w.length ≤ n by omega)]
      split_ifs <;> rfl
  funext p
  unfold zoneTrackTape
  rw [hr, hl, key, key]
  rfl

/-- Writing the home changes the represented tape at the head only. -/
theorem zoneTrackTape_homeWrite {ℓ : ℕ} (z : ZoneContents ℓ) (v : Option Bool) (p : ℤ) :
    zoneTrackTape (zoneHomeWrite z v) p = if p = 0 then v else zoneTrackTape z p := by
  unfold zoneTrackTape zoneHomeWrite
  split_ifs <;> rfl

/-- **The cascade's new home is the right word's first cell** under the classical pre-state of
`Turing.zoneSide_cascadeRight`. This is the one fact about the cascade's internals that the
represented-tape statement needs beyond the landed lemmas. [AB09, §1.7]: the leftmost symbol of
`R_{i₀}` is put in position `0`.

**Proof sketch.** The shift pairs keep the home (both wrappers copy it), so the home after the
cascade is the one the central move sets: the head of zone `0` on the right after the
descending pass. That pass keeps the right concatenation (`Turing.zoneSide_shiftInW`) and
leaves zone `0` nonempty (each descending prefix of the nonempty donor is nonempty, as in the
proof of `Turing.zoneSide_cascadeRight`), and zone `0` is the first word of the
concatenation, so its head is the right word's head. -/
theorem zoneCascadeRight_home {ℓ : ℕ} (j : ℕ) (hj : j < ℓ) (z : ZoneContents ℓ)
    (hr : ∀ k (hk : k < j), z.right ⟨k, by omega⟩ = [])
    (hl : ∀ k (hk : k < j), (z.left ⟨k, by omega⟩).length = zoneCapacity k)
    (hdonor : z.right ⟨j, hj⟩ ≠ [])
    (hroom : (z.left ⟨j, hj⟩).length + 2 ^ (j - 1) ≤ zoneCapacity j) :
    (zoneCascadeRight j z).home = (zoneSide z.right).headI := by
  sorry

/-- **The cascade realizes the virtual right move on the represented tape**: under the
classical pre-state of `Turing.zoneSide_cascadeRight`, the tape the cascade represents is the
old one read one cell further right. [AB09, §1.7]

**Proof sketch.** Positions `p > 0` read the right word's tail one place earlier, `p = 0` the
new home, which is the right word's head (`Turing.zoneCascadeRight_home`; the right word is
nonempty because it contains the donor), and `p < 0` the left word with the old home pushed
on (`Turing.zoneSide_cascadeRight`). -/
theorem zoneTrackTape_cascadeRight {ℓ : ℕ} (j : ℕ) (hj : j < ℓ) (z : ZoneContents ℓ)
    (hr : ∀ k (hk : k < j), z.right ⟨k, by omega⟩ = [])
    (hl : ∀ k (hk : k < j), (z.left ⟨k, by omega⟩).length = zoneCapacity k)
    (hdonor : z.right ⟨j, hj⟩ ≠ [])
    (hroom : (z.left ⟨j, hj⟩).length + 2 ^ (j - 1) ≤ zoneCapacity j) (p : ℤ) :
    zoneTrackTape (zoneCascadeRight j z) p = zoneTrackTape z (p + 1) := by
  obtain ⟨hL, hR⟩ := zoneSide_cascadeRight j hj z hr hl hdonor hroom
  have hH := zoneCascadeRight_home j hj z hr hl hdonor hroom
  have hne : zoneSide z.right ≠ [] := by
    intro he
    apply hdonor
    have hmem : z.right ⟨j, hj⟩ ⊆ zoneSide z.right := by
      intro a ha
      exact List.mem_flatMap.mpr ⟨⟨j, hj⟩, List.mem_finRange _, ha⟩
    rw [he] at hmem
    exact List.eq_nil_iff_forall_not_mem.mpr fun a ha => by simpa using hmem ha
  obtain ⟨c, cs, hcs⟩ := List.exists_cons_of_ne_nil hne
  unfold zoneTrackTape
  rw [hL, hR, hH, hcs]
  by_cases h0 : p = 0
  · subst h0
    simp
  · by_cases hp : 0 < p
    · simp only [h0, hp, if_false, if_true, show p + 1 ≠ 0 by omega, show 0 < p + 1 by omega,
        List.tail_cons]
      rw [show (p + 1 - 1).toNat = (p - 1).toNat + 1 by omega, List.getElem?_cons_succ]
    · by_cases hm : p = -1
      · subst hm
        simp
      · simp only [h0, hp, if_false, show p + 1 ≠ 0 by omega, show ¬ 0 < p + 1 by omega]
        rw [show (-p - 1).toNat = (-(p + 1) - 1).toNat + 1 by omega, List.getElem?_cons_succ]

/-- **The left cascade realizes the virtual left move on the represented tape**: the mirror of
`Turing.zoneTrackTape_cascadeRight`. [AB09, §1.7]

**Proof sketch.** Through the mirror (`Turing.zoneTrackTape_swap`): the left cascade is the
right cascade of the mirror, read back through the mirror. -/
theorem zoneTrackTape_cascadeLeft {ℓ : ℕ} (j : ℕ) (hj : j < ℓ) (z : ZoneContents ℓ)
    (hl : ∀ k (hk : k < j), z.left ⟨k, by omega⟩ = [])
    (hr : ∀ k (hk : k < j), (z.right ⟨k, by omega⟩).length = zoneCapacity k)
    (hdonor : z.left ⟨j, hj⟩ ≠ [])
    (hroom : (z.right ⟨j, hj⟩).length + 2 ^ (j - 1) ≤ zoneCapacity j) (p : ℤ) :
    zoneTrackTape (zoneCascadeLeft j z) p = zoneTrackTape z (p - 1) := by
  have h := zoneTrackTape_cascadeRight j hj z.swap hl hr hdonor hroom (-p)
  unfold zoneCascadeLeft
  rw [zoneTrackTape_swap, h, zoneTrackTape_swap]
  congr 1
  omega

/-- **A left rebalance realizes one virtual left move** under the classical right state at its
index: the mirror of `Turing.zoneSide_rebalance_right`. [AB09, §1.7]

**Proof sketch.** `Turing.zoneRebalance_false` writes the left rebalance as the mirrored right
one; `Turing.zoneSide_rebalance_right` applies to the mirror. -/
theorem zoneSide_rebalance_left {ℓ : ℕ} (z : ZoneContents ℓ) (h : zoneShiftIndex false z < ℓ)
    (hr : ∀ k (hk : k < zoneShiftIndex false z),
      (z.right ⟨k, by omega⟩).length = zoneCapacity k)
    (hroom : (z.right ⟨zoneShiftIndex false z, h⟩).length + 2 ^ (zoneShiftIndex false z - 1) ≤
      zoneCapacity (zoneShiftIndex false z)) :
    zoneSide (zoneRebalance false z).right = z.home :: zoneSide z.right ∧
    zoneSide (zoneRebalance false z).left = (zoneSide z.left).tail := by
  rw [zoneRebalance_false]
  exact zoneSide_rebalance_right z.swap h hr hroom

/-- **A right rebalance moves the represented tape one cell**: under the classical left state
at its scanned index (below `ℓ`), the tape the rebalanced track represents is the old one read
one cell further right. This is the per-track honesty statement the Hennie–Stearns assembly
consumes. [AB09, §1.7]

**Proof sketch.** Unfold the guard; the scan supplies the right-side hypotheses
(`Turing.zoneShiftIndex_empty`, `Turing.zoneShiftIndex_nonempty`) and
`Turing.zoneTrackTape_cascadeRight` finishes. -/
theorem zoneTrackTape_rebalance_right {ℓ : ℕ} (z : ZoneContents ℓ)
    (h : zoneShiftIndex true z < ℓ)
    (hl : ∀ k (hk : k < zoneShiftIndex true z),
      (z.left ⟨k, by omega⟩).length = zoneCapacity k)
    (hroom : (z.left ⟨zoneShiftIndex true z, h⟩).length + 2 ^ (zoneShiftIndex true z - 1) ≤
      zoneCapacity (zoneShiftIndex true z)) (p : ℤ) :
    zoneTrackTape (zoneRebalance true z) p = zoneTrackTape z (p + 1) := by
  have hr : ∀ k (hk : k < zoneShiftIndex true z), z.right ⟨k, by omega⟩ = [] :=
    fun k hk => zoneShiftIndex_empty true z ⟨k, by omega⟩ hk
  have hd : z.right ⟨zoneShiftIndex true z, h⟩ ≠ [] := zoneShiftIndex_nonempty true z h
  simp only [zoneRebalance, if_pos h, zoneCascade, if_true]
  exact zoneTrackTape_cascadeRight _ h z hr hl hd hroom p

/-- **A left rebalance moves the represented tape one cell the other way**: the mirror of
`Turing.zoneTrackTape_rebalance_right`. [AB09, §1.7]

**Proof sketch.** Through `Turing.zoneRebalance_false` and `Turing.zoneTrackTape_swap`. -/
theorem zoneTrackTape_rebalance_left {ℓ : ℕ} (z : ZoneContents ℓ)
    (h : zoneShiftIndex false z < ℓ)
    (hr : ∀ k (hk : k < zoneShiftIndex false z),
      (z.right ⟨k, by omega⟩).length = zoneCapacity k)
    (hroom : (z.right ⟨zoneShiftIndex false z, h⟩).length + 2 ^ (zoneShiftIndex false z - 1) ≤
      zoneCapacity (zoneShiftIndex false z)) (p : ℤ) :
    zoneTrackTape (zoneRebalance false z) p = zoneTrackTape z (p - 1) := by
  rw [zoneRebalance_false, zoneTrackTape_swap,
    zoneTrackTape_rebalance_right z.swap h hr hroom, zoneTrackTape_swap]
  congr 1
  omega

end Turing

/-! ### §13d G2 — mapped sweeps, the cross-tape rows beside the scan

The one-tape core of G2 is the scan `Turing.scanTM` above (decision Q7, 2026-10-11). The two
rows below move **one component** of each cell across tapes, which whole-symbol copying cannot
do: they read each source cell through `extract : Symbol → Δ` and write
`merge (destination cell) (extract source cell)` into the destination, where the destination
cell may be blank (`merge : Option Symbol → Δ → Symbol`). **The tables are written once**: the
plain rows `Turing.transferTM` and `Turing.copyTM` are *defined* as these rows at the identity
map (`extract := id`, `merge := fun _ s => s`, and for the transfer the blank residue), so they
are definitional instances (`Turing.copyMapTM_plain`, `Turing.transferMapTM_plain`, by `rfl`).
The fills of the mapped framed rows share the two-pass sweep trace `Turing.catalog_trace_run`
that every sweep row cites (one trace per routine family, as §12.6 established). (Landing: this

/-- **§13d G2, the mapped copy** (design §13d, G2; [Bon26], the copy row's shape). Copy the word
stored on tape `src` onto tape `dst` through a map. For distinct tapes (`src ≠ dst`, the
hypothesis of the framed theorem) the forward pass reads each source cell `γ` and the
destination cell `c` under the destination head (`c` may be blank), writes
`merge c (extract γ)` there, and moves both heads right; the source's right blank turns both
heads; the return pass, guided by the intact source, brings both back; the left blank steps them
right into the live `done` anchor. The source is then never written. `Turing.copyTM` is this row
at `extract := id` and `merge := fun _ s => s`. -/
def copyMapTM {Symbol : Type*} {Δ : Type*} (k : ℕ) (src dst : Fin k) (extract : Symbol → Δ)
    (merge : Option Symbol → Δ → Symbol) : MultiTapeTM k Symbol SweepPhase where
  q₀ := .sweep
  tr := fun q _ w =>
    match q with
    | .sweep =>
      match w src with
      | some b =>
        ⟨0, fun j => if j = dst then (some (some (merge (w dst) (extract b))), SignType.pos)
            else if j = src then (none, SignType.pos) else (none, 0),
          none, some .sweep⟩
      | none =>
        ⟨0, fun j => if j = src ∨ j = dst then (none, SignType.neg)
            else (none, 0), none, some .rewind⟩
    | .rewind =>
      match w src with
      | some _ =>
        ⟨0, fun j => if j = src ∨ j = dst then (none, SignType.neg)
            else (none, 0), none, some .rewind⟩
      | none =>
        ⟨0, fun j => if j = src ∨ j = dst then (none, SignType.pos)
            else (none, 0), none, some .done⟩
    | .done => ⟨0, fun _ => (none, 0), none, some .done⟩

/-- **§13d G2, the mapped transfer** (design §13d, G2; [Bon26], the transfer row's shape). As
`Turing.copyMapTM` (again for `src ≠ dst`), except that the return pass also rewrites each source
cell `γ` to its residue `rest γ` (`none` blanks it). With a residue that keeps a cell's other
components (the per-track case), the source word stays a delimited word. `Turing.transferTM` is
this row at `extract := id`, `merge := fun _ s => s` and `rest := fun _ => none`. -/
def transferMapTM {Symbol : Type*} {Δ : Type*} (k : ℕ) (src dst : Fin k) (extract : Symbol → Δ)
    (merge : Option Symbol → Δ → Symbol) (rest : Symbol → Option Symbol) :
    MultiTapeTM k Symbol SweepPhase where
  q₀ := .sweep
  tr := fun q _ w =>
    match q with
    | .sweep =>
      match w src with
      | some b =>
        ⟨0, fun j => if j = dst then (some (some (merge (w dst) (extract b))), SignType.pos)
            else if j = src then (none, SignType.pos) else (none, 0),
          none, some .sweep⟩
      | none =>
        ⟨0, fun j => if j = src ∨ j = dst then (none, SignType.neg)
            else (none, 0), none, some .rewind⟩
    | .rewind =>
      match w src with
      | some b =>
        ⟨0, fun j => if j = src then (some (rest b), SignType.neg)
            else if j = dst then (none, SignType.neg) else (none, 0),
          none, some .rewind⟩
      | none =>
        ⟨0, fun j => if j = src ∨ j = dst then (none, SignType.pos)
            else (none, 0), none, some .done⟩
    | .done => ⟨0, fun _ => (none, 0), none, some .done⟩

end Turing

/-! ### One constructor for the framed sweep rows (§13d G2, shared with R3 clear)

Every two-pass sweep row of the §12.6 shape (`Turing.clearTM_run_ofCfg`, the mapped copy and the
mapped transfer) gives a contract in the same way: the index is the word read from one tape, the
time `2|w| + 2`, the exit `done`, the touched heads within one cell of the word interval and the
others fixed. `Turing.sweepContract` builds it once from the row's framed theorem, and
`Turing.sweepPost` is the endpoint shape they share (an interval of one tape overwritten); the
instances supply only the machine, the read tape, the touched tapes and the written values.
(Landing: this block is inserted before `Turing.clearTM_contract`, which is then defined through
it.) -/

/-- **The endpoint of a sweep row**: `d` at the exit `done`, with the word interval
`[pos, pos + |w|)` of tape `a` (from its head) overwritten cell by cell, cell `q` by `v w d j q`,
and every other cell, all work heads, the input position and the output unchanged. -/
def sweepPost {S : Type} {x : List Γ} (done : S) (a : Fin k)
    (v : List Γ → Cfg k Γ S x → Fin k → ℤ → Option Γ) (w : List Γ) (d : Cfg k Γ S x) :
    Cfg k Γ S x :=
  { d with
    state := some done
    workTapes := fun j q =>
      if j = a ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (w.length : ℤ) then v w d j q
      else d.workTapes j q }

/-- **The contract of a framed two-pass sweep row** (§12.6 shape; design §13d, G2). Given a
machine `M`, its entry and exit `done`, the tape `src` it reads, the predicate `touched` of the
tapes whose heads move, an endpoint `post` at `done`, and the row's framed theorem `hrun` (from
the entry, with a word `w` delimited at the head of `src`: the run reaches `post w d` at exactly
`2|w| + 2`, no `done` earlier, touched heads within `[pos - 1, pos + |w|]`, the others fixed), this
is the contract with index `w`, that precondition, time `2|w| + 2`, endpoint `post`, exits
`{some done}`, and those envelopes. [AFM26], design adaptation (full entry in
`Build/Contract/Basic`): a specification used in place of the run.

**Proof sketch.** Each field is the matching clause of `hrun`; the exit is `hpost`; a fixed head
lies in its one-point interval. -/
def sweepContract {S : Type} (M : MultiTapeTM k Γ S) (entry done : S) (src : Fin k)
    (touched : Fin k → Prop) [DecidablePred touched] (x : List Γ)
    (post : List Γ → Cfg k Γ S x → Cfg k Γ S x) (hpost : ∀ w d, (post w d).state = some done)
    (hrun : ∀ (w : List Γ) (d : Cfg k Γ S x), d.state = some entry →
      (∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
        d.workTapes src (d.workTapePos src + p) = FinTM.bufferTape w p) →
      M.runFrom d (2 * w.length + 2) = post w d ∧
      (∀ t < 2 * w.length + 2, (M.runFrom d t).state ≠ some done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * w.length + 2 →
        if touched j then
          (M.runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (w.length : ℤ))
        else (M.runFrom d t).workTapePos j = d.workTapePos j)) :
    Contract M entry {some done} (List Γ) x where
  pre w d := ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
    d.workTapes src (d.workTapePos src + p) = FinTM.bufferTape w p
  time w _ := 2 * w.length + 2
  post := post
  lo w d j := if touched j then d.workTapePos j - 1 else d.workTapePos j
  hi w d j := if touched j then d.workTapePos j + w.length else d.workTapePos j
  run w d hd hp := (hrun w d hd hp).1
  exit_mem w d _ _ := by
    show (post w d).state ∈ ({some done} : Set (Option S))
    rw [hpost]
    exact Set.mem_singleton _
  cut w d hd hp u hu h := (hrun w d hd hp).2.1 u hu (Set.mem_singleton_iff.mp h)
  env w d hd hp u hu j := by
    have h := (hrun w d hd hp).2.2 j u hu
    by_cases hj : touched j
    · simpa only [hj, if_true] using h
    · simp only [hj, if_false] at h ⊢
      rw [h]
      exact Finset.mem_Icc.mpr ⟨le_rfl, le_rfl⟩

/-- **The joint trajectory of every framed sweep contract**: every untouched head stays at its
start, a relation on the whole head vector.

**Proof sketch.** An untouched head's envelope is the one-point interval at its start. -/
theorem sweepContract_traj {S : Type} (M : MultiTapeTM k Γ S) (entry done : S) (src : Fin k)
    (touched : Fin k → Prop) [DecidablePred touched] (x : List Γ)
    (post : List Γ → Cfg k Γ S x → Cfg k Γ S x) (hpost : ∀ w d, (post w d).state = some done)
    (hrun : ∀ (w : List Γ) (d : Cfg k Γ S x), d.state = some entry →
      (∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
        d.workTapes src (d.workTapePos src + p) = FinTM.bufferTape w p) →
      M.runFrom d (2 * w.length + 2) = post w d ∧
      (∀ t < 2 * w.length + 2, (M.runFrom d t).state ≠ some done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * w.length + 2 →
        if touched j then
          (M.runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (w.length : ℤ))
        else (M.runFrom d t).workTapePos j = d.workTapePos j)) :
    (sweepContract M entry done src touched x post hpost hrun).Traj
      fun _ d p => ∀ j, ¬ touched j → p j = d.workTapePos j := by
  intro w d hd hp u hu j hj
  have h := (sweepContract M entry done src touched x post hpost hrun).env w d hd hp u hu j
  have hlo : (sweepContract M entry done src touched x post hpost hrun).lo w d j =
      d.workTapePos j := if_neg hj
  have hhi : (sweepContract M entry done src touched x post hpost hrun).hi w d j =
      d.workTapePos j := if_neg hj
  rw [hlo, hhi, Finset.mem_Icc] at h
  exact le_antisymm h.2 h.1

/-- **A cell outside an overwritten interval keeps its value**: the frame of every sweep row,
whose endpoint overwrites `[lo, lo + n)` on tape `a` (with any values) and keeps every other
cell. If `q` lies outside that footprint, the endpoint's cell is the start's. -/
theorem sweep_ite_keep {α : Type} {j a : Fin k} {lo : ℤ} {n : ℕ} {q : ℤ} (v f : α)
    (hq : q ∉ (if j = a then Set.Ico lo (lo + n) else ∅)) :
    (if j = a ∧ lo ≤ q ∧ q < lo + n then v else f) = f := by
  rw [if_neg]
  rintro ⟨rfl, h₁, h₂⟩
  rw [if_pos rfl] at hq
  exact hq ⟨h₁, h₂⟩

end Turing

/-! ### §13d G2: the mapped sweeps as contracts

The two G2 rows of `Build/Catalog` stated as `Turing.Contract`s from the start (design §16b,
item 5), through the shared constructor `Turing.sweepContract` and endpoint `Turing.sweepPost`. -/

/-- The values a mapped sweep writes into its destination interval: at cell `q`, the merge of the
cell's original content with the extract of the source word's symbol at offset `q - pos`. -/
def sweepMapWrite {S : Type} {x : List Γ} {Δ : Type} (extract : Γ → Δ) (merge : Option Γ → Δ → Γ)
    (w : List Γ) (d : Cfg k Γ S x) (j : Fin k) (q : ℤ) : Option Γ :=
  (FinTM.bufferTape w (q - d.workTapePos j)).map fun γ => merge (d.workTapes j q) (extract γ)

/-- **The mapped copy as a contract** (design §13d, G2): the index is the source word `w`.
Started in the sweep phase with `w`, delimited, at the head of tape `src`, the mapped copy
reaches `done` in exactly `2|w| + 2` steps, and not earlier, with the destination interval
`[pos, pos + |w|)` holding `merge c (extract γ)` cell by cell (`c` the original destination
cell, `γ` the source cell) and every other field unchanged; through the endpoint the heads of
`src` and `dst` stay within one cell of their intervals and every other head is fixed.

**Proof sketch.** `Turing.sweepContract` with the clauses of `Turing.copyMapTM_run_ofCfg`. -/
def copyMapTM_contract {Δ : Type} (src dst : Fin k) (hne : src ≠ dst) (extract : Γ → Δ)
    (merge : Option Γ → Δ → Γ) (x : List Γ) :
    Contract (copyMapTM k src dst extract merge) SweepPhase.sweep {some SweepPhase.done}
      (List Γ) x :=
  sweepContract _ _ _ src (fun j => j = src ∨ j = dst) x
    (sweepPost .done dst (sweepMapWrite extract merge)) (fun _ _ => rfl)
    (copyMapTM_run_ofCfg src dst hne extract merge)

/-- **The mapped transfer as a contract** (design §13d, G2): as `Turing.copyMapTM_contract`,
with the source interval holding the residues `rest γ` at the endpoint.

**Proof sketch.** `Turing.sweepContract` with the clauses of `Turing.transferMapTM_run_ofCfg`. -/
def transferMapTM_contract {Δ : Type} (src dst : Fin k) (hne : src ≠ dst) (extract : Γ → Δ)
    (merge : Option Γ → Δ → Γ) (rest : Γ → Option Γ) (x : List Γ) :
    Contract (transferMapTM k src dst extract merge rest) SweepPhase.sweep
      {some SweepPhase.done} (List Γ) x :=
  sweepContract _ _ _ src (fun j => j = src ∨ j = dst) x
    (fun w d => sweepPost .done src (fun w d j q => (FinTM.bufferTape w (q - d.workTapePos j)).bind
      rest) w (sweepPost .done dst (sweepMapWrite extract merge) w d)) (fun _ _ => rfl)
    (transferMapTM_run_ofCfg src dst hne extract merge rest)

/-- The mapped copy frames exactly the destination interval: its footprint is
`[pos, pos + |w|)` on tape `dst` and empty elsewhere.

**Proof sketch.** `Turing.sweep_ite_keep` at `dst`. -/
theorem copyMapTM_contract_frames {Δ : Type} (src dst : Fin k) (hne : src ≠ dst)
    (extract : Γ → Δ) (merge : Option Γ → Δ → Γ) (x : List Γ) :
    (copyMapTM_contract src dst hne extract merge x).Frames fun w d j =>
      if j = dst then Set.Ico (d.workTapePos j) (d.workTapePos j + w.length) else ∅ :=
  fun _ _ _ _ _ _ hq => sweep_ite_keep _ _ hq

/-- The mapped transfer frames exactly its two word intervals, on `src` and on `dst`: its
footprint is `[pos, pos + |w|)` on each and empty elsewhere.

**Proof sketch.** `Turing.sweep_ite_keep` at `src`, then at `dst`. -/
theorem transferMapTM_contract_frames {Δ : Type} (src dst : Fin k) (hne : src ≠ dst)
    (extract : Γ → Δ) (merge : Option Γ → Δ → Γ) (rest : Γ → Option Γ) (x : List Γ) :
    (transferMapTM_contract src dst hne extract merge rest x).Frames fun w d j =>
      if j = src ∨ j = dst then Set.Ico (d.workTapePos j) (d.workTapePos j + w.length)
      else ∅ := by
  intro w d _ _ j q hq
  have hs : q ∉ (if j = src then Set.Ico (d.workTapePos j) (d.workTapePos j + w.length)
      else ∅) := by
    by_cases h : j = src <;> simp_all
  have hd : q ∉ (if j = dst then Set.Ico (d.workTapePos j) (d.workTapePos j + w.length)
      else ∅) := by
    by_cases h : j = dst <;> simp_all
  exact (sweep_ite_keep _ _ hs).trans (sweep_ite_keep _ _ hd)

/-- **A joint trajectory of the mapped copy**: every head other than the two touched ones stays
at its start.

**Proof sketch.** `Turing.sweepContract_traj`. -/
theorem copyMapTM_contract_traj {Δ : Type} (src dst : Fin k) (hne : src ≠ dst)
    (extract : Γ → Δ) (merge : Option Γ → Δ → Γ) (x : List Γ) :
    (copyMapTM_contract src dst hne extract merge x).Traj
      fun _ d p => ∀ j, j ≠ src → j ≠ dst → p j = d.workTapePos j :=
  fun w d hd hp u hu j h₁ h₂ =>
    sweepContract_traj _ _ _ _ _ _ _ _ _ w d hd hp u hu j (by tauto)

end Turing

/-!
# Machine-construction library: the multi-track zoned carrier (§13d G3)

The physical representation of the Hennie–Stearns simulation ([AB09] §1.7) over a rich alphabet
(`machine-library-design.md` §13d, G3; decisions 13d.1 and 13d.5): `k` tracks of zone contents
(`Turing.ZoneContents`, the per-track pure layer of `Build/Zone`, reused verbatim) on **one**
work tape over `Turing.ZoneSym k := Bool ⊕ Turing.ZoneCell k`. One physical cell holds one
virtual slot of every track; the home, at cell `0`, holds every track's virtual symbol under its
head, so the simulator reads and writes all `k` virtual heads in one step.

**Status: proved.** Every declaration of the module is proved.

## Design

* **The layout** (decision 13d.1). One blank separator precedes each zone: the right zone `i`
  occupies `[zoneStart i, zoneStart i + zoneCapacity i)` with `zoneStart i = zoneBase i + i + 2
  = 2^(i+1) + i`, and the left zone `i` the mirror image. Each zone is therefore a delimited
  nonblank word (`Turing.zonedTape_right_word`, `Turing.zonedTape_left_word`), so the framed
  sweeps of `Build/Catalog` (the scan, and the G2 mapped sweeps) run on a zone with no save and
  restore, and the separators mark the zone boundaries, so no counter is needed.
* **Inner packing.** In each zone, track `t`'s stored cells occupy the innermost slots and the
  rest hold `⊠`: a zone is empty on a track exactly when its innermost slot is `⊠`, and full
  exactly when its outermost slot is occupied. The book leaves the placement of `⊠` free.
* **Per-track access** is `Turing.zoneSlotOf` (read a slot) and `Turing.zoneSlotSet` (write a
  slot, keeping the other tracks): the `extract` and `merge` of the per-track mapped sweeps.
* **The carrier is an encoding**, `Turing.zonedTape`, faithful (`Turing.zonedTape_injective`).
  The fullness invariant is not part of it; it is the Hennie–Stearns consumer's (§13d, G5).

## Main definitions

* `Turing.ZoneCell`, `Turing.ZoneSym`: the alphabet; `Turing.zoneSlotOf`, `Turing.zoneSlotSet`.
* `Turing.zoneStart`, `Turing.zoneOwner`: the layout.
* `Turing.zoneCells`, `Turing.zoneCellAt`, `Turing.zonedTape`: the carrier.

## Main results

* `Turing.zonedTape_zero`, `Turing.zonedTape_right`, `Turing.zonedTape_left`: the cells.
* `Turing.zonedTape_right_sep`, `Turing.zonedTape_left_sep`, `Turing.zonedTape_blank_outside`:
  the blanks.
* `Turing.zonedTape_right_word`, `Turing.zonedTape_left_word`: zones are delimited words.
* `Turing.zonedTape_update_slot`, `Turing.zonedTape_isSome`: track independence and a
  contents-independent layout.
* `Turing.zonedTape_homeWrite`, `Turing.zonedTape_extend`: what the home write and the extension
  change.
* `Turing.zonedTape_injective`: the encoding is faithful.

## References

* [AB09] S. Arora, B. Barak, *Computational Complexity: A Modern Approach*, Cambridge University
  Press, 2009. (§1.7, the Hennie–Stearns simulation; Claim 1.5, the alphabet reduction that
  finishes a machine over `Turing.ZoneSym k`.)
-/

/-! ### The alphabet -/

/-- **A zone cell** (design §13d, G3; decision 13d.5): one slot per track. In track `t`'s slot,
`none` is the buffer symbol `⊠` (an unoccupied slot) and `some v` an occupied slot storing the
virtual symbol `v`, where `v = none` is the virtual blank. [AB09, §1.7]: the tape alphabet
`Γ^k` of the parallel tracks, with the buffer symbol `⊠`. -/
abbrev ZoneCell (k : ℕ) : Type := Fin k → Option (Option Bool)

/-- **The simulator's alphabet** (decision 13d.5): a binary symbol (`Sum.inl`, the input and
output alphabet that `Turing.FinTM.ComputesFunInTimeVia`'s embedding uses) or a zone cell
(`Sum.inr`). It is finite and has decidable equality, so a machine over it is a `FinTM`, and
`Turing.FinTM.alphabet_reduction` turns it into a binary machine. [AB09, §1.7, with Claim 1.5
for the alphabet reduction] -/
abbrev ZoneSym (k : ℕ) : Type := Bool ⊕ ZoneCell k

/-- Track `t`'s slot of a physical symbol: the slot of a zone cell; `⊠` for a binary symbol.
This is the `extract` of the per-track mapped sweeps (design §13d, G2). -/
def zoneSlotOf {k : ℕ} (t : Fin k) : ZoneSym k → Option (Option Bool)
  | .inl _ => none
  | .inr c => c t

/-- The zone cell a physical cell holds: its cell if it holds one, all `⊠` otherwise (a blank
or a binary cell). -/
def zoneCellOfCell {k : ℕ} : Option (ZoneSym k) → ZoneCell k
  | some (.inr c) => c
  | _ => fun _ => none

/-- Overwrite track `t`'s slot of a physical cell, keeping its other tracks; a blank or binary
cell is read as all `⊠` first. This is the `merge` of the per-track mapped sweeps (design §13d,
G2; `merge : Option Γ → Δ → Γ`, so that blank destinations are admitted). -/
def zoneSlotSet {k : ℕ} (t : Fin k) (cell : Option (ZoneSym k)) (s : Option (Option Bool)) :
    ZoneSym k :=
  .inr (Function.update (zoneCellOfCell cell) t s)

/-- Setting a track's slot sets that slot. -/
@[simp] theorem zoneSlotOf_slotSet_self {k : ℕ} (t : Fin k) (cell : Option (ZoneSym k))
    (s : Option (Option Bool)) : zoneSlotOf t (zoneSlotSet t cell s) = s := by
  simp [zoneSlotOf, zoneSlotSet]

/-- Setting a track's slot keeps every other track's slot of a zone cell. -/
theorem zoneSlotOf_slotSet_ne {k : ℕ} {s t : Fin k} (h : s ≠ t) (c : ZoneCell k)
    (v : Option (Option Bool)) :
    zoneSlotOf s (zoneSlotSet t (some (.inr c)) v) = c s := by
  simp [zoneSlotOf, zoneSlotSet, zoneCellOfCell, Function.update_of_ne h]

/-! ### The layout -/

/-- **The first physical cell of zone `i`**, on the right; the left zone `i` starts at
`-zoneStart i`. Home is cell `0`, and one blank separator precedes each zone (decision 13d.1):
`zoneStart i = zoneBase i + i + 2`, the `i + 1` separators and the inner zones' cells beyond
the home. -/
def zoneStart (i : ℕ) : ℕ := zoneBase i + i + 2

/-- Consecutive zones are one separator apart. -/
theorem zoneStart_succ (i : ℕ) : zoneStart (i + 1) = zoneStart i + zoneCapacity i + 1 := by
  rw [zoneStart, zoneStart, zoneBase_succ]
  omega

/-- The closed form `zoneStart i = 2^(i+1) + i`. -/
theorem zoneStart_eq (i : ℕ) : zoneStart i = 2 ^ (i + 1) + i := by
  have h := Nat.two_pow_pos i
  simp only [zoneStart, zoneBase, pow_succ]
  omega

/-- A zone ends, with its trailing separator, before any later zone starts. -/
theorem zoneStart_add_lt {i j : ℕ} (h : i < j) : zoneStart i + zoneCapacity i < zoneStart j := by
  have hp := Nat.pow_le_pow_right (by decide : 0 < 2) (show i + 1 ≤ j by omega)
  rw [zoneStart_eq, zoneStart_eq]
  simp only [zoneCapacity, pow_succ] at *
  omega

/-- Zone starts are monotone. -/
theorem zoneStart_mono {i j : ℕ} (h : i ≤ j) : zoneStart i ≤ zoneStart j := by
  rcases Nat.eq_or_lt_of_le h with rfl | h
  · exact le_rfl
  · have := zoneStart_add_lt h
    omega

/-- Every zone starts at least two cells from home. -/
theorem two_le_zoneStart (i : ℕ) : 2 ≤ zoneStart i := by
  unfold zoneStart
  omega

/-- The laid-out zone owning physical distance `n` from home, if any: the unique `i < ℓ` with
`zoneStart i ≤ n < zoneStart i + zoneCapacity i`. Home, separators and cells beyond the layout
have no owner. -/
def zoneOwner (ℓ n : ℕ) : Option (Fin ℓ) :=
  Fin.find fun i : Fin ℓ => zoneStart i ≤ n ∧ n < zoneStart i + zoneCapacity i

/-- A cell inside a laid-out zone is owned by that zone.

**Proof sketch.** `Fin.find_eq_some_iff`; minimality holds because another owner `j` would
overlap zone `i`, which `Turing.zoneStart_add_lt` excludes in either order. -/
theorem zoneOwner_of_mem {ℓ : ℕ} (i : Fin ℓ) {n : ℕ} (h₁ : zoneStart i ≤ n)
    (h₂ : n < zoneStart i + zoneCapacity i) : zoneOwner ℓ n = some i := by
  rw [zoneOwner, Fin.find_eq_some_iff]
  refine ⟨⟨h₁, h₂⟩, fun j hj => ?_⟩
  by_contra hlt
  have := zoneStart_add_lt (show j.val < i.val by omega)
  omega

/-- A cell owned by no laid-out zone has no owner. -/
theorem zoneOwner_of_not_mem {ℓ n : ℕ}
    (h : ∀ i : Fin ℓ, ¬ (zoneStart i ≤ n ∧ n < zoneStart i + zoneCapacity i)) :
    zoneOwner ℓ n = none :=
  Fin.find_eq_none_iff.mpr h

/-- The separator before zone `i` (any `i`, laid out or not) has no owner.

**Proof sketch.** A zone `j < i` ends before it (`Turing.zoneStart_succ`,
`Turing.zoneStart_mono`); a zone `j ≥ i` starts after it. -/
theorem zoneOwner_sep (ℓ i : ℕ) : zoneOwner ℓ (zoneStart i - 1) = none := by
  apply zoneOwner_of_not_mem
  intro j ⟨h₁, h₂⟩
  have h2 := two_le_zoneStart i
  rcases Nat.lt_or_ge j.val i with hj | hj
  · have := zoneStart_mono (show j.val + 1 ≤ i by omega)
    have := zoneStart_succ j.val
    omega
  · have := zoneStart_mono hj
    omega

/-- Past the last laid-out zone nothing has an owner. -/
theorem zoneOwner_beyond {ℓ n : ℕ} (h : zoneStart ℓ - 1 ≤ n) : zoneOwner ℓ n = none := by
  apply zoneOwner_of_not_mem
  intro j ⟨_, h₂⟩
  have := zoneStart_mono (show j.val + 1 ≤ ℓ by omega)
  have := zoneStart_succ j.val
  omega

/-! ### The carrier -/

/-- **The physical cells of zone `i` on one side** (`side = true` the right), inner end first:
cell `p` holds, in track `t`'s slot, the `p`-th stored cell of track `t`'s zone word, or `⊠`
beyond its length. Stored cells are thus **packed at the inner end** of each zone, so a zone is
empty on a track exactly when its innermost slot is `⊠`. Every cell is a zone cell, hence
nonblank: each zone is a nonblank word of length `zoneCapacity i`. -/
def zoneCells {k ℓ : ℕ} (side : Bool) (z : Fin k → ZoneContents ℓ) (i : Fin ℓ) :
    List (ZoneSym k) :=
  (List.range (zoneCapacity i)).map fun p => .inr fun t => ((z t).words side i)[p]?

/-- The cell at distance `n ≥ 1` from home on one side: the matching zone cell when a laid-out
zone owns it, blank otherwise (a separator, or beyond the layout). -/
def zoneCellAt {k ℓ : ℕ} (side : Bool) (z : Fin k → ZoneContents ℓ) (n : ℕ) :
    Option (ZoneSym k) :=
  (zoneOwner ℓ n).map fun i => .inr fun t => ((z t).words side i)[n - zoneStart i]?

/-- **The zoned tape** (design §13d, G3): the physical work tape that represents `k` tracks of
zone contents on `ℓ` laid-out levels. Cell `0` is the home, holding every track's virtual
symbol under its head; the right zone `i` occupies `[zoneStart i, zoneStart i + zoneCapacity i)`
and the left zone `i` the mirror image, each holding `Turing.zoneCells`; every other cell is
blank (the separators, one before each zone, and everything beyond the layout). [AB09, §1.7]:
the zones `L_i`, `R_i` of the single work tape, with the tracks in parallel. Deviations: one
blank separator between zones (decision 13d.1, so that zones are delimited words); stored cells
packed at each zone's inner end (the book leaves the placement of `⊠` free). -/
def zonedTape {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (c : ℤ) : Option (ZoneSym k) :=
  if c = 0 then some (.inr fun t => some (z t).home)
  else if 0 < c then zoneCellAt true z c.toNat
  else zoneCellAt false z (-c).toNat

/-- A zone has exactly `zoneCapacity i` cells. -/
@[simp] theorem zoneCells_length {k ℓ : ℕ} (side : Bool) (z : Fin k → ZoneContents ℓ)
    (i : Fin ℓ) : (zoneCells side z i).length = zoneCapacity i := by
  simp [zoneCells]

/-- The home cell holds every track's virtual symbol. -/
@[simp] theorem zonedTape_zero {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) :
    zonedTape z 0 = some (.inr fun t => some (z t).home) := by
  simp [zonedTape]

/-- A right zone cell holds the zone words' cells at its offset.

**Proof sketch.** The cell is positive and owned by zone `i` (`Turing.zoneOwner_of_mem`). -/
theorem zonedTape_right {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (i : Fin ℓ) (p : ℕ)
    (hp : p < zoneCapacity i) :
    zonedTape z ((zoneStart i + p : ℕ) : ℤ) =
      some (.inr fun t => ((z t).words true i)[p]?) := by
  have h2 := two_le_zoneStart i
  have ho := zoneOwner_of_mem (ℓ := ℓ) i (n := zoneStart i + p) (by omega) (by omega)
  simp only [zonedTape, zoneCellAt]
  rw [if_neg (by omega), if_pos (by omega), Int.toNat_natCast, ho]
  simp

/-- A left zone cell holds the zone words' cells at its offset. -/
theorem zonedTape_left {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (i : Fin ℓ) (p : ℕ)
    (hp : p < zoneCapacity i) :
    zonedTape z (-((zoneStart i + p : ℕ) : ℤ)) =
      some (.inr fun t => ((z t).words false i)[p]?) := by
  have h2 := two_le_zoneStart i
  have ho := zoneOwner_of_mem (ℓ := ℓ) i (n := zoneStart i + p) (by omega) (by omega)
  simp only [zonedTape, zoneCellAt]
  rw [if_neg (by omega), if_neg (by omega), neg_neg, Int.toNat_natCast, ho]
  simp

/-- The right separator before zone `i` is blank, for every `i` (laid out or not). -/
theorem zonedTape_right_sep {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (i : ℕ) :
    zonedTape z ((zoneStart i : ℤ) - 1) = none := by
  have h2 := two_le_zoneStart i
  simp only [zonedTape, zoneCellAt]
  rw [if_neg (by omega), if_pos (by omega), show ((zoneStart i : ℤ) - 1).toNat =
    zoneStart i - 1 by omega, zoneOwner_sep]
  rfl

/-- The left separator before zone `i` is blank. -/
theorem zonedTape_left_sep {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (i : ℕ) :
    zonedTape z (-(zoneStart i : ℤ) + 1) = none := by
  have h2 := two_le_zoneStart i
  simp only [zonedTape, zoneCellAt]
  rw [if_neg (by omega), if_neg (by omega), show (-(-(zoneStart i : ℤ) + 1)).toNat =
    zoneStart i - 1 by omega, zoneOwner_sep]
  rfl

/-- **Blank outside the layout**: every cell at distance at least `zoneStart ℓ - 1` from home
(the separator after the last laid-out zone, and beyond) is blank. With `ℓ` levels the
nonblank cells lie in `[-(zoneStart ℓ - 2), zoneStart ℓ - 2]`; this replaces the retired
`Turing.zoneTape_blank_outside` as the carrier's extent. -/
theorem zonedTape_blank_outside {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (c : ℤ)
    (hc : (zoneStart ℓ : ℤ) - 1 ≤ |c|) : zonedTape z c = none := by
  have h2 := two_le_zoneStart ℓ
  have h0 : c ≠ 0 := by
    intro h
    rw [h, abs_zero] at hc
    omega
  unfold zonedTape zoneCellAt
  rw [if_neg h0]
  by_cases hp : 0 < c
  · rw [abs_of_pos hp] at hc
    rw [if_pos hp, zoneOwner_beyond (by omega)]
    rfl
  · rw [abs_of_nonpos (by omega)] at hc
    rw [if_neg hp, zoneOwner_beyond (by omega)]
    rfl

/-- **Each right zone is a delimited word** (the carrier invariant that lets the framed sweeps
run on a zone, with no save and restore; design §13d, G3): from its first cell, the right zone
`i` holds `Turing.zoneCells true z i`, with blanks at relative positions `-1` and
`zoneCapacity i`. This is the precondition form of `Turing.copyTM_run_ofCfg` and of the mapped
sweeps with the head at `zoneStart i`.

**Proof sketch.** Inside the zone, `Turing.zonedTape_right` and the cells' definition; position
`-1` is the separator before zone `i` and position `zoneCapacity i` the one before zone `i + 1`
(`Turing.zonedTape_right_sep`, `Turing.zoneStart_succ`). -/
theorem zonedTape_right_word {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (i : Fin ℓ) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (zoneCapacity i : ℤ) →
      zonedTape z ((zoneStart i : ℤ) + p) = FinTM.bufferTape (zoneCells true z i) p := by
  intro p h₁ h₂
  rcases (show p = -1 ∨ (0 ≤ p ∧ p < zoneCapacity i) ∨ p = zoneCapacity i by omega) with
    h | ⟨h₁, h₂⟩ | h
  · subst h
    rw [show (zoneStart i : ℤ) + -1 = (zoneStart i : ℤ) - 1 by omega, zonedTape_right_sep]
    simp [FinTM.bufferTape]
  · obtain ⟨n, rfl⟩ : ∃ n : ℕ, p = n := ⟨p.toNat, by omega⟩
    rw [show (zoneStart i : ℤ) + n = ((zoneStart i + n : ℕ) : ℤ) by omega,
      zonedTape_right z i n (by omega), FinTM.bufferTape_nat]
    simp [zoneCells, show n < zoneCapacity i by omega]
  · subst h
    have hs := zoneStart_succ i.val
    rw [show (zoneStart i : ℤ) + (zoneCapacity i : ℕ) = (zoneStart (i.val + 1) : ℤ) - 1 by
      omega, zonedTape_right_sep, FinTM.bufferTape_nat]
    simp

/-- **Each left zone is a delimited word**, read in the sweep direction (left to right): from
the zone's outermost cell `-(zoneStart i + zoneCapacity i - 1)`, the left zone `i` holds the
reversed `Turing.zoneCells false z i`, with blanks at relative positions `-1` and
`zoneCapacity i`.

**Proof sketch.** Position `p` inside is the cell at offset `zoneCapacity i - 1 - p` from the
zone's inner end, which the reversed list holds at `p`; the delimiters are the separators
before zones `i + 1` and `i`. -/
theorem zonedTape_left_word {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (i : Fin ℓ) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (zoneCapacity i : ℤ) →
      zonedTape z (-((zoneStart i : ℤ) + zoneCapacity i - 1) + p) =
        FinTM.bufferTape (zoneCells false z i).reverse p := by
  intro p h₁ h₂
  have hc : 0 < zoneCapacity i := by
    have := Nat.two_pow_pos i.val
    unfold zoneCapacity
    omega
  rcases (show p = -1 ∨ (0 ≤ p ∧ p < zoneCapacity i) ∨ p = zoneCapacity i by omega) with
    h | ⟨h₁, h₂⟩ | h
  · subst h
    have hs := zoneStart_succ i.val
    rw [show -((zoneStart i : ℤ) + zoneCapacity i - 1) + -1 =
      -(zoneStart (i.val + 1) : ℤ) + 1 by omega, zonedTape_left_sep]
    simp [FinTM.bufferTape]
  · obtain ⟨n, rfl⟩ : ∃ n : ℕ, p = n := ⟨p.toNat, by omega⟩
    rw [show -((zoneStart i : ℤ) + zoneCapacity i - 1) + n =
      -((zoneStart i + (zoneCapacity i - 1 - n) : ℕ) : ℤ) by omega,
      zonedTape_left z i _ (by omega), FinTM.bufferTape_nat, List.getElem?_reverse
        (by simp; omega)]
    simp [zoneCells, show zoneCapacity i - 1 - n < zoneCapacity i by omega]
  · subst h
    rw [show -((zoneStart i : ℤ) + zoneCapacity i - 1) + (zoneCapacity i : ℕ) =
      -(zoneStart i : ℤ) + 1 by omega, zonedTape_left_sep, FinTM.bufferTape_nat]
    simp

/-- **Track independence**: changing one track's contents changes no other track's slot,
anywhere on the tape. -/
theorem zonedTape_update_slot {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) {s t : Fin k}
    (h : s ≠ t) (w : ZoneContents ℓ) (c : ℤ) :
    (zonedTape (Function.update z t w) c).map (zoneSlotOf s) =
      (zonedTape z c).map (zoneSlotOf s) := by
  unfold zonedTape zoneCellAt
  split_ifs <;> simp [zoneSlotOf, Function.update_of_ne h] <;>
    cases zoneOwner ℓ _ <;> simp [zoneSlotOf, Function.update_of_ne h]

/-- **The layout does not depend on the contents**: a cell is blank on one family's tape
exactly when it is blank on another's, for the same number of levels. -/
theorem zonedTape_isSome {k ℓ : ℕ} (z z' : Fin k → ZoneContents ℓ) (c : ℤ) :
    (zonedTape z c).isSome = (zonedTape z' c).isSome := by
  unfold zonedTape zoneCellAt
  split_ifs <;> simp

/-- **Writing the home** is a one-cell write: overwriting cell `0` with the new virtual
symbols gives the zoned tape of the home-written family. -/
theorem zonedTape_homeWrite {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (v : Fin k → Option Bool) :
    Function.update (zonedTape z) 0 (some (.inr fun t => some (v t))) =
      zonedTape (fun t => zoneHomeWrite (z t) (v t)) := by
  funext c
  by_cases h : c = 0
  · subst h
    simp [zoneHomeWrite]
  · rw [Function.update_of_ne h]
    unfold zonedTape zoneCellAt
    simp [h, zoneHomeWrite, ZoneContents.words]

/-- Zone contents are determined by their home and words. -/
theorem ZoneContents.ext' {ℓ : ℕ} {a b : ZoneContents ℓ} (h₀ : a.home = b.home)
    (h₁ : a.left = b.left) (h₂ : a.right = b.right) : a = b := by
  cases a
  cases b
  simp_all

/-- **The encoding is faithful**: two families on the same levels with the same zoned tape are
equal. (A sanity theorem: the carrier loses nothing.)

**Proof sketch.** The homes are read at cell `0`. A zone word is determined by its lookups: at
an offset below the capacity, `Turing.zonedTape_right`/`_left` expose it in the track's slot;
beyond the capacity both lookups are `none`, since words fit their zones. -/
theorem zonedTape_injective {k ℓ : ℕ} {z z' : Fin k → ZoneContents ℓ}
    (h : zonedTape z = zonedTape z') : z = z' := by
  funext t
  have hw : ∀ (side : Bool) (i : Fin ℓ), (z t).words side i = (z' t).words side i := by
    intro side i
    apply List.ext_getElem?
    intro p
    by_cases hp : p < zoneCapacity i
    · cases side
      · have e := congrFun h (-((zoneStart i + p : ℕ) : ℤ))
        rw [zonedTape_left z i p hp, zonedTape_left z' i p hp] at e
        simpa using congrFun (Sum.inr_injective (Option.some_injective _ e)) t
      · have e := congrFun h ((zoneStart i + p : ℕ) : ℤ)
        rw [zonedTape_right z i p hp, zonedTape_right z' i p hp] at e
        simpa using congrFun (Sum.inr_injective (Option.some_injective _ e)) t
    · rw [List.getElem?_eq_none (by have := (z t).words_le side i; omega),
        List.getElem?_eq_none (by have := (z' t).words_le side i; omega)]
  apply ZoneContents.ext'
  · have e := congrFun h 0
    simp only [zonedTape_zero] at e
    simpa using congrFun (Sum.inr_injective (Option.some_injective _ e)) t
  · funext i
    exact hw false i
  · funext i
    exact hw true i

/-- Ownership after laying out one more level: the old zones own what they owned, the new zone
`ℓ` owns its own cells, and nothing else changes.

**Proof sketch.** The old levels embed by `Fin.castSucc`, with the same starts; the new level
is `Fin.last ℓ`; uniqueness of owners is `Turing.zoneOwner_of_mem`. -/
theorem zoneOwner_succ (ℓ n : ℕ) :
    zoneOwner (ℓ + 1) n =
      if zoneStart ℓ ≤ n ∧ n < zoneStart ℓ + zoneCapacity ℓ then some (Fin.last ℓ)
      else (zoneOwner ℓ n).map Fin.castSucc := by
  split_ifs with h
  · exact zoneOwner_of_mem (Fin.last ℓ) h.1 h.2
  · cases ho : zoneOwner ℓ n with
    | none =>
      apply zoneOwner_of_not_mem
      intro i hi
      refine Fin.lastCases (motive := fun i : Fin (ℓ + 1) =>
        ¬ (zoneStart i ≤ n ∧ n < zoneStart i + zoneCapacity i)) ?_ (fun j => ?_) i hi
      · simpa using h
      · intro hj
        rw [zoneOwner_of_mem j hj.1 hj.2] at ho
        exact Option.some_ne_none _ ho
    | some j =>
      rw [Fin.find_eq_some_iff.mp ho |>.1 |> fun hj => zoneOwner_of_mem j.castSucc hj.1 hj.2]
      rfl

/-- **What the extension writes**: laying out level `ℓ` on every track changes exactly the
cells of the two new zones, each of which then holds, on every track, a stored virtual blank in
its inner `2^ℓ` slots and `⊠` in its outer `2^ℓ` slots. Every other cell is unchanged.

**Proof sketch.** `Turing.zoneOwner_succ` splits the cells into the new zone's and the old
owners'; the old words are kept (`Turing.ZoneContents.extend_words_castSucc`) and the new ones
are `replicate (2^ℓ) none` (`Turing.ZoneContents.extend_words_last`). -/
theorem zonedTape_extend {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (c : ℤ) :
    zonedTape (fun t => (z t).extend) c =
      if (zoneStart ℓ : ℤ) ≤ |c| ∧ |c| < zoneStart ℓ + zoneCapacity ℓ then
        some (.inr fun _ => if (|c|).toNat - zoneStart ℓ < 2 ^ ℓ then some none else none)
      else zonedTape z c := by
  by_cases h0 : c = 0
  · subst h0
    have := two_le_zoneStart ℓ
    simp only [abs_zero, zonedTape_zero]
    rw [if_neg (by omega)]
    rfl
  · unfold zonedTape zoneCellAt
    rw [if_neg h0, if_neg h0]
    by_cases hp : 0 < c
    · rw [if_pos hp, if_pos hp, zoneOwner_succ, abs_of_pos hp]
      by_cases hz : zoneStart ℓ ≤ c.toNat ∧ c.toNat < zoneStart ℓ + zoneCapacity ℓ
      · rw [if_pos hz, if_pos (by omega)]
        simp [ZoneContents.extend_words_last, List.getElem?_replicate]
      · rw [if_neg hz, if_neg (by omega)]
        cases zoneOwner ℓ c.toNat <;> simp [ZoneContents.extend_words_castSucc]
    · rw [if_neg hp, if_neg hp, zoneOwner_succ, abs_of_nonpos (by omega)]
      by_cases hz : zoneStart ℓ ≤ (-c).toNat ∧ (-c).toNat < zoneStart ℓ + zoneCapacity ℓ
      · rw [if_pos hz, if_pos (by omega)]
        simp [ZoneContents.extend_words_last, List.getElem?_replicate]
      · rw [if_neg hz, if_neg (by omega)]
        cases zoneOwner ℓ (-c).toNat <;> simp [ZoneContents.extend_words_castSucc]

end Turing

# Machine-construction library: the rebalance and extension rows (§13d G4)

The machine rows of the Hennie–Stearns zone layer ([AB09] §1.7; `machine-library-design.md`
§13d, G4; decisions 13d.2–13d.4), stated as `Turing.Contract`s from the start (§16b, item 5).
Both rows are two-tape machines over `Turing.ZoneSym k`: tape `0` carries the zoned tape
(`Turing.zonedTape`), tape `1` is a blank scratch tape, the buffer.

* **The rebalance row**, one machine per direction and track (the track in finite control,
  decision 13d.4): from the home, scan the feeding side for the scanned index
  `Turing.zoneShiftIndex`, realize the total cascade there (`Turing.zoneRebalance`, decision
  13d.3), return home; exit `done`, or `short` when every feeding zone is empty. Time
  `O(2^{i₀} + 1)`, the zoned head within the level-`i₀` extent.
* **The extension row** (decision 13d.2): lay out level `ℓ` on every track, half full of virtual
  blanks, in `O(2^ℓ + 1)`.

**Status: statements.** The two rows are sorried under their proof sketches (the fill plan);
everything else here is proved. These rows replace the retired binary rows
`exists_zoneShiftInTM` and `exists_zoneShiftOutTM` of `Build/Zone`.

## How the assembly consumes the rows (§13d, G5; sketched for fit, not in this module)

One simulated step reads the home tuple, applies the simulated transition in finite control,
writes the home tuple (`Turing.zonedTape_homeWrite`), and for each moving track runs its
rebalance row; on `short` it runs the extension row and the rebalance again, which then exits
`done` (`Turing.zoneRebalanceIndex_extend_of_short`). Every row's endpoint satisfies the next
row's precondition (`Turing.zoneRowPre_rebalancePost`, `Turing.zoneRowPre_extendPost`), and the
represented tapes move as the simulated heads do (`Turing.zoneTrackTape_rebalance_right` and its
mirror). The amortization `Σ_i T/2^{i-1} · 2^i = O(T log T)` is the consumer's mathematics.

## Main definitions

* `Turing.ZoneData`, `Turing.zoneRowPre`: the index and the precondition.
* `Turing.zoneRebalanceIndex`, `Turing.zoneRebalanceData`, `Turing.zoneExtendData`.
* `Turing.zoneRebalancePost`, `Turing.zoneExtendPost`, `Turing.zoneRowLo`, `Turing.zoneRowHi`:
  the endpoints and envelopes.

## Main results

* `Turing.exists_zoneRebalanceTM`, `Turing.exists_zoneExtendTM`: the rows (sorried).
* `Turing.zoneRowPre_rebalancePost`, `Turing.zoneRowPre_extendPost`: rows chain.
* `Turing.zoneRebalancePost_state_done`, `Turing.zoneRebalanceData_of_short`,
  `Turing.zoneRebalanceIndex_extend_of_short`: the two exits.
* `Turing.zoneRebalancePost_frames`, `Turing.zoneExtendPost_frames`: the footprints.

/-! ### The data, the precondition, the endpoints and the envelopes -/

/-- **The data index of the zone rows**: a level count `ℓ` and a contents family over `ℓ` laid
out levels, one `Turing.ZoneContents` per track. -/
abbrev ZoneData (k : ℕ) : Type := Σ ℓ : ℕ, Fin k → ZoneContents ℓ

/-- **The precondition of every zone row**: work tape `0` holds the zoned tape of the data,
with its head at home (cell `0`); work tape `1`, the scratch tape, is blank everywhere, its
head anywhere. Nothing is required of the control state beyond the contract's entry, of the
input tape or of the output. -/
def zoneRowPre {k : ℕ} {S : Type} {x : List (ZoneSym k)} (D : ZoneData k)
    (d : Cfg 2 (ZoneSym k) S x) : Prop :=
  d.workTapes 0 = zonedTape D.2 ∧ d.workTapePos 0 = 0 ∧ ∀ q, d.workTapes 1 q = none

/-- The scanned index of the rebalance of track `t` in direction `dir`. -/
def zoneRebalanceIndex {k : ℕ} (dir : Bool) (t : Fin k) (D : ZoneData k) : ℕ :=
  zoneShiftIndex dir (D.2 t)

/-- The data after the rebalance of track `t` in direction `dir`: track `t` rebalanced
(`Turing.zoneRebalance`), every other track and the level count unchanged. -/
def zoneRebalanceData {k : ℕ} (dir : Bool) (t : Fin k) (D : ZoneData k) : ZoneData k :=
  ⟨D.1, Function.update D.2 t (zoneRebalance dir (D.2 t))⟩

/-- The data after the extension: one more level, laid out on every track
(`Turing.ZoneContents.extend`). -/
def zoneExtendData {k : ℕ} (D : ZoneData k) : ZoneData k :=
  ⟨D.1 + 1, fun t => (D.2 t).extend⟩

/-- **The rebalance row's endpoint**: from `d`, the state is the exit `done` when the scanned
index is below `ℓ` and `short` otherwise; tape `0` holds the zoned tape of the rebalanced data;
every other field of `d` (the scratch tape, both heads, the input position, the output) is
unchanged. -/
def zoneRebalancePost {k : ℕ} (dir : Bool) (t : Fin k) {S : Type} {x : List (ZoneSym k)}
    (done short : S) (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x) : Cfg 2 (ZoneSym k) S x :=
  { d with
    state := some (if zoneRebalanceIndex dir t D < D.1 then done else short)
    workTapes := fun j => if j = 0 then zonedTape (zoneRebalanceData dir t D).2
      else d.workTapes j }

/-- **The extension row's endpoint**: the exit `done`; tape `0` holds the zoned tape of the
extended data; every other field of `d` is unchanged. -/
def zoneExtendPost {k : ℕ} {S : Type} {x : List (ZoneSym k)} (done : S) (D : ZoneData k)
    (d : Cfg 2 (ZoneSym k) S x) : Cfg 2 (ZoneSym k) S x :=
  { d with
    state := some done
    workTapes := fun j => if j = 0 then zonedTape (zoneExtendData D).2 else d.workTapes j }

/-- **The lower envelope of a zone row reaching level `r D`**: the zoned head stays at or
right of `-zoneStart (r D + 1)` (inside the level-`r D` extent and its outer separator, with one
cell to spare), and the scratch head at or right of its start minus `C · (2^(r D) + 1)`. -/
def zoneRowLo {k : ℕ} {S : Type} {x : List (ZoneSym k)} (r : ZoneData k → ℕ) (C : ℕ)
    (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x) (j : Fin 2) : ℤ :=
  if j = 0 then -(zoneStart (r D + 1) : ℤ) else d.workTapePos j - C * (2 ^ r D + 1)

/-- The upper envelope of a zone row reaching level `r D`, mirroring `Turing.zoneRowLo`. -/
def zoneRowHi {k : ℕ} {S : Type} {x : List (ZoneSym k)} (r : ZoneData k → ℕ) (C : ℕ)
    (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x) (j : Fin 2) : ℤ :=
  if j = 0 then (zoneStart (r D + 1) : ℤ) else d.workTapePos j + C * (2 ^ r D + 1)

/-! ### The rows -/

/-- **§13d G4, the rebalance rows** (design §13d, G4; decisions 13d.2–13d.4). For every track
count `k` there is a constant `C` such that for every direction `dir` (`true` is a right head
move) and every track `t` (in finite control) there is one two-tape machine over
`Turing.ZoneSym k`, with an entry and two distinct live exits `done` and `short`, whose run is a
`Turing.Contract` with index `Turing.ZoneData k`:

* **precondition** `Turing.zoneRowPre`: the zoned tape of `⟨ℓ, z⟩` on tape `0`, head at home;
  the scratch tape blank;
* **endpoint** `Turing.zoneRebalancePost`: tape `0` holds the zoned tape of `z` with track `t`
  replaced by `Turing.zoneRebalance dir (z t)` — the **total** pure cascade at the scanned index
  `Turing.zoneShiftIndex dir (z t)` when that index is below `ℓ` (exit `done`), the identity
  when every feeding zone is empty (exit `short`, the extension's cue); every other track, every
  separator and every cell beyond the layout unchanged; the scratch tape blank again; both
  heads back; input position and output unchanged;
* **exact time** with the no-earlier-exit cut (the contract's fields), bounded by
  `C · (2^i₀ + 1)` for the scanned index `i₀`;
* **envelopes** `Turing.zoneRowLo`/`Turing.zoneRowHi` at `i₀`: the zoned head within
  `zoneStart (i₀ + 1)` of home, the scratch head within `C · (2^i₀ + 1)` of its start.

For every contents, not only the classical ones (decision 13d.3). [AB09, §1.7]: one shift of
index `i₀`, at cost `O(2^{i₀})`. Deviation: the book's machine assumes the fullness invariant;
this row realizes the guarded cascade on every input, and reports an exhausted feeding side.

**Proof sketch.** Fill plan (design note §4): scan outward on the feeding side from home: zone
`i` is nonempty on track `t` exactly when its innermost slot is occupied (inner packing), the
separators delimit the zones, and two consecutive blanks mark the end of the layout, so the
scan finds `i₀` in `O(2^{i₀})` steps or reaches the end (`short`: rewind and exit). Then run the
cascade's `2i₀ + 1` guarded steps in its order — the descending pairs `i₀, …, 1`, the home move,
the ascending pairs `1, …, i₀` — each pair at level `i` acting on zones `i - 1` and `i` of both
sides in `O(2^i)` steps: a guard is a test of one or two slots (inner packing); a pairwise move
copies the track-`t` column of the zones involved to the scratch tape with the mapped copy
(`extract := zoneSlotOf t`), re-packs it there by scans, and writes it back with the mapped copy
(`merge := zoneSlotSet t`), which keeps the other tracks and needs no save and restore, since
zones are delimited words (`Turing.zonedTape_right_word`, `Turing.zonedTape_left_word`); the
mirror side is reached through a scratch ruler of the distance to home. The scratch tape is
cleared before the exit. The sum `Σ_{i ≤ i₀} O(2^i)` is `O(2^{i₀})`
(`Turing.zoneCascade_cost_le`). -/
theorem exists_zoneRebalanceTM (k : ℕ) :
    ∃ C : ℕ, ∀ (dir : Bool) (t : Fin k),
      ∃ (S : Type) (_ : Fintype S) (_ : DecidableEq S) (M : MultiTapeTM 2 (ZoneSym k) S)
        (entry done short : S), done ≠ short ∧
        ∀ x : List (ZoneSym k),
          ∃ R : Contract M entry {some done, some short} (ZoneData k) x,
            R.pre = zoneRowPre ∧
            R.post = zoneRebalancePost dir t done short ∧
            R.lo = zoneRowLo (zoneRebalanceIndex dir t) C ∧
            R.hi = zoneRowHi (zoneRebalanceIndex dir t) C ∧
            ∀ D d, d.state = some entry → R.pre D d →
              R.time D d ≤ C * (2 ^ zoneRebalanceIndex dir t D + 1) := by
  sorry

/-- **§13d G4, the extension row** (design §13d, decision 13d.2). For every track count `k`
there are a constant `C` and one two-tape machine over `Turing.ZoneSym k`, with an entry and a
live exit `done`, whose run is a `Turing.Contract` with index `Turing.ZoneData k`:

* **precondition** `Turing.zoneRowPre`;
* **endpoint** `Turing.zoneExtendPost`: tape `0` holds the zoned tape of the data extended by
  level `ℓ` on every track — on each side a separator and a fresh zone half full of virtual
  blanks (`Turing.zonedTape_extend` says which cells change); every other field unchanged;
* **exact time**, bounded by `C · (2^ℓ + 1)`;
* **envelopes** at level `ℓ`: the zoned head within `zoneStart (ℓ + 1)` of home, the scratch
  head within `C · (2^ℓ + 1)` of its start.

[AB09, §1.7]: zones are made half full "the first time we encounter it".

**Proof sketch.** Fill plan: walk right from home across the layout to the first of two
consecutive blanks (the separator after zone `ℓ - 1`); measure the new zone with a scratch
ruler of zone `ℓ - 1`'s length doubled (zone `ℓ` has twice its capacity; at `ℓ = 0` finite
control writes the two cells); write the inner half as all-tracks virtual blanks and the outer
half as all-`⊠` cells; return through home and do the same on the left, through a ruler of the
distance; clear the ruler and return home. Each phase is `O(2^ℓ)`. -/
theorem exists_zoneExtendTM (k : ℕ) :
    ∃ (C : ℕ) (S : Type) (_ : Fintype S) (_ : DecidableEq S)
      (M : MultiTapeTM 2 (ZoneSym k) S) (entry done : S),
      ∀ x : List (ZoneSym k),
        ∃ R : Contract M entry {some done} (ZoneData k) x,
          R.pre = zoneRowPre ∧
          R.post = zoneExtendPost done ∧
          R.lo = zoneRowLo (fun D => D.1) C ∧
          R.hi = zoneRowHi (fun D => D.1) C ∧
          ∀ D d, d.state = some entry → R.pre D d → R.time D d ≤ C * (2 ^ D.1 + 1) := by
  sorry

/-! ### What the rows' endpoints say (proved from the definitions) -/

/-- **Rows chain**: the rebalance row's endpoint, with its state replaced (as every seam
does), satisfies the row precondition at the rebalanced data. -/
theorem zoneRowPre_rebalancePost {k : ℕ} (dir : Bool) (t : Fin k) {S S' : Type}
    {x : List (ZoneSym k)} (done short : S) (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x)
    (hp : zoneRowPre D d) (f : S → S') :
    zoneRowPre (zoneRebalanceData dir t D) ((zoneRebalancePost dir t done short D d).mapState f) :=
  ⟨rfl, hp.2.1, hp.2.2⟩

/-- **Rows chain**: the extension row's endpoint satisfies the row precondition at the
extended data. -/
theorem zoneRowPre_extendPost {k : ℕ} {S S' : Type} {x : List (ZoneSym k)} (done : S)
    (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x) (hp : zoneRowPre D d) (f : S → S') :
    zoneRowPre (zoneExtendData D) ((zoneExtendPost done D d).mapState f) :=
  ⟨rfl, hp.2.1, hp.2.2⟩

/-- The rebalance exits at `done` exactly when its scanned index is below `ℓ`. -/
theorem zoneRebalancePost_state_done {k : ℕ} (dir : Bool) (t : Fin k) {S : Type}
    {x : List (ZoneSym k)} {done short : S} (hds : done ≠ short) (D : ZoneData k)
    (d : Cfg 2 (ZoneSym k) S x) :
    (zoneRebalancePost dir t done short D d).state = some done ↔
      zoneRebalanceIndex dir t D < D.1 := by
  simp only [zoneRebalancePost]
  split_ifs with h <;> simp [h, hds.symm]

/-- **A short exit leaves the data unchanged**: when every feeding zone of track `t` is empty
the rebalanced data is the data. -/
theorem zoneRebalanceData_of_short {k : ℕ} (dir : Bool) (t : Fin k) (D : ZoneData k)
    (h : ¬ zoneRebalanceIndex dir t D < D.1) : zoneRebalanceData dir t D = D := by
  obtain ⟨ℓ, z⟩ := D
  simp only [zoneRebalanceData, zoneRebalanceIndex] at h ⊢
  rw [zoneRebalance_of_le dir (z t) (by omega), Function.update_eq_self]

/-- **After a short exit, extend and retry**: the retried rebalance of the extended data scans
to the new level `ℓ`, below `ℓ + 1`, so it exits at `done`.

**Proof sketch.** A short exit means scanned index `ℓ` (`Turing.zoneShiftIndex_le`);
`Turing.zoneShiftIndex_extend`. -/
theorem zoneRebalanceIndex_extend_of_short {k : ℕ} (dir : Bool) (t : Fin k) (D : ZoneData k)
    (h : ¬ zoneRebalanceIndex dir t D < D.1) :
    zoneRebalanceIndex dir t (zoneExtendData D) = D.1 := by
  have hle := zoneShiftIndex_le dir (D.2 t)
  exact zoneShiftIndex_extend dir (D.2 t) (by unfold zoneRebalanceIndex at h; omega)

/-- **The rebalance frames the touched levels**: from an admissible start, its endpoint agrees
with the start outside the zoned tape's cells within `zoneStart (i₀ + 1)` of home (the levels up
to the scanned index `i₀`), and on the whole scratch tape.

**Proof sketch.** The scratch tape is the start's. On tape `0`, a cell farther out than
`zoneStart (i₀ + 1)` is a separator, beyond the layout, or owned by a zone above `i₀`, whose words
the rebalance keeps on track `t` (`Turing.zoneRebalance_frame`) and whose other tracks are
untouched. -/
theorem zoneRebalancePost_frames {k : ℕ} (dir : Bool) (t : Fin k) {S : Type}
    {x : List (ZoneSym k)} (done short : S) (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x)
    (hp : zoneRowPre D d) :
    Cfg.SameOutside (fun j => if j = 0 then
        Set.Icc (-(zoneStart (zoneRebalanceIndex dir t D + 1) : ℤ))
          (zoneStart (zoneRebalanceIndex dir t D + 1)) else ∅)
      d (zoneRebalancePost dir t done short D d) := by
  obtain ⟨ℓ, z⟩ := D
  intro j q hq
  simp only [zoneRebalancePost]
  split_ifs with hj
  · subst hj
    rw [hp.1]
    simp only [if_true, Set.mem_Icc, not_and_or, not_le] at hq
    set i₀ := zoneRebalanceIndex dir t ⟨ℓ, z⟩ with hi₀
    have key : ∀ (side : Bool) (n : ℕ), zoneStart (i₀ + 1) < n →
        zoneCellAt side (zoneRebalanceData dir t ⟨ℓ, z⟩).2 n = zoneCellAt side z n := by
      intro side n hn
      simp only [zoneCellAt, zoneRebalanceData]
      cases ho : zoneOwner ℓ n with
      | none => rfl
      | some i =>
        have hi := (Fin.find_eq_some_iff.mp ho).1
        have hgt : i₀ < i.val := by
          by_contra hle
          have := zoneStart_mono (show i.val + 1 ≤ i₀ + 1 by omega)
          have := zoneStart_succ i.val
          omega
        have hfr := zoneRebalance_frame dir (z t) i hgt
        simp only [Option.map_some]
        congr 2
        funext s
        by_cases hs : s = t
        · subst hs
          rw [Function.update_self]
          cases side
          · simp [ZoneContents.words, hfr.1]
          · simp [ZoneContents.words, hfr.2]
        · rw [Function.update_of_ne hs]
    unfold zonedTape
    have h0 : q ≠ 0 := by
      have := two_le_zoneStart (i₀ + 1)
      omega
    rw [if_neg h0, if_neg h0]
    split_ifs with hpos
    · exact key true _ (by omega)
    · exact key false _ (by omega)
  · rfl

/-- **The extension frames the new level**: from an admissible start, its endpoint agrees with
the start outside the two new zones on tape `0`, and on the whole scratch tape.

**Proof sketch.** `Turing.zonedTape_extend`: outside the new zones the extended tape is the old
one. -/
theorem zoneExtendPost_frames {k : ℕ} {S : Type} {x : List (ZoneSym k)} (done : S)
    (D : ZoneData k) (d : Cfg 2 (ZoneSym k) S x) (hp : zoneRowPre D d) :
    Cfg.SameOutside (fun j => if j = 0 then
        {c : ℤ | (zoneStart D.1 : ℤ) ≤ |c| ∧ |c| < zoneStart D.1 + zoneCapacity D.1} else ∅)
      d (zoneExtendPost done D d) := by
  intro j q hq
  simp only [zoneExtendPost]
  split_ifs with hj
  · subst hj
    rw [hp.1]
    simp only [if_true, Set.mem_setOf_eq] at hq
    rw [zoneExtendData, zonedTape_extend, if_neg hq]
  · rfl

end Turing
