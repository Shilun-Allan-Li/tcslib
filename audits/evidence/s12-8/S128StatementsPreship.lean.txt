/-! # §12.8 statement layer — pre-ship probe (executed checks against the landed modules)

Run against the s12.8 olean tree with `LEAN_PATH` (no olean output). Every `example` is
decided by evaluation on concrete configurations over `Fin 3` (or `Bool` for the counter
loop), at displaced heads, over nonblank outer frames, and at the degenerate sizes
(`w = []`, `u = []`, `ws = []`, `s = 0`, `k = 0`, a halted source, counter value `0`,
empty pattern). Expected output: nothing (0 errors, 0 warnings). -/

/-! ## Pre-ship executed checks: Catalog Part 3 (one or two tapes over `Fin 3`) -/
namespace S128Check
open Turing

/-- A one-tape configuration with tape contents `t` and head `h`. -/
def c1 {S : Type} (q : S) (t : ℤ → Option (Fin 3)) (h : ℤ) : Cfg 1 (Fin 3) S ([] : List (Fin 3)) :=
  ⟨some q, 1, fun _ => t, fun _ => h, []⟩

-- a tape: the given cells, and `2` (nonblank frame) at the listed outer cells
def tp (cells : List (ℤ × Fin 3)) (frame : List ℤ) (z : ℤ) : Option (Fin 3) :=
  match cells.find? (·.1 = z) with
  | some (_, a) => some a
  | none => if z ∈ frame then some 2 else none

-- skip: word [0,1,2] at 5, nonblank frame at 4 and 9 (outside `[pos, pos+|w|]`)
example :
    let d := c1 SeekPhase.run (tp [(5, 0), (6, 1), (7, 2)] [4, 9]) 5
    ((skipTM 1 0).runFrom d 4).state = some .done ∧
    ((skipTM 1 0).runFrom d 3).state ≠ some .done ∧
    ((skipTM 1 0).runFrom d 4).workTapePos 0 = 9 ∧
    ((skipTM 1 0).runFrom d 4).workTapes 0 9 = some 2 := by decide
-- skip, w = []: one step
example :
    let d := c1 SeekPhase.run (tp [] [4, 6]) 5
    ((skipTM 1 0).runFrom d 1).state = some .done ∧ ((skipTM 1 0).runFrom d 1).workTapePos 0 = 6 := by
  decide
-- skip back: u = [1,2] at [7,9), separator 9, delimiter 6, current cell 10, frame at 5
example :
    let d := c1 SeekPhase.run (tp [(7, 1), (8, 2), (10, 0)] [5, 11]) 10
    ((skipBackTM 1 0).runFrom d 5).state = some .done ∧
    ((skipBackTM 1 0).runFrom d 4).state ≠ some .done ∧
    ((skipBackTM 1 0).runFrom d 5).workTapePos 0 = 7 := by decide
-- skip back, u = []
example :
    let d := c1 SeekPhase.run (tp [(10, 0)] [7]) 10
    ((skipBackTM 1 0).runFrom d 3).state = some .done ∧
    ((skipBackTM 1 0).runFrom d 2).state ≠ some .done ∧
    ((skipBackTM 1 0).runFrom d 3).workTapePos 0 = 9 := by decide
-- seek end: ws = [[1],[2,0]] at 3 (D = 5), frame beyond at 9
example :
    let d := c1 SeekPhase.run (tp [(3, 1), (5, 2), (6, 0)] [9, 2]) 3
    fieldOffset ([[1], [2, 0]] : List (List (Fin 3))) 2 = 5 ∧
    ((seekEndTM 1 0).runFrom d 6).state = some .done ∧
    ((seekEndTM 1 0).runFrom d 5).state ≠ some .done ∧
    ((seekEndTM 1 0).runFrom d 6).workTapePos 0 = 8 := by decide
-- seek start: ws = [[1],[2,0]] at 15 (D = 5), current cell 20, frame at 12 and 21
example :
    let d := c1 SeekPhase.run (tp [(15, 1), (17, 2), (18, 0), (20, 1)] [12, 21]) 20
    ((seekStartTM 1 0).runFrom d 9).state = some .done ∧
    ((seekStartTM 1 0).runFrom d 8).state ≠ some .done ∧
    ((seekStartTM 1 0).runFrom d 9).workTapePos 0 = 15 := by decide
-- seek start, ws = []: four steps, back where it was
example :
    let d := c1 SeekPhase.run (tp [(20, 1)] [17]) 20
    ((seekStartTM 1 0).runFrom d 4).state = some .done ∧
    ((seekStartTM 1 0).runFrom d 3).state ≠ some .done ∧
    ((seekStartTM 1 0).runFrom d 4).workTapePos 0 = 20 := by decide
-- the field layout: field 1 of [[1],[2,0],[1]] starts at 2, framed
example :
    let fs : List (List (Fin 3)) := [[1], [2, 0], [1]]
    fieldOffset fs 1 = 2 ∧ fieldsTape fs 1 = none ∧ fieldsTape fs 2 = some 2 ∧
    fieldsTape fs 3 = some 0 ∧ fieldsTape fs 4 = none ∧ fieldsTape fs 5 = some 1 ∧
    fieldsTape fs 6 = none ∧ fieldsTape fs (-1) = none := by decide
-- scan: fold (sum mod 3) over [2,0,2] at displaced head 4, nonblank frame at 2 and 8
example :
    let d := c1 (ScanPhase.fwd (0 : Fin 3)) (tp [(4, 2), (5, 0), (6, 2)] [2, 8]) 4
    let M := scanTM 1 0 (scanFold fun a γ => a + γ) scanKeep (0 : Fin 3)
    (M.runFrom d 8).state = some (.done 1) ∧ (M.runFrom d 7).state ≠ some (.done 1) ∧
    (M.runFrom d 8).workTapes 0 6 = some 2 ∧ (M.runFrom d 8).workTapePos 0 = 4 ∧
    (M.runFrom d 8).workTapes 0 8 = some 2 := by decide
-- scan, w = []: two steps, verdict a₀
example :
    let d := c1 (ScanPhase.fwd (1 : Fin 3)) (tp [] [2, 6]) 4
    ((scanTM 1 0 (scanMap (· + 1)) scanKeep (1 : Fin 3)).runFrom d 2).state = some (.done 1) := by
  decide
-- the instance identities on concrete words
example : scanResult (scanFold fun (a γ : Fin 3) => a + γ) scanKeep 0 [2, 0, 2] = (1, [2, 0, 2]) ∧
    scanResult (scanMap (· + 1)) scanKeep (0 : Fin 3) ([2, 0, 1] : List (Fin 3)) = (0, [0, 1, 2]) ∧
    scanResult (scanShift (fun γ : Fin 3 => decide (γ = 2)) (fun γ b => if b then 1 else γ))
      scanKeep false [2, 0, 0] = (false, [2, 1, 0]) ∧
    scanResult scanKeep (scanShift (fun γ : Fin 3 => decide (γ = 2)) (fun γ b => if b then 1 else γ))
      false [0, 0, 2] = (false, [0, 1, 2]) ∧
    scanResult (scanCarry (fun γ : Fin 3 => decide (γ = 2)) (· + 1)) scanKeep true [2, 2, 1, 2] =
      (false, [0, 0, 2, 2]) ∧
    scanResult (scanCarry (fun γ : Fin 3 => decide (γ = 2)) (· + 1)) scanKeep true [2, 2] =
      (true, [0, 0]) := by decide
-- the identities' right-hand sides on the same words
example :
    ((([2, 0, 0] : List (Fin 3)).map (fun γ => decide (γ = 2))).getLast?.getD false,
      List.zipWith (fun (γ : Fin 3) (b : Bool) => if b then 1 else γ) [2, 0, 0]
        (false :: ([2, 0, 0] : List (Fin 3)).map (fun γ => decide (γ = 2)))) = (false, [2, 1, 0]) ∧
    ((([0, 0, 2] : List (Fin 3)).map (fun γ => decide (γ = 2))).head?.getD false,
      List.zipWith (fun (γ : Fin 3) (b : Bool) => if b then 1 else γ) [0, 0, 2]
        (([0, 0, 2] : List (Fin 3)).tail.map (fun γ => decide (γ = 2)) ++ [false])) =
      (false, [0, 1, 2]) ∧
    (let w : List (Fin 3) := [2, 2, 1, 2]
     (w.all (fun γ => decide (γ = 2)),
      (w.takeWhile (fun γ => decide (γ = 2))).map (· + 1) ++
        ((w.dropWhile (fun γ => decide (γ = 2))).take 1).map (· + 1) ++
        (w.dropWhile (fun γ => decide (γ = 2))).drop 1)) = (false, [0, 0, 2, 2]) := by decide
-- framed compare: equal words at displaced heads with nonblank outer frames
def c2 (t₀ t₁ : ℤ → Option (Fin 3)) (h₀ h₁ : ℤ) : Cfg 2 (Fin 3) FlagPhase ([] : List (Fin 3)) :=
  ⟨some .run, 1, fun j => if j = 0 then t₀ else t₁, fun j => if j = 0 then h₀ else h₁, []⟩
example :
    let d := c2 (tp [(5, 1), (6, 2)] [3, 8]) (tp [(2, 1), (3, 2)] [0, 5]) 5 2
    ((compareTM 2 0 1).runFrom d 6).state = some (FlagPhase.done true) ∧
    ((compareTM 2 0 1).runFrom d 5).state ≠ some (FlagPhase.done true) ∧
    ((compareTM 2 0 1).runFrom d 6).workTapePos 0 = 5 ∧
    ((compareTM 2 0 1).runFrom d 6).workTapePos 1 = 2 := by decide
-- u a proper prefix of v (δ = |u| = 1): false at 4
example :
    let d := c2 (tp [(5, 1)] [3, 7]) (tp [(2, 1), (3, 2)] [0, 5]) 5 2
    ((compareTM 2 0 1).runFrom d 4).state = some (FlagPhase.done false) ∧
    ((compareTM 2 0 1).runFrom d 3).state ≠ some (FlagPhase.done false) := by decide
-- u = v = []: true at 2
example :
    let d := c2 (tp [] [3, 6]) (tp [] [0, 3]) 5 2
    ((compareTM 2 0 1).runFrom d 2).state = some (FlagPhase.done true) := by decide

end S128Check

/-- A one-tape configuration over input `x`, input head at `1`. -/
def ci {S : Type} (x : List (Fin 3)) (q : S) (t : ℤ → Option (Fin 3)) (h : ℤ) :
    Cfg 1 (Fin 3) S x :=
  ⟨some q, 1, fun _ => t, fun _ => h, []⟩

-- inputCopy of [1,2] at displaced head 7 over nonblank cells, frame at 6 and 9
example :
    let d := ci [1, 2] SweepPhase.sweep (tp [(7, 0), (8, 0)] [6, 9]) 7
    let M := inputCopyTM (Symbol := Fin 3) 1 0
    (M.runFrom d 6).state = some .done ∧ (M.runFrom d 5).state ≠ some .done ∧
    (M.runFrom d 6).workTapes 0 7 = some 1 ∧ (M.runFrom d 6).workTapes 0 8 = some 2 ∧
    (M.runFrom d 6).workTapes 0 6 = some 2 ∧ (M.runFrom d 6).workTapes 0 9 = some 2 ∧
    (M.runFrom d 6).workTapePos 0 = 7 ∧ ((M.runFrom d 6).inputPos : ℕ) = 1 := by decide
-- inputCopy, x = []: two steps
example :
    let d := ci [] SweepPhase.sweep (tp [] [6]) 7
    ((inputCopyTM (Symbol := Fin 3) 1 0).runFrom d 2).state = some .done ∧
    ((inputCopyTM (Symbol := Fin 3) 1 0).runFrom d 1).state ≠ some .done ∧
    (((inputCopyTM (Symbol := Fin 3) 1 0).runFrom d 2).inputPos : ℕ) = 1 := by decide
-- inputFold: validator "no symbol is 2" over [1,0,2] and over [1,0]
example :
    let δ : Bool → Fin 3 → Bool := fun ok γ => ok && decide (γ ≠ 2)
    ((inputFoldTM 1 δ true).runFrom (ci [1, 0, 2] (ScanPhase.fwd true) (tp [] [0]) 3) 8).state =
      some (.done false) ∧
    ((inputFoldTM 1 δ true).runFrom (ci [1, 0, 2] (ScanPhase.fwd true) (tp [] [0]) 3) 7).state ≠
      some (.done false) ∧
    ((inputFoldTM 1 δ true).runFrom (ci [1, 0] (ScanPhase.fwd true) (tp [] [0]) 3) 6).state =
      some (.done true) ∧
    (((inputFoldTM 1 δ true).runFrom (ci [1, 0] (ScanPhase.fwd true) (tp [] [0]) 3) 6).inputPos : ℕ)
      = 1 ∧
    ((inputFoldTM 1 δ true).runFrom (ci [] (ScanPhase.fwd true) (tp [] [0]) 3) 2).state =
      some (.done true) := by decide

end S128CheckScan

/-! ## Pre-ship executed checks: expandTM -/
namespace S128CheckExpand
open Turing

/-- The body configuration: one Bool tape, blank except the listed cells, head `h`. -/
def cb (cells : List (ℤ × Bool)) (h : ℤ) : Cfg 1 Bool (Fin 3) ([] : List Bool) :=
  ⟨some 0, 1, fun _ z => (cells.find? (·.1 = z)).map (·.2), fun _ => h, []⟩

/-- A counter tape holding `w` from `cp`, with a nonblank frame cell at `cp + |w| + 1`. -/
def ctape (w : List Bool) (cp : ℤ) (z : ℤ) : Option Bool :=
  if z = cp + w.length + 1 then some true else FinTM.bufferTape w (z - cp)

-- counter [false, true] (value 2), pattern [true, false]: time 2·3 + overhead
example :
    let w := [false, true]
    let T := 2 * 3 + counterOverhead w 3
    let d := counterLoopCfg (some (.dec .run)) (cb [(5, false), (8, true)] 5) (ctape w 3) 3
    let M := expandTM 1 0 [true, false]
    (M.runFrom d T).state = some .done ∧ (M.runFrom d (T - 1)).state ≠ some .done ∧
    (M.runFrom d T).workTapes 0 5 = some true ∧ (M.runFrom d T).workTapes 0 6 = some false ∧
    (M.runFrom d T).workTapes 0 7 = some true ∧ (M.runFrom d T).workTapes 0 8 = some false ∧
    (M.runFrom d T).workTapes 0 9 = none ∧
    (M.runFrom d T).workTapePos 0 = 9 ∧ (M.runFrom d T).workTapePos 1 = 3 ∧
    (M.runFrom d T).workTapes 1 3 = some true ∧ (M.runFrom d T).workTapes 1 4 = some true ∧
    (M.runFrom d T).workTapes 1 6 = some true := by decide
-- counter value 0 ([false]): only the underflow sweep, nothing written
example :
    let w := [false]
    let T := 0 * 3 + counterOverhead w 1
    let d := counterLoopCfg (some (.dec .run)) (cb [] 5) (ctape w 3) 3
    let M := expandTM 1 0 [true, false]
    (M.runFrom d T).state = some .done ∧ (M.runFrom d (T - 1)).state ≠ some .done ∧
    (M.runFrom d T).workTapePos 0 = 5 ∧ (M.runFrom d T).workTapes 0 5 = none := by decide
-- empty pattern, counter value 1 ([true]): one stationary round
example :
    let w := [true]
    let T := 1 * 1 + counterOverhead w 2
    let d := counterLoopCfg (some (.dec .run)) (⟨some 0, 1, fun _ _ => none, fun _ => 5, []⟩ :
      Cfg 1 Bool (Fin 1) ([] : List Bool)) (ctape w 3) 3
    let M := expandTM 1 0 []
    (M.runFrom d T).state = some .done ∧ (M.runFrom d (T - 1)).state ≠ some .done ∧
    (M.runFrom d T).workTapePos 0 = 5 := by decide

end S128CheckExpand

/-! ## Pre-ship executed checks: the two-exit seam -/
namespace S128CheckSeam
open Turing S128Check

-- host: framed compare (exits `done true` / `done false`); continuations: skip tape 0 / skip tape 1
-- equal words [1,2] at heads 5 and 2: host exit at 6, dispatch, skip tape 0 in 3 steps
example :
    let c := c2 (tp [(5, 1), (6, 2)] [3, 8]) (tp [(2, 1), (3, 2)] [0, 5]) 5 2
    let M := seamCompTM₂ (compareTM 2 0 1) (FlagPhase.done true) (FlagPhase.done false)
      (skipTM 2 0) SeekPhase.run (skipTM 2 1) SeekPhase.run
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (6 + 1 + 3)).state =
      some (Sum.inl (Sum.inr SeekPhase.done)) ∧
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (6 + 1 + 3)).workTapePos 0 = 8 ∧
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (6 + 1 + 3)).workTapePos 1 = 2 ∧
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (6 + 1 + 2)).state =
      some (Sum.inl (Sum.inr SeekPhase.run)) := by decide
-- unequal words [1] and [1,2]: host exit `done false` at 4, then skip tape 1 in 3 steps
example :
    let c := c2 (tp [(5, 1)] [3, 7]) (tp [(2, 1), (3, 2)] [0, 5]) 5 2
    let M := seamCompTM₂ (compareTM 2 0 1) (FlagPhase.done true) (FlagPhase.done false)
      (skipTM 2 0) SeekPhase.run (skipTM 2 1) SeekPhase.run
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (4 + 1 + 3)).state =
      some (Sum.inr SeekPhase.done) ∧
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (4 + 1 + 3)).workTapePos 1 = 5 ∧
    (M.runFrom ((c.mapState Sum.inl).mapState Sum.inl) (4 + 1 + 3)).workTapePos 0 = 5 := by decide

end S128CheckSeam
