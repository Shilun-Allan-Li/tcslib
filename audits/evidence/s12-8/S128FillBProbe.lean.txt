/-! ## Part 3 — `inputFoldTM_run_ofCfg` as a trace instance -/

section Fold

variable {Γ : Type*} {A : Type} {k : ℕ} {x : List Γ}

/-- Forward family: after `r` steps the state is the fold of the first `r` symbols and
the input head is at `r + 1` (`min` makes the family total; only `r ≤ |x|` is used). -/
def probeFoldF (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x) (r : ℕ) :
    Cfg k Γ (ScanPhase A) x :=
  { d with state := some (.fwd ((x.take r).foldl δ a₀))
           inputPos := ⟨min r x.length + 1, by omega⟩ }

/-- Return family, indexed by the input position `r`, counted down to the left blank. -/
def probeFoldR (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x) (r : ℕ) :
    Cfg k Γ (ScanPhase A) x :=
  { d with state := some (.bwd (x.foldl δ a₀))
           inputPos := ⟨min r x.length, by omega⟩ }

/-- Exit: the target's finish configuration. -/
def probeFoldD (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x) :
    Cfg k Γ (ScanPhase A) x :=
  { d with state := some (.done (x.foldl δ a₀)) }

theorem probe_fold_F0 (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x)
    (hstate : d.state = some (.fwd a₀)) (hin : d.inputPos = 1) :
    probeFoldF δ a₀ d 0 = d := by
  apply Cfg.ext <;> simp [probeFoldF, hstate, hin]

/-- **The forward step equation** (`hF`, at every `r < |x|`; the required check). -/
theorem probe_fold_hF (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x)
    (r : ℕ) (hr : r < x.length) :
    (inputFoldTM k δ a₀).step (probeFoldF δ a₀ d r) = probeFoldF δ a₀ d (r + 1) := by
  have hs : (probeFoldF δ a₀ d r).inputSymbol = some x[r] :=
    inputSymbolInner r (by simp [probeFoldF]; omega) hr
  change ((inputFoldTM k δ a₀).tr (.fwd _) _ _).apply _ = _
  simp only [inputFoldTM, hs]
  apply Cfg.ext
  case state =>
    simp only [Action.apply, inputMoveAction, probeFoldF]
    rw [List.take_succ, List.getElem?_eq_getElem hr, List.foldl_append]
    rfl
  case inputPos =>
    simp only [Action.apply, inputMoveAction, probeFoldF]
    rw [← SignType.pos_eq_one, moveInputPos_pos_of_ne_right _ (by simp; omega)]
    simp
    omega
  all_goals simp [Action.apply, inputMoveAction, probeFoldF]

theorem probe_fold_hturn (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x) :
    (inputFoldTM k δ a₀).step (probeFoldF δ a₀ d x.length) =
      probeFoldR δ a₀ d x.length := by
  have hs : (probeFoldF δ a₀ d x.length).inputSymbol = none := by
    simp [Cfg.inputSymbol, probeFoldF]
  change ((inputFoldTM k δ a₀).tr (.fwd _) _ _).apply _ = _
  simp only [inputFoldTM, hs]
  apply Cfg.ext
  case state => simp [Action.apply, inputMoveAction, probeFoldF, probeFoldR]
  case inputPos =>
    simp only [Action.apply, inputMoveAction, probeFoldF, probeFoldR]
    apply Fin.ext
    rw [← SignType.neg_eq_neg_one, FinTM.moveInputPos_neg_val]
    simp
  all_goals simp [Action.apply, inputMoveAction, probeFoldF, probeFoldR]

theorem probe_fold_hR (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x)
    (r : ℕ) (hr : r < x.length) :
    (inputFoldTM k δ a₀).step (probeFoldR δ a₀ d (r + 1)) = probeFoldR δ a₀ d r := by
  have hs : (probeFoldR δ a₀ d (r + 1)).inputSymbol = some x[r] :=
    inputSymbolInner r (by simp [probeFoldR]; omega) hr
  change ((inputFoldTM k δ a₀).tr (.bwd _) _ _).apply _ = _
  simp only [inputFoldTM, hs]
  apply Cfg.ext
  case inputPos =>
    simp only [Action.apply, inputMoveAction, probeFoldR]
    apply Fin.ext
    rw [← SignType.neg_eq_neg_one, FinTM.moveInputPos_neg_val]
    simp
    omega
  all_goals simp [Action.apply, inputMoveAction, probeFoldR]

theorem probe_fold_hentry (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x)
    (hin : d.inputPos = 1) :
    (inputFoldTM k δ a₀).step (probeFoldR δ a₀ d 0) = probeFoldD δ a₀ d := by
  have hs : (probeFoldR δ a₀ d 0).inputSymbol = none := by
    simp [Cfg.inputSymbol, probeFoldR]
  change ((inputFoldTM k δ a₀).tr (.bwd _) _ _).apply _ = _
  simp only [inputFoldTM, hs]
  apply Cfg.ext
  case inputPos =>
    simp only [Action.apply, inputMoveAction, probeFoldR, probeFoldD, hin]
    rw [← SignType.pos_eq_one, moveInputPos_pos_of_ne_right _ (by simp)]
    apply Fin.ext
    simp
  all_goals simp [Action.apply, inputMoveAction, probeFoldR, probeFoldD]

theorem probe_fold_hD (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x) :
    (inputFoldTM k δ a₀).step (probeFoldD δ a₀ d) = probeFoldD δ a₀ d := by
  apply Cfg.ext <;> simp [MultiTapeTM.step, inputFoldTM, Action.apply, inputMoveAction,
    probeFoldD]

/-- The input fold's run is the two-pass trace at every time. -/
theorem probe_fold_trace (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x)
    (hstate : d.state = some (.fwd a₀)) (hin : d.inputPos = 1) (t : ℕ) :
    (inputFoldTM k δ a₀).runFrom d t =
      catalogTrace (probeFoldF δ a₀ d) (probeFoldR δ a₀ d) (probeFoldD δ a₀ d) x.length t := by
  conv_lhs => rw [← probe_fold_F0 δ a₀ d hstate hin]
  exact catalog_trace_run _ _ _ _ _ (probe_fold_hF δ a₀ d) (probe_fold_hturn δ a₀ d)
    (probe_fold_hR δ a₀ d) (probe_fold_hentry δ a₀ d hin) (probe_fold_hD δ a₀ d) t

/-- `inputFoldTM_run_ofCfg` (statement copied from `Build/Scan.lean:265`) from the trace. -/
theorem probe_inputFold_contract (δ : A → Γ → A) (a₀ : A) (d : Cfg k Γ (ScanPhase A) x)
    (hstate : d.state = some (.fwd a₀)) (hin : d.inputPos = 1) :
    (inputFoldTM k δ a₀).runFrom d (2 * x.length + 2) =
        { d with state := some (.done (x.foldl δ a₀)) } ∧
      (∀ t < 2 * x.length + 2, ∀ a,
        ((inputFoldTM k δ a₀).runFrom d t).state ≠ some (.done a)) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * x.length + 2 →
        ((inputFoldTM k δ a₀).runFrom d t).workTapePos j = d.workTapePos j) := by
  refine ⟨?_, ?_, ?_⟩
  · rw [probe_fold_trace δ a₀ d hstate hin]
    simp [catalogTrace, probeFoldD, show ¬2 * x.length + 2 ≤ x.length by omega]
  · intro t ht a
    rw [probe_fold_trace δ a₀ d hstate hin]
    simp only [catalogTrace]
    split_ifs <;> simp [probeFoldF, probeFoldR] <;> omega
  · intro j t _
    rw [probe_fold_trace δ a₀ d hstate hin]
    simp only [catalogTrace]
    split_ifs <;> simp [probeFoldF, probeFoldR, probeFoldD]

end Fold

/-! ## Part 4 — the shared write family, the two writers and the copier -/

section Write

variable {Symbol : Type*} {k : ℕ} {x : List Symbol}

/-- The framed write family shared by the three writing rows: `r` cells of `w` written
from the head of tape `i`, that head advanced by `r`, control `q`, input position `p`,
and everything else as in `d`. -/
def probeWriteCfg {S : Type*} (i : Fin k) (w : List Symbol) (d : Cfg k Symbol S x) (q : S)
    (p : Fin (x.length + 2)) (r : ℕ) : Cfg k Symbol S x :=
  { d with state := some q
           inputPos := p
           workTapes := fun j z =>
             if j = i ∧ d.workTapePos j ≤ z ∧ z < d.workTapePos j + (r : ℤ)
               then FinTM.bufferTape w (z - d.workTapePos j) else d.workTapes j z
           workTapePos := fun j => d.workTapePos j + if j = i then (r : ℤ) else 0 }

/-- The one write step shared by `writeConstTM`, `writeAdvTM` and `inputCopyTM`: writing
`w[r]` and moving right extends the written prefix by one cell. -/
theorem probe_write_step {S : Type*} (i : Fin k) (w : List Symbol) (d : Cfg k Symbol S x)
    (q q' : S) (p : Fin (x.length + 2)) (inp : SignType) (r : ℕ) (hr : r < w.length) :
    (tapeAction i inp (some (some w[r])) .pos q').apply (probeWriteCfg i w d q p r) =
      probeWriteCfg i w d q' (moveInputPos p inp) (r + 1) := by
  apply Cfg.ext
  case workTapes =>
    simp only [Action.apply, tapeAction, probeWriteCfg]
    funext j
    by_cases hj : j = i
    · subst j
      simp only [if_true, true_and]
      exact catalog_write_take _ _ w r hr
    · simp [hj]
  case workTapePos =>
    funext j
    by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeWriteCfg, hj] <;> omega
  all_goals simp [Action.apply, tapeAction, probeWriteCfg]

/-! ### `writeConstTM` -/

def probeWcF (i : Fin k) (w₀ : List Symbol) (d : Cfg k Symbol (WritePhase w₀.length) x)
    (r : ℕ) : Cfg k Symbol (WritePhase w₀.length) x :=
  probeWriteCfg i w₀ d (.wr ⟨min r w₀.length, by omega⟩) d.inputPos r

def probeWcR (i : Fin k) (w₀ : List Symbol) (d : Cfg k Symbol (WritePhase w₀.length) x)
    (r : ℕ) : Cfg k Symbol (WritePhase w₀.length) x :=
  { probeWriteCfg i w₀ d (.bk ⟨min r w₀.length, by omega⟩) d.inputPos w₀.length with
      workTapePos := fun j => d.workTapePos j + if j = i then (r : ℤ) else 0 }

def probeWcD (i : Fin k) (w₀ : List Symbol) (d : Cfg k Symbol (WritePhase w₀.length) x) :
    Cfg k Symbol (WritePhase w₀.length) x :=
  { d with state := some .done
           workTapes := (probeWcF i w₀ d w₀.length).workTapes }

theorem probe_wc_trace (i : Fin k) (w₀ : List Symbol) (d : Cfg k Symbol (WritePhase w₀.length) x)
    (hstate : d.state = some (.wr 0)) (t : ℕ) :
    (writeConstTM k i w₀).runFrom d t =
      catalogTrace (probeWcF i w₀ d) (probeWcR i w₀ d) (probeWcD i w₀ d) w₀.length t := by
  have h0 : probeWcF i w₀ d 0 = d := by
    apply Cfg.ext
    case state => simp [probeWcF, probeWriteCfg, hstate]
    case workTapes =>
      funext j z
      simp [probeWcF, probeWriteCfg]
    all_goals simp [probeWcF, probeWriteCfg]
  conv_lhs => rw [← h0]
  apply catalog_trace_run
  · -- forward: one write step
    intro r hr
    change ((writeConstTM k i w₀).tr (.wr _) _ _).apply _ = _
    simp only [writeConstTM, probeWcF]
    rw [dif_pos (show min r w₀.length < w₀.length by omega)]
    simp only [show min r w₀.length = r by omega, show min (r + 1) w₀.length = r + 1 by omega]
    rw [probe_write_step i w₀ d _ _ _ _ r hr, moveInputPos_zero]
  · -- turn: stationary, into the walk back
    change ((writeConstTM k i w₀).tr (.wr _) _ _).apply _ = _
    simp only [writeConstTM]
    rw [dif_neg (show ¬ min w₀.length w₀.length < w₀.length by omega)]
    apply Cfg.ext <;> simp [Action.apply, inputMoveAction, probeWcF, probeWcR, probeWriteCfg]
  · -- return: one left move
    intro r hr
    change ((writeConstTM k i w₀).tr (.bk _) _ _).apply _ = _
    simp only [writeConstTM]
    rw [if_neg (show ¬ min (r + 1) w₀.length = 0 by omega)]
    apply Cfg.ext
    case state => simp [Action.apply, tapeAction, probeWcR, probeWriteCfg]; omega
    case workTapes =>
      funext j
      by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeWcR, probeWriteCfg, hj]
    case workTapePos =>
      funext j
      by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeWcR, probeWriteCfg, hj] <;>
        omega
    all_goals simp [Action.apply, tapeAction, probeWcR, probeWriteCfg]
  · -- entry: stationary, into `done`
    change ((writeConstTM k i w₀).tr (.bk _) _ _).apply _ = _
    simp only [writeConstTM]
    rw [if_pos (show min 0 w₀.length = 0 by omega)]
    apply Cfg.ext <;>
      simp [Action.apply, inputMoveAction, probeWcR, probeWcD, probeWcF, probeWriteCfg]
  · apply Cfg.ext <;> simp [MultiTapeTM.step, writeConstTM, Action.apply, inputMoveAction,
      probeWcD]

/-! ### `writeAdvTM` (cyclic: one forward induction, no trace instance) -/

def probeWaF (i : Fin k) (w₀ : List Symbol) (d : Cfg k Symbol (Fin (w₀.length + 1)) x)
    (r : ℕ) : Cfg k Symbol (Fin (w₀.length + 1)) x :=
  probeWriteCfg i w₀ d ⟨min r w₀.length, by omega⟩ d.inputPos r

theorem probe_wa_run (i : Fin k) (w₀ : List Symbol) (d : Cfg k Symbol (Fin (w₀.length + 1)) x)
    (hstate : d.state = some 0) :
    ∀ r ≤ w₀.length, (writeAdvTM k i w₀).runFrom d r = probeWaF i w₀ d r := by
  intro r hr
  induction r with
  | zero =>
    apply Cfg.ext
    case state => simp [probeWaF, probeWriteCfg, hstate]
    case workTapes =>
      funext j z
      simp [probeWaF, probeWriteCfg]
    all_goals simp [probeWaF, probeWriteCfg]
  | succ r ih =>
    rw [MultiTapeTM.runFrom_succ_eq_step', ih (by omega)]
    change ((writeAdvTM k i w₀).tr _ _ _).apply _ = _
    simp only [writeAdvTM, probeWaF]
    rw [dif_pos (show min r w₀.length < w₀.length by omega)]
    simp only [show min r w₀.length = r by omega, show min (r + 1) w₀.length = r + 1 by omega]
    rw [probe_write_step i w₀ d _ _ _ _ r (by omega), moveInputPos_zero]

/-- `writeAdvTM_run_ofCfg` (statement copied from `Build/Scan.lean:162`). -/
theorem probe_writeAdv_contract (i : Fin k)
    (w₀ : List Symbol) (d : Cfg k Symbol (Fin (w₀.length + 1)) x)
    (hstate : d.state = some 0) :
    (writeAdvTM k i w₀).runFrom d (w₀.length + 1) =
        { d with
          state := some 0
          workTapes := fun j q =>

            else d.workTapes j q

      (∀ t, 0 < t → t < w₀.length + 1 → ((writeAdvTM k i w₀).runFrom d t).state ≠ some 0) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ w₀.length + 1 →
        if j = i then
          ((writeAdvTM k i w₀).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j) (d.workTapePos j + (w₀.length : ℤ))
        else ((writeAdvTM k i w₀).runFrom d t).workTapePos j = d.workTapePos j) := by
  -- the last, stationary step returns state `|w₀|` to `0`
  have hlast : (writeAdvTM k i w₀).runFrom d (w₀.length + 1) =
      { probeWaF i w₀ d w₀.length with state := some 0 } := by
    rw [MultiTapeTM.runFrom_succ_eq_step', probe_wa_run i w₀ d hstate _ le_rfl]
    change ((writeAdvTM k i w₀).tr _ _ _).apply _ = _
    simp only [writeAdvTM, probeWaF, probeWriteCfg]
    rw [dif_neg (show ¬ min w₀.length w₀.length < w₀.length by omega)]
    apply Cfg.ext <;> simp [Action.apply, inputMoveAction]
  refine ⟨?_, ?_, ?_⟩
  · rw [hlast]
    apply Cfg.ext
    case workTapePos =>
      funext j
      by_cases hj : j = i
      · subst j; simp [probeWaF, probeWriteCfg]
      · simp [probeWaF, probeWriteCfg, hj]
    all_goals simp [probeWaF, probeWriteCfg]
  · intro t ht0 ht
    rw [probe_wa_run i w₀ d hstate t (by omega)]
    simp only [probeWaF, probeWriteCfg, ne_eq, Option.some.injEq, Fin.ext_iff, Fin.val_zero]
    omega
  · intro j t ht
    rcases Nat.lt_or_ge t (w₀.length + 1) with h | h
    · rw [probe_wa_run i w₀ d hstate t (by omega)]
      split_ifs with hj <;> simp [probeWaF, probeWriteCfg, hj, Finset.mem_Icc] <;> omega
    · rw [show t = w₀.length + 1 by omega, hlast]
      split_ifs with hj <;> simp [probeWaF, probeWriteCfg, hj, Finset.mem_Icc]

/-! ### `inputCopyTM` -/

def probeIcF (i : Fin k) (d : Cfg k Symbol SweepPhase x) (r : ℕ) : Cfg k Symbol SweepPhase x :=
  probeWriteCfg i x d .sweep ⟨min r x.length + 1, by omega⟩ r

def probeIcR (i : Fin k) (d : Cfg k Symbol SweepPhase x) (r : ℕ) : Cfg k Symbol SweepPhase x :=
  { probeWriteCfg i x d .rewind ⟨min r x.length, by omega⟩ x.length with
      workTapePos := fun j => d.workTapePos j + if j = i then (r : ℤ) - 1 else 0 }

def probeIcD (i : Fin k) (d : Cfg k Symbol SweepPhase x) : Cfg k Symbol SweepPhase x :=
  { d with state := some .done
           workTapes := (probeIcF i d x.length).workTapes }

theorem probe_ic_trace (i : Fin k) (d : Cfg k Symbol SweepPhase x)
    (hstate : d.state = some .sweep) (hin : d.inputPos = 1) (t : ℕ) :
    (inputCopyTM k i).runFrom d t =
      catalogTrace (probeIcF i d) (probeIcR i d) (probeIcD i d) x.length t := by
  have h0 : probeIcF i d 0 = d := by
    apply Cfg.ext
    case inputPos => rw [hin]; apply Fin.ext; simp [probeIcF, probeWriteCfg]
    case workTapes =>
      funext j z
      simp [probeIcF, probeWriteCfg]
    all_goals simp [probeIcF, probeWriteCfg, hstate]
  conv_lhs => rw [← h0]
  apply catalog_trace_run
  · -- forward: one lockstep write step
    intro r hr
    have hs : (probeIcF i d r).inputSymbol = some x[r] :=
      inputSymbolInner r (by simp [probeIcF, probeWriteCfg]; omega) hr
    change ((inputCopyTM k i).tr .sweep _ _).apply _ = _
    simp only [inputCopyTM, hs]
    rw [probeIcF, probe_write_step i x d _ _ _ _ r hr]
    congr 1
    rw [← SignType.pos_eq_one, moveInputPos_pos_of_ne_right _ (by simp; omega)]
    simp
    omega
  · -- turn at the right end marker: both heads left
    have hs : (probeIcF i d x.length).inputSymbol = none := by
      simp [Cfg.inputSymbol, probeIcF, probeWriteCfg]
    change ((inputCopyTM k i).tr .sweep _ _).apply _ = _
    simp only [inputCopyTM, hs]
    apply Cfg.ext
    case inputPos =>
      simp only [Action.apply, tapeAction, probeIcF, probeIcR, probeWriteCfg]
      apply Fin.ext
      rw [← SignType.neg_eq_neg_one, FinTM.moveInputPos_neg_val]
      simp
    case workTapes =>
      funext j
      by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeIcF, probeIcR, probeWriteCfg, hj]
    case workTapePos =>
      funext j
      by_cases hj : j = i <;>
        simp [Action.apply, tapeAction, probeIcF, probeIcR, probeWriteCfg, hj] <;> omega
    all_goals simp [Action.apply, tapeAction, probeIcF, probeIcR, probeWriteCfg]
  · -- return: both heads left
    intro r hr
    have hs : (probeIcR i d (r + 1)).inputSymbol = some x[r] :=
      inputSymbolInner r (by simp [probeIcR, probeWriteCfg]; omega) hr
    change ((inputCopyTM k i).tr .rewind _ _).apply _ = _
    simp only [inputCopyTM, hs]
    apply Cfg.ext
    case inputPos =>
      simp only [Action.apply, tapeAction, probeIcR, probeWriteCfg]
      apply Fin.ext
      rw [← SignType.neg_eq_neg_one, FinTM.moveInputPos_neg_val]
      simp
      omega
    case workTapes =>
      funext j
      by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeIcR, probeWriteCfg, hj]
    case workTapePos =>
      funext j
      by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeIcR, probeWriteCfg, hj] <;>
        omega
    all_goals simp [Action.apply, tapeAction, probeIcR, probeWriteCfg]
  · -- entry at the left end marker: both heads right, into `done`
    have hs : (probeIcR i d 0).inputSymbol = none := by
      simp [Cfg.inputSymbol, probeIcR, probeWriteCfg]
    change ((inputCopyTM k i).tr .rewind _ _).apply _ = _
    simp only [inputCopyTM, hs]
    apply Cfg.ext
    case inputPos =>
      simp only [Action.apply, tapeAction, probeIcR, probeIcD, probeWriteCfg, hin]
      rw [← SignType.pos_eq_one, moveInputPos_pos_of_ne_right _ (by simp)]
      apply Fin.ext
      simp
    case workTapes =>
      funext j
      by_cases hj : j = i <;>
        simp [Action.apply, tapeAction, probeIcR, probeIcD, probeIcF, probeWriteCfg, hj]
    case workTapePos =>
      funext j
      by_cases hj : j = i <;> simp [Action.apply, tapeAction, probeIcR, probeIcD, probeWriteCfg, hj]
    all_goals simp [Action.apply, tapeAction, probeIcR, probeIcD, probeIcF, probeWriteCfg]
  · apply Cfg.ext <;> simp [MultiTapeTM.step, inputCopyTM, Action.apply, inputMoveAction,
      probeIcD]

/-- `inputCopyTM_run_ofCfg` (statement copied from `Build/Scan.lean:215`). -/
theorem probe_inputCopy_contract (i : Fin k)
    (d : Cfg k Symbol SweepPhase x) (hstate : d.state = some .sweep) (hin : d.inputPos = 1) :
    (inputCopyTM k i).runFrom d (2 * x.length + 2) =
        { d with
          state := some .done
          workTapes := fun j q =>
            if j = i ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (x.length : ℤ)
              then FinTM.bufferTape x (q - d.workTapePos j)
            else d.workTapes j q } ∧
      (∀ t < 2 * x.length + 2, ((inputCopyTM k i).runFrom d t).state ≠ some .done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * x.length + 2 →
        if j = i then
          ((inputCopyTM k i).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (x.length : ℤ))
        else ((inputCopyTM k i).runFrom d t).workTapePos j = d.workTapePos j) := by
  refine ⟨?_, ?_, ?_⟩
  · rw [probe_ic_trace i d hstate hin]
    simp [catalogTrace, probeIcD, probeIcF, probeWriteCfg,
      show ¬2 * x.length + 2 ≤ x.length by omega]
  · intro t ht
    rw [probe_ic_trace i d hstate hin]
    simp only [catalogTrace]
    split_ifs <;> simp [probeIcF, probeIcR, probeIcD, probeWriteCfg] <;> omega
  · intro j t ht
    rw [probe_ic_trace i d hstate hin]
    simp only [catalogTrace]
    split_ifs <;> simp_all [probeIcF, probeIcR, probeIcD, probeWriteCfg, Finset.mem_Icc] <;>
      omega

end Write

The shared write step at the `M.step` level, a writer's forward step and turn, the copier's

/-- the shared write step, stated at `M.step` and proved through the step lemma -/
theorem probe_write_step2 {S : Type*} (M : MultiTapeTM k Symbol S) (i : Fin k)
    (w : List Symbol) (d : Cfg k Symbol S x) (q q' : S) (p : Fin (x.length + 2))
    (inp : SignType) (r : ℕ) (hr : r < w.length)
    (htr : M.tr q (probeWriteCfg i w d q p r).inputSymbol
        (probeWriteCfg i w d q p r).workTapeSymbols = tapeAction i inp (some (some w[r])) .pos q') :
    M.step (probeWriteCfg i w d q p r) = probeWriteCfg i w d q' (moveInputPos p inp) (r + 1) := by
  rw [step_of_tapeAction_write M _ i inp _ .pos rfl htr]
  apply Cfg.ext
  case workTapes =>
    funext j
    by_cases hj : j = i
    · subst j
      simpa [probeWriteCfg] using catalog_write_take (d.workTapes i) (d.workTapePos i) w r hr
    · simp [probeWriteCfg, hj]
  case workTapePos =>
    funext j
    by_cases hj : j = i <;> simp [probeWriteCfg, hj, Function.update_apply, SignType.cast] <;> omega
  all_goals simp [probeWriteCfg]

-- writeConst hF through the shared step
