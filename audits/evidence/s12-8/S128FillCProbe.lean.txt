/-! ### R-SEEK-LIFT: a lifted `Bool` module on dedicated digit tapes of a rich bank

Round-1 note 32 / round-2 note 31: isolate the dedicated digit tapes by R1; the rich record
tapes are R1's frame. The transport is two rewrites. -/

example {K : ℕ} {Γ : Type} [DecidableEq Γ] (e : Bool ↪ Γ) {S : Type}
    (R : MultiTapeTM 2 Bool S) (ι : Fin 2 ↪ Fin (K + 1))
    (tapes : Fin (K + 1) → ℤ → Option Γ) (heads : Fin (K + 1) → ℤ) (pre : List Γ)
    {x : List Bool} (c : Cfg 2 Bool S x) (t : ℕ) :
    (embedEmitTM ι (R.liftSymbols e)).runFrom (embedEmitCfg ι tapes heads pre (c.liftSymbols e)) t =
      embedEmitCfg ι tapes heads pre ((R.runFrom c t).liftSymbols e) := by
  rw [embedEmitTM_runFrom, MultiTapeTM.liftSymbols_runFrom]

/-- The seek's components, lifted along the record digits (the record step's `R8.5b` call). -/
example (Q T : Type) [DecidableEq Q] [DecidableEq T] : True := by
  obtain ⟨S, _, _, R, _entry, _exit, _C, _hR⟩ := exists_inputSeekTM
  let _R' : MultiTapeTM 2 (RecSym Q T) S := R.liftSymbols (RecSym.digitEmb Q T)
  trivial

/-- **Restriction to a dedicated digit bank** (the start of every lifted phase of the record
step). A rich host configuration whose selected tapes hold only digits and blanks *is* the R1
transport of the lift of its decoded `Bool` restriction, with the host's other tapes, heads and
output as the frame. Proved from the public API (`embedEmitCfg_selected_*`, `embedEmitTM_frame`
at time `0`, `decodeBool_apply`); a candidate shared lemma for `Build/Embed` or `SymbolLift`. -/
theorem probeRestrict {K m : ℕ} {Γ : Type} [DecidableEq Γ] (e : Bool ↪ Γ) {S H : Type}
    (ι : Fin m ↪ Fin K) {x : List Bool} (D : Cfg K Γ H (x.map e)) (s : S) (h₀ : H)
    (hD : D.state = some h₀)
    (hdig : ∀ i p γ, D.workTapes (ι i) p = some γ → ∃ b, γ = e b) :
    D.mapState (fun _ => s) =
      embedEmitCfg ι D.workTapes D.workTapePos D.output
        (Cfg.liftSymbols e (⟨some s, Fin.cast (by simp) D.inputPos,
          fun i p => (D.workTapes (ι i) p).bind (decodeBool e),
          fun i => D.workTapePos (ι i), []⟩ : Cfg m Bool S x)) := by
  set c : Cfg m Bool S x := ⟨some s, Fin.cast (by simp) D.inputPos,
    fun i p => (D.workTapes (ι i) p).bind (decodeBool e), fun i => D.workTapePos (ι i), []⟩
  have hframe := fun (j : Fin K) (hj : j ∉ Set.range ι) =>
    (embedEmitTM_frame ι
      (⟨s, fun _ _ _ => ⟨0, fun _ => (none, 0), none, none⟩⟩ : MultiTapeTM m Γ S)
      D.workTapes D.workTapePos D.output (c.liftSymbols e) 0).1 j hj
  apply Cfg.ext
  · simp [Cfg.mapState, hD, embedEmitCfg, Cfg.liftSymbols, c]
  · simp [Cfg.mapState, embedEmitCfg, Cfg.liftSymbols, c]
  · funext j
    by_cases hj : j ∈ Set.range ι
    · obtain ⟨i, rfl⟩ := hj
      rw [embedEmitCfg_selected_tape]
      funext p
      simp only [Cfg.mapState, Cfg.liftSymbols, c]
      cases hcell : D.workTapes (ι i) p with
      | none => rfl
      | some γ =>
        obtain ⟨b, rfl⟩ := hdig i p γ hcell
        simp
    · simpa [Cfg.mapState] using (hframe j hj).1.symm
  · funext j
    by_cases hj : j ∈ Set.range ι
    · obtain ⟨i, rfl⟩ := hj
      rw [embedEmitCfg_selected_pos]
      rfl
    · simpa [Cfg.mapState] using (hframe j hj).2.symm
  · simp [Cfg.mapState, embedEmitCfg, Cfg.liftSymbols, c]

/-- A lifted delimited word is the delimited word of the lift: the landed `FinTM.bufferTape_map`. -/
example {Γ : Type} (e : Bool ↪ Γ) (w : List Bool) (p : ℤ) :

/-! ### R-SEEK-BODY: the counter-loop round "input head right" (round-2 note 4)

One state, one step, cyclic: the round shape `counterLoopTM_run_done` needs, with `τ := 1`. -/

/-- Probe sketch of the seek's private round body. -/
def probeSeekBody : MultiTapeTM 1 Bool Unit where
  q₀ := ()
  tr := fun _ _ _ => inputMoveAction 1 ()

theorem probeSeekBody_orbit_state {x : List Bool} (c : Cfg 1 Bool Unit x)
    (hc : c.state = some ()) (r : ℕ) :
    (counterLoopOrbit probeSeekBody (fun _ => 1) c r).state = some () := by
  induction r with
  | zero => simpa [counterLoopOrbit] using hc
  | succ r ih =>
    rw [counterLoopOrbit, Function.iterate_succ_apply']
    change (probeSeekBody.step (counterLoopOrbit probeSeekBody (fun _ => 1) c r)).state = _
    simp only [MultiTapeTM.step, ih]
    rfl

theorem probeSeekBody_orbit_input {x : List Bool} (c : Cfg 1 Bool Unit x)
    (hc : c.state = some ()) (r : ℕ) (hr : c.inputPos.val + r ≤ x.length + 1) :
    (counterLoopOrbit probeSeekBody (fun _ => 1) c r).inputPos.val = c.inputPos.val + r := by
  induction r with
  | zero => simp [counterLoopOrbit]
  | succ r ih =>
    have ih' := ih (by omega)
    rw [counterLoopOrbit, Function.iterate_succ_apply']
    change (probeSeekBody.step (counterLoopOrbit probeSeekBody (fun _ => 1) c r)).inputPos.val = _
    have hs := probeSeekBody_orbit_state c hc r
    generalize counterLoopOrbit probeSeekBody (fun _ => 1) c r = o at ih' hs ⊢
    simp only [MultiTapeTM.step, hs]
    change (moveInputPos o.inputPos SignType.pos).val = _
    rw [moveInputPos_pos_of_ne_right _ (by omega)]
    simp only
    omega

/-- The round hypotheses of `counterLoopTM_run_done` for the seek body, at every round. -/
theorem probeSeekBody_hround {x : List Bool} (c : Cfg 1 Bool Unit x) (hc : c.state = some ())
    (d : ℕ) :
    ∀ r < d,
      (counterLoopOrbit probeSeekBody (fun _ => 1) c r).state = some () ∧
      0 < (fun _ => 1 : Cfg 1 Bool Unit x → ℕ) (counterLoopOrbit probeSeekBody (fun _ => 1) c r) ∧
      (∀ u, 0 < u → u < (fun _ => 1 : Cfg 1 Bool Unit x → ℕ)
          (counterLoopOrbit probeSeekBody (fun _ => 1) c r) →
        ∃ s, (probeSeekBody.runFrom (counterLoopOrbit probeSeekBody (fun _ => 1) c r) u).state =
          some s ∧ s ≠ () ∧ (none : Option Unit) ≠ some s) ∧
      (counterLoopOrbit probeSeekBody (fun _ => 1) c (r + 1)).state = some () := by
  intro r _
  refine ⟨probeSeekBody_orbit_state c hc r, Nat.one_pos, ?_, probeSeekBody_orbit_state c hc _⟩
  intro u hu hu'
  simp only at hu'
  omega

/-- The seek's countdown phase, from the rewound input (position `0`), lands at `p`: the
`counterLoopTM_run_done` call with the body above. -/
example {x : List Bool} (c : Cfg 1 Bool Unit x) (hc : c.state = some ()) (h0 : c.inputPos.val = 0)
    (w : List Bool) (ct : ℤ → Option Bool) (cp : ℤ)
    (hw : ∀ q : ℤ, -1 ≤ q → q ≤ (w.length : ℤ) → ct (cp + q) = FinTM.bufferTape w q)
    (p : ℕ) (hp : Nat.ofDigits 2 (w.map Bool.toNat) = p) (hpx : p ≤ x.length + 1) :
    ((counterLoopTM probeSeekBody () none).runFrom (counterLoopCfg (some (.dec .run)) c ct cp)
        (∑ _r ∈ Finset.range p, 1 + counterOverhead w (p + 1))).inputPos.val = p := by
  have h := (counterLoopTM_run_done probeSeekBody () none (fun _ => 1) c w ct cp hw p hp
    (probeSeekBody_hround c hc p) _ rfl).1
  rw [h]
  have := probeSeekBody_orbit_input c hc p (by omega)
  simpa [counterLoopCfg, embedEmitCfg, Cfg.mapState, h0] using this

/-! ### R-SEEK-REWIND: `timed_rewind` on the seek's private rewind table (round-1 minor 2)

`timed_rewind` stops at position `1` within `pos + 2`; one further left move reaches `0`. -/

/-- Probe sketch of the rewind phases. -/
inductive ProbeRw where
  | start | scan | left | done
deriving DecidableEq, Fintype

/-- Probe sketch of the seek's private rewind table, in `timed_rewind`'s required shape. -/
def probeRewindTM (k : ℕ) : MultiTapeTM k Bool ProbeRw where
  q₀ := .start
  tr := fun q inp _ =>
    match q with
    | .start => FinTM.controlAction .neg (some .scan)
    | .scan =>
      match inp with
      | some _ => FinTM.controlAction .neg (some .scan)
      | none => FinTM.controlAction .pos (some .left)
    | .left => FinTM.controlAction .neg (some .done)
    | .done => FinTM.controlAction 0 (some .done)

example {k : ℕ} {x : List Bool} (c : Cfg k Bool ProbeRw x) (hs : c.state = some .start) :
    ∃ r ≤ c.inputPos.val + 2,
      (probeRewindTM k).runFrom c r = {c with state := some .left, inputPos := 1} :=
  FinTM.timed_rewind (probeRewindTM k) .start .scan (some .left) (fun _ _ => rfl)
    (fun inp _ => by cases inp <;> rfl) c hs

/-- The landed `FinTM.rewind_heads` on the same table: the run and the fixed work heads. -/
example {k : ℕ} {x : List Bool} (c : Cfg k Bool ProbeRw x) (hs : c.state = some .start) :
    ∃ r ≤ c.inputPos.val + 2,
      (probeRewindTM k).runFrom c r = {c with state := some .left, inputPos := 1} ∧
      ∀ u ≤ r, ((probeRewindTM k).runFrom c u).workTapePos = c.workTapePos :=
  FinTM.rewind_heads (probeRewindTM k) .start .scan (some .left) (fun _ _ => rfl)
    (fun inp _ => by cases inp <;> rfl) c hs

/-- The extra left move after `timed_rewind`: from position `1` to `0`, into the stationary exit. -/
example {k : ℕ} {x : List Bool} (c : Cfg k Bool ProbeRw x) :
    ((probeRewindTM k).runFrom
        ({c with state := some ProbeRw.left, inputPos := 1} : Cfg k Bool ProbeRw x) 1).inputPos.val
          = 0 ∧
      ((probeRewindTM k).runFrom
        ({c with state := some ProbeRw.left, inputPos := 1} : Cfg k Bool ProbeRw x) 1).state =
          some ProbeRw.done := by
  refine ⟨?_, rfl⟩
  change (moveInputPos (1 : Fin (x.length + 2)) .neg).val = 0
  rw [FinTM.moveInputPos_neg_val]
  simp

/-! ### R-SEEK-ASSEMBLY: the four phases glue by R2 at two tapes

Copy `w` from tape `0` to tape `1`; rewind; count down on tape `1` (the counter loop's last
tape, `Fin.last 1`) moving the input right; clear tape `1`. The `1 + 1` tapes of
`counterLoopTM` unify with the `2` of the R3 rows, and the state type is finite. -/

/-- Probe sketch of the seek module's machine. -/
def probeSeekTM : MultiTapeTM 2 Bool
    (((SweepPhase ⊕ ProbeRw) ⊕ CounterLoopState Unit) ⊕ SweepPhase) :=
  seamCompTM
    (seamCompTM (seamCompTM (copyTM 2 0 1) SweepPhase.done (probeRewindTM 2) ProbeRw.start)
      (Sum.inr ProbeRw.done) (counterLoopTM probeSeekBody () none) (.dec .run))
    (Sum.inr CounterLoopState.done) (clearTM 2 1) SweepPhase.sweep

example : Fintype (((SweepPhase ⊕ ProbeRw) ⊕ CounterLoopState Unit) ⊕ SweepPhase) ×
    DecidableEq (((SweepPhase ⊕ ProbeRw) ⊕ CounterLoopState Unit) ⊕ SweepPhase) :=
  ⟨inferInstance, inferInstance⟩

/-! ### R-CALL: `exists_inputCallTM` = `inputCopyTM` + `exists_installCallTM` (round-1 minor 3,

theorem probeCopy_seam (k : ℕ) (hk : 0 < k) (x : List Bool) :
    (inputCopyTM k ⟨0, hk⟩).runFrom (Cfg.ofWords (input := x) SweepPhase.sweep (fun _ => []))

      Cfg.ofWords SweepPhase.done (stateWord k x) := by

  · rfl
  · rfl
  · funext j q

  · rfl

/-- The `entry ≠ exit` branch: the composite run of the input call, by
`seamCompTM_run_ofCfg` and `seamCompTM_firstReturn_ofCfg`. -/
example (M : FinTM Bool) (f : List Bool → List Bool) (T : ℕ → ℕ) (hM : M.ComputesFunInTime f T) :
    True := by
  obtain ⟨C, entry, exit, c, hk, hC⟩ := FinTM.exists_installCallTM M f T hM
  have key : ∀ x : List Bool, entry ≠ exit → ∃ t,
      (seamCompTM (inputCopyTM C.k ⟨0, hk⟩) SweepPhase.done C.tm entry).runFrom
          (Cfg.ofWords (input := x) (Sum.inl SweepPhase.sweep) (fun _ => []))
          (2 * x.length + 2 + 1 + t) =
        Cfg.ofWords (Sum.inr exit) (stateWord C.k (f x)) ∧
      ∀ u < 2 * x.length + 2 + 1 + t,
        ((seamCompTM (inputCopyTM C.k ⟨0, hk⟩) SweepPhase.done C.tm entry).runFrom
          (Cfg.ofWords (input := x) (Sum.inl SweepPhase.sweep) (fun _ => [])) u).state ≠
          some (Sum.inr exit) := by
    intro x hne
    obtain ⟨t, _ht, _hpos, hfirst, hrun⟩ := hC x x
    have hcopy := probeCopy_seam C.k hk x
    have hcut := (inputCopyTM_run_ofCfg (x := x) ⟨0, hk⟩
      (Cfg.ofWords SweepPhase.sweep (fun _ => [])) rfl rfl).2.1
    refine ⟨t, ?_, ?_⟩
    · exact seamCompTM_run_ofCfg (inputCopyTM C.k ⟨0, hk⟩) SweepPhase.done C.tm entry hcopy rfl
        hcut hrun
    · refine seamCompTM_firstReturn_ofCfg (inputCopyTM C.k ⟨0, hk⟩) SweepPhase.done C.tm entry
        exit hcopy rfl hcut hrun rfl ?_
      intro u hu
      rcases Nat.eq_zero_or_pos u with h0 | hpos
      · subst h0
        simpa [Cfg.ofWords, Cfg.mapState] using hne
      · exact hfirst u hpos hu
  trivial

/-- The `entry = exit` branch: the release adapter supplies the time-zero cut. -/
example (M : FinTM Bool) (f : List Bool → List Bool) (T : ℕ → ℕ) (hM : M.ComputesFunInTime f T) :
    True := by
  obtain ⟨C, entry, exit, c, hk, hC⟩ := FinTM.exists_installCallTM M f T hM
  have key : ∀ x : List Bool, entry = exit → ∃ t,
      (seamReleaseTM C.tm exit).runFrom
          ((Cfg.ofWords (input := x) entry (stateWord C.k x)).mapState fun _ => Sum.inl ())
          t = (Cfg.ofWords (input := x) exit (stateWord C.k (f x))).mapState Sum.inr ∧
      ∀ u < t, ((seamReleaseTM C.tm exit).runFrom
          ((Cfg.ofWords (input := x) entry (stateWord C.k x)).mapState fun _ => Sum.inl ())
          u).state ≠ some (Sum.inr exit) := by
    intro x he
    obtain ⟨t, _ht, hpos, hfirst, hrun⟩ := hC x x
    subst he
    exact ⟨t, seamReleaseTM_firstReturn C.tm entry rfl hpos hrun rfl hfirst⟩
  trivial

/-- A halted source steps to itself and emits nothing, so `hupd` keeps the tag. -/
example {k : ℕ} {Q : Type} (M : MultiTapeTM k Bool Q) {x : List Bool} (c : Cfg k Bool Q x)

  ⟨MultiTapeTM.step_of_halt h, MultiTapeTM.outputSymbol_of_halt h⟩

/-- Every input position has a code at the position width. -/
theorem probe_pos_lt (n : ℕ) : n + 1 < 2 ^ posWidth n := by
  unfold posWidth
  rw [Nat.size_eq_bits_len]
  exact Nat.lt_size_self _

/-- The one-digit width is exactly the empty input (round-2 note 3: the `|x| = 0` branch is
detected from `F1`'s width, with no native movement). -/
theorem probe_posWidth_eq_one (n : ℕ) : posWidth n = 1 ↔ n = 0 := by
  unfold posWidth
  rw [Nat.size_eq_bits_len]
  constructor
  · intro h
    by_contra hn
    have : 1 < Nat.size (n + 1) := Nat.lt_size.mpr (by omega)
    omega
  · rintro rfl
    simp [Nat.size_one]

/-- The position field's value is the position (the seek's `hp` premise): `binDigits` is the
little-endian code below `2 ^ w`. Induction on `w` with `Nat.testBit_add_one`. -/
theorem probe_ofDigits_binDigits (w v : ℕ) (hv : v < 2 ^ w) :
    Nat.ofDigits 2 ((binDigits w v).map Bool.toNat) = v := by
  induction w generalizing v with
  | zero => simp at hv; subst hv; simp [binDigits]
  | succ w ih =>
    have hrange : binDigits (w + 1) v = v.testBit 0 :: binDigits w (v / 2) := by
      simp only [binDigits, List.range_succ_eq_map, List.map_cons, List.map_map]
      congr 1
      apply List.map_congr_left
      intro i _
      simp [Nat.testBit_add_one]
    rw [hrange, List.map_cons, Nat.ofDigits_cons, ih (v / 2) (by rw [pow_succ] at hv; omega)]
    rw [Nat.testBit_zero]
    rcases Nat.mod_two_eq_zero_or_one v with h | h <;> simp [h] <;> omega

/-- `RecordAt` plus `fieldsTape_field` (stream A) frames field `j` of the record at its offset,
given that the field lies inside the layout (a private span lemma of the fill). -/
example {K : ℕ} {Γ S : Type} {x : List Γ} (d : Cfg K Γ S x) (r : Fin K) (fs : List (List Γ))
    (hrec : RecordAt d r fs) (j : ℕ) (hj : j < fs.length)
    (hspan : fieldOffset fs j + fs[j].length ≤ (joinFields fs).length) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (fs[j].length : ℤ) →
      d.workTapes r ((d.workTapePos r + fieldOffset fs j) + p) = FinTM.bufferTape fs[j] p := by
  intro p hp1 hp2
  rw [← fieldsTape_field fs j hj p hp1 hp2, add_assoc]
  exact hrec _ (by omega) (by omega)

/-- The record alphabet's phase types are finite where the record rows need them. -/
example (Q T : Type) [Fintype Q] [DecidableEq Q] [Fintype T] [DecidableEq T] :
    Fintype (ScanPhase (Option (Option Q))) × Fintype (ScanPhase (Option T)) ×
      Fintype (CounterLoopState Unit) × DecidableEq (RecSym Q T) × Fintype (RecSym Q T) :=
  ⟨inferInstance, inferInstance, inferInstance, inferInstance, inferInstance⟩

/-- Record equality, field `0`: the framed compare applies at the two heads (the row's first
phase; `fieldOffset fs 0 = 0`). Cites the sorried `compareTM_run_ofCfg` and
`fieldsTape_field` (stream A). -/
example {Γ : Type} [DecidableEq Γ] {x : List Γ} (fs gs : List (List Γ)) (hf : 0 < fs.length)
    (hg : 0 < gs.length) (d : Cfg 2 Γ FlagPhase x) (hstate : d.state = some FlagPhase.run)
    (hrf : RecordAt d 0 fs) (hrg : RecordAt d 1 gs)
    (hsf : fs[0].length ≤ (joinFields fs).length) (hsg : gs[0].length ≤ (joinFields gs).length) :
    ∃ δ ≤ min fs[0].length gs[0].length,
      (compareTM 2 0 1).runFrom d (2 * δ + 2) =
        { d with state := some (FlagPhase.done (decide (fs[0] = gs[0]))) } := by
  have hfr : ∀ p : ℤ, -1 ≤ p → p ≤ (fs[0].length : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape fs[0] p := by
    intro p h1 h2
    have := fieldsTape_field fs 0 hf p h1 h2
    simp only [fieldOffset, List.take_zero, List.map_nil, List.sum_nil, Nat.cast_zero,
      zero_add] at this
    rw [← this]
    exact hrf p h1 (by omega)
  have hgr : ∀ p : ℤ, -1 ≤ p → p ≤ (gs[0].length : ℤ) →
      d.workTapes 1 (d.workTapePos 1 + p) = FinTM.bufferTape gs[0] p := by
    intro p h1 h2
    have := fieldsTape_field gs 0 hg p h1 h2
    simp only [fieldOffset, List.take_zero, List.map_nil, List.sum_nil, Nat.cast_zero,
      zero_add] at this
    rw [← this]
    exact hrg p h1 (by omega)
  obtain ⟨δ, hδ, hrun, -, -⟩ := compareTM_run_ofCfg 0 1 (by decide) fs[0] gs[0] d hstate hfr hgr
  exact ⟨δ, hδ, hrun⟩

/-! ### R-SEEK-LAYOUT: the counter-loop host layout from the public R1 API

`counterLoopCfg` is `embedEmitCfg Fin.castSuccEmb` with the counter as the frame. A consumer that
starts the loop from an arbitrary two-tape configuration must identify it with a
`counterLoopCfg`. CounterLoop's `counter_cfg_eq` does this but is `private`; the same identity
follows from `embedEmitTM_frame` at time `0` and `embedEmitCfg_selected_*`, as below. **The fill
should request `counter_cfg_eq`'s promotion rather than keep a second proof** (duplication
hazard, see the dossier). -/

theorem probeCounterLayout {S : Type} {x : List Bool} (q : Option (CounterLoopState S))
    (c : Cfg 1 Bool S x) (ct : ℤ → Option Bool) (cp : ℤ) :
    (counterLoopCfg q c ct cp).workTapes (Fin.castSucc 0) = c.workTapes 0 ∧
      (counterLoopCfg q c ct cp).workTapes (Fin.last 1) = ct ∧
      (counterLoopCfg q c ct cp).workTapePos (Fin.castSucc 0) = c.workTapePos 0 ∧
      (counterLoopCfg q c ct cp).workTapePos (Fin.last 1) = cp ∧
      (counterLoopCfg q c ct cp).inputPos = c.inputPos ∧
      (counterLoopCfg q c ct cp).output = c.output ∧ (counterLoopCfg q c ct cp).state = q := by
  rw [counter_cfg_eq]
  refine ⟨?_, ?_, ?_, ?_, rfl, rfl, rfl⟩ <;> simp only [Fin.lastCases_castSucc, Fin.lastCases_last]

/-! ### R-STEP-SAVE: saving the native input position (round-2 note 3)

The record step must restore an arbitrary host input position `pos`. A counter-loop round "if
the input is blank, exit; else move left" (zero body tapes; the host's only tape is the fuel
counter) escapes at round `q` from position `q`, leaving the counter at `value - (q + 1)`;
complementing gives `q + 1`. The two blank boundaries are told apart beforehand by a one-step
right probe (`|x| ≥ 1`); `|x| = 0` is detected from `F1`'s width and needs no movement. -/

/-- Probe sketch of the save loop's round body: anchor `false`, exit `true`. -/
def probeSaveBody : MultiTapeTM 0 Bool Bool where
  q₀ := false
  tr := fun q inp _ =>
    match q, inp with
    | false, some _ => inputMoveAction (-1) false
    | false, none => inputMoveAction 0 true
    | true, _ => inputMoveAction 0 true

theorem probeSaveBody_orbit {x : List Bool} (c : Cfg 0 Bool Bool x) (hc : c.state = some false)
    (hq : c.inputPos.val ≤ x.length) (r : ℕ) (hr : r ≤ c.inputPos.val) :
    (counterLoopOrbit probeSaveBody (fun _ => 1) c r).state = some false ∧
      (counterLoopOrbit probeSaveBody (fun _ => 1) c r).inputPos.val = c.inputPos.val - r := by
  induction r with
  | zero => exact ⟨by simpa [counterLoopOrbit] using hc, by simp [counterLoopOrbit]⟩
  | succ r ih =>
    obtain ⟨hs, hp⟩ := ih (by omega)
    rw [counterLoopOrbit, Function.iterate_succ_apply']
    change (probeSaveBody.step (counterLoopOrbit probeSaveBody (fun _ => 1) c r)).state = _ ∧
      (probeSaveBody.step (counterLoopOrbit probeSaveBody (fun _ => 1) c r)).inputPos.val = _
    generalize counterLoopOrbit probeSaveBody (fun _ => 1) c r = o at hs hp ⊢
    have hsym : o.inputSymbol = some x[o.inputPos.val - 1] :=
      inputSymbolInner (o.inputPos.val - 1) (by omega) (by omega)
    simp only [MultiTapeTM.step, hs]
    rw [hsym]
    refine ⟨rfl, ?_⟩
    change (moveInputPos o.inputPos SignType.neg).val = _
    rw [FinTM.moveInputPos_neg_val]
    omega

/-- At round `q` the head is on the left end marker, and the round exits. -/
theorem probeSaveBody_exit {x : List Bool} (c : Cfg 0 Bool Bool x) (hc : c.state = some false)
    (hq : c.inputPos.val ≤ x.length) :
    (counterLoopOrbit probeSaveBody (fun _ => 1) c (c.inputPos.val + 1)).state = some true ∧
      (counterLoopOrbit probeSaveBody (fun _ => 1) c (c.inputPos.val + 1)).inputPos.val = 0 := by
  obtain ⟨hs, hp⟩ := probeSaveBody_orbit c hc hq c.inputPos.val le_rfl
  rw [counterLoopOrbit, Function.iterate_succ_apply']
  change (probeSaveBody.step (counterLoopOrbit probeSaveBody (fun _ => 1) c c.inputPos.val)).state
      = _ ∧
    (probeSaveBody.step (counterLoopOrbit probeSaveBody (fun _ => 1) c c.inputPos.val)).inputPos.val
      = _
  generalize counterLoopOrbit probeSaveBody (fun _ => 1) c c.inputPos.val = o at hs hp ⊢
  have h0 : o.inputPos = 0 := Fin.ext (by simp only [Fin.val_zero]; omega)
  have hsym : o.inputSymbol = none := by
    unfold Cfg.inputSymbol
    rw [dif_pos h0]
  simp only [MultiTapeTM.step, hs]
  rw [hsym]
  refine ⟨rfl, ?_⟩
  change (moveInputPos o.inputPos 0).val = 0
  rw [moveInputPos_zero, h0]
  rfl

/-- The save loop: `counterLoopTM_run_escape` with the body above escapes at round `q = pos`,
the head on `0`, the fuel counter at `counterWord w (q + 1)`. -/
example {x : List Bool} (c : Cfg 0 Bool Bool x) (hc : c.state = some false)
    (hq : c.inputPos.val ≤ x.length) (w : List Bool) (ct : ℤ → Option Bool) (cp : ℤ)
    (hw : ∀ q : ℤ, -1 ≤ q → q ≤ (w.length : ℤ) → ct (cp + q) = FinTM.bufferTape w q)
    (hfuel : c.inputPos.val < Nat.ofDigits 2 (w.map Bool.toNat)) :
    (counterLoopTM probeSaveBody false (some true)).runFrom
        (counterLoopCfg (some (.dec .run)) c ct cp)
        (∑ _r ∈ Finset.range (c.inputPos.val + 1), 1 + counterOverhead w (c.inputPos.val + 1)) =
      counterLoopCfg (some .escape)
        (counterLoopOrbit probeSaveBody (fun _ => 1) c (c.inputPos.val + 1))
        (fun q => if cp ≤ q ∧ q < cp + (w.length : ℤ)
          then FinTM.bufferTape (counterWord w (c.inputPos.val + 1)) (q - cp) else ct q) cp := by
  refine (counterLoopTM_run_escape probeSaveBody false true (by decide) (fun _ => 1) c w ct cp hw
    c.inputPos.val hfuel ?_ (probeSaveBody_exit c hc hq).1 _ rfl).1
  intro r hr
  refine ⟨(probeSaveBody_orbit c hc hq r hr).1, Nat.one_pos, ?_⟩
  intro u hu hu'
  simp only at hu'
  omega

/-- Complementing a counter word: values add to `2 ^ |u| - 1`, so the complement of
`counterWord w (q + 1)` (value `2 ^ |w| - 1 - (q + 1)` from an all-`true` fuel `w`) has value
`q + 1`, the restore seek's target. -/
theorem probe_ofDigits_not (u : List Bool) :
    Nat.ofDigits 2 ((u.map not).map Bool.toNat) + Nat.ofDigits 2 (u.map Bool.toNat) =
      2 ^ u.length - 1 := by
  induction u with
  | nil => simp
  | cons b u ih =>
    simp only [List.map_cons, Nat.ofDigits_cons, List.length_cons, pow_succ]
    have h2 : 1 ≤ 2 ^ u.length := Nat.one_le_two_pow
    cases b <;>
      simp only [Bool.not_true, Bool.not_false, Bool.toNat_true, Bool.toNat_false] <;> omega
