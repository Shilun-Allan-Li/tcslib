open Lean Elab Command in
/-- Every constant in the dependency closure of `root` whose own type or value
mentions `sorryAx` (maintainer's read-only admission walk). -/
def sorrySourcesOf (env : Environment) (root : Name) : Array Name := Id.run do
  let mut visited : NameSet := {}
  let mut stack : Array Name := #[root]
  let mut srcs : Array Name := #[]
  while !stack.isEmpty do
    let c := stack.back!
    stack := stack.pop
    if visited.contains c then continue
    visited := visited.insert c
    if let some ci := env.find? c then
      let used := ci.getUsedConstantsAsSet
      if used.contains ``sorryAx then srcs := srcs.push c
      for d in used.toList do
        unless visited.contains d do stack := stack.push d
  return srcs

open Lean Elab Command in
elab "#sorry_sources " id:ident : command => do
  let n ← liftCoreM <| realizeGlobalConstNoOverload id
  logInfo m!"{n}: direct sorryAx users in closure: {sorrySourcesOf (← getEnv) n}"
