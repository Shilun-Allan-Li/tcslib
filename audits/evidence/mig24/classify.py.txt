"""MIG-24 triage, read-only: for each candidate pair, extract both declarations and compare tokens.
rename-mirror: equal up to a consistent renaming of identifiers; variant: otherwise (equal-token %)."""
import re, sys, os, difflib, csv
root, cand, pop, out = sys.argv[1:5]
src = open(os.path.join(os.path.dirname(os.path.abspath(__file__)), 'declfreeze_chunks.py')).read()
ns = {}; exec(src, ns); chunks = ns['chunks']
paths = [l.strip() for l in open(pop) if l.strip()]
def resolve(f):
    if os.path.exists(os.path.join(root, f)): return f
    hits = [p for p in paths if p.endswith('/' + f)]
    return hits[0] if len(hits) == 1 else None
cache = {}
def decl(f, n):
    p = resolve(f)
    if p is None: return None, 'unresolved file'
    if p not in cache:
        cache[p] = {name: t for name, _, t in chunks(open(os.path.join(root, p), encoding='utf-8').read())}
    t = cache[p].get(n)
    return (p, t) if t else (p, None)
TOK = re.compile(r"[A-Za-z_][A-Za-z0-9_'.₀-₉ᵢ]*|\d+|\S")
def toks(t):
    body = t.split('-/', 1)[1] if t.lstrip().startswith('/--') else t
    body = re.sub(r'--[^\n]*', '', body)
    return TOK.findall(body)
rows = []
for r in csv.DictReader(open(cand, encoding='utf-8'), delimiter='\t'):
    pa, ta = decl(r['file_a'], r['decl_a']); pb, tb = decl(r['file_b'], r['decl_b'])
    if not ta or not tb:
        rows.append({**r, 'class': 'unresolved', 'equal_pct': '', 'renames': '', 'path_a': pa or '', 'path_b': pb or ''}); continue
    A, B = toks(ta), toks(tb)
    sm = difflib.SequenceMatcher(a=A, b=B, autojunk=False)
    eq = sum(bl.size for bl in sm.get_matching_blocks())
    pct = round(100 * eq / max(len(A), len(B)))
    mapping, ok = {}, True
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == 'equal': continue
        if tag != 'replace' or i2 - i1 != j2 - j1: ok = False; break
        for x, y in zip(A[i1:i2], B[j1:j2]):
            if not (re.match(r'[A-Za-z_]', x) and re.match(r'[A-Za-z_]', y)): ok = False; break
            if mapping.setdefault(x, y) != y: ok = False; break
        if not ok: break
    cls = 'rename-mirror' if ok else 'variant'
    rows.append({**r, 'class': cls, 'equal_pct': pct, 'renames': ' '.join(f'{a}->{b}' for a, b in list(mapping.items())[:8]) if ok else '',
                 'path_a': pa, 'path_b': pb})
keys = list(rows[0].keys())
with open(out, 'w', encoding='utf-8') as f:
    w = csv.DictWriter(f, fieldnames=keys, delimiter='\t'); w.writeheader(); w.writerows(rows)
from collections import Counter
print(Counter(r['class'] for r in rows))
