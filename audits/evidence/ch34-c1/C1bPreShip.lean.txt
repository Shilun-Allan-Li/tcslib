section Shared
open Turing Complexity

example {s₁ s₂ : ℕ → ℕ} {a : ℕ} (h : ∀ n, s₁ n ≤ a * s₂ n) : SPACE s₁ ⊆ SPACE s₂ :=
  SPACE.mono_const h

example {M : FinTM Bool} {f : List Bool → List Bool} (hk : M.k = 0)
    (h : ∀ x, ∃ t, M.ComputesInTime x (f x) t) : M.ComputesInSpace f fun _ => 0 :=
  FinTM.computesInSpace_zero_of_k_eq_zero hk h

-- Z3 `⊆` shape. Z2 is applied by dot notation through `DecidesInSpace`
-- (brief pitfall). The bridge is called with an explicit `M`, and `.mono`
-- fixes `f`.
example {L : Language Bool} {s : ℕ → ℕ} {c : ℕ} {M : FinTM Bool}
    (hM : M.DecidesInSpace L fun n => c * s n) (hz : ∃ n, s n = 0) :
    L ∈ SPACE fun _ => 0 := by
  have hk : M.k = 0 := hM.k_eq_zero_of_exists_zero (by
    obtain ⟨n, hn⟩ := hz
    exact ⟨n, by simp [hn]⟩)
  refine ⟨0, M, ?_⟩
  refine (FinTM.computesInSpace_zero_of_k_eq_zero (M := M) hk ?_).mono ?_
  · intro x
    exact (hM x).imp fun _ h => h.1
  · intro n
    exact Nat.zero_le _

-- Z3 `⊇` shape.
example (s : ℕ → ℕ) : SPACE (fun _ => 0) ⊆ SPACE s :=
  SPACE.mono fun _ => Nat.zero_le _

-- Z5 `⊆` and `⊇` shapes (`mono_const` at `a = 2`).
example (s : ℕ → ℕ) : SPACE s ⊆ SPACE (fun n => s n + 1) :=
  SPACE.mono fun n => Nat.le_succ (s n)

example {s : ℕ → ℕ} (hpos : ∀ n, 0 < s n) : SPACE (fun n => s n + 1) ⊆ SPACE s :=
  SPACE.mono_const (a := 2) fun n => show s n + 1 ≤ 2 * s n by
    have := hpos n
    omega

-- Z6 shapes (`omega` with `max`).
example (s : ℕ → ℕ) : SPACE (fun n => s n + 1) ⊆ SPACE (fun n => max 1 (s n)) :=
  SPACE.mono_const (a := 2) fun n => show s n + 1 ≤ 2 * max 1 (s n) by omega

example (s : ℕ → ℕ) : SPACE (fun n => max 1 (s n)) ⊆ SPACE (fun n => s n + 1) :=
  SPACE.mono fun n => show max 1 (s n) ≤ s n + 1 by omega

end Shared

/-! ### 5a. The Z8/Z10 constant machine: emit `true` and halt, in one action -/

/-- Probe stand-in for C1b's private constant machine. -/
def constTrueTM : FinTM Bool where
  k := 0
  State := Unit
  tm := { q₀ := (), tr := fun _ _ _ => ⟨0, fun i => i.elim0, some true, none⟩ }

/-- The one-step contract (`t = 0` guard: halted at time `1` with `[true]`). -/
theorem constTrueTM_computesInTime (x : List Bool) : constTrueTM.ComputesInTime x [true] 1 :=
  (FinTM.computesInTime_iff _ _ _ _).mpr ⟨rfl, rfl⟩

-- Z10 shape.
example : ∃ M : FinTM Bool, M.k = 0 ∧ ∀ x : List Bool, M.ComputesInTime x [true] 1 :=
  ⟨constTrueTM, rfl, constTrueTM_computesInTime⟩

-- Z8 shape: the bridge, plus `indicator {x | True} x = true`.
example : ({x | True} : Language Bool) ∈ SPACE fun _ => 0 := by
  refine ⟨0, constTrueTM, ?_⟩
  refine (FinTM.computesInSpace_zero_of_k_eq_zero (M := constTrueTM) rfl ?_).mono ?_
  · intro x
    refine ⟨1, ?_⟩
    simpa [MultiTapeTM.indicator] using constTrueTM_computesInTime x
  · intro n
    exact Nat.zero_le _

/-! ### 5b. The `Examples` export shape (`evenScanTM`, `evenScanTM_decidesInTime`) -/

/-- Probe stand-in for `Complexity.evenScanTM`. The state `true` means odd
parity; scan transitions go through `controlAction`. -/
def scanTM : FinTM Bool where
  k := 0
  State := Bool
  tm :=
    { q₀ := false
      tr := fun q inp _ => match inp with
        | some b => FinTM.controlAction .pos (some (if b then !q else q))
        | none => ⟨0, fun i => i.elim0, some (!q), none⟩ }

/-- Probe stand-in for `Complexity.evenScanTM_decidesInTime`. It is the
fill's to prove; the stand-in is sorried.

**Proof sketch.** The brief's route 1 (scan invariant on `j ≤ x.length`). -/
theorem scanTM_decidesInTime : scanTM.DecidesInTime evenLang fun n => n + 1 := by
  sorry

-- Z11 shape.
example : ∃ M : FinTM Bool, M.k = 0 ∧ M.DecidesInTime evenLang fun n => n + 1 :=
  ⟨scanTM, rfl, scanTM_decidesInTime⟩

-- Z9 shape: the export plus the bridge. `0 * (fun _ => 0) n` needs only `.mono`.
example : evenLang ∈ SPACE fun _ => 0 := by
  refine ⟨0, scanTM, ?_⟩
  refine (FinTM.computesInSpace_zero_of_k_eq_zero (M := scanTM) rfl ?_).mono ?_
  · intro x
    exact ⟨x.length + 1, scanTM_decidesInTime x⟩
  · intro n
    exact Nat.zero_le _

-- E1 shape: the same, at `logSpace`, inside `Examples`' import surface.
example : evenLang ∈ LOGSPACE := by
  unfold LOGSPACE
  refine ⟨0, scanTM, ?_⟩
  refine (FinTM.computesInSpace_zero_of_k_eq_zero (M := scanTM) rfl ?_).mono ?_
  · intro x
    exact ⟨x.length + 1, scanTM_decidesInTime x⟩
  · intro n
    exact Nat.zero_le _

end C1bProbe

/-! ## 6. C1: the S9 statement, and the headroom-1 route's step shapes -/

section C1
open Complexity.CounterProg

variable {R : ℕ} {Λ : Type} (P : Λ → Instr R Λ) (x : List Bool)
  [Fintype Λ] [DecidableEq Λ] (l₀ : Λ)

-- The frozen statement, restated.
example (B t : ℕ) (s : St R Λ) (hp : s.pos ≤ x.length)
    (hB : ∀ j < t, ∀ r, (run P x s j).regs r ≤ B) :
    ∃ t' ≤ t * (2 * B + 5), (toTM P l₀).tm.runFrom (enc x s) t' = enc x (run P x s t) :=
  sim_run_of_regs_le P x l₀ B t s hp hB

-- One step via `sim_run` at headroom 1: the bound is `1 * (2 * (B + 1) + 3)`.
example (B : ℕ) (s : St R Λ) (hp : s.pos ≤ x.length)
    (h0 : ∀ r, (run P x s 0).regs r ≤ B) :
    ∃ t' ≤ 1 * (2 * (B + 1) + 3),
      (toTM P l₀).tm.runFrom (enc x s) t' = enc x (run P x s 1) :=
  sim_run P x l₀ (B + 1) 1 s hp fun r => by
    have h := h0 r
    rw [run_zero] at h
    omega

-- `run P x s 1 = step P x s`, as the pitfall prescribes.
example (s : St R Λ) : run P x s 1 = step P x s :=
  (run_succ P x s 0).trans (run_zero P x _)

-- The suffix keeps the position bound.
example (s : St R Λ) (hp : s.pos ≤ x.length) : (run P x s 1).pos ≤ x.length := by
  have h1 : run P x s 1 = step P x s := (run_succ P x s 0).trans (run_zero P x _)
  rw [h1]
  exact step_pos_le P x s hp

-- The register hypothesis shifts via `run_add`.
example (B t : ℕ) (s : St R Λ) (hB : ∀ j < t + 1, ∀ r, (run P x s j).regs r ≤ B) :
    ∀ j < t, ∀ r, (run P x (run P x s 1) j).regs r ≤ B := by
  intro j hj r
  have h := hB (1 + j) (by omega) r
  rwa [run_add] at h

-- Composition of the physical runs.
example (t t₁ t₂ : ℕ) (s : St R Λ)
    (h₁ : (toTM P l₀).tm.runFrom (enc x s) t₁ = enc x (run P x s 1))
    (h₂ : (toTM P l₀).tm.runFrom (enc x (run P x s 1)) t₂ =
      enc x (run P x (run P x s 1) t)) :
    (toTM P l₀).tm.runFrom (enc x s) (t₁ + t₂) = enc x (run P x s (t + 1)) := by
  rw [Turing.MultiTapeTM.runFrom_add, h₁, h₂, ← run_add, Nat.add_comm 1 t]

-- The time arithmetic: exactly the stated constant.
example (B t t₁ t₂ : ℕ) (h₁ : t₁ ≤ 1 * (2 * (B + 1) + 3)) (h₂ : t₂ ≤ t * (2 * B + 5)) :
    t₁ + t₂ ≤ (t + 1) * (2 * B + 5) := by
  nlinarith

end C1
