/-! Review probe (#116 gate, finding 1): the U2-S route's cleaned canonizer without reading
`cleanTM`'s initial state. The two `probe_*` lemmas stand in for the proposed public
`Turing.MultiTapeTM.runFrom_with_q₀` / `initCfg_with_q₀` (same statements, same `rfl` proofs;
they mention no definition of Hydroxyi's). The example then uses only `cleanTM_run`,

open Turing

theorem probe_runFrom_with_q₀ {k : ℕ} {Symbol State : Type*} {input : List Symbol}
    (tm : MultiTapeTM k Symbol State) (q : State) (cfg : Cfg k Symbol State input) (t : ℕ) :
    ({ tm with q₀ := q } : MultiTapeTM k Symbol State).runFrom cfg t = tm.runFrom cfg t :=
  rfl

theorem probe_initCfg_with_q₀ {k : ℕ} {Symbol State : Type*}
    (tm : MultiTapeTM k Symbol State) (q : State) (x : List Symbol) :
    ({ tm with q₀ := q } : MultiTapeTM k Symbol State).initCfg x = Cfg.init q x :=
  rfl

/-- The cleaned machine, restarted at `init q`, computes `M`'s output on `V` (with blank work
tapes), as a `FinTM` would state it from `initCfg`. -/

    (hs : ∀ i, (M.visitedByTapeHead (Cfg.init q V) T i).card ≤ s) :
    ∃ T', (({ Complexity.LogProg.cleanTM M q with q₀ := .init q } :
          MultiTapeTM (k + k) Bool (Complexity.LogProg.CleanSt S k)).runFrom
          (({ Complexity.LogProg.cleanTM M q with q₀ := .init q } :
            MultiTapeTM (k + k) Bool (Complexity.LogProg.CleanSt S k)).initCfg V) T').state =
        none ∧
      (({ Complexity.LogProg.cleanTM M q with q₀ := .init q } :
          MultiTapeTM (k + k) Bool (Complexity.LogProg.CleanSt S k)).runFrom
          (({ Complexity.LogProg.cleanTM M q with q₀ := .init q } :
            MultiTapeTM (k + k) Bool (Complexity.LogProg.CleanSt S k)).initCfg V) T').output =
        (M.runFrom (Cfg.init q V) T).output := by
  obtain ⟨T', h1, h2, -⟩ := Complexity.LogProg.cleanTM_run M q q V T hT s hs
  refine ⟨T', ?_, ?_⟩

  · rw [probe_initCfg_with_q₀, probe_runFrom_with_q₀]; exact h2
