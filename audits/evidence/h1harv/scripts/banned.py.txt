"""Banned-construct screen (standard library only).

Usage: python3 -I banned.py <file.lean> ...

Reports, per file, occurrences in code (comments removed, nesting-aware) of the tokens
`#eval`, `run_cmd`, `IO`, `unsafe`, `implemented_by`, `extern`, `native_decide`, and of
`syntax`, `macro`, `macro_rules`, `elab`, `elab_rules`, `notation` used as commands (at the
start of a line, after optional attributes and `local`/`scoped`). Exit 1 if any is found.
"""
import re
import sys

TOK = re.compile(r"(#eval\b|\brun_cmd\b|\bIO\b|\bunsafe\b|\bimplemented_by\b|\bextern\b|"
                 r"\bnative_decide\b)")
CMD = re.compile(r"^\s*(?:@\[[^\]]*\]\s*)*(?:(?:local|scoped)\s+)?(syntax|macro|macro_rules|"
                 r"elab|elab_rules|notation)\b")


def strip(src):
    out, i, depth = [], 0, 0
    while i < len(src):
        if src.startswith("/-", i):
            depth += 1; i += 2; continue
        if src.startswith("-/", i) and depth > 0:
            depth -= 1; i += 2; continue
        if depth == 0:
            out.append(src[i])
        elif src[i] == "\n":
            out.append("\n")
        i += 1
    return "\n".join(l.split("--")[0] for l in "".join(out).split("\n"))


bad = 0
for p in sys.argv[1:]:
    code = strip(open(p, encoding="utf-8").read()).split("\n")
    hits = [(i + 1, m.group(0)) for i, l in enumerate(code) for m in TOK.finditer(l)]
    hits += [(i + 1, m.group(1)) for i, l in enumerate(code) if (m := CMD.match(l))]
    print(f"{p}: {len(hits)}" + ("".join(f"\n   line {i}: {t}" for i, t in hits)))
    bad += len(hits)
print("banned constructs in code:", bad)
sys.exit(1 if bad else 0)
