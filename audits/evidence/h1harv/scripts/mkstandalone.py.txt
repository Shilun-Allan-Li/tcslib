"""Build a standalone file from the unit's Part A (standard library only).

Usage: python3 -I mkstandalone.py <unit.lean> <out> [--extra-import M ...] [--tail <file>]

<out> is: the unit's copyright block; the imports its `-- @imports` lines list, then each
extra import; the three standard options; the Part A body verbatim (everything after its last
`-- @imports` line); and the tail, if given. With no extras and no tail this is the module
exactly as it would land. Every textual operation is asserted.
"""
import sys

COPYRIGHT_END = "-/\n"
OPTIONS = ("set_option maxHeartbeats 0\nset_option relaxedAutoImplicit false\n"
           "set_option autoImplicit false\n")


def main():
    unit_path, out = sys.argv[1], sys.argv[2]
    args = sys.argv[3:]
    extra, tail = [], None
    while args:
        if args[0] == "--extra-import":
            extra.append(args[1]); args = args[2:]
        elif args[0] == "--tail":
            tail = open(args[1], encoding="utf-8").read(); args = args[2:]
        else:
            raise SystemExit(f"unknown argument {args[0]}")
    unit = open(unit_path, encoding="utf-8").read()
    assert unit.startswith("/-\nCopyright"), "copyright block first"
    copyright = unit[:unit.index(COPYRIGHT_END) + len(COPYRIGHT_END)]
    lines = unit.split("\n")
    assert sum(1 for l in lines if l.startswith("-- @landing ")) == 1
    imps = [w for l in lines if l.startswith("-- @imports ") for w in l.split()[2:]]
    last = max(i for i, l in enumerate(lines) if l.startswith("-- @imports "))
    body = "\n".join(lines[last + 1:]).strip("\n") + "\n"
    head = copyright + "".join(f"import {m}\n" for m in imps + extra) + "\n" + OPTIONS + "\n"
    text = head + body + ("\n" + tail if tail is not None else "")
    assert body in text
    open(out, "w", encoding="utf-8").write(text)
    print(f"{out}: imports {imps + extra}; {text.count(chr(10)) + 1} lines")


if __name__ == "__main__":
    main()
