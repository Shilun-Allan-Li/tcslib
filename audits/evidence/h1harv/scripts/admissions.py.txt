"""Admission census (standard library only).

Usage: python3 -I admissions.py <file.lean>=<expected sorries> ...

Counts, in code (comments removed, nesting-aware), the `sorry` tokens and the `axiom`
declarations of each file and compares the `sorry` count with the expectation; any `axiom`
declaration fails. Exit 1 on any mismatch.
"""
import re
import sys


def strip(src):
    out, i, depth = [], 0, 0
    while i < len(src):
        if src.startswith("/-", i):
            depth += 1; i += 2; continue
        if src.startswith("-/", i) and depth > 0:
            depth -= 1; i += 2; continue
        if depth == 0:
            out.append(src[i])
        elif src[i] == "\n":
            out.append("\n")
        i += 1
    return "\n".join(l.split("--")[0] for l in "".join(out).split("\n"))


bad = 0
for arg in sys.argv[1:]:
    path, want = arg.rsplit("=", 1)
    code = strip(open(path, encoding="utf-8").read())
    s = len(re.findall(r"(?<![\w.'])sorry(?![\w'])", code))
    a = len(re.findall(r"^\s*(?:@\[[^\]]*\]\s*)*(?:(?:private|protected|noncomputable)\s+)*axiom\b",
                       code, re.M))
    ok = s == int(want) and a == 0
    bad += not ok
    print(f"{path}: sorry {s} (expected {want}), axiom declarations {a}: {'ok' if ok else 'FAIL'}")
sys.exit(1 if bad else 0)
