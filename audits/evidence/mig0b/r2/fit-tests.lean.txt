/-- *Fit copy* of `Build/CfgRecord.lean:365` (`RecordRun`, private there), verbatim. -/
def RecordRun {K : ℕ} {Γ S : Type} {x : List Γ} (R : MultiTapeTM K Γ S)
    (yes no : S) (base : Fin K → ℤ) (radius : ℕ) (a b : Cfg K Γ S x) (t : ℕ) : Prop :=
  R.runFrom a t = b ∧
    (∀ u < t, (R.runFrom a u).state ≠ some yes ∧ (R.runFrom a u).state ≠ some no) ∧
    ∀ u ≤ t, ∀ j, |(R.runFrom a u).workTapePos j - base j| ≤ (radius : ℤ)

/-- **`RecordRun` is a segment** against the verdict pair, with the radius relation. -/
theorem recordRun_iff_segment {K : ℕ} {Γ S : Type} {x : List Γ} (R : MultiTapeTM K Γ S)
    (yes no : S) (base : Fin K → ℤ) (radius : ℕ) (a b : Cfg K Γ S x) (t : ℕ) :
    RecordRun R yes no base radius a b t ↔
      Segment R {some yes, some no} (fun p => ∀ j, |p j - base j| ≤ (radius : ℤ)) a b t := by
  simp only [RecordRun, Segment, Set.mem_insert_iff, Set.mem_singleton_iff, not_or]

/-- *Fit:* `CfgRecord`'s `recordRun_trans` (`Build/CfgRecord.lean:385`) is
`Turing.Segment.trans`. -/
theorem recordRun_trans {K : ℕ} {Γ S : Type} {x : List Γ}
    {R : MultiTapeTM K Γ S} {yes no : S} {base : Fin K → ℤ} {radius : ℕ}
    {a b c : Cfg K Γ S x} {t v : ℕ}
    (h : RecordRun R yes no base radius a b t)
    (g : RecordRun R yes no base radius b c v) :
    RecordRun R yes no base radius a c (t + v) :=
  (recordRun_iff_segment _ _ _ _ _ _ _ _).mpr
    (((recordRun_iff_segment _ _ _ _ _ _ _ _).mp h).trans
      ((recordRun_iff_segment _ _ _ _ _ _ _ _).mp g))

/-- *Fit:* `CfgRecord`'s `recordRun_zero` (`Build/CfgRecord.lean:372`) is
`Turing.Segment.zero`. -/
theorem recordRun_zero {K : ℕ} {Γ S : Type} {x : List Γ}
    (R : MultiTapeTM K Γ S) (yes no : S) (base : Fin K → ℤ) (radius : ℕ)
    (a : Cfg K Γ S x) (h : ∀ j, |a.workTapePos j - base j| ≤ (radius : ℤ)) :
    RecordRun R yes no base radius a a 0 :=
  (recordRun_iff_segment _ _ _ _ _ _ _ _).mpr (Segment.zero h)

/-- The **stationary machine** with no work tapes: every live state stays put, silently. -/
def idleTM {S : Type} (q₀ : S) : MultiTapeTM 0 Γ S :=
  ⟨q₀, fun q _ _ => ⟨0, fun _ => (none, 0), none, some q⟩⟩

/-- The stationary machine's run from a live configuration is constant.

**Proof sketch.** One step changes nothing (the action is stationary, silent and keeps the state);
induct on the time. -/
theorem idleTM_runFrom {S : Type} {x : List Γ} (q₀ : S) (c : Cfg 0 Γ S x) (hc : c.state ≠ none)
    (u : ℕ) : (idleTM (Γ := Γ) q₀).runFrom c u = c := by
  obtain ⟨q, hq⟩ := Option.ne_none_iff_exists'.mp hc
  have hstep : (idleTM (Γ := Γ) q₀).step c = c := by
    apply Cfg.ext
    · simp [idleTM, MultiTapeTM.step, hq, Action.apply]
    · simp [idleTM, MultiTapeTM.step, hq, Action.apply, moveInputPos]
    · funext j
      exact j.elim0
    · funext j
      exact j.elim0
    · simp [idleTM, MultiTapeTM.step, hq, Action.apply]
  induction u with
  | zero => rfl
  | succ u ih => rw [MultiTapeTM.runFrom_succ_eq_step', ih, hstep]

/-- *Fit:* the auditor's idling segment (round 1, counterexample 4), which neither a positive
`Turing.Contract` nor a `Turing.ReturnContract` packages, is a segment: the stationary machine,
run for two steps from `a`, avoids any verdict set not containing `a`. -/
theorem idle_segment {S : Type} {x : List Γ} (a : S) (F : Set (Option S)) (ha : some a ∉ F)
    (c : Cfg 0 Γ S x) (hc : c.state = some a) :
    Segment (idleTM a) F (fun _ => True) c c 2 := by
  have hrun := idleTM_runFrom (Γ := Γ) a c (by rw [hc]; exact Option.some_ne_none _)
  refine ⟨hrun 2, fun u _ => ?_, fun _ _ => trivial⟩
  rw [hrun u, hc]
  exact ha

/-- *Fit (MIG-0a gate, round 2):* a positive stationary run away from the anchor `a` is an
endpoint-inclusive segment against `a`, of every length `t`. -/
theorem stationary_safe {S : Type} {x : List Γ} (a q : S) (hq : q ≠ a) (c : Cfg 0 Γ S x)
    (hc : c.state = some q) (t : ℕ) :
    SafeSegment (idleTM q) {some a} (fun _ => True) c c t := by
  have hrun := idleTM_runFrom (Γ := Γ) q c (by rw [hc]; exact Option.some_ne_none _)
  refine safeSegment_iff.mpr ⟨hrun t, fun u _ hm => ?_, fun _ _ => trivial⟩
  rw [hrun u, hc] at hm
  exact hq (Option.some.inj hm)

/-- *Fit (MIG-0a gate, round 2):* but no first-exit contract of the stationary machine with exit
the anchor has an admissible start there: it would have to end at the anchor.

**Proof sketch.** The contract's endpoint is the run's, which is the start, at `q`; `exit_mem`
puts it at the anchor. -/
theorem stationary_no_contract {S : Type} {x : List Γ} {ι : Type} (a q : S) (hq : q ≠ a)
    (C : Contract (idleTM (Γ := Γ) q) q {some a} ι x) (i : ι) (c : Cfg 0 Γ S x)
    (hc : c.state = some q) : ¬ C.pre i c := by
  intro hp
  have hx := C.exit_mem i c hc hp
  rw [← C.run i c hc hp, idleTM_runFrom q c (by rw [hc]; exact Option.some_ne_none _), hc] at hx
  exact hq (Option.some.inj hx)

/-- *Fit copy* of `Build/EmitIterEmbed.lean:67` (`SafeRun`, public there but outside this file's
import closure), verbatim. Its avoidance is strict: before the end only. -/
def SafeRun {k : ℕ} {B : Type} {x : List Bool} (H : MultiTapeTM k Bool B) (avoid : B)
    (c : Cfg k Bool B x) (t : ℕ) (c' : Cfg k Bool B x) : Prop :=
  H.runFrom c t = c' ∧ ∀ t' < t, (H.runFrom c t').state ≠ some avoid

/-- **`SafeRun` is a segment** against the one avoided state; its `SafeRun.zero`, `cons` and
`trans` (`Build/EmitIterEmbed.lean:72`, `:77`, `:94`) are `Turing.Segment.zero`, `cons` and
`trans`. -/
theorem safeRun_iff_segment {k : ℕ} {B : Type} {x : List Bool} (H : MultiTapeTM k Bool B)
    (avoid : B) (c : Cfg k Bool B x) (t : ℕ) (c' : Cfg k Bool B x) :
    SafeRun H avoid c t c' ↔ Segment H {some avoid} (fun _ => True) c c' t := by
  simp [SafeRun, Segment]

/-- *Fit copy* of `Build/Primitives.lean:3505` (`splitSafe`, private there; `Catalog`'s
`f2_splitSafe`, `Build/Catalog.lean:9182`, is the same), verbatim. Its avoidance is
endpoint-inclusive. -/
def splitSafe {k : ℕ} {S : Type} {w : List Bool}
    (tm : MultiTapeTM k Bool S) (anchor : S) (c : Cfg k Bool S w) (t : ℕ) : Prop :=
  ∀ j ≤ t, (tm.runFrom c j).state ≠ some anchor

/-- **`splitSafe` is an endpoint-inclusive segment** against the anchor, ending at the run's own
configuration. -/
theorem splitSafe_iff_safeSegment {k : ℕ} {S : Type} {w : List Bool}
    (tm : MultiTapeTM k Bool S) (anchor : S) (c : Cfg k Bool S w) (t : ℕ) :
    splitSafe tm anchor c t ↔
      SafeSegment tm {some anchor} (fun _ => True) c (tm.runFrom c t) t := by
  rw [safeSegment_iff]
  simp [splitSafe]

/-- *Fit:* `Primitives`' `splitSafe_add` (`Build/Primitives.lean:3510`), verbatim, is
`Turing.SafeSegment.trans` at the configuration seam. -/
theorem splitSafe_add {k : ℕ} {S : Type} {w : List Bool}
    (tm : MultiTapeTM k Bool S) (anchor : S) (c : Cfg k Bool S w) (u v : ℕ)
    (hu : splitSafe tm anchor c u) (hv : splitSafe tm anchor (tm.runFrom c u) v) :
    splitSafe tm anchor c (u + v) := by
  have h := ((splitSafe_iff_safeSegment _ _ _ _).mp hu).trans
    ((splitSafe_iff_safeSegment _ _ _ _).mp hv)
  rw [← MultiTapeTM.runFrom_add] at h
  exact (splitSafe_iff_safeSegment _ _ _ _).mpr h

/-- *Fit copy* of `Robustness/SingleTape.lean:1182` (`DGSpan`, private there), verbatim. -/
def DGSpan {A Q : Type} {x : List A} (tm : MultiTapeTM 1 A Q)
    (c d : Cfg 1 A Q x) (t : ℕ) (lo hi : ℤ) : Prop :=
  tm.runFrom c t = d ∧ ∀ u ≤ t,
    lo ≤ (tm.runFrom c u).workTapePos 0 ∧ (tm.runFrom c u).workTapePos 0 ≤ hi

/-- **`DGSpan` is a segment** with nothing forbidden and an interval on the one tape; its
`dgSpan_add` (`Robustness/SingleTape.lean:1190`) is `Turing.Segment.trans`. -/
theorem dgSpan_iff_segment {A Q : Type} {x : List A} (tm : MultiTapeTM 1 A Q)
    (c d : Cfg 1 A Q x) (t : ℕ) (lo hi : ℤ) :
    DGSpan tm c d t lo hi ↔ Segment tm ∅ (fun p => lo ≤ p 0 ∧ p 0 ≤ hi) c d t := by
  simp [DGSpan, Segment]

/-- *Fit:* the conclusion of `RecordTable`'s `rt_find_round_entry`
(`Build/RecordTable.lean:1810`, private there), for any machine, anchor `a` and escape `e`: a
positive time, the live cut off both, and the endpoint at the escape or the anchor with moved
heads. It is a round with the time-derived box. -/
theorem findRound_round {S : Type} {x : List Γ} (M : MultiTapeTM k Γ S) (a e : S)
    (c : Cfg k Γ S x) (hc : c.state = some a) (t : ℕ) (b : Bool) (pos : Fin k → ℤ)
    (ht : 0 < t)
    (hcut : ∀ u, 0 < u → u < t → ∃ s, (M.runFrom c u).state = some s ∧ s ≠ a ∧ s ≠ e)
    (hrun : M.runFrom c t = { c with state := some (if b then e else a), workTapePos := pos }) :
    Round M a {some e} (HeadBox (fun j => c.workTapePos j - t) (fun j => c.workTapePos j + t))
      c (M.runFrom c t) t := by
  refine (Round.of_cut hc ht hcut ?_).time_box
  rw [hrun]
  cases b
  · exact Or.inl rfl
  · exact Or.inr rfl

/-- *Fit:* the round hypothesis of `CounterLoop`'s `counterLoopTMVia_run_done`
(`Build/CounterLoop.lean:1220`) follows from the round layer: rounds of the orbit with escape set
`{exit}`, each returning. The adapter probe feeds it to the public theorem. -/
theorem counterLoop_hround_of_rounds {S : Type} {x : List Γ} (M : MultiTapeTM k Γ S) (a : S)
    (exit : Option S) (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) (d : ℕ)
    {P : (Fin k → ℤ) → Prop}
    (h : ∀ r < d, Round M a {exit} P (roundOrbit M τ c r) (roundOrbit M τ c (r + 1))
        (τ (roundOrbit M τ c r)) ∧ (roundOrbit M τ c (r + 1)).state = some a) :
    ∀ r < d,
      (roundOrbit M τ c r).state = some a ∧
      0 < τ (roundOrbit M τ c r) ∧
      (∀ u, 0 < u → u < τ (roundOrbit M τ c r) →
        ∃ s, (M.runFrom (roundOrbit M τ c r) u).state = some s ∧
          s ≠ a ∧ exit ≠ some s) ∧
      (roundOrbit M τ c (r + 1)).state = some a := by
  intro r hr
  obtain ⟨hR, hret⟩ := h r hr
  refine ⟨hR.1, hR.2.1, fun u hu hut => ?_, hret⟩
  obtain ⟨s, hs, hsa, hsE⟩ := hR.cut_some u hu hut
  exact ⟨s, hs, hsa, fun h' => hsE (h' ▸ rfl)⟩

/-- *Fit copy* of `Build/CounterLoop.lean:713` (`counterSpan`, private there), verbatim. -/
def MIG0bFit.counterSpan {k : ℕ} {Γ S : Type} {x : List Γ}
    (H : MultiTapeTM (k + 1) Γ (CounterLoopState S))
    (z : Cfg (k + 1) Γ (CounterLoopState S) x) (t : ℕ) (cp : ℤ) (width : ℕ)
    (R : (Fin k → ℤ) → Prop) : Prop :=
  (∀ u < t, (H.runFrom z u).state ≠ some .done ∧
    (H.runFrom z u).state ≠ some .escape) ∧
  (∀ u ≤ t, (H.runFrom z u).workTapePos (Fin.last k) ∈
    Finset.Icc (cp - 1) (cp + (width : ℤ))) ∧
  (∀ u ≤ t, R (fun j => (H.runFrom z u).workTapePos j.castSucc))

/-- **`counterSpan` is a segment** (MIG-0a gate, round 2): a finite segment against both host
exits, ending at the run's own configuration, with the counter interval and the body relation as
one joint relation. It has no entry or endpoint constraint and admits time `0`. -/
theorem MIG0bFit.counterSpan_iff_segment {S : Type} {x : List Γ}
    (H : MultiTapeTM (k + 1) Γ (CounterLoopState S))
    (z : Cfg (k + 1) Γ (CounterLoopState S) x) (t : ℕ) (cp : ℤ) (width : ℕ)
    (R : (Fin k → ℤ) → Prop) :
    MIG0bFit.counterSpan H z t cp width R ↔
      Segment H {some .done, some .escape}
        (fun p => p (Fin.last k) ∈ Finset.Icc (cp - 1) (cp + (width : ℤ)) ∧
          R fun j => p j.castSucc) z (H.runFrom z t) t := by
  constructor
  · rintro ⟨hc, hi, hr⟩
    refine ⟨rfl, fun u hu hm => ?_, fun u hu => ⟨hi u hu, hr u hu⟩⟩
    rcases hm with hm | hm
    · exact (hc u hu).1 hm
    · exact (hc u hu).2 hm
  · rintro ⟨-, hc, hp⟩
    exact ⟨fun u hu => ⟨fun h => hc u hu (Or.inl h), fun h => hc u hu (Or.inr h)⟩,
      fun u hu => (hp u hu).1, fun u hu => (hp u hu).2⟩

/-- *Fit:* `CounterLoop`'s `counter_span_append` (`Build/CounterLoop.lean:727`), verbatim, is
`Turing.Segment.trans` after weakening each relation (`Turing.Segment.mono`).

**Proof sketch.** Read both spans as segments, weaken each relation to `R`, concatenate at the
seam, and read the result back. -/
theorem MIG0bFit.counter_span_append {S : Type} {x : List Γ}
    (H : MultiTapeTM (k + 1) Γ (CounterLoopState S))
    (z : Cfg (k + 1) Γ (CounterLoopState S) x) (s t : ℕ) (cp : ℤ) (width : ℕ)
    (R₁ R₂ R : (Fin k → ℤ) → Prop)
    (h₁ : MIG0bFit.counterSpan H z s cp width R₁)
    (h₂ : MIG0bFit.counterSpan H (H.runFrom z s) t cp width R₂)
    (hr₁ : ∀ p, R₁ p → R p) (hr₂ : ∀ p, R₂ p → R p) :
    MIG0bFit.counterSpan H z (s + t) cp width R := by
  have g₁ : Segment H {some .done, some .escape}
      (fun p => p (Fin.last k) ∈ Finset.Icc (cp - 1) (cp + (width : ℤ)) ∧ R fun j => p j.castSucc)
      z (H.runFrom z s) s :=
    ((MIG0bFit.counterSpan_iff_segment _ _ _ _ _ _).mp h₁).mono subset_rfl
      fun p hp => ⟨hp.1, hr₁ _ hp.2⟩
  have g₂ : Segment H {some .done, some .escape}
      (fun p => p (Fin.last k) ∈ Finset.Icc (cp - 1) (cp + (width : ℤ)) ∧ R fun j => p j.castSucc)
      (H.runFrom z s) (H.runFrom (H.runFrom z s) t) t :=
    ((MIG0bFit.counterSpan_iff_segment _ _ _ _ _ _).mp h₂).mono subset_rfl
      fun p hp => ⟨hp.1, hr₂ _ hp.2⟩
  have g := g₁.trans g₂
  rw [← MultiTapeTM.runFrom_add] at g
  exact (MIG0bFit.counterSpan_iff_segment _ _ _ _ _ _).mpr g

/-- *Fit (MIG-0a gate, round 2):* `counterSpan`'s genuine zero-time case, the base of
`counter_rounds`: duration, counter position and width `0`, the counter head at `0`, and the
constant-true relation. It is `Turing.Segment.zero`. -/
theorem MIG0bFit.counterSpan_zero {S : Type} {x : List Γ}
    (H : MultiTapeTM (k + 1) Γ (CounterLoopState S))
    (z : Cfg (k + 1) Γ (CounterLoopState S) x) (hz : z.workTapePos (Fin.last k) = 0) :
    MIG0bFit.counterSpan H z 0 0 0 fun _ => True :=
  (MIG0bFit.counterSpan_iff_segment _ _ _ _ _ _).mpr
    (Segment.zero ⟨by rw [hz]; simp, trivial⟩)

example {S : Type} {x : List Γ} (M : MultiTapeTM k Γ S) (anchor : S) (c : Cfg k Γ S x)

/-- *Finding 3:* with the identity encoding and a target entry `b` distinct from the source entry
`a`, the transported contract has no admissible start (the auditor's adversarial check 6). -/
example {S : Type} {x : List Γ} {M : MultiTapeTM k Γ S} {a b : S} (hab : a ≠ b)
    {X : Set (Option S)} {ι : Type} (C : Contract M a X ι x)
    (L : Lockstep M M id (fun _ => 1) X X) (blo bhi : Cfg k Γ S x → Fin k → ℤ)
    (hbox : ∀ c v, (v = 0 ∨ (c.state ∉ X ∧ v ≤ 1)) → ∀ j,
      (M.runFrom (id c) v).workTapePos j ∈ Finset.Icc (blo c j) (bhi c j))
    (p : ι × Cfg k Γ S x) (d : Cfg k Γ S x) (hd : d.state = some b) :
    ¬ (C.lockstep L b blo bhi hbox).pre p d := by
  rintro ⟨rfl, hs, -⟩
  exact hab (Option.some.inj (hs.symm.trans hd))

/-- *Finding 4:* for a single live escape `e ≠ anchor`, the release's exits are two distinct live
states, the shape of `Turing.Contract.branch`'s first phase. -/
example {S : Type} (a e : S) (he : e ≠ a) :
    Option.map Sum.inr '' insert (some a) ({some e} : Set (Option S)) =
        ({some (Sum.inr a), some (Sum.inr e)} : Set (Option (Unit ⊕ S))) ∧
      (Sum.inr a : Unit ⊕ S) ≠ Sum.inr e := by

  rw [Set.image_insert_eq, Set.image_singleton]
