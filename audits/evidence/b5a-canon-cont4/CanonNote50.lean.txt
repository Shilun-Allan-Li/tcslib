import TCSlib.Complexity.TuringMachine.CodeSimulator

/-!
# Note 50 on the complete physical canonizer (B5a canonizer continuation 4)

Permanent sanity cases at fill (wave-1 statement gate, note 50), executed on the canonizer
of `Turing.MachineFormat.exists_polyCanonizer` for a format whose own guard is `0`: one machine
`C` and one degree `d` handle all four cases, each within `d · (|α| + 1)^d` steps.

* (a) the header of `2^128` with a one-bit remainder: the fallback, in time polynomial in the
  input's length (about 261 bits), never in `2^128`;
* (b) an encoded empty header with nothing after it: the fallback;
* (c) a malformed pair encoding: the fallback;
* (d) every well-formed code followed by any `true` padding: its own serialization.

The four cases follow from the public headline and the public decoding laws only. The file
is complete when `#print axioms note50_cases` shows no `sorryAx`. No `#eval`, `native_decide`
or `IO`.
-/

open Turing

/-- The one-tape machine format with its own guard set to `0`. -/
def note50Format : MachineFormat :=
  { oneTapeMachineFormat with
    guard := 0
    tableBits_length := fun _ _ => by simp }

/-- The variant's guard is `0`. -/
theorem note50_guard : note50Format.guard = 0 := rfl

/-- The table reader fails on an empty remainder, at any declared state count. -/
theorem note50_readTable_nil (n : ℕ) : note50Format.readTable n [] = none := rfl

/-- A canonical header followed by a single `false` decodes to the fallback: the initial state
`0` is read and the table is empty. -/
theorem note50_decode_short (n : ℕ) :
    note50Format.decode (pairEncode n.bits [false]) = note50Format.fallback := by
  apply CodeFormat.decode_of_parseFull_none
  simp [CodeFormat.parseFull, pairDecode_pairEncode, codeBitsNat_bits, codeReadFin,
    codeReadUnary, note50_readTable_nil, note50_guard]

/-- An encoded empty header with nothing after it decodes to the fallback. -/
theorem note50_decode_empty :
    note50Format.decode (pairEncode [] []) = note50Format.fallback := by
  apply CodeFormat.decode_of_parseFull_none
  simp [CodeFormat.parseFull, pairDecode_pairEncode, codeBitsNat, codeReadFin, codeReadUnary]

/-- A malformed pair encoding decodes to the fallback. -/
theorem note50_decode_malformed :
    note50Format.decode [true, false] = note50Format.fallback := by
  apply CodeFormat.decode_of_parseFull_none
  simp [CodeFormat.parseFull, pairDecode]

/-- **Note 50.** One canonizer machine and one degree for all four cases. -/
theorem note50_cases :
    ∃ (C : FinTM Bool) (d : ℕ),
      C.ComputesInTime (pairEncode (2 ^ 128).bits [false])
          (note50Format.serialize note50Format.fallback)
          (d * ((pairEncode (2 ^ 128).bits [false]).length + 1) ^ d) ∧
      C.ComputesInTime (pairEncode [] []) (note50Format.serialize note50Format.fallback)
          (d * ((pairEncode [] []).length + 1) ^ d) ∧
      C.ComputesInTime [true, false] (note50Format.serialize note50Format.fallback)
          (d * ([true, false].length + 1) ^ d) ∧
      ∀ (M : note50Format.Code) (m : ℕ),
        C.ComputesInTime (note50Format.serialize M ++ List.replicate m true)
          (note50Format.serialize M)
          (d * ((note50Format.serialize M ++ List.replicate m true).length + 1) ^ d) := by
  obtain ⟨C, d, canon⟩ := note50Format.exists_polyCanonizer
  refine ⟨C, d, ?_, ?_, ?_, ?_⟩
  · have run := canon (pairEncode (2 ^ 128).bits [false])
    dsimp only at run
    rwa [note50_decode_short] at run
  · have run := canon (pairEncode [] [])
    dsimp only at run
    rwa [note50_decode_empty] at run
  · have run := canon [true, false]
    dsimp only at run
    rwa [note50_decode_malformed] at run
  · intro M m
    have run := canon (note50Format.serialize M ++ List.replicate m true)
    dsimp only at run
    rwa [CodeFormat.decode_serialize_pad] at run

#print axioms note50_cases
