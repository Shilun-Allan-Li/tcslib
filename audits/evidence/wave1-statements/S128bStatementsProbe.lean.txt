/-! ## Executed checks (W1 pre-ship): layout facts on concrete tables and records

Every `example` is decided by evaluation (`decide`), or by `simp` where `Nat.bits`
(well-founded recursion) must be unfolded first. Tables are over `Fin 3`, at displaced

namespace W1Check
open Turing

/-- A one-tape configuration with tape contents `t` and head `h`. -/
def c1 {S : Type} (q : S) (t : ℤ → Option (Fin 3)) (h : ℤ) : Cfg 1 (Fin 3) S ([] : List (Fin 3)) :=
  ⟨some q, 1, fun _ => t, fun _ => h, []⟩

/-- A header and two entries of `m = 2` fields. -/
def h₀ : List (Fin 3) := [2]
def es₀ : List (List (List (Fin 3))) := [[[1], [0]], [[2], [1, 1]]]

/-- The table at origin `3`, with nonblank frame cells at `0` and `15`, just beyond the
two sentinels on each side. -/
def tab₀ (z : ℤ) : Option (Fin 3) :=
  if z = 0 ∨ z = 15 then some 2 else fieldsTape (tableFields h₀ es₀) (z - 3)

/-- The empty table (a header alone) at origin `3`, frame at `0` and `7`. -/
def tabE (z : ℤ) : Option (Fin 3) :=
  if z = 0 ∨ z = 7 then some 2 else fieldsTape (tableFields h₀ []) (z - 3)

-- layout lengths: `joinFields_length`, `fieldOffset_length` and its empty exception
example : (joinFields ([[1], [2, 0], [1]] : List (List (Fin 3)))).length =
      ((([[1], [2, 0], [1]] : List (List (Fin 3))).map List.length).sum + 3 - 1) ∧
    fieldOffset ([[1], [2, 0], [1]] : List (List (Fin 3))) 3 =
      (joinFields ([[1], [2, 0], [1]] : List (List (Fin 3)))).length + 1 ∧
    (joinFields ([[], [1], []] : List (List (Fin 3)))).length = 3 ∧

/-! ### An executed witness candidate for R8.5c -/

/-- **A witness candidate for R8.5c** (probe only): a five-state framed counter, the
counting phase of Catalog's private `f2_counterTM` without its emission, plus an input
return. States: count `0`, carry `1`, counter rewind `2`, input return `3`, exit `4`.
On the instances below it meets every clause of `Turing.exists_inputLengthTM` with
`K = 0`, `C = 5`; its exit time is `5n + 2 - 2·popcount n`. -/
def lenTM : MultiTapeTM 1 Bool (Fin 5) where
  q₀ := 0
  tr := fun q inp w =>
    if q = 0 then
      match inp with
      | none => ⟨-1, fun _ => (none, 0), none, some 3⟩
      | some _ =>
        if w 0 = some true then ⟨1, fun _ => (some (some false), 1), none, some 1⟩
        else ⟨1, fun _ => (some (some true), -1), none, some 2⟩
    else if q = 1 then
      if w 0 = some true then ⟨0, fun _ => (some (some false), 1), none, some 1⟩
      else ⟨0, fun _ => (some (some true), -1), none, some 2⟩
    else if q = 2 then
      match w 0 with
      | some _ => ⟨0, fun _ => (none, -1), none, some 2⟩
      | none => ⟨0, fun _ => (none, 1), none, some 0⟩
    else if q = 3 then
      match inp with
      | some _ => ⟨-1, fun _ => (none, 0), none, some 3⟩
      | none => ⟨1, fun _ => (none, 0), none, some 4⟩
    else ⟨0, fun _ => (none, 0), none, some 4⟩

/-- Start: head at `5`, input head at `1`, nonblank frame at `3` (two left of the head) and at
`5 + |w| + 1` (one beyond the slot), `w` the expected word. -/
def st' (x w : List Bool) : Cfg 1 Bool (Fin 5) x :=
  ⟨some 0, 1, fun _ z => if z = 3 ∨ z = 5 + w.length + 1 then some true else none, fun _ => 5, []⟩

/-- Every clause of `exists_inputLengthTM` at `C = 5`, on cells `[0, 20)`. -/
def chk (x w : List Bool) (T : ℕ) : Bool :=
  let d := st' x w
  let f := lenTM.runFrom d T
  f.state == some 4 &&
  (List.range T).all (fun t => (lenTM.runFrom d t).state != some 4) &&
  decide (T ≤ 5 * (x.length + 1)) &&
  (List.range 20).all (fun z => f.workTapes 0 z ==
      (if 5 ≤ (z : ℤ) ∧ (z : ℤ) < 5 + w.length then FinTM.bufferTape w ((z : ℤ) - 5)
       else d.workTapes 0 z)) &&
  f.workTapePos 0 == 5 && (f.inputPos : ℕ) == 1 && f.output == [] &&
  (List.range (T + 1)).all (fun t => decide (4 ≤ (lenTM.runFrom d t).workTapePos 0 ∧
      (lenTM.runFrom d t).workTapePos 0 ≤ 5 + w.length))

example : chk [] [] 2 = true := by decide
example : chk [true] [true] 5 = true := by decide
example : chk [true, false] [false, true] 10 = true := by decide
example : chk [true, false, true] [true, true] 13 = true := by decide
example : chk (List.replicate 4 false) [false, false, true] 20 = true := by decide
example : chk (List.replicate 7 true) [true, true, true] 31 = true := by decide
example : chk (List.replicate 8 true) [false, false, false, true] 40 = true := by decide
example : Nat.bits 0 = [] ∧ Nat.bits 1 = [true] ∧ Nat.bits 2 = [false, true] ∧
    Nat.bits 3 = [true, true] ∧ Nat.bits 4 = [false, false, true] ∧
    Nat.bits 7 = [true, true, true] ∧ Nat.bits 8 = [false, false, false, true] := by simp [Nat.bits, Nat.binaryRec]

-- negative controls: one step early, the wrong word, a slot one cell too short on the right
example : chk (List.replicate 8 true) [false, false, false, true] 39 = false := by decide
example : chk (List.replicate 3 true) [true, false] 13 = false := by decide
example : chk [true] [true] 5 = true ∧ chk [true] [true] 4 = false := by decide
example : chk [true] [true] 6 = false := by decide
