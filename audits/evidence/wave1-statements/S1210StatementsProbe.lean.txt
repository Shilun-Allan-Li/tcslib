/-! ### Budget arithmetic -/

/-- A constant bound is exponential-polynomial. -/
theorem probe_expPoly_const (A : ℕ) : ExpPoly fun _ => A :=
  (expPoly_poly A 0).of_le fun n => by simp

/-- `A · 2^(K·(n + 1)^k)` is exponential-polynomial. -/
theorem probe_expPoly_mul_two_pow (A K k : ℕ) : ExpPoly fun n => A * 2 ^ (K * (n + 1) ^ k) := by
  refine ⟨A + K, k, fun n => ?_⟩
  have h1 : A < 2 ^ A := Nat.lt_two_pow_self
  have h2 : 1 ≤ (n + 1) ^ k := Nat.one_le_pow _ _ (Nat.succ_pos _)
  calc A * 2 ^ (K * (n + 1) ^ k) ≤ 2 ^ A * 2 ^ (K * (n + 1) ^ k) := Nat.mul_le_mul_right _ h1.le
    _ = 2 ^ (A + K * (n + 1) ^ k) := by rw [pow_add]
    _ ≤ 2 ^ ((A + K) * (n + 1) ^ k) := Nat.pow_le_pow_right (by omega) (by nlinarith)

/-- `c·(n^k + 1) + 1 ≤ (2c + 1)·(n + 1)^k`. -/
theorem probe_poly_succ_le (c k n : ℕ) : c * (n ^ k + 1) + 1 ≤ (2 * c + 1) * (n + 1) ^ k := by
  have hX1 : 1 ≤ (n + 1) ^ k := Nat.one_le_pow _ _ (Nat.succ_pos _)
  have hnX : n ^ k ≤ (n + 1) ^ k := Nat.pow_le_pow_left (Nat.le_succ n) k
  have : c * (n ^ k + 1) ≤ c * (2 * (n + 1) ^ k) := Nat.mul_le_mul_left _ (by omega)
  calc c * (n ^ k + 1) + 1 ≤ c * (2 * (n + 1) ^ k) + (n + 1) ^ k := by omega
    _ = (2 * c + 1) * (n + 1) ^ k := by ring

/-- **B1's ledger** (the P3.1 round-1 finding-2 shape): with oracle-machine time
`t = c·(n^k + 1)` and decider time `C·(m + 1)^d`, the uniform simulation bound
`a·(t + 1)·(t + 1 + C·(t + 1)^d)` is a polynomial of degree `k·(d + 2)`. -/
theorem probe_b1_budget (a c k C d n : ℕ) :
    a * (c * (n ^ k + 1) + 1) * (c * (n ^ k + 1) + 1 + C * (c * (n ^ k + 1) + 1) ^ d) ≤
      a * (1 + C) * (2 * c + 1) ^ (d + 2) * 2 ^ (k * (d + 2)) * (n ^ (k * (d + 2)) + 1) := by
  have hYX := probe_poly_succ_le c k n
  generalize hY : c * (n ^ k + 1) + 1 = Y at hYX ⊢
  have hY1 : 1 ≤ Y := by omega
  have hYd : Y + C * Y ^ d ≤ (1 + C) * Y ^ (d + 1) := by
    have h1 : Y ≤ Y ^ (d + 1) := by
      calc Y = Y ^ 1 := (pow_one Y).symm
        _ ≤ Y ^ (d + 1) := Nat.pow_le_pow_right hY1 (by omega)
    have h3 : C * Y ^ d ≤ C * Y ^ (d + 1) :=
      Nat.mul_le_mul_left _ (Nat.pow_le_pow_right hY1 (Nat.le_succ d))
    calc Y + C * Y ^ d ≤ Y ^ (d + 1) + C * Y ^ (d + 1) := Nat.add_le_add h1 h3
      _ = (1 + C) * Y ^ (d + 1) := by ring
  have hpow : Y ^ (d + 2) ≤ (2 * c + 1) ^ (d + 2) * (n + 1) ^ (k * (d + 2)) := by
    calc Y ^ (d + 2) ≤ ((2 * c + 1) * (n + 1) ^ k) ^ (d + 2) := Nat.pow_le_pow_left hYX _
      _ = (2 * c + 1) ^ (d + 2) * (n + 1) ^ (k * (d + 2)) := by rw [mul_pow, ← pow_mul]
  calc a * Y * (Y + C * Y ^ d) ≤ a * Y * ((1 + C) * Y ^ (d + 1)) := Nat.mul_le_mul_left _ hYd
    _ = a * (1 + C) * Y ^ (d + 2) := by ring
    _ ≤ a * (1 + C) * ((2 * c + 1) ^ (d + 2) * (n + 1) ^ (k * (d + 2))) :=
        Nat.mul_le_mul_left _ hpow
    _ ≤ a * (1 + C) * ((2 * c + 1) ^ (d + 2) * (2 ^ (k * (d + 2)) * (n ^ (k * (d + 2)) + 1))) :=
        Nat.mul_le_mul_left _ (Nat.mul_le_mul_left _ (succ_pow_le _ _))
    _ = a * (1 + C) * (2 * c + 1) ^ (d + 2) * 2 ^ (k * (d + 2)) * (n ^ (k * (d + 2)) + 1) := by
        ring

/-- **The `P^EXPCOM` ledger**: with oracle-machine time `t = c·(n^k + 1)` and decider time
`A·2^(B(m + 1))`, the uniform simulation bound is exponential-polynomial. -/
theorem probe_pexpcom_budget (a c k A B : ℕ) :
    ExpPoly fun n => a * (c * (n ^ k + 1) + 1) *
      (c * (n ^ k + 1) + 1 + A * 2 ^ (B * (c * (n ^ k + 1) + 1))) := by
  have eG : ExpPoly fun n => (2 * c + 1) * (n + 1) ^ k := expPoly_poly _ _
  refine (((probe_expPoly_const a).mul eG).mul
    (eG.add (probe_expPoly_mul_two_pow A (B * (2 * c + 1)) k))).of_le fun n => ?_
  have hY := probe_poly_succ_le c k n
  have hexp : A * 2 ^ (B * (c * (n ^ k + 1) + 1)) ≤ A * 2 ^ (B * (2 * c + 1) * (n + 1) ^ k) := by
    apply Nat.mul_le_mul_left
    apply Nat.pow_le_pow_right (by omega)
    rw [Nat.mul_assoc]
    exact Nat.mul_le_mul_left _ hY
  exact Nat.mul_le_mul (Nat.mul_le_mul_left _ hY) (Nat.add_le_add hY hexp)

/-- **The summit's ledger** ([AB09, Example 3.6(3)]; the P3.2 round-2 uniform ledger in
this route's shape): the bound returned by `Complexity.exists_proj_decider` around the
certificate shell, the simulated choice machine and the `EXPCOM` decider is
exponential-polynomial. -/
theorem probe_summit_budget (a' A B b e b' C k : ℕ) :
    ExpPoly fun n => b' * 2 ^ (C * (n + 1) ^ k) *
      (b * ((n + C * (n + 1) ^ k + 1) ^ e +
          a' * (2 * (n + C * (n + 1) ^ k) + 2 + 1) *
            (2 * (n + C * (n + 1) ^ k) + 2 + 1 +
              A * 2 ^ (B * (2 * (n + C * (n + 1) ^ k) + 2 + 1)))) +
        (n + C * (n + 1) ^ k + 1) ^ (k + 1)) := by
  have hG : ∀ n : ℕ, n + C * (n + 1) ^ k + 1 ≤ (C + 1) * (n + 1) ^ (k + 1) := by
    intro n
    have h1 : n + 1 ≤ (n + 1) ^ (k + 1) := by
      calc n + 1 = (n + 1) ^ 1 := (pow_one _).symm
        _ ≤ (n + 1) ^ (k + 1) := Nat.pow_le_pow_right (Nat.succ_pos n) (by omega)
    have h3 : C * (n + 1) ^ k ≤ C * (n + 1) ^ (k + 1) :=
      Nat.mul_le_mul_left _ (Nat.pow_le_pow_right (Nat.succ_pos n) (by omega))
    calc n + C * (n + 1) ^ k + 1 = (n + 1) + C * (n + 1) ^ k := by ring
      _ ≤ (n + 1) ^ (k + 1) + C * (n + 1) ^ (k + 1) := Nat.add_le_add h1 h3
      _ = (C + 1) * (n + 1) ^ (k + 1) := by ring
  have eG : ∀ d, ExpPoly fun n => ((C + 1) * (n + 1) ^ (k + 1)) ^ d := fun d =>
    (expPoly_poly ((C + 1) ^ d) ((k + 1) * d)).of_le fun n =>
      le_of_eq (by rw [mul_pow, ← pow_mul])
  have e3G : ExpPoly fun n => 3 * ((C + 1) * (n + 1) ^ (k + 1)) :=
    (expPoly_poly (3 * (C + 1)) (k + 1)).of_le fun n => le_of_eq (by ring)
  have eExp := probe_expPoly_mul_two_pow A (3 * B * (C + 1)) (k + 1)
  refine ((probe_expPoly_mul_two_pow b' C k).mul (((probe_expPoly_const b).mul ((eG e).add
    (((probe_expPoly_const a').mul e3G).mul (e3G.add eExp)))).add (eG (k + 1)))).of_le
    fun n => ?_
  have hg := hG n
  generalize hgdef : n + C * (n + 1) ^ k = g at hg ⊢
  generalize hGdef : (C + 1) * (n + 1) ^ (k + 1) = G at hg ⊢
  have h3 : 2 * g + 2 + 1 ≤ 3 * G := by omega
  have hexp : A * 2 ^ (B * (2 * g + 2 + 1)) ≤ A * 2 ^ (3 * B * (C + 1) * (n + 1) ^ (k + 1)) := by
    apply Nat.mul_le_mul_left
    apply Nat.pow_le_pow_right (by omega)
    calc B * (2 * g + 2 + 1) ≤ B * (3 * G) := Nat.mul_le_mul_left _ h3
      _ = 3 * B * (C + 1) * (n + 1) ^ (k + 1) := by rw [← hGdef]; ring
  apply Nat.mul_le_mul_left
  apply Nat.add_le_add _ (Nat.pow_le_pow_left hg _)
  apply Nat.mul_le_mul_left
  apply Nat.add_le_add (Nat.pow_le_pow_left hg _)
  exact Nat.mul_le_mul (Nat.mul_le_mul_left _ h3) (Nat.add_le_add h3 hexp)

/-! ### B1: `Complexity.POracle_eq_P_of_mem_P` -/

/-- **B1's last target, derived** ([AB09, Example 3.6(2)]). `⊇` is
`Complexity.P_subset_POracle`. For `⊆`, `Turing.FinOracleTM.exists_plain_decider` turns the
`Pᴼ` machine and the `P` decider of `O` into a plain decider within
`a·(t + 1)·(t + 1 + C·(t + 1)^d)`, `t = c·(n^k + 1)`, a polynomial
(`probe_b1_budget`); `Complexity.mem_P_of_dtime_le` concludes. -/
theorem POracle_eq_P_of_mem_P' {O : Language Bool} (h : O ∈ P) : POracle O = P := by
  refine Set.Subset.antisymm ?_ (P_subset_POracle O)
  intro L hL
  obtain ⟨k, c, M, hM⟩ := Set.mem_iUnion.mp hL
  obtain ⟨C, d, D, hD⟩ := mem_P_iff.mp h
  have hτ : Monotone fun m => C * (m + 1) ^ d := fun m m' hm =>
    Nat.mul_le_mul_left _ (Nat.pow_le_pow_left (by omega) _)
  obtain ⟨Pm, a, hP⟩ := FinOracleTM.exists_plain_decider hτ hM hD
  have hDT : L ∈ DTIME fun n => a * (c * (n ^ k + 1) + 1) *
      (c * (n ^ k + 1) + 1 + C * (c * (n ^ k + 1) + 1) ^ d) :=
    ⟨1, Pm, fun x => (hP x).mono (Nat.le_of_eq (Nat.one_mul _).symm)⟩
  exact mem_P_of_dtime_le hDT _ _ (probe_b1_budget a c k C d)

example : @POracle_eq_P_of_mem_P = @POracle_eq_P_of_mem_P' := rfl

/-! ### B5: the `EXPCOM` cluster -/

/-- **`EXPCOM` is decidable in time `2^(O(m))`** — B5's own consumer-side lemma, not part
of the shared layer (proposed home `Diagonalization/EXPCOM.lean`, the "query-dispatcher
routine" of the summit's sketch). [AB09, Example 3.6(3)]

**Proof sketch.** Parse `z` as `pairEncode α (pairEncode x 1ⁿ)` (aligned pair parsing;
malformed strings and non-unary tails reject), so `2|α| + 2|x| + n + 4 = |z|`. Build
`pairEncode (pairEncode (bits 2ⁿ) α) x`, with `bits 2ⁿ` of `n + 1` bits, and run
`expCode.simulator`: its two uniform clauses decide bounded acceptance within
`d·(|α| + |x| + 2ⁿ + 1)^d ≤ d·5^d·2^(d|z|)`, and parse uniqueness (`pairEncode_injective`
twice, `List.length` on the unary tails) identifies the witness. Phases by
`Turing.computesFunInTime_comp`. -/
theorem exists_EXPCOM_decider :
    ∃ (D : FinTM Bool) (A B : ℕ), D.DecidesInTime EXPCOM fun m => A * 2 ^ (B * (m + 1)) := by
  sorry

/-- **`P^EXPCOM ⊆ EXP` directly from the deterministic row** (no nondeterminism needed):
the `Pᴼ` machine simulated with `EXPCOM` answered by `Complexity.exists_EXPCOM_decider`
within an exponential-polynomial bound (`probe_pexpcom_budget`). -/
theorem POracle_EXPCOM_subset_EXP' : POracle EXPCOM ⊆ EXP := by
  intro L hL
  obtain ⟨k, c, M, hM⟩ := Set.mem_iUnion.mp hL
  obtain ⟨D, A, B, hD⟩ := exists_EXPCOM_decider
  have hτ : Monotone fun m => A * 2 ^ (B * (m + 1)) := fun m m' hm =>
    Nat.mul_le_mul_left _ (Nat.pow_le_pow_right (by omega) (Nat.mul_le_mul_left _ (by omega)))
  obtain ⟨Pm, a, hP⟩ := FinOracleTM.exists_plain_decider hτ hM hD
  exact ExpPoly.mem_EXP (probe_pexpcom_budget a c k A B)
    ⟨1, Pm, fun x => (hP x).mono (Nat.le_of_eq (Nat.one_mul _).symm)⟩

/-- **B5's summit, derived** ([AB09, Example 3.6(3)], the last inclusion). Choice-word
enumeration reduces it to the deterministic row: the choice-word machine of `N`
(`Turing.FinOracleNDTM.exists_choiceTM`), simulated with `EXPCOM` answered by
`Complexity.exists_EXPCOM_decider` (`Turing.FinOracleTM.exists_plain_simulation_of_monotone`
from the preloaded certificate), wrapped by the certificate shell
(`Turing.FinTM.exists_certificateShell`), enumerated by `Complexity.exists_proj_decider`,
recognized by `Turing.FinOracleNDTM.mem_iff_exists_choiceOut` and
`Turing.FinTM.certificateShell_proj`, and normalized into `EXP` by
`Complexity.ExpPoly.mem_EXP` (`probe_summit_budget`). -/
theorem NPOracle_EXPCOM_subset_EXP' : NPOracle EXPCOM ⊆ EXP := by
  intro L hL
  obtain ⟨k, a, N, hN⟩ := Set.mem_iUnion.mp hL
  obtain ⟨Mc, jc, hMc⟩ := FinOracleNDTM.exists_choiceTM N
  obtain ⟨D, A, B, hD⟩ := exists_EXPCOM_decider
  have hτ : Monotone fun m => A * 2 ^ (B * (m + 1)) := fun m m' hm =>
    Nat.mul_le_mul_left _ (Nat.pow_le_pow_right (by omega) (Nat.mul_le_mul_left _ (by omega)))
  obtain ⟨P, a', hk, hP⟩ :=
    FinOracleTM.exists_plain_simulation_of_monotone Mc D EXPCOM _ hτ hD
  let jP : Fin P.k := ⟨jc.castSucc, lt_of_lt_of_le jc.castSucc.isLt hk⟩
  have hshell : ∀ x u : List Bool, u.length = 2 * a * (x.length + 1) ^ k →
      (P.tm.runFrom (Cfg.ofWords (input := x) P.tm.q₀ (Function.update (fun _ => []) jP u))
          ((fun m => a' * (2 * m + 2 + 1) *
            (2 * m + 2 + 1 + A * 2 ^ (B * (2 * m + 2 + 1)))) (x.length + u.length))).state =
          none ∧
        (P.tm.runFrom (Cfg.ofWords (input := x) P.tm.q₀ (Function.update (fun _ => []) jP u))
          ((fun m => a' * (2 * m + 2 + 1) *
            (2 * m + 2 + 1 + A * 2 ^ (B * (2 * m + 2 + 1)))) (x.length + u.length))).output =
          N.choiceOut EXPCOM x u := by
    intro x u _
    obtain ⟨h1, h2⟩ := hMc EXPCOM x u
    have hw : Function.update (fun _ : Fin (Mc.k + 1) => ([] : List Bool)) jc.castSucc u
        (Fin.last Mc.k) = [] := Function.update_of_ne (Fin.castSucc_ne_last jc).symm _ _
    obtain ⟨s, hs, hs1, hs2⟩ := hP x _ hw _ h1
    rw [FinTM.prefixWords_update _ _ hk] at hs1 hs2
    have hle : s ≤ a' * (2 * (x.length + u.length) + 2 + 1) *
        (2 * (x.length + u.length) + 2 + 1 +
          A * 2 ^ (B * (2 * (x.length + u.length) + 2 + 1))) := by
      refine hs.trans ?_
      have hu : u.length ≤ x.length + u.length := Nat.le_add_left _ _
      have h3 : 2 * u.length + 2 + 1 ≤ 2 * (x.length + u.length) + 2 + 1 := by omega
      exact Nat.mul_le_mul (Nat.mul_le_mul_left _ h3)
        (Nat.add_le_add h3 (hτ (by omega : 2 * u.length + 2 ≤ 2 * (x.length + u.length) + 2)))
    refine ⟨?_, ?_⟩
    · dsimp only
      rw [← Nat.add_sub_of_le hle, MultiTapeTM.runFrom_add, MultiTapeTM.runFrom_of_halt _ hs1]
      exact hs1
    · dsimp only
      rw [← Nat.add_sub_of_le hle, MultiTapeTM.runFrom_add, MultiTapeTM.runFrom_of_halt _ hs1]
      exact hs2.trans h2
  obtain ⟨V, bV, e, hV⟩ :=
    FinTM.exists_certificateShell P jP (2 * a) k (N.choiceOut EXPCOM)
      (fun m => a' * (2 * m + 2 + 1) * (2 * m + 2 + 1 + A * 2 ^ (B * (2 * m + 2 + 1)))) hshell
  obtain ⟨b', E, hE⟩ := exists_proj_decider (2 * a) k _ _ V hV
  rw [FinTM.certificateShell_proj] at hE
  have hLeq : {x : List Bool | ∃ u, u.length = 2 * a * (x.length + 1) ^ k ∧
      N.choiceOut EXPCOM x u = [true]} = L :=
    Set.ext fun x => (FinOracleNDTM.mem_iff_exists_choiceOut hN x).symm
  rw [hLeq] at hE
  exact ExpPoly.mem_EXP (probe_summit_budget a' A B bV e b' (2 * a) k)
    ⟨1, E, fun x => (hE x).mono (Nat.le_of_eq (Nat.one_mul _).symm)⟩

example : @NPOracle_EXPCOM_subset_EXP = @NPOracle_EXPCOM_subset_EXP' := rfl

/-- `P^EXPCOM = EXP`, as sketched: `⊇` is `Complexity.EXP_subset_POracle_EXPCOM` (B5's
padding reduction, sorried in the tip); `⊆` chains `Complexity.POracle_subset_NPOracle`
with the summit. -/
theorem POracle_EXPCOM_eq_EXP' : POracle EXPCOM = EXP :=
  Set.Subset.antisymm ((POracle_subset_NPOracle EXPCOM).trans NPOracle_EXPCOM_subset_EXP')
    EXP_subset_POracle_EXPCOM

example : @POracle_EXPCOM_eq_EXP = @POracle_EXPCOM_eq_EXP' := rfl

/-- `NP^EXPCOM = EXP`, as sketched. -/
theorem NPOracle_EXPCOM_eq_EXP' : NPOracle EXPCOM = EXP :=
  Set.Subset.antisymm NPOracle_EXPCOM_subset_EXP'
    (EXP_subset_POracle_EXPCOM.trans (POracle_subset_NPOracle EXPCOM))

example : @NPOracle_EXPCOM_eq_EXP = @NPOracle_EXPCOM_eq_EXP' := rfl

/-- `P^EXPCOM = NP^EXPCOM`, as sketched. -/
theorem POracle_EXPCOM_eq_NPOracle_EXPCOM' : POracle EXPCOM = NPOracle EXPCOM :=
  POracle_EXPCOM_eq_EXP'.trans NPOracle_EXPCOM_eq_EXP'.symm

example : @POracle_EXPCOM_eq_NPOracle_EXPCOM = @POracle_EXPCOM_eq_NPOracle_EXPCOM' := rfl

/-- The Baker-Gill-Solovay theorem, as sketched: `A := EXPCOM`, and `B` from the proved
stage construction `Complexity.exists_oracle_ne`. -/
theorem baker_gill_solovay' :
    (∃ A : Language Bool, POracle A = NPOracle A) ∧
    ∃ B : Language Bool, POracle B ≠ NPOracle B := by
  refine ⟨⟨EXPCOM, POracle_EXPCOM_eq_NPOracle_EXPCOM'⟩, ?_⟩
  obtain ⟨B, h1, h2⟩ := exists_oracle_ne
  exact ⟨B, fun h => h2 (h ▸ h1)⟩

example : @baker_gill_solovay = @baker_gill_solovay' := rfl

/-! ## Sorry sources of the derived consumers

For each consumer, the declarations in its dependency closure whose own body uses
`sorryAx`: the W2 rows it rests on, plus the upstream sorried statements of the tip. -/

open Lean Elab Command in
/-- List the declarations in the dependency closure of a constant whose own type or body
mentions `sorryAx`. -/
elab "#sorry_sources " id:ident : command => do
  let env ← getEnv
  let root := id.getId
  let mut visited : NameSet := {}
  let mut stack : List Name := [root]
  let mut out : Array Name := #[]
  while true do
    match stack with
    | [] => break
    | n :: rest =>
      stack := rest
      if visited.contains n then continue
      visited := visited.insert n
      match env.find? n with
      | none => if n == root then throwError "#sorry_sources: unknown constant {n}"
      | some ci =>
        let consts := ci.getUsedConstantsAsSet
        if consts.contains ``sorryAx then out := out.push n
        for c in consts do
          if !visited.contains c then stack := c :: stack
  logInfo m!"{root}: {out.qsort (fun a b => a.toString < b.toString)}"
