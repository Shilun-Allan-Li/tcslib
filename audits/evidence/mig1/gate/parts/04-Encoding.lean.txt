/-- **The diagonal pairing machine** (landing: `Turing.pairDiagTM`, made public in place at
`TuringMachine/Encoding.lean:299`, body unchanged). A six-state machine with no work tapes that,
on input `α`, writes each symbol of `α` twice, then the separator `false, true`, rewinds the
input, and copies `α` once more, so that it outputs `pairEncode α α`. It is also the witness of
`Turing.FinTM.computesFunInTime_pairDup` (consolidation: Primitives' private `pairDupTM`, which
computes the same function, retires). [AB09, proof of Theorem 1.11: the self-application input
of the `HALT` reduction] -/
-- @@verbatim Encoding - --public --nodoc pairDiagTM

/-- **The diagonal pairing machine's time** (landing: `Turing.pairDiagTM_computes`, new in
`Encoding` before `computesFunInTime_pairEncode_diag`). On input `α` the diagonal pairing
machine halts within `4|α| + 5` steps with output `pairEncode α α`.

**Proof sketch.** The `have h` of today's `computesFunInTime_pairEncode_diag`
(`Encoding.lean:491`), from the private `pairDiag_run`: the doubling pass takes `2|α|` steps,
the two separator steps and the first left move `3`, the rewind `|α| + 1`, the copy `|α|`, and
the halting transition `1`. -/
theorem pairDiagTM_computes (x : List Bool) :
    pairDiagTM.ComputesInTime x (pairEncode x x) (4 * x.length + 5) := by
  sorry

/-- **The code normal form of a one-work-tape machine** (landing: `Turing.FinTM.toCodeTM`, new
in `Encoding` before `exists_codeTM`; the body is the witness term of today's `exists_codeTM`,
`Encoding.lean:618-627`). For a binary machine `M` with one work tape, the coded machine with
`|Q| - 1` as its state index bound, whose transition function is `M`'s transported along the
canonical bijection `Q ≃ Fin |Q|` (`Fintype.equivFinOfCardEq`; `Q` is inhabited by the start
state). [AB09, §1.4] Noncomputable, through `Fintype.equivFinOfCardEq`. -/
noncomputable def FinTM.toCodeTM : (M : FinTM Bool) → M.k = 1 → CodeTM
  | @FinTM.mk _ _ Q hQ dQ tm, rfl =>
    letI : Fintype Q := hQ
    letI : DecidableEq Q := dQ
    have hcard : Fintype.card Q = (Fintype.card Q - 1) + 1 := by
      have : 0 < Fintype.card Q := Fintype.card_pos_iff.mpr ⟨tm.q₀⟩
      omega
    let e := Fintype.equivFinOfCardEq hcard
    ⟨Fintype.card Q - 1, tm.relabelState e⟩

/-- **The code normal form is equivalent** (landing: `Turing.FinTM.toCodeTM_computesInTime_iff`,
new in `Encoding`; proved, the row's body moved). For a one-work-tape binary machine `M`, its
code normal form halts on `x` with output `w` within `t` steps if and only if `M` does.
[AB09, §1.4]

**Proof sketch.** The relabelled machine's run is the image of `M`'s run under the state
bijection (`Turing.MultiTapeTM.relabelState_runFrom_init`); halting and output are preserved. -/
theorem FinTM.toCodeTM_computesInTime_iff (M : FinTM Bool) (hk : M.k = 1)
    (x output : List Bool) (t : ℕ) :
    (M.toCodeTM hk).toFinTM.ComputesInTime x output t ↔ M.ComputesInTime x output t := by
  rcases M with @⟨k, Q, hQ, dQ, tm⟩
  dsimp only at hk
  subst k
  simp only [FinTM.toCodeTM, CodeTM.toFinTM, FinTM.ComputesInTime,
    MultiTapeTM.ComputesInTimeAndSpace, MultiTapeTM.relabelState_runFrom_init, Cfg.mapState,
    Option.map_eq_none_iff]
  constructor
  · rintro ⟨s, hhalt, hout, -⟩
    exact ⟨_, hhalt, hout, rfl⟩
  · rintro ⟨s, hhalt, hout, -⟩
    exact ⟨_, hhalt, hout, rfl⟩

theorem MIG1Gate.row_exists_codeTM : type_of% @Turing.exists_codeTM :=
