/-- **The copy endpoint from a canonical seam** (landing: `Turing.Cfg.ofWords_copy_endpoint`, new
in `Build/Catalog.lean`, Part 1, after `copyTM_run_ofCfg`). If the destination word of the seam
`Cfg.ofWords q w` is empty, then the framed copy endpoint of `Turing.copyTM_run_ofCfg` started
there (state `q'`, the destination interval overwritten with `w src`, every other cell, head and
the output unchanged) is the seam `Cfg.ofWords q' (Function.update w dst (w src))`.

**Proof sketch.** Today's `Cfg.ext` computation in the proof of `Turing.copyTM_run`
(`Build/Catalog.lean:1382`): at the origin heads of `Cfg.ofWords` the destination interval is
`[0, |w src|)`, which `bufferTape (w src)` fills exactly, and outside it the blank destination
word agrees with `bufferTape (w src)`'s blanks; every other tape is untouched. -/
theorem Cfg.ofWords_copy_endpoint {k : ℕ} {S : Type*} {x : List Bool} (q q' : S)
    (w : Fin k → List Bool) (src dst : Fin k) (hdst : w dst = []) :
    ({ (Cfg.ofWords (input := x) q w) with
        state := some q'
        workTapes := fun j p =>
          if j = dst ∧ (Cfg.ofWords (input := x) q w).workTapePos j ≤ p ∧
              p < (Cfg.ofWords (input := x) q w).workTapePos j + ((w src).length : ℤ)
            then FinTM.bufferTape (w src) (p - (Cfg.ofWords (input := x) q w).workTapePos j)
          else (Cfg.ofWords (input := x) q w).workTapes j p } : Cfg k Bool S x) =
      Cfg.ofWords q' (Function.update w dst (w src)) := by
  apply Cfg.ext <;> simp [Cfg.ofWords]
  funext j q
  by_cases hj : j = dst
  · subst j
    simp only [Function.update_self, hdst, FinTM.bufferTape_nil, true_and]
    split_ifs with hq
    · rfl
    · by_cases h0 : 0 ≤ q
      · simp [FinTM.bufferTape, h0, List.getElem?_eq_none]
        omega
      · simp [FinTM.bufferTape, h0]
  · simp [hj]

/-- **The transfer endpoint from a canonical seam** (landing:
`Turing.Cfg.ofWords_transfer_endpoint`, new in `Build/Catalog.lean`, Part 1, after
`transferTM_run_ofCfg`). If `src ≠ dst` and the destination word of the seam `Cfg.ofWords q w` is
empty, then the framed transfer endpoint of `Turing.transferTM_run_ofCfg` started there (state
`q'`, the source interval blanked, the destination interval overwritten with `w src`, everything
else unchanged) is the seam
`Cfg.ofWords q' (Function.update (Function.update w src []) dst (w src))`.

**Proof sketch.** Today's `Cfg.ext` computation in the proof of `Turing.transferTM_run`
(`Build/Catalog.lean:1297`): as the copy endpoint, with the source interval `[0, |w src|)`
blanked, which together with the blanks of `bufferTape (w src)` outside it is
`bufferTape []`. -/
theorem Cfg.ofWords_transfer_endpoint {k : ℕ} {S : Type*} {x : List Bool} (q q' : S)
    (w : Fin k → List Bool) (src dst : Fin k) (hne : src ≠ dst) (hdst : w dst = []) :
    ({ (Cfg.ofWords (input := x) q w) with
        state := some q'
        workTapes := fun j p =>
          if j = src ∧ (Cfg.ofWords (input := x) q w).workTapePos j ≤ p ∧
              p < (Cfg.ofWords (input := x) q w).workTapePos j + ((w src).length : ℤ)
            then none
          else if j = dst ∧ (Cfg.ofWords (input := x) q w).workTapePos j ≤ p ∧
              p < (Cfg.ofWords (input := x) q w).workTapePos j + ((w src).length : ℤ)
            then FinTM.bufferTape (w src) (p - (Cfg.ofWords (input := x) q w).workTapePos j)
          else (Cfg.ofWords (input := x) q w).workTapes j p } : Cfg k Bool S x) =
      Cfg.ofWords q' (Function.update (Function.update w src []) dst (w src)) := by
  apply Cfg.ext <;> simp [Cfg.ofWords]
  funext j q
  by_cases hs : j = src
  · subst j
    simp only [Function.update_of_ne hne, Function.update_self,
      FinTM.bufferTape_nil, hne, false_and, ite_false, true_and]
    split_ifs with hq
    · rfl
    · by_cases h0 : 0 ≤ q
      · simp [FinTM.bufferTape, h0, List.getElem?_eq_none]
        omega
      · simp [FinTM.bufferTape, h0]
  · by_cases hd : j = dst
    · subst j
      simp only [Function.update_self, hdst, FinTM.bufferTape_nil,
        hs, false_and, ite_false, true_and]
      split_ifs with hq
      · rfl
      · by_cases h0 : 0 ≤ q
        · simp [FinTM.bufferTape, h0, List.getElem?_eq_none]
          omega
        · simp [FinTM.bufferTape, h0]
    · simp [hs, hd]
