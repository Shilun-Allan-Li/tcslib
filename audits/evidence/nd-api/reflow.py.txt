import re, sys
MARK = 'the ND inventory API, 2026-10-17)'
def tokens(s):
    out, i = [], 0
    parts = re.split(r'(`[^`]*`)', s)
    cur = ''
    for p in parts:
        if p.startswith('`'):
            cur += p
        else:
            ws = p.split(' ')
            cur += ws[0]
            for w in ws[1:]:
                out.append(cur); cur = w
    out.append(cur)
    return [t for t in out if t != '']
def wrap(text, width=100):
    lines, cur = [], ''
    for t in tokens(text):
        if cur and len(cur) + 1 + len(t) > width:
            lines.append(cur); cur = t
        else:
            cur = t if not cur else cur + ' ' + t
    lines.append(cur)
    if lines[-1] == '-/' and len(lines) > 1:
        last = lines[-2].rsplit(' ', 1)
        if len(last) == 2:
            lines[-2] = last[0]; lines[-1] = last[1] + ' -/'
    return lines
n = 0
for p in sys.argv[1:]:
    t = open(p, encoding='utf-8').read()
    def fix(m):
        global n
        block = m.group(0)
        if MARK not in block: return block
        opener, body = block[:4], block[4:-3]
        assert opener in ('/-- ', '/-! ') and block.endswith(' -/') or block.endswith('-/'), block[:40]
        body = block[4:]
        assert body.endswith('-/'); body = body[:-2].rstrip()
        flat = ' '.join(l.strip() for l in body.split('\n'))
        new = '\n'.join(wrap(opener + flat + ' -/'))
        n += 1
        return new
    t2 = re.sub(r'/-[-!] .*?-/', fix, t, flags=re.S)
    open(p, 'w', encoding='utf-8').write(t2)
print('reflowed', n)
