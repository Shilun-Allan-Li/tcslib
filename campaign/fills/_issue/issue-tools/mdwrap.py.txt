import re, sys
def spaces_outside_ticks(s):
    out, inside = [], False
    for i, ch in enumerate(s):
        if ch == '`': inside = not inside
        elif ch == ' ' and not inside: out.append(i)
    return out
def wrap_line(line, width=100):
    m = re.match(r'^(\s*)(- |\d+\. )?', line)
    indent = ' ' * (len(m.group(1)) + len(m.group(2) or ''))
    res = []
    while len(line) > width:
        cands = [i for i in spaces_outside_ticks(line) if len(indent) < i <= width]
        if not cands: break
        i = cands[-1]
        res.append(line[:i]); line = indent + line[i+1:]
    res.append(line)
    return res
for p in sys.argv[1:]:
    lines = open(p, encoding='utf-8').read().split('\n'); out = []; fence = False
    for l in lines:
        if l.lstrip().startswith('```'): fence = not fence; out.append(l); continue
        if fence or l.startswith('|') or l.startswith('#') or len(l) <= 100: out.append(l); continue
        out += wrap_line(l)
    open(p, 'w', encoding='utf-8').write('\n'.join(out))
    left = [(i+1, len(l)) for i, l in enumerate(out) if len(l) > 100 and not l.startswith('#') and not l.startswith('|')]
    print(p.split('/')[-1], 'remaining over 100:', left)
