    guessCtlRunTM P pexit 0 q₀ δ stop b V qe a r =
      NDTM.phaseHostND (guessCtlRunPhases P 0 q₀ δ stop b V)

/-! The three host controls whose statements need `CtlState` to reduce, sealed without it: each
should fail at its `rfl`, not at elaboration. -/
section SealedNoCtlState
attribute [local irreducible] Turing.guessCtlRunTM Turing.guessCtlRunPhases Turing.guessCtlRunRed

-- guessCtlRunTM (with the start state written out)
example {SP SQ Q : Type} [DecidableEq SP] [DecidableEq SQ] (P : MultiTapeTM 2 Bool SP) (pexit : SP) (q₀ : Q) (δ : Q → Bool → Q) (stop : Q → Bool) (b : Bool) (V : MultiTapeTM 2 Bool SQ) (qe a r : SQ) :
    guessCtlRunTM P pexit 0 q₀ δ stop b V qe a r =
      NDTM.phaseHostND (guessCtlRunPhases P 0 q₀ δ stop b V) ⟨.pre, P.q₀⟩ (guessCtlRunRed pexit q₀ qe a r) := rfl

-- guessCtlRunPhases
example {SP SQ Q : Type} (P : MultiTapeTM 2 Bool SP) (q₀ : Q) (δ : Q → Bool → Q) (stop : Q → Bool) (b : Bool) (V : MultiTapeTM 2 Bool SQ) :
    guessCtlRunPhases P 0 q₀ δ stop b V .skip = (skipTM 2 0).toNDTM := rfl

-- guessCtlRunRed

end SealedNoCtlState
