import TCSlib.Complexity.TuringMachine.NDCodes
import TCSlib.Complexity.TuringMachine.Build.NDHost
import TCSlib.Complexity.TuringMachine.Build.StepBlock

/-! Review probe: a one-sweep verifier in the route's shape, composed by `Turing.phaseHostTM`
(sweep 0 as the scan fold, then a verdict park), meets the forward lemma's `hacc`/`hfirst`/`hnil`
and the backward lemma's `hV` shape from the entry configuration, through published lemmas only
(sealed). -/

open Turing

/-- The toy verifier's phases: `false` the sweep, `true` the verdict park. -/
def rvPh (A : Type) : Bool → Type
  | false => ScanPhase A
  | true => Bool

/-- The verdict park: stays at its state. -/
def rvPark : MultiTapeTM 2 Bool Bool := ⟨true, fun q _ _ => ⟨0, fun _ => (none, 0), none, some q⟩⟩

/-- The toy verifier's phase machines. -/
def rvM {A : Type} (δ : A → Bool → A) (a₀ : A) : (i : Bool) → MultiTapeTM 2 Bool (rvPh A i)
  | false => scanTM 2 0 (scanFold δ) scanKeep a₀
  | true => rvPark

/-- The toy verifier's redirects: the sweep's verdict `done a` to the park at `pass a`. -/
def rvRed {A : Type} (pass : A → Bool) : (i : Bool) → rvPh A i → Option (Σ i, rvPh A i)
  | false, .done a => some ⟨true, pass a⟩
  | false, .fwd _ => none
  | false, .bwd _ => none
  | true, _ => none

/-- The toy verifier. -/
def rvV {A : Type} (δ : A → Bool → A) (a₀ : A) (pass : A → Bool) :
    MultiTapeTM 2 Bool (Σ i, rvPh A i) :=
  phaseHostTM (rvM δ a₀) ⟨false, ScanPhase.fwd a₀⟩ (rvRed pass)

section RV
attribute [-simp] Turing.Action.apply Turing.Cfg.init Turing.MultiTapeTM.initCfg Turing.NDTM.initCfg
attribute [local irreducible] Turing.Cfg.mapState Turing.MultiTapeTM.runFrom Turing.phaseHostTM
  Turing.scanTM Turing.scanResult Turing.scanFold Turing.scanKeep Turing.streamStartCfg
  Turing.FinTM.bufferTape Turing.MultiTapeTM.step Turing.Action.apply Turing.Cfg.inputSymbol
  Turing.Cfg.workTapeSymbols

/-- A stream-start configuration with a tagged state is the tagged stream start. -/
theorem rv_tag {A : Type} (x s : List Bool) (q : ScanPhase A) :
    streamStartCfg x (0 : Fin 2) (⟨false, q⟩ : Σ i, rvPh A i) s =
      (streamStartCfg x (0 : Fin 2) (q : rvPh A false) s).mapState (Sigma.mk false) := by
  apply Cfg.ext
  · rw [Cfg.mapState_state, streamStartCfg_state, streamStartCfg_state]; rfl
  · rw [Cfg.mapState_inputPos, streamStartCfg_inputPos, streamStartCfg_inputPos]
  · funext i p; rw [Cfg.mapState_workTapes, streamStartCfg_workTapes, streamStartCfg_workTapes]
  · funext i; rw [Cfg.mapState_workTapePos, streamStartCfg_workTapePos, streamStartCfg_workTapePos]
  · rw [Cfg.mapState_output, streamStartCfg_output, streamStartCfg_output]

/-- **The toy verifier's first verdict.** From the entry configuration, it is at the park's
`pass (s.foldl δ a₀)` at exactly `2|s| + 3`, at neither verdict before, with an empty output. -/
theorem rv_V_verdict {A : Type} (δ : A → Bool → A) (a₀ : A) (pass : A → Bool) (x s : List Bool) :
    let c := streamStartCfg x (0 : Fin 2) (⟨false, ScanPhase.fwd a₀⟩ : Σ i, rvPh A i) s
    ((rvV δ a₀ pass).runFrom c (2 * s.length + 3)).state =
        some ⟨true, pass (s.foldl δ a₀)⟩ ∧
      (∀ u < 2 * s.length + 3, ((rvV δ a₀ pass).runFrom c u).state ≠ some ⟨true, true⟩ ∧
        ((rvV δ a₀ pass).runFrom c u).state ≠ some ⟨true, false⟩) ∧
      ((rvV δ a₀ pass).runFrom c (2 * s.length + 3)).output = [] := by
  intro c
  have hc : c = (streamStartCfg x 0 (ScanPhase.fwd a₀ : rvPh A false) s).mapState
      (Sigma.mk false) := rv_tag x s _
  -- the sweep is the scan fold
  have hw : ∀ p : ℤ, -1 ≤ p → p ≤ (s.length : ℤ) →
      (streamStartCfg x (0 : Fin 2) (ScanPhase.fwd a₀ : rvPh A false) s).workTapes 0
        ((streamStartCfg x (0 : Fin 2) (ScanPhase.fwd a₀ : rvPh A false) s).workTapePos 0 + p) =
          FinTM.bufferTape s p := by
    intro p _ _
    rw [streamStartCfg_workTapes, streamStartCfg_workTapePos, if_pos rfl, zero_add]
  obtain ⟨h1, h2, -⟩ := scanTM_run_ofCfg (k := 2) (0 : Fin 2) (scanFold δ) scanKeep a₀ s
    (streamStartCfg x 0 (ScanPhase.fwd a₀ : rvPh A false) s) (streamStartCfg_state _ _ _ _) hw
  rw [scanResult_fold] at h1
  obtain ⟨hpre, hend⟩ := phaseHostTM_run (rvM δ a₀) ⟨false, ScanPhase.fwd a₀⟩ (rvRed pass) false
    (streamStartCfg x 0 (ScanPhase.fwd a₀ : rvPh A false) s) _ (2 * s.length + 2) h1
    (ScanPhase.done (s.foldl δ a₀)) rfl ⟨true, pass (s.foldl δ a₀)⟩ rfl (by
      intro u hu s' hs'
      cases s' with
      | done a => exact absurd hs' (h2 u hu a)
      | fwd a => rfl
      | bwd a => rfl)
  refine ⟨?_, ?_, ?_⟩
  · show ((phaseHostTM (rvM δ a₀) ⟨false, ScanPhase.fwd a₀⟩ (rvRed pass)).runFrom c
      (2 * s.length + 2 + 1)).state = _
    rw [hc, hend, Cfg.mapState_state]
    rfl
  · intro u hu
    have hu' : u ≤ 2 * s.length + 2 := by omega
    show ((phaseHostTM (rvM δ a₀) ⟨false, ScanPhase.fwd a₀⟩ (rvRed pass)).runFrom c u).state ≠ _ ∧
      ((phaseHostTM (rvM δ a₀) ⟨false, ScanPhase.fwd a₀⟩ (rvRed pass)).runFrom c u).state ≠ _
    rw [hc, hpre u hu', Cfg.mapState_state]
    constructor <;> intro h <;>
      rcases hst : ((rvM δ a₀ false).runFrom
        (streamStartCfg x 0 (ScanPhase.fwd a₀ : rvPh A false) s) u).state with _ | q <;>
      rw [hst] at h <;> simp at h
  · show ((phaseHostTM (rvM δ a₀) ⟨false, ScanPhase.fwd a₀⟩ (rvRed pass)).runFrom c
      (2 * s.length + 2 + 1)).output = _
    rw [hc, hend, Cfg.mapState_output]
    show (streamStartCfg x 0 (ScanPhase.fwd a₀ : rvPh A false) s).output = []
    rw [streamStartCfg_output]

end RV
