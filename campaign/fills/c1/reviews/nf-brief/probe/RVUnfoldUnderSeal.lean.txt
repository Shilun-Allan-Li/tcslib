import TCSlib.Complexity.ClassNP.NTIME
import TCSlib.Complexity.TuringMachine.NDCodes
import TCSlib.Complexity.TuringMachine.Build.NDHost
import TCSlib.Complexity.TuringMachine.Build.StepBlock

/-! Review: does the seal (`local irreducible`) stop an explicit `unfold`/`simp only [·]` of
`MultiTapeTM.step` and `Action.apply`? (It should not: those use equation lemmas, which is why
the brief pairs the seal with an inspection table.) -/

open Turing

section S
attribute [-simp] Turing.Action.apply
attribute [local irreducible] Turing.MultiTapeTM.step Turing.Action.apply

-- explicit unfold of step: expected to succeed under the seal
example {S : Type} {x : List Bool} (M : MultiTapeTM 2 Bool S) (p : Fin (x.length + 2))
    (t : Fin 2 → ℤ → Option Bool) (h : Fin 2 → ℤ) (o : List Bool) :
    M.step (⟨none, p, t, h, o⟩ : Cfg 2 Bool S x) = ⟨none, p, t, h, o⟩ := by
  unfold MultiTapeTM.step
  rfl

-- simp only [Action.apply]: expected to succeed under the seal
example {S : Type} {x : List Bool} (a : Action 2 Bool S) (c : Cfg 2 Bool S x) :
    (a.apply c).state = a.state := by
  simp only [Action.apply]

-- the defeq closure: expected to FAIL under the seal
example {S : Type} {x : List Bool} (a : Action 2 Bool S) (c : Cfg 2 Bool S x) :
    (a.apply c).state = a.state := rfl
end S
