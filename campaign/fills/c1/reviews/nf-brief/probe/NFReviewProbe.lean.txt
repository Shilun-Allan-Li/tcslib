/-- The route probe's packaging, verbatim (`nfr_pack`). -/
theorem rv_pack {S : Type} [Fintype S] (H : NDTM 2 Bool S) :
    ∃ M : CodeNDTM, ∀ (x : List Bool) (t : ℕ), M.toFinNDTM.AcceptsWithin x t ↔
      ∃ w : List Bool, w.length = t ∧ (H.runWith w (H.initCfg x)).state = none ∧
        (H.runWith w (H.initCfg x)).output = [true] := by
  have hpos : 0 < Fintype.card S := Fintype.card_pos_iff.mpr ⟨H.q₀⟩
  let e : S ≃ Fin (Fintype.card S - 1 + 1) :=
    (Fintype.equivFin S).trans (finCongr (by omega))
  refine ⟨⟨Fintype.card S - 1, H.relabelState e⟩, fun x t => ?_⟩
  rw [FinNDTM.acceptsWithin_iff]
  simp only [CodeNDTM.toFinNDTM_tm, CodeNDTM.toFinNDTM_k, CodeNDTM.toFinNDTM_State]
  have hinit : (H.relabelState e).initCfg x = (H.initCfg x).mapState e := by
    rw [NDTM.initCfg_eq, NDTM.initCfg_eq, NDTM.relabelState_q₀]
    apply Cfg.ext
    · rw [Cfg.init_state, Cfg.mapState_state, Cfg.init_state]; rfl
    · rw [Cfg.init_inputPos, Cfg.mapState_inputPos, Cfg.init_inputPos]
    · funext i p; rw [Cfg.mapState_workTapes, Cfg.init_workTapes, Cfg.init_workTapes]
    · funext i; rw [Cfg.mapState_workTapePos, Cfg.init_workTapePos, Cfg.init_workTapePos]
    · rw [Cfg.mapState_output, Cfg.init_output, Cfg.init_output]
  rw [hinit]
  constructor
  · rintro ⟨w, hw, hh, ho⟩
    refine ⟨w, hw, ?_, ?_⟩
    · rw [NDTM.relabelState_runWith, Cfg.mapState_state] at hh
      exact Option.map_eq_none_iff.mp hh
    · rwa [NDTM.relabelState_runWith, Cfg.mapState_output] at ho
  · rintro ⟨w, hw, hh, ho⟩
    refine ⟨w, hw, ?_, ?_⟩
    · rw [NDTM.relabelState_runWith, Cfg.mapState_state, hh]; rfl
    · rwa [NDTM.relabelState_runWith, Cfg.mapState_output]

/-- The route probe's trivial setup, verbatim (`nfr_setup`). -/
theorem rv_setup {SP : Type} (P : MultiTapeTM 2 Bool SP) (z : List Bool) (j : Fin 2) :
    P.runFrom (P.initCfg z) 0 = P.initCfg z ∧ (P.initCfg z).state = some P.q₀ ∧
      (∀ u < 0, (P.runFrom (P.initCfg z) u).state ≠ some P.q₀) ∧
      (∀ p : ℕ, (P.initCfg z).workTapes j ((P.initCfg z).workTapePos j + p) = none) ∧
      (P.initCfg z).workTapes j ((P.initCfg z).workTapePos j - 1) = none := by
  refine ⟨MultiTapeTM.runFrom_zero, ?_, fun u hu => absurd hu (Nat.not_lt_zero u), fun p => ?_, ?_⟩
  · rw [MultiTapeTM.initCfg_eq, Cfg.init_state]
  · rw [MultiTapeTM.initCfg_eq, Cfg.init_workTapes]
  · rw [MultiTapeTM.initCfg_eq, Cfg.init_workTapes]

/-- A trivial setup: one state, its own exit. -/
def rvP : MultiTapeTM 2 Bool Unit := ⟨(), fun _ _ _ => ⟨0, fun _ => (none, 0), none, some ()⟩⟩

/-- **The target from the host's interface.** For any verifier `V` and controller, the
packaging, the trivial setup and the two acceptance lemmas give the target's statement from
three obligations: the forward display with its budget `0 + 2|s| + TQ + 11 ≤ C (t + 1)`, the
backward lemma's totality `hV` (stated per input), and "a passing stopped stream is an accepting
run". Nothing at the host level is composed. -/
theorem rv_skeleton (N : FinNDTM Bool) {SQ Q : Type} [Fintype SQ] [DecidableEq SQ] [Fintype Q]
    (V : MultiTapeTM 2 Bool SQ) (qentry acc rej : SQ) (q₀ : Q) (δ : Q → Bool → Q)
    (stop : Q → Bool) (C : ℕ)
    (hfwd : ∀ (x : List Bool) (t : ℕ), N.AcceptsWithin x t → ∃ s : List Bool,
      (∀ i < s.length, stop ((s.take i).foldl δ q₀) = false) ∧ stop (s.foldl δ q₀) = true ∧
      ∃ TQ : ℕ, (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) TQ).state = some acc ∧
        (∀ u < TQ, (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) u).state ≠ some acc ∧
          (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) u).state ≠ some rej) ∧
        (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) TQ).output = [] ∧
        0 + 2 * s.length + TQ + 11 ≤ C * (t + 1))
    (hV : ∀ (x s : List Bool), (∀ i < s.length, stop ((s.take i).foldl δ q₀) = false) →
      stop (s.foldl δ q₀) = true → ∃ TQ : ℕ,
        ((V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) TQ).state = some acc ∨
          (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) TQ).state = some rej) ∧
        ∀ u < TQ, (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) u).state ≠ some acc ∧
          (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) u).state ≠ some rej)
    (hbwd : ∀ (x s : List Bool), (∀ i < s.length, stop ((s.take i).foldl δ q₀) = false) →
      stop (s.foldl δ q₀) = true → ∀ TQ : ℕ,
        (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) TQ).state = some acc →
        (∀ u < TQ, (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) u).state ≠ some acc ∧
          (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) u).state ≠ some rej) →
        (V.runFrom (guessRunCfg (rvP.initCfg x) 0 s qentry) TQ).output = [] →
        ∃ t', N.AcceptsWithin x t') :
    ∃ (M : CodeNDTM) (C : ℕ),
      (∀ (x : List Bool) (t : ℕ),
        N.AcceptsWithin x t → M.toFinNDTM.AcceptsWithin x (C * (t + 1))) ∧
      (∀ (x : List Bool) (t : ℕ),
        M.toFinNDTM.AcceptsWithin x t → ∃ t', N.AcceptsWithin x t') := by
  obtain ⟨M, hM⟩ := rv_pack (guessCtlRunTM rvP rvP.q₀ 0 q₀ δ stop false V qentry acc rej)
  refine ⟨M, C, fun x t hN => ?_, fun x t hMt => ?_⟩
  · obtain ⟨s, hl, hs, TQ, hacc, hfirst, hnil, hB⟩ := hfwd x t hN
    obtain ⟨h0, hst, hfi, hbl, hle⟩ := rv_setup rvP x 0
    obtain ⟨w, hw, hh, ho⟩ := guessCtlRunTM_accepts_of rvP rvP.q₀ 0 q₀ δ stop V qentry acc rej
      0 (rvP.initCfg x) h0 hst hfi hbl hle s hl hs TQ hacc hfirst hnil (C * (t + 1)) hB
    exact (hM x _).2 ⟨w, hw, hh, ho⟩
  · obtain ⟨w, -, hh, ho⟩ := (hM x t).1 hMt
    obtain ⟨h0, hst, hfi, hbl, hle⟩ := rv_setup rvP x 0
    obtain ⟨s, hl, hs, TQ, hacc, hfirst, hnil⟩ := guessCtlRunTM_stream_of_accepts rvP rvP.q₀ 0
      q₀ δ stop V qentry acc rej 0 (rvP.initCfg x) h0 hst hfi hbl hle (hV x) w hh ho
    exact hbwd x s hl hs TQ hacc hfirst hnil

/-- **The verifier's entry is the stream start.** After the trivial setup, the run's entry
configuration is `streamStartCfg x 0 q s`: state `q`, the input head at `1`, `s` on tape `0`
from cell `0` (blank elsewhere), tape `1` blank, both heads at `0`, empty output. -/
theorem rv_entry {SP S : Type} (P : MultiTapeTM 2 Bool SP) (x s : List Bool) (q : S) :
    guessRunCfg (P.initCfg x) 0 s q = streamStartCfg x 0 q s := by
  rw [MultiTapeTM.initCfg_eq]
  apply Cfg.ext
  · rw [guessRunCfg_state, streamStartCfg_state]
  · rw [guessRunCfg_inputPos, streamStartCfg_inputPos, Cfg.init_inputPos]
  · funext i p
    rw [guessRunCfg_workTapes, streamStartCfg_workTapes, Cfg.init_workTapePos]
    by_cases hi : i = 0
    · subst hi
      rw [Function.update_self, if_pos rfl]
      by_cases hp : (0 : ℤ) ≤ p ∧ p < 0 + (s.length : ℤ)
      · rw [if_pos hp, sub_zero]
      · rw [if_neg hp, Cfg.init_workTapes]
        by_cases hneg : p < 0
        · rw [FinTM.bufferTape_of_neg s hneg]
        · rw [FinTM.bufferTape_eq, if_pos (by omega)]
          exact (List.getElem?_eq_none (by omega)).symm
    · rw [Function.update_of_ne hi, if_neg hi, Cfg.init_workTapes]
  · funext i
    rw [guessRunCfg_workTapePos, streamStartCfg_workTapePos, Cfg.init_workTapePos]
  · rw [guessRunCfg_output, streamStartCfg_output, Cfg.init_output]

/-- **Sweep 0 is the published scan fold.** From the entry configuration with the scan's
forward state, the fold instance of `Turing.scanTM` on tape `0` reaches `done (s.foldl δ a₀)` at
exactly `2|s| + 2`, not at any `done` before, with every tape, head, the input head and the
output as at the entry: A's head is back at `0` with no separate rewind. -/
theorem rv_sweep0 {A : Type} (δ : A → Bool → A) (a₀ : A) (x s : List Bool) :
    (scanTM 2 0 (scanFold δ) scanKeep a₀).runFrom (streamStartCfg x 0 (ScanPhase.fwd a₀) s)
        (2 * s.length + 2) = streamStartCfg x 0 (ScanPhase.done (s.foldl δ a₀)) s ∧
      ∀ t < 2 * s.length + 2, ∀ a,
        ((scanTM 2 0 (scanFold δ) scanKeep a₀).runFrom
          (streamStartCfg x 0 (ScanPhase.fwd a₀) s) t).state ≠ some (ScanPhase.done a) := by
  have hw : ∀ p : ℤ, -1 ≤ p → p ≤ (s.length : ℤ) →
      (streamStartCfg x (0 : Fin 2) (ScanPhase.fwd a₀) s).workTapes 0
        ((streamStartCfg x (0 : Fin 2) (ScanPhase.fwd a₀) s).workTapePos 0 + p) =
          FinTM.bufferTape s p := by
    intro p _ _
    rw [streamStartCfg_workTapes, streamStartCfg_workTapePos, if_pos rfl, zero_add]
  obtain ⟨h1, h2, -⟩ := scanTM_run_ofCfg (k := 2) (0 : Fin 2) (scanFold δ) scanKeep a₀ s
    (streamStartCfg x 0 (ScanPhase.fwd a₀) s) (streamStartCfg_state _ _ _ _) hw
  refine ⟨?_, h2⟩
  rw [h1, scanResult_fold]
  apply Cfg.ext
  · rw [streamStartCfg_state]
  · rw [streamStartCfg_inputPos, streamStartCfg_inputPos]
  · funext j q
    show (if j = 0 ∧ (streamStartCfg x 0 (ScanPhase.fwd a₀) s).workTapePos j ≤ q ∧
        q < (streamStartCfg x 0 (ScanPhase.fwd a₀) s).workTapePos j + (s.length : ℤ)
        then FinTM.bufferTape s (q - (streamStartCfg x 0 (ScanPhase.fwd a₀) s).workTapePos j)
        else (streamStartCfg x 0 (ScanPhase.fwd a₀) s).workTapes j q) = _
    rw [streamStartCfg_workTapes, streamStartCfg_workTapes, streamStartCfg_workTapePos]
    split_ifs with h hj <;> simp_all
  · funext j
    rw [streamStartCfg_workTapePos, streamStartCfg_workTapePos]
  · rw [streamStartCfg_output, streamStartCfg_output]

/-- **The input read depends on the input head only**, across machines with any work tapes and
states: the reading the input sweep needs to transfer a checked claim from the verifier's
configuration to `N`'s. -/
theorem rv_inputSymbol_congr {k k' : ℕ} {S S' : Type} {x : List Bool} (c : Cfg k Bool S x)
    (c' : Cfg k' Bool S' x) (h : c.inputPos.val = c'.inputPos.val) :
    c.inputSymbol = c'.inputSymbol := by
  by_cases h0 : c.inputPos.val = 0
  · have h0' : c'.inputPos = 0 := Fin.ext (by rw [← h]; exact h0)
    rw [Cfg.inputSymbol_of_zero (Fin.ext h0), Cfg.inputSymbol_of_zero h0']
  · by_cases he : c.inputPos.val = x.length + 1
    · rw [Cfg.inputSymbol_of_end he, Cfg.inputSymbol_of_end (by omega)]
    · have hlt := c.inputPos.isLt
      rw [inputSymbolInner (c.inputPos.val - 1) (by omega) (by omega),
        inputSymbolInner (cfg := c') (c.inputPos.val - 1) (by omega) (by omega)]

/-- **An input move changes the head by at most one cell**, read through the published boundary
and interior lemmas (`moveInputPos` sealed): the bound the ledger needs for the input head. -/
theorem rv_moveInputPos_le {n : ℕ} (p : Fin (n + 2)) (m : SignType) :
    (moveInputPos p m).val ≤ p.val + 1 ∧ p.val ≤ (moveInputPos p m).val + 1 := by
  rcases m with _ | _ | _
  · rw [show (SignType.zero : SignType) = 0 from rfl, moveInputPos_zero]; omega
  · by_cases h0 : p = 0
    · subst h0
      rw [show (SignType.neg : SignType) = -1 from rfl, moveInputPos_leftBoundary]; omega
    · rw [moveInputPos_neg_of_ne_left p h0]
      have : p.val ≠ 0 := fun h => h0 (Fin.ext h)
      show p.val - 1 ≤ p.val + 1 ∧ p.val ≤ p.val - 1 + 1
      omega
  · by_cases hr : p.val = n + 1
    · have hp : p = ⟨n + 1, by omega⟩ := Fin.ext hr
      rw [hp, show (SignType.pos : SignType) = 1 from rfl, moveInputPos_rightBoundary]
      show n + 1 ≤ n + 1 + 1 ∧ n + 1 ≤ n + 1 + 1
      omega
    · rw [moveInputPos_pos_of_ne_right p hr]
      show p.val + 1 ≤ p.val + 1 ∧ p.val ≤ p.val + 1 + 1
      omega

end RV
