open Lean Elab Command Meta in
elab "#kinds " s:str : command => do
  let env ← getEnv
  let names := (s.getString.splitOn " ").filter (· ≠ "") |>.map String.toName
  let st ← liftTermElabM getSimpTheorems
  for n in names do
    match env.find? n with
    | none => pure ()
    | some ci =>
      let kind := match ci with
        | .defnInfo _ => "def" | .thmInfo _ => "theorem" | .axiomInfo _ => "axiom"
        | .opaqueInfo _ => "opaque" | .quotInfo _ => "quot" | .inductInfo _ => "inductive"
        | .ctorInfo _ => "ctor" | .recInfo _ => "rec"
      let red ← getReducibilityStatus n
      let reds := match red with
        | .reducible => "reducible" | .semireducible => "semireducible" | .irreducible => "irreducible"
      let proj := (env.getProjectionFnInfo? n).isSome
      let inst := Lean.Meta.isInstanceCore env n
      let matcher := Lean.Meta.isMatcherCore env n
      let simpUnfold := st.isDeclToUnfold n
      let simpLemma := st.isLemma (.decl n)
      let mod := match env.getModuleIdxFor? n with
        | some i => (env.header.moduleNames[i.toNat]?).getD `unknown
        | none => `this
      logInfo m!"KIND {n}: {kind}; reducibility {reds}; projection {proj}; instance {inst}; matcher {matcher}; simp-unfold {simpUnfold}; simp-lemma {simpLemma}; module {mod}"
