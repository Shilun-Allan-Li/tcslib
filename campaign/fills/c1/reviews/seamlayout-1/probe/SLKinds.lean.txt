open Lean Meta Elab Command in
/-- Kind, reducibility, projection/instance/matcher flags, module and default-simp-set
membership of each constant (maintainer's read-only probe). -/
elab "#kinds " ids:ident* : command => do
  for id in ids do
    let n ← liftCoreM <| realizeGlobalConstNoOverload id
    let env ← getEnv
    let some ci := env.find? n | logInfo m!"{n}: not found"
    let kind := match ci with
      | .defnInfo _ => "def" | .thmInfo _ => "theorem" | .axiomInfo _ => "axiom"
      | .opaqueInfo _ => "opaque" | .quotInfo _ => "quot" | .inductInfo _ => "inductive"
      | .ctorInfo _ => "ctor" | .recInfo _ => "rec"
    let red ← liftCoreM <| getReducibilityStatus n
    let proj := (env.getProjectionFnInfo? n).isSome
    let inst := isInstanceCore env n
    let matcher := isMatcherCore env n
    let mod := (env.getModuleIdxFor? n).bind (env.header.moduleNames[·]?)
    let st ← liftTermElabM <| getSimpTheorems
    let unf := st.isDeclToUnfold n
    let eqs := st.isLemma (.decl (n ++ `eq_1))
    logInfo m!"KIND {n}: {kind}; reducibility {repr red}; projection {proj}; instance {inst}; matcher {matcher}; simpUnfold {unf}; simpEq1 {eqs}; module {mod}"
