-- S6: `nf_sweep_display_pass` is not vacuous: a toy machine (no work tape, one state) that halts
-- at its first step emitting `true` meets its three hypotheses on the one-bit word
def nfToyN : FinNDTM Bool where
  k := 0
  State := Unit
  tm := ⟨(), fun _ _ _ _ => ⟨0, fun i => i.elim0, some true, none⟩⟩

example (x : List Bool) :
    nf_sweepPass nfToyN
      ((nf_display nfToyN x [false]).foldl (nf_sweepStep nfToyN) (nf_sweepInit nfToyN)) = true := by
  have hinit : (nfToyN.tm.initCfg x).state = some () := by
    rw [NDTM.initCfg_eq, Cfg.init_state]; rfl
  refine nf_sweep_display_pass nfToyN x [false] ?_ ?_ ?_
  · intro r hr
    have : r = 0 := by simp at hr; omega
    subst this
    rw [List.take_zero, NDTM.runWith_nil, hinit]
    simp
  · rw [NDTM.runWith_cons, NDTM.runWith_nil, NDTM.stepWith_of_some hinit, Action.apply_state]
    rfl
  · rw [NDTM.runWith_cons, NDTM.runWith_nil, NDTM.stepWith_of_some hinit, Action.apply_output,
      NDTM.initCfg_eq, Cfg.init_output]
    rfl

-- S7: the display of that run is exactly one record, `false`, blank input read `00`... the
-- input is read at the input head; on `x = []` the read is the right blank
example : nf_display nfToyN [] [false] = [false, false, false, true] := by
  decide

-- S8: a stream whose claim is wrong still runs sweep 0 to its verdict (totality of the scan):
