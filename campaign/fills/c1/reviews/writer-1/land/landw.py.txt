import sys
base_p, v4_p, out_p = sys.argv[1:4]
base = open(base_p).read(); v4 = open(v4_p).read()
bl = base.split("\n"); vl = v4.split("\n")
# the helper block: v4 lines 188..312 (1-based), inserted after base line 187
helpers = vl[187:312]
assert helpers[0].startswith("/-- The configuration after writing the word"), helpers[0]
assert vl[312].startswith("/-- **The bounded choice writer's contract.**"), vl[312]
assert bl[187] == vl[312], (bl[187], vl[312])
# the proof body: v4 from the line after 'some .done := by' following the target, to the end of the proof
i_t = next(i for i, l in enumerate(vl) if l.startswith("theorem ndWriterTM_run_block"))
i_by = next(i for i in range(i_t, len(vl)) if vl[i].endswith("some .done := by"))
i_end = next(i for i in range(i_by + 1, len(vl)) if vl[i] == "")
body = vl[i_by + 1:i_end]
assert body[0].strip().startswith("have hm"), body[0]
b_t = next(i for i, l in enumerate(bl) if l.startswith("theorem ndWriterTM_run_block"))
b_by = next(i for i in range(b_t, len(bl)) if bl[i].endswith("some .done := by"))
assert bl[b_by + 1] == "  sorry" and bl[b_by + 2] == "", bl[b_by + 1:b_by + 3]
new = bl[:187] + helpers + bl[187:b_by + 1] + body + bl[b_by + 2:]
t = "\n".join(new)
def rep(old, new_):
    global t
    assert t.count(old) == 1, old[:60]
    t = t.replace(old, new_)
rep("* `Turing.ndWriterTM_run_block` — the bounded flavor's contract (sorried).",
    "* `Turing.ndWriterTM_run_block` — the bounded flavor's contract (proved).")
rep("  `Turing.guessRunTM_haltsWithin`, `Turing.guessRunTM_accepts_iff` — the guess-then-run assembly\n"
    "  (proved modulo `Turing.ndWriterTM_run_block`); `Turing.guessRunTM_runWith_setup_halt` — a\n"
    "  halting setup (proved).",
    "  `Turing.guessRunTM_haltsWithin`, `Turing.guessRunTM_accepts_iff` — the guess-then-run assembly\n"
    "  (proved); `Turing.guessRunTM_runWith_setup_halt` — a halting setup (proved).")
rep("Every theorem here is proved. The seven that pass the writer use `Turing.ndWriterTM_run_block`,\n"
    "its sorried contract, and so are proved modulo it; `Turing.guessRunTM_runWith_setup_halt` and\n"
    "`Turing.guessRunCfg_clean` do not use it.",
    "Every theorem here is proved. The writer's contract `Turing.ndWriterTM_run_block` is proved by\n"
    "one induction over the written prefix (C1-W, 2026-10-17); the seven theorems that pass the\n"
    "writer use it, and `Turing.guessRunTM_runWith_setup_halt` and `Turing.guessRunCfg_clean` do not.")
rep("writer's branch is `Turing.ndWriterTM_run_block` (sorried in this module, so the theorem is proved\nmodulo it), whose second clause is the guard.",
    "writer's branch is `Turing.ndWriterTM_run_block`, whose second clause is the guard.")
assert "Fill appendix" not in t and "sorry" not in t.split("theorem ndWriterTM_run_block")[1].split("\n\n")[0]
open(out_p, "w").write(t); print("ok", len(new), "lines")
