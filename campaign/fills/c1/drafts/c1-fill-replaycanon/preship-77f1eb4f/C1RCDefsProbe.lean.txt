import TCSlib.Complexity.TuringMachine.NDCodeSimulator

/-! C1, the replay canonizer: the brief's recommended definitions elaborate, and on two concrete
coded machines (one state, and two states with a choice-dependent table) `rc_g` maps the fixed
ND serialization to the replay serialization. An unsealed sanity check of the definitions'
shape, by evaluation (`#eval`); not a route step and not a proof. -/

namespace Turing

def rc_widen (r : List Bool) : List Bool :=
  r.take 10 ++ [false, false, true, false] ++ r.drop 10

def rc_round (n : ℕ) (p : List Bool × List Bool × List Bool) :
    List Bool × List Bool × List Bool :=
  match codeSkipActionK 2 n p.2.1, codeSkipActionK 2 n p.2.2 with
  | some f, some t =>
    let rf := p.2.1.take (p.2.1.length - f.length)
    let rt := p.2.2.take (p.2.2.length - t.length)
    (p.1 ++ (rc_widen rf ++ rc_widen rf ++ rc_widen rt), f, t)
  | _, _ => p

def rc_table (n : ℕ) (body : List Bool) : List Bool :=
  match codeSkipRepeat (codeSkipActionK 2 n) (27 * (n + 1)) body with
  | none => []
  | some tT => ((rc_round n)^[27 * (n + 1)] ([], body, tT)).1

def rc_g (xs : List Bool) : List Bool :=
  match pairDecode xs with
  | none => []
  | some (u, v) =>
    match codeReadUnary v with
    | none => []
    | some (_, body) => xs.take (xs.length - body.length) ++ rc_table (codeBitsNat u) body

/-- One state; the `true` slice writes and moves where the `false` slice halts. -/
def rcM1 : CodeNDTM := ⟨0, ⟨0, fun b _ inp w =>
  if b then ⟨1, fun i => (some (w i), if i = 0 then 1 else -1), inp, some 0⟩
  else ⟨0, fun _ => (none, 0), none, none⟩⟩⟩

/-- Two states, initial state `1`, the successor and the output depending on the choice. -/
def rcM2 : CodeNDTM := ⟨1, ⟨1, fun b q inp w =>
  ⟨if b then -1 else 1, fun i => (if q = 0 then some (w (i + 1)) else none, 0),
    if b then some true else w 0, if b then some (q + 1) else (if inp = none then none else some q)⟩⟩⟩

-- Evaluation, not proof (the definitions' shape on concrete machines): each line prints
-- `(true, |serialize|, |replay serialization|)`.
#eval (rc_g rcM1.serialize == serializeK rcM1.tm.replayTM, rcM1.serialize.length,
  (serializeK rcM1.tm.replayTM).length)
#eval (rc_g rcM2.serialize == serializeK rcM2.tm.replayTM, rcM2.serialize.length,
  (serializeK rcM2.tm.replayTM).length)
-- Off the serializations the value is irrelevant; for the record:
#eval (rc_g [], rc_g [true, true, false, true])

end Turing
