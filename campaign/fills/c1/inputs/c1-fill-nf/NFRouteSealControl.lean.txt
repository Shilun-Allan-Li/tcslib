/-! The controls for NF's route-probe seal: unsealed every goal closes; sealed every goal is
expected to FAIL, each at its own goal. -/

section Unsealed

-- CodeNDTM.toFinNDTM
example (M : Turing.CodeNDTM) : M.toFinNDTM.k = 2 := rfl

-- NDTM.relabelState
example {S S' : Type} (H : Turing.NDTM 2 Bool S) (e : S ≃ S') :
    (H.relabelState e).q₀ = e H.q₀ := rfl

-- NDTM.initCfg
example {S : Type} (H : Turing.NDTM 2 Bool S) (x : List Bool) :
    H.initCfg x = Turing.Cfg.init H.q₀ x := rfl

-- Cfg.init
example {S : Type} (q : S) (x : List Bool) :
    (Turing.Cfg.init q x : Turing.Cfg 2 Bool S x).inputPos = 1 := rfl

-- Cfg.mapState
example {S S' : Type} (f : S → S') (x : List Bool) (c : Turing.Cfg 2 Bool S x) :
    (c.mapState f).output = c.output := rfl

-- NDTM.runWith (a usage control: the empty word)
example {S : Type} (H : Turing.NDTM 2 Bool S) (x : List Bool) (c : Turing.Cfg 2 Bool S x) :
    H.runWith [] c = c := rfl

-- FinNDTM.AcceptsWithin
example (N : Turing.FinNDTM Bool) (x : List Bool) (t : ℕ) :
    N.AcceptsWithin x t = ∃ w : List Bool, w.length = t ∧
      (N.tm.runWith w (N.tm.initCfg x)).state = none ∧
      (N.tm.runWith w (N.tm.initCfg x)).output = [true] := rfl

-- MultiTapeTM.runFrom (a usage control: time zero)
example {S : Type} (P : Turing.MultiTapeTM 2 Bool S) (x : List Bool) (c : Turing.Cfg 2 Bool S x) :
    P.runFrom c 0 = c := rfl

-- MultiTapeTM.initCfg
example {S : Type} (P : Turing.MultiTapeTM 2 Bool S) (x : List Bool) :
    P.initCfg x = Turing.Cfg.init P.q₀ x := rfl

end Unsealed
