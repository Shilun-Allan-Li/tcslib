/-! U2, the `k`-tape universal machine: the pre-ship's route probe, **against the proposed API**.

The route reads two public definitions that have **no published lemma** (the style lint's
check 8 lists both): H2-F's encoding `Turing.StepBlock.forwardLoopCfg` (`Build/StepBlockForward`)
and H5's presentation `Turing.rulerHostCfg` (`Build/InputDevice`). The sections `Proposed` and
`ProposedRuler` state the field equations an API PR would publish beside them (what
`simps (config := { isSimp := false })` generates there; Lean refuses `simps` on another module's
declaration), proved here by `rfl` before the seal: they are **stand-ins**, not citations. With
them, the seam premises of both phases follow, sealed over 8 TCSlib definitions (the two
`@[simp]` ones among them, `MultiTapeTM.initCfg` and `Cfg.init`, also erased from `simp`):

* `u2_encode_init_tapes`, `u2_encode_init_fields`: **the startup's endpoint is the encoding.**
  `Turing.exists_selfStartupTMK`'s endpoint, on the forwarding loop's `k + 2` tapes, is
  `forwardLoopCfg entry M (M.initCfg _)`: the table on tape `k`, the unary initial state from
  cell `2` of tape `k + 1` (the blank summary cell is `bufferTape`'s left blank:
  `StepBlock.summaryCells_eq`, `FinTM.bufferTape_of_neg`), blank mirrored tapes, every head at
  `0`, the input head at `1`, an empty output, the control at `entry`;
* `u2_encode_halt`: **an encoded configuration halts exactly when the source does, with the
  source's output** (the forward ledger's endpoint and the converse);
* `u2_ruler_fields`: **the ruler host's presentation**: the physical head at `ℓ +` the source's,
  the ruler last, the source's bank first, the source's output.

Without the stand-ins each of these closes only by unfolding: `U2UESealControl.lean.txt`'s
controls for `forwardLoopCfg` and `rulerHostCfg` fail sealed. -/

namespace Turing

section Proposed

variable {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q) (M : MultiTapeTM k Bool (Fin (n + 1)))
  (c : Cfg k Bool (Fin (n + 1)) x)

theorem u2p_forwardLoopCfg_state :
    (StepBlock.forwardLoopCfg entry M c).state = c.state.map fun _ => entry := rfl
theorem u2p_forwardLoopCfg_inputPos :
    (StepBlock.forwardLoopCfg entry M c).inputPos = c.inputPos := rfl
theorem u2p_forwardLoopCfg_workTapes (i : Fin (k + 2)) :
    (StepBlock.forwardLoopCfg entry M c).workTapes i =
      Fin.addCases (motive := fun _ => ℤ → Option Bool) c.workTapes (fun j =>
        if j = 0 then FinTM.bufferTape (tableBitsK M.tr)
        else StepBlock.summaryCells none ((c.state.map unaryFin).getD [])) i := rfl
theorem u2p_forwardLoopCfg_workTapePos (i : Fin (k + 2)) :
    (StepBlock.forwardLoopCfg entry M c).workTapePos i =
      Fin.addCases (motive := fun _ => ℤ) c.workTapePos (fun _ => 0) i := rfl
theorem u2p_forwardLoopCfg_output :
    (StepBlock.forwardLoopCfg entry M c).output = c.output := rfl

end Proposed

section ProposedRuler

variable {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ) (hz : z.length = ℓ + y.length)
  (q : Option Q) (c : Cfg m Bool S y) (rt : ℤ → Option Bool) (rp : ℤ)

theorem u2p_rulerHostCfg_state : (rulerHostCfg ℓ hz q c rt rp).state = q := rfl
theorem u2p_rulerHostCfg_inputPos_val :
    (rulerHostCfg ℓ hz q c rt rp).inputPos.val = ℓ + c.inputPos.val := rfl
theorem u2p_rulerHostCfg_workTapes (i : Fin (m + 1)) :
    (rulerHostCfg ℓ hz q c rt rp).workTapes i = Fin.lastCases rt c.workTapes i := rfl
theorem u2p_rulerHostCfg_workTapePos (i : Fin (m + 1)) :
    (rulerHostCfg ℓ hz q c rt rp).workTapePos i = Fin.lastCases rp c.workTapePos i := rfl
theorem u2p_rulerHostCfg_output : (rulerHostCfg ℓ hz q c rt rp).output = c.output := rfl

end ProposedRuler

section U2Route

attribute [local irreducible] StepBlock.forwardLoopCfg rulerHostCfg MultiTapeTM.initCfg Cfg.init
  FinTM.bufferTape StepBlock.summaryCells tableBitsK unaryFin
attribute [-simp] MultiTapeTM.initCfg Cfg.init

/-- The startup's endpoint tapes, restricted to the forwarding loop's `k + 2`, are the encoding's
tapes at the decoded machine's initial configuration. -/
theorem u2_encode_init_tapes {k n : ℕ} {Q : Type} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (x : List Bool) (j : Fin (k + 2)) :
    (StepBlock.forwardLoopCfg entry M (M.initCfg x)).workTapes j =
      if j.val = k then FinTM.bufferTape (tableBitsK M.tr)
      else if j.val = k + 1 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
      else fun _ => none := by
  rw [u2p_forwardLoopCfg_workTapes]
  refine Fin.addCases (fun i => ?_) (fun j' => ?_) j
  · rw [Fin.addCases_left, MultiTapeTM.initCfg_eq]
    have h1 : (Fin.castAdd 2 i).val ≠ k := by simp only [Fin.coe_castAdd]; omega
    have h2 : (Fin.castAdd 2 i).val ≠ k + 1 := by simp only [Fin.coe_castAdd]; omega
    rw [if_neg h1, if_neg h2]
    funext p
    rw [Cfg.init_workTapes]
  · rw [Fin.addCases_right, MultiTapeTM.initCfg_eq, Cfg.init_state, Option.map_some,
      Option.getD_some]
    by_cases h : j' = 0
    · subst h
      have h1 : (Fin.natAdd k (0 : Fin 2)).val = k := by simp only [Fin.coe_natAdd]; omega
      rw [if_pos rfl, if_pos h1]
    · have hj : j' = 1 := by omega
      subst hj
      have h1 : (Fin.natAdd k (1 : Fin 2)).val ≠ k := by simp only [Fin.coe_natAdd]; omega
      have h2 : (Fin.natAdd k (1 : Fin 2)).val = k + 1 := by simp only [Fin.coe_natAdd]; omega
      rw [if_neg (by decide), if_neg h1, if_pos h2]
      funext p
      rw [StepBlock.summaryCells_eq]
      by_cases hp : p = 0
      · rw [if_pos hp, hp]
        exact (FinTM.bufferTape_of_neg _ (by omega)).symm
      · rw [if_neg hp]

/-- The encoding of the initial configuration is at `entry`, with input head `1`, every head at
`0` and an empty output. -/
theorem u2_encode_init_fields {k n : ℕ} {Q : Type} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (x : List Bool) :
    (StepBlock.forwardLoopCfg entry M (M.initCfg x)).state = some entry ∧
      (StepBlock.forwardLoopCfg entry M (M.initCfg x)).inputPos.val = 1 ∧
      (∀ j, (StepBlock.forwardLoopCfg entry M (M.initCfg x)).workTapePos j = 0) ∧
      (StepBlock.forwardLoopCfg entry M (M.initCfg x)).output = [] := by
  refine ⟨?_, ?_, fun j => ?_, ?_⟩
  · rw [u2p_forwardLoopCfg_state, MultiTapeTM.initCfg_eq, Cfg.init_state, Option.map_some]
  · rw [u2p_forwardLoopCfg_inputPos, MultiTapeTM.initCfg_eq, Cfg.init_inputPos]; rfl
  · rw [u2p_forwardLoopCfg_workTapePos]
    refine Fin.addCases (fun i => ?_) (fun j' => ?_) j
    · rw [Fin.addCases_left, MultiTapeTM.initCfg_eq, Cfg.init_workTapePos]
    · rw [Fin.addCases_right]
  · rw [u2p_forwardLoopCfg_output, MultiTapeTM.initCfg_eq, Cfg.init_output]

/-- An encoded configuration is halted exactly when the source is, and carries its output. -/
theorem u2_encode_halt {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x) :
    ((StepBlock.forwardLoopCfg entry M c).state = none ↔ c.state = none) ∧
      (StepBlock.forwardLoopCfg entry M c).output = c.output := by
  rw [u2p_forwardLoopCfg_state, u2p_forwardLoopCfg_output, Option.map_eq_none_iff]
  exact ⟨Iff.rfl, rfl⟩

/-- The ruler host's presented configuration: physical head at `ℓ +` the source's, the source's
bank first, the ruler last, the source's output. -/
theorem u2_ruler_fields {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (q : Option Q) (c : Cfg m Bool S y) (rt : ℤ → Option Bool)
    (rp : ℤ) :
    (rulerHostCfg ℓ hz q c rt rp).inputPos.val = ℓ + c.inputPos.val ∧
      (rulerHostCfg ℓ hz q c rt rp).workTapes (Fin.last m) = rt ∧
      (rulerHostCfg ℓ hz q c rt rp).workTapePos (Fin.last m) = rp ∧
      (∀ i : Fin m, (rulerHostCfg ℓ hz q c rt rp).workTapes i.castSucc = c.workTapes i) ∧
      (rulerHostCfg ℓ hz q c rt rp).output = c.output := by
  refine ⟨u2p_rulerHostCfg_inputPos_val _ _ _ _ _ _, ?_, ?_, fun i => ?_,
    u2p_rulerHostCfg_output _ _ _ _ _ _⟩
  · rw [u2p_rulerHostCfg_workTapes, Fin.lastCases_last]
  · rw [u2p_rulerHostCfg_workTapePos, Fin.lastCases_last]
  · rw [u2p_rulerHostCfg_workTapes, Fin.lastCases_castSucc]

end U2Route
