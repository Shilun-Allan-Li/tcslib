import TCSlib.Complexity.TuringMachine.CodeSimulator
import TCSlib.Complexity.TuringMachine.Nondeterministic
import TCSlib.Complexity.TuringMachine.Build.InterpreterStartup

/-! B5a, the uniform simulator: the controls for the seal of the junction probe
(`B5aJunctionProbe`, section `B5aUSJunction`). Each of its twenty sealed definitions has one goal
closing through it (by `rfl`, or for `StepBlock.Rep` by a projection). Unsealed (the first section)
every goal closes; sealed (the second, the junction's seal) every goal is expected to FAIL, so
that the junction's success is not by reduction. -/

namespace Turing

section Unsealed

-- MultiTapeTM.runFrom
example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (c : Cfg k Bool S x) :
    M.runFrom c 0 = c := rfl

-- MultiTapeTM.initCfg
example {k : ℕ} {S : Type} (M : MultiTapeTM k Bool S) (x : List Bool) :
    M.initCfg x = Cfg.init M.q₀ x := rfl

-- vhostCfg
example {m : ℕ} {S : Type} {x y : List Bool} (c : Cfg m Bool S y) (b : Bool)
    (p : Fin (x.length + 2)) (pre : List Bool) : (vhostCfg c b p pre).inputPos = p := rfl

-- counterLoopCfg
example {k : ℕ} {S : Type} {x : List Bool} (q : Option (CounterLoopState S))
    (c : Cfg k Bool S x) (ct : ℤ → Option Bool) (cp : ℤ) :
    (counterLoopCfg q c ct cp).state = q := rfl

-- FinTM.bufferTape
example : FinTM.bufferTape [true] 0 = some true := rfl

-- tableBitsK
example {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1))) :
    tableBitsK tr = (List.finRange (n + 1)).flatMap fun q =>
      ([none, some false, some true] : List (Option Bool)).flatMap fun inp =>
        (workReadsK k).flatMap fun w => actionBitsK (tr q inp w) := rfl

-- Cfg.init
example {k : ℕ} {S : Type} (q : S) (x : List Bool) :
    (Cfg.init q x : Cfg k Bool S x).inputPos = 1 := rfl

-- Cfg.mapState
example {k : ℕ} {S T : Type} {x : List Bool} (f : S → T) (c : Cfg k Bool S x) :
    (c.mapState f).inputPos = c.inputPos := rfl

-- embedEmitCfg
example {m k : ℕ} {S : Type} {x : List Bool} (ι : Fin m ↪ Fin k)
    (tapes : Fin k → ℤ → Option Bool) (heads : Fin k → ℤ) (pre : List Bool)
    (c : Cfg m Bool S x) : (embedEmitCfg ι tapes heads pre c).inputPos = c.inputPos := rfl

-- embedEmitTM
example {m k : ℕ} {S : Type} (ι : Fin m ↪ Fin k) (M : MultiTapeTM m Bool S) :
    (embedEmitTM ι M).q₀ = M.q₀ := rfl

-- seamCompTM
example {k : ℕ} {S₁ S₂ : Type} [DecidableEq S₁] (M₁ : MultiTapeTM k Bool S₁) (e : S₁)
    (M₂ : MultiTapeTM k Bool S₂) (en : S₂) :
    (seamCompTM M₁ e M₂ en).q₀ = Sum.inl M₁.q₀ := rfl

-- StepBlock.summaryCells
example (s : Option Bool) (w : List Bool) : StepBlock.summaryCells s w 0 = s := rfl

-- StepBlock.Rep
example {k n : ℕ} {x : List Bool}
    {tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1))}
    {c : Cfg k Bool (Fin (n + 1)) x} {d : Cfg (k + 2) Bool (Σ key, StepBlock.Ph k key) x}
    (h : StepBlock.Rep tr c d) : d.inputPos = c.inputPos := h.1

-- StepBlock.tableTape
example (k : ℕ) : StepBlock.tableTape k = Fin.natAdd k (0 : Fin 2) := rfl

-- StepBlock.summaryTape
example (k : ℕ) : StepBlock.summaryTape k = Fin.natAdd k (1 : Fin 2) := rfl

-- StepBlock.exit
example (k : ℕ) : StepBlock.exit k true = ⟨.entry, ()⟩ := rfl

-- StepBlock.outputSummary
example : StepBlock.outputSummary [] = none := rfl

-- unaryFin
example : unaryFin (0 : Fin 1) = [false] := rfl

-- FinTM.ComputesInTime
example (M : FinTM Bool) (x y : List Bool) (t : ℕ) :
    M.ComputesInTime x y t = ∃ s, M.tm.ComputesInTimeAndSpace x y t s := rfl

-- pairEncode
example : pairEncode [] [] = [false, true] := rfl

end Unsealed
