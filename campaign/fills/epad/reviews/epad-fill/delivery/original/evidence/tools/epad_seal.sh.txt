#!/usr/bin/env bash
# E-PAD rule-7 seal test. Usage: epad_seal.sh <tag>   (tag: cp1, cp2, ..., final)
# Requires: Padding.olean of the current source in /workspace/epad-oleans (chkE.py run first).
set -u
TAG="$1"
F=/workspace/tcslib-fill
E=/workspace/epad-evidence
cd $F
export PATH=$HOME/.elan/bin:$PATH
python3 /workspace/bin/chkE.py Probe/EPadForeignDefs > $E/seal/foreign-defs-by-decl-$TAG.txt 2>&1
python3 /workspace/bin/chkE.py Probe/EPadSimpDefs > $E/seal/simp-defs-$TAG.txt 2>&1
python3 - "$TAG" <<'PY'
import re, sys
tag = sys.argv[1]
E = '/workspace/epad-evidence'
F = '/workspace/tcslib-fill'
txt = open(f'{E}/seal/foreign-defs-by-decl-{tag}.txt').read()
names = set()
for line in txt.splitlines():
    m = re.match(r"^(\S+)\t\[(.*)\]$", line)
    if m:
        names.update(x.strip() for x in m.group(2).split(',') if x.strip())
# Not sealed, with the reason (each is disclosed in REPORT.md).
exempt = {
  'Complexity.LogProg.ARM': 'type synonym (`abbrev ARM m d Λ := Λ → Ins m d Λ`): every published ARM run lemma applies `A l`; sealing it makes the published statements themselves ill-typed at a concrete machine',
  'Complexity.LogProg.AConf': 'type synonym (`abbrev AConf m Λ := Option Λ × (Fin m → ℕ) × Option Bool`): the published run lemmas take configurations as triples',
  'Complexity.LogProg.BankS': 'type synonym (`abbrev BankS := Option (Σ j, (Ms j).State)`), the state type in `compile_decides`\'s statement; `[reducible]` by declaration (Lean refuses `[local irreducible]` on it); never unfolded',
}
proj = re.compile(r"^(Turing\.FinTM\.(k|State|tm|decEqState|fintypeState))$")
seal, notsealed = [], []
for n in sorted(names):
    if n in exempt: notsealed.append((n, exempt[n])); continue
    if proj.match(n): notsealed.append((n, 'structure projection')); continue
    if '.match_' in n:
        notsealed.append((n, 'auto-generated `match` auxiliary of a published statement (its `match` on `pairDecode w`); Lean reduces matchers by its matcher reduction whatever their reducibility, so `[local irreducible]` has no effect (logs/seal-control-matcher-demo.log: its control closes sealed)')); continue
    if re.search(r'\.(casesOn|recOn|brecOn|below|ibelow|noConfusionType)$', n):
        notsealed.append((n, 'auto-generated eliminator of an inductive type (from this file\'s own `match` on it), `[reducible]` by construction: Lean refuses `[local irreducible]` on it')); continue
    if n.endswith('.noConfusion'):
        notsealed.append((n, 'auto-generated `noConfusion` of an inductive type (constructor injectivity/disjointness; `[reducible]` by construction: Lean refuses `[local irreducible]` on it)')); continue
    seal.append(n)
open(f'{E}/seal/seal-list-{tag}.txt', 'w').write('\n'.join(seal) + '\n')
open(f'{E}/seal/not-sealed-{tag}.txt', 'w').write('\n'.join(f'{a}\t{b}' for a, b in notsealed) + '\n')
# every TCSlib definition the default simp set may unfold (Probe/EPadSimpDefs.lean), plus the
# brief's four named ones where present in the import closure
simpdefs = [l.strip() for l in open(f'{E}/seal/simp-defs-{tag}.txt') if re.match(r'^[A-Z][A-Za-z.]+$', l.strip())]
nosimp = sorted(set(simpdefs))
block = ('\n-- RULE-7 SEAL (evidence copy; not part of the delivery source)\n'
         'attribute [local irreducible]\n  ' + '\n  '.join(seal) + '\n')
if nosimp:
    block += 'attribute [-simp] ' + ' '.join(nosimp) + '\n'
src = open(f'{F}/TCSlib/Complexity/SpaceComplexity/Padding.lean').read()
anchor = '\nnamespace Complexity\n\nopen Turing\n'
assert anchor in src
src = src.replace(anchor, anchor + block, 1)
open(f'{F}/Probe/EPadSealed.lean', 'w').write(src)
ctrl_cmd = '''open Lean Elab Command Term Meta in
/-- `#epad_seal_rfl f` takes the first defining equation of `f` (`f.eq_1`, generated by Lean
from the definition: `f xs = body`, at the first pattern for a recursive definition; for a
`match` auxiliary, its first match equation) and closes its statement by `intros; rfl`, a
definitional check at default transparency. It closes exactly when `f` may be unfolded;
otherwise it reports an error. -/
elab "#epad_seal_rfl " id:ident : command => do
  let n ← liftCoreM <| realizeGlobalConstNoOverloadWithInfo id
  liftTermElabM do
    let e ← if (← Meta.isMatcher n) then
        pure (← Meta.Match.getEquationsFor n).eqnNames[0]!
      else
        let some eqs ← Meta.getEqnsFor? n | throwError "no equations for {n}"
        pure eqs[0]!
    let ty ← inferType (← mkConstWithLevelParams e)
    let ok ← try
        Term.withoutErrToSorry do
          let pf ← Term.elabTermEnsuringType (← `(by intros; rfl)) ty
          Term.synthesizeSyntheticMVarsNoPostponing
          let _ ← instantiateMVars pf
        pure true
      catch _ => pure false
    if ok then logInfo m!"rfl closes: {n}   [{e} : {ty}]"
    else throwError m!"rfl FAILS: {n}   [{e} : {ty}]"
'''
imports = ''.join(l + '\n' for l in src.splitlines() if l.startswith('import '))
checks = '\n'.join(f'#epad_seal_rfl {n}' for n in seal) + '\n'
open(f'{F}/Probe/EPadSealControlUnsealed.lean', 'w').write(
    imports + '\n/-! Seal controls, unsealed: each sealed definition closes its defining equation by `rfl`. -/\n'
    + ctrl_cmd + '\n' + checks)
open(f'{F}/Probe/EPadSealControlSealed.lean', 'w').write(
    imports + '\n/-! Seal controls, sealed: the same equations fail under the seal list. -/\n'
    + ctrl_cmd + block + '\n' + checks)
print('sealed', len(seal), 'not sealed', len(notsealed), 'no-simp', nosimp)
PY
cp Probe/EPadSealed.lean $E/seal/EPadSealed-$TAG.lean
cp Probe/EPadSealControlUnsealed.lean $E/seal/EPadSealControlUnsealed-$TAG.lean
cp Probe/EPadSealControlSealed.lean $E/seal/EPadSealControlSealed-$TAG.lean
python3 /workspace/bin/chkE.py Probe/EPadSealed > $E/logs/seal-main-$TAG.log 2>&1
{ echo "== unsealed"; python3 /workspace/bin/chkE.py Probe/EPadSealControlUnsealed;
  echo "== sealed (every #epad_seal_rfl must fail)"; python3 /workspace/bin/chkE.py Probe/EPadSealControlSealed; } > $E/logs/seal-control-$TAG.log 2>&1
echo "seal main:"; grep -c "error" $E/logs/seal-main-$TAG.log; tail -1 $E/logs/seal-main-$TAG.log
echo "controls unsealed closes:"; grep -c "rfl closes" $E/logs/seal-control-$TAG.log
echo "controls sealed errors:"; grep -c "error" $E/logs/seal-control-$TAG.log
