import TCSlib.Complexity.Diagonalization.NTimeHierarchy
import TCSlib.Complexity.Diagonalization.NTimeDiagonalizer

namespace Complexity

open Turing

section Unsealed

-- NTIME
example (T : ℕ → ℕ) :
    NTIME T = {L | ∃ (c : ℕ) (N : FinNDTM Bool), N.DecidesInTime L fun n => c * T n} := rfl

-- FinNDTM.DecidesInTime
example (N : FinNDTM Bool) (L : Language Bool) (T : ℕ → ℕ) :
    N.DecidesInTime L T = ∀ x : List Bool,
      N.tm.HaltsWithin x (T x.length) ∧ (x ∈ L ↔ N.AcceptsWithin x (T x.length)) := rfl

-- FinNDTM.AcceptsWithin
example (N : FinNDTM Bool) (x : List Bool) (t : ℕ) :
    N.AcceptsWithin x t = ∃ w : List Bool, w.length = t ∧
      (N.tm.runWith w (N.tm.initCfg x)).state = none ∧
      (N.tm.runWith w (N.tm.initCfg x)).output = [true] := rfl

-- NDTM.HaltsWithin
example (N : FinNDTM Bool) (x : List Bool) (t : ℕ) :
    N.tm.HaltsWithin x t = ∀ w : List Bool, w.length = t →
      (N.tm.runWith w (N.tm.initCfg x)).state = none := rfl

end Unsealed
