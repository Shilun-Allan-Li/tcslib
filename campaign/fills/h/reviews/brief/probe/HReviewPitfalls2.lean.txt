/-
H brief review, the membership pitfall in the target's own shape (review sub-agent, 2026-10-27):
`L` introduced by `set` from `acceptLang_mem_NTIME`'s set, then read by `simp only`.
-/
import TCSlib.Complexity.Diagonalization.NTimeHierarchy
import TCSlib.Complexity.Diagonalization.NTimeDiagonalizer

open Turing

-- (a) through `set ... with hL` and `simp only [hL, Set.mem_setOf_eq, List.length_replicate]`
example (D : FinNDTM Bool) (K : ℕ) (g : ℕ → ℕ)
    (hhalt : ∀ x : List Bool, D.tm.HaltsWithin x (K * (g x.length + 1))) (m : ℕ) : True := by
  have h := NTimeDiag.acceptLang_mem_NTIME D K g hhalt
  set L : Language Bool := {x | D.AcceptsWithin x (K * (g x.length + 1))} with hL
  have : List.replicate m true ∈ L ↔ D.AcceptsWithin (List.replicate m true) (K * (g m + 1)) := by
    simp only [hL, Set.mem_setOf_eq, List.length_replicate]
  trivial

-- (b) the same, read by `show` and `List.length_replicate` (expected: closes)
example (D : FinNDTM Bool) (K : ℕ) (g : ℕ → ℕ)
    (hhalt : ∀ x : List Bool, D.tm.HaltsWithin x (K * (g x.length + 1))) (m : ℕ) : True := by
  have h := NTimeDiag.acceptLang_mem_NTIME D K g hhalt
  set L : Language Bool := {x | D.AcceptsWithin x (K * (g x.length + 1))} with hL
  have : List.replicate m true ∈ L ↔ D.AcceptsWithin (List.replicate m true) (K * (g m + 1)) := by
    show D.AcceptsWithin (List.replicate m true) (K * (g (List.replicate m true).length + 1)) ↔ _
    rw [List.length_replicate]
  trivial
