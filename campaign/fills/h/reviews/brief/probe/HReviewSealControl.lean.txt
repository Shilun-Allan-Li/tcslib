section Unsealed

-- NDTM.runWith
example (N : FinNDTM Bool) (x : List Bool) : N.tm.runWith [] (N.tm.initCfg x) = N.tm.initCfg x := rfl

-- NDTM.initCfg
example (N : FinNDTM Bool) (x : List Bool) : N.tm.initCfg x = Cfg.init N.tm.q₀ x := rfl

-- NTimeDiag.ladder
example (f : ℕ → ℕ) : NTimeDiag.ladder f 0 = 2 := rfl

-- NTimeDiag.budget
example (f : ℕ → ℕ) (l : ℕ) : NTimeDiag.budget f l = (f (l + 1) + (l + 1) + 1) ^ 2 := rfl

-- NTimeDiag.stage
example (f : ℕ → ℕ) (n : ℕ) :
    NTimeDiag.stage f n = Nat.find (NTimeDiag.exists_le_ladder_succ f n) := rfl

-- NTimeDiag.stageCode
example (i : ℕ) : NTimeDiag.stageCode i = (Nat.bits ((Nat.unpair i).1 + 1)).dropLast := rfl

-- TimeConstructible
example (T : ℕ → ℕ) : TimeConstructible T = ((∀ n, n ≤ T n) ∧ ∃ c : ℕ, 0 < c ∧
    ∃ M : FinTM Bool, ∀ x : List Bool,
      M.ComputesInTime x (T x.length).bits (c * (T x.length + 1))) := rfl

-- CodeNDTM.toFinNDTM
example (M : CodeNDTM) : M.toFinNDTM.k = 2 := rfl

end Unsealed
