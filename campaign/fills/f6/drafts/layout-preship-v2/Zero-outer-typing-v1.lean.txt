/-- The full-bank emitting embedding at the zero-outer-frame specialization
is reconstructed by the existing public configuration contract. -/
theorem full_bank_embedding {m : ℕ} {Q : Type} {z : List Bool} (c : Cfg m Bool Q z) :
    embedEmitCfg (Fin.castLEEmb (le_refl m)) (fun _ _ => none) (fun _ => 0) [] c = c := by
  apply embedEmitCfg_eq
  · rfl
  · rfl
  · exact List.nil_append _
  · intro i
    exact ⟨rfl, rfl⟩
  · intro j hj
    exact False.elim (hj ⟨j, rfl⟩)

/-- Actual constructor typing at k=K+2, outer K=0 gives K+4 physical tapes.
This is a constructor/type probe only; it supplies no run, cut or space premise. -/
theorem zero_outer_constructor (K : ℕ) {P Q : Type} [DecidableEq P]
    (A : MultiTapeTM (K + 4) Bool P) (finish : P)
    (H : MultiTapeTM (K + 4) Bool Q) (entry : Q) :
    Nonempty (MultiTapeTM (K + 4) Bool (P ⊕ Q)) := by
  let A' : MultiTapeTM (0 + ((K + 2) + 2)) Bool P := by
    simpa only [Nat.zero_add] using A
  refine ⟨?_⟩
  simpa only [Nat.zero_add] using
    (startupLoopTM (k := K + 2) (K := 0) A' finish H entry)

end F6LayoutPreship
