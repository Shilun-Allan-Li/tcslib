import TCSlib.Complexity.TuringMachine.Encoding
/-! Conditional consumer of the exact proposed whole-bundle equation.
The equation is a hypothesis, not a published or assumed new theorem. -/
set_option autoImplicit false
attribute [local irreducible] Turing.CodeTM.toFinTM
namespace F6DraftProposedReader
open Turing

/-- Frozen-target computation and space premises convert through the proposed
bundle equation with the foreign definition sealed. -/
theorem conditional_bundle_consumer (M : CodeTM)
    (hEq : M.toFinTM = { k := 1, State := Fin (M.numStates + 1), tm := M.tm })
    (x output : List Bool) (t s : ℕ)
    (hcomp : M.toFinTM.ComputesInTime x output t)
    (hspace : M.toFinTM.tm.spaceUsed (M.toFinTM.tm.initCfg x) t ≤ s) :
    (M.tm.runFrom (M.tm.initCfg x) t).state = none ∧
    (M.tm.runFrom (M.tm.initCfg x) t).output = output ∧
    M.tm.spaceUsed (M.tm.initCfg x) t ≤ s := by
  rw [hEq] at hcomp hspace
  obtain ⟨hhalt, hout⟩ := (FinTM.computesInTime_iff
    { k := 1, State := Fin (M.numStates + 1), tm := M.tm } x output t).mp hcomp
  exact ⟨hhalt, hout, hspace⟩

end F6DraftProposedReader
