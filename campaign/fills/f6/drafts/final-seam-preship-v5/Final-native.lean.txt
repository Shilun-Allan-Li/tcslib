/-- Pure layout reconstruction for the last-ruler route. The three H2-F tapes
occupy the first three of K+3 bank tapes; H5 adds the ruler last. No run/cut
premise for the unfinished header/startup prefix is supplied by this equation. -/
theorem last_ruler_layout (K n ℓ : ℕ) {P Q V : Type}
    (finish : P) (entry : Q) (inj : Q → Bool → V)
    (M : MultiTapeTM 1 Bool (Fin (n + 1))) (x z : List Bool)
    (hz : z.length = ℓ + x.length) :
    let σ : Fin 3 ↪ Fin (K + 3) := Fin.castLEEmb (by omega)
    let ec := embedEmitCfg σ (fun _ _ => none) (fun _ => 0) []
      (StepBlock.forwardLoopCfg entry M (M.initCfg x))
    let bank : Cfg (K + 3) Bool P x :=
      ⟨some finish, 1,
        fun j => if j = 1 then FinTM.bufferTape (tableBitsK M.tr)
          else if j = 2 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
          else fun _ => none,
        fun _ => 0, []⟩
    let physical : Cfg ((K + 3) + 1) Bool P z :=
      ⟨some finish, ⟨ℓ + 1, by omega⟩,
        Fin.lastCases (FinTM.bufferTape (List.replicate ℓ true)) bank.workTapes,
        fun _ => 0, []⟩
    physical.mapState (fun _ => inj entry false) =
      rulerHostCfg ℓ hz (ec.state.map fun q => inj q false) ec
        (FinTM.bufferTape (List.replicate ℓ true)) 0 := by
  intro σ ec bank physical
  dsimp only [physical]
  have h1 (j : Fin (K + 3)) : (j = 1) ↔ j.val = 1 := by
    rw [Fin.ext_iff]
    change j.val = 1 % (K + 3) ↔ j.val = 1
    rw [Nat.mod_eq_of_lt (by omega : 1 < K + 3)]
  have h2 (j : Fin (K + 3)) : (j = 2) ↔ j.val = 2 := by
    rw [Fin.ext_iff]
    change j.val = 2 % (K + 3) ↔ j.val = 2
    rw [Nat.mod_eq_of_lt (by omega : 2 < K + 3)]
  have hbank : ec = bank.mapState (fun _ => entry) := by
    have h := embedEmitCfg_forwardLoop_init (k := 1) (K := K) finish entry M x
    simpa only [ec, bank, σ, h1, h2] using h
  apply Cfg.ext
  · rw [Cfg.mapState_state, rulerHostCfg_state, hbank, Cfg.mapState_state]
    all_goals rfl
  · apply Fin.ext
    rw [Cfg.mapState_inputPos, (rulerHostCfg_fields ℓ hz _ ec _ 0).1,
      hbank, Cfg.mapState_inputPos]
    simp only [bank, Fin.val_one]
  · funext i p
    rw [Cfg.mapState_workTapes, rulerHostCfg_workTapes, hbank]
    change Fin.lastCases (motive := fun _ => ℤ → Option Bool) (FinTM.bufferTape (List.replicate ℓ true)) bank.workTapes i p =
      Fin.lastCases (motive := fun _ => ℤ → Option Bool)
        (FinTM.bufferTape (List.replicate ℓ true)) (bank.mapState (fun _ => entry)).workTapes i p
    refine Fin.lastCases ?_ (fun j => ?_) i
    · simp only [Fin.lastCases_last]
    · simp only [Fin.lastCases_castSucc, Cfg.mapState_workTapes]
  · funext i
    rw [Cfg.mapState_workTapePos, rulerHostCfg_workTapePos, hbank]
    refine Fin.lastCases ?_ (fun j => ?_) i
    · rw [Fin.lastCases_last]
    · rw [Fin.lastCases_castSucc, Cfg.mapState_workTapePos]
  · rw [Cfg.mapState_output, rulerHostCfg_output, hbank, Cfg.mapState_output]


/-- The full-bank emitting embedding at the zero-outer-frame specialization
is reconstructed by the existing public configuration contract. -/
theorem full_bank_embedding {m : ℕ} {Q : Type} {z : List Bool} (c : Cfg m Bool Q z) :
    embedEmitCfg (Fin.castLEEmb (le_refl m)) (fun _ _ => none) (fun _ => 0) [] c = c := by
  apply embedEmitCfg_eq
  · rfl
  · rfl
  · exact List.nil_append _
  · intro i
    exact ⟨rfl, rfl⟩
  · intro j hj
    exact False.elim (hj ⟨j, rfl⟩)

/-- Equality induction transports the indexed operations without opening them. -/
private theorem cast_init_run {P : Type} {m n : ℕ} (h : m = n)
    (A : MultiTapeTM m Bool P) (z : List Bool) (t : ℕ) :
    (h ▸ A : MultiTapeTM n Bool P).runFrom
      ((h ▸ A : MultiTapeTM n Bool P).initCfg z) t =
        (h ▸ A.runFrom (A.initCfg z) t : Cfg n Bool P z) := by
  cases h
  rfl

private theorem cast_init_space {P : Type} {m n : ℕ} (h : m = n)
    (A : MultiTapeTM m Bool P) (z : List Bool) (t : ℕ) :
    (h ▸ A : MultiTapeTM n Bool P).spaceUsed
      ((h ▸ A : MultiTapeTM n Bool P).initCfg z) t =
        A.spaceUsed (A.initCfg z) t := by
  cases h
  rfl

private theorem cast_cfg_state {P : Type} {m n : ℕ} (h : m = n) {z : List Bool}
    (c : Cfg m Bool P z) : (h ▸ c : Cfg n Bool P z).state = c.state := by
  cases h
  rfl

private theorem cast_cfg_mapState {P Q : Type} {m n : ℕ} (h : m = n) {z : List Bool}
    (c : Cfg m Bool P z) (f : P → Q) :
    (h ▸ c : Cfg n Bool P z).mapState f = (h ▸ c.mapState f : Cfg n Bool Q z) := by
  cases h
  rfl

private theorem cast_cfg_embedding {Q : Type} {m n : ℕ} (h : m = n) {z : List Bool}
    (c : Cfg m Bool Q z) :
    (h ▸ c : Cfg n Bool Q z) =
      embedEmitCfg (Fin.castLEEmb (Nat.le_of_eq h)) (fun _ _ => none) (fun _ => 0) [] c := by
  cases h
  exact (full_bank_embedding c).symm

private theorem cast_back_embedding {Q : Type} {m n : ℕ} (h : m = n) {z : List Bool}
    (c : Cfg m Bool Q z) :
    (h.symm ▸ embedEmitCfg (Fin.castLEEmb (Nat.le_of_eq h))
      (fun _ _ => none) (fun _ => 0) [] c : Cfg m Bool Q z) = c := by
  cases h
  exact full_bank_embedding c

/-- Conditional consumer at the actual last-ruler endpoint. Prefix run/cut/space
and the hosted loop's per-tape bound are explicit OWN inputs, not asserted facts.

**Proof sketch.** Reconstruct the complete last-ruler endpoint via public fields,
transport the prefix and endpoint along zero-addition by equality induction,
apply the published seam run and per-tape space contracts, then transport their
conclusions back to the physical bank. No OWN premise is proved here. -/
theorem conditional_final_seam (K n ℓ : ℕ) {P Q V : Type} [DecidableEq P]
    (A : MultiTapeTM (K + 4) Bool P) (finish : P)
    (H : MultiTapeTM (K + 4) Bool V) (entry : Q) (inj : Q → Bool → V)
    (M : MultiTapeTM 1 Bool (Fin (n + 1))) (x z : List Bool)
    (hz : z.length = ℓ + x.length) :
    let σ : Fin 3 ↪ Fin (K + 3) := Fin.castLEEmb (by omega)
    let ec := embedEmitCfg σ (fun _ _ => none) (fun _ => 0) []
      (StepBlock.forwardLoopCfg entry M (M.initCfg x))
    let bank : Cfg (K + 3) Bool P x :=
      ⟨some finish, 1,
        fun j => if j = 1 then FinTM.bufferTape (tableBitsK M.tr)
          else if j = 2 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
          else fun _ => none,
        fun _ => 0, []⟩
    let d : Cfg (K + 4) Bool P z :=
      ⟨some finish, ⟨ℓ + 1, by omega⟩,
        Fin.lastCases (FinTM.bufferTape (List.replicate ℓ true)) bank.workTapes,
        fun _ => 0, []⟩
    let c := rulerHostCfg ℓ hz (ec.state.map fun q => inj q false) ec
      (FinTM.bufferTape (List.replicate ℓ true)) 0
    ∀ (T t B S : ℕ),
      (aRun : A.runFrom (A.initCfg z) T = d) →
      (aCut : ∀ u < T, (A.runFrom (A.initCfg z) u).state ≠ some finish) →
      (hB : A.spaceUsed (A.initCfg z) T ≤ B) →
      (hS : 1 ≤ S) →
      (hloop : ∀ i, H.spaceUsedByTape c t i ≤ S) →
      let A0 : MultiTapeTM (0 + ((K + 2) + 2)) Bool P :=
        (Nat.zero_add ((K + 2) + 2)).symm ▸ A
      let U0 := startupLoopTM (k := K + 2) (K := 0) A0 finish H (inj entry false)
      let U : MultiTapeTM (K + 4) Bool (P ⊕ V) :=
        (Nat.zero_add ((K + 2) + 2)) ▸ U0
      U.runFrom (U.initCfg z) (T + 1 + t) = (H.runFrom c t).mapState Sum.inr ∧
        U.spaceUsed (U.initCfg z) (T + 1 + t) ≤ B + (K + 4) * S := by
  intro σ ec bank d c T t B S aRun aCut hB hS hloop A0 U0 U
  have hexit : d.state = some finish := rfl
  have hphysical : d.mapState (fun _ => inj entry false) = c :=
    last_ruler_layout K n ℓ finish entry inj M x z hz
  have hcfg : d.mapState (fun _ => inj entry false) =
      embedEmitCfg (Fin.castLEEmb (le_refl (K + 4)))
        (fun _ _ => none) (fun _ => 0) [] c :=
    hphysical.trans (full_bank_embedding c).symm
  let e : (K + 4) = 0 + ((K + 2) + 2) := (Nat.zero_add ((K + 2) + 2)).symm
  let d0 : Cfg (0 + ((K + 2) + 2)) Bool P z := e ▸ d
  have aRun0 : A0.runFrom (A0.initCfg z) T = d0 := by
    calc
      _ = (e ▸ A.runFrom (A.initCfg z) T : Cfg (0 + ((K + 2) + 2)) Bool P z) :=
        cast_init_run e A z T
      _ = d0 := congrArg (fun v : Cfg (K + 4) Bool P z =>
        (e ▸ v : Cfg (0 + ((K + 2) + 2)) Bool P z)) aRun
  have hexit0 : d0.state = some finish := (cast_cfg_state e d).trans hexit
  have aCut0 : ∀ u < T, (A0.runFrom (A0.initCfg z) u).state ≠ some finish := by
    intro u hu
    rw [show A0.runFrom (A0.initCfg z) u =
      (e ▸ A.runFrom (A.initCfg z) u : Cfg (0 + ((K + 2) + 2)) Bool P z) from
        cast_init_run e A z u, cast_cfg_state]
    exact aCut u hu
  have hcfg0 : d0.mapState (fun _ => inj entry false) =
      embedEmitCfg (Fin.castLEEmb (by omega : (K + 2) + 2 ≤ 0 + ((K + 2) + 2)))
        (fun _ _ => none) (fun _ => 0) [] c := by
    calc
      _ = (e ▸ d.mapState (fun _ => inj entry false) :
          Cfg (0 + ((K + 2) + 2)) Bool V z) := cast_cfg_mapState e d _
      _ = (e ▸ c : Cfg (0 + ((K + 2) + 2)) Bool V z) :=
        congrArg (fun v : Cfg (K + 4) Bool V z =>
          (e ▸ v : Cfg (0 + ((K + 2) + 2)) Bool V z)) hphysical
      _ = _ := cast_cfg_embedding e c
  have hB0 : A0.spaceUsed (A0.initCfg z) T ≤ B :=
    (cast_init_space e A z T).trans_le hB
  have hr := startupLoopTM_runFrom (k := K + 2) (K := 0)
    A0 finish H (inj entry false) c d0 T aRun0 hexit0 aCut0 hcfg0 t
  have hs := startupLoopTM_spaceUsed_le (k := K + 2) (K := 0)
    A0 finish H (inj entry false) c d0 T aRun0 hexit0 aCut0 hcfg0 t B S hB0 hS hloop
  constructor
  · calc
      _ = (e.symm ▸ U0.runFrom (U0.initCfg z) (T + 1 + t) : Cfg (K + 4) Bool (P ⊕ V) z) :=
        cast_init_run e.symm U0 z (T + 1 + t)
      _ = (e.symm ▸ (embedEmitCfg (Fin.castLEEmb (Nat.le_of_eq e))
          (fun _ _ => none) (fun _ => 0) [] (H.runFrom c t)).mapState Sum.inr :
            Cfg (K + 4) Bool (P ⊕ V) z) :=
        congrArg (fun v : Cfg (0 + ((K + 2) + 2)) Bool (P ⊕ V) z =>
          (e.symm ▸ v : Cfg (K + 4) Bool (P ⊕ V) z)) hr
      _ = _ := by
        rw [← cast_cfg_mapState, cast_back_embedding]
        exact e
  · calc
      _ = U0.spaceUsed (U0.initCfg z) (T + 1 + t) :=
        cast_init_space e.symm U0 z (T + 1 + t)
      _ ≤ B + (0 + ((K + 2) + 2)) * S := hs
      _ = B + (K + 4) * S := by rw [Nat.zero_add]

/-- The suggested common loop budget is positive without a source-space premise.
This arithmetic fact does not establish its per-tape bound. -/
theorem common_budget_positive (s D ℓ : ℕ) : 1 ≤ s + (2 * D + 1) + ℓ + 3 := by
  omega

/-- Consumer of the implemented, still-unmerged prerequisite reader, with
CodeTM.toFinTM sealed before this header. No assumed reader hypothesis. -/
theorem actual_bundle_consumer (M : CodeTM) (x output : List Bool) (t s : ℕ)
    (hcomp : M.toFinTM.ComputesInTime x output t)
    (hspace : M.toFinTM.tm.spaceUsed (M.toFinTM.tm.initCfg x) t ≤ s) :
    (M.tm.runFrom (M.tm.initCfg x) t).state = none ∧
      (M.tm.runFrom (M.tm.initCfg x) t).output = output ∧
      M.tm.spaceUsed (M.tm.initCfg x) t ≤ s := by
  rw [CodeTM.toFinTM_eq] at hcomp hspace
  obtain ⟨halted, out⟩ := (FinTM.computesInTime_iff
    { k := 1, State := Fin (M.numStates + 1), tm := M.tm } x output t).mp hcomp
  exact ⟨halted, out, hspace⟩

/-! Read-only compiled inventory of the eleven bounded proof roots and their descendants.
Imported theorem proof values are not traversed. Existing same-home legacy helpers remain
outside the changed proof scope. Root types and compiled values are both inspected. -/
open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  let wanted : List String := ["last_ruler_layout", "full_bank_embedding", "cast_init_run", "cast_init_space", "cast_cfg_state", "cast_cfg_mapState", "cast_cfg_embedding", "cast_back_embedding", "conditional_final_seam", "common_budget_positive", "actual_bundle_consumer"]
  let kind := fun (ci : ConstantInfo) => match ci with
    | .axiomInfo _ => "axiom" | .defnInfo _ => "definition" | .thmInfo _ => "theorem"
    | .opaqueInfo _ => "opaque" | .quotInfo _ => "quotient" | .inductInfo _ => "inductive"
    | .ctorInfo _ => "constructor" | .recInfo _ => "recursor"
  let mut roots : Array Name := #[]
  for (n, _) in env.constants.toList do
    match n with
    | .str _ base => if wanted.contains base then roots := roots.push n
    | _ => pure ()
  if roots.size != wanted.length then
    throwError "Root count mismatch: {roots.size} for {wanted.length} source roots"
  for root in roots.qsort Name.lt do
    logInfo m!"ROOT|{root}|private={isPrivateName root}"
  let mut selected : Array Name := #[]
  for (n, _) in env.constants.toList do
    if roots.any (fun r => r.isPrefixOf n) then selected := selected.push n
  for n in selected.qsort Name.lt do
    let some ci := env.find? n | throwError "selected constant missing {n}"
    let axs ← liftCoreM <| Lean.collectAxioms n
    logInfo m!"PROOFAX|{n}|kind={kind ci}|{(axs.qsort Name.lt).toList}"
    let used := ci.type.getUsedConstants ++ (ci.value?.map (·.getUsedConstants)).getD #[]
    let mut seen : Array Name := #[]
    for c in used do
      if seen.contains c then continue
      seen := seen.push c
      match env.getModuleIdxFor? c, env.find? c with
      | some j, some (.defnInfo di) =>
        let owner := env.header.moduleNames[j.toNat]!
        if (`TCSlib).isPrefixOf owner then
          let hint := match di.hints with
            | .abbrev => "abbrev" | .opaque => "opaque" | .regular _ => "regular"
          logInfo m!"FOREIGNDEF|{n}|{c}|{owner}|hints={hint}"
      | none, some (.defnInfo _) =>
        logInfo m!"HOMEDEF|{n}|{c}"
      | _, _ => pure ()
