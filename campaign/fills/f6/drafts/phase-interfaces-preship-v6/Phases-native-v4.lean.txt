/-- The second scan extends an already nonblank ruler; its physical start presentation is an
explicit prefix OWN interface, not a supplied proof that the entire prefix reaches it.

**Proof sketch.** Apply the public scan contract and ruler-extension reader at the existing
nonblank length. Public field equations and integer arithmetic identify the common ruler origin
and the two endpoint heads; their update equations preserve every other bank field. -/
theorem second_scan {k : ℕ} {z : List Bool} (i : Fin k)
    (d : Cfg k Bool (Fin 4) z) (ell1 : ℕ)
    (hp : d.inputPos.val = 1 + ell1) (hq : d.state = some 0)
    (hh : d.workTapePos i = (ell1 : ℤ))
    (hw : ∀ a, d.workTapes i a = FinTM.bufferTape (List.replicate ell1 true) a)
    (u v : List Bool) (hz : z.drop ell1 = pairEncode u v) :
    let ell2 := 2 * u.length + 2
    let ht : d.inputPos.val + ell2 < z.length + 2 := by
      have h := congrArg List.length hz
      rw [List.length_drop, length_pairEncode] at h
      dsimp only [ell2]
      omega
    let c := headerScanCfg i d 3 ell2 ht
    (headerScanTM i).runFrom d ell2 = c ∧
      (∀ t < ell2, ((headerScanTM i).runFrom d t).state ≠ some 3) ∧
      c.inputPos.val = 1 + (ell1 + ell2) ∧
      c.workTapePos i = ((ell1 + ell2 : ℕ) : ℤ) ∧
      (∀ a, c.workTapes i a = FinTM.bufferTape (List.replicate (ell1 + ell2) true) a) ∧
      (∀ j, j ≠ i → c.workTapes j = d.workTapes j ∧ c.workTapePos j = d.workTapePos j) ∧
      (∀ t ≤ ell2, ((headerScanTM i).runFrom d t).output = d.output) := by
  intro ell2 ht c
  obtain ⟨run, cut, silent⟩ := headerScanTM_run i d ell1 hp hq u v hz
  refine ⟨run, cut, ?_, ?_, ?_, ?_, silent⟩
  · rw [headerScanCfg_inputPos]
    change d.inputPos.val + ell2 = 1 + (ell1 + ell2)
    omega
  · rw [headerScanCfg_workTapePos]
    change Function.update d.workTapePos i (d.workTapePos i + ell2) i = _
    rw [Function.update_self, hh, Nat.cast_add]
    dsimp only [ell2]
    omega
  · intro a
    have hpre (b : ℤ) : d.workTapes i (d.workTapePos i - (ell1 : ℤ) + b) =
        FinTM.bufferTape (List.replicate ell1 true) b := by
      simpa only [hh, sub_self, zero_add] using hw b
    have h := headerScanCfg_ruler i d 3 ell2 ht ell1 hpre a
    simpa only [hh, sub_self, zero_add] using h
  · intro j hj
    constructor
    · funext a
      rw [headerScanCfg_workTapes, Function.update_of_ne hj]
    · rw [headerScanCfg_workTapePos]
      exact Function.update_of_ne hj _ _

/-- The actual skip ends one past the first ruler. Its start presentation is a named prefix
OWN interface. The published framed row preserves every field except control and this head. -/
theorem skip_endpoint {k : ℕ} {z : List Bool} (i : Fin k)
    (d : Cfg k Bool SeekPhase z) (ell1 : ℕ)
    (hq : d.state = some .run) (hh : d.workTapePos i = 0)
    (hw : ∀ a, d.workTapes i a = FinTM.bufferTape (List.replicate ell1 true) a) :
    (skipTM k i).runFrom d (ell1 + 1) =
      { d with
        state := some .done
        workTapePos := Function.update d.workTapePos i ((ell1 : ℤ) + 1)} ∧
      (∀ t < ell1 + 1, ((skipTM k i).runFrom d t).state ≠ some .done) := by
  obtain ⟨run, cut, _⟩ := skipTM_run_ofCfg i (List.replicate ell1 true) d hq
    (fun p _ _ => by simpa only [hh, zero_add] using hw p)
  simpa only [List.length_replicate, hh, zero_add] using And.intro run cut

/-- A scanned ruler of total length N supplies the exact framed-rewind window, including its
left blank. The row itself consumes N+2 steps and preserves all unselected heads. -/
theorem final_rewind {k : ℕ} {z : List Bool} (i : Fin k)
    (d : Cfg k Bool RewindPhase z) (N : ℕ)
    (hq : d.state = some .start) (hh : d.workTapePos i = (N : ℤ))
    (hw : ∀ a, d.workTapes i a = FinTM.bufferTape (List.replicate N true) a) :
    (rewindTM k i).runFrom d (N + 2) =
      {d with state := some .done, workTapePos := Function.update d.workTapePos i 0} ∧
      (∀ t < N + 2, ((rewindTM k i).runFrom d t).state ≠ some .done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ N + 2 → j ≠ i →
        ((rewindTM k i).runFrom d t).workTapePos j = d.workTapePos j) := by
  obtain ⟨run, cut, frame⟩ := rewindTM_run_ofCfg i (List.replicate N true) N
    (by rw [List.length_replicate]) d hq
    (fun p _ _ => by simpa only [hh, sub_self, zero_add] using hw p)
  refine ⟨?_, cut, fun j t ht hj => ?_⟩
  · simpa only [hh, sub_self] using run
  · simpa only [if_neg hj] using frame j t ht

/-- Relabelling an actual second-scan endpoint supplies all final rewind entry fields, while
keeping its arbitrary installed bank and physical input/output unchanged. -/
theorem to_rewind {k : ℕ} {z : List Bool} (i : Fin k)
    (c : Cfg k Bool (Fin 4) z) (N : ℕ)
    (hq : c.state = some 3) (hh : c.workTapePos i = (N : ℤ))
    (hw : ∀ a, c.workTapes i a = FinTM.bufferTape (List.replicate N true) a) :
    let r := c.mapState (fun _ => RewindPhase.start)
    r.state = some .start ∧ r.workTapePos i = (N : ℤ) ∧
      (∀ a, r.workTapes i a = FinTM.bufferTape (List.replicate N true) a) ∧
      r.inputPos = c.inputPos ∧ r.output = c.output := by
  intro r
  refine ⟨?_, ?_, fun a => ?_, ?_, ?_⟩
  · rw [Cfg.mapState_state, hq, Option.map_some]
  · rw [Cfg.mapState_workTapePos]; exact hh
  · rw [Cfg.mapState_workTapes]; exact hw a
  · rw [Cfg.mapState_inputPos]
  · rw [Cfg.mapState_output]

/-- Installed-bank fields follow from the actual public H2-F embedding contract. The source
one-tape model uses bank indices 0/1/2; arbitrary extra K tapes remain blank and stationary. -/
theorem installed_bank (K n : ℕ) {Q : Type} (entry : Q)
    (M : MultiTapeTM 1 Bool (Fin (n + 1))) (x : List Bool) :
    let sigma : Fin 3 ↪ Fin (K + 3) := Fin.castLEEmb (by omega)
    let e := embedEmitCfg sigma (fun _ _ => none) (fun _ => 0) []
      (StepBlock.forwardLoopCfg entry M (M.initCfg x))
    e.state = some entry ∧ e.inputPos.val = 1 ∧
      (∀ j, e.workTapePos j = 0) ∧ e.output = [] ∧
      (∀ (j : Fin (K + 3)) (a : ℤ), e.workTapes j a =
        if j.val = 1 then FinTM.bufferTape (tableBitsK M.tr) a
        else if j.val = 2 then FinTM.bufferTape (unaryFin M.q₀) (a - 2)
        else none) := by
  intro sigma e
  have h := embedEmitCfg_forwardLoop_init (k := 1) (K := K) () entry M x
  dsimp only [e, sigma] at *
  refine ⟨?_, ?_, fun j => ?_, ?_, fun j a => ?_⟩
  · rw [h, Cfg.mapState_state]; rfl
  · rw [h, Cfg.mapState_inputPos]; rfl
  · rw [h, Cfg.mapState_workTapePos]
  · rw [h, Cfg.mapState_output]
  · rw [h, Cfg.mapState_workTapes]
    change (if j.val = 1 then FinTM.bufferTape (tableBitsK M.tr)
      else if j.val = 2 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
      else fun _ => none) a = _
    split_ifs <;> rfl

/-- An elementary native correction leaf moves only the last work head left once.
The machine is fixed before the input/configuration; no payload is copied.

**Proof sketch.** Use the published tape-move step equation and the public zero/succ
run equations. The initial label is false and the fresh endpoint label is true. -/
theorem last_ruler_correction (K : ℕ) :
    ∃ C : MultiTapeTM (K + 4) Bool Bool, C.q₀ = false ∧
      ∀ (z : List Bool) (d : Cfg (K + 4) Bool Bool z), d.state = some false →
        C.runFrom d 1 = {d with
          state := some true
          workTapePos := Function.update d.workTapePos (Fin.last (K + 3))
            (d.workTapePos (Fin.last (K + 3)) - 1)} ∧
        (∀ u < 1, (C.runFrom d u).state ≠ some true) ∧
        (∀ ℓ : ℕ, d.workTapePos (Fin.last (K + 3)) = (ℓ : ℤ) + 1 →
          (C.runFrom d 1).workTapePos (Fin.last (K + 3)) = (ℓ : ℤ)) := by
  classical
  let C : MultiTapeTM (K + 4) Bool Bool :=
    ⟨false, fun _ _ _ => tapeAction (Fin.last (K + 3)) 0 none (-1) true⟩
  refine ⟨C, rfl, fun z d hs => ?_⟩
  have hr : C.runFrom d 1 = {d with
      state := some true
      workTapePos := Function.update d.workTapePos (Fin.last (K + 3))
        (d.workTapePos (Fin.last (K + 3)) - 1)} := by
    rw [MultiTapeTM.runFrom_succ_eq_step', MultiTapeTM.runFrom_zero]
    have hstep := step_of_tapeAction_move C d (Fin.last (K + 3)) 0 (-1) hs rfl
    simpa only [moveInputPos_zero, SignType.coe_neg, SignType.coe_one, sub_eq_add_neg]
      using hstep
  refine ⟨hr, ?_, ?_⟩
  · intro u hu
    have hu0 : u = 0 := by omega
    subst u
    rw [MultiTapeTM.runFrom_zero, hs]
    exact fun h => Bool.noConfusion (Option.some.inj h)
  · intro ℓ hℓ
    rw [hr]
    change Function.update d.workTapePos (Fin.last (K + 3))
      (d.workTapePos (Fin.last (K + 3)) - 1) (Fin.last (K + 3)) = (ℓ : ℤ)
    rw [Function.update_self, hℓ]
    omega

/-- Skip, then the fixed one-cell correction, yields a literal next-scan presentation.
The full prefix run, dispatches and global space ledger remain Claude OWN obligations.

**Proof sketch.** Specialize the framed skip equation, relabel its live endpoint for the
already checked correction leaf, and use its actual one-step result. Public state-map fields
show the next scan starts at zero while every tape, native head and output is preserved. -/
theorem skip_correction_connection (K : ℕ) :
    ∃ C : MultiTapeTM (K + 4) Bool Bool, C.q₀ = false ∧
      ∀ (z : List Bool) (d : Cfg (K + 4) Bool SeekPhase z) (ell1 : ℕ),
        d.state = some .run → d.workTapePos (Fin.last (K + 3)) = 0 →
        (∀ a, d.workTapes (Fin.last (K + 3)) a =
          FinTM.bufferTape (List.replicate ell1 true) a) →
        let s := (skipTM (K + 4) (Fin.last (K + 3))).runFrom d (ell1 + 1)
        let e := s.mapState (fun _ => false)
        let f := (C.runFrom e 1).mapState (fun _ => (0 : Fin 4))
        f.state = some 0 ∧ f.workTapePos (Fin.last (K + 3)) = (ell1 : ℤ) ∧
          f.workTapes = d.workTapes ∧ f.inputPos = d.inputPos ∧ f.output = d.output := by
  obtain ⟨C, q, correct⟩ := last_ruler_correction K
  refine ⟨C, q, ?_⟩
  intro z d ell1 hs hh hw s e f
  have skip := (skip_endpoint (Fin.last (K + 3)) d ell1 hs hh hw).1
  change s = _ at skip
  have es : e.state = some false := by
    rw [Cfg.mapState_state, skip]
    rfl
  have eh : e.workTapePos (Fin.last (K + 3)) = (ell1 : ℤ) + 1 := by
    rw [Cfg.mapState_workTapePos, skip]
    exact Function.update_self _ _ _
  obtain ⟨run, cut, head⟩ := correct z e es
  refine ⟨?_, ?_, ?_, ?_, ?_⟩
  · rw [Cfg.mapState_state, run]
    rfl
  · rw [Cfg.mapState_workTapePos]
    exact head ell1 eh
  · funext j a
    rw [Cfg.mapState_workTapes, run]
    change e.workTapes j a = d.workTapes j a
    rw [Cfg.mapState_workTapes, skip]
  · rw [Cfg.mapState_inputPos, run]
    change e.inputPos = d.inputPos
    rw [Cfg.mapState_inputPos, skip]
  · rw [Cfg.mapState_output, run]
    change e.output = d.output
    rw [Cfg.mapState_output, skip]

/-! Read-only compiled inventory of the single proposed proof root and their descendants.
Imported theorem proof values are not traversed. Existing same-home legacy helpers remain
outside the changed proof scope. Root types and compiled values are both inspected. -/
open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  let wanted : List String := ["second_scan", "skip_endpoint", "final_rewind", "to_rewind", "installed_bank", "last_ruler_correction", "skip_correction_connection"]
  let kind := fun (ci : ConstantInfo) => match ci with
    | .axiomInfo _ => "axiom" | .defnInfo _ => "definition" | .thmInfo _ => "theorem"
    | .opaqueInfo _ => "opaque" | .quotInfo _ => "quotient" | .inductInfo _ => "inductive"
    | .ctorInfo _ => "constructor" | .recInfo _ => "recursor"
  let mut roots : Array Name := #[]
  for (n, _) in env.constants.toList do
    match n with
    | .str _ base => if wanted.contains base then roots := roots.push n
    | _ => pure ()
  if roots.size != wanted.length then
    throwError "Root count mismatch: {roots.size} for {wanted.length} source roots"
  for root in roots.qsort Name.lt do
    logInfo m!"ROOT|{root}|private={isPrivateName root}"
  let mut selected : Array Name := #[]
  for (n, _) in env.constants.toList do
    if roots.any (fun r => r.isPrefixOf n) then selected := selected.push n
  for n in selected.qsort Name.lt do
    let some ci := env.find? n | throwError "selected constant missing {n}"
    let axs ← liftCoreM <| Lean.collectAxioms n
    logInfo m!"PROOFAX|{n}|kind={kind ci}|{(axs.qsort Name.lt).toList}"
    let used := ci.type.getUsedConstants ++ (ci.value?.map (·.getUsedConstants)).getD #[]
    let mut seen : Array Name := #[]
    for c in used do
      if seen.contains c then continue
      seen := seen.push c
      match env.getModuleIdxFor? c, env.find? c with
      | some j, some (.defnInfo di) =>
        let owner := env.header.moduleNames[j.toNat]!
        if (`TCSlib).isPrefixOf owner then
          let hint := match di.hints with
            | .abbrev => "abbrev" | .opaque => "opaque" | .regular _ => "regular"
          logInfo m!"FOREIGNDEF|{n}|{c}|{owner}|hints={hint}"
      | none, some (.defnInfo _) =>
        logInfo m!"HOMEDEF|{n}|{c}"
      | _, _ => pure ()
