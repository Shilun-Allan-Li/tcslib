namespace Turing.NTimeDiag

private theorem rv_core_word (B α N : List Bool) :
    qyBody false 3 B ++ [false, false, true, true] ++ qyBody false 1 α ++ [false, true] ++
      qyBody false 0 N = pairEncode (pairEncode B α) N := by
  have h := qy_writer_word true B (α ++ [true]) N
  have hb : qyBody true 1 (α ++ [true]) = qyBody false 1 α := by
    unfold qyBody
    rw [if_pos rfl, if_neg Bool.false_ne_true, List.dropLast_concat]
  rw [hb, List.dropLast_concat] at h
  simp only [List.nil_append, List.append_nil, List.singleton_append, List.cons_append,
    List.append_assoc] at h
  simpa only [List.append_assoc] using (List.cons.inj h).2

end Turing.NTimeDiag
