Seal controls for the review probe's five extra sealed definitions (`Cfg.ofWords`, `stateWord`,
`Cfg.mapState`, `seamReleaseTM`, `Cfg.Stopped`): each goal closes by `rfl` unsealed and fails
sealed. Expected: exactly five errors, all in the sealed section.
-/

namespace Turing.NTimeDiag

section Unsealed

example (k : ℕ) (S : Type) (x : List Bool) (q : S) (w : Fin k → List Bool) :
    (Cfg.ofWords (input := x) q w).state = some q := rfl
example (k : ℕ) (s : List Bool) (i : Fin k) : stateWord k s i = if (i : ℕ) = 0 then s else [] :=
  rfl
example (k : ℕ) (S T : Type) (x : List Bool) (f : S → T) (c : Cfg k Bool S x) :
    (c.mapState f).state = c.state.map f := rfl
example (k : ℕ) (S : Type) (M : MultiTapeTM k Bool S) (a : S) :
    (seamReleaseTM M a).q₀ = Sum.inl () := rfl
example (k : ℕ) (S : Type) (x : List Bool) (p : S → Bool) (c : Cfg k Bool S x) :
    c.Stopped p = (c.state = none ∨ ∃ s, c.state = some s ∧ p s = true) := rfl

end Unsealed
