
open Lean Meta Elab Command in
elab "#eq_kinds " ids:ident* : command => do
  let env ← getEnv
  let mut lines : Array String := #[]
  for id in ids do
    let n := id.getId
    let cands ← liftCoreM <| (do
      try
        let r ← resolveGlobalConst id
        pure r
      catch _ => pure [])
    if cands.isEmpty then
      lines := lines.push s!"{n} | not a global constant (local hypothesis, let or bound name)"
    else
      for c in cands do
        let k ← liftTermElabM <| EQAudit.kindOf c
        let m := (EQAudit.modName? env c).getD `this_module
        lines := lines.push s!"{n} | {c} | {k} | {m}"
  logInfo m!"{String.intercalate "\n" lines.toList}"

open Lean Meta Elab Command in
elab "#eq_simpdefs" : command => do
  let env ← getEnv
  let st ← liftTermElabM getSimpTheorems
  let mut out : Array String := #[]
  for (n, ci) in env.constants.map₁.toList do
    if let .defnInfo _ := ci then
      if EQAudit.isTCSlib env n && (st.isDeclToUnfold n || st.isLemma (.decl n)) then
        out := out.push s!"{n} | {(EQAudit.modName? env n).getD `none}"
  logInfo m!"TCSLIB @[simp] DEFINITIONS {out.size}:\n{String.intercalate "\n" (out.qsort (· < ·)).toList}"
