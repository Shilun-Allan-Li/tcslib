
/-! ## Review audit (appended by the reviewer; not part of the module) -/

open Lean Meta Elab Command

namespace EQAudit2

def modName? (env : Environment) (n : Name) : Option Name :=
  (env.getModuleIdxFor? n).bind fun i => env.header.moduleNames[i.toNat]?

def isTCSlib (env : Environment) (n : Name) : Bool :=
  match modName? env n with
  | some m => m.getRoot == `TCSlib
  | none => false

def isLocal (env : Environment) (n : Name) : Bool :=
  (env.getModuleIdxFor? n).isNone && (env.find? n).isSome

def usedOf (env : Environment) (n : Name) : NameSet :=
  match env.find? n with
  | some ci => ci.getUsedConstantsAsSet
  | none => {}

def typeConsts (env : Environment) (n : Name) : NameSet :=
  match env.find? n with
  | some ci => ci.type.getUsedConstantsAsSet
  | none => {}

/-- The name with any private prefix removed. -/
def pub (n : Name) : Name := (privateToUserName? n).getD n

def sortNames (s : Array Name) : Array Name := s.qsort (fun a b => a.toString < b.toString)

def kindOf (n : Name) : MetaM String := do
  let env ← getEnv
  match env.find? n with
  | some (.thmInfo _) => return "thm"
  | some (.defnInfo _) =>
    let r ← getReducibilityStatus n
    let inst ← isInstance n
    let proj := env.isProjectionFn n
    let matcher := Lean.Meta.isMatcherCore env n
    let st ← getSimpTheorems
    let simpd := st.isDeclToUnfold n || st.isLemma (.decl n)
    let rs := match r with
      | .reducible => "reducible" | .semireducible => "semireducible" | .irreducible => "irreducible"
    return s!"def[{rs}{if inst then ",instance" else ""}{if proj then ",projection" else ""}{if matcher then ",matcher" else ""}{if simpd then ",@[simp]" else ""}]"
  | some (.ctorInfo _) => return "ctor"
  | some (.inductInfo _) => return "inductive"
  | some (.recInfo _) => return "recursor"
  | some (.opaqueInfo _) => return "opaque"
  | some (.axiomInfo _) => return "axiom"
  | some (.quotInfo _) => return "quot"
  | none => return "?"

/-- Local constants reached from `roots` (not entering `stops`), and the external constants
(including `stops`) that they use directly. -/
def walkLocal (env : Environment) (roots : Array Name) (stops : NameSet) : NameSet × NameSet := Id.run do
  let mut visited : NameSet := {}
  let mut ext : NameSet := {}
  let mut stack := roots
  while !stack.isEmpty do
    let c := stack.back!
    stack := stack.pop
    if visited.contains c then continue
    visited := visited.insert c
    for d in (usedOf env c).toList do
      if isLocal env d && !stops.contains d then
        unless visited.contains d do stack := stack.push d
      else
        ext := ext.insert d
  return (visited, ext)

/-- Every local constant's user name and kind, sorted (for the base/head comparison). -/
elab "#eq_locals" : command => do
  let env ← getEnv
  let mut out : Array String := #[]
  for (n, _) in env.constants.map₂.toList do
    let k ← liftTermElabM <| kindOf n
    out := out.push s!"LOCAL {if isPrivateName n then "private" else "public"} {pub n} | {k}"
  out := out.qsort (· < ·)
  logInfo m!"LOCALS {out.size}:\n{String.intercalate "\n" out.toList}"

/-- Axioms of every local constant: the ones outside the standard three, and the counts. -/
elab "#eq_axioms_all" : command => do
  let env ← getEnv
  let mut std : Nat := 0
  let mut bad : Array String := #[]
  for (n, _) in env.constants.map₂.toList do
    let axs ← liftCoreM <| collectAxioms n
    if axs.all (stdAx.contains ·) then std := std + 1
    else bad := bad.push s!"{pub n}: {axs}"
  bad := bad.qsort (· < ·)
  logInfo m!"AXIOMS-ALL: {std} local constants within the standard three; outside it {bad.size}:\n{String.intercalate "\n" bad.toList}"

/-- The walk from every new local constant and the target, stopping at the frozen declarations. -/
elab "#eq_walk" : command => do
  let env ← getEnv
  let frz := frozen
  let mut stops : NameSet := {}
  let mut roots : Array Name := #[]
  for (n, _) in env.constants.map₂.toList do
    let u := pub n
    -- a frozen declaration or one of its auxiliaries
    if frz.any (fun f => f == u || f.isPrefixOf u) then stops := stops.insert n
    else roots := roots.push n
  let (loc, ext) := walkLocal env roots stops
  logInfo m!"ROOTS {roots.size} (every local constant other than the frozen declarations and their auxiliaries); reached {loc.size}"
  let rootsU := sortNames (roots.map pub)
  logInfo m!"ROOT NAMES:\n{rootsU}"
  let extA := sortNames ext.toList.toArray
  let tcs := extA.filter (isTCSlib env ·)
  let stopHits := extA.filter (stops.contains ·)
  logInfo m!"FROZEN DECLARATIONS USED by the new code: {sortNames (stopHits.map pub)}"
  let other := extA.filter (fun n => !isTCSlib env n && !isLocal env n)
  let mut lines : Array String := #[]
  for n in tcs do
    let k ← liftTermElabM <| kindOf n
    lines := lines.push s!"{n} | {k} | {(modName? env n).getD `none}"
  logInfo m!"DIRECT-TCSLIB {tcs.size}:\n{String.intercalate "\n" lines.toList}"
  let mut olines : Array String := #[]
  for n in other do
    let k ← liftTermElabM <| kindOf n
    olines := olines.push s!"{n} | {k} | {(modName? env n).getD `none}"
  logInfo m!"DIRECT-CORE/MATHLIB {other.size}:\n{String.intercalate "\n" olines.toList}"
  let priv := extA.filter fun n => isPrivateName n && !isLocal env n
  logInfo m!"DIRECT external private names: {priv}"
  let r3 := extA.filter fun n =>
    match n with
    | .str _ s => ["copyTM_run", "transferTM_run", "clearTM_run", "compareTM_run",
        "incrementTM_run_succ", "incrementTM_run_overflow"].contains s
    | _ => false
  logInfo m!"DIRECT R3 rows: {r3}"
  -- TCSlib definitions in the statements of the directly used TCSlib constants, and of the
  -- new local constants' types (statement-reached)
  let mut d2 : NameSet := {}
  for n in tcs ++ (sortNames loc.toList.toArray) do
    for d in (typeConsts env n).toList do
      if isTCSlib env d && !ext.contains d then d2 := d2.insert d
  let mut l2 : Array String := #[]
  for n in sortNames d2.toList.toArray do
    let k ← liftTermElabM <| kindOf n
    if k.startsWith "def" || k == "opaque" then
      l2 := l2.push s!"{n} | {k} | {(modName? env n).getD `none}"
  logInfo m!"STATEMENT-REACHED TCSlib definitions (not direct) {l2.size}:\n{String.intercalate "\n" l2.toList}"

/-- Every TCSlib definition outside this module that is semireducible, not an instance, not a
projection and not a matcher: the maximal seal's candidates. -/
elab "#eq_alldefs" : command => do
  let env ← getEnv
  let st ← liftTermElabM getSimpTheorems
  let mut out : Array String := #[]
  let mut simpd : Array String := #[]
  for (n, ci) in env.constants.map₁.toList do
    if let .defnInfo _ := ci then
      if isTCSlib env n && !isPrivateName n && !n.isInternalDetail then
        let r ← getReducibilityStatus n
        let inst ← liftTermElabM <| isInstance n
        let proj := env.isProjectionFn n
        let matcher := Lean.Meta.isMatcherCore env n
        if st.isDeclToUnfold n || st.isLemma (.decl n) then
          simpd := simpd.push s!"{n}"
        if r == .semireducible && !inst && !proj && !matcher then
          out := out.push s!"{n}"
  out := out.qsort (· < ·)
  simpd := simpd.qsort (· < ·)
  logInfo m!"ALLDEFS {out.size}:\n{String.intercalate "\n" out.toList}"
  logInfo m!"SIMPDEFS {simpd.size}:\n{String.intercalate "\n" simpd.toList}"
