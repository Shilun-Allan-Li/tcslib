
open Lean Meta Elab Command in
elab "#eq_kinds " ids:ident* : command => do
  let env ← getEnv
  let mut lines : Array String := #[]
  for id in ids do
    let n := id.getId
    let cands ← liftCoreM <| (do
      try
        let r ← resolveGlobalConst id
        pure r
      catch _ => pure [])
    if cands.isEmpty then
      lines := lines.push s!"KIND {n} | - | local (not a global constant) | -"
    else
      for c in cands do
        let k ← liftTermElabM <| EQAudit2.kindOf c
        let m := (EQAudit2.modName? env c).getD `this_module
        lines := lines.push s!"KIND {n} | {EQAudit2.pub c} | {k} | {m}"
  logInfo m!"{String.intercalate "\n" lines.toList}"
