/-! Reviewer probe: resolve the identifiers named in the new code's rw/simp/unfold lists and show
terms (kind, reducibility, simp status, module). -/
open Lean Elab Command

elab "#dz_origin " ids:ident* : command => do
  let env ← getEnv
  let sth ← liftCoreM Meta.getSimpTheorems
  let mut out := ""
  for id in ids do
    let r : Option (List Name) ← try (some <$> resolveGlobalConst id) catch _ => pure none
    match r with
    | none => out := out ++ s!"UNRESOLVED\t{id.getId}\n"
    | some ns =>
      for n in ns do
        let mod := match env.getModuleIdxFor? n with
          | some i => toString (env.header.moduleNames[i.toNat]!)
          | none => "(this file)"
        let kind := match env.find? n with
          | some (.thmInfo _) => "theorem" | some (.defnInfo _) => "def"
          | some (.axiomInfo _) => "axiom" | some (.opaqueInfo _) => "opaque"
          | some (.inductInfo _) => "inductive" | some (.ctorInfo _) => "ctor"
          | some (.recInfo _) => "recursor" | some (.quotInfo _) => "quot" | none => "?"
        let red ← getReducibilityStatus n
        let redS := match red with | .reducible => "reducible" | .semireducible => "semireducible" | .irreducible => "irreducible"
        let simpS := (if sth.isLemma (.decl n) then "simp-lemma " else "") ++
          (if sth.isDeclToUnfold n then "simp-unfold" else "")
        out := out ++ s!"{id.getId}\t{n}\t{kind}\t{redS}\t{simpS}\t{mod}\n"
  logInfo out
