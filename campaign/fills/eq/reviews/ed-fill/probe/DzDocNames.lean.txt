/-! Reviewer probe: resolve the names cited in the new docstrings. -/
open Lean Elab Command

elab "#rv_resolve " ids:ident* : command => do
  let mut out := ""
  for id in ids do
    let r : Option (List Name) ← try (some <$> resolveGlobalConst id) catch _ => pure none
    match r with
    | none => out := out ++ s!"UNRESOLVED {id.getId}\n"
    | some ns => out := out ++ s!"ok {id.getId} -> {ns}\n"
  logInfo out
