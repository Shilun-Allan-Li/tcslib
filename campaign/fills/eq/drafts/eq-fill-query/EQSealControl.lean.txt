section Unsealed

example (T : ℕ → ℕ) : Complexity.TimeConstructible T =
    ((∀ n, n ≤ T n) ∧ ∃ c : ℕ, 0 < c ∧ ∃ M : FinTM Bool, ∀ x : List Bool,
      M.ComputesInTime x (T x.length).bits (c * (T x.length + 1))) := rfl
example (M : FinTM Bool) (x y : List Bool) (t : ℕ) :
    M.ComputesInTime x y t = ∃ s, M.tm.ComputesInTimeAndSpace x y t s := rfl
example (M : FinTM Bool) (f : List Bool → List Bool) (T : ℕ → ℕ) :
    M.ComputesFunInTime f T = ∀ x, M.ComputesInTime x (f x) (T x.length) := rfl
example (M : FinTM Bool) (x : List Bool) (c : Cfg M.k Bool M.State x) (t : ℕ) :
    M.tm.runFrom c t = M.tm.step^[t] c := rfl
example (M : FinTM Bool) (x : List Bool) : M.tm.initCfg x = Cfg.init M.tm.q₀ x := rfl
example (w : List Bool) : counterValue w = Nat.ofDigits 2 (w.map Bool.toNat) := rfl
example (u v : List Bool) : pairEncode u v = (u.flatMap fun b => [b, b]) ++ [false, true] ++ v :=
  rfl

end Unsealed
