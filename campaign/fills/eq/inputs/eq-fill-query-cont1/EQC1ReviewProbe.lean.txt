/-- The witness, through A1's reading: `n ≤ f n`, and a machine computing `bits (f |x|)` within
`c (f |x| + 1)`, as a function-level computation. -/
theorem qy_witness {f : ℕ → ℕ} (hf : Complexity.TimeConstructible f) :
    (∀ n, n ≤ f n) ∧ ∃ c : ℕ, 0 < c ∧ ∃ M : FinTM Bool,
      M.ComputesFunInTime (fun y => (f y.length).bits) fun m => c * (f m + 1) := by
  obtain ⟨hle, c, hc, M, hM⟩ := (Complexity.timeConstructible_iff f).mp hf
  exact ⟨hle, c, hc, M, (FinTM.computesFunInTime_iff _ _ _).mpr hM⟩

/-- The locator's three outcomes at a rung. -/
inductive QyStep
  | interior
  | top
  | next
  deriving DecidableEq

/-- The locator's decision at the rung `ℓ` on a length `n` (the dossier's `eqDecide`). -/
def qyDecide (f : ℕ → ℕ) (n ℓ : ℕ) : QyStep :=
  if n < f (ℓ + 1) then .interior
  else if Nat.log 2 n < (f (ℓ + 1) + (ℓ + 1) + 1) ^ 4 then .interior
  else if (f (ℓ + 1) + (ℓ + 1) + 1) ^ 4 = Nat.log 2 n then
    (if n = 2 ^ Nat.log 2 n then .top else .next)
  else .next

theorem qy_ladder_succ_eq (f : ℕ → ℕ) (i : ℕ) :
    ladder f (i + 1) = 2 ^ (f (ladder f i + 1) + (ladder f i + 1) + 1) ^ 4 := by
  rw [ladder_succ, budget_eq, ← pow_mul]

/-- The stop rule: at a rung below `n`, the decision is the comparison with the next rung. -/
theorem qy_decide_spec (f : ℕ → ℕ) {n i : ℕ} (hn : 1 ≤ n) :
    (qyDecide f n (ladder f i) = .interior ↔ n < ladder f (i + 1)) ∧
    (qyDecide f n (ladder f i) = .top ↔ n = ladder f (i + 1)) ∧
    (qyDecide f n (ladder f i) = .next ↔ ladder f (i + 1) < n) := by
  rw [qy_ladder_succ_eq]
  set X := f (ladder f i + 1) + (ladder f i + 1) + 1 with hX
  set L := Nat.log 2 n with hL
  have hlo : 2 ^ L ≤ n := Nat.pow_log_le_self 2 (by omega)
  have hhi : n < 2 ^ (L + 1) := Nat.lt_pow_succ_log_self (by norm_num) n
  have hX1 : 1 ≤ X := by omega
  have hX4 : X ≤ X ^ 4 := Nat.le_self_pow (by norm_num) X
  by_cases h1 : n < f (ladder f i + 1)
  · have hbig : n < 2 ^ X ^ 4 := by
      have := Nat.lt_two_pow_self (n := X ^ 4)
      omega
    simp only [qyDecide, h1, if_true]
    refine ⟨by simp; omega, by simp; omega, by simp; omega⟩
  · by_cases h2 : L < X ^ 4
    · have hbig : n < 2 ^ X ^ 4 :=
        lt_of_lt_of_le hhi (Nat.pow_le_pow_right (by norm_num) (by omega))
      simp only [qyDecide, h1, if_false, ← hL, ← hX, h2, if_true]
      refine ⟨by simp; omega, by simp; omega, by simp; omega⟩
    · by_cases h3 : X ^ 4 = L
      · simp only [qyDecide, h1, if_false, ← hL, ← hX, h3, if_true]
        by_cases h4 : n = 2 ^ L
        · simp only [h4, if_true]
          refine ⟨by simp, by simp, by simp⟩
        · simp only [h4, if_false]
          have : 2 ^ L < n := lt_of_le_of_ne hlo (Ne.symm h4)
          refine ⟨by simp; omega, by simp, by simp; omega⟩
      · have hlt : X ^ 4 < L := by omega
        have hsmall : 2 ^ X ^ 4 < n :=
          lt_of_lt_of_le (Nat.pow_lt_pow_right (by norm_num) hlt) hlo
        simp only [qyDecide, h1, if_false, ← hL, ← hX, h2, h3]
        refine ⟨by simp; omega, by simp; omega, by simp; omega⟩

/-- At a rung `ℓᵢ < n`, a decision other than `next` locates the stage: `stage f n = i`. -/
theorem qy_stage_of_stop (f : ℕ → ℕ) {n i : ℕ} (hlo : ladder f i < n)
    (hstop : qyDecide f n (ladder f i) ≠ .next) : stage f n = i := by
  have hn : 1 ≤ n := by have := le_ladder f i; omega
  obtain ⟨-, -, hnext⟩ := qy_decide_spec f (i := i) hn
  have hle : n ≤ ladder f (i + 1) := by
    by_contra h
    exact hstop (hnext.mpr (by omega))
  exact stage_eq f hlo hle

/-- A passed rung's operands are at most the next rung. -/
theorem qy_round_le (f : ℕ → ℕ) (i : ℕ) :
    f (ladder f i + 1) + (ladder f i + 1) + 1 ≤ ladder f (i + 1) := by
  rw [qy_ladder_succ_eq]
  set X := f (ladder f i + 1) + (ladder f i + 1) + 1
  have h4 : X ≤ X ^ 4 := Nat.le_self_pow (by norm_num) X
  have := Nat.lt_two_pow_self (n := X ^ 4)
  omega

/-- The ledger: the rungs passed before stage `s` sum to at most twice its bottom. -/
theorem qy_ledger (f : ℕ → ℕ) (s : ℕ) :
    ∑ h ∈ Finset.range s, ladder f (h + 1) ≤ 2 * ladder f s := by
  induction s with
  | zero => simp
  | succ s ih =>
    rw [Finset.sum_range_succ]
    have := two_mul_lt_ladder_succ f s
    omega

/-! #### The note's new steps -/

/-- **The all-ones counter's value**: `m` ones are worth `2 ^ m - 1`. -/
theorem qy_ones_val (m : ℕ) :
    Nat.ofDigits 2 ((List.replicate m true).map Bool.toNat) = 2 ^ m - 1 := by
  induction m with
  | zero => rfl
  | succ m ih =>
    rw [List.replicate_succ, List.map_cons, Nat.ofDigits_cons, ih, Bool.toNat_true, pow_succ]
    have := Nat.one_le_two_pow (n := m)
    omega

/-- **The next rung's counter value**: `m` zeros then a one are worth `2 ^ m`. -/
theorem qy_rung_val (m : ℕ) :
    Nat.ofDigits 2 ((List.replicate m false ++ [true]).map Bool.toNat) = 2 ^ m := by
  induction m with
  | zero => rfl
  | succ m ih =>
    rw [List.replicate_succ, List.cons_append, List.map_cons, Nat.ofDigits_cons, ih,
      Bool.toNat_false, pow_succ]
    ring

/-- **The counter is large enough**: the all-ones word of width `|bits n| + |bits D|` is worth at
least `D (n + 1)`. -/
theorem qy_ones_ge (n D : ℕ) :
    D * (n + 1) ≤ 2 ^ ((Nat.bits n).length + (Nat.bits D).length) - 1 := by
  rw [Nat.size_eq_bits_len, Nat.size_eq_bits_len, pow_add]
  have hn := Nat.lt_size_self n
  have hD := Nat.lt_size_self D
  have hprod : (n + 1) * (D + 1) ≤ 2 ^ Nat.size n * 2 ^ Nat.size D :=
    Nat.mul_le_mul hn hD
  have : D * (n + 1) + 1 ≤ (n + 1) * (D + 1) := by nlinarith
  omega

/-- **The counter is not too large**: `2 ^ |bits n| ≤ 2 n + 1`. -/
theorem qy_two_pow_bits_le (n : ℕ) : 2 ^ (Nat.bits n).length ≤ 2 * n + 1 := by
  rcases Nat.eq_zero_or_pos n with rfl | hn
  · simp
  · rw [Nat.size_eq_bits_len]
    have hsz : Nat.size n = Nat.log 2 n + 1 := by
      apply le_antisymm
      · exact Nat.size_le.mpr (Nat.lt_pow_succ_log_self (by norm_num) n)
      · by_contra h
        have hlt : n < 2 ^ Nat.log 2 n :=
          Nat.size_le.mp (by omega)
        exact absurd (Nat.pow_log_le_self 2 (by omega)) (not_le.mpr hlt)
    rw [hsz, pow_succ]
    have := Nat.pow_log_le_self 2 (show n ≠ 0 by omega)
    omega

/-- **`⌊log₂ n⌋` from `bits n`**: for `n ≥ 1`, `Nat.log 2 n + 1 = |bits n|`. -/
theorem qy_log_bits {n : ℕ} (hn : 1 ≤ n) : Nat.log 2 n + 1 = (Nat.bits n).length := by
  rw [Nat.size_eq_bits_len]
  apply le_antisymm
  · by_contra h
    have hlt : n < 2 ^ Nat.log 2 n := Nat.size_le.mp (by omega)
    exact absurd (Nat.pow_log_le_self 2 (by omega)) (not_le.mpr hlt)
  · exact Nat.size_le.mpr (Nat.lt_pow_succ_log_self (by norm_num) n)

/-- **The power-of-two test on `bits n`**: `n = 2 ^ L` exactly when `bits n` is `L` zeros then a
one, where `L = ⌊log₂ n⌋`. -/
theorem qy_pow_two_bits (n : ℕ) :
    n = 2 ^ Nat.log 2 n ↔ Nat.bits n = List.replicate (Nat.log 2 n) false ++ [true] := by
  constructor
  · intro h
    calc Nat.bits n = Nat.bits (2 ^ Nat.log 2 n) := congrArg Nat.bits h
      _ = List.replicate (Nat.log 2 n) false ++ [true] := bits_two_pow_eq_replicate _
  · intro h
    have hv := counterValue_bits n
    rw [h, counterValue_eq, qy_rung_val] at hv
    exact hv.symm

/-- **The install call's bound under `f a ≤ n`** (the timeout certificate's arithmetic): with
`a ≤ n` and `f a ≤ n`, the call's time bound `c' (c (f a + 1) + a + |bits (f a)| + 1)` is at most
`c' (c + 2) (n + 1)`. -/
theorem qy_call_bound (f : ℕ → ℕ) (c c' a n : ℕ) (ha : a ≤ n) (hfa : f a ≤ n) :
    c' * (c * (f a + 1) + a + (Nat.bits (f a)).length + 1) ≤ c' * (c + 2) * (n + 1) := by
  have hb : (Nat.bits (f a)).length ≤ f a := by
    rw [Nat.size_eq_bits_len]
    exact Nat.size_le.mpr (Nat.lt_two_pow_self)
  have h1 : c * (f a + 1) ≤ c * (n + 1) := Nat.mul_le_mul_left c (by omega)
  have h2 : c * (f a + 1) + a + (Nat.bits (f a)).length + 1 ≤ (c + 2) * (n + 1) := by nlinarith
  calc c' * (c * (f a + 1) + a + (Nat.bits (f a)).length + 1) ≤ c' * ((c + 2) * (n + 1)) :=
        Nat.mul_le_mul_left c' h2
    _ = c' * (c + 2) * (n + 1) := by ring

/-- The ladder is at least `2 ^ (i + 1)`. -/
theorem qy_two_pow_le_ladder (f : ℕ → ℕ) (i : ℕ) : 2 ^ (i + 1) ≤ ladder f i := by
  induction i with
  | zero => rw [ladder_zero]; norm_num
  | succ i ih =>
    have := two_mul_lt_ladder_succ f i
    rw [pow_succ]
    omega

/-- **The stage index is below `log₂ n`**: for `n ≥ 3`, `stage f n < Nat.log 2 n`. -/
theorem qy_stage_lt_log (f : ℕ → ℕ) {n : ℕ} (hn : 3 ≤ n) : stage f n < Nat.log 2 n := by
  have hlo := (stage_spec f hn).1
  have hp := qy_two_pow_le_ladder f (stage f n)
  have : stage f n + 1 ≤ Nat.log 2 n := Nat.le_log_of_pow_le (by norm_num) (by omega)
  omega


/-- **F1: the clocked call through the release adapter.** From the install call's published
clauses alone (a positive exit time, the positive cut, the endpoint), the call wrapped by
`seamReleaseTM` at its entry, started at the fresh state and stopped at `Sum.inr exit`, first
stops exactly at the call's exit time. No `entry ≠ exit` is needed. -/
theorem rv_release_first_stop (C : FinTM Bool) (entry exit : C.State) (x arg out : List Bool)
    (t : ℕ) (hpos : 0 < t)
    (hcut : ∀ t', 0 < t' → t' < t →
      (C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) t').state ≠ some exit)
    (hrun : C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) t =
      Cfg.ofWords exit (stateWord C.k out)) :
    ((seamReleaseTM C.tm entry).runFrom
        ((Cfg.ofWords (input := x) entry (stateWord C.k arg)).mapState fun _ => Sum.inl ())
        t).Stopped (fun s => decide (s = Sum.inr exit)) ∧
    (∀ u < t, ¬((seamReleaseTM C.tm entry).runFrom
        ((Cfg.ofWords (input := x) entry (stateWord C.k arg)).mapState fun _ => Sum.inl ())
        u).Stopped (fun s => decide (s = Sum.inr exit))) ∧
    (seamReleaseTM C.tm entry).runFrom
        ((Cfg.ofWords (input := x) entry (stateWord C.k arg)).mapState fun _ => Sum.inl ()) t =
      (Cfg.ofWords (input := x) exit (stateWord C.k out)).mapState Sum.inr := by
  have hc : (Cfg.ofWords (input := x) entry (stateWord C.k arg)).state = some entry :=
    Cfg.ofWords_state entry _
  have hend := seamReleaseTM_runFrom_pos C.tm entry _ hc t hpos
  rw [hrun] at hend
  have hlive : ∀ u ≤ t,
      (C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) u).state ≠ none := by
    refine MultiTapeTM.runFrom_state_ne_none_of_le C.tm _ t ?_
    rw [hrun, Cfg.ofWords_state]
    exact Option.some_ne_none _
  refine ⟨?_, fun u hu => ?_, hend⟩
  · rw [hend, Cfg.stopped_iff, Cfg.mapState_state, Cfg.ofWords_state]
    exact Or.inr ⟨Sum.inr exit, rfl, decide_eq_true rfl⟩
  · rw [Cfg.stopped_iff]
    rintro (hnone | ⟨s, hs, hstop⟩)
    · rcases Nat.eq_zero_or_pos u with rfl | hu0
      · rw [MultiTapeTM.runFrom_zero, Cfg.mapState_state, hc] at hnone
        exact Option.some_ne_none _ hnone
      · rw [seamReleaseTM_runFrom_pos C.tm entry _ hc u hu0, Cfg.mapState_state] at hnone
        exact hlive u hu.le (Option.map_eq_none_iff.mp hnone)
    · have hse : s = Sum.inr exit := of_decide_eq_true hstop
      subst hse
      rcases Nat.eq_zero_or_pos u with rfl | hu0
      · rw [MultiTapeTM.runFrom_zero, Cfg.mapState_state, hc] at hs
        exact Sum.inl_ne_inr (Option.some.inj hs)
      · rw [seamReleaseTM_runFrom_pos C.tm entry _ hc u hu0, Cfg.mapState_state] at hs
        obtain ⟨q, hq, hqe⟩ := Option.map_eq_some_iff.mp hs
        exact hcut u hu0 hu (hq.trans (congrArg some (Sum.inr.inj hqe)))

/-- **F2: a timed-out call is silent.** Every configuration of the install call before its exit
has empty output (output only grows, and the exit's output is `[]`). Its input head, by
contrast, is not determined by any published clause. -/
theorem rv_call_output_nil (C : FinTM Bool) (entry exit : C.State) (x arg out : List Bool)
    (t v : ℕ) (hv : v ≤ t)
    (hrun : C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) t =
      Cfg.ofWords exit (stateWord C.k out)) :
    (C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) v).output = [] := by

  rw [hrun, Cfg.ofWords_output] at h

/-- **F2's fix: the input head goes home from anywhere.** The published input seek
(`Turing.exists_inputSeekTM`, `Build/InputCall`, already imported) at `p = 1`, with the word
`[true]` on its tape `0` and its tape `1` blank, returns the input head to position `1` from
any position, within `C · (|[true]| + pos + 2)` steps, restoring the output (and every work tape
and head). -/
theorem rv_seek_home :
    ∃ (S : Type) (_ : Fintype S) (_ : DecidableEq S) (R : MultiTapeTM 2 Bool S) (entry exit : S)
      (C : ℕ), ∀ {x : List Bool} (d : Cfg 2 Bool S x), d.state = some entry →
        (∀ q : ℤ, -1 ≤ q → q ≤ (([true] : List Bool).length : ℤ) →
          d.workTapes 0 (d.workTapePos 0 + q) = FinTM.bufferTape [true] q) →
        (∀ q, d.workTapes 1 q = none) →
        ∃ t ≤ C * ([true].length + d.inputPos.val + 1 + 1),
          (R.runFrom d t).state = some exit ∧ (R.runFrom d t).inputPos = 1 ∧
            (R.runFrom d t).output = d.output := by
  obtain ⟨S, hS, hdS, R, entry, exit, C, h⟩ := exists_inputSeekTM
  refine ⟨S, hS, hdS, R, entry, exit, C, fun {x} d hd hw hblank => ?_⟩
  have hp : Nat.ofDigits 2 (([true] : List Bool).map Bool.toNat) = 1 := by
    rw [List.map_cons, List.map_nil, Bool.toNat_true, Nat.ofDigits_singleton]
  obtain ⟨t, ht, -, -, hexit, hpos, -, -, hout⟩ :=
    h (x := x) 1 [true] hp (by omega) d hd hw hblank
  exact ⟨t, ht, hexit, Fin.ext (by rw [hpos]; rfl), hout⟩

/-- Doubling twice is quadrupling. -/
theorem rv_flatMap_dbl_dbl (B : List Bool) :
    (B.flatMap fun b => [b, b]).flatMap (fun b => [b, b]) = B.flatMap fun b => [b, b, b, b] := by
  induction B with
  | nil => rfl
  | cons b B ih =>
    rw [List.flatMap_cons, List.flatMap_append, ih, List.flatMap_cons, List.flatMap_cons,
      List.flatMap_cons, List.flatMap_nil, List.append_nil]
    rfl

/-- **F3: the query word.** `pairEncode (pairEncode B α) t` is `B` with every bit written four
times, the inner separator doubled (`[false, false, true, true]`), `α` doubled, the outer
separator `[false, true]`, then `t`. -/
theorem rv_query_word (B α t : List Bool) :
    pairEncode (pairEncode B α) t =
      (B.flatMap fun b => [b, b, b, b]) ++ [false, false, true, true] ++
        (α.flatMap fun b => [b, b]) ++ [false, true] ++ t := by
  rw [pairEncode_eq, pairEncode_eq, List.flatMap_append, List.flatMap_append,
    rv_flatMap_dbl_dbl]
  rfl

/-- **F4: the round's `X`.** With `a = ℓ + 1`, `X = f a + a + 1 = f a + ℓ + 2`: one copy of the
rung and two cells, not "the rung twice and one cell", which would be `f a + 2ℓ + 1`. -/
theorem rv_X_cells (fa ℓ : ℕ) : fa + (ℓ + 1) + 1 = fa + ℓ + 2 := by omega

example : 0 + (2 + 1) + 1 ≠ 0 + 2 * 2 + 1 := by decide

/-- **F9: the ledger's polylogarithmic term is linear**: `(L + 1)³ ≤ 8 · 2^L`, so with
`2^L ≤ n` (`Nat.pow_log_le_self`) the `s · O((L + 1)²)` term, `s < L`, is at most `8 n`. -/
theorem rv_cube_le (L : ℕ) : (L + 1) ^ 3 ≤ 8 * 2 ^ L := by
  induction L with
  | zero => norm_num
  | succ L ih =>
    rcases Nat.lt_or_ge L 3 with h | h
    · have hL : L = 0 ∨ L = 1 ∨ L = 2 := by omega
      rcases hL with rfl | rfl | rfl <;> norm_num
    · have h2 : (L + 1 + 1) ^ 3 ≤ 2 * (L + 1) ^ 3 := by nlinarith [sq_nonneg (L : ℕ)]
      calc (L + 1 + 1) ^ 3 ≤ 2 * (L + 1) ^ 3 := h2
        _ ≤ 2 * (8 * 2 ^ L) := Nat.mul_le_mul_left 2 ih
        _ = 8 * 2 ^ (L + 1) := by rw [pow_succ]; ring



/-! #### The cont1 review's steps -/

/-- **The timeout certificate.** If the install call's exit time `t` is within its published
bound `c' (c (f a + 1) + a + |bits (f a)| + 1)`, `a ≤ n`, and the clock's all-ones counter of
width `|bits n| + |bits (c' (c + 2))|` (value `2 ^ m - 1`) runs out before `t`, then `n < f a`. -/
theorem qy_timeout_cert (f : ℕ → ℕ) (c c' a n t : ℕ) (ha : a ≤ n)
    (ht : t ≤ c' * (c * (f a + 1) + a + (Nat.bits (f a)).length + 1))
    (hlate : 2 ^ ((Nat.bits n).length + (Nat.bits (c' * (c + 2))).length) - 1 < t) :
    n < f a := by
  by_contra h
  have hfa : f a ≤ n := by omega
  have h1 := qy_call_bound f c c' a n ha hfa
  have h2 := qy_ones_ge n (c' * (c + 2))
  exact absurd (hlate.trans_le (ht.trans (h1.trans h2))) (lt_irrefl _)

/-- **The clocked, released install call** (the first-arrival cut). Clock the release adapter of
a call `C` at its entry, with the stop state `Sum.inr exit`, by the step clock with the identity
digit embedding. From the call's published clauses (a positive exit time `t`, the positive cut,
the endpoint) and an all-ones counter of width `m` on the clock's tape:
* if `t ≤ 2 ^ m - 1`, the host reaches `finish` at the call's endpoint, right-tagged, with the
  counter tape patched by a word of width `m`, and no exit earlier;
* if `t > 2 ^ m - 1`, the host reaches some `timeout r`, silent, and no exit earlier.
No `entry ≠ exit` is used. -/
theorem qy_clocked_call (C : FinTM Bool) (entry exit : C.State) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (H : MultiTapeTM (C.k + 1) Bool Q)
      (inj : Unit ⊕ C.State → Q) (finish timeout : Option (Unit ⊕ C.State) → Q),
      Function.Injective finish ∧ Function.Injective timeout ∧ (∀ a b, finish a ≠ timeout b) ∧
      ∀ (x arg out : List Bool) (t : ℕ), 0 < t →
        (∀ t', 0 < t' → t' < t →
          (C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) t').state ≠
            some exit) →
        C.tm.runFrom (Cfg.ofWords (input := x) entry (stateWord C.k arg)) t =
          Cfg.ofWords exit (stateWord C.k out) →
        ∀ (m : ℕ) (ct : ℤ → Option Bool) (cp : ℤ),
          (∀ p : ℤ, -1 ≤ p → p ≤ (m : ℤ) →
            ct (cp + p) = FinTM.bufferTape (List.replicate m true) p) →
        ∀ c₀ : Cfg C.k Bool (Unit ⊕ C.State) x,
          c₀ = (Cfg.ofWords (input := x) entry (stateWord C.k arg)).mapState (fun _ => Sum.inl ()) →
          (t ≤ 2 ^ m - 1 →
            ∃ T ≤ 5 * t + 2 * m + 2, ∃ w' : List Bool, w'.length = m ∧
              H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) T =
                clockCfg (some (finish (some (Sum.inr exit))))
                  ((Cfg.ofWords (input := x) exit (stateWord C.k out)).mapState
                    (Sum.inr : C.State → Unit ⊕ C.State))
                  (wordPatch ct cp w') cp ∧
              ∀ u < T, ∀ r,
                (H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) u).state ≠
                    some (finish r) ∧
                  (H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) u).state ≠
                    some (timeout r)) ∧
          (2 ^ m - 1 < t →
            ∃ T ≤ 5 * (2 ^ m - 1) + 2 * m + 2, ∃ r : Option (Unit ⊕ C.State),
              (H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) T).state =
                  some (timeout r) ∧
                (H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) T).output = [] ∧
              ∀ u < T, ∀ r,
                (H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) u).state ≠
                    some (finish r) ∧
                  (H.runFrom (clockCfg (some (inj (Sum.inl ()))) c₀ ct cp) u).state ≠
                    some (timeout r)) := by
  obtain ⟨Q, hQ, hdQ, H, inj, finish, timeout, hfi, hti, hft, -, hH⟩ :=
    exists_stepClockTMVia (Function.Embedding.refl Bool) (seamReleaseTM C.tm entry)
      (fun s => decide (s = Sum.inr exit))
  refine ⟨Q, hQ, hdQ, H, inj, finish, timeout, hfi, hti, hft, ?_⟩
  intro x arg out t hpos hcut hrun m ct cp hct c₀ hc₀
  obtain ⟨hstop, hfirst, hend⟩ := rv_release_first_stop C entry exit x arg out t hpos hcut hrun
  have hc0state : c₀.state = some (Sum.inl ()) := by
    rw [hc₀, Cfg.mapState_state, Cfg.ofWords_state]
    rfl
  have hid : (List.replicate m true).map (Function.Embedding.refl Bool) =
      List.replicate m true := by
    rw [Function.Embedding.coe_refl, List.map_id]
  have hct' : ∀ p : ℤ, -1 ≤ p → p ≤ ((List.replicate m true).length : ℤ) →
      ct (cp + p) = FinTM.bufferTape ((List.replicate m true).map (Function.Embedding.refl Bool)) p := by
    intro p h1 h2
    rw [hid]
    rw [List.length_replicate] at h2
    exact hct p h1 h2
  obtain ⟨hfin, hto⟩ := hH c₀ (Sum.inl ()) hc0state (List.replicate m true) ct cp hct'
  rw [qy_ones_val m] at hfin hto
  rw [List.length_replicate] at hfin hto
  subst hc₀
  refine ⟨fun hle => ?_, fun hlt => ?_⟩
  · obtain ⟨T, hT, w', hw', hrunH, hcutH, -⟩ := hfin t hle hstop hfirst
    refine ⟨T, hT, w'.map (Function.Embedding.refl Bool), by rw [List.length_map, hw'], ?_, hcutH⟩
    rw [hrunH, hend, Cfg.mapState_state, Cfg.ofWords_state]
    rfl
  · have hns : ∀ u ≤ 2 ^ m - 1, ¬((seamReleaseTM C.tm entry).runFrom
        ((Cfg.ofWords (input := x) entry (stateWord C.k arg)).mapState fun _ => Sum.inl ())
        u).Stopped (fun s => decide (s = Sum.inr exit)) := fun u hu => hfirst u (by omega)
    obtain ⟨T, hT, w', -, hrunH, hcutH, -⟩ := hto hns
    refine ⟨T, hT, ((seamReleaseTM C.tm entry).runFrom
      ((Cfg.ofWords (input := x) entry (stateWord C.k arg)).mapState fun _ => Sum.inl ())
      (2 ^ m - 1)).state, by rw [hrunH, clockCfg_state], ?_, hcutH⟩
    rw [hrunH, clockCfg_output]
    rcases Nat.eq_zero_or_pos (2 ^ m - 1) with h0 | h0
    · rw [h0, MultiTapeTM.runFrom_zero, Cfg.mapState_output, Cfg.ofWords_output]
    · rw [seamReleaseTM_runFrom_pos C.tm entry _ (Cfg.ofWords_state entry _) _ h0,
        Cfg.mapState_output]
      exact rv_call_output_nil C entry exit x arg out t _ hlt.le hrun

/-- **The stop rule along the ladder.** For `n ≥ 3` and `s = stage f n`: every rung up to `s` is
below `n`, the decision at every rung below `s` is `next`, the decision at `s` is not `next`, and
it is `top` exactly at the top `n = ladder f (s + 1)`. -/
theorem qy_rounds_spec (f : ℕ → ℕ) {n : ℕ} (hn : 3 ≤ n) :
    (∀ r ≤ stage f n, ladder f r < n) ∧
    (∀ r < stage f n, qyDecide f n (ladder f r) = .next) ∧
    qyDecide f n (ladder f (stage f n)) ≠ .next ∧
    (qyDecide f n (ladder f (stage f n)) = .top ↔ n = ladder f (stage f n + 1)) := by
  obtain ⟨hlo, hhi⟩ := stage_spec f hn
  have h1 : 1 ≤ n := by omega
  refine ⟨fun r hr => lt_of_le_of_lt ((ladder_strictMono f).monotone hr) hlo,
    fun r hr => ?_, fun h => ?_, (qy_decide_spec f (i := stage f n) h1).2.1⟩
  · have hle : ladder f (r + 1) ≤ ladder f (stage f n) :=
      (ladder_strictMono f).monotone (by omega)
    exact (qy_decide_spec f (i := r) h1).2.2.mpr (by omega)
  · have := (qy_decide_spec f (i := stage f n) h1).2.2.mp h
    omega

/-- **The stop rule, converse**: a run that passes the rungs below `s` and stops at `s` stops at
the stage of `n`. -/
theorem qy_stage_of_run (f : ℕ → ℕ) {n s : ℕ} (hn : 3 ≤ n)
    (hpass : ∀ r < s, qyDecide f n (ladder f r) = .next)
    (hstop : qyDecide f n (ladder f s) ≠ .next) : s = stage f n := by
  have h1 : 1 ≤ n := by omega
  have hlo : ladder f s < n := by
    rcases s with _ | s
    · rw [ladder_zero]
      omega
    · exact (qy_decide_spec f (i := s) h1).2.2.mp (hpass s (by omega))
  exact (qy_stage_of_stop f hlo hstop).symm

/-- The cutoff fourth power: `min (min (X X) K · min (X X) K) K`. -/
def qyCut (X K : ℕ) : ℕ := min (min (X * X) K * min (X * X) K) K

/-- **The cutoff product decides the comparison.** With the cutoff `K = L + 2`, the cut fourth
power is below, equal to, or above `L` exactly when `X ⁴` is; and when `X ⁴ ≤ L`, the inner cut
square is `X ²`. -/
theorem qy_cut_spec (X L : ℕ) :
    (qyCut X (L + 2) < L ↔ X ^ 4 < L) ∧ (qyCut X (L + 2) = L ↔ X ^ 4 = L) ∧
    (X ^ 4 ≤ L → min (X * X) (L + 2) = X ^ 2) := by
  have h4 : X ^ 4 = X * X * (X * X) := by ring
  have h2 : X ^ 2 = X * X := by ring
  unfold qyCut
  rw [h4, h2]
  rcases Nat.lt_or_ge (X * X) (L + 2) with h | h
  · rw [min_eq_left h.le]
    rcases Nat.lt_or_ge (X * X * (X * X)) (L + 2) with h' | h'
    · rw [min_eq_left h'.le]
      exact ⟨Iff.rfl, Iff.rfl, fun _ => rfl⟩
    · rw [min_eq_right h']
      refine ⟨⟨fun h'' => by omega, fun h'' => by omega⟩,
        ⟨fun h'' => by omega, fun h'' => by omega⟩, fun _ => rfl⟩
  · rw [min_eq_right h]
    have hsq : L + 2 ≤ (L + 2) * (L + 2) := Nat.le_mul_self _
    have hX4 : (L + 2) * (L + 2) ≤ X * X * (X * X) := Nat.mul_le_mul h h
    rw [min_eq_right hsq]
    refine ⟨⟨fun h'' => by omega, fun h'' => by omega⟩,
      ⟨fun h'' => by omega, fun h'' => by omega⟩, fun h'' => by omega⟩

/-- The locator's decision as the machine computes it: `n < f a` (the binary compare), then
`L < X` (the unary compare), then the cut fourth power against `L`, with the power-of-two fold's
verdict `pow2` at equality. -/
def qyMachineDecide (n fa X L : ℕ) (pow2 : Bool) : QyStep :=
  if n < fa then .interior
  else if L < X then .interior
  else if qyCut X (L + 2) < L then .next
  else if qyCut X (L + 2) = L then (if pow2 then .top else .next)
  else .interior

/-- **The machine's decision is the stop rule's**: with `L = |bits n| - 1`, `X = f a + ℓ + 2`
and the fold's verdict on `bits n`, the machine decides exactly `qyDecide`. -/
theorem qy_machineDecide_eq (f : ℕ → ℕ) {n ℓ : ℕ} (hn : 1 ≤ n) :
    qyMachineDecide n (f (ℓ + 1)) (f (ℓ + 1) + ℓ + 2) ((Nat.bits n).length - 1)
        (decide (Nat.bits n = List.replicate ((Nat.bits n).length - 1) false ++ [true])) =
      qyDecide f n ℓ := by
  have hL : (Nat.bits n).length - 1 = Nat.log 2 n := by
    have := qy_log_bits hn
    omega
  have hpow : decide (Nat.bits n = List.replicate ((Nat.bits n).length - 1) false ++ [true]) =
      decide (n = 2 ^ Nat.log 2 n) :=
    decide_eq_decide.mpr (by rw [hL]; exact (qy_pow_two_bits n).symm)
  rw [hpow, hL, ← rv_X_cells (f (ℓ + 1)) ℓ]
  obtain ⟨hlt, heq, -⟩ := qy_cut_spec (f (ℓ + 1) + (ℓ + 1) + 1) (Nat.log 2 n)
  have hX4 : f (ℓ + 1) + (ℓ + 1) + 1 ≤ (f (ℓ + 1) + (ℓ + 1) + 1) ^ 4 :=
    Nat.le_self_pow (by norm_num) _
  unfold qyMachineDecide qyDecide
  by_cases h1 : n < f (ℓ + 1)
  · rw [if_pos h1, if_pos h1]
  · rw [if_neg h1, if_neg h1]
    by_cases h2 : Nat.log 2 n < f (ℓ + 1) + (ℓ + 1) + 1
    · rw [if_pos h2, if_pos (lt_of_lt_of_le h2 hX4)]
    · rw [if_neg h2]
      set X := f (ℓ + 1) + (ℓ + 1) + 1 with hXdef
      set L := Nat.log 2 n with hLdef
      rcases lt_trichotomy (X ^ 4) L with h3 | h3 | h3
      · rw [if_pos (hlt.mpr h3), if_neg (show ¬L < X ^ 4 by omega),
          if_neg (show ¬X ^ 4 = L by omega)]
      · rw [if_neg (show ¬qyCut X (L + 2) < L from fun h => absurd (hlt.mp h) (by omega)),
          if_pos (heq.mpr h3), if_neg (show ¬L < X ^ 4 by omega), if_pos h3]
        by_cases hp : n = 2 ^ L
        · rw [if_pos (decide_eq_true hp), if_pos hp]
        · rw [if_neg (show ¬decide (n = 2 ^ L) = true by simpa using hp), if_neg hp]
      · rw [if_neg (show ¬qyCut X (L + 2) < L from fun h => absurd (hlt.mp h) (by omega)),
          if_neg (show ¬qyCut X (L + 2) = L from fun h => absurd (heq.mp h) (by omega)),
          if_pos h3]

/-- **At the top the cut square is the budget**: when `X ⁴ = L`, with `X = f (ℓ + 1) + ℓ + 2`,
the inner cut square is `budget f ℓ`. -/
theorem qy_top_budget (f : ℕ → ℕ) (ℓ L : ℕ) (h : (f (ℓ + 1) + ℓ + 2) ^ 4 = L) :
    min ((f (ℓ + 1) + ℓ + 2) * (f (ℓ + 1) + ℓ + 2)) (L + 2) = budget f ℓ := by
  rw [(qy_cut_spec (f (ℓ + 1) + ℓ + 2) L).2.2 h.le, budget_eq, rv_X_cells (f (ℓ + 1)) ℓ]

/-- **The polylogarithmic terms are linear**: for `1 ≤ n` and `h ≤ ⌊log₂ n⌋`,
`(h + 1) ³ ≤ 8 n`. -/
theorem qy_poly_log_le {n : ℕ} (hn : 1 ≤ n) {h : ℕ} (hh : h ≤ Nat.log 2 n) :
    (h + 1) ^ 3 ≤ 8 * n := by
  have h1 : (h + 1) ^ 3 ≤ (Nat.log 2 n + 1) ^ 3 := Nat.pow_le_pow_left (by omega) 3
  have h2 := rv_cube_le (Nat.log 2 n)
  have h3 : 2 ^ Nat.log 2 n ≤ n := Nat.pow_log_le_self 2 (by omega)
  omega

/-- **The loop's ledger.** If each passed round `r < stage f n` costs at most
`A₁ (ℓ_{r+1} + (L + 1)²)` and the stopping round at most `A₂ (n + (L + 1)²)`, with
`L = ⌊log₂ n⌋`, the loop costs at most `(10 A₁ + 9 A₂)(n + 1)`. -/
theorem qy_loop_ledger (f : ℕ → ℕ) {n : ℕ} (hn : 3 ≤ n) (A₁ A₂ τs : ℕ) (τ : ℕ → ℕ)
    (hτ : ∀ r < stage f n, τ r ≤ A₁ * (ladder f (r + 1) + (Nat.log 2 n + 1) ^ 2))
    (hs : τs ≤ A₂ * (n + (Nat.log 2 n + 1) ^ 2)) :
    ∑ r ∈ Finset.range (stage f n), τ r + τs ≤ (10 * A₁ + 9 * A₂) * (n + 1) := by
  have hsL : stage f n < Nat.log 2 n := qy_stage_lt_log f hn
  have hlo : ladder f (stage f n) < n := (stage_spec f hn).1
  have hcube : (Nat.log 2 n + 1) ^ 3 ≤ 8 * n := qy_poly_log_le (by omega) le_rfl
  have hsum : ∑ r ∈ Finset.range (stage f n), τ r ≤
      A₁ * (∑ r ∈ Finset.range (stage f n), ladder f (r + 1) +
        stage f n * (Nat.log 2 n + 1) ^ 2) := by
    calc ∑ r ∈ Finset.range (stage f n), τ r
        ≤ ∑ r ∈ Finset.range (stage f n), A₁ * (ladder f (r + 1) + (Nat.log 2 n + 1) ^ 2) :=
          Finset.sum_le_sum fun r hr => hτ r (Finset.mem_range.mp hr)
      _ = A₁ * (∑ r ∈ Finset.range (stage f n), ladder f (r + 1) +
            stage f n * (Nat.log 2 n + 1) ^ 2) := by
          have key : ∀ s : ℕ, ∑ r ∈ Finset.range s, A₁ * (ladder f (r + 1) + (Nat.log 2 n + 1) ^ 2) =
              A₁ * (∑ r ∈ Finset.range s, ladder f (r + 1) + s * (Nat.log 2 n + 1) ^ 2) := by
            intro s
            induction s with
            | zero => simp
            | succ s ih => rw [Finset.sum_range_succ, Finset.sum_range_succ, ih]; ring
          exact key _
  have hled := qy_ledger f (stage f n)
  have hsq : stage f n * (Nat.log 2 n + 1) ^ 2 ≤ (Nat.log 2 n + 1) ^ 3 := by
    calc stage f n * (Nat.log 2 n + 1) ^ 2 ≤ (Nat.log 2 n + 1) * (Nat.log 2 n + 1) ^ 2 :=
          Nat.mul_le_mul_right _ (by omega)
      _ = (Nat.log 2 n + 1) ^ 3 := by ring
  have hsq2 : (Nat.log 2 n + 1) ^ 2 ≤ (Nat.log 2 n + 1) ^ 3 :=
    Nat.pow_le_pow_right (by omega) (by omega)
  have e1 : ∑ r ∈ Finset.range (stage f n), ladder f (r + 1) +
      stage f n * (Nat.log 2 n + 1) ^ 2 ≤ 10 * n := by omega
  have e2 : n + (Nat.log 2 n + 1) ^ 2 ≤ 9 * n := by omega
  calc ∑ r ∈ Finset.range (stage f n), τ r + τs ≤ A₁ * (10 * n) + A₂ * (9 * n) :=
        Nat.add_le_add (hsum.trans (Nat.mul_le_mul_left _ e1))
          (hs.trans (Nat.mul_le_mul_left _ e2))
    _ ≤ (10 * A₁ + 9 * A₂) * (n + 1) := by nlinarith

/-- **The main machine's bound absorbs into `g`**: phase costs linear in `n + 1` and in
`g n + 1` sum to a constant times `g n + 1` once `n ≤ g n`. -/
theorem qy_main_bound (g : ℕ → ℕ) (hg : ∀ n, n ≤ g n) (B₁ B₂ n T₁ T₂ : ℕ)
    (h₁ : T₁ ≤ B₁ * (n + 1)) (h₂ : T₂ ≤ B₂ * (g n + 1)) :
    T₁ + T₂ ≤ (B₁ + B₂) * (g n + 1) := by
  have := Nat.mul_le_mul_left B₁ (Nat.add_le_add_right (hg n) 1)
  nlinarith

/-- **The writer's emission order**: for `n ≥ 3` the stage query of `1ⁿ` is the flag, the budget
field with every bit written four times, `[false, false, true, true]`, the code `α` doubled (the
binary word of `j + 1` without its top bit, `j = (Nat.unpair i).1`), `[false, true]`, and the
tail. At the top the budget is `(f (ℓ + 1) + ℓ + 2)²` and the tail `1^{ℓ+1}`; in the middle the
budget is `g n` and the tail `1ⁿ` followed by one more `true`. -/
theorem qy_writer_word (f g : ℕ → ℕ) {n : ℕ} (hn : 3 ≤ n) :
    query f g (List.replicate n true) =
      if n = ladder f (stage f n + 1) then
        true :: ((Nat.bits ((f (ladder f (stage f n) + 1) + ladder f (stage f n) + 2) ^ 2)).flatMap
            (fun b => [b, b, b, b]) ++ [false, false, true, true] ++
          ((Nat.bits ((Nat.unpair (stage f n)).1 + 1)).dropLast.flatMap fun b => [b, b]) ++
          [false, true] ++ List.replicate (ladder f (stage f n) + 1) true)
      else
        false :: ((Nat.bits (g n)).flatMap (fun b => [b, b, b, b]) ++ [false, false, true, true] ++
          ((Nat.bits ((Nat.unpair (stage f n)).1 + 1)).dropLast.flatMap fun b => [b, b]) ++
          [false, true] ++ List.replicate n true ++ [true]) := by
  rw [query_eq f g hn, budget_eq, stageCode_eq, rv_query_word, rv_query_word,
    rv_X_cells (f (ladder f (stage f n) + 1)) (ladder f (stage f n)),
    List.replicate_succ' (n := n)]
  simp only [List.append_assoc]

/-- The expansion's unary block: `d` copies of `[true]` are `1ᵈ`. -/
theorem qy_flatten_replicate (d : ℕ) :
    (List.replicate d [true]).flatten = List.replicate d true := by
  induction d with
  | zero => rfl
  | succ d ih => rw [List.replicate_succ, List.flatten_cons, ih, List.singleton_append,
      List.replicate_succ]
