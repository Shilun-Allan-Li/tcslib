
/-! Elaborated-constant walk (scratch; not library code). For every constant this copy declares
whose name is the fill's (`qy`/`Qy` prefixed, their auxiliaries, and the target), collect the
constants its type and value reference, recursing through this copy's own fill constants and
stopping at constants of other modules. Print each other-module TCSlib constant with its kind. -/

open Lean Meta Elab Command in
private def qyWalkMine (n : Name) : Bool :=
  n.components.any fun c =>
    let s := c.toString
    s.startsWith "qy" || s.startsWith "Qy" || s.startsWith "instDecidableEqQy" ||
      s.startsWith "instFintypeQy" || s.startsWith "instReprQy" || s == "exists_queryTM"

open Lean Meta Elab Command in
elab "#qy_walk" : command => do
  let env ← getEnv
  let mut roots : Array Name := #[]
  for (n, _) in env.constants.map₂.toList do
    if qyWalkMine n then roots := roots.push n
  let mut seen : NameSet := {}
  let mut ext : NameSet := {}
  let mut stack := roots
  while !stack.isEmpty do
    let n := stack.back!
    stack := stack.pop
    if seen.contains n then continue
    seen := seen.insert n
    let some ci := env.find? n | continue
    let used := ci.type.getUsedConstants ++ (ci.value?.map (·.getUsedConstants)).getD #[]
    for c in used do
      match env.getModuleIdxFor? c with
      | none => if qyWalkMine c then stack := stack.push c
      | some idx =>
        let m := env.header.moduleNames[idx.toNat]!
        if (`TCSlib).isPrefixOf m then ext := ext.insert c
  let mut lines : Array String := #[]
  for c in ext.toList do
    let some ci := env.find? c | continue
    let idx := (env.getModuleIdxFor? c).get!
    let m := env.header.moduleNames[idx.toNat]!
    let kind := match ci with
      | .thmInfo _ => "theorem" | .defnInfo _ => "def" | .inductInfo _ => "inductive"
      | .ctorInfo _ => "ctor" | .recInfo _ => "rec" | .axiomInfo _ => "axiom"
      | .opaqueInfo _ => "opaque" | .quotInfo _ => "quot"
    let proj := if (env.getProjectionFnInfo? c).isSome then " projection" else ""
    let inst := if isInstanceCore env c then " instance" else ""
    let mtch := if (Match.Extension.getMatcherInfo? env c).isSome then " matcher" else ""
    let red ← liftTermElabM do
      match (← getReducibilityStatus c) with
      | .reducible => pure " reducible" | .irreducible => pure " irreducible"
      | .semireducible => pure ""
    let simp ← liftTermElabM do
      let s ← getSimpTheorems
      pure (if s.isLemma (.decl c) || s.isDeclToUnfold c then " simp" else "")
    lines := lines.push s!"{kind}{proj}{inst}{mtch}{red}{simp} {c} [{m}]"
  logInfo m!"QYWALK roots={roots.size} visited={seen.size} external={lines.size}\n{
    "\n".intercalate (lines.qsort (· < ·)).toList}"

#qy_walk

open Lean Elab Command in
/-- The public (unmangled) names among this copy's fill constants: the derived instances of the
private inductives, if any, and the target. -/
elab "#qy_public" : command => do
  let env ← getEnv
  let mut pub : Array String := #[]
  for (n, _) in env.constants.map₂.toList do
    if qyWalkMine n && !isPrivateName n then pub := pub.push n.toString
  logInfo m!"QYPUBLIC {pub.size}\n{"\n".intercalate (pub.qsort (· < ·)).toList}"

#qy_public
