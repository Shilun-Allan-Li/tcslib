def bounds(s,n):
    m=re.search(r'(?m)^(?:private )?(?:theorem|lemma) '+re.escape(n)+r'\b',s);assert m
    e=re.search(r'(?m)^(?:/--|(?:private )?(?:theorem|lemma|def|instance|structure|inductive)\b|end\b|namespace\b)',s[m.end():]);en=m.end()+e.start() if e else len(s)
    return m.start(),en
def split_args(s):
    args=[];depth=0;st=0
    for i,c in enumerate(s):
        if c in '([{':depth+=1
        elif c in ')]}':depth-=1
        elif c==',' and depth==0:args.append(s[st:i]);st=i+1
    args.append(s[st:]);return args
for target in MAP:
    f=target['file'];n=target['name'].split('.')[-1]
    samehome[f]=re.findall(r'(?m)^(?:private )?(?:noncomputable )?(?:def|abbrev)\s+([A-Za-z_][A-Za-z_0-9.\u0080-\uffff\']*)',actual[f])
    for generation,sources in [('actual-ea1c51',actual),('proposed',proposed)]:
        st,en=bounds(sources[f],n);block=sources[f][st:en]
        for m in re.finditer(r'\b(simp_all|simp_rw|simpa|simp|erw|rw|dsimp|unfold)\b',block):

            args=''
            if pos<len(block) and block[pos]=='[':
                depth=1;e=pos+1
                while depth:
                    if block[e]=='[':depth+=1
                    if block[e]==']':depth-=1
