def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()


def write(name,data):

def parse(path):
    lines=path.read_text().splitlines();i=0
    data=dict(census={},public_axioms={},all_axioms={},root_axioms={},cited=[],pending_absent=[])
    seen_cited=set()
    ax_tags={'PUBLICAX':'public_axioms','ALLAX':'all_axioms','ROOTAX':'root_axioms'}
    while i<len(lines):
        s=lines[i];line=i+1;i+=1
        if s.startswith('CENSUS|'):
            tag,owner,name,kind,pr,internal=s.split('|')
            assert kind in {'theorem','def','axiom','opaque','constructor','recursor','inductive','quotient'}
            assert pr in {'private=true','private=false'} and internal in {'internal=true','internal=false'}
            assert name not in data['census'],('duplicate CENSUS',name)
            data['census'][name]=dict(owner=owner,kind=kind,private=pr=='private=true',internal=internal=='internal=true',line=line)
        elif any(s.startswith(t+'|') for t in ax_tags):
            tag,name,ax=s.split('|',2)
            while not ax.rstrip().endswith(']'):
                assert i<len(lines) and '|' not in lines[i],('broken axiom continuation',line)
                ax+=' '+lines[i].strip();i+=1
            assert ax.startswith('[') and ax.endswith(']')
            values=[v.strip() for v in ax[1:-1].split(',') if v.strip()]
            assert len(values)==len(set(values)) and values==sorted(values)
            key=ax_tags[tag];assert name not in data[key],(tag,name)
            data[key][name]=dict(axioms=values,line=line)
        elif s.startswith('CITED|'):
            _,root,name,owner,kind=s.split('|');assert (root,name) not in seen_cited
            seen_cited.add((root,name));data['cited'].append(dict(root=root,name=name,owner=owner,kind=kind,line=line))
        elif s.startswith('PENDINGABSENT|'):
            _,name=s.split('|');assert name not in data['pending_absent'];data['pending_absent'].append(name)
        else:raise AssertionError(('unrecognized metadata line',line,s))
    census=data['census'];public={n for n,c in census.items() if not c['private']}
    assert set(data['public_axioms'])==public
    assert set(data['root_axioms'])<=set(census)
    assert all(c['root'] in data['root_axioms'] for c in data['cited'])
    if data['all_axioms']:
        assert set(data['all_axioms'])==set(census)
        assert all(data['all_axioms'][n]['axioms']==a['axioms'] for n,a in data['public_axioms'].items())
        assert all(data['all_axioms'][n]['axioms']==a['axioms'] for n,a in data['root_axioms'].items())
    assert set(data['pending_absent'])=={'Turing.pairDiagTM_spaceUsed','Turing.FinTM.incFixedTM_spaceUsed','Turing.MultiTapeTM.exists_first_entry_of_fixed'}
    return data

def classify(n,c,label):
    if c['private']:return 'private kernel name; exact-name bookkeeping only'
    if n in source_names[label] and not source_names[label][n]['private']:return 'source-declared export'
    if n in {'Turing.solveSplit.eq_def','Turing.solveSplitWith.eq_def'}:return 'generated foreign-home equation theorem'
    if n=='Fin.castLEEmb.congr_simp':return 'generated foreign-home congruence theorem'
    if n.startswith(('Turing.FinTM.instDecidableEqFwdMapState','Turing.FinTM.instDecidableEqA2_MapState')):return 'generated deriving-instance family'
    if c['internal']:return 'generated proof/matcher/equation auxiliary'
    return 'generated or synthesized nonprivate constant; no lexical source root'


def dist(d):
    rows=Counter((v['owner'],v['kind'],v['private'],v['internal']) for v in d.values())
    return [dict(owner=o,kind=k,private=p,internal=i,count=n) for (o,k,p,i),n in sorted(rows.items())]


def compare(a,b,la,lb):
    A=a['census'];B=b['census'];shared=set(A)&set(B)
    attrs=['owner','kind','private','internal']
    changed={n:dict(before=A[n],after=B[n],changed=[x for x in attrs if A[n][x]!=B[n][x]])
             for n in sorted(shared) if any(A[n][x]!=B[n][x] for x in attrs)}
    added={n:dict(B[n],classification=classify(n,B[n],lb)) for n in sorted(set(B)-set(A))}
    removed={n:dict(A[n],classification=classify(n,A[n],la)) for n in sorted(set(A)-set(B))}
    axchanged={n:dict(before=a['public_axioms'][n]['axioms'],after=b['public_axioms'][n]['axioms'])
               for n in sorted(set(a['public_axioms'])&set(b['public_axioms']))
               if a['public_axioms'][n]['axioms']!=b['public_axioms'][n]['axioms']}
    return dict(added=added,removed=removed,shared_names=sorted(shared),shared_attribute_changes=changed,
                public_axiom_set_changes=axchanged,added_distribution=dist(added),removed_distribution=dist(removed),
                correspondence_limit='Exact printed kernel names only. Shared name does not prove unchanged body; removed/added private names are not matched as declaration identities. No privacy/home inference across renames.')
