"""Final independent data/hash audit. No Lean, Git mutation, or delivered-code execution."""
from pathlib import Path
import ast,datetime,hashlib,json
P=Path('/private/tmp/mig-post234-consumer-seals-final')
sha=lambda b:hashlib.sha256(b).hexdigest()
summary=json.loads((P/'SUMMARY.json').read_text());checks=json.loads((P/'FINAL-CHECKS.json').read_text())
assert summary['targets']==14 and summary['selected_roots_and_descendants']==31
assert sum(len(x['targets']) for x in checks['full_seals'])==14
assert sum(x['guards'] for x in checks['full_seals'])==116
assert len(checks['selected_AX'])==31

    data=(P/row['packet']).read_bytes();assert len(data)==row['bytes'] and sha(data)==row['sha256']
for row in json.loads((P/'RUN-INDEX.json').read_text()):
    run=json.loads((P/row['result']).read_text());data=(P/row['input']).read_bytes();raw=(P/row['raw']).read_bytes()
    assert sha(data)==run['input_sha256'] and sha(raw)==run['output_sha256'] and len(raw)==run['output_bytes']
for row in json.loads((P/'SOURCE-BINDINGS.json').read_text()):
    data=(P/'api-home-sources'/(row['path']+'.txt')).read_bytes();assert sha(data)==row['sha256'] and len(data)==row['bytes']

    data=(P/row['patch']).read_bytes();assert sha(data)==row['sha256'] and len(data)==row['bytes']
    assert row['forward_exact'] and row['reverse_exact']
for record in checks['original_proposed_header_equalities']:
    assert record['exact_binders'] and 'complete forall theorem types' in record['identity']

# Include the last two own assembly programs as review data, with full stdlib provenance.
meta=json.loads((P/'TOOL-PROVENANCE.json').read_text())
for p in [Path('/private/tmp/mig-write-consumer-seal-report.py'),Path('/private/tmp/mig-finalize-consumer-manifest.py')]:
    b=p.read_bytes();(P/'tools'/(p.name+'.txt')).write_bytes(b)
    t=ast.parse(b);imports=sorted({a.name.split('.')[0] for n in ast.walk(t) if isinstance(n,ast.Import) for a in n.names}|{n.module.split('.')[0] for n in ast.walk(t) if isinstance(n,ast.ImportFrom)})
    meta['own_scripts'].append(dict(path=str(p),bytes=len(b),sha256=sha(b),imports=imports,stdlib_only=True,execution='python3 -I'))
(P/'TOOL-PROVENANCE.json').write_text(json.dumps(meta,ensure_ascii=False,indent=2)+'\n')
f=P/'README.md';s=f.read_text().replace('The actual pushed source remains','The evaluated pushed source pin is');f.write_text(s)

    all_executed_input_and_raw_hashes_verified=True,all_copy_receipts_exact=True,
    all_TCSlib_source_bindings_from_pinned_git_objects=True,all_three_patch_roundtrips_exact=True,

    zero_changed_region_warnings=True,all_895_context_objects_unchanged=True,

files=[]
for p in sorted(P.rglob('*')):
    if not p.is_file():continue
    assert p.suffix not in ['.lean','.olean','.py','.sh']
    b=p.read_bytes();b.decode('utf-8')
    files.append(dict(path=p.relative_to(P).as_posix(),bytes=len(b),sha256=sha(b)))

              files=files,payload_files=len(files),payload_bytes=sum(x['bytes'] for x in files),
