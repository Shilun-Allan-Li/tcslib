def bounds(s,name):
    m=re.search(r'(?m)^(?:private )?(?:theorem|lemma) '+re.escape(name)+r'\b',s);assert m,name
    e=re.search(r'(?m)^(?:/--|(?:private )?(?:theorem|lemma|def|instance|structure|inductive)\b|end\b|namespace\b)',s[m.end():])
    en=m.end()+e.start() if e else len(s)
    doc=s.rfind('/--',0,m.start());assert doc>=0
    depth=0;assign=None
    for i in range(m.end(),en-1):
        if s[i] in '({[':depth+=1
        elif s[i] in ')}]':depth-=1
        elif s[i:i+2]==':=' and depth==0:assign=i;break
    assert assign is not None and assign<en
    return doc,m.start(),assign,en
def diff(before,after):
    return ''.join(''.join(difflib.unified_diff(before[f].splitlines(True),after[f].splitlines(True),fromfile='a/'+f,tofile='b/'+f)) for f in sorted(before) if before[f]!=after[f])
def apply_patch_data(before,patch,reverse=False):
    out=before.copy();lines=patch.splitlines(True);i=0
    while i<len(lines):
        assert lines[i].startswith('--- a/');old=lines[i][6:].rstrip('\n');i+=1
        assert lines[i].startswith('+++ b/');new=lines[i][6:].rstrip('\n');i+=1
        assert old==new and old in before
        source=before[old].splitlines(True);dst=[];cursor=0
        while i<len(lines) and not lines[i].startswith('--- a/'):
            m=re.fullmatch(r'@@ -(\d+)(?:,(\d+))? \+(\d+)(?:,(\d+))? @@[^\n]*\n',lines[i]);assert m,lines[i]
            a,ac,b,bc=m.groups();ac=1 if ac is None else int(ac);bc=1 if bc is None else int(bc)
            start=int(b if reverse else a)-1
            if (bc if reverse else ac)==0:start+=1
            assert start>=cursor;dst.extend(source[cursor:start]);cursor=start;i+=1
            consumed=produced=0
            while i<len(lines) and not lines[i].startswith(('@@ ','--- a/')):
                kind=lines[i][0];text=lines[i][1:];assert kind in ' +-'
                if reverse:kind={'+':'-','-':'+',' ':' '}[kind]
                if kind in ' -':assert cursor<len(source) and source[cursor]==text;(consumed:=consumed+1);cursor+=1
                if kind in ' +':dst.append(text);produced+=1
                i+=1
            assert consumed==(bc if reverse else ac) and produced==(ac if reverse else bc)
        dst.extend(source[cursor:]);out[old]=''.join(dst)
    return out
