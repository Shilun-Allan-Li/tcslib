import TCSlib.Complexity.TuringMachine.Simulation

/-! Proposed additions in Simulation's home, tested through an import.
Only branchTM/AgreeOn and their same-home fields may reduce in these proofs.
The other-file runFrom definition is sealed; its published agreement theorem is used.
This is a proposal probe, not an actual library addition. -/
set_option autoImplicit false
set_option relaxedAutoImplicit false

open Turing Turing.FinTM

namespace BranchReaderProposal

section ProposedHome
attribute [local irreducible] Turing.MultiTapeTM.runFrom

theorem branchTM_k (M₁ M₂ : FinTM Bool) (b : Bool) :
    (branchTM M₁ M₂ b).k = M₁.k + M₂.k := rfl

theorem branchTM_State (M₁ M₂ : FinTM Bool) (b : Bool) :
    (branchTM M₁ M₂ b).State = (M₁.State ⊕ M₂.State) := rfl

theorem branchTM_runFrom_independent (M₁ M₂ : FinTM Bool) (b b' : Bool)
    {x : List Bool}
    (c : Cfg (M₁.k + M₂.k) Bool (M₁.State ⊕ M₂.State) x) (t : ℕ) :
    (branchTM M₁ M₂ b).tm.runFrom c t =
      (branchTM M₁ M₂ b').tm.runFrom c t := by
  have agree : (branchTM M₁ M₂ b').tm.AgreeOn (branchTM M₁ M₂ b).tm Set.univ := by
    rw [Turing.MultiTapeTM.agreeOn_iff]
    intro q _ inp work
    rfl
  exact Turing.MultiTapeTM.runFrom_eq_of_agreeOn agree c t
    (fun _ _ q _ => Set.mem_univ q)

end ProposedHome

#print axioms branchTM_k
#print axioms branchTM_State
#print axioms branchTM_runFrom_independent

section SealedConsumer
attribute [local irreducible] Turing.FinTM.branchTM Turing.MultiTapeTM.runFrom

private def toBranchCfg (M₁ M₂ : FinTM Bool) (b : Bool) {x : List Bool}
    (c : Cfg (M₁.k + M₂.k) Bool (M₁.State ⊕ M₂.State) x) :
    Cfg (branchTM M₁ M₂ b).k Bool (branchTM M₁ M₂ b).State x := by
  rw [branchTM_k, branchTM_State]
  exact c

private def fromBranchCfg (M₁ M₂ : FinTM Bool) (b : Bool) {x : List Bool}
    (c : Cfg (branchTM M₁ M₂ b).k Bool (branchTM M₁ M₂ b).State x) :
    Cfg (M₁.k + M₂.k) Bool (M₁.State ⊕ M₂.State) x := by
  rw [← branchTM_k M₁ M₂ b, ← branchTM_State M₁ M₂ b]
  exact c

private def branchView (M₁ M₂ : FinTM Bool) (b : Bool) :
    MultiTapeTM (M₁.k + M₂.k) Bool (M₁.State ⊕ M₂.State) := by
  rw [← branchTM_k M₁ M₂ b, ← branchTM_State M₁ M₂ b]
  exact (branchTM M₁ M₂ b).tm

-- This is the actual proposed reader applied with explicit raw input.
example (M₁ M₂ : FinTM Bool) (b b' : Bool) {x : List Bool}
    (c : Cfg (M₁.k + M₂.k) Bool (M₁.State ⊕ M₂.State) x) (t : ℕ) :=
  branchTM_runFrom_independent M₁ M₂ b b' c t

-- This checks whether the shape transports cancel while branchTM stays sealed.
example (M₁ M₂ : FinTM Bool) (b : Bool) {x : List Bool}
    (c : Cfg (M₁.k + M₂.k) Bool (M₁.State ⊕ M₂.State) x) :
    fromBranchCfg M₁ M₂ b (toBranchCfg M₁ M₂ b c) = c := by
  unfold fromBranchCfg toBranchCfg
  rfl

end SealedConsumer
end BranchReaderProposal
