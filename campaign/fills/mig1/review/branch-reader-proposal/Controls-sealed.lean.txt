import TCSlib.Complexity.TuringMachine.Simulation
set_option autoImplicit false
set_option relaxedAutoImplicit false
open Turing Turing.FinTM

attribute [local irreducible] Turing.FinTM.branchTM Turing.MultiTapeTM.runFrom
  Turing.FinTM.rightCfg Turing.FinTM.rightAction

-- runFrom: delta at zero, then its complete defining equation.
example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) : M.runFrom c 0 = c := rfl
example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) (t : ℕ) : M.runFrom c t = M.step^[t] c := rfl

-- branchTM: bundle projection delta, then complete constructor equation.
example (M₁ M₂ : FinTM Bool) (b : Bool) :
    (branchTM M₁ M₂ b).k = M₁.k + M₂.k := rfl
example (M₁ M₂ : FinTM Bool) (b : Bool) : branchTM M₁ M₂ b =
    { k := M₁.k + M₂.k
      State := M₁.State ⊕ M₂.State
      tm :=
        { q₀ := cond b (.inl M₁.tm.q₀) (.inr M₂.tm.q₀)
          tr := fun q inp work => match q with
            | .inl q => leftAction M₂.k Sum.inl
                (M₁.tm.tr q inp (fun i => work (Fin.castAdd M₂.k i)))
            | .inr q => rightAction M₁.k Sum.inr
                (M₂.tm.tr q inp (fun i => work (Fin.natAdd M₁.k i))) } } := rfl

-- rightCfg: state delta, then complete configuration equation.
example {k l : ℕ} {S S' : Type} {x : List Bool} (f : S → S')
    (c : Cfg l Bool S x) (tapes : Fin k → ℤ → Option Bool) (heads : Fin k → ℤ) :
    (rightCfg f c tapes heads).state = c.state.map f := rfl
example {k l : ℕ} {S S' : Type} {x : List Bool} (f : S → S')
    (c : Cfg l Bool S x) (tapes : Fin k → ℤ → Option Bool) (heads : Fin k → ℤ) :
    rightCfg f c tapes heads =
      { state := c.state.map f, inputPos := c.inputPos,
        workTapes := Fin.addCases tapes c.workTapes,
        workTapePos := Fin.addCases heads c.workTapePos, output := c.output } := rfl

-- rightAction: input-head delta, then complete action equation.
example {l : ℕ} {S S' : Type} (k : ℕ) (f : S → S') (a : Action l Bool S) :
    (rightAction k f a).inputTape = a.inputTape := rfl
example {l : ℕ} {S S' : Type} (k : ℕ) (f : S → S') (a : Action l Bool S) :
    rightAction k f a =
      { inputTape := a.inputTape,
        workTapes := Fin.addCases (fun _ => (none, 0)) a.workTapes,
        output := a.output, state := a.state.map f } := rfl
