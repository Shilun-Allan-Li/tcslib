-- These are conditional consumer checks. The unfilled specifications are
-- explicit hypotheses, never asserted as proved by this pre-ship.
theorem idRow
    (h : ∀ x, idTM.ComputesInTime x x (x.length + 1)) :
    ∃ (M : FinTM Bool) (c : ℕ), M.ComputesFunInTime id (fun n => c * (n + 1)) := by
  refine ⟨idTM, 1, ?_⟩
  intro x
  simpa only [one_mul] using h x

theorem pairRow
    (h : ∀ x, pairDiagTM.ComputesInTime x (pairEncode x x) (4 * x.length + 5)) :
    ∃ (M : FinTM Bool) (c : ℕ),
      M.ComputesFunInTime (fun x => pairEncode x x) (fun n => c * (n + 1)) := by
  refine ⟨pairDiagTM, 5, ?_⟩
  intro x
  exact (h x).mono (by change 4 * x.length + 5 ≤ 5 * (x.length + 1); omega)

theorem appendRow (b : Bool)
    (h : ∀ x, (appendBitTM b).ComputesInTime x (x ++ [b]) (x.length + 1)) :
    ∃ (M : FinTM Bool) (c : ℕ),
      M.ComputesFunInTime (fun x => x ++ [b]) (fun n => c * (n + 1)) := by
  refine ⟨appendBitTM b, 1, ?_⟩
  intro x
  simpa only [one_mul] using h x

-- The three forwarding-map exports were gated but deliberately left for F-B.
-- Check their exact statement shapes with an abstract witness, and check both
-- existing consumer rows conditionally; this does not construct the witness.
theorem mapRows (W : FinTM Bool → FinTM Bool)
    (ht : ∀ {Mg : FinTM Bool} {g : List Bool → List Bool} {Tg : ℕ → ℕ},
      Mg.ComputesFunInTime g Tg → Monotone Tg →
      (W Mg).ComputesFunInTime
        (fun z => match pairDecode z with
          | some (a, b) => pairEncode a (g b)
          | none => [])
        (fun n => 22 * (n + 1 + Tg n)))
    (hs : ∀ {Mg : FinTM Bool} (Sg : ℕ → ℕ),
      (∀ (y : List Bool) (t : ℕ), Mg.tm.spaceUsed (Mg.tm.initCfg y) t ≤ Sg y.length) →
      Monotone Sg → ∀ (x : List Bool) (t : ℕ),
      (W Mg).tm.spaceUsed ((W Mg).tm.initCfg x) t ≤ Sg x.length + 22 * (x.length + 1)) :
    type_of% @computesFunInTime_pairMapSnd_spaceUsed := by
  intro Mg g Tg Sg hg hTg hgs hSg
  exact ⟨W Mg, 22, ht hg hTg, hs Sg hgs hSg⟩

#print axioms mapRows
