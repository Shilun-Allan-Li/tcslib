import TCSlib.Complexity.TuringMachine.Build.OracleSim

set_option maxHeartbeats 0
set_option relaxedAutoImplicit false
set_option autoImplicit false

open private Turing.oracleChoiceMachine Turing.oracleChoiceSlot from
  TCSlib.Complexity.TuringMachine.Build.OracleSim

namespace OracleSimSanity
open Turing

/-- An immediate query; taking an ordinary source transition would visibly emit true. -/
def source : FinOracleNDTM Bool where
  k := 0
  State := Fin 3
  tm := {
    q₀ := 0
    qQuery := 0
    qYes := 1
    qNo := 2
    tr := fun _ _ _ _ => ⟨0, fun _ => (none, 0), some true, none⟩ }
  wf := by constructor <;> decide

/-- Execute the actual W2-C query row and then its blank-choice answer row.
Proof sketch: reduce the first query and both possible answer controls. -/
theorem exhausted_at_query (O : Language Bool) (x : List Bool) :
    let M := Turing.oracleChoiceMachine source
    (M.tm.runFrom O (M.tm.initCfg x) 1).state ≠ none ∧
      M.tm.runFrom O (M.tm.initCfg x) 2 =
        {M.tm.initCfg x with state := none, output := [false]} := by
  classical
  dsimp only
  let M := Turing.oracleChoiceMachine source
  have first := M.tm.step_of_qQuery O (cfg := M.tm.initCfg x) rfl
  constructor
  · change (M.tm.step O (M.tm.initCfg x)).state ≠ none
    rw [first]
    simp
  · change M.tm.step O (M.tm.step O (M.tm.initCfg x)) = _
    rw [first]
    split_ifs <;>
      simp [M, Turing.oracleChoiceMachine, source, OracleTM.step, OracleTM.initCfg, Cfg.init,
        Cfg.inputSymbol, Cfg.workTapeSymbols, Turing.oracleChoiceSlot, OracleTM.answerAction,
        Action.apply, moveInputPos]

example (x : List Bool) := exhausted_at_query (fun _ : List Bool => False) x
example (x : List Bool) := exhausted_at_query (fun _ : List Bool => True) x

#print axioms exhausted_at_query
end OracleSimSanity
