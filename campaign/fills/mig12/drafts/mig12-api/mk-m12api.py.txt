"""MIG-12's API: `Cfg.liftSymbols`'s and `Action.liftSymbols`'s field equations (SymbolLift),

Asserted edits. Usage: python3 -I mk-m12api.py <worktree root>"""
import sys
root = sys.argv[1]
P = f'{root}/TCSlib/Complexity/TuringMachine'
def edit(path, pairs):
    t = open(path, encoding='utf-8').read()
    for a, b in pairs:
        assert t.count(a) == 1, (path, a[:70], t.count(a))
        t = t.replace(a, b)
    open(path, 'w', encoding='utf-8').write(t)

edit(f'{P}/Build/Convention.lean', [
("""def incFixed : List Bool → Option (List Bool)
  | [] => none
  | false :: rest => some (true :: rest)
  | true :: rest => (incFixed rest).map (false :: ·)
""",
"""def incFixed : List Bool → Option (List Bool)
  | [] => none
  | false :: rest => some (true :: rest)
  | true :: rest => (incFixed rest).map (false :: ·)

/-- The empty word overflows: `incFixed [] = none`. -/
theorem incFixed_nil : incFixed [] = none :=
  rfl

/-- A low `false` absorbs the carry: `incFixed (false :: w) = some (true :: w)`. -/
theorem incFixed_false_cons (w : List Bool) : incFixed (false :: w) = some (true :: w) :=
  rfl

/-- A low `true` passes the carry on: `incFixed (true :: w) = (incFixed w).map (false :: ·)`. -/
theorem incFixed_true_cons (w : List Bool) :
    incFixed (true :: w) = (incFixed w).map (false :: ·) :=
  rfl

/-- **The increment keeps the width:** `incFixed w = some v → |v| = |w|`.

**Proof sketch.** Induct on `w` through the three case equations: the empty word overflows, a
low `false` becomes `true`, and a low `true` becomes `false` before the incremented tail. -/
theorem incFixed_length {w v : List Bool} (h : incFixed w = some v) : v.length = w.length := by
  induction w generalizing v with
  | nil =>
    rw [incFixed_nil] at h
    exact absurd h (by simp)
  | cons b w ih =>
    cases b with
    | false =>
      rw [incFixed_false_cons, Option.some.injEq] at h
      subst v
      rfl
    | true =>
      rw [incFixed_true_cons] at h
      cases he : incFixed w with
      | none =>
        rw [he] at h
        exact absurd h (by simp)
      | some z =>
        rw [he, Option.map_some, Option.some.injEq] at h
        subst v
        rw [List.length_cons, List.length_cons, ih he]
"""),
])

edit(f'{P}/Build/CounterLoop.lean', [
("""      | some v =>
        simp only [hd] at ih
        rcases ih with ⟨hl, hv, hc⟩
        simp [decFixed, hd, Nat.ofDigits_cons, hl]
        omega

/-- The next frozen counter word is obtained by one optional decrement. -/""",
"""      | some v =>
        simp only [hd] at ih
        rcases ih with ⟨hl, hv, hc⟩
        simp [decFixed, hd, Nat.ofDigits_cons, hl]
        omega

/-- **The decrement keeps the width:** `decFixed w = some v → |v| = |w|`, the width clause of
`Turing.counter_dec_spec`'s successful branch.

**Proof sketch.** Specialize `Turing.counter_dec_spec` at `w`; its match reduces by `h`. -/
theorem decFixed_length {w v : List Bool} (h : decFixed w = some v) : v.length = w.length := by
  have hs := counter_dec_spec w
  simp only [h] at hs
  exact hs.1

/-- The next frozen counter word is obtained by one optional decrement. -/"""),
])
print('ok')
