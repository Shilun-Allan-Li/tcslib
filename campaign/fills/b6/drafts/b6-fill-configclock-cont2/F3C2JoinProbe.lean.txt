import TCSlib.Complexity.TuringMachine.Build.StepClock

/-! F3 cont2's join, from public lemmas only (sealed): a constant write on an interval, in the
`if` form of `cw_copy_constant`'s endpoint, is a `wordPatch` of a constant word, and two adjacent
constant writes join into one. `wordPatch_append` is inlined (renamed) because the snapshot
predates it; at issue the probe cites it. -/

namespace Turing

section F3C2Join
attribute [local irreducible] Turing.FinTM.bufferTape Turing.wordPatch

theorem wordPatch_append_inl {Γ : Type*} (ct : ℤ → Option Γ) (cp : ℤ) (u v : List Γ) :
    wordPatch (wordPatch ct cp u) (cp + u.length) v = wordPatch ct cp (u ++ v) := by
  funext q
  have hlen : ((u ++ v).length : ℤ) = u.length + v.length := by simp
  by_cases h₁ : cp ≤ q ∧ q < cp + (u.length : ℤ)
  · have hn : 0 ≤ q - cp := by omega
    rw [wordPatch_of_not_mem _ _ _ (by omega), wordPatch_of_mem _ _ _ h₁.1 h₁.2,
      wordPatch_of_mem _ _ _ h₁.1 (by omega), ← Int.toNat_of_nonneg hn, FinTM.bufferTape_nat,
      FinTM.bufferTape_nat, List.getElem?_append_left (by omega)]
  · by_cases h₂ : cp + (u.length : ℤ) ≤ q ∧ q < cp + u.length + (v.length : ℤ)
    · have hn : 0 ≤ q - (cp + u.length) := by omega
      have hm : 0 ≤ q - cp := by omega
      rw [wordPatch_of_mem _ _ _ h₂.1 h₂.2, wordPatch_of_mem _ _ _ (by omega) (by omega),
        ← Int.toNat_of_nonneg hn, ← Int.toNat_of_nonneg hm, FinTM.bufferTape_nat,
        FinTM.bufferTape_nat, List.getElem?_append_right (by omega)]
      congr 1
      omega
    · rw [wordPatch_of_not_mem _ _ _ h₂, wordPatch_of_not_mem _ _ _ h₁,
        wordPatch_of_not_mem _ _ _ (by omega)]

/-- The constant write's `if` form is the patch of the constant word. -/
theorem cw_probe_fill_eq {Γ : Type} (ct : ℤ → Option Γ) (p : ℤ) (n : ℕ) (e : Γ) :
    (fun q => if p ≤ q ∧ q < p + (n : ℤ) then some e else ct q) =
      wordPatch ct p (List.replicate n e) := by
  funext q
  by_cases h : p ≤ q ∧ q < p + (n : ℤ)
  · have hn : 0 ≤ q - p := by omega
    rw [if_pos h, wordPatch_of_mem _ _ _ h.1 (by rw [List.length_replicate]; exact h.2),
      ← Int.toNat_of_nonneg hn, FinTM.bufferTape_nat,
      List.getElem?_replicate_of_lt (by omega)]
  · rw [if_neg h, wordPatch_of_not_mem _ _ _ (by rw [List.length_replicate]; exact h)]

/-- Two adjacent constant writes join into one. -/
theorem cw_probe_fill_join {Γ : Type} (ct : ℤ → Option Γ) (p : ℤ) (a b : ℕ) (e : Γ) :
    wordPatch (wordPatch ct p (List.replicate a e)) (p + a) (List.replicate b e) =
      wordPatch ct p (List.replicate (a + b) e) := by
  have h := wordPatch_append_inl ct p (List.replicate a e) (List.replicate b e)
  rw [List.length_replicate, ← List.replicate_add] at h
  exact h

end F3C2Join

#print axioms Turing.cw_probe_fill_eq
#print axioms Turing.cw_probe_fill_join

end Turing
