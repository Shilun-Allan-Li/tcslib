/-- Control 1: closes only by unfolding `Turing.Cfg.InWindow`. -/
example {k : ℕ} {S : Type} {x : List Bool} (s : ℕ) (c : Cfg k Bool S x) :
    c.InWindow s = ((∀ i, |c.workTapePos i| ≤ (s : ℤ)) ∧
      ∀ (i : Fin k) (z : ℤ), (s : ℤ) < |z| → c.workTapes i z = none) := rfl

/-- Control 2: closes only by unfolding `Turing.Cfg.inputSymbol`. -/
example : Cfg.inputSymbol
    (⟨none, 0, fun _ _ => none, fun _ => 0, []⟩ : Cfg 0 Bool Unit [true]) = none := rfl

/-- Control 3: closes only by unfolding `Turing.Cfg.mapState`. -/
example {k : ℕ} {S S' : Type} {x : List Bool} (f : S → S') (c : Cfg k Bool S x) :
    (c.mapState f).workTapes = c.workTapes := rfl

/-- Control 4: closes only by unfolding `Turing.Cfg.workTapeSymbols`. -/
example {k : ℕ} {S : Type} {x : List Bool} (c : Cfg k Bool S x) (i : Fin k) :
    c.workTapeSymbols i = c.workTapes i (c.workTapePos i) := rfl

/-- Control 5: closes only by unfolding `Turing.FinTM.bufferTape`. -/
example : FinTM.bufferTape [true] 0 = some true := rfl

/-- Control 6: closes only by unfolding `Turing.FinTM.configBound`. -/
example (M : FinTM Bool) (n s : ℕ) : M.configBound n s =
    (Fintype.card M.State + 1) * (n + 2) * 3 ^ (M.k * (2 * s + 1)) * (2 * s + 1) ^ M.k := rfl

/-- Control 7: closes only by unfolding `Turing.MultiTapeTM.runFrom`. -/
example {k : ℕ} {S : Type} {x : List Bool} (tm : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) : tm.runFrom c 1 = tm.step c := rfl

/-- Control 8: closes only by unfolding `Turing.MultiTapeTM.spaceUsed`. -/
example {k : ℕ} {S : Type} {x : List Bool} (tm : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) (t : ℕ) : tm.spaceUsed c t = ∑ i, tm.spaceUsedByTape c t i := rfl

/-- Control 9: closes only by unfolding `Turing.MultiTapeTM.step`. -/
example {k : ℕ} {S : Type} {x : List Bool} (tm : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) : tm.step { c with state := none } = { c with state := none } := rfl

/-- Control 10: closes only by unfolding `Turing.MultiTapeTM.spaceUsedByTape`. -/
example {k : ℕ} {S : Type} {x : List Bool} (tm : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) (t : ℕ) (i : Fin k) :
    tm.spaceUsedByTape c t i = (tm.visitedByTapeHead c t i).card := rfl

/-- Control 11: closes only by unfolding `Turing.MultiTapeTM.visitedByTapeHead`. -/
example {k : ℕ} {S : Type} {x : List Bool} (tm : MultiTapeTM k Bool S)
    (c : Cfg k Bool S x) (t : ℕ) (i : Fin k) : tm.visitedByTapeHead c t i =
      (Finset.range (t + 1)).image fun t' => (tm.runFrom c t').workTapePos i := rfl

/-- Control 12: closes only by unfolding `Turing.RecSym.digitEmb`. -/
example (Q T : Type) (b : Bool) : RecSym.digitEmb Q T b = RecSym.digit b := rfl

/-- Control 13: closes only by unfolding `Turing.RecordAt`. -/
example {K : ℕ} {S : Type} {x : List Bool} (d : Cfg K Bool S x) (r : Fin K)
    (fs : List (List Bool)) : RecordAt d r fs = ∀ p : ℤ, -1 ≤ p →
      p ≤ ((joinFields fs).length : ℤ) → d.workTapes r (d.workTapePos r + p) = fieldsTape fs p :=
  rfl

/-- Control 14: closes only by unfolding `Turing.Segment`. -/
example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (F : Set (Option S))
    (P : (Fin k → ℤ) → Prop) (a b : Cfg k Bool S x) (t : ℕ) : Segment M F P a b t =
      (M.runFrom a t = b ∧ (∀ u < t, (M.runFrom a u).state ∉ F) ∧
        ∀ u ≤ t, P (M.runFrom a u).workTapePos) := rfl

/-- Control 15: closes only by unfolding `Turing.cfgRecord`. -/
example {k : ℕ} {Q T : Type} {x : List Bool} (s : ℕ) (c : Cfg k Bool Q x) (τ : T) :
    (cfgRecord s c τ).head? = some [RecSym.st c.state] := rfl

/-- Control 16: closes only by unfolding `Turing.clearTM`. -/
example {k : ℕ} (i : Fin k) : (clearTM (Symbol := Bool) k i).q₀ = .sweep := rfl

/-- Control 17: closes only by unfolding `Turing.copyMapTM`. -/
example {k : ℕ} (src dst : Fin k) :
    (copyMapTM (Symbol := Bool) k src dst (fun _ => ()) (fun _ _ => true)).q₀ = .sweep := rfl

/-- Control 18: closes only by unfolding `Turing.fieldOffset`. -/
example : fieldOffset ([[true], [false, true]] : List (List Bool)) 1 = 2 := rfl

/-- Control 19: closes only by unfolding `Turing.fieldsTape`. -/
example (fs : List (List Bool)) (z : ℤ) :
    fieldsTape fs z = if 0 ≤ z then ((joinFields fs)[z.toNat]?).join else none := rfl

/-- Control 20: closes only by unfolding `Turing.joinFields`. -/
example (fs : List (List Bool)) :
    joinFields fs = List.intercalate [none] (fs.map (List.map some)) := rfl

/-- Control 21: closes only by unfolding `Turing.moveInputPos`. -/
example : (moveInputPos (0 : Fin 3) .pos).val = 1 := rfl

/-- Control 22: closes only by unfolding `Turing.phaseHostTM`. -/
example {k : ℕ} {I : Type} {Ph : I → Type} (M : (i : I) → MultiTapeTM k Bool (Ph i))
    (q0 : Σ i, Ph i) (red : (i : I) → Ph i → Option (Σ i, Ph i)) :
    (phaseHostTM M q0 red).q₀ = q0 := rfl

/-- Control 23: closes only by unfolding `Turing.posWidth`. -/
example (n : ℕ) : posWidth n = (Nat.bits (n + 1)).length := rfl

/-- Control 24: closes only by unfolding `Turing.recordLength`. -/
example (k n s : ℕ) : recordLength k n s = k * (2 * s + 1) + posWidth n + k + 4 := rfl

/-- Control 25: closes only by unfolding `Turing.skipBackTM`. -/

/-- Control 26: closes only by unfolding `Turing.skipTM`. -/

/-- Control 27: closes only by unfolding `Turing.tapeAction`. -/
example {k : ℕ} {S : Type} (i : Fin k) (q : S) :

/-- Control 28: closes only by unfolding `Turing.windowRun`. -/
example {k : ℕ} {Q T : Type} {x : List Bool} (M : MultiTapeTM k Bool Q) (s : ℕ)
    (upd : T → Option Bool → T) (c : Cfg k Bool Q x) (τ : T) :
    windowRun M s upd 0 c τ = (.exhausted, c, τ) := rfl

/-- Control 29: closes only by unfolding `Turing.wordPatch`. -/
example (ct : ℤ → Option Bool) (cp : ℤ) (u : List Bool) (q : ℤ) : wordPatch ct cp u q =
    if cp ≤ q ∧ q < cp + (u.length : ℤ) then FinTM.bufferTape u (q - cp) else ct q := rfl
