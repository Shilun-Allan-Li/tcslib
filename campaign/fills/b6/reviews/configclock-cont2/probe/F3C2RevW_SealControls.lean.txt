import TCSlib.Complexity.SpaceComplexity.WindowClock

/-! Rule-10 seal controls: each sealed definition unfolds to its body by `rfl`-level
definitional unfolding (`isDefEq` at default transparency) with nothing sealed, and does not
under the seal. -/

open Lean Meta Elab Command

elab "#cw_control " id:ident : command => do
  let n ← liftCoreM <| realizeGlobalConstNoOverload id
  liftTermElabM do
    let ci ← getConstInfo n
    let some v := ci.value? | throwError "no value"
    let ok ← withTransparency .default <| isDefEq (mkConst n (ci.levelParams.map mkLevelParam)) v
    logInfo m!"control {n}: {if ok then "unfolds (rfl closes)" else "sealed (rfl fails)"}"
