import TCSlib.Complexity.SpaceComplexity.WindowClock

/-! B6 F3, continuation 2: the pre-ship discharge of the route's premises, from public lemmas
only, with every other-file definition the statements mention sealed. Each example is one
prescribed step's premise, derived from what the previous step supplies.

* `f3c2_silence`: an endpoint equation gives the output at every intermediate time
  (`MultiTapeTM.output_prefix`, both ends).
* `f3c2_field`: a record at the head of a tape presents field `j` at its offset as a delimited
  word, the premise of `copyMapTM_run_ofCfg` and `skipTM_run_ofCfg` on the source.
* `f3c2_field_back`: from field `j + 1`'s offset, field `j` is `skipBackTM_run_ofCfg`'s word.
* `f3c2_patch_word`: a word patched onto a blank tape is a delimited word at the patch's start,
  F2's clock precondition and the clock rewind's premise.
* `f3c2_seam_heads`: the head positions of a seam composite at every time, from the two
  lockstep lemmas.
* The space control at the end: the composite's `spaceUsed` from per-tape bounds closes only
  through `MultiTapeTM.spaceUsed`'s definition (kept as a control: it still fails sealed).
* The space step at issue (after #149): the target's space clause, in its exact form, from its
  envelope clause by the published radius lemma, with the space measure sealed. -/

namespace Turing

section F3C2Route

attribute [local irreducible] FinTM.bufferTape wordPatch fieldsTape fieldOffset joinFields
  RecordAt copyMapTM skipTM skipBackTM clearTM MultiTapeTM.runFrom seamCompTM

theorem f3c2_silence {k : ℕ} {Γ S : Type*} {x : List Γ} (M : MultiTapeTM k Γ S)
    (d : Cfg k Γ S x) (T : ℕ) (hT : (M.runFrom d T).output = d.output) :
    ∀ u ≤ T, (M.runFrom d u).output = d.output := by
  intro u hu
  have h0 := MultiTapeTM.output_prefix M d (Nat.zero_le u)
  have h1 := MultiTapeTM.output_prefix M d hu
  rw [MultiTapeTM.runFrom_zero] at h0
  rw [hT] at h1
  exact h1.eq_of_length (le_antisymm h1.length_le h0.length_le)

/-- The mapped copy is silent at every time up to its exit. -/
example {k : ℕ} {Γ Δ : Type*} {x : List Γ} (src dst : Fin k) (hne : src ≠ dst)
    (extract : Γ → Δ) (merge : Option Γ → Δ → Γ) (w : List Γ) (d : Cfg k Γ SweepPhase x)
    (hd : d.state = some .sweep)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
      d.workTapes src (d.workTapePos src + p) = FinTM.bufferTape w p) :
    ∀ u ≤ 2 * w.length + 2,
      ((copyMapTM k src dst extract merge).runFrom d u).output = d.output :=
  f3c2_silence _ d _ (by rw [(copyMapTM_run_ofCfg src dst hne extract merge w d hd hw).1])

theorem f3c2_field {K : ℕ} {Γ S : Type*} {x : List Γ} (d : Cfg K Γ S x) (r : Fin K)
    (fs : List (List Γ)) (hd : RecordAt d r fs) (j : ℕ) (hj : j < fs.length) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (fs[j].length : ℤ) →
      d.workTapes r (d.workTapePos r + fieldOffset fs j + p) = FinTM.bufferTape fs[j] p := by
  intro p hp hlen
  have hle := fieldOffset_add_length_le fs j hj
  rw [add_assoc, (recordAt_iff d r fs).1 hd _ (by omega) (by omega)]
  exact fieldsTape_field fs j hj p hp hlen

theorem f3c2_field_back {K : ℕ} {Γ S : Type*} {x : List Γ} (d : Cfg K Γ S x) (r : Fin K)
    (fs : List (List Γ)) (hd : RecordAt d r fs) (j : ℕ) (hj : j < fs.length) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (fs[j].length : ℤ) →
      d.workTapes r (d.workTapePos r + fieldOffset fs (j + 1) - fs[j].length - 1 + p) =
        FinTM.bufferTape fs[j] p := by
  intro p hp hlen
  have he : d.workTapePos r + (fieldOffset fs (j + 1) : ℤ) - fs[j].length - 1 + p =
      d.workTapePos r + fieldOffset fs j + p := by
    rw [fieldOffset_succ fs j hj]
    push_cast
    ring
  rw [he]
  exact f3c2_field d r fs hd j hj p hp hlen

/-- The record's end: past the last field, `fieldOffset` is the layout's length plus one,
so the source rewind starts one cell past the record's right delimiter. -/
example {Γ : Type*} (fs : List (List Γ)) (hfs : fs ≠ []) :
    fieldOffset fs fs.length = (joinFields fs).length + 1 :=
  (length_joinFields_add_one fs hfs).symm

theorem f3c2_patch_word {Γ : Type*} (ct : ℤ → Option Γ) (hct : ∀ q, ct q = none) (p0 : ℤ)
    (u : List Γ) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (u.length : ℤ) → wordPatch ct p0 u (p0 + p) = FinTM.bufferTape u p := by
  intro p hp hlen
  by_cases hin : 0 ≤ p ∧ p < (u.length : ℤ)
  · rw [wordPatch_of_mem _ _ _ (by omega) (by omega)]
    congr 1
    ring
  · rw [wordPatch_of_not_mem _ _ _ (by omega), hct]
    rcases (show p = -1 ∨ p = (u.length : ℤ) by omega) with rfl | rfl
    · rw [FinTM.bufferTape_left]
    · rw [FinTM.bufferTape_nat, List.getElem?_length]

/-- The clock rewind's first step: a skip over the empty word at the clock's right delimiter
(one step right), whose premise is that delimiter. -/
example {Γ : Type*} (ct : ℤ → Option Γ) (hct : ∀ q, ct q = none) (p0 : ℤ) (u : List Γ) :
    ∀ p : ℤ, 0 ≤ p → p ≤ (([] : List Γ).length : ℤ) →
      wordPatch ct p0 u (p0 + u.length + p) = FinTM.bufferTape ([] : List Γ) p := by
  intro p hp hlen
  have hp0 : p = 0 := by simp at hlen; omega
  subst hp0
  rw [add_zero, f3c2_patch_word ct hct p0 u u.length (by omega) le_rfl, FinTM.bufferTape_nat,
    List.getElem?_length, FinTM.bufferTape_nil]

/-- The clock rewind's second step: from one cell past the right delimiter, the word is
`skipBackTM_run_ofCfg`'s. -/
example {Γ : Type*} (ct : ℤ → Option Γ) (hct : ∀ q, ct q = none) (p0 : ℤ) (u : List Γ) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (u.length : ℤ) →
      wordPatch ct p0 u ((p0 + u.length + 1) - u.length - 1 + p) = FinTM.bufferTape u p := by
  intro p hp hlen
  have he : p0 + (u.length : ℤ) + 1 - u.length - 1 + p = p0 + p := by ring
  rw [he]
  exact f3c2_patch_word ct hct p0 u p hp hlen

/-- F2's clock precondition from the writer's clock: the all-`true` digit word. -/
example (M : FinTM Bool) {T : Type} (ct : ℤ → Option (RecSym M.State T))
    (hct : ∀ q, ct q = none) (p0 : ℤ) (W : ℕ) :
    ∀ p : ℤ, -1 ≤ p → p ≤ ((List.replicate W true).length : ℤ) →
      wordPatch ct p0 (List.replicate W (RecSym.digitEmb M.State T true)) (p0 + p) =
        FinTM.bufferTape ((List.replicate W true).map (RecSym.digitEmb M.State T)) p := by
  intro p hp hlen
  rw [List.map_replicate]
  exact f3c2_patch_word ct hct p0 _ p hp (by simpa using hlen)

/-- The join of a pass's writes into one patch. -/
example {Γ : Type*} (ct : ℤ → Option Γ) (p : ℤ) (e : Γ) (a b : ℕ) :
    wordPatch (wordPatch ct p (List.replicate a e)) (p + a) (List.replicate b e) =
      wordPatch ct p (List.replicate (a + b) e) := by
  have h := wordPatch_append ct p (List.replicate a e) (List.replicate b e)
  rw [List.length_replicate, ← List.replicate_add] at h
  exact h

theorem f3c2_seam_heads {k : ℕ} {Γ : Type*} {x : List Γ} {S₁ S₂ : Type*} [DecidableEq S₁]
    (M₁ : MultiTapeTM k Γ S₁) (ex : S₁) (M₂ : MultiTapeTM k Γ S₂) (en : S₂)
    {c₀ c₁ : Cfg k Γ S₁ x} {T₁ : ℕ} (h₁ : M₁.runFrom c₀ T₁ = c₁) (hexit : c₁.state = some ex)
    (hcut : ∀ t < T₁, (M₁.runFrom c₀ t).state ≠ some ex) :
    (∀ u ≤ T₁, ((seamCompTM M₁ ex M₂ en).runFrom (c₀.mapState Sum.inl) u).workTapePos =
        (M₁.runFrom c₀ u).workTapePos) ∧
      ∀ v, ((seamCompTM M₁ ex M₂ en).runFrom (c₀.mapState Sum.inl) (T₁ + 1 + v)).workTapePos =
        (M₂.runFrom (c₁.mapState fun _ => en) v).workTapePos := by
  refine ⟨fun u hu => ?_, fun v => ?_⟩
  · funext j
    rw [seamCompTM_runFrom_left M₁ ex M₂ en hcut u hu, Cfg.mapState_workTapePos]
  · funext j
    rw [seamCompTM_runFrom_right M₁ ex M₂ en h₁ hexit hcut v, Cfg.mapState_workTapePos]

end F3C2Route

/-! ### The space control

The per-tape bound is public (`MultiTapeTM.spaceUsedByTape_le_card_Icc`). Summing it into the
target's `spaceUsed` clause reads `MultiTapeTM.spaceUsed` as its defining sum: the first example
closes with nothing sealed; the second, the same proof with `MultiTapeTM.spaceUsed` sealed, is
expected to FAIL. No published equation gives the sum. -/

theorem f3c2_space_unsealed {k : ℕ} {Γ P : Type} (R : MultiTapeTM k Γ P) {x : List Γ}
    (d : Cfg k Γ P x) (t r : ℕ)
    (hj : ∀ j, R.spaceUsedByTape d t j ≤ 2 * r + 1) :
    R.spaceUsed d t ≤ k * (2 * r + 1) := by
  calc
    R.spaceUsed d t ≤ ∑ _j : Fin k, (2 * r + 1) := Finset.sum_le_sum (fun j _ => hj j)
    _ = k * (2 * r + 1) := by simp

/-! ### The space step through the space API (#149) -/

section F3C2SpaceAPI
attribute [local irreducible] MultiTapeTM.spaceUsed MultiTapeTM.spaceUsedByTape
  MultiTapeTM.visitedByTapeHead

theorem f3c2_space_api {K : ℕ} {Γ P : Type} (R : MultiTapeTM (K + 1) Γ P) {x : List Γ}
    (d : Cfg (K + 1) Γ P x) (t C L : ℕ)
    (henv : ∀ u ≤ t, ∀ j : Fin (K + 1),
      |(R.runFrom d u).workTapePos j - d.workTapePos j| ≤ ((C * (L + 1) : ℕ) : ℤ)) :
    R.spaceUsed d t ≤ (K + 1) * (2 * C * (L + 1) + 1) := by
  have h := MultiTapeTM.spaceUsed_le_of_abs_le R d t (C * (L + 1)) henv
  rw [mul_assoc 2 C]
  exact h

end F3C2SpaceAPI
