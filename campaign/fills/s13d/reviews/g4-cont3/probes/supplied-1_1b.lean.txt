/-- Target 1 (endpoint): the right pass writes the new right zone (virtual blanks in its inner
half, empty slots in its outer half), leaves the ruler on scratch, restores the scratch head, and
leaves the zoned head at the outer half's first cell. -/
example {k : ℕ} {x : List (ZoneSym k)} (D : zr_RightData k) (d)
    (hp : (zr_rightPassContract x).pre D d) :
    (zr_rightPassContract x).post D d =
      { d with
        state := some (Sum.inr (Sum.inr (Sum.inr SweepPhase.done)))
        workTapes := fun j q =>
          if j = 0 ∧ (zoneStart (D.1 + 1) : ℤ) ≤ q ∧
              q < zoneStart (D.1 + 1) + zoneCapacity (D.1 + 1) then
            some (.inr fun _ => if q < zoneStart (D.1 + 1) + 2 ^ (D.1 + 1) then some none else none)
          else if j = 1 ∧ d.workTapePos 1 ≤ q ∧ q < d.workTapePos 1 + zoneCapacity D.1 then
            some (.inl false)
          else d.workTapes j q
        workTapePos := Function.update d.workTapePos 0
          (zoneStart (D.1 + 1) + zoneCapacity D.1) } := by
  exact zr_rightPass_post D d hp

/-- Target 2 (envelope): the zoned head stays inside the extension row's level-`ℓ` box. -/
example {k : ℕ} {x : List (ZoneSym k)} (D : zr_RightData k) (d)
    (hp : (zr_rightPassContract x).pre D d) :
    -(zoneStart (D.1 + 2) : ℤ) ≤ (zr_rightPassContract x).lo D d 0 ∧
      (zr_rightPassContract x).hi D d 0 ≤ zoneStart (D.1 + 2) := by
  exact zr_rightPass_envelope D d hp
