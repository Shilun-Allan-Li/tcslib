/-- Target 1 (endpoint): the right pass writes the new right zone (virtual blanks in its inner
half, empty slots in its outer half), leaves the ruler on scratch, restores the scratch head, and
leaves the zoned head at the outer half's first cell. -/
example {k : ℕ} {x : List (ZoneSym k)} (D : zr_RightData k) (d)
    (hp : (zr_rightPassContract x).pre D d) :
    (zr_rightPassContract x).post D d =
      { d with
        state := some (Sum.inr (Sum.inr (Sum.inr SweepPhase.done)))
        workTapes := fun j q =>
          if j = 0 ∧ (zoneStart (D.1 + 1) : ℤ) ≤ q ∧
              q < zoneStart (D.1 + 1) + zoneCapacity (D.1 + 1) then
            some (.inr fun _ => if q < zoneStart (D.1 + 1) + 2 ^ (D.1 + 1) then some none else none)
          else if j = 1 ∧ d.workTapePos 1 ≤ q ∧ q < d.workTapePos 1 + zoneCapacity D.1 then
            some (.inl false)
          else d.workTapes j q
        workTapePos := Function.update d.workTapePos 0
          (zoneStart (D.1 + 1) + zoneCapacity D.1) } := by
  exact zr_rightPass_post D d hp

/-- Level-one arithmetic used below (published equations only). -/
private lemma g4c3_levelOne_numbers :
    zoneStart 1 = 5 ∧ zoneStart 2 = 10 ∧ zoneCapacity 0 = 2 ∧ zoneCapacity 1 = 4 := by
  refine ⟨?_, ?_, ?_, ?_⟩
  · norm_num [zoneStart_eq]
  · norm_num [zoneStart_eq]
  · norm_num [zoneCapacity_eq]
  · norm_num [zoneCapacity_eq]

/-- Level-one instance (gate 58, note 11), with scratch head `-7`: ruler cells `-7, -6`; inner
cells `5, 6`; outer cells `7, 8`; neighbours unchanged; heads `7` and `-7`; time `48`. -/
example {k : ℕ} {x : List (ZoneSym k)} (z : Fin k → ZoneContents (0 + 1)) (d)
    (hp : (zr_rightPassContract x).pre ⟨0, z⟩ d) (hq : d.workTapePos 1 = -7) :
    ((zr_rightPassContract x).post ⟨0, z⟩ d).state =
        some (Sum.inr (Sum.inr (Sum.inr SweepPhase.done))) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 0 5 = some (.inr fun _ => some none) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 0 6 = some (.inr fun _ => some none) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 0 7 = some (.inr fun _ => none) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 0 8 = some (.inr fun _ => none) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 0 4 = d.workTapes 0 4 ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 0 9 = d.workTapes 0 9 ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 1 (-7) = some (.inl false) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 1 (-6) = some (.inl false) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 1 (-8) = d.workTapes 1 (-8) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapes 1 (-5) = d.workTapes 1 (-5) ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapePos 0 = 7 ∧
      ((zr_rightPassContract x).post ⟨0, z⟩ d).workTapePos 1 = -7 ∧
      (zr_rightPassContract x).time ⟨0, z⟩ d = 48 := by
  obtain ⟨hs1, -, hc0, hc1⟩ := g4c3_levelOne_numbers
  rw [zr_rightPass_post _ d hp, zr_rightPass_time]
  norm_num [hs1, hc0, hc1, hq, Function.update_of_ne (show (1 : Fin 2) ≠ 0 by decide)]

/-- Level-one envelope: tape 0 within `[0, 9]` (inside `±zoneStart 2 = ±10`), scratch within
`[-8, -4]`. -/
example {k : ℕ} {x : List (ZoneSym k)} (z : Fin k → ZoneContents (0 + 1)) (d)
    (hp : (zr_rightPassContract x).pre ⟨0, z⟩ d) (hq : d.workTapePos 1 = -7) :
    0 ≤ (zr_rightPassContract x).lo ⟨0, z⟩ d 0 ∧ (zr_rightPassContract x).hi ⟨0, z⟩ d 0 ≤ 9 ∧
      -8 ≤ (zr_rightPassContract x).lo ⟨0, z⟩ d 1 ∧
      (zr_rightPassContract x).hi ⟨0, z⟩ d 1 ≤ -4 := by
  obtain ⟨hs1, -, hc0, hc1⟩ := g4c3_levelOne_numbers
  have h := zr_rightPass_bounds (x := x) ⟨0, z⟩ d hp
  simp only [Nat.zero_add, hs1, hc0, hc1, hq] at h
  omega
