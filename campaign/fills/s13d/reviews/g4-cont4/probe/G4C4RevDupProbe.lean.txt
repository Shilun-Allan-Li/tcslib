-- P1 zr_contents_ext
example {ℓ : ℕ} {u v : ZoneContents ℓ} (h1 : u.home = v.home) (h2 : u.left = v.left)
    (h3 : u.right = v.right) : u = v := by exact?

example {k : ℕ} {Γ S S' : Type} {x : List Γ} {c : Cfg k Γ S x} {c' : Cfg k Γ S' x}
    (h : c.inputPos = c'.inputPos) : c.inputSymbol = c'.inputSymbol := by exact?

-- P3 zr_slot_track
example {k ℓ : ℕ} {z z' : Fin k → ZoneContents ℓ} {t : Fin k} (h : z t = z' t) (q : ℤ) :
    (zonedTape z q).map (zoneSlotOf t) = (zonedTape z' q).map (zoneSlotOf t) := by exact?

-- P4 zr_zonedTape_inr
example {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (q : ℤ) (γ : ZoneSym k)

example {k : ℕ} (t : Fin k) (c : Option (ZoneSym k)) (v w : Option (Option Bool)) :

-- P8 zr_joinFields_append
example {Γ : Type} (ws vs : List (List Γ)) (hw : ws ≠ []) (hv : vs ≠ []) :

-- P9 zr_fieldsTape_end
example {Γ : Type} (ws : List (List Γ)) (z : ℤ) (hz : (fieldOffset ws ws.length : ℤ) ≤ z + 1) :

-- P10 zr_bufferTape_map
example {α β : Type} (w : List α) (f : α → β) (q : ℤ) :
    FinTM.bufferTape (w.map f) q = (FinTM.bufferTape w q).map f := by exact?

-- P11 zr_bufferTape_reverse
example {α : Type} (w : List α) (q : ℤ) (h1 : -1 ≤ q) (h2 : q ≤ w.length) :
    FinTM.bufferTape w.reverse q = FinTM.bufferTape w (w.length - 1 - q) := by exact?

-- P12 zr_getElem?_mid
example {α : Type} (P Q : List α) (x y : α) (n : ℕ) :
    (P ++ x :: Q)[n]? = if n = P.length then some x else (P ++ y :: Q)[n]? := by exact?

-- P13 zr_map_pair
example {S T : Type} (f : S → T) (a b : S) :

-- P16 zr_bindState_eq
example {n : ℕ} {Γ S S' : Type} {x : List Γ} (e : Cfg n Γ S x) (r : S → Option S')
    (y : Cfg n Γ S' x) (hs : e.state.bind r = y.state) (hi : e.inputPos = y.inputPos)
    (ht : e.workTapes = y.workTapes) (hp : e.workTapePos = y.workTapePos)
    (ho : e.output = y.output) : e.bindState r = y := by exact?

-- P17 zr_round_of_body
example {n : ℕ} {Γ S I : Type} {x : List Γ} {M : MultiTapeTM n Γ S}
    {a entry : S} {E X : Set (Option S)} {P : (Fin n → ℤ) → Prop} (C : Contract M entry X I x)
    (i : I) (c c₁ : Cfg n Γ S x) (hc : c.state = some a) (h1 : M.runFrom c 1 = c₁)
    (hc₁ : c₁.state = some entry) (hp : C.pre i c₁) (haX : some a ∈ X) (hEX : E ⊆ X)
    (hend : (C.post i c₁).state ∈ insert (some a) E) (hP0 : P c.workTapePos)
    (hPC : ∀ p, HeadBox (C.lo i c₁) (C.hi i c₁) p → P p) :
    Round M a E P c (C.post i c₁) (1 + C.time i c₁) := by exact?

-- P18 the iteration inside zr_pinLoop: rounds along any sequence of round starts
example {k : ℕ} {Γ S : Type} {x : List Γ} {M : MultiTapeTM k Γ S} {a : S}
    {E : Set (Option S)} {P : (Fin k → ℤ) → Prop} (cf : ℕ → Cfg k Γ S x) (t : ℕ → ℕ)
    (ha : some a ∉ E) (h0 : P (cf 0).workTapePos) (n : ℕ)
    (hround : ∀ m < n, Round M a E P (cf m) (cf (m + 1)) (t m)) :
    Segment M E P (cf 0) (cf n) (∑ m ∈ Finset.range n, t m) := by exact?

-- P19 zr_fieldOffset_append (second half)
example {Γ : Type} (L R : List (List Γ)) (j : ℕ) (hj : j ≤ R.length) :
    fieldOffset (L ++ R) (L.length + j) = fieldOffset L L.length + fieldOffset R j := by exact?

-- P20 zr_zonedTape_update_same
example {k ℓ : ℕ} (z : Fin k → ZoneContents ℓ) (t : Fin k) (u : ZoneContents ℓ) (q : ℤ)
    (h : (zonedTape (fun _ => u) q).map (zoneSlotOf t) = (zonedTape z q).map (zoneSlotOf t)) :
    zonedTape (Function.update z t u) q = zonedTape z q := by exact?
