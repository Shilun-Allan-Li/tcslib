import TCSlib.Complexity.Diagonalization.NTimeHierarchy

/-! Review probe: the direct `sorryAx` sources reachable from the target's two landed consumers.
The brief claims both are proved modulo exactly `Turing.exists_replayCanonizer`. -/

open Lean Meta Elab Command in
elab "#sorry_sources " id:ident : command => do
  let env ← getEnv
  let root := id.getId
  let mut todo : Array Name := #[root]
  let mut seen : NameSet := {}
  let mut srcs : NameSet := {}
  while !todo.isEmpty do
    let n := todo.back!
    todo := todo.pop
    if seen.contains n then continue
    seen := seen.insert n
    let some ci := env.find? n | continue
    let cs := ci.type.getUsedConstants ++ (ci.value?.map (·.getUsedConstants)).getD #[]
    if cs.contains ``sorryAx then srcs := srcs.insert n
    for c in cs do
      unless seen.contains c do todo := todo.push c
  logInfo m!"{root}: direct sorryAx sources reachable: {srcs.toList}"

#sorry_sources Turing.exists_ndInterpreterCore
#sorry_sources Turing.exists_timed_universal_NDTM
