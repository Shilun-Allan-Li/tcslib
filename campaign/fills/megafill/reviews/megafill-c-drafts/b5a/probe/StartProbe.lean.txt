/-! Review probe: the start of `S`. `S.tm` is a seam composite restarted at its first phase's
initial state; its initial configuration is the first phase's transported start, read only
through published equations (`MultiTapeTM.initCfg_with_q₀`, `MultiTapeTM.runFrom_with_q₀`, the
`Cfg.init` and `Cfg.mapState` fields) and U2-S's capture-initial identity (`ss_captureInitial`,
copied here as the proposed public `embedSilentCfg_initCfg`). Sealed. -/

namespace Turing

section StartSeal

attribute [local irreducible] embedSilentRetTM embedSilentCfg seamCompTM inputRewindTM
  Cfg.mapState MultiTapeTM.runFrom MultiTapeTM.initCfg Cfg.init FinTM.bufferTape
attribute [-simp] MultiTapeTM.initCfg Cfg.init

/-- Capturing a fresh run starts with blank host tapes and zero work heads.

**Proof sketch.** Partition host tapes into selected tapes, the capture tape, and the frame.
The selected tapes inherit the blank initial configuration; the empty captured output makes
its tape blank; the frame is blank by construction. Compare all configuration fields. -/
theorem embedSilentCfg_initCfg {m n : ℕ} {S : Type} (M : MultiTapeTM m Bool S)
    (ι : Fin m ↪ Fin n) (cap : Fin n) (hcap : cap ∉ Set.range ι) (x : List Bool) :
    (embedSilentCfg ι cap (fun _ _ => none) (fun _ => 0) [] []
      (M.initCfg x)).mapState Sum.inl =
      Cfg.init (Sum.inl M.q₀ : S ⊕ Unit) x := by
  classical
  have hfields (j : Fin n) :
      (embedSilentCfg ι cap (fun _ _ => none) (fun _ => 0) [] []
        (M.initCfg x)).workTapes j = (fun _ => none) ∧
      (embedSilentCfg ι cap (fun _ _ => none) (fun _ => 0) [] []
        (M.initCfg x)).workTapePos j = 0 := by
    by_cases hj : j ∈ Set.range ι
    · obtain ⟨i, rfl⟩ := hj
      rw [embedSilentCfg_selected_tape, embedSilentCfg_selected_pos,
        MultiTapeTM.initCfg_eq]
      exact ⟨funext fun z => Cfg.init_workTapes _ _ i z, Cfg.init_workTapePos _ _ i⟩
    · by_cases he : j = cap
      · subst j
        have hh := embedSilentCfg_cap ι cap hcap (fun _ _ => none) (fun _ => 0)
          [] [] (M.initCfg x)
        simpa only [MultiTapeTM.initCfg_eq, Cfg.init_output, List.nil_append,
          FinTM.bufferTape_nil, List.length_nil, Nat.cast_zero] using hh
      · exact embedSilentCfg_unselected ι cap (fun _ _ => none) (fun _ => 0)
          [] [] (M.initCfg x) j hj he
  apply Cfg.ext
  · rw [Cfg.mapState_state, embedSilentCfg_state, MultiTapeTM.initCfg_eq,
      Cfg.init_state, Option.map_some, Cfg.init_state]
  · rw [Cfg.mapState_inputPos, embedSilentCfg_inputPos, MultiTapeTM.initCfg_eq,
      Cfg.init_inputPos, Cfg.init_inputPos]
  · funext j z
    rw [Cfg.mapState_workTapes, Cfg.init_workTapes, (hfields j).1]
  · funext j
    rw [Cfg.mapState_workTapePos, Cfg.init_workTapePos, (hfields j).2]
  · rw [Cfg.mapState_output, embedSilentCfg_output, Cfg.init_output]

/-- Relabelling an initial configuration relabels its state. -/
private theorem init_mapState {k : ℕ} {S T : Type} (f : S → T) (q : S) (x : List Bool) :
    (Cfg.init (k := k) (Symbol := Bool) q x).mapState f = Cfg.init (f q) x := by
  apply Cfg.ext
  · rw [Cfg.mapState_state, Cfg.init_state, Option.map_some, Cfg.init_state]
  · rw [Cfg.mapState_inputPos, Cfg.init_inputPos, Cfg.init_inputPos]
  · funext j z
    rw [Cfg.mapState_workTapes, Cfg.init_workTapes, Cfg.init_workTapes]
  · funext j
    rw [Cfg.mapState_workTapePos, Cfg.init_workTapePos, Cfg.init_workTapePos]
  · rw [Cfg.mapState_output, Cfg.init_output, Cfg.init_output]

example {K : ℕ} (M : FinTM Bool) (ι : Fin M.k ↪ Fin K) (cap : Fin K)
    (hcap : cap ∉ Set.range ι) (z : List Bool) (T : ℕ) :
    let W := seamCompTM (embedSilentRetTM ι cap M.tm) (Sum.inr ()) (inputRewindTM K) .start
    ({ W with q₀ := Sum.inl (Sum.inl M.tm.q₀) } : MultiTapeTM K Bool _).runFrom
        (({ W with q₀ := Sum.inl (Sum.inl M.tm.q₀) } : MultiTapeTM K Bool _).initCfg z) T =
      W.runFrom (((embedSilentCfg ι cap (fun _ _ => none) (fun _ => 0) [] []
        (M.tm.initCfg z)).mapState Sum.inl).mapState Sum.inl) T := by
  intro W
  rw [MultiTapeTM.initCfg_with_q₀, MultiTapeTM.runFrom_with_q₀,
    embedSilentCfg_initCfg M.tm ι cap hcap z, init_mapState]

end StartSeal

end Turing
