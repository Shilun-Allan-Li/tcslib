section U2Route

attribute [local irreducible] StepBlock.forwardLoopCfg rulerHostCfg MultiTapeTM.initCfg Cfg.init
  FinTM.bufferTape StepBlock.summaryCells tableBitsK unaryFin embedEmitCfg Cfg.mapState
attribute [-simp] MultiTapeTM.initCfg Cfg.init

theorem r_encode_init_tapes {k n : ℕ} {Q : Type} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (x : List Bool) (j : Fin (k + 2)) :
    (StepBlock.forwardLoopCfg entry M (M.initCfg x)).workTapes j =
      if j.val = k then FinTM.bufferTape (tableBitsK M.tr)
      else if j.val = k + 1 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
      else fun _ => none := by
  funext p
  rw [StepBlock.forwardLoopCfg_workTapes]
  refine Fin.addCases (fun i => ?_) (fun j' => ?_) j
  · rw [Fin.addCases_left, MultiTapeTM.initCfg_eq, Cfg.init_workTapes]
    have h1 : (Fin.castAdd 2 i).val ≠ k := by simp only [Fin.coe_castAdd]; omega
    have h2 : (Fin.castAdd 2 i).val ≠ k + 1 := by simp only [Fin.coe_castAdd]; omega
    rw [if_neg h1, if_neg h2]
  · rw [Fin.addCases_right, MultiTapeTM.initCfg_eq, Cfg.init_state, Option.map_some,
      Option.getD_some]
    by_cases h : j' = 0
    · subst h
      have h1 : (Fin.natAdd k (0 : Fin 2)).val = k := by simp only [Fin.coe_natAdd]; omega
      rw [if_pos rfl, if_pos h1]
    · have hj : j' = 1 := by omega
      subst hj
      have h1 : (Fin.natAdd k (1 : Fin 2)).val ≠ k := by simp only [Fin.coe_natAdd]; omega
      have h2 : (Fin.natAdd k (1 : Fin 2)).val = k + 1 := by simp only [Fin.coe_natAdd]; omega
      rw [if_neg (by decide), if_neg h1, if_pos h2, StepBlock.summaryCells_eq]
      by_cases hp : p = 0
      · rw [if_pos hp, hp]
        exact (FinTM.bufferTape_of_neg _ (by omega)).symm
      · rw [if_neg hp]

theorem r_encode_init_fields {k n : ℕ} {Q : Type} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (x : List Bool) :
    (StepBlock.forwardLoopCfg entry M (M.initCfg x)).state = some entry ∧
      (StepBlock.forwardLoopCfg entry M (M.initCfg x)).inputPos.val = 1 ∧
      (∀ j, (StepBlock.forwardLoopCfg entry M (M.initCfg x)).workTapePos j = 0) ∧
      (StepBlock.forwardLoopCfg entry M (M.initCfg x)).output = [] := by
  refine ⟨?_, ?_, fun j => ?_, ?_⟩
  · rw [StepBlock.forwardLoopCfg_state, MultiTapeTM.initCfg_eq, Cfg.init_state, Option.map_some]
  · rw [StepBlock.forwardLoopCfg_inputPos, MultiTapeTM.initCfg_eq, Cfg.init_inputPos]; rfl
  · rw [StepBlock.forwardLoopCfg_workTapePos]
    refine Fin.addCases (fun i => ?_) (fun j' => ?_) j
    · rw [Fin.addCases_left, MultiTapeTM.initCfg_eq, Cfg.init_workTapePos]
    · rw [Fin.addCases_right]
  · rw [StepBlock.forwardLoopCfg_output, MultiTapeTM.initCfg_eq, Cfg.init_output]

theorem r_encode_halt {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x) :
    ((StepBlock.forwardLoopCfg entry M c).state = none ↔ c.state = none) ∧
      (StepBlock.forwardLoopCfg entry M c).output = c.output := by
  rw [StepBlock.forwardLoopCfg_state, StepBlock.forwardLoopCfg_output, Option.map_eq_none_iff]
  exact ⟨Iff.rfl, rfl⟩

theorem r_ruler_fields {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (q : Option Q) (c : Cfg m Bool S y) (rt : ℤ → Option Bool)
    (rp : ℤ) :
    (rulerHostCfg ℓ hz q c rt rp).inputPos.val = ℓ + c.inputPos.val ∧
      (rulerHostCfg ℓ hz q c rt rp).workTapes (Fin.last m) = rt ∧
      (rulerHostCfg ℓ hz q c rt rp).workTapePos (Fin.last m) = rp ∧
      (∀ i : Fin m, (rulerHostCfg ℓ hz q c rt rp).workTapes i.castSucc = c.workTapes i) ∧
      (rulerHostCfg ℓ hz q c rt rp).output = c.output := by
  refine ⟨by rw [rulerHostCfg_inputPos], ?_, ?_, fun i => ?_, rulerHostCfg_output _ _ _ _ _ _⟩
  · funext p; rw [rulerHostCfg_workTapes, Fin.lastCases_last]
  · rw [rulerHostCfg_workTapePos, Fin.lastCases_last]
  · funext p; rw [rulerHostCfg_workTapes, Fin.lastCases_castSucc]

/-! ### Beyond the draft: the full seam configurations of both phases (sealed)

Phase 1: the startup's endpoint, relabelled to H2-F's entry, is the R1 embedding of the encoding
of the initial configuration into blank frames. -/

theorem r_seam_phase1 {k K n : ℕ} {P Q : Type} (entry : Q) (exit : P)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (z : List Bool)
    (ι : Fin (k + 2) ↪ Fin (K + (k + 2))) (hι : ∀ j, (ι j).val = j.val)
    (D : Cfg (K + (k + 2)) Bool P z)
    (hD : D = ⟨some exit, 1,
      fun j => if j.val = k then FinTM.bufferTape (tableBitsK M.tr)
        else if j.val = k + 1 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
        else fun _ => none,
      fun _ => 0, []⟩) :
    embedEmitCfg ι (fun _ _ => none) (fun _ => 0) []
        (StepBlock.forwardLoopCfg entry M (M.initCfg z)) = D.mapState fun _ => entry := by
  obtain ⟨hs, hin, hh, ho⟩ := r_encode_init_fields entry M z
  subst hD
  apply embedEmitCfg_eq
  · rw [hs, Cfg.mapState_state]; rfl
  · rw [Cfg.mapState_inputPos]; exact Fin.ext hin
  · rw [ho, Cfg.mapState_output, List.nil_append]
  · intro i
    refine ⟨?_, ?_⟩
    · funext p
      rw [r_encode_init_tapes, Cfg.mapState_workTapes]
      show _ = (if (ι i).val = k then (_ : ℤ → Option Bool) else _) p
      rw [hι i]
    · rw [hh i, Cfg.mapState_workTapePos]
  · intro j hj
    have hge : k + 2 ≤ j.val := by
      by_contra hlt
      exact hj ⟨⟨j.val, by omega⟩, Fin.ext (by rw [hι])⟩
    refine ⟨?_, ?_⟩
    · funext p
      rw [Cfg.mapState_workTapes]
      show _ = (if j.val = k then (_ : ℤ → Option Bool) else _) p
      rw [if_neg (by omega), if_neg (by omega)]
    · rw [Cfg.mapState_workTapePos]

/-! Phase 2: after the startup, the header ruler and its rewind, the configuration relabelled to
the host's entry `inj entry false` is the R1 embedding of the ruler host's presentation of the
encoding of the initial configuration on the suffix `x`. -/

theorem r_seam_phase2 {k N n : ℕ} {P Q H : Type} (entry : Q) (inj : Q → Bool → H)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (x z : List Bool) (ℓ : ℕ)
    (hz : z.length = ℓ + x.length)
    (σ : Fin ((k + 2) + 1) ↪ Fin N) (hσ : ∀ j : Fin (k + 2), (σ j.castSucc).val = j.val)
    (r : Fin N) (hr : σ (Fin.last (k + 2)) = r) (hrk : k + 2 ≤ r.val)
    (D : Cfg N Bool P z) (e : P) (hst : D.state = some e) (hin : D.inputPos.val = ℓ + 1)
    (htapes : D.workTapes = fun j => if j.val = k then FinTM.bufferTape (tableBitsK M.tr)
        else if j.val = k + 1 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
        else if j = r then FinTM.bufferTape (List.replicate ℓ true)
        else fun _ => none)
    (hheads : D.workTapePos = fun _ => 0) (hout : D.output = []) :
    embedEmitCfg σ (fun _ _ => none) (fun _ => 0) []
        (rulerHostCfg ℓ hz
          ((StepBlock.forwardLoopCfg entry M (M.initCfg x)).state.map fun q => inj q false)
          (StepBlock.forwardLoopCfg entry M (M.initCfg x))
          (FinTM.bufferTape (List.replicate ℓ true)) 0) =
      D.mapState fun _ => inj entry false := by
  obtain ⟨hs, hin', hh, ho⟩ := r_encode_init_fields entry M x
  apply embedEmitCfg_eq
  · rw [rulerHostCfg_state, hs, Option.map_some, Cfg.mapState_state, hst, Option.map_some]
  · rw [Cfg.mapState_inputPos]
    apply Fin.ext
    rw [hin, rulerHostCfg_inputPos]
    show ℓ + _ = ℓ + 1
    rw [hin']
  · rw [rulerHostCfg_output, ho, Cfg.mapState_output, hout, List.nil_append]
  · intro i
    refine Fin.lastCases ?_ (fun j => ?_) i
    · refine ⟨?_, ?_⟩
      · funext p
        rw [rulerHostCfg_workTapes, Fin.lastCases_last, Cfg.mapState_workTapes, htapes, hr]
        show _ = (if r.val = k then (_ : ℤ → Option Bool) else _) p
        rw [if_neg (by omega), if_neg (by omega), if_pos rfl]
      · rw [rulerHostCfg_workTapePos, Fin.lastCases_last, Cfg.mapState_workTapePos, hheads]
    · refine ⟨?_, ?_⟩
      · funext p
        rw [rulerHostCfg_workTapes, Fin.lastCases_castSucc, Cfg.mapState_workTapes, htapes,
          r_encode_init_tapes]
        have hne : σ j.castSucc ≠ r := by
          rw [← hr]; exact fun h => (Fin.castSucc_ne_last j) (σ.injective h)
        show (if j.val = k then (_ : ℤ → Option Bool) else _) p =
          (if (σ j.castSucc).val = k then (_ : ℤ → Option Bool) else _) p
        rw [hσ j]
        by_cases h1 : j.val = k
        · rw [if_pos h1, if_pos h1]
        · rw [if_neg h1, if_neg h1]
          by_cases h2 : j.val = k + 1
          · rw [if_pos h2, if_pos h2]
          · rw [if_neg h2, if_neg h2, if_neg hne]
      · rw [rulerHostCfg_workTapePos, Fin.lastCases_castSucc, hh, Cfg.mapState_workTapePos,
          hheads]
  · intro j hj
    have hge : k + 2 ≤ j.val := by
      by_contra hlt
      exact hj ⟨(⟨j.val, by omega⟩ : Fin (k + 2)).castSucc, Fin.ext (by rw [hσ])⟩
    have hjr : j ≠ r := fun h => hj ⟨Fin.last (k + 2), hr.trans h.symm⟩
    refine ⟨?_, ?_⟩
    · funext p
      rw [Cfg.mapState_workTapes, htapes]
      show _ = (if j.val = k then (_ : ℤ → Option Bool) else _) p
      rw [if_neg (by omega), if_neg (by omega), if_neg hjr]
    · rw [Cfg.mapState_workTapePos, hheads]

end U2Route
