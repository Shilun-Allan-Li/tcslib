Megafill D drafting, 2026-10-25: the route probe for the fill E-Q,
`Turing.NTimeDiag.exists_queryTM` (`Diagonalization/NTimeDiagonalizer.lean`). A probe record,
not library code.

It checks the mathematical steps of the capped locator (the P3.3 gate's round 2, §2) that a
machine-level route rests on, from published declarations only, under a seal of every other-file
TCSlib definition they touch:
* the witness elimination through the proposed API lemma `timeConstructible_iff` (a local
  stand-in here, proved by `Iff.rfl` before the seal; it is not published at the drafting base);
* the timeout deduction: a capped run that has not halted by `c (n + 1)` certifies `n < f a`;
* the stop rule: the locator's decision at a rung `ℓᵢ < n` (`n < f a`, then
  `(f a + a + 1)⁴` against `⌊log₂ n⌋` with the power-of-two test) is exactly the comparison of `n`
  with the next rung, so the stage it stops at is `stage f n`, and it knows whether `n` is the top;
* the ledger: a passed rung's work is at most the next rung, and the passed rungs sum to at most
  twice the last one, so the completed attempts cost `O(n)`.
The machine assembly itself (rounds as contracts, the clock, the calls) is not probed.

`ladder`, `budget`, `stage`, `stageCode` and `query` are the owned file's own definitions; the
probe reads them through their published laws anyway.

/-! ### Stand-in for the proposed API lemma (proved before the seal) -/

theorem eq_timeConstructible_iff (T : ℕ → ℕ) :
    Complexity.TimeConstructible T ↔
      (∀ n, n ≤ T n) ∧ ∃ c : ℕ, 0 < c ∧ ∃ M : FinTM Bool, ∀ x : List Bool,
        M.ComputesInTime x (T x.length).bits (c * (T x.length + 1)) :=
  Iff.rfl

section Sealed

attribute [local irreducible] Complexity.TimeConstructible Turing.FinTM.ComputesInTime
  Turing.FinTM.ComputesFunInTime Turing.MultiTapeTM.runFrom Turing.MultiTapeTM.initCfg

/-- The witness, through the proposed API lemma. -/
theorem eq_witness {f : ℕ → ℕ} (hf : Complexity.TimeConstructible f) :
    ∃ c : ℕ, 0 < c ∧ ∃ M : FinTM Bool, ∀ x : List Bool,
      M.ComputesInTime x (f x.length).bits (c * (f x.length + 1)) := by
  obtain ⟨-, c, hc, M, hM⟩ := (eq_timeConstructible_iff f).mp hf
  exact ⟨c, hc, M, hM⟩

/-- **The timeout deduction** (round 2, §2): if the witness has not halted on `x` by time
`c (n + 1)`, then `n < f |x|`. -/
theorem eq_timeout {f : ℕ → ℕ} {c n : ℕ} {M : FinTM Bool} {x : List Bool}
    (hM : M.ComputesInTime x (f x.length).bits (c * (f x.length + 1)))
    (hrun : ∀ u ≤ c * (n + 1), (M.tm.runFrom (M.tm.initCfg x) u).state ≠ none) :
    n < f x.length := by
  by_contra hle
  push_neg at hle
  have hstop := ((FinTM.computesInTime_iff _ _ _ _).mp hM).1
  exact hrun _ (Nat.mul_le_mul_left c (Nat.succ_le_succ hle)) hstop

/-- The locator's three outcomes at a rung. -/
inductive EqStep
  | interior
  | top
  | next
  deriving DecidableEq

/-- **The locator's decision at the rung `ℓ`** on a length `n`, as the machine takes it: an output
`f a` above `n` (or a timeout, which certifies it) stops inside the stage; otherwise
`(f a + a + 1)⁴` is compared with `⌊log₂ n⌋`: above it the stage is found, below it the locator
advances, and at equality it stops at the top exactly when `n` is a power of two (otherwise the next
rung `2 ^ ⌊log₂ n⌋` is below `n`, and it advances). -/
def eqDecide (f : ℕ → ℕ) (n ℓ : ℕ) : EqStep :=
  if n < f (ℓ + 1) then .interior
  else if Nat.log 2 n < (f (ℓ + 1) + (ℓ + 1) + 1) ^ 4 then .interior
  else if (f (ℓ + 1) + (ℓ + 1) + 1) ^ 4 = Nat.log 2 n then
    (if n = 2 ^ Nat.log 2 n then .top else .next)
  else .next

/-- The next rung is `2 ^ X⁴` with `X = f a + a + 1`. -/
theorem eq_ladder_succ_eq (f : ℕ → ℕ) (i : ℕ) :
    ladder f (i + 1) = 2 ^ (f (ladder f i + 1) + (ladder f i + 1) + 1) ^ 4 := by
  rw [ladder_succ, budget_eq, ← pow_mul]

/-- **The stop rule** (round 2, §2): at a rung below `n`, the decision is exactly the comparison
of `n` with the next rung. -/
theorem eq_decide_spec (f : ℕ → ℕ) {n i : ℕ} (hn : 1 ≤ n) :
    (eqDecide f n (ladder f i) = .interior ↔ n < ladder f (i + 1)) ∧
    (eqDecide f n (ladder f i) = .top ↔ n = ladder f (i + 1)) ∧
    (eqDecide f n (ladder f i) = .next ↔ ladder f (i + 1) < n) := by
  rw [eq_ladder_succ_eq]
  set X := f (ladder f i + 1) + (ladder f i + 1) + 1 with hX
  set L := Nat.log 2 n with hL
  have hlo : 2 ^ L ≤ n := Nat.pow_log_le_self 2 (by omega)
  have hhi : n < 2 ^ (L + 1) := Nat.lt_pow_succ_log_self (by norm_num) n
  have hX1 : 1 ≤ X := by omega
  have hX4 : X ≤ X ^ 4 := Nat.le_self_pow (by norm_num) X
  by_cases h1 : n < f (ladder f i + 1)
  · -- the output is above `n`: then `2 ^ X⁴ > X⁴ ≥ X > n`
    have hbig : n < 2 ^ X ^ 4 := by
      have := Nat.lt_two_pow_self (n := X ^ 4)
      omega
    simp only [eqDecide, h1, if_true]
    refine ⟨by simp; omega, by simp; omega, by simp; omega⟩
  · by_cases h2 : L < X ^ 4
    · have hbig : n < 2 ^ X ^ 4 :=
        lt_of_lt_of_le hhi (Nat.pow_le_pow_right (by norm_num) (by omega))
      simp only [eqDecide, h1, if_false, ← hL, ← hX, h2, if_true]
      refine ⟨by simp; omega, by simp; omega, by simp; omega⟩
    · by_cases h3 : X ^ 4 = L
      · simp only [eqDecide, h1, if_false, ← hL, ← hX, h3, if_true]
        by_cases h4 : n = 2 ^ L
        · simp only [h4, if_true]
          refine ⟨by simp, by simp, by simp⟩
        · simp only [h4, if_false]
          have : 2 ^ L < n := lt_of_le_of_ne hlo (Ne.symm h4)
          refine ⟨by simp; omega, by simp, by simp; omega⟩
      · have hlt : X ^ 4 < L := by omega
        have hsmall : 2 ^ X ^ 4 < n :=
          lt_of_lt_of_le (Nat.pow_lt_pow_right (by norm_num) hlt) hlo
        simp only [eqDecide, h1, if_false, ← hL, ← hX, h2, h3]
        refine ⟨by simp; omega, by simp; omega, by simp; omega⟩

/-- **The stage the locator stops at**: at a rung `ℓᵢ < n`, a decision `interior` or `top` gives
`stage f n = i`; `top` says `n` is the top. -/
theorem eq_stage_of_stop (f : ℕ → ℕ) {n i : ℕ} (hlo : ladder f i < n)
    (hstop : eqDecide f n (ladder f i) ≠ .next) : stage f n = i := by
  have hn : 1 ≤ n := by have := le_ladder f i; omega
  obtain ⟨-, -, hnext⟩ := eq_decide_spec f (i := i) hn
  have hle : n ≤ ladder f (i + 1) := by
    by_contra h
    exact hstop (hnext.mpr (by omega))
  exact stage_eq f hlo hle

/-- **A passed rung's work** (round 2, §2): the witness input `a = ℓ + 1`, its output `f a` and
the arithmetic operand `f a + a + 1` are all at most the next rung. -/
theorem eq_round_le (f : ℕ → ℕ) (i : ℕ) :
    f (ladder f i + 1) + (ladder f i + 1) + 1 ≤ ladder f (i + 1) := by
  rw [eq_ladder_succ_eq]
  set X := f (ladder f i + 1) + (ladder f i + 1) + 1
  have h4 : X ≤ X ^ 4 := Nat.le_self_pow (by norm_num) X
  have := Nat.lt_two_pow_self (n := X ^ 4)
  omega

/-- **The ledger** (round 2, §2): the rungs passed before stage `s` sum to at most twice its
bottom, `∑_{h < s} ℓ_{h+1} ≤ 2 ℓ_s`. -/
theorem eq_ledger (f : ℕ → ℕ) (s : ℕ) :
    ∑ h ∈ Finset.range s, ladder f (h + 1) ≤ 2 * ladder f s := by
  induction s with
  | zero => simp
  | succ s ih =>
    rw [Finset.sum_range_succ]
    have := two_mul_lt_ladder_succ f s
    omega
