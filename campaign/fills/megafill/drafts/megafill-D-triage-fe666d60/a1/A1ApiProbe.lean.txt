open Turing

/-- `T` is time-constructible exactly when `n ≤ T n` for every `n` and some machine writes
`bits (T |x|)` within `c · (T |x| + 1)` steps on every `x`, for some `c > 0`. -/
theorem Complexity.a1_timeConstructible_iff (T : ℕ → ℕ) :
    Complexity.TimeConstructible T ↔
      (∀ n, n ≤ T n) ∧ ∃ c : ℕ, 0 < c ∧ ∃ M : FinTM Bool, ∀ x : List Bool,
        M.ComputesInTime x (T x.length).bits (c * (T x.length + 1)) :=
  Iff.rfl

/-- `S` is space-constructible exactly when `logSpace n ≤ S n` for every `n` and some machine
computes `x ↦ bits (S |x|)` in space `c · S`, for some `c > 0`. -/
theorem Complexity.a1_spaceConstructible_iff (S : ℕ → ℕ) :
    Complexity.SpaceConstructible S ↔
      (∀ n, Complexity.logSpace n ≤ S n) ∧ ∃ c : ℕ, 0 < c ∧ ∃ M : FinTM Bool,
        M.ComputesInSpace (fun x => (S x.length).bits) fun n => c * S n :=
  Iff.rfl

/-- `M` computes `f` in space `s` exactly when on every `x` there is a time `t` at which it has
halted with output `f x`, having visited at most `s |x|` work cells by then. -/
theorem Turing.FinTM.a1_computesInSpace_iff (M : FinTM Bool) (f : List Bool → List Bool)
    (s : ℕ → ℕ) :
    M.ComputesInSpace f s ↔
      ∀ x : List Bool, ∃ t, M.ComputesInTime x (f x) t ∧
        M.tm.spaceUsed (M.tm.initCfg x) t ≤ s x.length :=
  Iff.rfl

/-- The normalized logarithm: `logSpace n = Nat.log 2 n + 1`. -/
theorem Complexity.a1_logSpace_eq (n : ℕ) : Complexity.logSpace n = Nat.log 2 n + 1 := rfl

/-- A transducer emits nothing on the empty word. -/
theorem Complexity.a1_transduce_nil {σ : Type} (δ : σ → Bool → σ) (o : σ → Bool → Option Bool)
    (q : σ) : Complexity.transduce δ o q [] = [] := rfl

/-- On `b :: l` a transducer emits `o q b`, then runs on `l` from `δ q b`. -/
theorem Complexity.a1_transduce_cons {σ : Type} (δ : σ → Bool → σ)
    (o : σ → Bool → Option Bool) (q : σ) (b : Bool) (l : List Bool) :
    Complexity.transduce δ o q (b :: l) =
      (o q b).toList ++ Complexity.transduce δ o (δ q b) l := rfl

/-- **A copying state is the identity**: if from `q` every bit is emitted unchanged and `q` is
kept, the transducer copies its input. (It generalizes the two published instances
`Complexity.KarpLipton.transduce_drop_true` and `transduce_fT_copy`, `MeyerVerifierGlue`.) -/
theorem Complexity.a1_transduce_of_copy {σ : Type} (δ : σ → Bool → σ)
    (o : σ → Bool → Option Bool) (q : σ) (hδ : ∀ b, δ q b = q) (ho : ∀ b, o q b = some b)
    (l : List Bool) : Complexity.transduce δ o q l = l := by
  induction l with
  | nil => exact Complexity.a1_transduce_nil δ o q
  | cons b l ih =>
    rw [Complexity.a1_transduce_cons, hδ, ho, ih]
    rfl
