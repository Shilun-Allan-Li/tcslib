/-!
# The step block's forwarding round loop (H2-F)

**Status: proved** (fill H2-F, 2026-10-16; the fill gate, with U2-S and U2-U, is pending). The
statement gate passed on 2026-10-15 (`audits/u2-statements-findings.md`: PASS, 0 blockers,
0 majors, 2 minors; resolutions `audits/u2-statements-resolutions.md`). The contract
`Turing.StepBlock.exists_forwardLoopTM` is proved: the step block, released by
`Turing.seamReleaseTM`, and a forwarding leaf, as one machine on the public phase host. The
encoding is defined.

**H2's unclocked loop form** (host H2, `machine-library-design.md` §14): the step block
`Turing.StepBlock.roundTM` followed by a forwarding leaf, run until the simulated machine halts.
One finite machine per work-tape count `k` simulates any `k`-work-tape binary transition table,
held as data in the `Turing.tableBitsK` layout, in macro-step lockstep (`Turing.Lockstep`) through
the encoding `Turing.StepBlock.forwardLoopCfg`, forwarding each simulated emission to its own
output, with a per-step cost bounded by the table's length and a per-head space ledger. It is the
loop of the `k`-generic universal machine U2 (`TCSlib.Complexity.TuringMachine.UniversalK`;
design note `audits/evidence/u2/U2-design.md` §2.2), and of B6's `U₀` at `k = 1`.

H2's clocked loop form, the counter loop over the step block, is
`TCSlib.Complexity.TuringMachine.Build.StepBlockLoop`. The forwarding form has its own module
(user, 2026-10-15) so that the clocked form's consumers, `CodeSimulator` among them, do not depend
on it.

## Main definitions

* `Turing.StepBlock.forwardLoopCfg` — H2-F's encoding of a source configuration, with its field
  equations (`Turing.StepBlock.forwardLoopCfg_state` and the like).

## Main results

* `Turing.StepBlock.exists_forwardLoopTM` — H2-F, the forwarding round loop: a macro-step
  lockstep of any `k`-tape table with its emissions forwarded, per-step cost
  `≤ 50 · (|table| + 1) + C` (with `C = 3`), and the space ledger.

## References

* [AB09] S. Arora, B. Barak, *Computational Complexity: A Modern Approach*,
  Cambridge University Press, 2009. (§1.4.1, proof of Theorem 1.9.)
-/

/-! ## The forwarding round loop (H2-F) -/

/-- **The forwarding round loop's configuration for a source configuration** (H2-F's encoding):
for a `k`-work-tape binary machine `M` with `n + 1` states and a configuration `c` of `M` on the
input `x`, the `(k + 2)`-tape configuration whose input head and mirrored tapes `0, …, k - 1`
(contents and heads) are `c`'s, whose tape `k` holds the table `tableBitsK M.tr` and tape
`k + 1` the blank summary cell followed by the unary current state (empty when `c` is halted),
both auxiliary heads at `0`, whose output is `c`'s output, and whose control is `entry` while
`c` is live and halted once `c` is. It is the step block's seam relation `Turing.StepBlock.Rep`
for `c` with its output erased (so the summary cell is blank), with the physical output
restored: the reset invariant of `Turing.exists_selfUniversal`'s sketch (the H2-S gate's
finding 4). [AB09, §1.4.1, proof of Theorem 1.9] -/
def forwardLoopCfg {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x) :
    Cfg (k + 2) Bool Q x where
  state := c.state.map fun _ => entry
  inputPos := c.inputPos
  workTapes := Fin.addCases (motive := fun _ => ℤ → Option Bool) c.workTapes fun j =>
    if j = 0 then FinTM.bufferTape (tableBitsK M.tr)
    else summaryCells none ((c.state.map unaryFin).getD [])
  workTapePos := Fin.addCases (motive := fun _ => ℤ) c.workTapePos fun _ => 0
  output := c.output

/-! The forwarding configuration's fields as equations, generated by `simps` without the `simp`
attribute: `Turing.StepBlock.forwardLoopCfg_state`, `_inputPos`, `_workTapes` (pointwise in the
tape and the cell), `_workTapePos` and `_output`. -/
attribute [simps (config := { isSimp := false }) state inputPos workTapes workTapePos output]
  forwardLoopCfg

/-- Erasing only the simulated output gives the block's reset invariant; the physical
output prefix of the forwarding configuration is retained. -/
private lemma fl_reset {k n : ℕ} {x : List Bool}
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x) :
    Rep M.tr {c with output := []} (forwardLoopCfg (exit k true) M c) := by
  simp [Rep, forwardLoopCfg, tableTape, summaryTape, outputSummary]

/-- The data change of a forwarding leaf: clear the scanned summary cell and append its
optional bit to the physical output, with the supplied successor control. -/
private def fl_flush {k : ℕ} {S Q : Type} {x : List Bool} (next : Option Q)
    (d : Cfg (k + 2) Bool S x) : Cfg (k + 2) Bool Q x where
  state := next
  inputPos := d.inputPos
  workTapes := Function.update d.workTapes (summaryTape k)
    (Function.update (d.workTapes (summaryTape k)) (d.workTapePos (summaryTape k)) none)
  workTapePos := d.workTapePos
  output := d.output ++ (d.workTapeSymbols (summaryTape k)).toList

/-- The forwarding leaf. `some live` reads, emits and clears; `none` is the live return
anchor. There is no public row for this atomic emission and clearing: the step block and
its clocked loop are silent, and `tapeAction` itself emits nothing. The leaf changes only
that action's emission and successor fields and moves no head. -/
private def fl_leafTM (k : ℕ) : MultiTapeTM (k + 2) Bool (Option Bool) where
  q₀ := some true
  tr q _ w := match q with
    | some live => { tapeAction (summaryTape k) 0 (some none) 0 (none : Option Bool) with
        output := w (summaryTape k)
        state := if live then some none else none }
    | none => inputMoveAction 0 none

/-- The leaf's single step has exactly the clearing and emission effect, including when
it halts. All input and work heads keep their positions.

**Proof sketch.** Apply the one-tape action with its emission and successor fields replaced.
Only the selected work tape is written; its head displacement and every other displacement
are zero. -/
private lemma fl_leaf_step {k : ℕ} {x : List Bool} (d : Cfg (k + 2) Bool (Option Bool) x)
    (live : Bool) (hd : d.state = some (some live)) :
    (fl_leafTM k).step d = fl_flush (if live then some none else none) d := by
  simp only [MultiTapeTM.step, hd, fl_leafTM]
  apply Cfg.ext
  · rfl
  · simp [Action.apply, tapeAction, fl_flush]
  · funext j
    by_cases hj : j = summaryTape k <;> simp [Action.apply, tapeAction, fl_flush, hj]
  · funext j
    by_cases hj : j = summaryTape k <;> simp [Action.apply, tapeAction, fl_flush, hj]
  · rfl

/-- A full leaf run takes one step. -/
private lemma fl_leaf_run {k : ℕ} {x : List Bool} (d : Cfg (k + 2) Bool (Option Bool) x)
    (live : Bool) (hd : d.state = some (some live)) :
    (fl_leafTM k).runFrom d 1 = fl_flush (if live then some none else none) d :=
  fl_leaf_step d live hd

/-- Relabelling the forwarding configuration changes only its designated live control. -/
private lemma fl_encode_map {k n : ℕ} {Q R : Type} {x : List Bool} (entry : Q)
    (f : Q → R) (M : MultiTapeTM k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) :
    (forwardLoopCfg entry M c).mapState f = forwardLoopCfg (f entry) M c := by
  apply Cfg.ext <;> simp [forwardLoopCfg, Cfg.mapState, Option.map_map, Function.comp_def]

/-- Releasing the block and forwarding its summary are the two phases. -/
private def fl_Ph (k : ℕ) : Bool → Type
  | false => Unit ⊕ (Σ key, Ph k key)
  | true => Option Bool

private instance fl_fintype (k : ℕ) (b : Bool) : Fintype (fl_Ph k b) := by
  cases b <;> dsimp [fl_Ph] <;> infer_instance

private instance fl_decidableEq (k : ℕ) (b : Bool) : DecidableEq (fl_Ph k b) := by
  cases b <;> dsimp [fl_Ph] <;> infer_instance

/-- Each invocation starts at the release adapter's fresh control. -/
private def fl_entry (k : ℕ) : Σ b, fl_Ph k b := ⟨false, Sum.inl ()⟩

/-- The two machines supplied to the existing phase host. -/
private def fl_part (k : ℕ) : (b : Bool) → MultiTapeTM (k + 2) Bool (fl_Ph k b)
  | false => seamReleaseTM (roundTM k) (exit k true)
  | true => fl_leafTM k

/-- Only returned block exits and the leaf's live return anchor dispatch. In particular,
the release adapter's fresh state is never redirected. -/
private def fl_red (k : ℕ) : (b : Bool) → fl_Ph k b → Option (Σ b, fl_Ph k b)
  | false, Sum.inl _ => none
  | false, Sum.inr s =>
    if s = exit k true then some ⟨true, some true⟩
    else if s = exit k false then some ⟨true, some false⟩ else none
  | true, some _ => none
  | true, none => some (fl_entry k)

/-- The forwarding machine uses exactly one public phase host over the released block and
its forwarding leaf. Its finite control depends only on the tape count. -/
private def fl_TM (k : ℕ) : MultiTapeTM (k + 2) Bool (Σ b, fl_Ph k b) :=
  phaseHostTM (fl_part k) (fl_entry k) (fl_red k)

/-- The released block runs unchanged until its positive exit and then dispatches to the
appropriate forwarding leaf.

**Proof sketch.** The release adapter agrees with the block at positive times. At time zero
its fresh control cannot dispatch; at interior positive times the block's strict cut excludes
both redirect states. Apply the public phase lemma to this guard. -/
private lemma fl_block_phase {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x) (T : ℕ) (live : Bool)
    (hd : d.state = some (exit k true)) (hT : 0 < T)
    (he : ((roundTM k).runFrom d T).state = some (exit k live))
    (hc : ∀ u, 0 < u → u < T → ∃ s, ((roundTM k).runFrom d u).state = some s ∧
      s ≠ exit k true ∧ s ≠ exit k false) :
    (∀ u ≤ T, (fl_TM k).runFrom (d.mapState fun _ => fl_entry k) u =
      ((fl_part k false).runFrom (d.mapState fun _ => Sum.inl ()) u).mapState
        (Sigma.mk false)) ∧
    (fl_TM k).runFrom (d.mapState fun _ => fl_entry k) (T + 1) =
      ((roundTM k).runFrom d T).mapState (fun _ => ⟨true, some live⟩) := by
  have released := seamReleaseTM_runFrom_pos (roundTM k) (exit k true) d hd T hT
  have dispatched := phaseHostTM_run (fl_part k) (fl_entry k) (fl_red k) false
    (d.mapState fun _ => Sum.inl ()) (((roundTM k).runFrom d T).mapState Sum.inr)
    T released (Sum.inr (exit k live))
    (by simp [Cfg.mapState, he]) ⟨true, some live⟩
    (by cases live <;> simp [fl_red, exit]) (by
      intro u hu s hs
      by_cases hz : u = 0
      · subst u
        simp only [MultiTapeTM.runFrom_zero, Cfg.mapState, hd, Option.map_some,
          Option.some.injEq] at hs
        subst s
        rfl
      · have hp : 0 < u := Nat.pos_of_ne_zero hz
        change ((seamReleaseTM (roundTM k) (exit k true)).runFrom _ u).state = _ at hs
        rw [seamReleaseTM_runFrom_pos (roundTM k) (exit k true) d hd u hp] at hs
        obtain ⟨r, hr, hn, hh⟩ := hc u hp hu
        simp only [Cfg.mapState, hr, Option.map_some, Option.some.injEq] at hs
        subst s
        simp [fl_red, hn, hh])
  simpa only [fl_TM, Cfg.mapState, Option.map_map, Function.comp_def] using dispatched

/-- The leaf keeps its one-step effect inside the public phase host. -/
private lemma fl_hosted_leaf {k : ℕ} {S : Type} {x : List Bool}
    (d : Cfg (k + 2) Bool S x) (live : Bool) (hd : d.state ≠ none) :
    (fl_TM k).step (d.mapState fun _ => ⟨true, some live⟩) =
      fl_flush (if live then some ⟨true, none⟩ else none) d := by
  obtain ⟨q, hq⟩ := Option.ne_none_iff_exists'.mp hd
  have lifted : (fl_TM k).step (d.mapState fun _ => ⟨true, some live⟩) =
      ((fl_leafTM k).step (d.mapState fun _ => some live)).mapState (Sigma.mk true) := by
    simp only [MultiTapeTM.step, Cfg.mapState, hq, Option.map_some]
    rfl
  rw [lifted, fl_leaf_step _ live (by simp [Cfg.mapState, hq])]
  cases live <;> rfl

/-- The live leaf's return anchor dispatches without moving a head or changing data. -/
private lemma fl_return {k : ℕ} {S : Type} {x : List Bool} (d : Cfg (k + 2) Bool S x) :
    (fl_TM k).step (fl_flush (some ⟨true, none⟩) d) =
      fl_flush (some (fl_entry k)) d := by
  rw [step_of_inputMoveAction (fl_TM k) _ 0 rfl (by rfl)]
  simp only [fl_flush, moveInputPos_zero]

/-- Clearing a represented summary recovers the forwarding encoding, with exactly the
physical prefix and the one summary emission in its output.

**Proof sketch.** The mirrored bank and table are unchanged. At the summary head, which is
home, `summaryCells_update` removes only cell zero. The state word is already the word of
the represented source, even at a halt. -/
private lemma fl_restore {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x) (hr : Rep M.tr c d) :
    fl_flush (c.state.map fun _ => entry) d =
      forwardLoopCfg entry M {c with output := d.output ++ (outputSummary c.output).toList} := by
  obtain ⟨hin, hbank, htable, hsummary, ht, hs⟩ := hr
  have read : d.workTapeSymbols (summaryTape k) = outputSummary c.output := by
    simp only [Cfg.workTapeSymbols, hs, hsummary, summaryCells, ↓reduceIte]
  apply Cfg.ext
  · rfl
  · exact hin
  · funext j
    induction j using Fin.addCases with
    | left i =>
      have hne : Fin.castAdd 2 i ≠ summaryTape k := by
        intro he
        have hv := congrArg Fin.val he
        change i.val = k + 1 at hv
        have hi := i.isLt
        omega
      simpa only [fl_flush, forwardLoopCfg, Fin.addCases_left,
        Function.update_of_ne hne] using (hbank i).1
    | right j =>
      fin_cases j
      · have hne : tableTape k ≠ summaryTape k := by
          simp [tableTape, summaryTape, Fin.ext_iff]
        change Function.update d.workTapes (summaryTape k) _ (tableTape k) = _
        rw [Function.update_of_ne hne, htable]
        simp [forwardLoopCfg]
      · change Function.update d.workTapes (summaryTape k) _ (summaryTape k) = _
        rw [Function.update_self, hs, hsummary, summaryCells_update]
        simp [forwardLoopCfg]
  · funext j
    induction j using Fin.addCases with
    | left i =>
      simpa only [fl_flush, forwardLoopCfg, Fin.addCases_left] using (hbank i).2
    | right j =>
      fin_cases j
      · simpa only [fl_flush, forwardLoopCfg, Fin.addCases_right] using ht
      · simpa only [fl_flush, forwardLoopCfg, Fin.addCases_right] using hs
  · simp only [fl_flush, forwardLoopCfg, read]

/-- One released block followed by its leaf returns or halts with the summary forwarded,
never halts early, and visits no work-head position beyond the block's visited sets.

**Proof sketch.** The phase lemma supplies the block prefix and its one-step dispatch.
The leaf takes one stationary step; a live leaf takes one more stationary dispatch. Split
interior and visiting times at the block exit; all later head positions are its final ones.
This composes the public hosts and does not iterate rounds. -/
private lemma fl_round {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x) (T : ℕ) (live : Bool)
    (hd : d.state = some (exit k true)) (hT : 0 < T)
    (he : ((roundTM k).runFrom d T).state = some (exit k live))
    (hc : ∀ u, 0 < u → u < T → ∃ s, ((roundTM k).runFrom d u).state = some s ∧
      s ≠ exit k true ∧ s ≠ exit k false) :
    (fl_TM k).runFrom (d.mapState fun _ => fl_entry k) (T + if live then 3 else 2) =
        fl_flush (if live then some (fl_entry k) else none) ((roundTM k).runFrom d T) ∧
    (∀ u < T + if live then 3 else 2,
      ((fl_TM k).runFrom (d.mapState fun _ => fl_entry k) u).state ≠ none) ∧
    ∀ j, (fl_TM k).visitedByTapeHead (d.mapState fun _ => fl_entry k)
        (T + if live then 3 else 2) j ⊆ (roundTM k).visitedByTapeHead d T j := by
  obtain ⟨hprefix, dispatch⟩ := fl_block_phase d T live hd hT he hc
  have emitted : (fl_TM k).runFrom (d.mapState fun _ => fl_entry k) (T + 2) =
      fl_flush (if live then some ⟨true, none⟩ else none) ((roundTM k).runFrom d T) := by
    rw [show T + 2 = (T + 1) + 1 by omega, MultiTapeTM.runFrom_succ_eq_step', dispatch,
      fl_hosted_leaf _ live (by rw [he]; exact Option.some_ne_none _)]
  have finish : (fl_TM k).runFrom (d.mapState fun _ => fl_entry k)
      (T + if live then 3 else 2) =
      fl_flush (if live then some (fl_entry k) else none) ((roundTM k).runFrom d T) := by
    cases live
    · exact emitted
    · change (fl_TM k).runFrom _ (T + 3) = _
      rw [show T + 3 = (T + 2) + 1 by omega, MultiTapeTM.runFrom_succ_eq_step', emitted]
      exact fl_return _
  refine ⟨finish, ?_, ?_⟩
  · intro u hu
    by_cases hb : u ≤ T
    · rw [hprefix u hb]
      by_cases hz : u = 0
      · simp [hz, Cfg.mapState, hd]
      · change (((seamReleaseTM (roundTM k) (exit k true)).runFrom _ u).mapState _).state ≠ _
        rw [seamReleaseTM_runFrom_pos (roundTM k) (exit k true) d hd u
          (Nat.pos_of_ne_zero hz)]
        by_cases hend : u = T
        · simp [hend, Cfg.mapState, he]
        · obtain ⟨s, hs, _, _⟩ := hc u (Nat.pos_of_ne_zero hz) (by omega)
          simp [Cfg.mapState, hs]
    · have ht : u = T + 1 ∨ (live = true ∧ u = T + 2) := by
        cases live <;> simp at hu ⊢ <;> omega
      rcases ht with rfl | ⟨rfl, rfl⟩
      · rw [dispatch]
        simp [Cfg.mapState, he]
      · rw [emitted]
        simp [fl_flush]
  · intro j p hp
    obtain ⟨u, hu, hpos⟩ := Finset.mem_image.mp hp
    have hu' : u ≤ T + if live then 3 else 2 := by
      simpa only [Finset.mem_range, Nat.lt_succ_iff] using hu
    have heads : ((fl_TM k).runFrom (d.mapState fun _ => fl_entry k) u).workTapePos j =
        ((roundTM k).runFrom d (min u T)).workTapePos j := by
      by_cases hb : u ≤ T
      · rw [min_eq_left hb, hprefix u hb]
        by_cases hz : u = 0
        · simp [hz, Cfg.mapState]
        · rw [show (fl_part k false) = seamReleaseTM (roundTM k) (exit k true) from rfl,
            seamReleaseTM_runFrom_pos (roundTM k) (exit k true) d hd u
              (Nat.pos_of_ne_zero hz)]
          rfl
      · rw [min_eq_right (by omega)]
        have ht : u = T + 1 ∨ u = T + 2 ∨ (live = true ∧ u = T + 3) := by
          cases live <;> simp at hu' ⊢ <;> omega
        rcases ht with rfl | rfl | ⟨rfl, rfl⟩
        · rw [dispatch]; rfl
        · rw [emitted]; rfl
        · rw [show T + 3 = T + if true then 3 else 2 from rfl, finish]; rfl
    apply Finset.mem_image.mpr
    refine ⟨min u T, Finset.mem_range.mpr (by omega), ?_⟩
    exact heads.symm.trans hpos

/-- A live source step has a positive bounded forwarding simulation, its exact successor,
a strict live interior, and the mirrored and auxiliary visited-set bounds.

**Proof sketch.** Apply the block's visited-set contract to the source with erased output,
using the reset invariant but retaining the physical output. Compose its exit with the leaf.
The summary is the single current emission; restoring the blank summary therefore gives
exactly the actual successor, including its original output prefix. -/
private lemma fl_step_exists {k n : ℕ} {x : List Bool}
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x)
    (hl : c.state ≠ none) :
    ∃ t, 0 < t ∧ t ≤ 50 * ((tableBitsK M.tr).length + 1) + 3 ∧
      (fl_TM k).runFrom (forwardLoopCfg (fl_entry k) M c) t =
        forwardLoopCfg (fl_entry k) M (M.step c) ∧
      (∀ u < t, ((fl_TM k).runFrom (forwardLoopCfg (fl_entry k) M c) u).state ≠ none) ∧
      (∀ i : Fin k, (fl_TM k).visitedByTapeHead (forwardLoopCfg (fl_entry k) M c) t
        (Fin.castAdd 2 i) ⊆ {c.workTapePos i, (M.step c).workTapePos i}) ∧
      ∀ j : Fin (k + 2), k ≤ j.val →
        (fl_TM k).visitedByTapeHead (forwardLoopCfg (fl_entry k) M c) t j ⊆
          Finset.Icc (-((50 * ((tableBitsK M.tr).length + 1) + 3 : ℕ) : ℤ))
            ((50 * ((tableBitsK M.tr).length + 1) + 3 : ℕ) : ℤ) := by
  obtain ⟨q, hq⟩ := Option.ne_none_iff_exists'.mp hl
  let a := M.tr q c.inputSymbol c.workTapeSymbols
  let d := forwardLoopCfg (exit k true) M c
  have hd : d.state = some (exit k true) := by simp [d, forwardLoopCfg, hq]
  obtain ⟨T, hpos, hbound, he, hr, hc, ho, hm, ha⟩ :=
    roundTM_visited M.tr {c with output := []} q hq d hd (fl_reset M c)
  have joined := fl_round d T a.state.isSome hd hpos he hc
  have start : d.mapState (fun _ => fl_entry k) = forwardLoopCfg (fl_entry k) M c :=
    fl_encode_map (exit k true) (fun _ => fl_entry k) M c
  rw [start] at joined
  have emission : outputSummary (a.apply {c with output := []}).output = a.output := by
    simpa only [Action.apply, outputSummary, summaryEmit_none] using
      outputSummary_append [] a.output
  have out : ((roundTM k).runFrom d T).output = c.output := ho T le_rfl
  have source :
      {a.apply {c with output := []} with
        output := ((roundTM k).runFrom d T).output ++
          (outputSummary (a.apply {c with output := []}).output).toList} = M.step c := by
    rw [emission, out]
    simp only [MultiTapeTM.step, hq]
    rfl
  have control : (a.apply {c with output := []}).state.map (fun _ => fl_entry k) =
      if a.state.isSome then some (fl_entry k) else none := by
    cases hs : a.state <;> simp [Action.apply, hs]
  have restored := fl_restore (fl_entry k) M (a.apply {c with output := []})
    ((roundTM k).runFrom d T) hr
  rw [source, control] at restored
  refine ⟨T + if a.state.isSome then 3 else 2, ?_, ?_, joined.1.trans restored,
    joined.2.1, ?_, ?_⟩
  · split <;> omega
  · split <;> omega
  · intro i p hp
    have h := hm i (joined.2.2 (Fin.castAdd 2 i) hp)
    simpa only [MultiTapeTM.step, hq, Action.apply] using h
  · intro j hj p hp
    have h := ha j hj (joined.2.2 j hp)
    simp only [Finset.mem_Icc] at h ⊢
    push_cast
    omega

/-- Both auxiliary heads of every encoding are home, including for halted sources and
zero work tapes. -/
private lemma fl_home {k n : ℕ} {Q : Type} {x : List Bool} (entry : Q)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x)
    (j : Fin (k + 2)) (hj : k ≤ j.val) : (forwardLoopCfg entry M c).workTapePos j = 0 := by
  induction j using Fin.addCases with
  | left i => have hi := i.isLt; simp only [Fin.coe_castAdd] at hj; omega
  | right j => simp only [forwardLoopCfg, Fin.addCases_right]

/-- **H2-F, the forwarding round loop** (host H2's unclocked loop form; `machine-library-design.md`
§14; U2 design note `audits/evidence/u2/U2-design.md` §2.2). For each work-tape count `k` there
is one finite `(k + 2)`-tape binary machine `L`, with an entry state and a constant `C`, such
that for every `k`-work-tape binary machine `M` with `n + 1` states and every input `x`, `L`
simulates `M` step by step through `forwardLoopCfg`: there is a cost function on `M`'s
configurations with
* **a macro-step lockstep** (`Turing.Lockstep`, exits `{none}` on both sides): from the encoding
  of a live configuration `c`, `L` reaches the encoding of `M.step c` after exactly `cost c`
  steps, without halting before; the encoding of a halted configuration is halted. So `L` halts
  exactly when `M` does, with `M`'s output, which it emits step by step
  (`Turing.Lockstep.runFrom`, `Turing.Lockstep.cut`);
* **a per-step bound**: `0 < cost c ≤ 50 · (|tableBitsK M.tr| + 1) + C` for live `c`, a constant
  for a fixed table (the step block's bound plus the leaf's);
* **a space ledger**: over `t` source steps that are live before `t`, each mirrored head visits
  only cells that `M`'s head visits, and the two auxiliary heads stay within
  `50 · (|tableBitsK M.tr| + 1) + C` of their home.

The control of `L` does not depend on `M`, `n` or `x`. [AB09, §1.4.1, proof of Theorem 1.9:
each step of `M` is simulated in a number of steps depending only on `M`'s table]

**Proof sketch.** `L` is the step block `roundTM k` followed, at either of its exits, by a
forwarding leaf on the summary tape: read the summary cell, emit its bit if it holds one, blank
it, then return to the block's entry (exit `entry`) or halt (exit `halt`). For a live `c`, the
block-control part of `forwardLoopCfg entry M c` represents `{c with output := []}` (`Rep`: a blank
summary, the unary state word, both auxiliary heads home), so `roundTM_effect` gives a first exit
after `T ≤ 50 · (|tableBitsK M.tr| + 1)` steps at the representation of
`a.apply {c with output := []}`, silent before, with summary cell
`summaryEmit none a.output = a.output` (`summaryEmit_none`); the exit time is unique
(`roundTM_firstExit_unique`). The leaf emits `a.output`, making the physical output
`(a.apply c).output`, blanks the cell, and enters `entry` or halts according to the exit: the
result is exactly `forwardLoopCfg entry M (M.step c)`. Take `cost c` the block's exit time plus the
leaf's length (at most `C`); the inner times are live by the block's cut and the leaf's. The space
ledger is `roundTM_visited` per round, summed by `MultiTapeTM.visitedByTapeHead_add_subset`; the
leaf moves no head.

**Fill appendix (H2-F).** The two phases are the released block and the forwarding leaf;
`fl_round` proves their composition from the public phase lemma and the block's cut. The
leaf alone emits and clears, and all its heads are stationary. One exit-time witness is chosen
once per live source configuration, together with all its run and visited-set properties.
The cost is that block time plus three for a live successor, or plus two for a halt, so
`C = 3`. `Lockstep.runFrom` supplies iteration; the mirrored ledger uses visited-set addition,
and the auxiliary ledger uses `Lockstep.locate`. -/
theorem exists_forwardLoopTM (k : ℕ) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM (k + 2) Bool Q)
      (entry : Q) (C : ℕ),
      ∀ {n : ℕ} {x : List Bool} (M : MultiTapeTM k Bool (Fin (n + 1))),
        ∃ cost : Cfg k Bool (Fin (n + 1)) x → ℕ,
          Lockstep M L (forwardLoopCfg entry M) cost {none} {none} ∧
          (∀ c, c.state ≠ none → 0 < cost c ∧
            cost c ≤ 50 * ((tableBitsK M.tr).length + 1) + C) ∧
          ∀ (c : Cfg k Bool (Fin (n + 1)) x) (t : ℕ), (∀ u < t, (M.runFrom c u).state ≠ none) →
            (∀ i : Fin k, L.visitedByTapeHead (forwardLoopCfg entry M c)
                (∑ u ∈ Finset.range t, cost (M.runFrom c u)) (Fin.castAdd 2 i) ⊆
              M.visitedByTapeHead c t i) ∧
            ∀ j : Fin (k + 2), k ≤ j.val →
              L.visitedByTapeHead (forwardLoopCfg entry M c)
                  (∑ u ∈ Finset.range t, cost (M.runFrom c u)) j ⊆
                Finset.Icc (-((50 * ((tableBitsK M.tr).length + 1) + C : ℕ) : ℤ))
                  ((50 * ((tableBitsK M.tr).length + 1) + C : ℕ) : ℤ) := by
  classical
  refine ⟨(Σ b, fl_Ph k b), inferInstance, inferInstance, fl_TM k, fl_entry k, 3, ?_⟩
  intro n x M
  let cost (c : Cfg k Bool (Fin (n + 1)) x) : ℕ :=
    if h : c.state ≠ none then (fl_step_exists M c h).choose else 1
  have chosen (c : Cfg k Bool (Fin (n + 1)) x) (h : c.state ≠ none) :
      cost c = (fl_step_exists M c h).choose := dif_pos h
  have one (c : Cfg k Bool (Fin (n + 1)) x) (h : c.state ≠ none) :=
    (fl_step_exists M c h).choose_spec
  have lock : Lockstep M (fl_TM k) (forwardLoopCfg (fl_entry k) M) cost {none} {none} := by
    constructor
    · intro c h
      have alive : c.state ≠ none := by simpa only [Set.mem_singleton_iff] using h
      rw [chosen c alive]
      exact (one c alive).2.2.1
    · intro c h
      simp only [Set.mem_singleton_iff] at h ⊢
      simp [forwardLoopCfg, h]
    · intro c h u hu
      have alive : c.state ≠ none := by simpa only [Set.mem_singleton_iff] using h
      rw [chosen c alive] at hu
      exact (one c alive).2.2.2.1 u hu
  refine ⟨cost, lock, ?_, ?_⟩
  · intro c h
    rw [chosen c h]
    exact ⟨(one c h).1, (one c h).2.1⟩
  · intro c t ht
    constructor
    · -- Split visited sets at each completed round; the lockstep supplies its start.
      induction t with
      | zero => simp [MultiTapeTM.visitedByTapeHead, forwardLoopCfg]
      | succ t ih =>
        have ht' : ∀ u < t, (M.runFrom c u).state ≠ none := fun u hu => ht u (by omega)
        have alive := ht t (by omega)
        have atRound := lock.runFrom c t ht'
        intro i p hp
        rw [Finset.sum_range_succ] at hp
        have split := MultiTapeTM.visitedByTapeHead_add_subset (fl_TM k)
          (forwardLoopCfg (fl_entry k) M c)
          (∑ u ∈ Finset.range t, cost (M.runFrom c u)) (cost (M.runFrom c t))
          (Fin.castAdd 2 i) hp
        rcases Finset.mem_union.mp split with before | current
        · have old := ih ht' i before
          exact (Finset.image_subset_image (Finset.range_mono (Nat.le_succ (t + 1)))) old
        · rw [atRound, chosen (M.runFrom c t) alive] at current
          have h := (one (M.runFrom c t) alive).2.2.2.2.1 i current
          rcases Finset.mem_insert.mp h with rfl | hend
          · exact Finset.mem_image.mpr ⟨t, Finset.mem_range.mpr (by omega), rfl⟩
          · have heq := Finset.mem_singleton.mp hend
            apply Finset.mem_image.mpr
            refine ⟨t + 1, Finset.mem_range.mpr (by omega), ?_⟩
            simpa only [MultiTapeTM.runFrom_succ_eq_step'] using heq.symm
    · -- Locate an auxiliary-head visit in one round, or at an encoded start.
      intro j hj p hp
      obtain ⟨v, hv, rfl⟩ := Finset.mem_image.mp hp
      obtain ⟨u, _, w, hw, run⟩ := lock.locate c t ht v
        (Nat.lt_succ_iff.mp (Finset.mem_range.mp hv))
      rw [run]
      rcases hw with rfl | ⟨alive, hw⟩
      · rw [MultiTapeTM.runFrom_zero, fl_home (fl_entry k) M (M.runFrom c u) j hj]
        exact Finset.mem_Icc.mpr ⟨neg_nonpos.mpr (Nat.cast_nonneg _), Nat.cast_nonneg _⟩
      · rw [chosen (M.runFrom c u) alive] at hw
        exact (one (M.runFrom c u) alive).2.2.2.2.2 j hj
          (Finset.mem_image.mpr ⟨w, Finset.mem_range.mpr (Nat.lt_succ_iff.mpr hw), rfl⟩)

end StepBlock

/-!
# Machine-construction library: input devices (host H5)

An **input device** presents a *virtual* input `y` at a virtual head position `i` inside a host
configuration over a *physical* input `z`, and supplies the two operations a simulated machine
needs: read the virtual cell (`Turing.inputCell y i`: blank at `0` and past the end, the
input-tape convention of `Turing.Cfg.inputSymbol`), and move the virtual head by the clamped
move `Turing.moveInputPos`. It is host H5 of the registry (`machine-library-design.md` §14).
Two devices are stated here.
* **Native** (`Turing.NativeAt`): `z = y`, and the physical head is the virtual one. Nothing is
  stored, and a host step is a source step (literal lockstep). It is the device of H2's step
  block and of `U₀` (`TuringMachine/SelfUniversal.lean`).
* **Bounded-prefix ruler** (`Turing.RulerAt`): `z` is a header of length `ℓ > 0` followed by
  `y`, the physical head is at `ℓ + i`, and a work tape holds the unary ruler `true^ℓ`. Since
  `ℓ > 0`, the physical cell `p - ℓ` is blank exactly at the virtual left boundary, so a
  restoring sweep of `ℓ` cells tests it without clamping. Its storage is the ruler's `ℓ + 2`
  cells. It is the device of the addend-free `Complexity.space_universal`, whose header length
  is linear in the budget.

The buffered device is Z1 (`Build/VirtualInput.lean`). The binary-position device of the
registry row is not stated here; no current consumer needs it.

**The ruler host is a block simulation.** `Turing.exists_rulerHostTM` turns any native-input machine
into one on the suffix input: each source step is one host step, followed by a boundary test after a
left move, so `t` source steps take between `t` and `C · (ℓ + 1) · t` host steps. At every host time
the bank, the output and the ruler are those of the source at some source time (the intermediate
frames), the host does not halt before the source does (the exit cut), and the time and space
ledgers are separate clauses. Literal lockstep is the native special case.

**Status.** Everything here is proved: the devices' correctness facts, the restoring boundary
test `Turing.exists_rulerTestTM`, the call rows `Turing.exists_rulerDeviceTM` and the ruler host
`Turing.exists_rulerHostTM`.

## Main definitions

* `Turing.inputCell` — the virtual input cell.
* `Turing.NativeAt` — the native presentation.
* `Turing.RulerAt` — the bounded-prefix ruler presentation.
* `Turing.rulerHostCfg` — the ruler host's layout, with its field equations
  (`Turing.rulerHostCfg_state` and the like).

## Main results

* `Turing.Cfg.inputSymbol_eq_inputCell`, `Turing.NativeAt.inputSymbol_eq`,
  `Turing.NativeAt.move`, `Turing.nativeAt_init` — the native device (proved).
* `Turing.inputCell_sub_eq_none_iff`, `Turing.inputCell_of_drop` — the ruler device's
  correctness facts, the empty payload included (proved).
* `Turing.pairEncode_drop`, `Turing.pairEncode_pairEncode_length`,
  `Turing.pairEncode_pairEncode_drop` — the payload after one pairing's header, and the header of
  `Complexity.space_universal`'s input (proved).
* `Turing.exists_rulerTestTM` — the restoring boundary test (proved).
* `Turing.exists_rulerDeviceTM`, `Turing.exists_rulerHostTM` — the call rows and the host
  (proved).

/-! ### The interface and the native device -/

/-- **The cell of a virtual input at position `i`** (the input-tape convention of
`Turing.Cfg.inputSymbol`): blank at `0` and beyond the word, the `i`-th symbol otherwise.
[AB09, §1.2: the input tape, with blanks at both ends] -/
def inputCell {Γ : Type*} (y : List Γ) (i : ℕ) : Option Γ :=
  if i = 0 then none else y[i - 1]?

/-- The native input symbol is the input cell at the head.

**Proof sketch.** Compare the three cases of `Turing.Cfg.inputSymbol`: position `0`, the right
blank `|y| + 1` (where the lookup is out of range), and the interior
(`Turing.inputSymbolInner`). -/
theorem Cfg.inputSymbol_eq_inputCell {k : ℕ} {Γ S : Type*} {y : List Γ}
    (c : Cfg k Γ S y) : c.inputSymbol = inputCell y c.inputPos.val := by
  have hlt := c.inputPos.isLt
  by_cases h0 : c.inputPos.val = 0
  · have hz : c.inputPos = 0 := Fin.ext h0
    simp [Cfg.inputSymbol, inputCell, hz]
  · by_cases hr : c.inputPos.val = y.length + 1
    · have hz : c.inputPos ≠ 0 := fun h => h0 (by rw [h]; rfl)
      simp only [Cfg.inputSymbol, inputCell, dif_neg hz, dif_pos hr, if_neg h0]
      rw [List.getElem?_eq_none (by omega)]
    · have hin := inputSymbolInner (cfg := c) (c.inputPos.val - 1) (by omega) (by omega)
      rw [hin]
      simp only [inputCell, if_neg h0]
      rw [List.getElem?_eq_getElem (by omega)]

/-- **H5, the native device**: a host configuration presents its own input `y` at virtual
position `i` when its physical input head is at `i`. Reading is the native read, and moving is
the native clamped move; there is nothing to store. This is the device of H2's step block
(its seam relation's input clause) and of `U₀`, whose virtual input is all of its input.
[AB09, §1.2] -/
def NativeAt {K : ℕ} {Γ S : Type*} {y : List Γ} (d : Cfg K Γ S y) (i : Fin (y.length + 2)) :
    Prop :=
  d.inputPos = i

/-- The native device reads the virtual input cell.

**Proof sketch.** `Turing.Cfg.inputSymbol_eq_inputCell` at the head. -/
theorem NativeAt.inputSymbol_eq {K : ℕ} {Γ S : Type*} {y : List Γ} {d : Cfg K Γ S y}
    {i : Fin (y.length + 2)} (h : NativeAt d i) : d.inputSymbol = inputCell y i.val := by
  rw [Cfg.inputSymbol_eq_inputCell, h]

/-- The native device moves the virtual head by the clamped move. -/
theorem NativeAt.move {K : ℕ} {Γ S : Type*} {y : List Γ} {d : Cfg K Γ S y}
    {i : Fin (y.length + 2)} (h : NativeAt d i) (m : SignType) :
    NativeAt { d with inputPos := moveInputPos d.inputPos m } (moveInputPos i m) := by
  unfold NativeAt at h ⊢
  rw [h]

/-- The initial configuration presents its input natively at position `1`, the position at
which `U₀`'s startup must leave its input head (the RD-3 gate's check 2). -/
theorem nativeAt_init {K : ℕ} {Γ S : Type*} (q₀ : S) (y : List Γ) :
    NativeAt (Cfg.init q₀ y : Cfg K Γ S y) 1 := rfl

/-! ### The bounded-prefix ruler device -/

/-- **The boundary test reads a blank exactly at the virtual left boundary.** If the header
length `ℓ` is positive and the physical head is at `p` with `ℓ ≤ p ≤ |z| + 1`, then the
physical input cell `p - ℓ` is blank exactly when `p = ℓ`. This holds for every `z`, so for an
empty payload as well (the RD-3 gate's suggested sanity obligation for the bounded-prefix
device).

**Proof sketch.** At `p = ℓ` the cell is position `0`. Otherwise `1 ≤ p - ℓ ≤ |z|`, since
`ℓ ≥ 1`, and the cell holds an input symbol. -/
theorem inputCell_sub_eq_none_iff {z : List Bool} {ℓ p : ℕ} (hℓ : 0 < ℓ) (hp : ℓ ≤ p)
    (hpz : p ≤ z.length + 1) : inputCell z (p - ℓ) = none ↔ p = ℓ := by
  unfold inputCell
  by_cases h : p - ℓ = 0
  · simp only [if_pos h, true_iff]
    omega
  · rw [if_neg h, List.getElem?_eq_getElem (by omega)]
    simp only [reduceCtorEq, false_iff]
    omega

/-- **Away from the boundary the virtual cell is the physical one.** If `y` is `z` without
its first `ℓ` symbols, the virtual cell `i ≠ 0` of `y` is the physical cell `ℓ + i` of `z`,
the right blank included.

**Proof sketch.** Both are the lookup of `z` at `ℓ + i - 1` (`List.getElem?_drop`). -/
theorem inputCell_of_drop {z y : List Bool} {ℓ : ℕ} (h : z.drop ℓ = y) {i : ℕ}
    (hi : i ≠ 0) : inputCell y i = inputCell z (ℓ + i) := by
  subst h
  unfold inputCell
  rw [if_neg hi, if_neg (by omega), List.getElem?_drop]
  congr 1
  omega

/-- **H5, the bounded-prefix ruler device** (the RD-3 gate's row 1): a host configuration `d`
over the physical input `z` presents the virtual input `y` at virtual position `i` with header
length `ℓ` and ruler tape `r` when `z` is a header of length `ℓ > 0` followed by `y`, the
physical head is at `ℓ + i`, and tape `r` holds, from its head, the delimited unary ruler
`true^ℓ`. Since `ℓ > 0`, the physical cell `i` is blank exactly when `i = 0`, so a sweep of
`ℓ` cells left, a read and a sweep back test the virtual left boundary without clamping. The
ruler's cells are the device's whole storage: `ℓ + 2` visited cells, however far the virtual
head travels. [AB09, §1.2; Exercise 4.1] -/
def RulerAt {K : ℕ} {S : Type*} {y z : List Bool} (ℓ : ℕ) (r : Fin K) (d : Cfg K Bool S z)
    (i : Fin (y.length + 2)) : Prop :=
  0 < ℓ ∧ ℓ ≤ z.length ∧ z.drop ℓ = y ∧ d.inputPos.val = ℓ + i.val ∧
    ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes r (d.workTapePos r + p) = FinTM.bufferTape (List.replicate ℓ true) p

/-- **The header length of `Complexity.space_universal`'s input.** The input
`⟨bits s, ⟨α, x⟩⟩ = pairEncode (bits s) (pairEncode α x)` is a header of length
`2|bits s| + 2|α| + 4` followed by `x`.

**Proof sketch.** Each pairing doubles its first word and adds two separator cells
(`Turing.universal_pair_length`). -/
theorem pairEncode_pairEncode_length (u α x : List Bool) :
    (pairEncode u (pairEncode α x)).length = (2 * u.length + 2 * α.length + 4) + x.length := by
  rw [universal_pair_length, universal_pair_length]
  omega

/-- **The payload after the header.** Dropping the header of length `2|u| + 2` from
`pairEncode u x` leaves `x`: the ruler device's `z.drop ℓ = y` for the code-first input.

**Proof sketch.** The pairing is the header `pairEncode u []` followed by `x`, and that header
has length `2|u| + 2` (`Turing.length_pairEncode`). -/
theorem pairEncode_drop (u x : List Bool) : (pairEncode u x).drop (2 * u.length + 2) = x := by
  have split : pairEncode u x = pairEncode u [] ++ x := by
    rw [pairEncode_eq, pairEncode_eq, List.append_nil]
  have width : (pairEncode u []).length = 2 * u.length + 2 := by
    rw [length_pairEncode, List.length_nil, Nat.add_zero]
  rw [split, ← width, List.drop_left]

/-- **The payload after the header.** Dropping the header of length `2|u| + 2|α| + 4` from
`pairEncode u (pairEncode α x)` leaves `x`: the ruler device's `z.drop ℓ = y`.

**Proof sketch.** Drop the two pairings' prefixes one after the other
(`Turing.pairEncode_drop`). -/
theorem pairEncode_pairEncode_drop (u α x : List Bool) :
    (pairEncode u (pairEncode α x)).drop (2 * u.length + 2 * α.length + 4) = x := by
  rw [show 2 * u.length + 2 * α.length + 4 = (2 * u.length + 2) + (2 * α.length + 2) by ring,
    ← List.drop_drop, pairEncode_drop, pairEncode_drop]

/-- The forward state, followed by return and verdict states carrying one Boolean. -/
private abbrev rd_State := Unit ⊕ Bool × Bool

/-- The ruler test is a leaf: scan jointly left/right, read once, and return jointly.
No catalog row moves the input head together with a work head. -/
private def rd_test : MultiTapeTM 1 Bool rd_State where
  q₀ := .inl ()
  tr q v w := match q with
    | .inl _ => if w 0 = some true then tapeAction 0 (-1) none 1 (.inl ())
        else tapeAction 0 0 none (-1) (.inr (v.isNone, false))
    | .inr (b, false) => if w 0 = some true then tapeAction 0 1 none (-1) (.inr (b, false))
        else tapeAction 0 0 none 1 (.inr (b, true))
    | .inr (b, true) => tapeAction 0 0 none 0 (.inr (b, true))

/-- A point of the restoring trace, with distance from the original input head. -/
private def rd_cfg {z : List Bool} (d : Cfg 1 Bool rd_State z) (q : rd_State)
    (n : ℕ) (e : ℤ) : Cfg 1 Bool rd_State z :=
  { d with
    state := some q
    inputPos := ⟨d.inputPos.val - n, by have := d.inputPos.isLt; omega⟩
    workTapePos := fun i => d.workTapePos i + e }

/-- The verdict says precisely whether the physical head equals the header length. -/
private def rd_verdict {z : List Bool} (d : Cfg 1 Bool rd_State z) (ℓ : ℕ) : Bool :=
  decide (d.inputPos.val = ℓ)

/-- The whole finite trace has a forward leg, a return leg and one restored endpoint. -/
private def rd_trace {z : List Bool} (d : Cfg 1 Bool rd_State z) (ℓ u : ℕ) :=
  if u ≤ ℓ then rd_cfg d (.inl ()) u u
  else if u ≤ 2 * ℓ + 1 then
    let n := 2 * ℓ + 1 - u
    rd_cfg d (.inr (rd_verdict d ℓ, false)) n ((n : ℤ) - 1)
  else rd_cfg d (.inr (rd_verdict d ℓ, true)) 0 0

/-- The ruler gives a true cell at each interior natural offset. -/
private theorem rd_read_nat {z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool rd_State z)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p)
    (n : ℕ) (hn : n < ℓ) : d.workTapes 0 (d.workTapePos 0 + n) = some true := by
  rw [hr n (by omega) (by omega), FinTM.bufferTape_nat]
  simp only [List.getElem?_replicate, if_pos hn]

/-- One table row moves between two trace points using `Turing.step_of_tapeAction_move`. -/
private theorem rd_cfg_step {z : List Bool} (d : Cfg 1 Bool rd_State z)
    (q q' : rd_State) (n n' : ℕ) (e e' : ℤ) (a b : SignType)
    (ht : rd_test.tr q (rd_cfg d q n e).inputSymbol
      (rd_cfg d q n e).workTapeSymbols = tapeAction 0 a none b q')
    (hi : moveInputPos (rd_cfg d q n e).inputPos a = (rd_cfg d q' n' e').inputPos)
    (he : e + (b : ℤ) = e') :
    rd_test.step (rd_cfg d q n e) = rd_cfg d q' n' e' := by
  rw [step_of_tapeAction_move rd_test (rd_cfg d q n e) 0 a b rfl ht]
  refine Cfg.ext rfl hi rfl ?_ rfl
  funext i
  rw [Subsingleton.elim i 0]
  simp only [Function.update_self, rd_cfg, add_assoc, he]

/-- Each successive point follows the table, including both delimiter turns.

**Proof sketch.** Interior ruler cells move both heads; the right delimiter records the
semantic boundary test, and the left delimiter restores the ruler head before the verdict. -/
private theorem rd_trace_step {z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool rd_State z)
    (hℓ : 0 < ℓ) (hp : ℓ ≤ d.inputPos.val)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p)
    (u : ℕ) (hu : u < 2 * ℓ + 2) :
    rd_test.step (rd_trace d ℓ u) = rd_trace d ℓ (u + 1) := by
  by_cases hf : u < ℓ
  · simp only [rd_trace, if_pos (show u ≤ ℓ by omega),
      if_pos (show u + 1 ≤ ℓ by omega)]
    refine rd_cfg_step d _ _ _ _ _ _ (-1) 1 ?_ ?_ (by simp only [SignType.coe_one]; omega)
    · simp only [rd_test, Cfg.workTapeSymbols_eq, rd_cfg, rd_read_nat ℓ d hr u hf,
        if_true]
    · rw [← SignType.neg_eq_neg_one]
      rw [moveInputPos_neg_of_ne_left (rd_cfg d (.inl ()) u u).inputPos (by
        intro h; have hh := congrArg Fin.val h; simp only [rd_cfg, Fin.val_zero] at hh
        omega)]
      apply Fin.ext
      simp only [rd_cfg]
      omega
  · by_cases ht : u = ℓ
    · subst u
      have hb : (inputCell z (d.inputPos.val - ℓ)).isNone = rd_verdict d ℓ := by
        rw [Bool.eq_iff_iff]
        simp only [Option.isNone_iff_eq_none, rd_verdict, decide_eq_true_eq]
        exact inputCell_sub_eq_none_iff hℓ hp (by have := d.inputPos.isLt; omega)
      have hend : d.workTapes 0 (d.workTapePos 0 + ℓ) = none := by
        rw [hr ℓ (by omega) (by omega), FinTM.bufferTape_nat]
        simp only [List.getElem?_replicate, lt_self_iff_false, if_false]
      simp only [rd_trace, if_pos (show ℓ ≤ ℓ from le_rfl),
        if_neg (show ¬ℓ + 1 ≤ ℓ by omega), if_pos (show ℓ + 1 ≤ 2 * ℓ + 1 by omega)]
      have hn : 2 * ℓ + 1 - (ℓ + 1) = ℓ := by omega
      rw [hn]
      refine rd_cfg_step d _ _ _ _ _ _ 0 (-1) ?_ ?_ (by
        simp only [SignType.coe_neg_one]; omega)
      · simp only [rd_test, Cfg.workTapeSymbols_eq, rd_cfg, hend, reduceCtorEq, if_false,
          Cfg.inputSymbol_eq_inputCell, hb]
      · exact moveInputPos_zero _
    · have hf' : ¬u ≤ ℓ := by omega
      have hf'' : ¬u + 1 ≤ ℓ := by omega
      have hrange : u ≤ 2 * ℓ + 1 := by omega
      by_cases he : u = 2 * ℓ + 1
      · subst u
        have hend : d.workTapes 0 (d.workTapePos 0 + (-1)) = none := by
          rw [hr (-1) (by omega) (by omega), FinTM.bufferTape_left]
        simp only [rd_trace, if_neg hf', if_neg hf'', if_pos (le_refl (2 * ℓ + 1)),
          if_neg (show ¬2 * ℓ + 1 + 1 ≤ 2 * ℓ + 1 by omega), Nat.sub_self,
          Nat.cast_zero, zero_sub]
        refine rd_cfg_step d _ _ _ _ _ _ 0 1 ?_ ?_ (by simp)
        · simp only [rd_test, Cfg.workTapeSymbols_eq, rd_cfg, hend, reduceCtorEq, if_false]
        · exact moveInputPos_zero _
      · have hnext : u + 1 ≤ 2 * ℓ + 1 := by omega
        let n := 2 * ℓ + 1 - u
        have hn : 0 < n ∧ n ≤ ℓ := by dsimp [n]; omega
        have hnextn : 2 * ℓ + 1 - (u + 1) = n - 1 := by dsimp [n]; omega
        have hcast : ((n - 1 : ℕ) : ℤ) = (n : ℤ) - 1 := by omega
        have hread : d.workTapes 0 (d.workTapePos 0 + ((n : ℤ) - 1)) = some true := by
          rw [← hcast]
          exact rd_read_nat ℓ d hr (n - 1) (by omega)
        simp only [rd_trace, if_neg hf', if_neg hf'', if_pos hrange, if_pos hnext]
        change rd_test.step (rd_cfg d (.inr (rd_verdict d ℓ, false)) n ((n : ℤ) - 1)) = _
        rw [hnextn]
        refine rd_cfg_step d _ _ _ _ _ _ 1 (-1) ?_ ?_ (by
          simp only [SignType.coe_neg_one]; omega)
        · simp only [rd_test, Cfg.workTapeSymbols_eq, rd_cfg, hread, if_true]
        · rw [← SignType.pos_eq_one]
          rw [moveInputPos_pos_of_ne_right
            (rd_cfg d (.inr (rd_verdict d ℓ, false)) n ((n : ℤ) - 1)).inputPos (by
            change d.inputPos.val - n ≠ z.length + 1
            have := d.inputPos.isLt; omega)]
          apply Fin.ext
          simp only [rd_cfg]
          omega

/-- The single induction that identifies every point of the finite restoring sweep. -/
private theorem rd_run {z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool rd_State z)
    (hd : d.state = some (.inl ())) (hℓ : 0 < ℓ) (hp : ℓ ≤ d.inputPos.val)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p)
    (u : ℕ) (hu : u ≤ 2 * ℓ + 2) : rd_test.runFrom d u = rd_trace d ℓ u := by
  induction u with
  | zero =>
    rw [MultiTapeTM.runFrom_zero]
    simp only [rd_trace, Nat.zero_le, if_true, rd_cfg, Nat.sub_zero, Nat.cast_zero, add_zero]
    exact Cfg.ext hd rfl rfl rfl rfl
  | succ u ih =>
    rw [MultiTapeTM.runFrom_succ_eq_step', ih (by omega)]
    exact rd_trace_step ℓ d hℓ hp hr u (by omega)

/-- **The restoring suffix-boundary test** (the ruler device's core). There is a one-tape binary
module with an entry and two distinct verdicts as follows. Start it at its entry with the
delimited ruler `true^ℓ`, `ℓ > 0`, at its head and the physical input head at a position `p ≥ ℓ`.
It reaches `yes` if `p = ℓ` and `no` otherwise, within `C · (ℓ + 1)` steps and no verdict
earlier, with nothing else changed: the input head is back at `p` and the ruler head at its
start, having stayed within `ℓ + 1` cells.
The input itself is arbitrary, so an empty payload is included. [AB09, §1.2]

**Proof sketch.** Move the input head left and the ruler head right in lockstep until the ruler
head reads the blank after the ruler: `ℓ` moves, none clamped since `p ≥ ℓ`. Read the input cell
at `p - ℓ`: it is blank exactly when `p - ℓ = 0`, since `p - ℓ ≤ |z| + 1 - ℓ ≤ |z|`
(`Turing.inputCell_sub_eq_none_iff`). Move both heads back `ℓ` cells (the ruler head stops at
the blank before the ruler, then steps right). A finite control carries the verdict.

**Implementation.** The test is a hand-built controller that moves the input head and the
ruler head in the same step (a `Turing.tapeAction`); no catalog row does that. One
forward-and-return trajectory, identified by a single induction, takes exactly `2ℓ + 2` steps, so
`C = 2`. The turn reads the boundary through `Turing.inputCell_sub_eq_none_iff`; the return leg
changes no tape and no output and restores both heads. -/
theorem exists_rulerTestTM :
    ∃ (P : Type) (_ : Fintype P) (_ : DecidableEq P) (R : MultiTapeTM 1 Bool P)
      (entry yes no : P) (C : ℕ), yes ≠ no ∧
      ∀ {z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool P z), d.state = some entry → 0 < ℓ →
        ℓ ≤ d.inputPos.val →
        (∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
          d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p) →
        ∃ t ≤ C * (ℓ + 1),
          (∀ u < t, (R.runFrom d u).state ≠ some yes ∧ (R.runFrom d u).state ≠ some no) ∧
          (∀ u ≤ t, |(R.runFrom d u).workTapePos 0 - d.workTapePos 0| ≤ ((ℓ + 1 : ℕ) : ℤ)) ∧
          R.runFrom d t = { d with state := some (if d.inputPos.val = ℓ then yes else no) } := by
  refine ⟨rd_State, inferInstance, inferInstance, rd_test, .inl (), .inr (true, true),
    .inr (false, true), 2, by decide, ?_⟩
  intro z ℓ d hd hℓ hp hr
  refine ⟨2 * ℓ + 2, by omega, ?_, ?_, ?_⟩
  · intro u hu
    rw [rd_run ℓ d hd hℓ hp hr u (by omega)]
    by_cases hf : u ≤ ℓ
    · simp only [rd_trace, if_pos hf, rd_cfg]
      constructor <;> intro h <;> cases h
    · have hu' : u ≤ 2 * ℓ + 1 := by omega
      simp only [rd_trace, if_neg hf, if_pos hu', rd_cfg]
      constructor <;> intro h
      all_goals exact Bool.false_ne_true (congrArg Prod.snd (Sum.inr.inj (Option.some.inj h)))
  · intro u hu
    rw [rd_run ℓ d hd hℓ hp hr u hu]
    by_cases hf : u ≤ ℓ
    · simp only [rd_trace, if_pos hf, rd_cfg, add_sub_cancel_left, abs_le]
      constructor <;> omega
    · by_cases hb : u ≤ 2 * ℓ + 1
      · simp only [rd_trace, if_neg hf, if_pos hb, rd_cfg, add_sub_cancel_left, abs_le]
        constructor <;> omega
      · simp only [rd_trace, if_neg hf, if_neg hb, rd_cfg, add_zero, sub_self, abs_zero]
        omega
  · rw [rd_run ℓ d hd hℓ hp hr (2 * ℓ + 2) (le_refl _)]
    simp only [rd_trace, if_neg (show ¬2 * ℓ + 2 ≤ ℓ by omega),
      if_neg (show ¬2 * ℓ + 2 ≤ 2 * ℓ + 1 by omega), rd_cfg, Nat.sub_zero, add_zero]
    refine Cfg.ext ?_ rfl rfl rfl rfl
    by_cases h : d.inputPos.val = ℓ
    · simp only [rd_verdict, h, decide_true, if_true]
    · simp only [rd_verdict, h, decide_false, if_false]

/-- The concrete test's frames and its two-sided head envelope, projected from `rd_run`.

Proof sketch: project each of the existing trace's three cases; no new trace induction. -/
private theorem rd_test_window {z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool rd_State z)
    (hd : d.state = some (.inl ())) (hℓ : 0 < ℓ) (hp : ℓ ≤ d.inputPos.val)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p)
    (u : ℕ) (hu : u ≤ 2 * ℓ + 2) :
    (rd_test.runFrom d u).workTapes = d.workTapes ∧
    (rd_test.runFrom d u).output = d.output ∧
    d.workTapePos 0 - 1 ≤ (rd_test.runFrom d u).workTapePos 0 ∧
    (rd_test.runFrom d u).workTapePos 0 ≤ d.workTapePos 0 + ℓ ∧
    (rd_test.runFrom d u).state ≠ none ∧
    (u < 2 * ℓ + 2 → ∀ b, (rd_test.runFrom d u).state ≠ some (.inr (b, true))) := by
  rw [rd_run ℓ d hd hℓ hp hr u hu]
  unfold rd_trace
  split_ifs with hf hb
  · simp only [rd_cfg, ne_eq, reduceCtorEq, not_false_eq_true, true_and]
    exact ⟨by omega, by omega, by simp⟩
  · simp only [rd_cfg, ne_eq, reduceCtorEq, not_false_eq_true, true_and]
    refine ⟨by omega, by omega, ?_⟩
    intro _ b h
    exact Bool.false_ne_true (congrArg Prod.snd (Sum.inr.inj (Option.some.inj h)))
  · simp only [rd_cfg, add_zero, ne_eq, reduceCtorEq, not_false_eq_true, true_and]
    exact ⟨by omega, by omega, fun h => False.elim (by omega)⟩

/-- The restored endpoint of the same test; no second trajectory argument is needed. -/
private theorem rd_test_return {z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool rd_State z)
    (hd : d.state = some (.inl ())) (hℓ : 0 < ℓ) (hp : ℓ ≤ d.inputPos.val)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p) :
    rd_test.runFrom d (2 * ℓ + 2) =
      { d with state := some (.inr (rd_verdict d ℓ, true)) } := by
  rw [rd_run ℓ d hd hℓ hp hr _ le_rfl]
  simp only [rd_trace, if_neg (show ¬2 * ℓ + 2 ≤ ℓ by omega),
    if_neg (show ¬2 * ℓ + 2 ≤ 2 * ℓ + 1 by omega), rd_cfg, Nat.sub_zero, add_zero]

/-- Read answers occupy three pairs; the unused fourth pair is the move exit. -/
private def rd_answer : Option Bool → Bool × Bool
  | none => (false, false)
  | some b => (true, b)

/-- Only a left move at the virtual boundary is suppressed. -/
private def rd_direction (m : SignType) (b : Bool) : SignType :=
  if b = true ∧ m = -1 then 0 else m

/-- The parameter selects a read or a move, sharing the same test and dispatch proof. -/
private def rd_leafMove (op : Option SignType) (b : Bool) : SignType :=
  op.elim 0 (fun m => rd_direction m b)

private def rd_leafAnswer (op : Option SignType) (b : Bool) (inp : Option Bool) : Bool × Bool :=
  op.elim (rd_answer (if b then none else inp)) (fun _ => (false, true))

/-- The leaf uses the public input-only action, including a stationary read. -/
private def rd_callLeaf (op : Option SignType) (b : Bool) : MultiTapeTM 1 Bool rd_State where
  q₀ := .inl ()
  tr := fun _ inp _ =>
    inputMoveAction (rd_leafMove op b) (.inr (rd_leafAnswer op b inp))

private abbrev rd_Phase := Option SignType ⊕ (Option SignType × Bool ⊕ Unit)
private abbrev rd_CallState := Σ _ : rd_Phase, rd_State

private def rd_family : rd_Phase → MultiTapeTM 1 Bool rd_State
  | .inr (.inl (op, b)) => rd_callLeaf op b
  | _ => rd_test

private def rd_redirect : (i : rd_Phase) → rd_State → Option rd_CallState
  | .inl op, .inr (b, true) => some ⟨.inr (.inl (op, b)), .inl ()⟩
  | .inr (.inl _), .inr a => some ⟨.inr (.inr ()), .inr a⟩
  | _, _ => none

private def rd_callTM : MultiTapeTM 1 Bool rd_CallState :=
  phaseHostTM rd_family ⟨.inl none, .inl ()⟩ rd_redirect

/-- Both call rows are two applications of the public phase lemma.

Proof sketch: concatenate the guarded test and one-step leaf, then distinguish their
phase tags before the final dispatch. The ruler envelope comes from `rd_test_window`. -/
private theorem rd_call_run {z : List Bool} (op : Option SignType) (ℓ : ℕ)
    (d : Cfg 1 Bool rd_CallState z) (hd : d.state = some ⟨.inl op, .inl ()⟩)
    (hℓ : 0 < ℓ) (hp : ℓ ≤ d.inputPos.val)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes 0 (d.workTapePos 0 + p) = FinTM.bufferTape (List.replicate ℓ true) p) :
    let b := decide (d.inputPos.val = ℓ)
    let a := rd_leafAnswer op b d.inputSymbol
    rd_callTM.runFrom d (2 * ℓ + 5) =
      { d with
        state := some ⟨.inr (.inr ()), .inr a⟩
        inputPos := moveInputPos d.inputPos (rd_leafMove op b) } ∧
    (∀ u < 2 * ℓ + 5, ∀ q, (rd_callTM.runFrom d u).state ≠
      some ⟨.inr (.inr ()), q⟩) ∧
    (∀ u ≤ 2 * ℓ + 5,
      |(rd_callTM.runFrom d u).workTapePos 0 - d.workTapePos 0| ≤ ((ℓ + 1 : ℕ) : ℤ)) := by
  let c : Cfg 1 Bool rd_State z :=
    ⟨some (.inl ()), d.inputPos, d.workTapes, d.workTapePos, d.output⟩
  let b := rd_verdict c ℓ
  let a := rd_leafAnswer op b c.inputSymbol
  let i : rd_Phase := .inr (.inl (op, b))
  let e : Cfg 1 Bool rd_State z := { c with state := some (.inr (b, true)) }
  let f : Cfg 1 Bool rd_State z := { c with
    state := some (.inr a)
    inputPos := moveInputPos c.inputPos (rd_leafMove op b) }
  have hc : c.state = some (.inl ()) := rfl
  have hstart : c.mapState (Sigma.mk (.inl op)) = d := by
    apply Cfg.ext
    · simp only [Cfg.mapState_state, hc, Option.map_some, hd]
    · exact Cfg.mapState_inputPos _ _
    · funext j p; rw [Cfg.mapState_workTapes]
    · funext j; rw [Cfg.mapState_workTapePos]
    · exact Cfg.mapState_output _ _
  have test := phaseHostTM_run rd_family ⟨.inl none, .inl ()⟩ rd_redirect
    (.inl op) c e (2 * ℓ + 2) (rd_test_return ℓ c hc hℓ hp hr)
    (.inr (b, true)) rfl ⟨i, .inl ()⟩ rfl (by
      intro u hu q hq
      have hcut := (rd_test_window ℓ c hc hℓ hp hr u (by omega)).2.2.2.2.2 hu
      cases q with
      | inl q => rfl
      | inr q =>
        rcases q with ⟨v, flag⟩
        cases flag
        · rfl
        · exact False.elim (hcut v hq))
  have hdispatch : e.mapState (fun _ => (⟨i, .inl ()⟩ : rd_CallState)) =
      c.mapState (Sigma.mk i) := by
    apply Cfg.ext
    · simp only [Cfg.mapState_state, e, hc, Option.map_some]
    · simp only [Cfg.mapState_inputPos, e]
    · funext j p; simp only [Cfg.mapState_workTapes, e]
    · funext j; simp only [Cfg.mapState_workTapePos, e]
    · simp only [Cfg.mapState_output, e]
  have hleaf : (rd_family i).runFrom c 1 = f := by
    rw [MultiTapeTM.runFrom_succ_eq_step', MultiTapeTM.runFrom_zero]
    exact step_of_inputMoveAction (rd_callLeaf op b) c (rd_leafMove op b) hc rfl
  have leaf := phaseHostTM_run rd_family ⟨.inl none, .inl ()⟩ rd_redirect i c f 1
    hleaf (.inr a) rfl ⟨.inr (.inr ()), .inr a⟩ rfl (by
      intro u hu q hq
      have hu0 : u = 0 := by omega
      rw [hu0, MultiTapeTM.runFrom_zero, hc] at hq
      have hq' := Option.some.inj hq
      subst q
      rfl)
  have hjoin : rd_callTM.runFrom d (2 * ℓ + 3) = c.mapState (Sigma.mk i) := by
    rw [← hstart]
    exact test.2.trans hdispatch
  have hlast : rd_callTM.runFrom d (2 * ℓ + 5) =
      f.mapState (fun _ => (⟨.inr (.inr ()), .inr a⟩ : rd_CallState)) := by
    rw [show 2 * ℓ + 5 = (2 * ℓ + 3) + 2 by omega, MultiTapeTM.runFrom_add, hjoin]
    exact leaf.2
  have hinp : c.inputSymbol = d.inputSymbol := by
    rw [Cfg.inputSymbol_eq_inputCell, Cfg.inputSymbol_eq_inputCell]
  refine ⟨?_, ?_, ?_⟩
  · rw [hlast]
    apply Cfg.ext
    · simp only [Cfg.mapState_state, f, Option.map_some, a, hinp, b, rd_verdict, c]
    · exact Cfg.mapState_inputPos _ _
    · funext j p; rw [Cfg.mapState_workTapes]
    · funext j; rw [Cfg.mapState_workTapePos]
    · exact Cfg.mapState_output _ _
  · intro u hu q hq
    by_cases ht : u ≤ 2 * ℓ + 2
    · rw [← hstart, test.1 u ht, Cfg.mapState_state] at hq
      obtain ⟨s, _, hs⟩ := Option.map_eq_some_iff.mp hq
      have := congrArg Sigma.fst hs
      cases this
    · have hv : u = (2 * ℓ + 3) + (u - (2 * ℓ + 3)) := by omega
      rw [hv, MultiTapeTM.runFrom_add, hjoin,
        leaf.1 (u - (2 * ℓ + 3)) (by omega), Cfg.mapState_state] at hq
      obtain ⟨s, _, hs⟩ := Option.map_eq_some_iff.mp hq
      have := congrArg Sigma.fst hs
      cases this
  · intro u hu
    by_cases ht : u ≤ 2 * ℓ + 2
    · rw [← hstart, test.1 u ht, Cfg.mapState_workTapePos,
        Cfg.mapState_workTapePos]
      have hw := rd_test_window ℓ c hc hℓ hp hr u ht
      rw [abs_le]
      constructor <;> have := hw.2.2 <;> push_cast <;> omega
    · by_cases he : u = 2 * ℓ + 5
      · rw [he, hlast, Cfg.mapState_workTapePos]
        simp only [f, c, sub_self, abs_zero]
        positivity
      · have hv : u = (2 * ℓ + 3) + (u - (2 * ℓ + 3)) := by omega
        rw [hv, MultiTapeTM.runFrom_add, hjoin,
          leaf.1 (u - (2 * ℓ + 3)) (by omega), Cfg.mapState_workTapePos]
        have hv' : u - (2 * ℓ + 3) = 0 ∨ u - (2 * ℓ + 3) = 1 := by omega
        rcases hv' with hv' | hv'
        · rw [hv', MultiTapeTM.runFrom_zero]
          simp only [c, sub_self, abs_zero]; positivity
        · rw [hv', hleaf]
          simp only [f, c, sub_self, abs_zero]; positivity

/-- Translation of the clamped input move, including both blanks of an empty payload.

Proof sketch: use the public left-decrement and right-boundary equations; the tagged
left blank suppresses its physical move. -/
private theorem rd_shift_move {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (p : Fin (z.length + 2)) (i : Fin (y.length + 2))
    (hp : p.val = ℓ + i.val) (b : Bool) (hb : b = true ↔ i.val = 0) (m : SignType) :
    (moveInputPos p (rd_direction m b)).val = ℓ + (moveInputPos i m).val := by
  cases m with
  | zero => simpa only [rd_direction, SignType.zero_eq_zero, reduceCtorEq,
      and_false, if_false, moveInputPos_zero] using hp
  | neg =>
    by_cases hi : i.val = 0
    · have hbt : b = true := hb.mpr hi
      have hm : rd_direction .neg b = 0 := by simp [rd_direction, hbt]
      rw [hm, moveInputPos_zero, FinTM.moveInputPos_neg_val]
      omega
    · have hbf : b ≠ true := fun h => hi (hb.mp h)
      have hm : rd_direction .neg b = .neg := by simp [rd_direction, hbf]
      rw [hm, FinTM.moveInputPos_neg_val, FinTM.moveInputPos_neg_val]
      omega
  | pos =>
    have hm : rd_direction .pos b = .pos := by simp [rd_direction]
    rw [hm]
    by_cases hi : i.val = y.length + 1
    · have he : i = ⟨y.length + 1, by omega⟩ := Fin.ext hi
      have hp' : p = ⟨z.length + 1, by omega⟩ := by
        apply Fin.ext; change p.val = z.length + 1; omega
      rw [he, hp', SignType.pos_eq_one, moveInputPos_rightBoundary,
        moveInputPos_rightBoundary]
      simp only; omega
    · rw [moveInputPos_pos_of_ne_right i hi,
        moveInputPos_pos_of_ne_right p (by omega)]
      simp only; omega

/-- The tagged physical read is the virtual read, including its right blank. -/
private theorem rd_tagged_read {k : ℕ} {S : Type} {y z : List Bool} (ℓ : ℕ)
    (hdrop : z.drop ℓ = y) (d : Cfg k Bool S z) (i : Fin (y.length + 2))
    (hp : d.inputPos.val = ℓ + i.val) (b : Bool) (hb : b = true ↔ i.val = 0) :
    (if b then none else d.inputSymbol) = inputCell y i.val := by
  cases b with
  | false =>
    have hi : i.val ≠ 0 := fun h => Bool.false_ne_true (hb.mpr h)
    change d.inputSymbol = inputCell y i.val
    rw [Cfg.inputSymbol_eq_inputCell, hp, ← inputCell_of_drop hdrop hi]
  | true =>
    have hi := hb.mp rfl
    simp only [if_true, hi, inputCell, if_true]

/-- **H5, the ruler device's call rows** ("read the virtual input cell", "move the virtual input
head"). There is a one-tape binary module with a read entry, three read verdicts
`readExit v`, a move entry per direction and a move exit, as follows, for every presentation
`Turing.RulerAt ℓ 0 d i` of a virtual input `y` at `i`.
* **Read**: from the read entry it reaches `readExit (inputCell y i)` within `C · (ℓ + 1)`
  steps, no read verdict earlier, with nothing else changed.
* **Move** by `m`: from `moveEntry m` it reaches the move exit within `C · (ℓ + 1)` steps, not
  earlier, with the physical head at `ℓ + (moveInputPos i m)` and every tape, every head and
  the output unchanged; so the result presents `y` at the clamped position `moveInputPos i m`.

In both, the ruler head stays within `ℓ + 1` cells of its start, so the device's work space is
`ℓ + 2` cells. These rows are the interface for a device-generic step block; the ruler host
below uses the boundary test directly. [AB09, §1.2]

**Proof sketch.** Both rows begin with the boundary test of `Turing.exists_rulerTestTM`. Read:
on `yes` answer blank; on `no` answer the native symbol under the head, which is
`y[i - 1]` or the right blank (`Turing.inputCell_of_drop`). Move: a right move is the native
clamped move (at `i = |y| + 1` the physical head is at `|z| + 1` and clamps exactly as
`moveInputPos` does); a stationary move does nothing; a left move is suppressed on `yes` and is
the native left move on `no`. Rows are joined by R2 seams; the dispatch steps add a constant.

**Implementation.** One phase host (`Turing.phaseHostTM_run`) runs the boundary test, a one-step
leaf built from `Turing.inputMoveAction` (stationary for a read; for a move the clamped move, with
a left move suppressed on `yes`), and a final dispatch. Both rows take exactly `2ℓ + 5` steps, so
`C = 4`; they are one argument with the operation as a parameter. -/
theorem exists_rulerDeviceTM :
    ∃ (P : Type) (_ : Fintype P) (_ : DecidableEq P) (R : MultiTapeTM 1 Bool P)
      (readEntry : P) (readExit : Option Bool → P) (moveEntry : SignType → P) (moveExit : P)
      (C : ℕ),
      Function.Injective readExit ∧ (∀ v, readExit v ≠ moveExit) ∧
      ∀ {y z : List Bool} (ℓ : ℕ) (d : Cfg 1 Bool P z) (i : Fin (y.length + 2)),
        RulerAt ℓ 0 d i →
        (d.state = some readEntry →
          ∃ t ≤ C * (ℓ + 1),
            (∀ u < t, ∀ v, (R.runFrom d u).state ≠ some (readExit v)) ∧
            (∀ u ≤ t, |(R.runFrom d u).workTapePos 0 - d.workTapePos 0| ≤ ((ℓ + 1 : ℕ) : ℤ)) ∧
            R.runFrom d t = { d with state := some (readExit (inputCell y i.val)) }) ∧
        (∀ m : SignType, d.state = some (moveEntry m) →
          ∃ t ≤ C * (ℓ + 1),
            (∀ u < t, (R.runFrom d u).state ≠ some moveExit) ∧
            (∀ u ≤ t, |(R.runFrom d u).workTapePos 0 - d.workTapePos 0| ≤ ((ℓ + 1 : ℕ) : ℤ)) ∧
            (R.runFrom d t).state = some moveExit ∧
            (R.runFrom d t).inputPos.val = ℓ + (moveInputPos i m).val ∧
            (R.runFrom d t).workTapes = d.workTapes ∧
            (R.runFrom d t).workTapePos = d.workTapePos ∧
            (R.runFrom d t).output = d.output) := by
  refine ⟨rd_CallState, inferInstance, inferInstance, rd_callTM,
    ⟨.inl none, .inl ()⟩, (fun v => ⟨.inr (.inr ()), .inr (rd_answer v)⟩),
    (fun m => ⟨.inl (some m), .inl ()⟩), ⟨.inr (.inr ()), .inr (false, true)⟩,
    4, ?_, ?_, ?_⟩
  · intro v w h
    cases v <;> cases w <;> simp_all [rd_answer]
  · intro v h
    cases v <;> simp_all [rd_answer]
  · intro y z ℓ d i h
    rcases h with ⟨hℓ, hlen, hdrop, hp, hr⟩
    have hz : z.length = ℓ + y.length := by
      have := congrArg List.length hdrop
      rw [List.length_drop] at this
      omega
    have hb : decide (d.inputPos.val = ℓ) = true ↔ i.val = 0 := by
      rw [decide_eq_true_eq]; omega
    constructor
    · intro hd
      obtain ⟨he, hcut, hspace⟩ := rd_call_run none ℓ d hd hℓ (by omega) hr
      refine ⟨2 * ℓ + 5, by omega, fun u hu v => hcut u hu (.inr (rd_answer v)),
        hspace, ?_⟩
      have hs := rd_tagged_read ℓ hdrop d i hp _ hb
      simpa only [rd_leafMove, rd_leafAnswer, Option.elim_none,
        moveInputPos_zero, hs] using he
    · intro m hd
      obtain ⟨he, hcut, hspace⟩ := rd_call_run (some m) ℓ d hd hℓ (by omega) hr
      refine ⟨2 * ℓ + 5, by omega, fun u hu => hcut u hu (.inr (false, true)),
        hspace, ?_⟩
      rw [he]
      refine ⟨rfl, ?_, rfl, rfl, rfl⟩
      exact rd_shift_move ℓ hz d.inputPos i hp _ hb m

/-- **The ruler host layout**: a configuration `c` of an `m`-tape machine on the virtual input
`y`, viewed over the physical input `z` with a header of length `ℓ`: the control `q`, the
physical input head at `ℓ + c`'s position, `c`'s work tapes and heads first, the ruler tape
`rt` with its head at `rp` last, and `c`'s output. -/
def rulerHostCfg {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (q : Option Q) (c : Cfg m Bool S y) (rt : ℤ → Option Bool)
    (rp : ℤ) : Cfg (m + 1) Bool Q z :=
  ⟨q, ⟨ℓ + c.inputPos.val, by have := c.inputPos.isLt; omega⟩, Fin.lastCases rt c.workTapes,
    Fin.lastCases rp c.workTapePos, c.output⟩

/-! The ruler host layout's fields as equations, generated by `simps` without the `simp`
attribute: `Turing.rulerHostCfg_state`, `_inputPos`, `_workTapes` (pointwise in the tape and the
cell), `_workTapePos` and `_output`. -/
attribute [simps (config := { isSimp := false }) state inputPos workTapes workTapePos output]
  rulerHostCfg

/-- Ordinary source controls and paused test controls carrying the source successor. -/
private abbrev rd_HostState (S : Type) := (S × Bool) ⊕ (S × rd_State)

/-- A test verdict resumes the source immediately, without an extra dispatch step. -/
private def rd_resume {S : Type} (q : S) : rd_State → rd_HostState S
  | .inr (b, true) => .inl (q, b)
  | s => .inr (q, s)

/-- The test's control injection, used by the public guarded state-transport theorem. -/
private def rd_pauseEmb {S : Type} (q : S) : rd_State ↪ rd_HostState S where
  toFun := rd_resume q
  inj' := by
    let decode : rd_HostState S → rd_State
      | .inl (_, b) => .inr (b, true)
      | .inr (_, s) => s
    have h : Function.LeftInverse decode (rd_resume q) := by
      intro s
      rcases s with ⟨⟩ | ⟨b, flag⟩
      · rfl
      · cases flag <;> rfl
    exact h.injective

/-- R1 selects just the last tape. -/
private def rd_slot (m : ℕ) : Fin 1 ↪ Fin (m + 1) :=
  ⟨fun _ => Fin.last m, fun _ _ _ => Subsingleton.elim _ _⟩

private theorem rd_slot_bank {m : ℕ} (i : Fin m) : i.castSucc ∉ Set.range (rd_slot m) := by
  rintro ⟨j, hj⟩
  exact Fin.castSucc_ne_last i hj.symm

/-- One source action on the bank; a live left move may start the restoring test.
A halted successor remains halted in this very action. -/
private def rd_hostAction {m : ℕ} {S : Type} (a : Action m Bool S) (b : Bool) :
    Action (m + 1) Bool (rd_HostState S) :=
  ⟨rd_direction a.inputTape b, Fin.lastCases (none, 0) a.workTapes, a.output,
    a.state.map (fun q => if b = false ∧ a.inputTape = -1 then rd_resume q (.inl ())
      else .inl (q, if a.inputTape = 1 then false else b))⟩

private def rd_hostTM {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S) :
    MultiTapeTM (m + 1) Bool (rd_HostState S) where
  q₀ := .inl (M.q₀, true)
  tr := fun s inp w => match s with
    | .inl (q, b) => rd_hostAction (M.tr q (if b then none else inp)
        (fun i => w i.castSucc)) b
    | .inr (q, r) => ((embedEmitTM (rd_slot m) rd_test).tr r inp w).mapState (rd_resume q)

/-- A paused test uses R1 and guarded state transport; the old trace supplies all frames.

Proof sketch: identify the selected one-tape start by the public reconstruction equation,
transport `rd_run` through R1, and redirect only its final verdict. -/
private theorem rd_pause_run {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S)
    {z : List Bool} (ℓ : ℕ) (d : Cfg (m + 1) Bool (rd_HostState S) z) (q : S)
    (hd : d.state = some (rd_resume q (.inl ()))) (hℓ : 0 < ℓ)
    (hp : ℓ ≤ d.inputPos.val)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      d.workTapes (Fin.last m) (d.workTapePos (Fin.last m) + p) =
        FinTM.bufferTape (List.replicate ℓ true) p) :
    (rd_hostTM M).runFrom d (2 * ℓ + 2) =
      { d with state := some (.inl (q, decide (d.inputPos.val = ℓ))) } ∧
    (∀ u ≤ 2 * ℓ + 2,
      ((rd_hostTM M).runFrom d u).state ≠ none ∧
      ((rd_hostTM M).runFrom d u).workTapes = d.workTapes ∧
      (∀ i : Fin m, ((rd_hostTM M).runFrom d u).workTapePos i.castSucc =
        d.workTapePos i.castSucc) ∧
      ((rd_hostTM M).runFrom d u).output = d.output ∧
      ((rd_hostTM M).runFrom d u).workTapePos (Fin.last m) ∈
        Finset.Icc (d.workTapePos (Fin.last m) - 1) (d.workTapePos (Fin.last m) + ℓ)) := by
  let c : Cfg 1 Bool rd_State z :=
    ⟨some (.inl ()), d.inputPos, fun _ => d.workTapes (Fin.last m),
      fun _ => d.workTapePos (Fin.last m), d.output⟩
  have hc : c.state = some (.inl ()) := rfl
  have hstart : (embedEmitCfg (rd_slot m) d.workTapes d.workTapePos [] c).mapState
      (rd_pauseEmb q) = d := by
    apply embedEmitCfg_mapState_eq
    · simp only [hc, Option.map_some]; exact hd.symm
    · rfl
    · exact List.nil_append _
    · intro j; exact ⟨rfl, rfl⟩
    · intro j _; exact ⟨rfl, rfl⟩
  have trajectory (u : ℕ) (hu : u ≤ 2 * ℓ + 2) :
      (rd_hostTM M).runFrom d u =
        (embedEmitCfg (rd_slot m) d.workTapes d.workTapePos []
          (rd_test.runFrom c u)).mapState (rd_pauseEmb q) := by
    conv_lhs => rw [← hstart]
    rw [MultiTapeTM.runFrom_mapState_of_agreeOn (embedEmitTM (rd_slot m) rd_test)
      (rd_hostTM M) (rd_pauseEmb q) (fun r => ∀ b, r ≠ .inr (b, true))]
    · rw [embedEmitTM_runFrom]
    · intro r good inp work
      rcases r with ⟨⟩ | ⟨b, flag⟩
      · rfl
      · cases flag
        · rfl
        · exact False.elim (good b rfl)
    · intro v hv r hs
      rw [embedEmitTM_runFrom, embedEmitCfg_state] at hs
      intro b hb
      subst r
      exact (rd_test_window ℓ c hc hℓ hp hr v (by omega)).2.2.2.2.2 (by omega) b hs
  constructor
  · rw [trajectory _ le_rfl, rd_test_return ℓ c hc hℓ hp hr]
    apply embedEmitCfg_mapState_eq
    · simp only [Option.map_some, rd_pauseEmb, Function.Embedding.coeFn_mk,
        rd_resume, rd_verdict, c]
    · rfl
    · exact List.nil_append _
    · intro j; exact ⟨rfl, rfl⟩
    · intro j _; exact ⟨rfl, rfl⟩
  · intro u hu
    have hw := rd_test_window ℓ c hc hℓ hp hr u hu
    rw [trajectory u hu]
    refine ⟨?_, ?_, ?_, ?_, ?_⟩
    · rw [Cfg.mapState_state, embedEmitCfg_state]
      exact fun h => hw.2.2.2.2.1 (Option.map_eq_none_iff.mp h)
    · funext j p
      rw [Cfg.mapState_workTapes]
      refine Fin.lastCases ?_ (fun i => ?_) j
      · rw [← show (rd_slot m) 0 = Fin.last m from rfl, embedEmitCfg_selected_tape, hw.1]
        rfl
      · exact congrFun (embedEmitCfg_unselected (rd_slot m) d.workTapes d.workTapePos []
          (rd_test.runFrom c u) i.castSucc (rd_slot_bank i)).1 p
    · intro i
      rw [Cfg.mapState_workTapePos]
      exact (embedEmitCfg_unselected (rd_slot m) d.workTapes d.workTapePos []
        (rd_test.runFrom c u) i.castSucc (rd_slot_bank i)).2
    · rw [Cfg.mapState_output, embedEmitCfg_output, hw.2.1, List.nil_append]
    · rw [Cfg.mapState_workTapePos, ← show (rd_slot m) 0 = Fin.last m from rfl,
        embedEmitCfg_selected_pos, Finset.mem_Icc]
      exact ⟨hw.2.2.1, hw.2.2.2.1⟩

/-- The source branch performs the source action on the bank and output in one step.

Proof sketch: read the virtual cell through its tag, then use the public action-field
and clamped-move equations. The last tape has a stationary, unwritten action. -/
private theorem rd_host_source_step {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S)
    {y z : List Bool} (ℓ : ℕ) (hz : z.length = ℓ + y.length) (hdrop : z.drop ℓ = y)
    (c : Cfg m Bool S y) (b : Bool) (hb : b = true ↔ c.inputPos.val = 0)
    (rt : ℤ → Option Bool) (rp : ℤ) (q : S) (hq : c.state = some q) :
    let a := M.tr q c.inputSymbol c.workTapeSymbols
    (rd_hostTM M).step (rulerHostCfg ℓ hz (some (.inl (q, b))) c rt rp) =
      rulerHostCfg ℓ hz (rd_hostAction a b).state (M.step c) rt rp := by
  let d : Cfg (m + 1) Bool (rd_HostState S) z :=
    rulerHostCfg ℓ hz (some (.inl (q, b))) c rt rp
  let a := M.tr q c.inputSymbol c.workTapeSymbols
  have hs : d.state = some (.inl (q, b)) := rfl
  have hread : (if b then none else d.inputSymbol) = c.inputSymbol := by
    rw [Cfg.inputSymbol_eq_inputCell c]
    exact rd_tagged_read ℓ hdrop d c.inputPos rfl b hb
  have hbank : (fun i : Fin m => d.workTapeSymbols i.castSucc) = c.workTapeSymbols := by
    funext i
    rw [Cfg.workTapeSymbols_eq, Cfg.workTapeSymbols_eq]
    simp only [d, rulerHostCfg, Fin.lastCases_castSucc]
  change (rd_hostTM M).step d = _
  rw [MultiTapeTM.step_of_some hs]
  change (rd_hostAction (M.tr q (if b then none else d.inputSymbol)
    (fun i => d.workTapeSymbols i.castSucc)) b).apply d = _
  rw [hread, hbank, MultiTapeTM.step_of_some hq]
  change (rd_hostAction a b).apply d = rulerHostCfg ℓ hz (rd_hostAction a b).state
    (a.apply c) rt rp
  apply Cfg.ext
  · rw [Action.apply_state]; rfl
  · apply Fin.ext
    rw [Action.apply_inputPos]
    change (moveInputPos d.inputPos (rd_direction a.inputTape b)).val =
      ℓ + (a.apply c).inputPos.val
    rw [Action.apply_inputPos]
    exact rd_shift_move ℓ hz d.inputPos c.inputPos rfl b hb a.inputTape
  · funext j
    refine Fin.lastCases ?_ (fun i => ?_) j
    · rw [Action.apply_workTapes]
      simp only [rd_hostAction, Fin.lastCases_last, Option.getD_none,
        Cfg.workTapeSymbols_eq, d, rulerHostCfg, Fin.lastCases_last, Function.update_eq_self]
    · rw [Action.apply_workTapes]
      simp only [rd_hostAction, Fin.lastCases_castSucc, d, rulerHostCfg,
        Fin.lastCases_castSucc, Cfg.workTapeSymbols_eq]
      rw [Action.apply_workTapes, Cfg.workTapeSymbols_eq]
  · funext j
    refine Fin.lastCases ?_ (fun i => ?_) j
    · simp only [Action.apply_workTapePos, rd_hostAction, d, rulerHostCfg,
        Fin.lastCases_last, SignType.coe_zero, add_zero]
    · simp only [Action.apply_workTapePos, rd_hostAction, d, rulerHostCfg,
        Fin.lastCases_castSucc]
  · simp only [Action.apply_output, rd_hostAction, d, rulerHostCfg]

/-- Away from a live left-move pause, the next boundary tag is known locally.

Proof sketch: the public move equations settle the three directions and the right clamp. -/
private theorem rd_next_tag {n : ℕ} (i : Fin (n + 2)) (b : Bool)
    (hb : b = true ↔ i.val = 0) (m : SignType) (hno : ¬(b = false ∧ m = -1)) :
    (if m = 1 then false else b) = decide ((moveInputPos i m).val = 0) := by
  cases m with
  | zero =>
    simp only [SignType.zero_eq_zero, reduceCtorEq, if_false, moveInputPos_zero]
    apply Bool.eq_iff_iff.mpr
    simpa only [decide_eq_true_eq] using hb
  | neg =>
    have hbt : b = true := by cases b <;> simp_all
    have hi := hb.mp hbt
    rw [FinTM.moveInputPos_neg_val]
    simp only [SignType.neg_eq_neg_one, reduceCtorEq, if_false, hbt, hi,
      Nat.zero_sub, decide_true]
  | pos =>
    have hi : (moveInputPos i .pos).val ≠ 0 := by
      by_cases hr : i.val = n + 1
      · have he : i = ⟨n + 1, by omega⟩ := Fin.ext hr
        rw [he, SignType.pos_eq_one, moveInputPos_rightBoundary]
        simp only; omega
      · rw [moveInputPos_pos_of_ne_right i hr]
        simp only; omega
    simp only [SignType.pos_eq_one, if_true]
    exact (decide_eq_false hi).symm

private def rd_hostEnc {m : ℕ} {S : Type} {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (rt : ℤ → Option Bool) (rp : ℤ)
    (c : Cfg m Bool S y) : Cfg (m + 1) Bool (rd_HostState S) z :=
  rulerHostCfg ℓ hz (c.state.map (fun q => .inl (q, decide (c.inputPos.val = 0)))) c rt rp

/-- The live left-move test is the only nonconstant part of a source step's cost. -/
private def rd_hostCost {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S) (ℓ : ℕ)
    {y : List Bool} (c : Cfg m Bool S y) : ℕ :=
  c.state.elim 1 (fun q =>
    let a := M.tr q c.inputSymbol c.workTapeSymbols
    if a.state.isSome = true ∧ decide (c.inputPos.val = 0) = false ∧ a.inputTape = -1
    then 2 * ℓ + 3 else 1)

private theorem rd_hostCost_bounds {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S)
    (ℓ : ℕ) {y : List Bool} (c : Cfg m Bool S y) :
    1 ≤ rd_hostCost M ℓ c ∧ rd_hostCost M ℓ c ≤ 3 * (ℓ + 1) := by
  unfold rd_hostCost
  cases c.state with
  | none => simp only [Option.elim_none]; omega
  | some q => simp only [Option.elim_some]; split_ifs <;> omega

/-- The bank and output are those of one source configuration; the ruler stays delimited. -/
private def rd_hostFrame {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ)
    (rt : ℤ → Option Bool) (rp : ℤ) (c : Cfg m Bool S y) (d : Cfg (m + 1) Bool Q z) : Prop :=
  (∀ i : Fin m, d.workTapes i.castSucc = c.workTapes i ∧
    d.workTapePos i.castSucc = c.workTapePos i) ∧
  d.output = c.output ∧ d.workTapes (Fin.last m) = rt ∧
  d.workTapePos (Fin.last m) ∈ Finset.Icc (rp - 1) (rp + ℓ)

private theorem rd_hostFrame_layout {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (q : Option Q) (c : Cfg m Bool S y)
    (rt : ℤ → Option Bool) (rp : ℤ) :
    rd_hostFrame ℓ rt rp c (rulerHostCfg ℓ hz q c rt rp) := by
  simp only [rd_hostFrame, rulerHostCfg, Fin.lastCases_castSucc, Fin.lastCases_last,
    true_and, Finset.mem_Icc]
  exact ⟨fun _ => trivial, by omega, by omega⟩

/-- One complete source step, with its cut and its two possible source frames.

Proof sketch: perform the source action once; a live left move then uses `rd_pause_run`.
Every other successor, including a halt, returns immediately. -/
private theorem rd_host_block {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S)
    {y z : List Bool} (ℓ : ℕ) (hz : z.length = ℓ + y.length) (hℓ : 0 < ℓ)
    (hdrop : z.drop ℓ = y) (rt : ℤ → Option Bool) (rp : ℤ)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      rt (rp + p) = FinTM.bufferTape (List.replicate ℓ true) p) (c : Cfg m Bool S y) :
    (rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) (rd_hostCost M ℓ c) =
      rd_hostEnc ℓ hz rt rp (M.step c) ∧
    (c.state ≠ none → ∀ u < rd_hostCost M ℓ c,
      ((rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) u).state ≠ none) ∧
    (∀ u ≤ rd_hostCost M ℓ c,
      rd_hostFrame ℓ rt rp c ((rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) u) ∨
      rd_hostFrame ℓ rt rp (M.step c)
        ((rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) u)) := by
  cases hq : c.state with
  | none =>
    have he : (rd_hostEnc ℓ hz rt rp c).state = none := by
      simp only [rd_hostEnc, rulerHostCfg, hq, Option.map_none]
    have fixed (u : ℕ) := MultiTapeTM.runFrom_of_halt (tm := rd_hostTM M)
      (rd_hostEnc ℓ hz rt rp c) he (n := u)
    refine ⟨?_, fun h => False.elim (h rfl), ?_⟩
    · rw [fixed, MultiTapeTM.step_of_halt hq]
    · intro u _
      rw [fixed]
      exact Or.inl (rd_hostFrame_layout ℓ hz _ c rt rp)
  | some q =>
    let b := decide (c.inputPos.val = 0)

/-- Locate every host time in a source step through the public lockstep API.

Proof sketch: for a nonzero horizon, locate within the penultimate prefix or split off
its last block. Thus a frame after the located source step is still within the horizon. -/
private theorem rd_host_frames {m : ℕ} {S : Type} (M : MultiTapeTM m Bool S)
    {y z : List Bool} (ℓ : ℕ) (hz : z.length = ℓ + y.length) (hℓ : 0 < ℓ)
    (hdrop : z.drop ℓ = y) (rt : ℤ → Option Bool) (rp : ℤ)
    (hr : ∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
      rt (rp + p) = FinTM.bufferTape (List.replicate ℓ true) p)
    (L : Lockstep M (rd_hostTM M) (rd_hostEnc ℓ hz rt rp) (rd_hostCost M ℓ) ∅ ∅)
    (c : Cfg m Bool S y) (t : ℕ) :
    ∀ u ≤ ∑ v ∈ Finset.range t, rd_hostCost M ℓ (M.runFrom c v),
      ∃ v ≤ t, rd_hostFrame ℓ rt rp (M.runFrom c v)
        ((rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) u) := by
  cases t with
  | zero =>
    intro u hu
    simp only [Finset.range_zero, Finset.sum_empty, Nat.le_zero] at hu
    rw [hu, MultiTapeTM.runFrom_zero]
    refine ⟨0, le_rfl, ?_⟩
    rw [MultiTapeTM.runFrom_zero]
    exact rd_hostFrame_layout ℓ hz _ c rt rp
  | succ t =>
    intro u hu
    have finish (v : ℕ) (hv : v ≤ t) (w : ℕ)
        (hw : w ≤ rd_hostCost M ℓ (M.runFrom c v))
        (he : (rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) u =
          (rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp (M.runFrom c v)) w) :
        ∃ r ≤ t + 1, rd_hostFrame ℓ rt rp (M.runFrom c r)
          ((rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c) u) := by
      rw [he]
      rcases (rd_host_block M ℓ hz hℓ hdrop rt rp hr (M.runFrom c v)).2.2 w hw with h | h
      · exact ⟨v, by omega, h⟩
      · refine ⟨v + 1, by omega, ?_⟩
        rw [MultiTapeTM.runFrom_succ_eq_step']
        exact h
    by_cases hp : u ≤ ∑ v ∈ Finset.range t, rd_hostCost M ℓ (M.runFrom c v)
    · obtain ⟨v, hv, w, hw, he⟩ := L.locate c t (by simp) u hp
      apply finish v hv w _ he
      rcases hw with rfl | ⟨_, hw⟩
      · omega
      · exact hw
    · let a := ∑ v ∈ Finset.range t, rd_hostCost M ℓ (M.runFrom c v)
      have ha : u = a + (u - a) := by dsimp only [a]; omega
      apply finish t le_rfl (u - a)
      · rw [Finset.sum_range_succ] at hu
        dsimp only [a]; omega
      · conv_lhs => rw [ha, MultiTapeTM.runFrom_add, L.runFrom c t (by simp)]

/-- **H5, the ruler host**: a positive, finite block simulation of any native-input machine on
a suffix input (the H2-S gate's finding 2). For any binary machine `M` on `m` tapes there is a
host on one more tape (the ruler, last), with a state map `inj` of the source states paired
with a boundary tag, as follows. Present a source configuration `c` on the virtual input `y`
inside `z = header ++ y` (header length `ℓ > 0`) through `Turing.rulerHostCfg`, with the ruler
`true^ℓ` at the ruler head and the tag `b` true exactly at the virtual left boundary. Then for
every `t`:
* **block endpoint**: after some `t'` host steps with `t ≤ t' ≤ C · (ℓ + 1) · t` the host
  presents the source's configuration after `t` steps, with a correct tag;
* **exit cut**: if the source is live before `t`, the host is live before `t'`;
* **intermediate frames**: at every host time up to `t'`, the bank tapes, the bank heads and
  the output are the source's at some source time no later than `t`, the ruler tape is
  unchanged, and the ruler head is on the ruler or one of its two delimiters;
* **space ledger**: the bank heads visit exactly the source's cells, and the ruler head
  `ℓ + 2` cells.

So the host halts exactly when the source does (the cut and `t ≤ t'` give the converse), with
the source's output, using the source's space plus `ℓ + 2` cells, within `C · (ℓ + 1)` times
the source's time. Literal lockstep (`t' = t`) is the native special case, and Z1's
`Turing.vhostEmitTM`. This is how H2's step block, whose input is native, runs on a suffix
input unchanged. [AB09, §1.2; Exercise 4.1]

**Proof sketch.** Each host step performs the source transition, reading the native symbol, or
blank when the tag is set, and suppressing a left move when the tag is set. A right move
clears the tag; after a left move from a cleared tag the host pauses for the boundary test of
`Turing.exists_rulerTestTM` (on the ruler tape, by R1) and sets the tag from its verdict. A
halting source transition halts the host in the same step, with no pause. The pause costs
`C · (ℓ + 1)` steps, moves no bank head, writes nothing and emits nothing, which gives the
frames and the cut. Induct on `t` as `Turing.vhostEmitTM_runFrom` does; the visited-set clauses
follow `Turing.vhostEmitTM_visitedByTapeHead_bank` and
`Turing.vhostEmitTM_visitedByTapeHead_buffer`.

**Implementation.** The host is a `Turing.Lockstep` over `Turing.rulerHostCfg` with the tag
`c.inputPos = 0`. A block costs one step, or `2ℓ + 3` after a live left move from a cleared tag:
the test runs on the last tape through `Turing.embedEmitTM` and the guarded state transport, and
its verdict resumes the source directly. Hence `C = 3`. The endpoint, the cut and the frames come
from `Turing.Lockstep.runFrom`, `.cut` and `.locate`, and the bank's visited sets are equal by
`Turing.MultiTapeTM.mem_visitedByTapeHead_iff`. -/
theorem exists_rulerHostTM {m : ℕ} {S : Type} [Fintype S] [DecidableEq S]
    (M : MultiTapeTM m Bool S) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (H : MultiTapeTM (m + 1) Bool Q)
      (inj : S → Bool → Q) (C : ℕ), 0 < C ∧
      ∀ {y z : List Bool} (ℓ : ℕ) (hz : z.length = ℓ + y.length), 0 < ℓ → z.drop ℓ = y →
      ∀ (rt : ℤ → Option Bool) (rp : ℤ),
        (∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
          rt (rp + p) = FinTM.bufferTape (List.replicate ℓ true) p) →
      ∀ (c : Cfg m Bool S y) (b : Bool), (b = true ↔ c.inputPos.val = 0) → ∀ t : ℕ,
        let d := rulerHostCfg ℓ hz (c.state.map fun q => inj q b) c rt rp
        ∃ t', t ≤ t' ∧ t' ≤ C * (ℓ + 1) * t ∧
          (∃ b' : Bool, (b' = true ↔ (M.runFrom c t).inputPos.val = 0) ∧
            H.runFrom d t' =
              rulerHostCfg ℓ hz ((M.runFrom c t).state.map fun q => inj q b')
                (M.runFrom c t) rt rp) ∧
          ((∀ u < t, (M.runFrom c u).state ≠ none) → ∀ u < t', (H.runFrom d u).state ≠ none) ∧
          (∀ u ≤ t', ∃ v ≤ t,
            (∀ i : Fin m, (H.runFrom d u).workTapes i.castSucc = (M.runFrom c v).workTapes i ∧
              (H.runFrom d u).workTapePos i.castSucc = (M.runFrom c v).workTapePos i) ∧
            (H.runFrom d u).output = (M.runFrom c v).output ∧
            (H.runFrom d u).workTapes (Fin.last m) = rt ∧
            (H.runFrom d u).workTapePos (Fin.last m) ∈ Finset.Icc (rp - 1) (rp + ℓ)) ∧
          (∀ i : Fin m, H.visitedByTapeHead d t' i.castSucc = M.visitedByTapeHead c t i) ∧
          H.visitedByTapeHead d t' (Fin.last m) ⊆ Finset.Icc (rp - 1) (rp + ℓ) := by
  refine ⟨rd_HostState S, inferInstance, inferInstance, rd_hostTM M,
    (fun q b => .inl (q, b)), 3, by omega, ?_⟩
  intro y z ℓ hz hℓ hdrop rt rp hr c b hb t
  have hb' : b = decide (c.inputPos.val = 0) := by
    rw [Bool.eq_iff_iff, decide_eq_true_eq]
    exact hb
  subst b
  have blocks := rd_host_block M ℓ hz hℓ hdrop rt rp hr
  let L : Lockstep M (rd_hostTM M) (rd_hostEnc ℓ hz rt rp) (rd_hostCost M ℓ) ∅ ∅ :=
    ⟨fun c _ => (blocks c).1, by simp, by simp⟩
  let live : Lockstep M (rd_hostTM M) (rd_hostEnc ℓ hz rt rp) (rd_hostCost M ℓ)
      {none} {none} := {
    step := fun c _ => (blocks c).1
    exit := by
      intro c hc
      simp only [Set.mem_singleton_iff] at hc ⊢
      simp only [rd_hostEnc, rulerHostCfg, hc, Option.map_none]
    inner := by
      intro c hc u hu
      simp only [Set.mem_singleton_iff] at hc ⊢
      exact (blocks c).2.1 hc u hu }
  let time := fun n => ∑ v ∈ Finset.range n, rd_hostCost M ℓ (M.runFrom c v)
  have bounds : t ≤ time t ∧ time t ≤ 3 * (ℓ + 1) * t := by
    constructor
    · calc
        t = ∑ _v ∈ Finset.range t, (1 : ℕ) := by simp
        _ ≤ time t := Finset.sum_le_sum (fun v _ => (rd_hostCost_bounds M ℓ _).1)
    · calc
        time t ≤ ∑ _v ∈ Finset.range t, 3 * (ℓ + 1) :=
          Finset.sum_le_sum (fun v _ => (rd_hostCost_bounds M ℓ _).2)
        _ = 3 * (ℓ + 1) * t := by simp [Nat.mul_comm]
  have frames := rd_host_frames M ℓ hz hℓ hdrop rt rp hr L c t
  refine ⟨time t, bounds.1, bounds.2,
    ⟨decide ((M.runFrom c t).inputPos.val = 0), by simp only [decide_eq_true_eq], ?_⟩,
    ?_, ?_, ?_, ?_⟩
  · exact L.runFrom c t (by simp)
  · intro ht
    exact live.cut c t (by simpa only [Set.mem_singleton_iff] using ht)
  · exact frames
  · intro i
    apply Finset.ext
    intro p
    rw [MultiTapeTM.mem_visitedByTapeHead_iff, MultiTapeTM.mem_visitedByTapeHead_iff]
    constructor
    · rintro ⟨u, hu, hp⟩
      obtain ⟨v, hv, hf⟩ := frames u hu
      exact ⟨v, hv, (hf.1 i).2.symm.trans hp⟩
    · rintro ⟨v, hv, hp⟩
      refine ⟨time v, Finset.sum_le_sum_of_subset (Finset.range_mono hv), ?_⟩
      change ((rd_hostTM M).runFrom (rd_hostEnc ℓ hz rt rp c)
        (∑ u ∈ Finset.range v, rd_hostCost M ℓ (M.runFrom c u))).workTapePos i.castSucc = p
      rw [L.runFrom c v (by simp)]
      simpa only [rd_hostEnc, rulerHostCfg, Fin.lastCases_castSucc] using hp
  · intro p hp
    obtain ⟨u, hu, he⟩ := (MultiTapeTM.mem_visitedByTapeHead_iff _ _ _ _).mp hp
    obtain ⟨v, _, hf⟩ := frames u hu
    rw [← he]
    exact hf.2.2.2

end Turing

    (M : MultiTapeTM k Bool (Fin (n + 1))) (c : Cfg k Bool (Fin (n + 1)) x) :
    ((StepBlock.forwardLoopCfg entry M c).state = none ↔ c.state = none) ∧
      (StepBlock.forwardLoopCfg entry M c).output = c.output := by
  rw [StepBlock.forwardLoopCfg_state, StepBlock.forwardLoopCfg_output, Option.map_eq_none_iff]
  exact ⟨Iff.rfl, rfl⟩

theorem r_ruler_fields {m : ℕ} {S Q : Type} {y z : List Bool} (ℓ : ℕ)
    (hz : z.length = ℓ + y.length) (q : Option Q) (c : Cfg m Bool S y) (rt : ℤ → Option Bool)
    (rp : ℤ) :
    (rulerHostCfg ℓ hz q c rt rp).inputPos.val = ℓ + c.inputPos.val ∧
      (rulerHostCfg ℓ hz q c rt rp).workTapes (Fin.last m) = rt ∧
      (rulerHostCfg ℓ hz q c rt rp).workTapePos (Fin.last m) = rp ∧
      (∀ i : Fin m, (rulerHostCfg ℓ hz q c rt rp).workTapes i.castSucc = c.workTapes i) ∧
      (rulerHostCfg ℓ hz q c rt rp).output = c.output := by
  refine ⟨by rw [rulerHostCfg_inputPos], ?_, ?_, fun i => ?_, rulerHostCfg_output _ _ _ _ _ _⟩
  · funext p; rw [rulerHostCfg_workTapes, Fin.lastCases_last]
  · rw [rulerHostCfg_workTapePos, Fin.lastCases_last]
  · funext p; rw [rulerHostCfg_workTapes, Fin.lastCases_castSucc]

/-! ### Beyond the draft: the full seam configurations of both phases (sealed)

Phase 1: the startup's endpoint, relabelled to H2-F's entry, is the R1 embedding of the encoding
of the initial configuration into blank frames. -/

theorem r_seam_phase1 {k K n : ℕ} {P Q : Type} (entry : Q) (exit : P)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (z : List Bool)
    (ι : Fin (k + 2) ↪ Fin (K + (k + 2))) (hι : ∀ j, (ι j).val = j.val)
    (D : Cfg (K + (k + 2)) Bool P z)
    (hD : D = ⟨some exit, 1,
      fun j => if j.val = k then FinTM.bufferTape (tableBitsK M.tr)
        else if j.val = k + 1 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
        else fun _ => none,
      fun _ => 0, []⟩) :
    embedEmitCfg ι (fun _ _ => none) (fun _ => 0) []
        (StepBlock.forwardLoopCfg entry M (M.initCfg z)) = D.mapState fun _ => entry := by
  obtain ⟨hs, hin, hh, ho⟩ := r_encode_init_fields entry M z
  subst hD
  apply embedEmitCfg_eq
  · rw [hs, Cfg.mapState_state]; rfl
  · rw [Cfg.mapState_inputPos]; exact Fin.ext hin
  · rw [ho, Cfg.mapState_output, List.nil_append]
  · intro i
    refine ⟨?_, ?_⟩
    · funext p
      rw [r_encode_init_tapes, Cfg.mapState_workTapes]
      show _ = (if (ι i).val = k then (_ : ℤ → Option Bool) else _) p
      rw [hι i]
    · rw [hh i, Cfg.mapState_workTapePos]
  · intro j hj
    have hge : k + 2 ≤ j.val := by
      by_contra hlt
      exact hj ⟨⟨j.val, by omega⟩, Fin.ext (by rw [hι])⟩
    refine ⟨?_, ?_⟩
    · funext p
      rw [Cfg.mapState_workTapes]
      show _ = (if j.val = k then (_ : ℤ → Option Bool) else _) p
      rw [if_neg (by omega), if_neg (by omega)]
    · rw [Cfg.mapState_workTapePos]

/-! Phase 2: after the startup, the header ruler and its rewind, the configuration relabelled to
the host's entry `inj entry false` is the R1 embedding of the ruler host's presentation of the
encoding of the initial configuration on the suffix `x`. -/

theorem r_seam_phase2 {k N n : ℕ} {P Q H : Type} (entry : Q) (inj : Q → Bool → H)
    (M : MultiTapeTM k Bool (Fin (n + 1))) (x z : List Bool) (ℓ : ℕ)
    (hz : z.length = ℓ + x.length)
    (σ : Fin ((k + 2) + 1) ↪ Fin N) (hσ : ∀ j : Fin (k + 2), (σ j.castSucc).val = j.val)
    (r : Fin N) (hr : σ (Fin.last (k + 2)) = r) (hrk : k + 2 ≤ r.val)
    (D : Cfg N Bool P z) (e : P) (hst : D.state = some e) (hin : D.inputPos.val = ℓ + 1)
    (htapes : D.workTapes = fun j => if j.val = k then FinTM.bufferTape (tableBitsK M.tr)
        else if j.val = k + 1 then fun p => FinTM.bufferTape (unaryFin M.q₀) (p - 2)
        else if j = r then FinTM.bufferTape (List.replicate ℓ true)
        else fun _ => none)
    (hheads : D.workTapePos = fun _ => 0) (hout : D.output = []) :
    embedEmitCfg σ (fun _ _ => none) (fun _ => 0) []
        (rulerHostCfg ℓ hz
          ((StepBlock.forwardLoopCfg entry M (M.initCfg x)).state.map fun q => inj q false)
          (StepBlock.forwardLoopCfg entry M (M.initCfg x))
          (FinTM.bufferTape (List.replicate ℓ true)) 0) =
      D.mapState fun _ => inj entry false := by
  obtain ⟨hs, hin', hh, ho⟩ := r_encode_init_fields entry M x
  apply embedEmitCfg_eq
  · rw [rulerHostCfg_state, hs, Option.map_some, Cfg.mapState_state, hst, Option.map_some]
  · rw [Cfg.mapState_inputPos]
    apply Fin.ext
    rw [hin, rulerHostCfg_inputPos]
    show ℓ + _ = ℓ + 1
    rw [hin']
  · rw [rulerHostCfg_output, ho, Cfg.mapState_output, hout, List.nil_append]
  · intro i
    refine Fin.lastCases ?_ (fun j => ?_) i
    · refine ⟨?_, ?_⟩
      · funext p
        rw [rulerHostCfg_workTapes, Fin.lastCases_last, Cfg.mapState_workTapes, htapes, hr]
        show _ = (if r.val = k then (_ : ℤ → Option Bool) else _) p
        rw [if_neg (by omega), if_neg (by omega), if_pos rfl]
      · rw [rulerHostCfg_workTapePos, Fin.lastCases_last, Cfg.mapState_workTapePos, hheads]
    · refine ⟨?_, ?_⟩
      · funext p
        rw [rulerHostCfg_workTapes, Fin.lastCases_castSucc, Cfg.mapState_workTapes, htapes,
          r_encode_init_tapes]
        have hne : σ j.castSucc ≠ r := by
          rw [← hr]; exact fun h => (Fin.castSucc_ne_last j) (σ.injective h)
        show (if j.val = k then (_ : ℤ → Option Bool) else _) p =
          (if (σ j.castSucc).val = k then (_ : ℤ → Option Bool) else _) p
        rw [hσ j]
        by_cases h1 : j.val = k
        · rw [if_pos h1, if_pos h1]
        · rw [if_neg h1, if_neg h1]
          by_cases h2 : j.val = k + 1
          · rw [if_pos h2, if_pos h2]
          · rw [if_neg h2, if_neg h2, if_neg hne]
      · rw [rulerHostCfg_workTapePos, Fin.lastCases_castSucc, hh, Cfg.mapState_workTapePos,
          hheads]
  · intro j hj
    have hge : k + 2 ≤ j.val := by
      by_contra hlt
      exact hj ⟨(⟨j.val, by omega⟩ : Fin (k + 2)).castSucc, Fin.ext (by rw [hσ])⟩
    have hjr : j ≠ r := fun h => hj ⟨Fin.last (k + 2), hr.trans h.symm⟩
    refine ⟨?_, ?_⟩
    · funext p
      rw [Cfg.mapState_workTapes, htapes]
      show _ = (if j.val = k then (_ : ℤ → Option Bool) else _) p
      rw [if_neg (by omega), if_neg (by omega), if_neg hjr]
    · rw [Cfg.mapState_workTapePos, hheads]

end U2Route
