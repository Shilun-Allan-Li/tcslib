-- Part 4 (the joint seal of `evidence/WordPatchSeal.lean.txt`): the proof elaborates; both
-- controls fail by `rfl` and close by the published lemmas; then both bare (expected to FAIL).
section WordPatchSeal
attribute [local irreducible] Turing.FinTM.bufferTape Turing.wordPatch

theorem wordPatch_append_sealed {Γ : Type*} (ct : ℤ → Option Γ) (cp : ℤ) (u v : List Γ) :
    wordPatch (wordPatch ct cp u) (cp + u.length) v = wordPatch ct cp (u ++ v) := by
  funext q
  have hlen : ((u ++ v).length : ℤ) = u.length + v.length := by simp
  by_cases h₁ : cp ≤ q ∧ q < cp + (u.length : ℤ)
  · have hn : 0 ≤ q - cp := by omega
    rw [wordPatch_of_not_mem _ _ _ (by omega), wordPatch_of_mem _ _ _ h₁.1 h₁.2,
      wordPatch_of_mem _ _ _ h₁.1 (by omega), ← Int.toNat_of_nonneg hn, FinTM.bufferTape_nat,
      FinTM.bufferTape_nat, List.getElem?_append_left (by omega)]
  · by_cases h₂ : cp + (u.length : ℤ) ≤ q ∧ q < cp + u.length + (v.length : ℤ)
    · have hn : 0 ≤ q - (cp + u.length) := by omega
      have hm : 0 ≤ q - cp := by omega
      rw [wordPatch_of_mem _ _ _ h₂.1 h₂.2, wordPatch_of_mem _ _ _ (by omega) (by omega),
        ← Int.toNat_of_nonneg hn, ← Int.toNat_of_nonneg hm, FinTM.bufferTape_nat,
        FinTM.bufferTape_nat, List.getElem?_append_right (by omega)]
      congr 1
      omega
    · rw [wordPatch_of_not_mem _ _ _ h₂, wordPatch_of_not_mem _ _ _ h₁,
        wordPatch_of_not_mem _ _ _ (by omega)]

example (u : List Bool) : FinTM.bufferTape u (-1) = none := by

open Lean Meta Elab Command in
#eval show CommandElabM Unit from do
  let env ← getEnv
  let root := ``Turing.wordPatch_append
  let valOf : ConstantInfo → Option Expr := fun ci => match ci with
    | .thmInfo v => some v.value
    | .defnInfo v => some v.value
    | _ => none
  let kindOf : Name → String := fun c => match env.find? c with
    | some (.defnInfo _) => "def"
    | some (.thmInfo _) => "theorem"
    | some (.opaqueInfo _) => "opaque"
    | some (.inductInfo _) => "inductive"
    | some (.ctorInfo _) => "constructor"
    | some _ => "other"
    | none => "?"
  let tcslibMod? : Name → Option Name := fun c => match env.getModuleIdxFor? c with
    | some idx =>
      let m := env.header.moduleNames[idx.toNat]!
      if (`TCSlib).isPrefixOf m then some m else none
    | none => none
  let mut todo : Array Name := #[root]
  let mut seen : NameSet := {}
  let mut direct : Array Name := #[]
  while !todo.isEmpty do
    let n := todo.back!
    todo := todo.pop
    if seen.contains n then continue
    seen := seen.insert n
    let some ci := env.find? n | continue
    let cs := ci.type.getUsedConstants ++ ((valOf ci).map Expr.getUsedConstants).getD #[]
    for c in cs do
      if root.isPrefixOf c && c != root then
        todo := todo.push c
      else if (tcslibMod? c).isSome && !direct.contains c then
        direct := direct.push c
  for c in direct do
    logInfo m!"G56 walk: {c} ({kindOf c}, {(tcslibMod? c).getD .anonymous})"
  for c in direct do
    if kindOf c == "theorem" then
      let some ci := env.find? c | continue
      let ds := ci.type.getUsedConstants.filter fun d =>
        (tcslibMod? d).isSome && kindOf d == "def" && !(env.isProjectionFn d)
      logInfo m!"G56 level 2: the statement of {c} names the TCSlib definitions {ds.toList.eraseDups}"
  logInfo m!"G56 walk: {direct.size} TCSlib constants referenced; {seen.size} declarations visited"

end Turing
