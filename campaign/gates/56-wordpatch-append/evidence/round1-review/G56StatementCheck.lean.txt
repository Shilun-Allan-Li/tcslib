section G56Seal
attribute [local irreducible] Turing.FinTM.bufferTape Turing.wordPatch

/-- The constant fill (the shape of the copy endpoint's write) is a patch of a constant word. -/
example {Γ : Type*} (ct : ℤ → Option Γ) (cp : ℤ) (a : ℕ) (e : Γ) :
    (fun q => if cp ≤ q ∧ q < cp + (a : ℤ) then some e else ct q) =
      wordPatch ct cp (List.replicate a e) := by
  funext q
  by_cases h : cp ≤ q ∧ q < cp + (a : ℤ)
  · have hn : 0 ≤ q - cp := by omega
    rw [if_pos h, wordPatch_of_mem _ _ _ h.1 (by simpa using h.2), ← Int.toNat_of_nonneg hn,
      FinTM.bufferTape_nat, List.getElem?_replicate_of_lt (by omega)]
  · rw [if_neg h, wordPatch_of_not_mem _ _ _ (by simpa using h)]

/-- Two adjacent constant patches join into one, by `wordPatch_append`. -/
example {Γ : Type*} (ct : ℤ → Option Γ) (cp : ℤ) (a b : ℕ) (e : Γ) :
    wordPatch (wordPatch ct cp (List.replicate a e)) (cp + a) (List.replicate b e) =
      wordPatch ct cp (List.replicate (a + b) e) := by
  have h := wordPatch_append ct cp (List.replicate a e) (List.replicate b e)
  rwa [List.length_replicate, List.replicate_append_replicate] at h

end G56Seal
