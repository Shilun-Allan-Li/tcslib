/-- **2. U2-S's use**: for the code `α`'s captured serialization (head one past it), one time `T`
serves every suffix `w` of the native input `pairEncode α w`, and the end is U2-S's tapes `k` and
`k + 1` exactly. -/
theorem seamLayout_u2s {kk K : ℕ} (E : EffectiveCodeK kk) (src tbl sm : Fin K) (hst : src ≠ tbl)
    (hss : src ≠ sm) (hts : tbl ≠ sm) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM K Bool Q)
      (entry done : Q),
      ∀ α : List Bool, ∃ T : ℕ, ∀ (w : List Bool) (d : Cfg K Bool Q (pairEncode α w)),
        d.state = some entry →
        d.workTapes src = FinTM.bufferTape (serializeK (E.machine α)) →
        d.workTapePos src = ((serializeK (E.machine α)).length : ℤ) →
        d.workTapes tbl = (fun _ => none) → d.workTapePos tbl = 0 →
        d.workTapes sm = (fun _ => none) → d.workTapePos sm = 0 →
        (L.runFrom d T).state = some done ∧
        (∀ u < T, (L.runFrom d u).state ≠ some done) ∧
        (∀ u ≤ T, (L.runFrom d u).output = d.output) ∧
        (L.runFrom d T).workTapes tbl = FinTM.bufferTape (tableBitsK (E.machine α).tr) ∧
        (L.runFrom d T).workTapes sm =
          (fun p => FinTM.bufferTape (unaryFin (E.machine α).q₀) (p - 2)) ∧
        (L.runFrom d T).workTapes src = (fun _ => none) ∧
        (L.runFrom d T).workTapePos src = 0 ∧ (L.runFrom d T).workTapePos tbl = 0 ∧
        (L.runFrom d T).workTapePos sm = 0 ∧ (L.runFrom d T).inputPos = d.inputPos := by
  obtain ⟨Q, hQ, hQ', L, entry, done, C, h⟩ := exists_seamLayoutTM K src tbl sm hst hss hts
  refine ⟨Q, hQ, hQ', L, entry, done, fun α => ?_⟩
  have hs : serializeK (E.machine α) =
      pairEncode (E.numStates α).bits
        ((List.replicate ((E.machine α).q₀ : ℕ) true ++ [false]) ++ tableBitsK (E.machine α).tr) := by
    rw [serializeK_eq, unaryFin_eq]
  obtain ⟨T, -, hrun⟩ := h (E.numStates α).bits (tableBitsK (E.machine α).tr) (E.machine α).q₀
    (serializeK (E.machine α)).length (by rw [hs])
  refine ⟨T, fun w d hstate htape hpos htbl htblp hsm hsmp => ?_⟩
  obtain ⟨hdone, hin, htapes, hheads, hout, hfirst⟩ :=
    hrun d hstate (by rw [htape, hs]) (by rw [hpos, hs]) htbl htblp hsm hsmp
  refine ⟨hdone, hfirst, hout, ?_, ?_, ?_, ?_, ?_, ?_, hin⟩
  · rw [htapes, Function.update_of_ne hts, Function.update_self]
  · rw [htapes, Function.update_self, unaryFin_eq]
  · rw [htapes, Function.update_of_ne hss, Function.update_of_ne hst, Function.update_self]
  · rw [hheads, Function.update_of_ne hss, Function.update_of_ne hst, Function.update_self]
  · rw [hheads, Function.update_of_ne hts, Function.update_self]
  · rw [hheads, Function.update_self]
