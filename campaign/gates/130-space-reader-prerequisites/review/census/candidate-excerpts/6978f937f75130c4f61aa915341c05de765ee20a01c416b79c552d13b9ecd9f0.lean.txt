/-- The selector's router: capture the read rank, enter the unary state word at cell `2` of
the summary tape, request one whole state group per `true`, and the rank suffix at the
terminating `false`. (`h1SeekRouter`.) -/
def seekRouterTM (k : ℕ) : MultiTapeTM (k + 2) Bool (SeekCtrl k) where
  q₀ := .observe
  tr := fun q inp w => match q with
    | .observe => tapeAction (summaryTape k) 0 none .pos
        (.advance (readRank inp (fun i => w (Fin.castAdd 2 i))))
    | .advance rank => tapeAction (summaryTape k) 0 none .pos (.state rank)
    | .state rank =>
        if w (summaryTape k) = some true then
          tapeAction (summaryTape k) 0 none .pos
            (.request rank ⟨3 ^ (k + 1) - 1, by
              have hp : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega); omega⟩ false)
        else inputMoveAction 0 (if hr : rank.val = 0 then .done
          else .request rank ⟨rank.val - 1, by have := rank.isLt; omega⟩ true)
    | .request rank rest last => inputMoveAction 0 (.request rank rest last)
    | .done => inputMoveAction 0 .done
