/-- The step block's phase machines; every variable scan is a landed catalog row
(`Turing.skipTM`, `Turing.skipBackTM`, `Turing.clearTM`). (`h1RoundPart`.) -/
def part (k : ℕ) : (key : Key) → MultiTapeTM (k + 2) Bool (Ph k key)
  | .entry | .halt => ⟨(), fun _ _ _ => inputMoveAction 0 ()⟩
  | .lookup => seekTM k
  | .stateTail | .stateGap _ => skipTM (k + 2) (summaryTape k)
  | .stateHome | .stateRestore _ => skipBackTM (k + 2) (summaryTape k)
  | .clearState => clearTM (k + 2) (summaryTape k)
  | .toSummary | .finish _ => twoMoveTM (summaryTape k) .neg
  | .toState => twoMoveTM (summaryTape k) .pos
  | .record => recordTM k
  | .successor => succFieldTM (tableTape k) (summaryTape k) true
  | .tableTail _ => skipTM (k + 2) (tableTape k)
  | .tableHome _ => skipBackTM (k + 2) (tableTape k)
