/-- *Fit:* `Primitives`' `emitterP2_frame` (`Build/Primitives.lean:5199`, private there),
verbatim, is `Turing.embedEmitCfg_mapState_eq`. -/
theorem emitterP2_frame {k l : ℕ} {S H : Type} {x : List Bool}
    (_q : S) (index : Fin k ↪ Fin l) (emb : S → H)
    (tapes : Fin l → ℤ → Option Bool) (heads : Fin l → ℤ)
    (c : Cfg k Bool S x) (d : Cfg l Bool H x)
    (hs : c.state.map emb = d.state) (hi : c.inputPos = d.inputPos)
    (ho : c.output = d.output)
    (selected : ∀ i, c.workTapes i = d.workTapes (index i) ∧
      c.workTapePos i = d.workTapePos (index i))
    (frame : ∀ j, j ∉ Set.range index → tapes j = d.workTapes j ∧ heads j = d.workTapePos j) :
    (embedEmitCfg index tapes heads [] c).mapState emb = d :=
  embedEmitCfg_mapState_eq index emb tapes heads [] c d hs hi ho selected frame
