/-- At the unary terminator, the selector skips exactly the read rank.
This includes rank zero: the router's single silent step exits immediately.

**Proof sketch.** The first step is `seekTM_stateStep` at a false bit. With no records the
rank is zero, that step lands on `done` with nothing skipped, and the cut and frame are
checked at times `0` and `1` directly. Otherwise the rank is positive and the step enters
the request with countdown `rank - 1 = records.length - 1` and `last` true, which is
`seekTM_many`'s entry for the whole list. Shifting its endpoint, cut and frame by that one
step (`MultiTapeTM.runFrom_succ_eq_step`) gives the claim; time `0` is the router control
`state`, checked directly. -/
private lemma seekTM_rank {k n : ℕ} {x : List Bool}
    (records : List (Action k Bool (Fin (n + 1))))
    (d : Cfg (k + 2) Bool (RecordState k) x) (rank : Fin (3 ^ (k + 1)))
    (hlen : records.length = rank.val)
    (hq : d.state = some (recordTM k).q₀)
    (hs : d.workTapeSymbols (summaryTape k) = some false)
    (hw : ∀ (r : ℕ) (hr : r < (records.flatMap actionBitsK).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((records.flatMap actionBitsK)[r]'hr)) :
    (seekTM k).runFrom (seekCfg d (.state rank)) (1 + seekCost records) =
      seekCfg ({d with
        workTapePos := Function.update d.workTapePos (tableTape k)
          (d.workTapePos (tableTape k) + (records.flatMap actionBitsK).length)}) .done ∧
    (∀ u < 1 + seekCost records,
      ((seekTM k).runFrom (seekCfg d (.state rank)) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ 1 + seekCost records,
      let f := (seekTM k).runFrom (seekCfg d (.state rank)) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ tableTape k → f.workTapePos j = d.workTapePos j) := by
  have step := seekTM_stateStep d rank false hq hs
  simp only [Bool.false_eq_true, ↓reduceIte] at step
  by_cases hn : records = []
  · subst records
    have hz : rank.val = 0 := hlen.symm
    simp only [hz, ↓reduceDIte] at step
    have he : (seekTM k).runFrom (seekCfg d (.state rank)) 1 = seekCfg d .done := step
    simp only [seekCost, List.flatMap_nil, List.length_nil, Nat.mul_zero,
      Nat.cast_zero, add_zero, Function.update_eq_self]
    refine ⟨he, ?_, ?_⟩
    · intro u hu
      have hu0 : u = 0 := by omega
      simp [hu0, seekCfg, Cfg.mapState, hq]
    · intro u hu
      rcases Nat.le_one_iff_eq_zero_or_eq_one.mp hu with rfl | rfl
      · exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
      · rw [he]
        exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
  · have hp : 0 < rank.val := by rw [← hlen]; exact List.length_pos_iff.mpr hn
    simp only [show rank.val ≠ 0 by omega, ↓reduceDIte] at step
    have spec := seekTM_many records hn (by rw [hlen]; exact Nat.le_of_lt rank.isLt)
      d rank true hq hw
    dsimp only at spec
    simp only [↓reduceIte] at spec
    have hm : (⟨rank.val - 1, by have := rank.isLt; omega⟩ : Fin (3 ^ (k + 1))) =
        ⟨records.length - 1, by have := rank.isLt; omega⟩ := Fin.ext (by dsimp only; omega)
    rw [hm] at step
    have advance (u : ℕ) := MultiTapeTM.runFrom_succ_eq_step (tm := seekTM k)
      (cfg := seekCfg d (.state rank)) (t := u)
    rw [step] at advance
    refine ⟨?_, ?_, ?_⟩
    · rw [Nat.add_comm 1, advance]
      exact spec.1
    · intro u hu
      cases u with
      | zero => simp [seekCfg, Cfg.mapState, hq]
      | succ u =>
        rw [advance u]
        exact spec.2.1 u (by omega)
    · intro u hu
      cases u with
      | zero => exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
      | succ u =>
        rw [advance u]
        exact spec.2.2 u (by omega)
