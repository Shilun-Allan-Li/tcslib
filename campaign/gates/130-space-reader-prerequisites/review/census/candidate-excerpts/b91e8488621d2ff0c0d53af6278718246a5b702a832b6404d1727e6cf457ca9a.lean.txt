/-- **The input rewind leaf, framed**: from a start whose input head is at `p ≥ 1`, it parks at
exactly `p + 1` and not earlier, with the input head at `1` and every other field unchanged. -/
private theorem un_inRew_run {k : ℕ} {z : List Bool} (d : Cfg k Bool (Fin 3) z)
    (hstate : d.state = some 0) (hp : 1 ≤ d.inputPos.val) :
    (un_inRewTM k).runFrom d (d.inputPos.val + 1) =
        { d with state := some 2, inputPos := 1 } ∧
      ∀ t < d.inputPos.val + 1, ((un_inRewTM k).runFrom d t).state ≠ some 2 := by
  classical
  have hlt := d.inputPos.isLt
  -- the scan's trajectory
  let T : ℕ → Cfg k Bool (Fin 3) z := fun r =>
    { d with
      state := some (if r = 0 then 0 else 1)
      inputPos := ⟨d.inputPos.val - r, by omega⟩ }
  have hT0 : T 0 = d := by
    refine Cfg.ext ?_ ?_ rfl rfl rfl
    · show some (if 0 = 0 then (0 : Fin 3) else 1) = d.state
      rw [if_pos rfl, hstate]
    · refine Fin.ext ?_
      show d.inputPos.val - 0 = d.inputPos.val
      omega
  have hstep : ∀ r : ℕ, r ≤ d.inputPos.val - 1 →
      (un_inRewTM k).step (T r) = T (r + 1) := by
    intro r hr
    have hs : (T r).state = some (if r = 0 then 0 else 1) := rfl
    rw [MultiTapeTM.step_of_some hs]
    have htr : (un_inRewTM k).tr (if r = 0 then 0 else 1) (T r).inputSymbol
        (T r).workTapeSymbols = FinTM.controlAction .neg (some 1) := by
      by_cases hr0 : r = 0
      · rw [if_pos hr0]
        trace "M615" <;> simp only [un_inRewTM, Fin.reduceEq, ↓reduceIte]
      · rw [if_neg hr0]
        have hread : (T r).inputSymbol = some (z[d.inputPos.val - r - 1]'(by omega)) := by
          refine inputSymbolInner (d.inputPos.val - r - 1) ?_ (by omega)
          show d.inputPos.val - r = 1 + (d.inputPos.val - r - 1)
          omega
        rw [hread]
        trace "M622" <;> simp only [un_inRewTM, Fin.reduceEq, ↓reduceIte]
    rw [htr, FinTM.controlAction_apply]
    refine Cfg.ext ?_ ?_ rfl rfl rfl
    · show some 1 = some (if r + 1 = 0 then (0 : Fin 3) else 1)
      rw [if_neg (Nat.succ_ne_zero r)]
    · show moveInputPos (T r).inputPos .neg = (T (r + 1)).inputPos
      rw [moveInputPos_neg_of_ne_left _ (by
        intro h
        have hv : d.inputPos.val - r = 0 := by
          have hvv := congrArg Fin.val h
          trace "M632" <;> simpa [T] using hvv
        omega)]
      exact Fin.ext (by show d.inputPos.val - r - 1 = d.inputPos.val - (r + 1); omega)
  have hrun : ∀ r : ℕ, r ≤ d.inputPos.val → (un_inRewTM k).runFrom d r = T r := by
    intro r hr
    induction r with
    | zero => rw [MultiTapeTM.runFrom_zero, hT0]
    | succ r ih =>
      rw [MultiTapeTM.runFrom_succ_eq_step', ih (by omega), hstep r (by omega)]
  have hfinal : (un_inRewTM k).step (T d.inputPos.val) =
      { d with state := some 2, inputPos := 1 } := by
    have hs : (T d.inputPos.val).state = some 1 := by
      show some (if d.inputPos.val = 0 then (0 : Fin 3) else 1) = some 1
      rw [if_neg (by omega)]
    rw [MultiTapeTM.step_of_some hs]
    have hzero : (T d.inputPos.val).inputPos = 0 := by
      refine Fin.ext ?_
      show d.inputPos.val - d.inputPos.val = (0 : Fin (z.length + 2)).val
      rw [Fin.val_zero]
      omega
    have hread : (T d.inputPos.val).inputSymbol = none := Cfg.inputSymbol_of_zero hzero
    have htr : (un_inRewTM k).tr 1 (T d.inputPos.val).inputSymbol
        (T d.inputPos.val).workTapeSymbols = FinTM.controlAction .pos (some 2) := by
      rw [hread]
      trace "M656" <;> simp only [un_inRewTM, Fin.reduceEq, ↓reduceIte]
    rw [htr, FinTM.controlAction_apply]
    refine Cfg.ext rfl ?_ rfl rfl rfl
    show moveInputPos (T d.inputPos.val).inputPos .pos = (1 : Fin (z.length + 2))
    rw [hzero, moveInputPos_pos_of_ne_right (0 : Fin (z.length + 2))
      (by rw [Fin.val_zero]; omega)]
    exact Fin.ext (by trace "M662" <;> simp [Fin.val_one])
  constructor
  · rw [MultiTapeTM.runFrom_succ_eq_step', hrun d.inputPos.val le_rfl, hfinal]
  · intro t ht
    rw [hrun t (by omega)]
    show some (if t = 0 then (0 : Fin 3) else 1) ≠ some 2
    by_cases ht0 : t = 0
    · rw [if_pos ht0]; decide
    · rw [if_neg ht0]; decide
