/-! ## Part C: alternative H2 (per-code linear, input-independent) and H2 ⇒ H1 -/

/-- H2: same verdicts, within `d · (|α| + 1)^d · (t + 1)` — no `|x|` term. -/
theorem exists_timedSimulator :
    ∃ (S : FinTM Bool) (d : ℕ), ∀ (α x : List Bool) (t : ℕ),
      ((F.toFinTM (F.decode α)).ComputesInTime x [true] t →
        S.ComputesInTime (pairEncode (pairEncode (Nat.bits t) α) x) [true]
          (d * (α.length + 1) ^ d * (t + 1))) ∧
      (¬(F.toFinTM (F.decode α)).ComputesInTime x [true] t →
        S.ComputesInTime (pairEncode (pairEncode (Nat.bits t) α) x) [false]
          (d * (α.length + 1) ^ d * (t + 1))) := by
  sorry

/-- The joint bound dominates the per-code-linear one (pure arithmetic). -/
lemma timed_le_joint (d a b t : ℕ) :
    d * (a + 1) ^ d * (t + 1) ≤ (d + 1) * (a + b + t + 1) ^ (d + 1) := by
  have h₁ : (a + 1) ^ d ≤ (a + b + t + 1) ^ d := Nat.pow_le_pow_left (by omega) d
  have h₂ : t + 1 ≤ a + b + t + 1 := by omega
  calc d * (a + 1) ^ d * (t + 1)
      ≤ (d + 1) * (a + b + t + 1) ^ d * (a + b + t + 1) :=
        Nat.mul_le_mul (Nat.mul_le_mul (Nat.le_succ d) h₁) h₂
    _ = (d + 1) * (a + b + t + 1) ^ (d + 1) := by rw [Nat.pow_succ, Nat.mul_assoc]

/-- H2 implies H1 (the instances need only H1). -/
theorem exists_uniformSimulator_of_timed :
    ∃ (S : FinTM Bool) (d : ℕ), ∀ (α x : List Bool) (t : ℕ),
      ((F.toFinTM (F.decode α)).ComputesInTime x [true] t →
        S.ComputesInTime (pairEncode (pairEncode (Nat.bits t) α) x) [true]
          (d * (α.length + x.length + t + 1) ^ d)) ∧
      (¬(F.toFinTM (F.decode α)).ComputesInTime x [true] t →
        S.ComputesInTime (pairEncode (pairEncode (Nat.bits t) α) x) [false]
          (d * (α.length + x.length + t + 1) ^ d)) := by
  obtain ⟨S, d, hS⟩ := F.exists_timedSimulator
  refine ⟨S, d + 1, fun α x t => ⟨fun h => ?_, fun h => ?_⟩⟩
  · exact ((hS α x t).1 h).mono (timed_le_joint d _ _ _)
  · exact ((hS α x t).2 h).mono (timed_le_joint d _ _ _)

end MachineFormat

/-! ## Part F: sorry sources (maintainer admission walk) -/

open Lean Elab Command in
/-- Every constant in the dependency closure of `root` whose own type or value
mentions `sorryAx` (maintainer's read-only admission walk). -/
def sorrySourcesOf (env : Environment) (root : Name) : Array Name := Id.run do
  let mut visited : NameSet := {}
  let mut stack : Array Name := #[root]
  let mut srcs : Array Name := #[]
  while !stack.isEmpty do
    let c := stack.back!
    stack := stack.pop
    if visited.contains c then continue
    visited := visited.insert c
    if let some ci := env.find? c then
      let used := ci.getUsedConstantsAsSet
      if used.contains ``sorryAx then srcs := srcs.push c
      for d in used.toList do
        unless visited.contains d do stack := stack.push d
  return srcs

open Lean Elab Command in
elab "#sorry_sources " id:ident : command => do
  let n ← liftCoreM <| realizeGlobalConstNoOverload id
  logInfo m!"{n}: direct sorryAx users in closure: {sorrySourcesOf (← getEnv) n}"
