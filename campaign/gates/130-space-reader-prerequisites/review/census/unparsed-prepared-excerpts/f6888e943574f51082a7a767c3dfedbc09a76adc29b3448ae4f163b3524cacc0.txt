open Lean Elab Command Meta in
elab "#rv_deep " ids:ident* : command => do
  let env ← getEnv
  let mut stack : Array Name := #[]
  for id in ids do stack := stack.push (← liftCoreM <| realizeGlobalConstNoOverload id)
  let roots := stack
  let mut seen : NameSet := {}
  let mut out : Array Name := #[]
  while !stack.isEmpty do
    let c := stack.back!
    stack := stack.pop
    if seen.contains c then continue
    seen := seen.insert c
    let some ci := env.find? c | continue
    let isT := match env.getModuleIdxFor? c with
      | some i => ((env.header.moduleNames[i]?).getD `none).getRoot == `TCSlib
      | none => false
    unless isT do continue
    match ci with
    | .defnInfo d =>
      let red := getReducibilityStatusCore env c
      unless roots.contains c do
        if red matches .semireducible && !env.isProjectionFn c then out := out.push c
      for u in d.value.getUsedConstants do
        unless seen.contains u do stack := stack.push u
    | _ => pure ()
  let simps ← liftCoreM Meta.getSimpTheorems
  for c in out.qsort (fun a b => a.toString < b.toString) do
    let modName := match env.getModuleIdxFor? c with
      | some i => (env.header.moduleNames[i]?).getD `none | none => `this
    logInfo m!"DEEP {c} | {modName} | inst={Meta.isInstanceCore env c} | simpUnfold={simps.isDeclToUnfold c}"
  logInfo m!"DEEP total {out.size}"
  for c in roots do
    logInfo m!"ROOT {c} | simpUnfold={simps.isDeclToUnfold c} | red={repr (getReducibilityStatusCore env c)}"
