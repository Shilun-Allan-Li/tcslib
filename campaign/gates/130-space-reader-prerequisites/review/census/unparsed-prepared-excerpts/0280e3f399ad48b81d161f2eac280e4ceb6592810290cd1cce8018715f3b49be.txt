/-- Existing H5 and R1' witnesses can be chosen before the input, ruler and
first-halting time. The source machine M is fixed; this is a component contract,
not a constructed F6 prefix. The identity returning embedding adds no tape. -/
theorem uniform_hosted_return {m : ℕ} {P : Type} [Fintype P] [DecidableEq P]
    (M : MultiTapeTM m Bool P) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q)
      (H : MultiTapeTM (m + 1) Bool Q) (inj : P → Bool → Q) (C : ℕ), 0 < C ∧
      ∀ {y z : List Bool} (ℓ : ℕ) (hz : z.length = ℓ + y.length),
        0 < ℓ → z.drop ℓ = y →
        ∀ (rt : ℤ → Option Bool) (rp : ℤ),
          (∀ p : ℤ, -1 ≤ p → p ≤ (ℓ : ℤ) →
            rt (rp + p) = FinTM.bufferTape (List.replicate ℓ true) p) →
        ∀ (c : Cfg m Bool P y) (b : Bool), (b = true ↔ c.inputPos.val = 0) →
        ∀ (t : ℕ), c.state ≠ none →
          (∀ u < t, (M.runFrom c u).state ≠ none) →
          (M.runFrom c t).state = none →
        ∃ t', t ≤ t' ∧ t' ≤ C * (ℓ + 1) * t ∧
          let d := rulerHostCfg ℓ hz (c.state.map fun q => inj q b) c rt rp
          let ι := Function.Embedding.refl (Fin (m + 1))
          let ec := embedEmitCfg ι (fun _ _ => none) (fun _ => 0) [] d
          let A := embedEmitRetTM ι H
          (A.runFrom (ec.mapState Sum.inl) t').state = some (Sum.inr ()) ∧
          (A.runFrom (ec.mapState Sum.inl) t').output = (M.runFrom c t).output ∧
          (∀ u < t', (A.runFrom (ec.mapState Sum.inl) u).state ≠ some (Sum.inr ())) ∧
          (∀ j : Fin (m + 1),
            A.visitedByTapeHead (ec.mapState Sum.inl) t' (ι j) =
              H.visitedByTapeHead d t' j) ∧
          (∀ j : Fin m, H.visitedByTapeHead d t' j.castSucc =
            M.visitedByTapeHead c t j) ∧
          H.visitedByTapeHead d t' (Fin.last m) ⊆ Finset.Icc (rp - 1) (rp + ℓ) := by
  classical
  obtain ⟨Q, fQ, dQ, H, inj, C, hC, hH⟩ := exists_rulerHostTM M
  letI : Fintype Q := fQ
  letI : DecidableEq Q := dQ
  refine ⟨Q, fQ, dQ, H, inj, C, hC, ?_⟩
  intro y z ℓ hz hℓ hdrop rt rp hr c b hb t hc hlive hhalt
  obtain ⟨t', hlo, hhi, ⟨b', _, hend⟩, hinner, _, hbank, hruler⟩ :=
    hH ℓ hz hℓ hdrop rt rp hr c b hb t
  let d := rulerHostCfg ℓ hz (c.state.map fun q => inj q b) c rt rp
  let ι := Function.Embedding.refl (Fin (m + 1))
  let ec := embedEmitCfg ι (fun _ _ => none) (fun _ => 0) [] d
  have hclive : d.state ≠ none := by
    rw [rulerHostCfg_state]
    intro h
    exact hc (Option.map_eq_none_iff.mp h)
  have hHlive : ∀ u < t', (H.runFrom d u).state ≠ none := hinner hlive
  have hHhalt : (H.runFrom d t').state = none := by
    rw [hend, rulerHostCfg_state, hhalt, Option.map_none]
  obtain ⟨_, hreturn, hcut⟩ := embedEmitRetTM_run ι H (fun _ _ => none)
    (fun _ => 0) [] d t' hclive hHlive hHhalt
  refine ⟨t', hlo, hhi, ?_, ?_, hcut, ?_, hbank, hruler⟩
  · rw [hreturn]
  · rw [hreturn, embedEmitCfg_output, List.nil_append, hend, rulerHostCfg_output]
  · intro j
    calc
      _ = (embedEmitTM ι H).visitedByTapeHead ec t' (ι j) :=
        embedEmitRetTM_visitedByTapeHead ι H (fun _ _ => none) (fun _ => 0) [] d t' (ι j)
      _ = H.visitedByTapeHead d t' j :=
        (embedEmitTM_visitedByTapeHead ι H (fun _ _ => none) (fun _ => 0) [] d t' j).1

end F6LayoutPreship
