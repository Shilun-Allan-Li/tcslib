namespace G106FPReaderCheck
open Complexity Turing
 theorem map_consumer {g : List Bool → List Bool} (hg : PolyTimeComputable g) :
    PolyTimeComputable (fun z => match pairDecode z with
      | some (a,b) => pairEncode a (g b)
      | none => []) := by
  simpa only [Complexity.pairMapSnd.eq_def] using PolyTimeComputable.pairMapSnd hg
 theorem pair_consumer {f g : List Bool → List Bool}
    (hf : PolyTimeComputable f) (hg : PolyTimeComputable g) :
    PolyTimeComputable (fun x => pairEncode (f x) (g x)) :=
  PolyTimeComputable.pairEncode hf hg
 theorem append_consumer {f g : List Bool → List Bool}
    (hf : PolyTimeComputable f) (hg : PolyTimeComputable g) :
    PolyTimeComputable (fun x => f x ++ g x) := PolyTimeComputable.append hf hg
 theorem pair_input_consumer {f : List Bool → List Bool} (hf : PolyTimeComputable f) :
    PolyTimeComputable (fun x => pairEncode (f x) x) :=
  PolyTimeComputable.pairEncode hf polyTimeComputable_id
 theorem unary_consumer (C e : ℕ) :
    PolyTimeComputable (fun x => List.replicate (C * (x.length + 1)^e) true) :=
  polyTimeComputable_polyUnary C e
 theorem unary_length_consumer :
    PolyTimeComputable (fun x => List.replicate x.length true) := polyTimeComputable_unary
end G106FPReaderCheck
