import TCSlib.Complexity.ClassNP.SAT
import TCSlib.Complexity.ClassNP.Tautology
import TCSlib.Complexity.CookLevin.Hardness
import TCSlib.Complexity.SpaceComplexity.Machines.Frag
import TCSlib.Complexity.TuringMachine.Build.Catalog
import TCSlib.Complexity.TuringMachine.Build.Contract.Basic
import TCSlib.Complexity.TuringMachine.Build.CounterLoop
import TCSlib.Complexity.TuringMachine.Build.Primitives
import TCSlib.Complexity.TuringMachine.Robustness.SingleTape

open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let wanted : List Name := [`TCSlib.Complexity.ClassNP.SAT, `TCSlib.Complexity.ClassNP.Tautology, `TCSlib.Complexity.CookLevin.Hardness, `TCSlib.Complexity.SpaceComplexity.Machines.Frag, `TCSlib.Complexity.TuringMachine.Build.Catalog, `TCSlib.Complexity.TuringMachine.Build.Contract.Basic, `TCSlib.Complexity.TuringMachine.Build.CounterLoop, `TCSlib.Complexity.TuringMachine.Build.Primitives, `TCSlib.Complexity.TuringMachine.Robustness.SingleTape]
  for mod in wanted do
    unless (env.getModuleIdx? mod).isSome do throwError "missing imported module {mod}"
  let names := env.constants.toList.filterMap fun (n, ci) =>
    match env.getModuleIdxFor? n with
    | some i =>
      let mod := env.header.moduleNames[i.toNat]!
      if wanted.contains mod then some (n, ci, mod) else none
    | none => none
  for (n, ci, mod) in names.toArray.qsort (fun a b => a.1.toString < b.1.toString) do
    let kind := match ci with
      | .thmInfo _ => "theorem" | .defnInfo _ => "definition"
      | .opaqueInfo _ => "opaque" | .axiomInfo _ => "axiom"
      | .ctorInfo _ => "constructor" | .recInfo _ => "recursor"
      | .inductInfo _ => "inductive" | .quotInfo _ => "quotient"
    logInfo m!"CONSTANT {n} module={mod} kind={kind} private={isPrivateName n} internal={n.isInternalDetail}"
    if n.toString.endsWith ".Complexity.sat_guard_add" || n.toString.endsWith ".Complexity.clA5_guard_add" then
      logInfo m!"GUARD {n} : {ci.type}"
      let ax ← Lean.collectAxioms n
      logInfo m!"GUARD_AXIOMS {n}: {ax}"
