run_cmd do
  let env ← getEnv
  let mut guardUnion : Array Name := #[]
  for suffix in ["Complexity.sat_live_prefix", "Complexity.clA5_live_prefix"] do
    let matches := env.constants.toList.filter fun (n, _) =>
      isPrivateName n && n.toString.endsWith ("." ++ suffix)
    unless matches.length == 1 do throwError "F4 private root mismatch {suffix}"
    let (root, ci) := matches.head!
    let mut pending := ci.type.getUsedConstants ++ ((ci.value?.map Expr.getUsedConstants).getD #[])
    let mut visited : Array Name := #[]
    let mut index := 0
    while index < pending.size do
      let n := pending[index]!
      index := index + 1
      if visited.contains n then continue
      visited := visited.push n
      let some idx := env.getModuleIdxFor? n | continue
      let home := env.header.moduleNames[idx.toNat]!
      unless (`TCSlib).isPrefixOf home do continue
      let some info := env.find? n | throwError "missing reachable name {n}"
      match info with
      | .defnInfo d =>
        let projection := (Lean.getProjectionFnInfo? env n).isSome
        let abbrev := match d.hints with | .abbrev => true | _ => false
        let ordinary := !projection && !abbrev
        if ordinary && !guardUnion.contains n then guardUnion := guardUnion.push n
        logInfo m!"F4DEF|{(Json.mkObj [("root", toJson root.toString), ("name", toJson n.toString),
          ("owner", toJson home.toString), ("private", toJson (isPrivateName n)),
          ("projection", toJson projection), ("kernel_abbrev", toJson abbrev),
          ("reducibility", toJson (reprStr (getReducibilityStatusCore env n))),
          ("ordinary_guard_required", toJson ordinary)]).compress}"
        pending := pending ++ d.type.getUsedConstants ++ d.value.getUsedConstants
      | .thmInfo _ =>
        logInfo m!"F4THEOREM_BOUNDARY|{(Json.mkObj [("root", toJson root.toString),
          ("name", toJson n.toString), ("owner", toJson home.toString),
          ("value_not_traversed", toJson true)]).compress}"
        pending := pending ++ info.type.getUsedConstants
      | _ => pure ()
  logInfo m!"F4_GUARD_UNION|{(toJson ((guardUnion.qsort Name.lt).toList.map Name.toString)).compress}"
