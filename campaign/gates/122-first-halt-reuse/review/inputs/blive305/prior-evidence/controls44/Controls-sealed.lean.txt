attribute [-simp] Turing.Cfg.init Turing.MultiTapeTM.initCfg
attribute [local irreducible] Turing.Cfg.init Turing.FinTM.ComputesInTime Turing.FinTM.VirtualTag Turing.FinTM.bufferTape Turing.FinTM.controlAction Turing.FinTM.leftAction Turing.FinTM.rightAction Turing.FinTM.tapeBlocks Turing.FinTM.virtualMove Turing.FinTM.virtualNextTag Turing.MultiTapeTM.initCfg Turing.MultiTapeTM.runFrom Turing.MultiTapeTM.spaceUsed Turing.MultiTapeTM.spaceUsedByTape Turing.MultiTapeTM.step Turing.MultiTapeTM.visitedByTapeHead Turing.NDTM.runWith Turing.NDTM.stepWith Turing.captureAction Turing.instDecidableEqUniversalControl Turing.instFintypeUniversalControl Turing.universalInterpreter
open Lean Elab Command Meta in
run_cmd do
  let targets : List Name := [`Turing.Cfg.init, `Turing.FinTM.ComputesInTime, `Turing.FinTM.VirtualTag, `Turing.FinTM.bufferTape, `Turing.FinTM.controlAction, `Turing.FinTM.leftAction, `Turing.FinTM.rightAction, `Turing.FinTM.tapeBlocks, `Turing.FinTM.virtualMove, `Turing.FinTM.virtualNextTag, `Turing.MultiTapeTM.initCfg, `Turing.MultiTapeTM.runFrom, `Turing.MultiTapeTM.spaceUsed, `Turing.MultiTapeTM.spaceUsedByTape, `Turing.MultiTapeTM.step, `Turing.MultiTapeTM.visitedByTapeHead, `Turing.NDTM.runWith, `Turing.NDTM.stepWith, `Turing.captureAction, `Turing.instDecidableEqUniversalControl, `Turing.instFintypeUniversalControl, `Turing.universalInterpreter]
  let mut i : Nat := 0
  for name in targets do
    let some (.defnInfo info) := (← getEnv).find? name | throwError "not ordinary definition {name}"
    for mode in ["point", "abstract"] do
      i := i + 1
      let result ← liftTermElabM do
        let levels := info.levelParams.map Level.param
        let head := mkConst name levels
        let solve := fun (lhs rhs : Expr) => do
          let ty ← mkEq lhs rhs
          let goal ← mkFreshExprSyntheticOpaqueMVar ty
          try
            goal.mvarId!.refl
            pure (some (ty, ← instantiateMVars goal))
          catch _ =>
            logInfo m!"CONTROL|{i}|{name}|{mode}|FAIL"
            logError m!"CONTROL_RFL|{i}|{name}|{mode}|rfl did not close reflected identity"
            pure none
        if mode == "abstract" then solve head info.value
        else forallTelescopeReducing info.type fun xs _ => do
          if xs.isEmpty then
            let unitTy := mkConst ``Unit
            solve (Expr.lam `_u unitTy head .default)
              (Expr.lam `_u unitTy info.value .default)
          else
            let result ← solve (mkAppN head xs) ((mkAppN info.value xs).headBeta)
            match result with
            | none => pure none
            | some (ty, value) => pure (some (← mkForallFVars xs ty, ← mkLambdaFVars xs value))
      if let some (ty, value) := result then
        let ty ← liftTermElabM <| instantiateMVars ty
        let value ← liftTermElabM <| instantiateMVars value
        if ty.hasFVar || ty.hasMVar || value.hasFVar || value.hasMVar then
          throwError "unclosed control {i}"
        let kernelName := Name.mkSimple ("FHFullControl_" ++ toString i)
        liftCoreM <| addDecl (.thmDecl { name := kernelName, levelParams := info.levelParams, type := ty, value := value })
        logInfo m!"CONTROL|{i}|{name}|{mode}|PASS|kernelChecked={kernelName}"
  logInfo m!"CONTROL_COMPLETE|{i}"
