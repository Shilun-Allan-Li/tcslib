"""The re-pointed exists_ndInterpreterCore: the startup at p = 4, then phase 12 (A+).
usage: mkcore.py <NDCodeSimulator.lean> <out proof file>"""
import sys, re
src, out = sys.argv[1:3]
L = open(src, encoding='utf-8').read().split('\n')
def find(pat, start=1):
    for i in range(start - 1, len(L)):
        if pat in L[i]: return i + 1
    raise SystemExit('not found: ' + pat)
def block(a, b): return '\n'.join(L[a - 1:b])
p12a = block(find('  -- Phase 12: the loop entry.'), find('  -- The complete tape inventory at the loop') - 1)
p12b = block(find('  -- The counter-loop configuration\'s fields'), find('  -- The target\'s start is the composite\'s start.') - 1)
verdict = block(find('  · -- the verdict'), find('    · simp only [hP, decide_false, if_false]'))
txt = p12a + '\n' + p12b
def rep(a, b, n=None):
    global txt
    c = txt.count(a); assert c >= 1 and (n is None or c == n), (c, a[:80]); txt = txt.replace(a, b)
rep('(e₁₁.mapState (Sum.inr : _ → St₁₁))', 'e')
txt = re.sub(r'(?<![A-Za-z_])e₁₁(?![₀-₉A-Za-z_])', 'e', txt)
rep('(hblank jb0 (Or.inl hjb0))', 'hb0')
rep('(hblank jb1 (Or.inr (Or.inl hjb1)))', 'hb1')
rep('(hblank capL (Or.inr (Or.inr hcapL)))', 'hcL')
rep("""      intro j n0 n1 n2 n3 n4 n5 n6
      rintro ⟨i, rfl⟩
      have hv := hι₇v i
      split_ifs at hv
      · exact n0 (Fin.ext (by omega))
      · exact n1 (Fin.ext (by omega))
      · exact n2 (Fin.ext (by omega))
      · exact n3 (Fin.ext (by omega))
      · exact n4 (Fin.ext (by omega))
      · exact n5 (Fin.ext (by omega))
      · exact n6 (Fin.ext (by omega))""",
"""      intro j n0 n1 n2 n3 n4 n5 n6
      rintro ⟨i, rfl⟩
      fin_cases i
      · exact n0 rfl
      · exact n1 rfl
      · exact n2 rfl
      · exact n3 rfl
      · exact n4 rfl
      · exact n5 rfl
      · exact n6 rfl""")
rep("""    · rw [Cfg.mapState_state, Cfg.mapState_state, embedSilentCfg_state, Cfg.mapState_state,
        vhostCfg_state, hcl_state, he₁₁]
      rfl""",
"""    · rw [Cfg.mapState_state, embedSilentCfg_state, Cfg.mapState_state,
        vhostCfg_state, hcl_state, hst]
      rfl""")
rep("""    s11run s11st s11cut h₂12""", """    he.symm hst hcut h₂12""")
rep("""  obtain ⟨sRrun, sRst, sRcut⟩ := seamCompTM_step _ (Sum.inr RewindPhase.done)""",
    """  obtain ⟨sRrun, sRst, sRcut⟩ := seamCompTM_step U doneU""")
open(out, 'w', encoding='utf-8').write(txt)
open(out + '.verdict', 'w', encoding='utf-8').write(verdict)
left = [l for l in txt.split('\n') if re.search(r'St₁₁|hι₇v|hjb0|hjb1|hcapL\b|s11|TPRE|hstream|hblank', l)]
print(len(txt.split('\n')), 'leftovers:'); print('\n'.join(left[:30]))
