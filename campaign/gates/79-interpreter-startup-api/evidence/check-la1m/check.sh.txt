import sys
W, D = sys.argv[1:3]
rd = lambda p: open(f'{W}/{p}', encoding='utf-8').read().split('\n')
kit = rd('TCSlib/Complexity/TuringMachine/Build/InterpreterStartup.lean')
nd = rd('TCSlib/Complexity/TuringMachine/NDCodeSimulator.lean')

    imps = []
    for p in parts: imps += [l for l in p if l.startswith('import ') and not any(d in l for d in drop)]
    imps = list(dict.fromkeys(imps))
    body = []
    for p in parts: body += [l for l in p if not l.startswith('import ')] + ['']
    open(f'{D}/{out}', 'w', encoding='utf-8').write('\n'.join(imps + [''] + body + prints))

PY
