"""Draft Turing.exists_interpreterStartup from exists_ndInterpreterCore's phases 1-11 (A+).
usage: mkstartup.py <NDCodeSimulator.lean> <out body file>"""
import sys, re
src, out = sys.argv[1:3]
L = open(src, encoding='utf-8').read().split('\n')
def block(a, b):  # 1-based inclusive line range
    return L[a - 1:b]
def find(pat, start=1):
    for i in range(start - 1, len(L)):
        if pat in L[i]: return i + 1
    raise SystemExit('not found: ' + pat)
setup_k = block(find('  set k₁ := (prefixCanonTM'), find('  set k₂ := (prefixCanonTM'))
win_lay = block(find('  obtain ⟨QW, instFW, instDW, W, entryW, doneW, CW, windows⟩ :='), find('  haveI := instDL'))
lets = block(find('  let A₁ := embedSilentRetTM'), find('  let A₁₁ : MultiTapeTM K Bool RewindPhase'))
sts = block(find('  let St₂ : Type :='), find('  let St₁₁ : Type := St₁₀ ⊕ RewindPhase'))
sss = block(find('  let S₂ := seamCompTM A₁'), find('  let S₁₁ := seamCompTM S₁₀'))
ranges = block(find('  -- Ranges of the two header scratch selections, by value.'), find('  -- Phase 12: the loop entry.') - 1)
inv = block(find('  -- The complete tape inventory at the loop\'s threshold.'), find('  -- The stream, the two loop banks') - 1)
named = block(find('  have hctr₁₁ : e₁₁.workTapes jctr'), find('  -- The counter-loop configuration\'s fields') - 1)
start = block(find('  -- The target\'s start is the composite\'s start.'), find('  have hmin : (x.take t).length'))
body = '\n'.join(ranges + inv + named + start)
lets_t = '\n'.join(l for l in lets if 'let A₁₂' not in l)
assert 'let A₁₂' not in lets_t
txt = '\n'.join(setup_k) + '\n' + '\n'.join(win_lay) + '\n' + lets_t + '\n' + '\n'.join(sts) + '\n' + '\n'.join(sss) + '\n' + body
def rep(a, b, n=None):
    global txt
    c = txt.count(a)
    assert c >= 1 and (n is None or c == n), (c, a[:80])
    txt = txt.replace(a, b)
# the installed machine
rep('(c.decode α).tm.replayTM', '(M α)')
rep('(M α).q₀', '(M α).q₀')
# the ranges by value
rep('11 ≤ j.val ∧ j.val < 11 + k₁', '7 ≤ j.val ∧ j.val < 7 + k₁')
rep('11 + k₁ ≤ j.val ∧ j.val < 11 + k₁ + k₂', '7 + k₁ ≤ j.val ∧ j.val < 7 + k₁ + k₂')
rep('j.val - 11', 'j.val - 7')
rep('show 11 + (j.val - 7) = j.val', 'show 7 + (j.val - 7) = j.val')
rep('j.val - (11 + k₁)', 'j.val - (7 + k₁)')
rep('show 11 + k₁ + (j.val - (7 + k₁)) = j.val', 'show 7 + k₁ + (j.val - (7 + k₁)) = j.val')
rep('∀ j : Fin K, j.val < 11 → j ≠ jbt → j ≠ jal →', '∀ j : Fin K, j.val < 7 → j ≠ jbt → j ≠ jal →')
rep('∀ j : Fin K, j.val ≠ 2 →\n      ¬(11 + k₁ + k₂ ≤ j.val ∧ j.val < 10 + k₁ + k₂ + CI.k) →',
    '∀ j : Fin K, j.val ≠ 1 →\n      ¬(7 + k₁ + k₂ ≤ j.val ∧ j.val < 6 + k₁ + k₂ + CI.k) →')
# the blank-off-the-passive-tapes lemma and its call sites
rep("""  have htapes₀blank : ∀ j : Fin K, j ≠ js → tapes₀ j = fun _ => none := by
    intro j hj
    rw [htapes₀]
    simp only [if_neg hj]""",
"""  have htapes₀blank : ∀ j : Fin K, j.val < 6 + k₁ + k₂ + CI.k → tapes₀ j = fun _ => none := by
    intro j hj
    exact htapes₀ j (by rintro ⟨i, rfl⟩; have := hιP i; omega)""")
txt = re.sub(r'htapes₀blank (\S+) \(by intro hmem; have := congrArg Fin\.val hmem; omega\)',
             r'htapes₀blank \1 (by omega)', txt)
txt = re.sub(r'htapes₀blank _ \(by intro hmem; have := congrArg Fin\.val hmem; have := (hι[₁₂]) i; omega\)',
             r'htapes₀blank _ (by have := \1 i; have := i.isLt; omega)', txt)
rep('htapes₀blank jal hnjs', 'htapes₀blank jal (by omega)', 1)
rep("""    have hnjs : jal ≠ js := by
      intro hmem
      have := congrArg Fin.val hmem
      omega
""", '', 1)
rep("""            htapes₀blank _ (by
              intro hmem
              have := congrArg Fin.val hmem
              have hv := hιI i
              rw [if_neg h0] at hv
              omega),""",
"""            htapes₀blank _ (by
              have hv := hιI i
              rw [if_neg h0] at hv
              have := i.isLt
              omega),""", 1)
open(out, 'w', encoding='utf-8').write(txt)
left = [m.start() for m in re.finditer(r'\bjs\b|jb0|jb1|capL|ι₇|ic_|c\.decode|htapes₀\]', txt)]
print('lines', txt.count('\n') + 1, 'leftovers', len(left))
for p in left[:40]:
    ln = txt.count('\n', 0, p) + 1
    print(ln, txt.split('\n')[ln - 1][:120])
