"""Build the core's whole-module seal test from the elaboration recipe's NDCSCat file (the kit's
text, then NDCodeSimulator's, imports merged): insert `section C2Seal` and the seal block (gate
66's 46 `local irreducible` attributes and its `-simp` line) after the core's `namespace Turing`,
close the section before the last `end Turing` (NDCSCat already prints the core's axioms).
Usage: mksealcat.py <NDCSCat.lean> <c2seal-block.lean> <out>"""
import sys
src, blk, out = sys.argv[1:4]
L = open(src, encoding='utf-8').read().split('\n')
B = open(blk, encoding='utf-8').read().rstrip('\n').split('\n')
assert sum(1 for l in B if l.startswith('attribute [local irreducible]')) == 46
assert sum(1 for l in B if l.startswith('attribute [-simp]')) == 1
ns = [i for i, l in enumerate(L) if l == 'namespace Turing']
assert len(ns) == 2, ns  # the kit's, then the core's
ends = [i for i, l in enumerate(L) if l == 'end Turing']
assert ends and ends[-1] > ns[1]
e = ends[-1]
new = L[:ns[1] + 1] + ['section C2Seal', ''] + B + [''] + L[ns[1] + 1:e] + ['end C2Seal', ''] + L[e:]
assert new[-1] == '#print axioms Turing.exists_ndInterpreterCore', new[-1]  # NDCSCat's own print
open(out, 'w', encoding='utf-8').write('\n'.join(new))
