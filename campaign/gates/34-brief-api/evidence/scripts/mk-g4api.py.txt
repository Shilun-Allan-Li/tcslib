"""G4 continuation API: the predicate `_iff` lemmas of Contract/Basic, `roundOrbit_zero`, the hull
formulas, and `Cfg.bindState`'s field equations (Contract/Seam). Asserted edits.
Usage: python3 -I mk-g4api.py <worktree root>"""

def edit(path, pairs):
    t = open(path, encoding='utf-8').read()
    for a, b in pairs:
        assert t.count(a) == 1, (path, a[:70], t.count(a))
        t = t.replace(a, b)
    open(path, 'w', encoding='utf-8').write(t)
