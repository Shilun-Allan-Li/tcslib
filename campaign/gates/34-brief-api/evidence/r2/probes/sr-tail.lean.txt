section BapiConsumer
open Turing
variable {k : ℕ} {Γ S S' : Type} {x : List Γ}

example (f : S → S') (c : Cfg k Γ S x) (q : S) (h : c.state = some q) :
    (c.mapState f).state = some (f q) := by
  rw [Cfg.mapState_state, h, Option.map_some]

example (f : S → S') (c : Cfg k Γ S x) (i : Fin k) :
    (c.mapState f).workTapes i ((c.mapState f).workTapePos i) =
      c.workTapes i (c.workTapePos i) := by
  rw [Cfg.mapState_workTapes, Cfg.mapState_workTapePos]

example (f : S → S') (a : Action k Γ S) :
    (a.mapState f).state = a.state.map f ∧ (a.mapState f).output = a.output :=
  ⟨Action.mapState_state f a, Action.mapState_output f a⟩

end BapiConsumer
