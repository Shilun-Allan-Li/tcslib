"""Name-based unfolding scan of a Lean file, across line breaks (review only; read-only).

Usage: python3 -I unfold_scan.py <name table TSV> <own module> <file.lean> [--all]

For every tactic argument list of `simp`/`simp only`/`simpa`/`simp_all`/`dsimp`/`simp_rw`/`rw`/
`rwa`/`erw`/`rewrite`/`nth_rw`/`norm_num`/`field_simp` (the whole balanced `[...]`, however many
lines it spans) and every `unfold`/`delta` argument, resolve each item's head identifier against
the name table (the file's namespace stack and its `open`s) and print one TSV row per item that
names a definition of another TCSlib module:

  line  definition  module  category  tactic  first-line-of-tactic

Categories: `primitive` (the vendored machine model: Configuration, Deterministic, Finite),
`abbrev` (reducible), `field` (a structure projection), `definition` (otherwise). Theorems,
instances, constructors, local hypotheses and Mathlib names are not definitions of another
TCSlib module and are skipped (`--all` prints the skipped TCSlib items too, as `other:<kind>`).
Comments, docstrings and string literals are masked first, so their text is never scanned.
"""
import re
import sys

table_path, own, path = sys.argv[1:4]
show_all = '--all' in sys.argv[4:]
table = {}
for line in open(table_path, encoding='utf-8'):
    parts = line.rstrip('\n').split('\t')
    if len(parts) == 4:
        table[parts[0]] = (parts[1], parts[2], parts[3])
PRIM = {'TCSlib.Complexity.TuringMachine.Configuration',
        'TCSlib.Complexity.TuringMachine.Deterministic',
        'TCSlib.Complexity.TuringMachine.Finite'}

src = open(path, encoding='utf-8').read()


def mask(t):
    out, i, depth, n = [], 0, 0, len(t)
    while i < n:
        if t.startswith('/-', i):
            depth += 1
            out.append('  ')
            i += 2
            continue
        if depth > 0:
            if t.startswith('-/', i):
                depth -= 1
                out.append('  ')
                i += 2
            else:
                out.append('\n' if t[i] == '\n' else ' ')
                i += 1
            continue
        if t.startswith('--', i):
            j = t.find('\n', i)
            j = n if j < 0 else j
            out.append(' ' * (j - i))
            i = j
            continue
        if t[i] == '"':
            j = i + 1
            while j < n and t[j] != '"':
                j += 2 if t[j] == '\\' else 1
            out.append(' ' * (j + 1 - i))
            i = j + 1
            continue
        out.append(t[i])
        i += 1
    return ''.join(out)


m = mask(src)
assert len(m) == len(src)
line_starts = [0] + [i + 1 for i, c in enumerate(src) if c == '\n']


def line_of(pos):
    lo, hi = 0, len(line_starts) - 1
    while lo < hi:
        mid = (lo + hi + 1) // 2
        if line_starts[mid] <= pos:
            lo = mid
        else:
            hi = mid - 1
    return lo + 1


# namespace stack and opens, per line
ns_at, opens_at = {}, {}
stack, opens = [], []
for ln, text in enumerate(m.split('\n'), 1):
    s = text.strip()
    mo = re.match(r'^namespace\s+(\S+)', s)
    if mo:
        stack.append(('ns', mo.group(1)))
    elif re.match(r'^(noncomputable\s+)?section\b', s):
        stack.append(('sec', s))
    elif re.match(r'^end\b', s) and stack:
        stack.pop()
    mo = re.match(r'^open\s+(.*?)(\s+in)?$', s)
    if mo and not mo.group(2):
        opens = opens + [x for x in mo.group(1).split() if not x.startswith('(')]
    ns_at[ln] = '.'.join(x[1] for x in stack if x[0] == 'ns')
    opens_at[ln] = list(opens)

IDENT = r"[^\W\d][\w.'!?₀-₉ₐ-ₜ]*"


def resolve(name, ln):
    ns = ns_at.get(ln, '')
    cands = []
    parts = ns.split('.') if ns else []
    for k in range(len(parts), -1, -1):
        pre = '.'.join(parts[:k])
        cands.append(pre + '.' + name if pre else name)
    for o in opens_at.get(ln, []):
        cands.append(o + '.' + name)
        if ns:
            cands.append(ns + '.' + o + '.' + name)
    for c in cands:
        if c in table:
            return c
    return None


def items(body):
    out, depth, cur, start = [], 0, [], 0
    for i, ch in enumerate(body):
        if ch in '([{⟨':
            depth += 1
        elif ch in ')]}⟩':
            depth -= 1
        if ch == ',' and depth == 0:
            out.append((start, ''.join(cur)))
            cur, start = [], i + 1
        else:
            cur.append(ch)
    out.append((start, ''.join(cur)))
    return out


TAC = re.compile(r"(?<![\w.'])(simp_all|simp_rw|simpa|simp|dsimp|norm_num|field_simp|rwa|erw|"
                 r"rewrite|rw|nth_rw\s+\d+|nth_rewrite\s+\d+)(\?|!)?(?![\w'])")
rows = []
for mo in TAC.finditer(m):
    j = mo.end()
    # optional config, `only`, `[`
    while True:
        while j < len(m) and m[j] in ' \t\n':
            j += 1
        if m.startswith('only', j) and not re.match(r"[\w']", m[j + 4:j + 5] or ' '):
            j += 4
            continue
        if m.startswith('(config', j) or m.startswith('+', j) or m.startswith('-', j) and m[j + 1:j + 2].isalpha():
            if m[j] == '(':
                d, k = 0, j
                while k < len(m):
                    d += m[k] == '('
                    d -= m[k] == ')'
                    k += 1
                    if d == 0:
                        break
                j = k
            else:
                k = j + 1
                while k < len(m) and (m[k].isalnum() or m[k] in '_.'):
                    k += 1
                j = k
            continue
        break
    if j >= len(m) or m[j] != '[':
        continue
    d, k = 0, j
    while k < len(m):
        if m[k] == '[':
            d += 1
        elif m[k] == ']':
            d -= 1
            if d == 0:
                break
        k += 1
    body = m[j + 1:k]
    tac = mo.group(1)
    tl = line_of(mo.start())
    first = src[line_starts[tl - 1]:src.find('\n', mo.start())].strip()
    for off, it in items(body):
        s = it.strip()
        s = re.sub(r'^(←|<-|↓|↑|-)\s*', '', s)
        if not s or s[0] in '(⟨':
            continue
        mi = re.match(IDENT, s)
        if not mi:
            continue
        name = mi.group(0).rstrip('.')
        pos = j + 1 + off + it.find(mi.group(0))
        ln = line_of(pos)
        r = resolve(name, ln)
        if r is None:
            continue
        kind, mod, red = table[r]
        if mod == own:
            continue
        if kind == 'def':
            cat = 'primitive' if mod in PRIM else ('abbrev' if red == 'reducible' else 'definition')
        elif kind == 'proj':
            cat = 'field'
        else:
            if show_all:
                rows.append((ln, r, mod, 'other:' + kind, tac, first))
            continue
        rows.append((ln, r, mod, cat, tac, first))

UNF = re.compile(r"(?<![\w.'])(unfold|delta|unfold_let)(?![\w'])([^\n;]*)")
for mo in UNF.finditer(m):
    ln = line_of(mo.start())
    args = re.split(r'\s+at\s+|<;>', mo.group(2))[0]
    first = src[line_starts[ln - 1]:src.find('\n', mo.start())].strip()
    for name in re.findall(IDENT, args):
        r = resolve(name, ln)
        if r is None:
            continue
        kind, mod, red = table[r]
        if mod == own:
            continue
        if kind == 'def':
            cat = 'primitive' if mod in PRIM else ('abbrev' if red == 'reducible' else 'definition')
        elif kind == 'proj':
            cat = 'field'
        else:
            continue
        rows.append((ln, r, mod, cat, mo.group(1), first))

rows.sort()
print('line\tdefinition\tmodule\tcategory\ttactic\tsource')
for r in rows:
    print('\t'.join([str(r[0]), r[1], r[2].removeprefix('TCSlib.Complexity.'), r[3], r[4],
                     r[5].replace('\t', ' ')[:160]]))
