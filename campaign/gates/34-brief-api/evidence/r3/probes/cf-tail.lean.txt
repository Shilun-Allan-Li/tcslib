section R3Consumer
open Turing
variable {k : ℕ} {Γ S : Type} {x : List Γ}
attribute [local irreducible] Turing.Action.apply in
example (a : Action k Γ S) (c : Cfg k Γ S x) (i : Fin k) (hm : (a.workTapes i).2 = 1) :
    (a.apply c).workTapePos i = c.workTapePos i + 1 ∧ (a.apply c).state = a.state ∧
      (a.apply c).output = c.output ++ a.output.toList ∧
      (a.apply c).inputPos = moveInputPos c.inputPos a.inputTape := by
  refine ⟨?_, Action.apply_state a c, Action.apply_output a c, Action.apply_inputPos a c⟩
  rw [Action.apply_workTapePos, hm]
  rfl
end R3Consumer
