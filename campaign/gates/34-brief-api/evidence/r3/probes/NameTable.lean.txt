/-! Review probe (read-only): every public TCSlib constant in the six owned files' joint import
context, with its kind, module and reducibility, written to a TSV for the unfolding scanner. -/
open Lean in
#eval show CoreM Unit from do
  let env ← getEnv
  let mut lines : Array String := #[]
  for (n, ci) in env.constants.toList do
    let some idx := env.getModuleIdxFor? n | continue
    let mod := env.header.moduleNames[idx.toNat]!.toString
    unless mod.startsWith "TCSlib" do continue
    if n.isInternal then continue
    let kind := match ci with
      | .defnInfo _ =>
        if env.isProjectionFn n then "proj"
        else if Lean.Meta.isInstanceCore env n then "inst" else "def"
      | .thmInfo _ => "thm"
      | .ctorInfo _ => "ctor"
      | .inductInfo _ => "ind"
      | .axiomInfo _ => "ax"
      | .opaqueInfo _ => "opaque"
      | .recInfo _ => "rec"
      | .quotInfo _ => "quot"
    let red ← match (← getReducibilityStatus n) with
      | .reducible => pure "reducible"
      | .irreducible => pure "irreducible"
      | _ => pure "default"
    lines := lines.push s!"{n}\t{kind}\t{mod}\t{red}"
  IO.FS.writeFile "/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/review/bapi-gate-r2/out/nametable.tsv" (String.intercalate "\n" lines.toList ++ "\n")
  IO.println s!"wrote {lines.size} constants"
