/-! Gate 57 review: the probe's six consumer examples against the PUBLISHED module (its olean in
`oleans-lbw2`), in the seam-layout fill's import context (its two imports plus the one authorized
import), plus three route reads the probe does not show (phase 5's reversed unary word, phases
7-8's window under the unary split's blank, the table tape's endpoint). Sealed. -/

namespace Turing

section BufferWordsConsumer
attribute [local irreducible] Turing.FinTM.bufferTape Turing.pairEncode

variable (a v : List Bool) (q : ℕ)

/-- The unary state's cells, the terminator included. -/
example (r : ℕ) (hr : r ≤ q) :
    FinTM.bufferTape (pairEncode a ((List.replicate q true ++ [false]) ++ v))
      (((a.flatMap fun b => [b, b]) ++ [false, true]).length + r) = some (decide (r < q)) := by
  rw [pairEncode_eq, List.append_assoc ((a.flatMap fun b => [b, b])) [false, true],
    ← List.append_assoc, FinTM.bufferTape_append_shift _ _ (Int.natCast_nonneg r),
    ← FinTM.bufferTape_take _ (q + 1) (by omega : (r : ℤ) < (q + 1 : ℕ)),
    List.take_left' (by simp), FinTM.bufferTape_unary q r hr]

/-- The table's cells, from its first cell on (its right blank and beyond included). -/
example (z : ℤ) (hz : 0 ≤ z) :
    FinTM.bufferTape (pairEncode a ((List.replicate q true ++ [false]) ++ v))
      (((a.flatMap fun b => [b, b]) ++ [false, true]).length + (q + 1 : ℕ) + z) =
      FinTM.bufferTape v z := by
  rw [pairEncode_eq, List.append_assoc ((a.flatMap fun b => [b, b])) [false, true],
    ← List.append_assoc, add_assoc, FinTM.bufferTape_append_shift _ _ (by omega),
    show ((q + 1 : ℕ) : ℤ) = (List.replicate q true ++ [false]).length by simp,
    FinTM.bufferTape_append_shift _ _ hz]

/-- A suffix of the serialization from cell `r`. -/
example (r : ℕ) (z : ℤ) (hz : 0 ≤ z) :
    FinTM.bufferTape ((pairEncode a ((List.replicate q true ++ [false]) ++ v)).drop r) z =
      FinTM.bufferTape (pairEncode a ((List.replicate q true ++ [false]) ++ v)) (r + z) :=
  FinTM.bufferTape_drop _ r hz

/-- A prefix of the serialization, read below its length. -/
example (n : ℕ) (z : ℤ) (hz : z < n) :
    FinTM.bufferTape ((pairEncode a ((List.replicate q true ++ [false]) ++ v)).take n) z =
      FinTM.bufferTape (pairEncode a ((List.replicate q true ++ [false]) ++ v)) z :=
  FinTM.bufferTape_take _ n hz

/-- Every cell past the serialization is blank. -/
example (z : ℤ) (hz : 0 ≤ z) :
    FinTM.bufferTape (pairEncode a ((List.replicate q true ++ [false]) ++ v))
      ((pairEncode a ((List.replicate q true ++ [false]) ++ v)).length + z) = none :=
  FinTM.bufferTape_eq_none_of_not_mem _ (by omega)

/-- A middle word's letters, here the separator's second cell. -/
example :
    FinTM.bufferTape (pairEncode a ((List.replicate q true ++ [false]) ++ v))
      ((a.flatMap fun b => [b, b]).length + 1) = some true := by
  rw [pairEncode_eq]
  exact FinTM.bufferTape_append_mid _ [false, true] _ 1 (by simp)

end BufferWordsConsumer

end Turing

namespace Turing
section BufferWordsRouteExtra
attribute [local irreducible] Turing.FinTM.bufferTape Turing.pairEncode

/-- Phase 5 (the walk on `sm`, `dir = neg`, at `w = m.reverse`): the cell `|w| - 1 - p` of `w`'s
buffer is the cell `p` of the reversed word's buffer, for `p ≤ |w|` (the last one both blanks). -/
example (w : List Bool) (p : ℕ) (hp : p ≤ w.length) :
    FinTM.bufferTape w ((w.length : ℤ) - 1 - p) = FinTM.bufferTape w.reverse p := by
  rcases Nat.lt_or_ge p w.length with h | h
  · rw [show (w.length : ℤ) - 1 - p = ((w.length - 1 - p : ℕ) : ℤ) by omega,
      FinTM.bufferTape_nat, FinTM.bufferTape_nat, List.getElem?_reverse h]
  · obtain rfl : p = w.length := by omega
    rw [show (w.length : ℤ) - 1 - (w.length : ℕ) = -1 by omega, FinTM.bufferTape_left,
      FinTM.bufferTape_eq_none_of_not_mem _ (by rw [List.length_reverse]; omega)]

/-- Phases 7-8 (skip back, then clear, at `u = s.take (h + q)`): with the cell `h + q` blanked by
the unary split, the window `-1 ≤ p ≤ h + q` reads the prefix's buffer. -/
example (s : List Bool) (n : ℕ) (hn : n < s.length) (p : ℤ) (h₁ : -1 ≤ p) (h₂ : p ≤ (n : ℤ)) :
    Function.update (FinTM.bufferTape s) (n : ℤ) none p = FinTM.bufferTape (s.take n) p := by
  rcases lt_or_eq_of_le h₂ with h | rfl
  · rw [Function.update_of_ne (ne_of_lt h), FinTM.bufferTape_take _ _ h]
  · rw [Function.update_self, FinTM.bufferTape_eq_none_of_not_mem _
      (by rw [List.length_take, min_eq_left (le_of_lt hn)]; omega)]

/-- Phase 6's endpoint on the blank table tape (head at `0`): the transfer's window form is the
buffer of `v`. -/
example (v : List Bool) :
    (fun q : ℤ => if (0 : ℤ) ≤ q ∧ q < 0 + (v.length : ℤ) then FinTM.bufferTape v (q - 0)
      else none) = FinTM.bufferTape v := by
  funext q
  split_ifs with h
  · rw [sub_zero]
  · exact (FinTM.bufferTape_eq_none_of_not_mem v (by omega)).symm

end BufferWordsRouteExtra
end Turing
