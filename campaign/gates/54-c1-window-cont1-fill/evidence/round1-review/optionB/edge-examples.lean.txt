/-- `d = 0`: an empty window; `n = 0`, the exhaustion branch with no round. -/
example {z : List Bool} (c : Cfg 1 Bool Bool z) (p : ℕ) (hc : c.state = some false)
    (hi : c.inputPos.val = p + 1) (hp : p ≤ z.length) (w : List Bool) (ct : ℤ → Option Bool)
    (cp : ℤ) (hw : ∀ q : ℤ, -1 ≤ q → q ≤ (w.length : ℤ) → ct (cp + q) = FinTM.bufferTape w q)
    (hd0 : Nat.ofDigits 2 (w.map Bool.toNat) = 0) :
    ∃ T ≤ 6 * (0 + 1) + 2 * w.length, ∃ v : List Bool, v.length = w.length ∧ True :=
  have ⟨T, hT, v, hv, _, _⟩ := iw_local c p hc hi hp w ct cp hw 0 (by rw [hd0]; simp)
  ⟨T, hT, v, hv, trivial⟩

/-- `p = |z|`: the head is on the right end marker; `n = 0`, and with `d ≥ 1` the escape
branch at round `1`. -/
example {z : List Bool} (c : Cfg 1 Bool Bool z) (hc : c.state = some false)
    (hi : c.inputPos.val = z.length + 1) (w : List Bool) (ct : ℤ → Option Bool) (cp : ℤ)
    (hw : ∀ q : ℤ, -1 ≤ q → q ≤ (w.length : ℤ) → ct (cp + q) = FinTM.bufferTape w q) :
    ∃ T ≤ 6 * (0 + 1) + 2 * w.length, ∃ v : List Bool, v.length = w.length ∧ True :=
  have ⟨T, hT, v, hv, _, _⟩ := iw_local c z.length hc hi le_rfl w ct cp hw 0 (by simp)
  ⟨T, hT, v, hv, trivial⟩

/-- `|y| < d`: the input ends inside the window; `n = |z| - p`. -/
example {z : List Bool} (c : Cfg 1 Bool Bool z) (p : ℕ) (hc : c.state = some false)
    (hi : c.inputPos.val = p + 1) (hp : p ≤ z.length) (w : List Bool) (ct : ℤ → Option Bool)
    (cp : ℤ) (hw : ∀ q : ℤ, -1 ≤ q → q ≤ (w.length : ℤ) → ct (cp + q) = FinTM.bufferTape w q)
    (hlt : z.length - p < Nat.ofDigits 2 (w.map Bool.toNat)) :
    ∃ T ≤ 6 * (z.length - p + 1) + 2 * w.length, ∃ v : List Bool, v.length = w.length ∧ True :=
  have ⟨T, hT, v, hv, _, _⟩ :=
    iw_local c p hc hi hp w ct cp hw (z.length - p) (Nat.min_eq_right hlt.le).symm
  ⟨T, hT, v, hv, trivial⟩

/-- A counter word with leading zero bits: `[true, false, false]` has value `1` and width `3`;
the final word keeps width `3`. -/
example {z : List Bool} (c : Cfg 1 Bool Bool z) (p : ℕ) (hc : c.state = some false)
    (hi : c.inputPos.val = p + 1) (hp : p < z.length) (ct : ℤ → Option Bool) (cp : ℤ)
    (hw : ∀ q : ℤ, -1 ≤ q → q ≤ (3 : ℤ) →
      ct (cp + q) = FinTM.bufferTape [true, false, false] q) :
    ∃ T ≤ 6 * (1 + 1) + 2 * 3, ∃ v : List Bool, v.length = 3 ∧ True :=
  have ⟨T, hT, v, hv, _, _⟩ := iw_local c p hc hi hp.le [true, false, false] ct cp hw 1
    (by have : Nat.ofDigits 2 ([true, false, false].map Bool.toNat) = 1 := by decide
        rw [this]; omega)
  ⟨T, hT, v, hv, trivial⟩
