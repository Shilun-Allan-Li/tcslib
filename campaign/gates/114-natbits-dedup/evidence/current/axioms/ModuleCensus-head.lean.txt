open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let wanted : List Name := [`TCSlib.Complexity.CircuitComplexity.Meyer, `TCSlib.Complexity.CircuitComplexity.MeyerTableau, `TCSlib.Complexity.ClassNP.ExhaustiveReplay, `TCSlib.Complexity.CookLevin.Hardness, `TCSlib.Complexity.Diagonalization.NTimeDiagonalizer, `TCSlib.Complexity.Randomized.QuasiPolyHierarchy, `TCSlib.Complexity.Randomized.SpaceAmplifyPoly, `TCSlib.Complexity.Randomized.UniformSample, `TCSlib.Complexity.SpaceComplexity.Codec.Word, `TCSlib.Complexity.SpaceComplexity.Constructible, `TCSlib.Complexity.SpaceComplexity.Logspace.Mult, `TCSlib.Complexity.SpaceComplexity.Logspace.QueryReshape, `TCSlib.Complexity.SpaceComplexity.Machines.Bin, `TCSlib.Complexity.SpaceComplexity.Machines.Frag, `TCSlib.Complexity.SpaceComplexity.SavitchARM.Deepen, `TCSlib.Complexity.TuringMachine, `TCSlib.Complexity.TuringMachine.Build.Catalog, `TCSlib.Complexity.TuringMachine.Build.CfgRecord, `TCSlib.Complexity.TuringMachine.Build.InputLength, `TCSlib.Complexity.TuringMachine.Build.RecordTable, `TCSlib.Complexity.TuringMachine.Encoding, `TCSlib.Complexity.TuringMachine.NatBits]
  for mod in wanted do
    unless (env.getModuleIdx? mod).isSome do throwError "missing imported module {mod}"
  let names := env.constants.toList.filterMap fun (n, ci) =>
    match env.getModuleIdxFor? n with
    | some i =>
      let mod := env.header.moduleNames[i.toNat]!
      if wanted.contains mod && !isPrivateName n then some (n, ci, mod) else none
    | none => none
  for (n, ci, mod) in names.toArray.qsort (fun a b => a.1.toString < b.1.toString) do
    let kind := match ci with
      | .thmInfo _ => "theorem" | .defnInfo _ => "definition"
      | .opaqueInfo _ => "opaque" | .axiomInfo _ => "axiom"
      | .ctorInfo _ => "constructor" | .recInfo _ => "recursor"
      | .inductInfo _ => "inductive" | .quotInfo _ => "quotient"
    logInfo m!"CONSTANT {n} module={mod} kind={kind} internal={n.isInternalDetail}"
