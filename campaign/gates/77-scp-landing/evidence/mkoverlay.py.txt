"""Build the SCP landing overlays: A1's ten lemmas (their exact added text at A1's head) declared
before Savitch.lean's landing text, imports merged. Modes: plain (the file as it lands) and sealed
(the SCP declarations inside a section sealing the 12 TCSlib definitions, with one rfl control per
sealed definition at the section's top)."""
import subprocess, sys, re
wt, base, a1, mode, out = sys.argv[1:6]
FILES = {
 'TCSlib/Complexity/ClassP/TimeConstructible.lean': ('Complexity', ''),
 'TCSlib/Complexity/SpaceComplexity/Constructible.lean': ('Complexity', ''),
 'TCSlib/Complexity/SpaceComplexity/Basic.lean': (None, ''),
 'TCSlib/Complexity/ClassNP/Transducer.lean': ('Complexity', 'variable {σ : Type}'),
}
NAMES = {'timeConstructible_iff','spaceConstructible_iff','computesInSpace_iff','decidesInSpace_iff',
 'mem_SPACE_iff','logSpace_eq','LOGSPACE_eq','transduce_nil','transduce_cons','transduce_of_copy'}
FINTM = {'computesInSpace_iff','decidesInSpace_iff'}
def git(*a): return subprocess.run(['git','-C',wt]+list(a),capture_output=True,text=True,check=True).stdout
blocks = {}
for f in FILES:
    d = git('diff','-U0',base,a1,'--',f)
    cur = []; groups = []
    for line in d.splitlines():
        if line.startswith('@@'):
            if cur: groups.append(cur); cur = []
        elif line.startswith('+') and not line.startswith('+++'):
            cur.append(line[1:])
        elif line.startswith('-') and not line.startswith('---'):
            raise SystemExit('A1 removes a line in '+f)
    if cur: groups.append(cur)
    for g in groups:
        txt = '\n'.join(g)
        names = re.findall(r'^theorem (\S+)', txt, re.M)
        if not names:
            assert all(l.startswith(('* `','  ')) for l in g), ('unexpected non-decl hunk', f, g)
            continue
        # a hunk may hold several declarations; split at docstring starts
        parts = re.split(r'\n(?=/-- )', txt.strip('\n'))
        for p in parts:
            n = re.findall(r'^theorem (\S+)', p, re.M); assert len(n) == 1, p
            blocks[n[0]] = (f, p.strip('\n'))
assert set(blocks) == NAMES, set(blocks) ^ NAMES
sav = open(wt + '/TCSlib/Complexity/SpaceComplexity/Savitch.lean').read().split('\n')
nimp = max(i for i, l in enumerate(sav) if l.startswith('import ')) + 1
imports, body = sav[:nimp], sav[nimp:]
o = imports + ['', '/-! ### A1 (#164), its ten lemmas verbatim from its head (absent from the olean tree) -/', '',
               'namespace Turing.FinTM', '']
for n in ['computesInSpace_iff','decidesInSpace_iff']:
    o += blocks[n][1].split('\n') + ['']
o += ['end Turing.FinTM', '', 'namespace Complexity', '', 'open Turing', '']
for n in ['timeConstructible_iff','spaceConstructible_iff','mem_SPACE_iff','logSpace_eq','LOGSPACE_eq']:
    o += blocks[n][1].split('\n') + ['']
o += ['section', 'variable {σ : Type}', '']
for n in ['transduce_nil','transduce_cons','transduce_of_copy']:
    o += blocks[n][1].split('\n') + ['']
o += ['end', '', 'end Complexity', '', '/-! ### Savitch.lean as it lands (after its imports) -/']
off = len(o) - nimp  # overlay line = landing line + off
SEAL = ['Complexity.SpaceConstructible','Turing.FinTM.ComputesInSpace','Complexity.logSpace',
 'Turing.FinTM.ComputesFunInTime','Turing.FinTM.ComputesInTime','Turing.FinTM.bufferedCompTM',
 'Turing.FinTM.polyUnaryTM','Turing.FinTM.prependTM','Complexity.transducerTM','Complexity.transduce',
 'Turing.MultiTapeTM.spaceUsed','Turing.MultiTapeTM.initCfg']
CONTROLS = open(sys.argv[6]).read().rstrip('\n').split('\n') if mode == 'sealed' else []
if mode == 'sealed':
    i0 = next(i for i, l in enumerate(body) if l.startswith('/-- The tail transducer\'s transition'))
    i1 = next(i for i, l in enumerate(body) if l.startswith('/-- **Savitch\'s theorem**'))
    seal = ['section Sealed', '', 'attribute [local irreducible] ' + ' '.join(SEAL), '',
            '/-! The controls: each closes by `rfl` unsealed and must fail here. -/'] + CONTROLS + ['']
    body = body[:i0] + seal + body[i0:i1] + ['end Sealed', ''] + body[i1:]
o += body
open(out, 'w').write('\n'.join(o))
print('offset', off, 'lines', len(o))
