out = src
for cid, old, new in R:
    n = out.count(old)
    if n != 1:
        sys.exit(f"{cid}: old text occurs {n} times")
    line = src[: src.index(old)].count("\n") + 1
    print(f"{cid}\tsrc line {line}")
    out = out.replace(old, new)

for i, l in enumerate(out.splitlines(), 1):
    if len(l) > 100:
        print(f"WARNING line {i} has {len(l)} chars")
open(DST, "w", encoding="utf-8").write(out)
