-- Cross-check of the generated names: every public, non-internal constant whose name begins
-- with the prefix, in name order. Beside the 23 names above that carry these prefixes, the sweep
-- prints the base's `zoneRebalancePost_state_done`, `zoneRebalancePost_frames`,
-- `zoneExtendPost_frames` and `zoneRebalanceData_of_short`; auxiliaries (`proof_N`, `match_N`,
-- `eq_N`) are skipped. A `simps` name above that the sweep does not print is a misnamed entry.
open Lean Elab Command in
elab "#ax_prefix " s:str : command => do
  let env ← getEnv
  let pre := s.getString
  let names := env.constants.map₁.toList.filterMap fun (n, _) =>
    if pre.isPrefixOf n.toString && !n.isInternalDetail && !isPrivateName n then some n else none
  for n in names.toArray.qsort (fun a b => a.toString < b.toString) do
    let axs ← collectAxioms n
    logInfo m!"SWEEP '{n}' depends on axioms: {axs.qsort (fun a b => a.toString < b.toString)}"
