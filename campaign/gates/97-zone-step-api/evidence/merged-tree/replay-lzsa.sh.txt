#!/usr/bin/env bash
# Replay lzsa (#197 (G5's consumer API) merged with the campaign after #190; branch lean/zone-step-api, stacked on the campaign after #190): its downstream closure (111 modules),
# fresh .oleans into oleans-lzsa, a copy of oleans-lcf3, via scripts/lean_check_tree.sh. The root TCSlib
# module is not listed (replay rule). The precheck pins the changed files' blobs.
S=/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad
WT="$S/wt-zsa"
LIST="/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/fills/replay/down-lzsa.txt"
LOG="$S/fills/replay/replay-lzsa.log"
PINS="TCSlib/Complexity/TuringMachine.lean:8dc81561b608e7e2c15bb8da7d669485a6d5ae7e
TCSlib/Complexity/TuringMachine/Build/Contract/Basic.lean:d7d40bdd2fa62dab0c4eca7ae02df0f8170f96f4
TCSlib/Complexity/TuringMachine/Build/ZoneRebalance.lean:ca48d3af975530679c669bc48154c7417ae014df
TCSlib/Complexity/TuringMachine/Build/ZoneStep.lean:9ae0da4c94f2db4bbe17f5912d023620f9989c4f"
[ -d "$WT/.lake/packages/mathlib/.lake/build/lib/lean/Mathlib" ] || { echo "precheck: $WT/.lake/packages has no Mathlib build; nothing copied" >&2; exit 2; }
git -C "$WT" fetch -q origin 2>/dev/null
[ "$(git -C "$WT" rev-parse HEAD)" = "$(git -C "$WT" rev-parse origin/lean/zone-step-api 2>/dev/null)" ] || { echo "precheck: $WT is not at the pushed PR head; nothing copied" >&2; exit 2; }
[ -z "$(git -C "$WT" status --porcelain --untracked-files=no)" ] || { echo "precheck: $WT has local changes; nothing copied" >&2; exit 2; }
git -C "$WT" merge-base --is-ancestor 683e2adf571e8b6a5c62280d4a3739ec89687546 HEAD || { echo "precheck: the head must contain the campaign after #190; nothing copied" >&2; exit 2; }
[ "$(git -C "$WT" diff --name-only 683e2adf571e8b6a5c62280d4a3739ec89687546 HEAD -- '*.lean' | sort)" = "$(printf '%s\n' "$PINS" | cut -d: -f1 | sort)" ] || { echo "precheck: the Lean diff from the campaign after #190 must be exactly the pinned files; nothing copied" >&2; exit 2; }
while IFS=: read -r f b; do
  [ "$(git -C "$WT" rev-parse "HEAD:$f")" = "$b" ] || { echo "precheck: $f is not the checked blob; nothing copied" >&2; exit 2; }
done <<< "$PINS"
tail -1 "/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/fills/replay/replay-lcf3.log" 2>/dev/null | grep -qx "# modules=[0-9]* failed=0" || { echo "precheck: the previous replay must have passed; nothing copied" >&2; exit 2; }
[ "$(wc -l < "$LIST" | tr -d ' ')" = "111" ] || { echo "precheck: the replay list must have 111 modules; nothing copied" >&2; exit 2; }
grep -qx "TCSlib" "$LIST" && { echo "precheck: the replay list must not contain the root TCSlib; nothing copied" >&2; exit 2; }
[ -d "$S/oleans-lzsa" ] && { echo "precheck: $S/oleans-lzsa already exists; remove it for a fresh copy" >&2; exit 2; }
[ -d "$S/oleans-lcf3" ] || { echo "precheck: oleans-lcf3 is missing; nothing copied" >&2; exit 2; }
cp -R "$S/oleans-lcf3" "$S/oleans-lzsa" || exit 1
cd "$WT" || exit 1
{ echo "# replay-lzsa at $(git rev-parse --short HEAD): #197 (G5's consumer API) merged with the campaign after #190 ($(wc -l < "$LIST" | tr -d ' ') modules), fresh .oleans into oleans-lzsa"
  n=0; f=0
  while read -r m; do
    n=$((n+1))
    out=$(TCSLIB_OLEANS=$S/oleans-lzsa nice -n 10 bash scripts/lean_check_tree.sh "$m" 2>&1); st=$?
    e=$(printf '%s\n' "$out" | grep -c "error")
    echo "$m exit=$st errors=$e sorry=$(printf '%s\n' "$out" | grep -c "declaration uses 'sorry'")"
    [ $st -ne 0 ] && { f=$((f+1)); printf '%s\n' "$out" | grep error | head -3 | sed "s#$PWD/##g"; }
  done < "$LIST"
  echo "# modules=$n failed=$f"
} > "$LOG" 2>&1
python3 -I "$S/api/staticsorry.py" "$WT" "$LOG" > "${LOG%.log}-static.txt"; sst=$?
tail -n 6 "$LOG"
tail -n 1 "${LOG%.log}-static.txt"
# the exit status: nonzero unless every listed module passed and the static check found no mismatch
tail -1 "$LOG" | grep -qx "# modules=111 failed=0" || { echo "replay: a module failed, or the count is not 111" >&2; exit 1; }
[ $sst -eq 0 ] && tail -1 "${LOG%.log}-static.txt" | grep -q "mismatches 0," || { echo "replay: the static sorry check failed" >&2; exit 1; }
exit 0
