import TCSlib.Complexity.SpaceComplexity.Codec.Deciders

/-! The non-private constants of `TCSlib.Complexity.SpaceComplexity.Codec.Deciders`, compiler auxiliaries
included, sorted. A census probe, not library code. -/

open Lean Elab Command in
run_cmd do
  let env ← getEnv
  let some idx := env.getModuleIdx? `TCSlib.Complexity.SpaceComplexity.Codec.Deciders | throwError "no module"
  let mut names : Array String := #[]
  for (n, _) in env.constants.map₁.toList do
    if env.getModuleIdxFor? n == some idx && !isPrivateName n then names := names.push n.toString
  let sorted := names.qsort (· < ·)
  logInfo m!"count {sorted.size}"
  for n in sorted do logInfo m!"CONST {n}"
