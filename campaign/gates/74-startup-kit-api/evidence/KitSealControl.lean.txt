-- FinTM.bufferTape

-- StepBlock.summaryCells

-- tableBitsK
example {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1))) :
    tableBitsK tr = (List.finRange (n + 1)).flatMap fun q =>
      ([none, some false, some true] : List (Option Bool)).flatMap fun inp =>

-- unaryFin
