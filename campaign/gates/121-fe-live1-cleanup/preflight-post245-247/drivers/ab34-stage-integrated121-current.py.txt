def raw(*args, env=None):
    return subprocess.check_output(['git','-C',str(R),*args],env=env)
common=raw('rev-parse','--path-format=absolute','--git-path','objects').decode().strip()
env=os.environ | {'GIT_OBJECT_DIRECTORY':str(objects),'GIT_ALTERNATE_OBJECT_DIRECTORIES':common,

    'GIT_COMMITTER_NAME':'Codex','GIT_COMMITTER_EMAIL':'codex@openai.com',
    'GIT_AUTHOR_DATE':'2026-11-09T00:00:00Z','GIT_COMMITTER_DATE':'2026-11-09T00:00:00Z'}
def git(*args):
    return raw(*args,env=env)
def sha(payload):
    return hashlib.sha256(payload).hexdigest()
def save(name,obj):
    (e/name).write_text(json.dumps(obj,ensure_ascii=False,indent=2)+'\n')
base=q['baseline']; reader=q['head']; joint=q['clean_prospective_tree']

assert h=='9936345ae64a4380bbec3ac6f3dad192f927420a'

prospective=git('merge-tree','--write-tree',parent,h).decode().splitlines()[0]
changes=git('diff','--name-only',joint,prospective,'--','*.lean').decode().splitlines()
assert len(changes)==4

    return {f.relative_to(W).as_posix():sha(f.read_bytes()) for f in W.rglob('*.lean')}

area=R/'campaign/items/dedup/fe-live1-repairs'

    assert all(p[6:].decode().split('	',1)[0] in allowed for p in payload.splitlines() if p.startswith((b'--- a/',b'+++ b/')))
    subprocess.run(['git','apply','--check',str(patch)],cwd=W,check=True)

imports_of=runpy.run_path(str(W/'devtools/seyoon/graph.py'))['imports_of']
def dependencies(module):
    seen=set()
    pending=[module]
    while pending:
        path=pending.pop()
        if path in seen:
            continue
        seen.add(path)
        f=W/path
        if f.is_file():
            pending.extend(n.replace('.','/')+'.lean' for n in imports_of(f))
    return seen
required={p:sorted(dependencies(p)&(set(changes)|changed)) for p in sorted(changed)}
for path in sorted(allowed):

    shutil_data=compiled.read_bytes()
