import TCSlib.Complexity.TuringMachine.Finite

/-! The space API PR's radius lemma, its proof verbatim (primed), against the snapshot without
the PR. `spaceUsed_eq_sum'` stands in for the PR's `Turing.MultiTapeTM.spaceUsed_eq_sum` in
`Deterministic.lean` (the same statement, by `rfl`), and is declared before the seal; the proof
then runs with the four space and run definitions sealed. -/

namespace Turing

theorem MultiTapeTM.spaceUsed_eq_sum' {k : ℕ} {Symbol State : Type*} {input : List Symbol}
    {tm : MultiTapeTM k Symbol State} (cfg : Cfg k Symbol State input) (t : ℕ) :
    tm.spaceUsed cfg t = ∑ i, tm.spaceUsedByTape cfg t i :=
  rfl

section Sealed
attribute [local irreducible] MultiTapeTM.spaceUsed MultiTapeTM.spaceUsedByTape
  MultiTapeTM.visitedByTapeHead MultiTapeTM.runFrom

theorem MultiTapeTM.spaceUsed_le_of_abs_le' {k : ℕ} {Symbol State : Type*}
    {input : List Symbol} (tm : MultiTapeTM k Symbol State)
    (d : Cfg k Symbol State input) (t r : ℕ)
    (h : ∀ u ≤ t, ∀ j, |(tm.runFrom d u).workTapePos j - d.workTapePos j| ≤ (r : ℤ)) :
    tm.spaceUsed d t ≤ k * (2 * r + 1) := by
  rw [MultiTapeTM.spaceUsed_eq_sum']
  calc
    ∑ j, tm.spaceUsedByTape d t j ≤ ∑ _j : Fin k, (2 * r + 1) := by
      refine Finset.sum_le_sum fun j _ => ?_
      have hb := tm.spaceUsedByTape_le_card_Icc d t j (d.workTapePos j - r)
        (d.workTapePos j + r) fun u hu => by
          have he := abs_le.mp (h u hu j)
          rw [Finset.mem_Icc]
          constructor <;> omega
      omega
    _ = k * (2 * r + 1) := by simp

end Sealed

#print axioms MultiTapeTM.spaceUsed_le_of_abs_le'

end Turing
