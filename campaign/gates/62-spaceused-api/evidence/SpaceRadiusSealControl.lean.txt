import TCSlib.Complexity.TuringMachine.Finite

/-! Controls for `SpaceRadiusSeal`'s seal: one `rfl` goal per sealed definition. Unsealed (first
section) all four close; sealed (second section, the same seal) all four are expected to FAIL. -/

namespace Turing

section Unsealed

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) (t : ℕ) :
    M.spaceUsed d t = ∑ i, M.spaceUsedByTape d t i := rfl

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) (t : ℕ)
    (i : Fin k) : M.spaceUsedByTape d t i = (M.visitedByTapeHead d t i).card := rfl

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) (t : ℕ)
    (i : Fin k) : M.visitedByTapeHead d t i =
      (Finset.range (t + 1)).image fun t' => (M.runFrom d t').workTapePos i := rfl

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) :
    M.runFrom d 0 = d := rfl

end Unsealed

section Sealed
attribute [local irreducible] MultiTapeTM.spaceUsed MultiTapeTM.spaceUsedByTape
  MultiTapeTM.visitedByTapeHead MultiTapeTM.runFrom

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) (t : ℕ) :
    M.spaceUsed d t = ∑ i, M.spaceUsedByTape d t i := rfl

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) (t : ℕ)
    (i : Fin k) : M.spaceUsedByTape d t i = (M.visitedByTapeHead d t i).card := rfl

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) (t : ℕ)
    (i : Fin k) : M.visitedByTapeHead d t i =
      (Finset.range (t + 1)).image fun t' => (M.runFrom d t').workTapePos i := rfl

example {k : ℕ} {S : Type} {x : List Bool} (M : MultiTapeTM k Bool S) (d : Cfg k Bool S x) :
    M.runFrom d 0 = d := rfl

end Sealed

end Turing
