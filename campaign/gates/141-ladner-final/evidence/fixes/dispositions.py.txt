"""Gate 138: classify every new copy pair of the narrow and wide screens (run after both)."""
import json, re, sys
from pathlib import Path
G = Path('/home/user/tcslib-scratch/g141')
DEFS = {'ladnerH', 'ladner_stage', 'ladner_candidate', 'ladner_wordVerdict', 'ladner_ctxGuard',
        'ladner_validQuery', 'ladner_testStep', 'ladner_countStep', 'ladner_queryCode',
        'ladner_queryInput', 'ladner_queryDeadline', 'ladner_tableSyntax', 'ladner_padSyntax',
        'ladner_selectStep', 'ladner_memoStep', 'ladner_selectInit', 'ladner_memoInit', 'ladner_selectCtx',
        'ladner_shrinkStep', 'ladner_verifier', 'ladner_smallSAT', 'ladner_ShrinkInv', 'ladner_shrinkMeasure'}
STEPS = {'ladner_testStep_FP', 'ladner_countStep_FP', 'ladner_selectStep_FP', 'ladner_memoStep_FP', 'ladner_shrinkStep_FP', 'polyTimeComputable_revStep',
         'polyTimeComputable_mulStep', 'polyTimeComputable_sqStep'}
HELPERS = {'ladner_index_lt_loglog_FP', 'ladner_padSyntax_FP', 'ladner_lengthBits_FP',
           'ladner_bits_prepend', 'ladner_stage_length_le', 'ladner_candidate_selectCtx', 'ladner_stage_spec',
           'ladner_shrink_terminal_result', 'ladner_shrink_initial'}
ORBITS = {'ladner_test_orbit', 'ladner_count_orbit', 'iterate_mulUStep', 'ladner_select_orbit', 'ladner_memo_orbit'}
def rows(p):
    out = []
    for l in Path(p).read_text().splitlines()[1:]:
        m = re.match(r'\s+(NEW|GONE)\s+(\S+)::(\S+)\s+reproduces\s+(\S+)::(\S+): (\d+)/(\d+) = (\d+)%', l)
        out.append(dict(status=m[1], target=f'{m[2]}::{m[3]}', source=f'{m[4]}::{m[5]}',
                        chars=int(m[6]), source_chars=int(m[7]), pct=int(m[8])))
    return out
SRC = (G/'screen/HEAD/export/TCSlib/Complexity/Diagonalization/Ladner.lean').read_text()
def body(name):
    import re as _re
    m = _re.search(r'^private (?:noncomputable )?(?:theorem|def) ' + _re.escape(name) + r'\b', SRC, _re.M)
    if not m: return ''
    n = _re.search(r'^(?:/--|private |theorem |def |end )', SRC[m.end():], _re.M)
    return SRC[m.start(): m.end() + (n.start() if n else len(SRC))]
def cites(user, helper):
    """gate 138 G138-01: a helper-citation pair needs the other endpoint to name the helper."""
    return helper in body(user)
def classify(r):
    t, s = r['target'].split('::')[1], r['source'].split('::')[1]
    # gate 141 G141-01: simp-set idioms described as such, not as FP closure chains
    SIMP = {frozenset({'ladner_shrink_progress', 'ladner_padWord_membership'}):
              'One simp call over the existing in-file padding-membership leaf ladner_pad_membership with option/product/existential equations, applied to different parses.',
            frozenset({'ladner_bounded_pad_length', 'ladner_parsed_length'}):
              'The core list-length simp set (List.length_append, List.length_cons, List.length_replicate) applied to different words.'}
    if r['status'] == 'NEW' and frozenset({t, s}) in SIMP:
        return 'simp-idiom', SIMP[frozenset({t, s})]
    if r['status'] == 'GONE':
        return 'removed', 'Present at base, gone at head.'
    if {t, s} <= DEFS:
        return 'spec-mirror', ('Definitions restating the route\'s specification: `ladner_stage` and `ladner_candidate` '
                               'mirror `ladnerH`\'s defining selection and test, which part B must prove equal; the guards '
                               'and verdict share their canonical-index and length conditions.')
    if {t, s} <= STEPS:
        return 'step-skeleton', ('The iterator-step FP skeleton (fixed on an empty counter, otherwise `pairEncode` of '
                                 'updated slots, closed by `convert polyTimeComputable_ite … using 1`), as among Mul\'s '
                                 'own step proofs at base. The step functions differ.')
    if {t, s} <= ORBITS:
        return 'orbit-idiom', 'The `Function.iterate` orbit induction opening shared with `iterate_mulUStep`; different orbits.'
    if (t in HELPERS and cites(s, t)) or (s in HELPERS and cites(t, s)):
        return 'helper-citation', ('Text shared with a new private helper is its call and the closure tail around it: '
                                   'a citation of the helper, not a copy of its proof.')
    return 'fp-idiom', (f"A {r['source_chars']}-character polynomial-time closure chain (`comp`, `pairEncode`, "
                        '`polyTimeComputable_and`/`_ite`, `convert … using 1; funext; simp only` over published '
                        'equations) shared by proofs of different predicates.')
def load(d):
    rs = rows(G/d/'pairdiff.txt')
    for r in rs: r['class'], r['note'] = classify(r)
    return rs
first = lambda p: Path(p).read_text().splitlines()[0].strip()
out = {}
for d, key in (('screen', ''), ('screen-wide', 'wide_')):
    if not (G/d/'pairdiff.txt').exists(): continue
    rs = load(d)
    out.update({key+'base_summary': first(G/d/'BASE/screen.txt'), key+'head_summary': first(G/d/'HEAD/screen.txt'),
                key+'new': sum(r['status']=='NEW' for r in rs), key+'gone': sum(r['status']=='GONE' for r in rs),
                key+'pairs': rs})
    from collections import Counter
    print(d, Counter(r['class'] for r in rs))
out['population'] = (G/'POPULATION.txt').read_text().split()
out['wide_population'] = (G/'POPULATION-wide.txt').read_text().split()
out['round_note'] = ('Gate 141: Codex delivery verbatim; no maintainer repair. ladner_shrink_progress / ladner_padWord_membership '
  'share one simp call over the published padding-membership leaf (ladner_pad_membership with option/product/existential '
  'equations), applied to different parses; ladner_bounded_pad_length / ladner_parsed_length share the list-length simp set.')
json.dump(out, open(G/'screen/PAIR-DISPOSITIONS.json', 'w'), indent=1, ensure_ascii=False)
print(out['head_summary'])
