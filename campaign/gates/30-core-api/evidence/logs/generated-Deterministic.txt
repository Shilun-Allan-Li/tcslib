@Turing.MultiTapeTM.mem_visitedByTapeHead_iff : ∀ {k : ℕ} {State : Type u_1} {Symbol : Type u_2} {input : List Symbol}
  {tm : Turing.MultiTapeTM k Symbol State} (cfg : Turing.Cfg k Symbol State input) (t : ℕ) (i : Fin k) (p : ℤ),
  p ∈ Turing.MultiTapeTM.visitedByTapeHead cfg t i ↔ ∃ u ≤ t, (Turing.MultiTapeTM.runFrom cfg u).workTapePos i = p
@Turing.MultiTapeTM.visitedByTapeHead_add_subset : ∀ {m : ℕ} {S : Type} {x : List Bool}
  (M : Turing.MultiTapeTM m Bool S) (d : Turing.Cfg m Bool S x) (T₁ T₂ : ℕ) (j : Fin m),
  Turing.MultiTapeTM.visitedByTapeHead d (T₁ + T₂) j ⊆
    Turing.MultiTapeTM.visitedByTapeHead d T₁ j ∪
      Turing.MultiTapeTM.visitedByTapeHead (Turing.MultiTapeTM.runFrom d T₁) T₂ j
@Turing.MultiTapeTM.visitedByTapeHead_step_subset : ∀ {m : ℕ} {S : Type} {x : List Bool}
  (M : Turing.MultiTapeTM m Bool S) (c : Turing.Cfg m Bool S x) (t : ℕ) (j : Fin m),
  insert (c.workTapePos j) (Turing.MultiTapeTM.visitedByTapeHead (Turing.MultiTapeTM.step c) t j) ⊆
    Turing.MultiTapeTM.visitedByTapeHead c (t + 1) j
