import sys, pathlib
root = pathlib.Path(sys.argv[1]) / "TCSlib/Complexity/TuringMachine"
def rep(rel, old, new):
    p = root / rel
    t = p.read_text()
    n = t.count(old)
    assert n == 1, (rel, n, old[:70])
    p.write_text(t.replace(old, new))
    print("ok", rel, old[:50].replace("\n", " "))
