def edit(path, pairs):
    t = open(path, encoding='utf-8').read()
    for anchor, add, where in pairs:
        assert t.count(anchor) == 1, (path, anchor[:70], t.count(anchor))
        t = t.replace(anchor, anchor + add if where == 'after' else add + anchor)
    open(path, 'w', encoding='utf-8').write(t)
    print('edited', path.split('TCSlib/')[1], len(pairs))
