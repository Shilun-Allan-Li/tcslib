#!/usr/bin/env bash
# Replay ldwa (#196 (the deepening's API) merged with the campaign after #190; branch lean/deepen-window-api, stacked on the campaign after #190): its downstream closure (73 modules),
# fresh .oleans into oleans-ldwa, a copy of oleans-lcf3, via scripts/lean_check_tree.sh. The root TCSlib
# module is not listed (replay rule). The precheck pins the changed files' blobs.
S=/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad
WT="$S/wt-dwa"
LIST="/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/fills/replay/down-ldwa.txt"
LOG="$S/fills/replay/replay-ldwa.log"
PINS="TCSlib/Complexity/SpaceComplexity.lean:fbfdcd9a90a210e0dc7d02d2acc9c87d6a2f3b59
TCSlib/Complexity/SpaceComplexity/Codec/Bridge.lean:6c3353684eac0a6cd5245f70634538398f08eeb1
TCSlib/Complexity/SpaceComplexity/ExpBFS/Radius.lean:0594fa007144a93949834ff083926aaf3d05fafa
TCSlib/Complexity/SpaceComplexity/Logspace/PathWalk.lean:5591f207cff8bcc51a905532086daed72b5efab3
TCSlib/Complexity/SpaceComplexity/SavitchARM/Reach.lean:43e9a4a4c5daf344d859d98c0f6090b803593f77
TCSlib/Complexity/SpaceComplexity/SavitchARM/Recursion.lean:274fdf6a8c38aa4369552c84e0ce8bbbddf734ac
TCSlib/Complexity/SpaceComplexity/SavitchARM/Search.lean:1edc38c784a10696de104c9abf2387ee2cc96e9c
TCSlib/Complexity/SpaceComplexity/SavitchARM/Setup.lean:60b3453070d7fc36a3059c5bd460abcb9df78c1a
TCSlib/Complexity/SpaceComplexity/SavitchARM/Window.lean:dc16e3f1eeee0270be2b7dc506a2d178bb8ad1bb
TCSlib/Complexity/TuringMachine/Build/CfgRecord.lean:adae6bf3ba45d5929164e1aac9f9711214a21338"
[ -d "$WT/.lake/packages/mathlib/.lake/build/lib/lean/Mathlib" ] || { echo "precheck: $WT/.lake/packages has no Mathlib build; nothing copied" >&2; exit 2; }
git -C "$WT" fetch -q origin 2>/dev/null
[ "$(git -C "$WT" rev-parse HEAD)" = "$(git -C "$WT" rev-parse origin/lean/deepen-window-api 2>/dev/null)" ] || { echo "precheck: $WT is not at the pushed PR head; nothing copied" >&2; exit 2; }
[ -z "$(git -C "$WT" status --porcelain --untracked-files=no)" ] || { echo "precheck: $WT has local changes; nothing copied" >&2; exit 2; }
git -C "$WT" merge-base --is-ancestor 4ef99963b8839877c843adc8a8b09da64ef1d8b6 HEAD || { echo "precheck: the head must contain the campaign after #190; nothing copied" >&2; exit 2; }
[ "$(git -C "$WT" diff --name-only 4ef99963b8839877c843adc8a8b09da64ef1d8b6 HEAD -- '*.lean' | sort)" = "$(printf '%s\n' "$PINS" | cut -d: -f1 | sort)" ] || { echo "precheck: the Lean diff from the campaign after #190 must be exactly the pinned files; nothing copied" >&2; exit 2; }
while IFS=: read -r f b; do
  [ "$(git -C "$WT" rev-parse "HEAD:$f")" = "$b" ] || { echo "precheck: $f is not the checked blob; nothing copied" >&2; exit 2; }
done <<< "$PINS"
tail -1 "/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/fills/replay/replay-lcf3.log" 2>/dev/null | grep -qx "# modules=[0-9]* failed=0" || { echo "precheck: the previous replay must have passed; nothing copied" >&2; exit 2; }
[ "$(wc -l < "$LIST" | tr -d ' ')" = "73" ] || { echo "precheck: the replay list must have 73 modules; nothing copied" >&2; exit 2; }
grep -qx "TCSlib" "$LIST" && { echo "precheck: the replay list must not contain the root TCSlib; nothing copied" >&2; exit 2; }
[ -d "$S/oleans-ldwa" ] && { echo "precheck: $S/oleans-ldwa already exists; remove it for a fresh copy" >&2; exit 2; }
[ -d "$S/oleans-lcf3" ] || { echo "precheck: oleans-lcf3 is missing; nothing copied" >&2; exit 2; }
cp -R "$S/oleans-lcf3" "$S/oleans-ldwa" || exit 1
cd "$WT" || exit 1
{ echo "# replay-ldwa at $(git rev-parse --short HEAD): #196 (the deepening's API) merged with the campaign after #190 ($(wc -l < "$LIST" | tr -d ' ') modules), fresh .oleans into oleans-ldwa"
  n=0; f=0
  while read -r m; do
    n=$((n+1))
    out=$(TCSLIB_OLEANS=$S/oleans-ldwa nice -n 10 bash scripts/lean_check_tree.sh "$m" 2>&1); st=$?
    e=$(printf '%s\n' "$out" | grep -c "error")
    echo "$m exit=$st errors=$e sorry=$(printf '%s\n' "$out" | grep -c "declaration uses 'sorry'")"
    [ $st -ne 0 ] && { f=$((f+1)); printf '%s\n' "$out" | grep error | head -3 | sed "s#$PWD/##g"; }
  done < "$LIST"
  echo "# modules=$n failed=$f"
} > "$LOG" 2>&1
python3 -I "$S/api/staticsorry.py" "$WT" "$LOG" > "${LOG%.log}-static.txt"; sst=$?
tail -n 6 "$LOG"
tail -n 1 "${LOG%.log}-static.txt"
# the exit status: nonzero unless every listed module passed and the static check found no mismatch
tail -1 "$LOG" | grep -qx "# modules=73 failed=0" || { echo "replay: a module failed, or the count is not 73" >&2; exit 1; }
[ $sst -eq 0 ] && tail -1 "${LOG%.log}-static.txt" | grep -q "mismatches 0," || { echo "replay: the static sorry check failed" >&2; exit 1; }
exit 0
