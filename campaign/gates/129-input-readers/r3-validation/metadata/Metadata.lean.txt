import TCSlib.Complexity.TuringMachine.Build.Scan
set_option maxHeartbeats 0
set_option pp.universes true
open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  let simp ← liftTermElabM getSimpTheorems
  let homeOf := fun (n : Name) => match env.getModuleIdxFor? n with
    | some i => env.header.moduleNames[i.toNat]!
    | none => Name.anonymous
  let kind := fun (ci : ConstantInfo) => match ci with
    | .axiomInfo _ => "axiom" | .defnInfo _ => "definition" | .thmInfo _ => "theorem"
    | .opaqueInfo _ => "opaque" | .quotInfo _ => "quotient" | .inductInfo _ => "inductive"
    | .ctorInfo _ => "constructor" | .recInfo _ => "recursor"
  let targets : List (Name × Name) := [(`Turing.catalog_apply_record, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.catalog_clear_trace, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.copyMapTM_step_eq, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.catalog_transferMap_trace, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.catalog_compare_trace, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.catalog_increment_trace, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.FinTM.catalogPrefixTM_emit, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPrefixTM_copy, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.scanCopy_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.scanCopy_finish, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.scanCopy_suffix, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_first, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_block, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_rewind, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_replay_finish, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_suffix, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_finish, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extract_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.pairExtractTM_computes, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_emit, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_rewind, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_loop, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_copy, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_setup, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_start, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPoly_unary_computes, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.lenSuffix_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.lenParse_block, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.lenParse_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.rawStrip_copy, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.rawStrip_rewind, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.rawStrip_finish, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.rawStrip_erase, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.anyTrue_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitRestore_scan, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitRestore_append, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitRestore_rewind, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitCount_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitPoly_loop_end, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitPrepare_scan, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitPrepare_extra, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitPrepare_rewind, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.splitEmit_double, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_erase_back, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_erase_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_prepare_candidate, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_prepare_suffix, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_prepare_rewind_bank, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_prepare_run, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.step_of_tapeAction_write, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.step_of_inputMoveAction, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.walkTM_run_ofCfg, `TCSlib.Complexity.TuringMachine.Build.Catalog), (`Turing.inputMoveAction_eq, (`Turing.FinTM.inputSymbol_at, `TCSlib.Complexity.TuringMachine.Simulation), (`Turing.FinTM.rewind_scan_heads, `TCSlib.Complexity.TuringMachine.Simulation), (`Turing.FinTM.bufferedSecondCfg_step, `TCSlib.Complexity.TuringMachine.Simulation), (`Turing.FinTM.incFixedTM_computes, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.catalogPrefixTM_halt, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.scanCfg_read, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.extractCfg_read, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.emitterP2_control, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.FinTM.pairExtractTM_fst_run, `TCSlib.Complexity.TuringMachine.Build.Primitives), (`Turing.scan_copy_trace, `TCSlib.Complexity.TuringMachine.Build.Scan), (`Turing.inputFoldTM_tr_bwd, `TCSlib.Complexity.TuringMachine.Build.Scan), (`Turing.scan_fold_trace, `TCSlib.Complexity.TuringMachine.Build.Scan)]
  let table := env.constants.toList
  let roots := table.filterMap fun (n, _) =>
    if targets.contains ((privateToUserName? n).getD n, homeOf n) then some n else none
  unless roots.length == targets.length do
    throwError "selected count {roots.length} for {targets.length}"
  for root in roots do
    let some ci := env.find? root | throwError "missing selected theorem {root}"
    let nameHits := table.filter fun (n, _) => n == root
    unless nameHits.length == 1 do throwError "selected theorem name count {nameHits.length}"
    let owner := homeOf root
    let axs ← liftCoreM <| collectAxioms root
    let pretty ← liftTermElabM <| ppExpr ci.type
    let ip ← liftCoreM <| getInstancePriority? root
    let propType ← liftTermElabM <| Meta.isProp ci.type
    let valueText := if root == `Turing.Action.apply_workTapes then (ci.value?.map reprStr).getD "" else "selected proof rendering omitted; full traversal retained"
    let canonicalLevels := (List.range ci.levelParams.length).map fun i =>
      Level.param (Name.str Name.anonymous s!"canonicalUniverse{i}")
    let canonicalType := ci.type.instantiateLevelParams ci.levelParams canonicalLevels
    let canonicalValue := if root == `Turing.Action.apply_workTapes then
      (ci.value?.map fun e => reprStr (e.instantiateLevelParams ci.levelParams canonicalLevels)).getD ""
      else "selected proof rendering omitted; full traversal retained"
    logInfo m!"SELECTED|{(Json.mkObj [
      ("name",toJson root.toString),("surface",toJson (((privateToUserName? root).getD root).toString)),("owner",toJson owner.toString),
      ("nameCount",toJson nameHits.length),("kind",toJson (kind ci)),
      ("private",toJson (isPrivateName root)),("internal",toJson root.isInternalDetail),
      ("levelParams",toJson (ci.levelParams.map Name.toString)),
      ("typeExpr",toJson (reprStr ci.type)),("typePretty",toJson pretty.pretty),
      ("valueExpr",toJson valueText),("typeIsProp",toJson propType),
      ("typeExpr_canonicalLevelParameters",toJson (reprStr canonicalType)),
      ("valueExpr_canonicalLevelParameters",toJson canonicalValue),
      ("unsafe",toJson ci.isUnsafe),("partial",toJson ci.isPartial),
      ("reducibility",toJson (reprStr (getReducibilityStatusCore env root))),
      ("simp",toJson (simp.isLemma (.decl root))),("unfold",toJson (simp.toUnfold.contains root)),
      ("protected",toJson (isProtected env root)),("instancePriority",toJson ip),
      ("axioms",toJson ((axs.qsort Name.lt).toList.map Name.toString))]).compress}"
    let mut todo := ci.type.getUsedConstants.toList ++
      (ci.value?.map Expr.getUsedConstants |>.getD #[]).toList
    let mut seen : List Name := [root]
    let mut localDefs : List Name := []
    let mut foreign : List Json := []
    let mut ordinary : List Name := []
    let mut theoremStops : List Name := []
    while !todo.isEmpty do
      let n := todo.head!
      todo := todo.tail!
      if seen.contains n then continue
      seen := n :: seen
      let some info := env.find? n | continue
      todo := info.type.getUsedConstants.toList ++ todo
      let home := homeOf n
      match info with
      | .defnInfo di =>
        if home == owner then
          localDefs := n :: localDefs
          todo := di.value.getUsedConstants.toList ++ todo
        else if home.toString.startsWith "TCSlib." then
          let isAbbrev := match di.hints with | .abbrev => true | _ => false
          let regular := !isAbbrev && !(env.isProjectionFn n)
          if regular then ordinary := n :: ordinary
          foreign := Json.mkObj [
            ("name",toJson n.toString),("owner",toJson home.toString),
            ("abbrev",toJson isAbbrev),("projection",toJson (env.isProjectionFn n)),
            ("ordinary",toJson regular),("simp",toJson (simp.isLemma (.decl n))),
            ("unfold",toJson (simp.toUnfold.contains n))] :: foreign
      | .thmInfo _ => theoremStops := n :: theoremStops
      | _ => pure ()
    logInfo m!"BOUNDARY|{(Json.mkObj [
      ("root",toJson root.toString),("surface",toJson (((privateToUserName? root).getD root).toString)),("owner",toJson owner.toString),
      ("own_definition_helpers",toJson ((localDefs.toArray.qsort Name.lt).toList.map Name.toString)),
      ("foreign_definitions",Json.arr foreign.reverse.toArray),
      ("ordinary_foreign_heads",toJson ((ordinary.toArray.qsort Name.lt).toList.map Name.toString)),
      ("other_theorem_proof_boundaries",toJson ((theoremStops.toArray.qsort Name.lt).toList.map Name.toString)),
      ("foreign_definition_values_traversed",toJson false),
      ("other_theorem_values_traversed",toJson false)]).compress}"
    for n in ordinary do
      let some (.defnInfo di) := env.find? n | throwError "ordinary head is missing"
      logInfo m!"IDENTITY|{(Json.mkObj [("name",toJson n.toString),("owner",toJson (homeOf n).toString),
        ("universes",toJson (di.levelParams.map Name.toString)),("typeExpr",toJson (reprStr di.type)),
        ("valueExpr",toJson (reprStr di.value))]).compress}"
  logInfo m!"COMPLETE|{roots.length}"
