/-- A control action has the requested input movement and successor state,
with no work-tape writes or head movement and no output symbol. -/
theorem controlAction_eq {k : ℕ} {S : Type} (m : SignType) (q : Option S) :
    controlAction (k := k) m q =
      (⟨m, fun _ => (none, 0), none, q⟩ : Action k Bool S) := by
  rfl
end Turing.FinTM

/-- An input-only action has the requested input movement and live successor,
with no work-tape writes or head movement and no output symbol. -/
theorem inputMoveAction_eq {k : ℕ} {Symbol State : Type*} (inp : SignType) (q : State) :
    inputMoveAction (k := k) (Symbol := Symbol) inp q =
      (⟨inp, fun _ => (none, 0), none, some q⟩ : Action k Symbol State) := by
  rfl

/-- In the backward phase the native fold moves left across an input symbol;
at the left blank it moves right and enters the completed phase, preserving
the accumulator. The work-tape reads do not affect this transition. -/
theorem inputFoldTM_tr_bwd {Γ : Type*} {A : Type} (k : ℕ)
    (δ : A → Γ → A) (a₀ a : A) (inp : Option Γ) (work : Fin k → Option Γ) :
    (inputFoldTM k δ a₀).tr (.bwd a) inp work =
      (match inp with
       | some _ => inputMoveAction (-1) (.bwd a)
       | none => inputMoveAction 1 (.done a)) := by
  rfl
end Turing
