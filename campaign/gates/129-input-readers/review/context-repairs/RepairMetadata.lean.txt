import Lean
import TCSlib.Complexity.TuringMachine.Simulation
import TCSlib.Complexity.TuringMachine.Build.Catalog
import TCSlib.Complexity.TuringMachine.Build.Scan
set_option maxHeartbeats 0
set_option maxRecDepth 4096
open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  let st ← liftTermElabM getSimpTheorems
  let homeOf := fun (n : Name) => match env.getModuleIdxFor? n with
    | some i => env.header.moduleNames[i.toNat]!
    | none => Name.anonymous
  let scanRoot := Name.str (Name.str
    (Name.num `_private.TCSlib.Complexity.TuringMachine.Build.Scan 0) "Turing") "scan_fold_trace"
  let roots : List Name := [`Turing.FinTM.controlAction_apply,
    `Turing.step_of_tapeAction_move, `Turing.step_of_tapeAction_write,
    `Turing.step_of_inputMoveAction, `Turing.walkTM_run_ofCfg, scanRoot]
  let mut union : List Name := []
  for root in roots do
    let some ci := env.find? root | throwError "MISSING_REPAIR_ROOT|{root}"
    let owner := homeOf root
    let ax ← liftTermElabM <| collectAxioms root
    let typ ← liftTermElabM <| ppExpr ci.type
    let directType := ci.type.getUsedConstants.toList
    let directBody := (ci.value?.map Expr.getUsedConstants |>.getD #[]).toList
    let mut todo := directType ++ directBody
    let mut seen : List Name := [root]
    let mut foreign : List Json := []
    let mut helpers : List Json := []
    let mut stopped : List Json := []
    while !todo.isEmpty do
      let n := todo.head!
      todo := todo.tail!
      if seen.contains n then continue
      seen := n :: seen
      let some info := env.find? n | continue
      todo := info.type.getUsedConstants.toList ++ todo
      let home := homeOf n
      match info with
      | .defnInfo di =>
        let regular := match di.hints with | .abbrev => false | _ => true
        let ordinary := regular && !(env.isProjectionFn n)
        let entry := Json.mkObj [("name",toJson n.toString),("owner",toJson home.toString),
          ("ordinary",toJson ordinary),("projection",toJson (env.isProjectionFn n)),
          ("hints",toJson (if regular then "non-abbrev" else "abbrev")),("type_Expr",toJson (reprStr di.type)),
          ("value_Expr",toJson (reprStr di.value)),
          ("universes",toJson (di.levelParams.map Name.toString)),
          ("type_mentions",toJson (di.type.getUsedConstants.toList.map Name.toString)),
          ("simp",toJson (st.isLemma (.decl n))),("unfold",toJson (st.toUnfold.contains n))]
        if home == owner then
          helpers := entry :: helpers
          todo := di.value.getUsedConstants.toList ++ todo
        else if home.toString.startsWith "TCSlib." then
          foreign := entry :: foreign
          if ordinary && !union.contains n then union := n :: union
      | .thmInfo _ =>
        stopped := Json.mkObj [("name",toJson n.toString),("owner",toJson home.toString),
          ("same_owner",toJson (home == owner)),("proof_value_traversed",toJson false)] :: stopped
      | _ => pure ()
    logInfo m!"REPAIR_METADATA|{(Json.mkObj [("name",toJson root.toString),
      ("owner",toJson owner.toString),("type_Expr",toJson (reprStr ci.type)),
      ("universes",toJson (ci.levelParams.map Name.toString)),
      ("type_pretty",toJson typ.pretty),("axioms",toJson (ax.toList.map Name.toString)),
      ("direct_type_mentions",toJson (directType.map Name.toString)),
      ("direct_body_mentions",toJson (directBody.map Name.toString)),
      ("simp",toJson (st.isLemma (.decl root))),("unfold",toJson (st.toUnfold.contains root)),
      ("foreign_definitions",Json.arr foreign.reverse.toArray),
      ("own_definition_helpers",Json.arr helpers.reverse.toArray),
      ("theorems_stopped_at_type",Json.arr stopped.reverse.toArray),
      ("imported_or_other_theorem_proof_values_traversed",toJson false),
      ("foreign_definition_values_recorded_but_not_traversed",toJson true)]).compress}"
  logInfo m!"REPAIR_COMPLETE|{roots.length}|{union.length}"
