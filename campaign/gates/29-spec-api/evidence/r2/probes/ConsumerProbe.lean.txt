import TCSlib.Complexity.TuringMachine.Build.Contract.Seam

/-! Review probe (#112 round-2 gate, the note): import-only consumer instantiations of the
repaired iteration interfaces, as G4's hosts use them. Only published lemmas are used: the
generated `_pre`/`_time`/`_post` equations, the contracts' own fields, `Contract.seq_time`,
`Round.iterate`; no definition is unfolded. -/

open Turing

/-- **The round release, applied.** A round contract's admissible start, mapped to the fresh
release state, satisfies the released precondition; the released run then reaches the
right-tagged round endpoint at the round's own time. -/
example {k : ℕ} {Γ S : Type} {M : MultiTapeTM k Γ S} {anchor : S} {E : Set (Option S)}
    {ι : Type} {x : List Γ} (R : RoundContract M anchor E ι x) (i : ι)
    (d : Cfg k Γ (Unit ⊕ S) x) (hd : d.state = some (Sum.inl ()))
    (hp : R.pre i (d.mapState fun _ => anchor)) :
    (seamReleaseTM M anchor).runFrom d (R.time i (d.mapState fun _ => anchor)) =
      (R.post i (d.mapState fun _ => anchor)).mapState Sum.inr := by
  have h := R.release.run i d hd (by rw [RoundContract.release_pre]; exact hp)
  rwa [RoundContract.release_time, RoundContract.release_post] at h

/-- **A returning round, released and sequenced.** The composite's time is the round's time at
the anchor-mapped start, one dispatch step, and the next contract's time at the released
endpoint, read through `Contract.seq_time` and the release equations. -/
example {k : ℕ} {Γ S S₂ : Type} [DecidableEq S] {M : MultiTapeTM k Γ S}
    {M₂ : MultiTapeTM k Γ S₂} {anchor : S} {entry₂ : S₂} {X₂ : Set (Option S₂)} {ι κ : Type}
    {x : List Γ} (C : ReturnContract M anchor ι x) (C₂ : Contract M₂ entry₂ X₂ κ x)
    (link : ι → κ)
    (hlink : ∀ i d, d.state = some (Sum.inl ()) → C.release.pre i d →
      C₂.pre (link i) ((C.release.post i d).mapState fun _ => entry₂))
    (i : ι) (d : Cfg k Γ ((Unit ⊕ S) ⊕ S₂) x) :
    (Contract.seq C.release C₂ link hlink).time i d =
      C.time i ((d.mapState fun _ => (Sum.inl () : Unit ⊕ S)).mapState fun _ => anchor) + 1 +
        C₂.time (link i)
          (((C.post i ((d.mapState fun _ => (Sum.inl () : Unit ⊕ S)).mapState
            fun _ => anchor)).mapState (Sum.inr : S → Unit ⊕ S)).mapState fun _ => entry₂) := by
  rw [Contract.seq_time, ReturnContract.release_time, ReturnContract.release_post]

/-- **The index-free route.** `Round.iterate`'s segment, passed whole to `Contract.ofSegment`,
gives a contract whose time and endpoint are the orbit's, read through the generated
equations. -/
example {k : ℕ} {Γ S : Type} {M : MultiTapeTM k Γ S} {a : S} {E : Set (Option S)}
    {x : List Γ} (τ : Cfg k Γ S x → ℕ) (n : ℕ) (lo hi : Unit → Cfg k Γ S x → Fin k → ℤ)
    (ha : some a ∉ E)
    (h0 : ∀ c : Cfg k Γ S x, HeadBox (lo () c) (hi () c) c.workTapePos)
    (hround : ∀ (c : Cfg k Γ S x), c.state = some a → ∀ r < n,
      Round M a E (HeadBox (lo () c) (hi () c)) (roundOrbit M τ c r)
        (roundOrbit M τ c (r + 1)) (τ (roundOrbit M τ c r)))
    (hmem : ∀ (_ : Unit) (c : Cfg k Γ S x), c.state = some a → True →
      (roundOrbit M τ c n).state ∈ E)
    (c : Cfg k Γ S x) :
    let C : Contract M a E Unit x :=
      Contract.ofSegment (fun _ _ => True)
        (fun _ c => ∑ r ∈ Finset.range n, τ (roundOrbit M τ c r))
        (fun _ c => roundOrbit M τ c n) lo hi
        (fun _ c hc _ => Round.iterate τ c ha (h0 c) n (hround c hc)) hmem
    C.time () c = ∑ r ∈ Finset.range n, τ (roundOrbit M τ c r) ∧
      C.post () c = roundOrbit M τ c n := by
  intro C
  exact ⟨Contract.ofSegment_time _ _ _ _ _ _ _ () c, Contract.ofSegment_post _ _ _ _ _ _ _ () c⟩
