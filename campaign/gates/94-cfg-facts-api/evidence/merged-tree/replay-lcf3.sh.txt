#!/usr/bin/env bash
# Replay lcf3 (#190 (the configuration facts) merged with the campaign after #188; branch lean/cfg-facts, stacked on the campaign tip after #188): its downstream closure (86 modules),
# fresh .oleans into oleans-lcf3, a copy of oleans-led4, via scripts/lean_check_tree.sh. The root TCSlib
# module is not listed (replay rule). The precheck pins the changed files' blobs.
S=/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad
WT="$S/wt-cf"
LIST="/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/fills/replay/down-lcf3.txt"
LOG="$S/fills/replay/replay-lcf3.log"
PINS="TCSlib/Complexity/ClassNP/ExhaustiveReplay.lean:2ce883de15e716b0649093bbcaecfda69fabd7c1
TCSlib/Complexity/TuringMachine.lean:4edcf9fef98189022fa83c8b0da0bcabe9b04b84
TCSlib/Complexity/TuringMachine/Build/CfgRecord.lean:c82b5bc6588f2cfdcfeff072a5821a2ac70c90b3
TCSlib/Complexity/TuringMachine/Build/SeamLayout.lean:aa46f40096b924d308276cc02c7368b6dbda4354
TCSlib/Complexity/TuringMachine/Build/ZoneRebalance.lean:9e687b1694cb8772cc78d490d4434a3edd67f2c5
TCSlib/Complexity/TuringMachine/CfgFacts.lean:6d6b47275ac2063c371fd9c52c9eeece51ac4268"
[ -d "$WT/.lake/packages/mathlib/.lake/build/lib/lean/Mathlib" ] || { echo "precheck: $WT/.lake/packages has no Mathlib build; nothing copied" >&2; exit 2; }
git -C "$WT" fetch -q origin 2>/dev/null
[ "$(git -C "$WT" rev-parse HEAD)" = "$(git -C "$WT" rev-parse origin/lean/cfg-facts 2>/dev/null)" ] || { echo "precheck: $WT is not at the pushed PR head; nothing copied" >&2; exit 2; }
[ -z "$(git -C "$WT" status --porcelain --untracked-files=no)" ] || { echo "precheck: $WT has local changes; nothing copied" >&2; exit 2; }
git -C "$WT" merge-base --is-ancestor 90836a96a6497829a969ef0fd7c2db47adbbeec7 HEAD || { echo "precheck: the head must contain the campaign tip after #188; nothing copied" >&2; exit 2; }
[ "$(git -C "$WT" diff --name-only 90836a96a6497829a969ef0fd7c2db47adbbeec7 HEAD -- '*.lean' | sort)" = "$(printf '%s\n' "$PINS" | cut -d: -f1 | sort)" ] || { echo "precheck: the Lean diff from the campaign tip after #188 must be exactly the pinned files; nothing copied" >&2; exit 2; }
while IFS=: read -r f b; do
  [ "$(git -C "$WT" rev-parse "HEAD:$f")" = "$b" ] || { echo "precheck: $f is not the checked blob; nothing copied" >&2; exit 2; }
done <<< "$PINS"
tail -1 "/private/tmp/claude-501/-Users-seyoonr-phd-experiments-tcslib/8f0076bc-454a-4666-b8cf-f4c7ed13c549/scratchpad/fills/replay/replay-led4.log" 2>/dev/null | grep -qx "# modules=[0-9]* failed=0" || { echo "precheck: the previous replay must have passed; nothing copied" >&2; exit 2; }
[ "$(wc -l < "$LIST" | tr -d ' ')" = "86" ] || { echo "precheck: the replay list must have 86 modules; nothing copied" >&2; exit 2; }
grep -qx "TCSlib" "$LIST" && { echo "precheck: the replay list must not contain the root TCSlib; nothing copied" >&2; exit 2; }
[ -d "$S/oleans-lcf3" ] && { echo "precheck: $S/oleans-lcf3 already exists; remove it for a fresh copy" >&2; exit 2; }
[ -d "$S/oleans-led4" ] || { echo "precheck: oleans-led4 is missing; nothing copied" >&2; exit 2; }
cp -R "$S/oleans-led4" "$S/oleans-lcf3" || exit 1
cd "$WT" || exit 1
{ echo "# replay-lcf3 at $(git rev-parse --short HEAD): #190 (the configuration facts) merged with the campaign after #188 ($(wc -l < "$LIST" | tr -d ' ') modules), fresh .oleans into oleans-lcf3"
  n=0; f=0
  while read -r m; do
    n=$((n+1))
    out=$(TCSLIB_OLEANS=$S/oleans-lcf3 nice -n 10 bash scripts/lean_check_tree.sh "$m" 2>&1); st=$?
    e=$(printf '%s\n' "$out" | grep -c "error")
    echo "$m exit=$st errors=$e sorry=$(printf '%s\n' "$out" | grep -c "declaration uses 'sorry'")"
    [ $st -ne 0 ] && { f=$((f+1)); printf '%s\n' "$out" | grep error | head -3 | sed "s#$PWD/##g"; }
  done < "$LIST"
  echo "# modules=$n failed=$f"
} > "$LOG" 2>&1
python3 -I "$S/api/staticsorry.py" "$WT" "$LOG" > "${LOG%.log}-static.txt"; sst=$?
tail -n 6 "$LOG"
tail -n 1 "${LOG%.log}-static.txt"
# the exit status: nonzero unless every listed module passed and the static check found no mismatch
tail -1 "$LOG" | grep -qx "# modules=86 failed=0" || { echo "replay: a module failed, or the count is not 86" >&2; exit 1; }
[ $sst -eq 0 ] && tail -1 "${LOG%.log}-static.txt" | grep -q "mismatches 0," || { echo "replay: the static sorry check failed" >&2; exit 1; }
exit 0
