end

section
-- CONTROL initCfg_equation
attribute [local irreducible] MultiTapeTM.initCfg

example {n : ℕ} {P : Type} (A : MultiTapeTM n Bool P) (x : List Bool) :

end

section
-- CONTROL Cfg_init_delta
attribute [local irreducible] Cfg.init
attribute [-simp] Cfg.init
example {n : ℕ} {P : Type} (q : P) (x : List Bool) :
    (Cfg.init q x : Cfg n Bool P x) = ⟨some q, 1, fun _ _ => none, fun _ => 0, []⟩ := rfl
end

section
-- CONTROL Cfg_init_equation

example {n : ℕ} {P : Type} (q : P) (x : List Bool) :

end
