set_option maxHeartbeats 0
open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  for n in [`Turing.seamCompTM.eq_1, `Turing.MultiTapeTM.indicator.eq_1] do
    match env.find? n with
    | none => logInfo m!"BEFORE_ABSENT|{n}"
    | some ci =>
      let owner := match env.getModuleIdxFor? n with
        | some i => env.header.moduleNames[i.toNat]!.toString
        | none => "CURRENT_PROBE"
      let ax ← liftCoreM <| Lean.collectAxioms n
      let row := Json.mkObj [("name",toJson n.toString),("owner",toJson owner),
        ("typeExpr",toJson (reprStr ci.type)),("axioms",toJson ((ax.qsort Name.lt).toList.map Name.toString))]
      logInfo m!"BEFORE|{row.compress}"
