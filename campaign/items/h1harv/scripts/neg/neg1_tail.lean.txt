/-! Negative control neg1 (must fail): S1 with the wrong exit, `escape` for a zero-round run. -/
namespace Turing
namespace StepBlock
example {k n : ℕ} {x : List Bool} (M : MultiTapeTM k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (live : c.state ≠ none)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x) (entry : d.state = some (exit k true))
    (rel : Rep M.tr c d) (w : List Bool) (ct : ℤ → Option Bool) (cp : ℤ)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) → ct (cp + p) = FinTM.bufferTape w p)
    (zero : Nat.ofDigits 2 (w.map Bool.toNat) = 0) :
    ∃ T e ct', T ≤ 2 * w.length + 2 ∧ Rep M.tr c e ∧
      (loopTM k).runFrom (counterLoopCfg (some (.dec .run)) d ct cp) T =
        counterLoopCfg (some .escape) e ct' cp := by
  obtain ⟨T, e, ct', bound, -, related, -, -, run, -⟩ :=
    loopTM_run M c live d entry rel w ct cp hw 0 zero
  rw [MultiTapeTM.runFrom_zero] at related run
  simp only [Option.isSome_iff_ne_none.mpr live, ↓reduceIte] at run
  exact ⟨T, e, ct', by simpa using bound, related, run⟩
end StepBlock
end Turing
