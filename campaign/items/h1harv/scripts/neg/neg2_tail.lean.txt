/-! Negative control neg2 (must fail): the early-escape bound without the counter-width term
`2|w|`, derived by the same steps. -/
namespace Turing
namespace StepBlock
example {k n : ℕ} {x : List Bool} (M : MultiTapeTM k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (live : c.state ≠ none)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x) (entry : d.state = some (exit k true))
    (rel : Rep M.tr c d) (w : List Bool) (ct : ℤ → Option Bool) (cp : ℤ)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) → ct (cp + p) = FinTM.bufferTape w p)
    (t : ℕ) (ht : Nat.ofDigits 2 (w.map Bool.toNat) = t) :
    ∃ T, ∀ t' ≤ t, (M.runFrom c t').state = none →
      T ≤ t' * (50 * ((tableBitsK M.tr).length + 1)) + 4 * t' := by
  obtain ⟨T, e, ct', -, early, -⟩ := loopTM_run M c live d entry rel w ct cp hw t ht
  exact ⟨T, fun t' h1 h2 => by have := early t' h1 h2; omega⟩
end StepBlock
end Turing
