"""Import closures for the landing (standard library only).

Usage: python3 -I closure.py <import graph json from tools/imports.py>

Adds the landing's edges to the checkout's graph (the new module `Build/StepBlockLoop` importing
`Build/StepBlock`, `Build/CounterLoop`, `Build/VirtualInput`; `CodeSimulator` and the
`TuringMachine` facade importing it), checks that the result is acyclic, and prints the forward
closure of the new module and the replay set (every module whose closure contains the new
module or `CodeSimulator`). The root `TCSlib.lean` is outside `TCSlib/` and is not listed: a
replay list never includes it (the snapshots cover only part of the library).
"""
import json
import sys

P = "TCSlib/Complexity/TuringMachine/"
NEW = P + "Build/StepBlockLoop"
g = json.load(open(sys.argv[1]))
print(f"checkout graph: {len(g)} modules under TCSlib/")
g[NEW] = [P + "Build/StepBlock", P + "Build/CounterLoop", P + "Build/VirtualInput"]
for m in (P + "CodeSimulator", "TCSlib/Complexity/TuringMachine"):
    assert NEW not in g[m]
    g[m] = g[m] + [NEW]


def fwd(m):
    seen, todo = set(), [m]
    while todo:
        x = todo.pop()
        if x in seen or x not in g:
            continue
        seen.add(x)
        todo += g[x]
    return seen


color = {}


def acyclic(m):
    color[m] = 1
    for d in g.get(m, []):
        if d not in g:
            continue
        if color.get(d) == 1 or (color.get(d) is None and not acyclic(d)):
            return False
    color[m] = 2
    return True


ok = all(color.get(m) == 2 or acyclic(m) for m in sorted(g))
print("acyclic with the landing edges:", ok)
F = fwd(NEW) - {NEW}
print(f"forward closure of the new module (excluding it): {len(F)} modules; contains "
      f"CodeSimulator: {P + 'CodeSimulator' in F}")
for m in sorted(F):
    print("   ", m)
R = sorted(m for m in g if NEW in fwd(m) or P + "CodeSimulator" in fwd(m))
print(f"replay set (modules whose closure contains the new module or CodeSimulator): {len(R)}")
for m in R:
    print("   ", m)
sys.exit(0 if ok else 1)
