#!/usr/bin/env bash
# Re-run every check of the H1 hosted-clock harvest statement gate and write its logs.
# Base: the campaign tip REF (default e831b5b0), exported read-only into <OUT>/base/tip by
# `git archive`; olean snapshot oleans-lm0b (post-MIG-0b, content-checked in section 0); the
# packages (Mathlib, ...) of wt-h1harv, whose lake-manifest and toolchain equal REF's (checked).
# Read-only on every checkout: `lean` only through tools/leanelab.sh (no -o, no olean written);
# Python only as `python3 -I`; git only archive, show, rev-parse, hash-object (no -w), diff,
# merge-base. Fails closed: every check states its expected outcome; the last line counts the
# failures, and the exit status is nonzero if any check failed.
# Usage: bash runall.sh <S> [REF]
set -u

REF=$(git -C "$R" rev-parse "$REF_IN"); B6=$(git -C "$R" rev-parse lean/b6-hosts)
ARM=$(git -C "$R" rev-parse lean/arm-minor-fixes)
SRC="$O/base/tip"; TIP="$SRC/TCSlib/Complexity/TuringMachine/CodeSimulator.lean"
LCS="$O/landed/TCSlib/Complexity/TuringMachine/CodeSimulator.lean"
LNEW="$O/landed/TCSlib/Complexity/TuringMachine/Build/StepBlockLoop.lean"
CHECKS=0; FAILS=0
el() { bash "$S/tools/leanelab.sh" "$WT" "$OL" "$1" 2>&1; }
elold() { bash "$S/tools/leanelab.sh" "$WT" "$S/oleans-lc1" "$1" 2>&1; }
san() { sed "s|$O/|<OUT>/|g; s|$S/|<S>/|g; s|$R/|<REPO>/|g"; }
counts() { local e w s
  e=$(grep -c ': error' <<<"$1"); w=$(grep -c ': warning' <<<"$1")
  s=$(grep -c "declaration uses 'sorry'" <<<"$1"); echo "$e $w $s"; }
# expect <label> <output> <exit> <errors> <sorry warnings> <other warnings>
expect() {
  local label="$1" out="$2" ex="$3" we="$4" ws="$5" wo="$6" e w s v=PASS
  read -r e w s <<<"$(counts "$out")"
  CHECKS=$((CHECKS + 1))
  if [ "$e" != "$we" ] || [ "$s" != "$ws" ] || [ "$((w - s))" != "$wo" ]; then v=FAIL; fi
  if [ "$we" = 0 ] && [ "$ex" != 0 ]; then v=FAIL; fi
  if [ "$we" != 0 ] && [ "$ex" = 0 ]; then v=FAIL; fi
  [ $v = FAIL ] && FAILS=$((FAILS + 1))
  echo "  [$v] $label: exit $ex | errors $e (expected $we) | \`declaration uses 'sorry'\`" \
       "$s (expected $ws) | other warnings $((w - s)) (expected $wo)"
}
check() { local label="$1"; shift; CHECKS=$((CHECKS + 1))
  if "$@"; then echo "  [PASS] $label"; else echo "  [FAIL] $label"; FAILS=$((FAILS + 1)); fi; }

  "$O/overlay/H1HarvestOverlay.lean" "$SC/consumer_tail.lean" "$O/probes/ConsumerProbe.lean" \

  --extra-import TCSlib.Complexity.TuringMachine.ChoiceReplay --tail "$SC/replay_tail.lean" \

  "$O/probes/PostHarvest-CodeSimulator.lean" > "$O/work/gen-posth.txt"
for n in neg1 neg2 neg3; do
  python3 -I "$SC/mkstandalone.py" "$O/H1Harvest.lean" "$O/work/$n.lean" \
    --tail "$SC/neg/${n}_tail.lean" > /dev/null
done
diff -u "$TIP" "$LCS" | sed "s|$SRC/|a/|; s|$O/landed/|b/|" > "$O/work/PostHarvest-CodeSimulator.diff"
mkdir -p "$O/work/snap"
cp "$TIP" "$O/work/snap/CodeSimulator_tip.lean"
cp "$SRC/TCSlib/Complexity/TuringMachine/Build/StepBlock.lean" "$O/work/snap/StepBlock_tip.lean"

  > "$O/work/snap/CodeSimulator_2c3c975a.lean"

HDR="Tool: \`bash <S>/tools/leanelab.sh <S>/wt-h1harv <S>/oleans-lm0b <file>\`, which runs, verbatim:
\`\`\`
$(cat "$S/tools/leanelab.sh")
\`\`\`
(direct \`lean\`, no \`-o\`, no olean written; \`wt-h1harv\` supplies only the packages and the
working directory, \`lean\` reads no TCSlib source but the file it is given). Base: the campaign
tip $REF, exported read-only to <OUT>/base/tip by \`git archive\`. Snapshot: \`oleans-lm0b\`
(post-MIG-0b; §0). Toolchain: \`$(cat "$SRC/lean-toolchain")\`. Pending PRs read as text only:
#92 \`lean/b6-hosts\` $B6, #93 \`lean/arm-minor-fixes\` $ARM. <OUT> is the gate's output
