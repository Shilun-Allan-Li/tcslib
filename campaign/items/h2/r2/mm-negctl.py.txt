"""Negative control for movemanifest.py: a simulated integration base (scratch directory with a
mutated copy of CodeSimulator.lean only), mimicking what an in-flight H1 continuation could
deliver. Usage: python3 -I mm-negctl.py CHECKOUT OUTDIR
  M1  a new private h1 lemma inside the step-block region (not in the 121-name inventory);
  M2  the headline's proof names h1Round_effect (renamed: a text edit) and h1SeekTM (tier C);
  M3  one moved proof changed (h1TwoMove_run: an extra `skip`), so the move comparison differs.
"""
import os
import sys

REL = 'TCSlib/Complexity/TuringMachine/CodeSimulator.lean'


def once(s, old, new):
    assert s.count(old) == 1, old[:60]
    return s.replace(old, new)


wt, out = sys.argv[1], sys.argv[2]
s = open(os.path.join(wt, REL), encoding='utf-8').read()
anchor = '/-- Number of fixed bits before a transition'
k = s.index('private lemma h1TwoMove_run')
k = s.index('\n/--', k)
s = s[:k] + ('\n\n/-- Simulated continuation helper. -/\nprivate lemma h1Round_timeAux (k : ℕ) :'
             '\n    h1RoundTM k = h1RoundTM k := rfl\n') + s[k:]
old = """          (d * (α.length + x.length + t + 1) ^ d)) := by
  sorry

end MachineFormat"""
new = """          (d * (α.length + x.length + t + 1) ^ d)) := by
  have _h := @h1Round_effect
  have _s := h1SeekTM
  sorry

end MachineFormat"""
s = once(s, old, new)
j = s.index('private lemma h1TwoMove_run')
j2 = s.index(':= by', j) + len(':= by')
s = s[:j2] + '\n  skip' + s[j2:]
d = os.path.join(out, 'TCSlib/Complexity/TuringMachine')
os.makedirs(d, exist_ok=True)
open(os.path.join(out, REL), 'w', encoding='utf-8').write(s)
print(os.path.join(out, REL))
