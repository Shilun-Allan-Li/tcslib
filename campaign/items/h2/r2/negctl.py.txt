"""Negative controls for fidelity-r2.py: write mutated copies of the round-2 draft.

Usage:  python3 -I negctl.py DRAFT.lean OUTDIR
Each mutation is a single asserted textual change; the expected fidelity-r2 verdict is FAIL.
  N1  definition:        recordWidth 4 * k + 4  ->  4 * k + 5
  N2  contract (bound):  roundTM_effect's round bound 50  ->  51 (before the statement's `let`)
  N3  contract (body):   roundTM_effect's mirrored envelope 1  ->  2 (after the `let`)
  N4  missing:           recordSkipTM_spec deleted (docstring and declaration)
  N5  unexpected:        an extra theorem appended
  N6  duplicate:         inputRank declared twice
  N7  admission:         the additive roundTM_firstExit_unique's proof replaced by `sorry`
"""
import os
import re
import sys


def once(s, old, new, start=0, label=''):
    k = s.find(old, start)
    assert k >= 0, f'{label}: pattern not found'
    assert s.find(old, k + 1) < 0 or start, f'{label}: pattern not unique'
    return s[:k] + new + s[k + len(old):]


def block(s, name):
    """Docstring start .. end of the declaration `name` (up to the next docstring)."""
    m = re.search(r'^(?:theorem|lemma|def|abbrev|inductive|instance) ' + re.escape(name) + r'\b',
                  s, re.M)
    assert m, name
    a = s.rfind('\n/--', 0, m.start()) + 1
    b = s.find('\n/--', m.end())
    b2 = s.find('\n/-!', m.end())
    b = min(x for x in (b, b2) if x >= 0) + 1
    return a, b


def main():
    draft, out = sys.argv[1], sys.argv[2]
    s = open(draft, encoding='utf-8').read()
    os.makedirs(out, exist_ok=True)
    muts = {}
    muts['N1'] = once(s, 'abbrev recordWidth (k : ℕ) : ℕ := 4 * k + 4',
                      'abbrev recordWidth (k : ℕ) : ℕ := 4 * k + 5', label='N1')
    eff = s.index('theorem roundTM_effect')
    t = s[:eff] + s[eff:].replace('∃ T, 0 < T ∧ T ≤ 50 *', '∃ T, 0 < T ∧ T ≤ 51 *', 1)
    assert t != s
    muts['N2'] = t
    old = 'if j.val < k then 1 else (50 * ((tableBitsK tr).length + 1) : ℕ)) ∧'
    k = s.index(old, eff)
    assert k < s.index('theorem abs_workTapePos_runFrom_sub_le') if \
        'theorem abs_workTapePos_runFrom_sub_le' in s else True
    muts['N3'] = s[:k] + old.replace('then 1 else', 'then 2 else') + s[k + len(old):]
    a, b = block(s, 'recordSkipTM_spec')
    muts['N4'] = s[:a] + s[b:]
    muts['N5'] = once(s, '\nend Turing.Draft.StepBlock',
                      '\n/-- An unexpected addition. -/\ntheorem extra_sanity : True := trivial\n'
                      '\nend Turing.Draft.StepBlock', label='N5')
    a, b = block(s, 'inputRank')
    muts['N6'] = s[:b] + s[a:b] + s[b:]
    m = s.index('theorem roundTM_firstExit_unique')
    p = s.index(':= by\n  obtain ⟨T, hpos, -, hexit', m)
    e = s.index('\nend Turing.Draft.StepBlock', p)
    muts['N7'] = s[:p] + ':= by\n  sorry\n' + s[e:]
    for k, v in muts.items():
        assert v != s, k
        path = os.path.join(out, f'StepBlockDraft-r2-{k}.lean')
        open(path, 'w', encoding='utf-8').write(v)
        print(path)


if __name__ == '__main__':
    main()
