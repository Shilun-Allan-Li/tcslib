/-- The fixed enumeration of the work reads of `k` work tapes: tape `0`
outermost, each read over `none`, `some false`, `some true`. [AB09, §1.4] -/
def workReadsK : (k : ℕ) → List (Fin k → Option Bool)
  | 0 => [Fin.elim0]
  | k + 1 => ([none, some false, some true] : List (Option Bool)).flatMap fun w =>
      (workReadsK k).map fun r => (Fin.cons w r : Fin (k + 1) → Option Bool)

/-- The fixed serialization of a `k`-work-tape transition table: states in `Fin`
order, then the input read, then the work reads in `Turing.workReadsK` order,
one `Turing.actionBitsK` record each. [AB09, §1.4] -/
def tableBitsK {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1))) :
    List Bool :=
  (List.finRange (n + 1)).flatMap fun q =>
    ([none, some false, some true] : List (Option Bool)).flatMap fun inp =>
      (workReadsK k).flatMap fun w => actionBitsK (tr q inp w)

namespace StepBlock

/-- Output summary: `none` means empty, `some true` means exactly `[true]`,
and `some false` means any other output. The last class is irreversible. -/
def outputSummary : List Bool → Option Bool
  | [] => none
  | [true] => some true
  | _ => some false

/-- Update the three-way summary for the emission of one source transition. -/
def summaryEmit : Option Bool → Option Bool → Option Bool
  | s, none => s
  | none, some true => some true
  | _, some _ => some false

/-- Summary update accounts for the symbol on a halting transition as well. -/
lemma outputSummary_append (w : List Bool) (b : Option Bool) :
    outputSummary (w ++ b.toList) = summaryEmit (outputSummary w) b := by
  cases b with
  | none => simp [summaryEmit]
  | some b =>
    cases w with
    | nil => cases b <;> rfl
    | cons a w =>
      cases w with
      | nil => cases a <;> cases b <;> rfl
      | cons c w => cases a <;> cases b <;> rfl

/-- The summary's accepting value characterizes the exact singleton output. -/
lemma outputSummary_true (w : List Bool) :
    outputSummary w = some true ↔ w = [true] := by
  cases w with
  | nil => simp [outputSummary]
  | cons b w =>
    cases w with
    | nil => cases b <;> simp [outputSummary]
    | cons c w => cases b <;> simp [outputSummary]

/-- Once the summary is other, no subsequent emission can repair it. -/
lemma summaryEmit_other (b : Option Bool) :
    summaryEmit (some false) b = some false := by
  cases b with
  | none => rfl
  | some b => cases b <;> rfl

/-- Number of fixed bits before a transition's variable-length successor field. -/
abbrev recordWidth (k : ℕ) : ℕ := 4 * k + 4

/-- Finite control for the selected-record component of H1. The pair stores
the number of bits read and a fixed-size bit buffer; `none` is its live exit.
This type depends only on the work-tape count, never on the coded state count. -/
abbrev RecordState (k : ℕ) :=
  Option (Fin (recordWidth k + 1) × (Fin (recordWidth k) → Bool))

/-- The record tape follows the mirrored work bank. -/
def tableTape (k : ℕ) : Fin (k + 2) := Fin.natAdd k (0 : Fin 2)

/-- A stationary cell after the record tape stores the three-valued output summary. -/
def summaryTape (k : ℕ) : Fin (k + 2) := Fin.natAdd k (1 : Fin 2)

/-- Every mirrored work tape is distinct from the record tape. -/
private lemma mirror_ne_table (k : ℕ) (i : Fin k) :
    Fin.castAdd 2 i ≠ tableTape k := by
  intro h
  have hv := congrArg Fin.val h
  have hi := i.isLt
  dsimp [tableTape] at hv
  omega

/-- The summary cell and the record head lie on distinct auxiliary tapes. -/
private lemma summary_ne_table (k : ℕ) : summaryTape k ≠ tableTape k := by
  intro h
  have hv := congrArg Fin.val h
  dsimp [summaryTape, tableTape] at hv
  omega

/-- Interpret the fixed fields with the public record reader. A dummy halt
field makes the parsed successor irrelevant; the actual successor remains
on the record tape for the next component. Only canonized records are used. -/
private def payload (k : ℕ) (bits : Fin (recordWidth k) → Bool) :
    Action k Bool (Fin 1) :=
  ((codeReadActionK k 0 (List.ofFn bits ++ [false])).map Prod.fst).getD
    ⟨0, fun _ => (none, 0), none, none⟩

/-- Apply the fixed fields to the mirrored bank and native input head,
updating the summary cell instead of physically emitting. The record tape
and both auxiliary heads remain fixed, and the next control is a live exit. -/
private def applyPayload {k : ℕ} (a : Action k Bool (Fin 1)) (s : Option Bool) :
    Action (k + 2) Bool (RecordState k) :=
  ⟨a.inputTape, Fin.addCases a.workTapes
    (fun i : Fin 2 => if i = 0 then (none, 0) else (some (summaryEmit s a.output), 0)),
    none, some none⟩

/-- H1's physical selected-record component: read the fixed fields, then
execute their work/input actions and summary update. Its exit points at the
successor field; selecting the record and installing its successor are separate
remaining parts of the single interpreter block. [AB09, Theorem 1.9]

No catalog row reads a fixed-length, non-delimited prefix into finite control:
the two-pass scan requires delimiters. This small controller supplies that
part, using `tapeAction` and the existing `codeReadActionK`, without a new
grammar validator, host, or clock. -/
def recordTM (k : ℕ) : MultiTapeTM (k + 2) Bool (RecordState k) where
  q₀ := some (0, fun _ => false)
  tr := fun q _ w => match q with
    | none => inputMoveAction 0 none
    | some (p, bits) =>
      if h : p.val < recordWidth k then
        tapeAction (tableTape k) 0 none .pos
          (some (⟨p.val + 1, by omega⟩,
            Function.update bits ⟨p.val, h⟩ ((w (tableTape k)).getD false)))
      else applyPayload (payload k bits) (w (summaryTape k))

/-- Configuration during the fixed-prefix read, with the first `r` bits
installed in finite control and every tape content unchanged. -/
private def readCfg {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (bits : Fin (recordWidth k) → Bool) (r : ℕ) (hr : r ≤ recordWidth k) :
    Cfg (k + 2) Bool (RecordState k) x :=
  { d with
    state := some (some (⟨r, by omega⟩, fun i => if i.val < r then bits i else false))
    workTapePos := Function.update d.workTapePos (tableTape k)
      (d.workTapePos (tableTape k) + r) }

/-- The public record reader recovers exactly the fixed action fields from
their finite buffer, regardless of the actual successor's state space.

**Proof sketch.** The movement, work actions and emission occupy exactly
`4k + 4` bits. Replace only the remaining successor field by the halt tag,
then invoke `codeReadActionK_append` at the one-state type. -/
private lemma payload_correct {k n : ℕ} (a : Action k Bool (Fin (n + 1)))
    (bits : Fin (recordWidth k) → Bool)
    (hbits : List.ofFn bits = (actionBitsK a).take (recordWidth k)) :
    payload k bits = ⟨a.inputTape, a.workTapes, a.output, none⟩ := by
  have moves (s : SignType) : (signBits s).length = 2 := by cases s <;> rfl
  have writes (b : Option (Option Bool)) : (optOptBoolBits b).length = 2 := by
    cases b with
    | none => rfl
    | some b => cases b with
      | none => rfl
      | some b => cases b <;> rfl
  have emits (b : Option Bool) : (optBoolBits b).length = 2 := by
    cases b with
    | none => rfl
    | some b => cases b <;> rfl
  let fixed := signBits a.inputTape ++
    (List.finRange k).flatMap (fun i => optOptBoolBits (a.workTapes i).1 ++
      signBits (a.workTapes i).2) ++ optBoolBits a.output
  have plen : fixed.length = recordWidth k := by
    simp [fixed, recordWidth, List.length_flatMap, moves, writes, emits]
    omega
  have he : List.ofFn bits ++ [false] =
      actionBitsK (⟨a.inputTape, a.workTapes, a.output, none⟩ : Action k Bool (Fin 1)) := by
    rw [hbits]
    change (fixed ++ optStateBits a.state).take (recordWidth k) ++ [false] =
      fixed ++ [false]
    rw [← plen, List.take_left]
  have hp := codeReadActionK_append
    (⟨a.inputTape, a.workTapes, a.output, none⟩ : Action k Bool (Fin 1)) []
  simp only [List.append_nil] at hp
  simp only [payload, he, hp, Option.map_some, Option.getD_some]

/-- The physical reader consumes any permitted prefix in exactly its length,
without changing the native input, any work-tape cell, or the physical output.
Only the record head advances. This is a framed statement: all unmentioned
tape cells, mirrored tape positions, and the prior output may be arbitrary.

**Proof sketch.** Induct on the number of consumed bits. The public single-tape
movement law advances the record head; updating the next finite buffer slot
extends precisely the prefix already read. There is no delimiter assumption. -/
private lemma recordTM_read {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (bits : Fin (recordWidth k) → Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ i : Fin (recordWidth k),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i))
    (r : ℕ) (hr : r ≤ recordWidth k) :
    (recordTM k).runFrom d r = readCfg d bits r hr := by
  induction r with
  | zero =>
    apply Cfg.ext <;> simp [readCfg, hq, recordTM]
  | succ r ih =>
    have hlt : r < recordWidth k := by omega
    rw [MultiTapeTM.runFrom_succ_eq_step', ih (by omega)]
    have hread : (readCfg d bits r (by omega)).workTapeSymbols (tableTape k) =
        some (bits ⟨r, hlt⟩) := by
      simpa [readCfg, Cfg.workTapeSymbols] using hw ⟨r, hlt⟩
    have hbuffer : Function.update (fun i : Fin (recordWidth k) =>
        if i.val < r then bits i else false) ⟨r, hlt⟩ (bits ⟨r, hlt⟩) =
        fun i => if i.val < r + 1 then bits i else false := by
      funext i
      by_cases he : i = ⟨r, hlt⟩
      · subst i
        simp
      · have hv : i.val ≠ r := fun h => he (Fin.ext h)
        rw [Function.update_of_ne he]
        by_cases hi : i.val < r <;> simp [hi, show (i.val < r + 1) ↔ (i.val < r) by omega]
    rw [step_of_tapeAction_move (recordTM k) (readCfg d bits r (by omega))
      (q' := some (⟨r + 1, by omega⟩,
        fun i => if i.val < r + 1 then bits i else false))
      (tableTape k) 0 .pos rfl (by
        simp only [recordTM, dif_pos hlt, hread, Option.getD_some, hbuffer])]
    apply Cfg.ext
    · rfl
    · simp [readCfg]
    · rfl
    · funext i
      by_cases he : i = tableTape k
      · subst i
        simp [readCfg, Nat.cast_add, SignType.pos_eq_one, add_assoc]
      · simp [readCfg, he]
    · rfl

/-- A selected, canonized record executes its fixed action fields in exactly
`4k + 5` physical steps. The table head then points at its successor field.
The equality includes the native-input movement, all mirrored writes/moves,
the finite summary update, and preservation of the physical output prefix.

**Proof sketch.** Apply the prefix trace at its full width. Its finite buffer
is the record's fixed prefix, so `payload_correct` identifies the public
reader's result. The next transition executes that payload silently. -/
private lemma recordTM_execute {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (a : Action k Bool (Fin (n + 1))) (bits : Fin (recordWidth k) → Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ i : Fin (recordWidth k),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i))
    (hbits : List.ofFn bits = (actionBitsK a).take (recordWidth k)) :
    (recordTM k).runFrom d (recordWidth k + 1) =
      (applyPayload (⟨a.inputTape, a.workTapes, a.output, none⟩ : Action k Bool (Fin 1))
        (d.workTapeSymbols (summaryTape k))).apply
        (readCfg d bits (recordWidth k) le_rfl) := by
  rw [MultiTapeTM.runFrom_succ_eq_step', recordTM_read d bits hq hw _ le_rfl]
  have hbuf : (fun i : Fin (recordWidth k) =>
      if i.val < recordWidth k then bits i else false) = bits := by
    funext i
    exact if_pos i.isLt
  have hne := summary_ne_table k
  have hs : (readCfg d bits (recordWidth k) le_rfl).workTapeSymbols
      (summaryTape k) = d.workTapeSymbols (summaryTape k) := by
    simp [readCfg, Cfg.workTapeSymbols, hne]
  simp only [MultiTapeTM.step, readCfg, recordTM, lt_self_iff_false,
    hbuf, payload_correct a bits hbits]
  change (applyPayload _
    ((readCfg d bits (recordWidth k) le_rfl).workTapeSymbols (summaryTape k))).apply _ = _
  rw [hs]

/-- Addendum 1's head envelope for the selected-record component. Mirrored
heads and the native input head remain at their starting positions until the
single payload transition, then move exactly as the source action does. Thus
they stay between their start and end positions, with displacement at most
one. The table head advances by the consumed fixed prefix; the summary head
never moves. The table head is not yet restored to a round home.

**Proof sketch.** Before the payload, project `recordTM_read`; at the payload,
project `recordTM_execute`. Use the public one-action work-head bound. The
native input movement is clamped at either boundary. -/
private lemma recordTM_envelope {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (a : Action k Bool (Fin (n + 1))) (bits : Fin (recordWidth k) → Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ i : Fin (recordWidth k),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i))
    (hbits : List.ofFn bits = (actionBitsK a).take (recordWidth k)) :
    ∀ u ≤ recordWidth k + 1,
      let e := (recordTM k).runFrom d u
      (∀ i : Fin k, e.workTapePos (Fin.castAdd 2 i) =
        d.workTapePos (Fin.castAdd 2 i) +
          if u = recordWidth k + 1 then ((a.workTapes i).2 : ℤ) else 0) ∧
      e.inputPos = (if u = recordWidth k + 1 then moveInputPos d.inputPos a.inputTape
        else d.inputPos) ∧
      (∀ j : Fin (k + 2), |e.workTapePos j - d.workTapePos j| ≤
        if j.val < k then 1 else (recordWidth k : ℤ)) ∧
      |(e.inputPos.val : ℤ) - d.inputPos.val| ≤ 1 ∧
      e.workTapePos (tableTape k) = d.workTapePos (tableTape k) + min u (recordWidth k) ∧
      e.workTapePos (summaryTape k) = d.workTapePos (summaryTape k) := by
  intro u hu
  dsimp only
  by_cases hend : u = recordWidth k + 1
  · subst u
    rw [recordTM_execute d a bits hq hw hbits]
    refine ⟨?_, ?_, ?_, ?_, ?_, ?_⟩
    · intro i
      simp [applyPayload, readCfg, mirror_ne_table k i]
    · simp [applyPayload, readCfg]
    · intro j
      refine Fin.addCases (fun i => ?_) (fun i => ?_) j
      · simpa [readCfg, mirror_ne_table k i, i.isLt] using
          workTapePos_apply_le
            (applyPayload (⟨a.inputTape, a.workTapes, a.output, none⟩ : Action k Bool (Fin 1))
              (d.workTapeSymbols (summaryTape k)))
            (readCfg d bits (recordWidth k) le_rfl) (Fin.castAdd 2 i)
      · fin_cases i <;>
          simp [applyPayload, readCfg, tableTape, recordWidth, abs_le] <;> omega
    · change |((moveInputPos d.inputPos a.inputTape).val : ℤ) - d.inputPos.val| ≤ 1
      have hp := d.inputPos.isLt
      cases a.inputTape <;> simp only [moveInputPos, SignType.cast] <;>
        split <;> simp_all [abs_le] <;> omega
    · simp [applyPayload, readCfg, tableTape]
    · simp [applyPayload, readCfg, summaryTape, tableTape]
  · have hread : u ≤ recordWidth k := by omega
    rw [recordTM_read d bits hq hw u hread]
    refine ⟨?_, ?_, ?_, ?_, ?_, ?_⟩
    · intro i
      simp [readCfg, mirror_ne_table k i, hend]
    · simp [readCfg, hend]
    · intro j
      by_cases hj : j = tableTape k
      · subst j
        simpa [readCfg, tableTape, abs_of_nonneg (Int.natCast_nonneg u)] using
          (show (u : ℤ) ≤ recordWidth k by omega)
      · simp [readCfg, hj]
        split <;> omega
    · simp [readCfg]
    · simp [readCfg, min_eq_left hread]
    · simp [readCfg, summary_ne_table k]

/-- The selected-record component has finite, decidable control for every
fixed tape count; it is a single physical machine for all coded state counts. -/
def recordFinTM (k : ℕ) : FinTM Bool where
  k := k + 2
  State := RecordState k
  tm := recordTM k

/-- Relation-level effect of the selected record's fixed fields. The first
`k` work tapes mirror an arbitrary source configuration, and a stationary
auxiliary cell stores its output summary. After exactly `4k + 5` steps the
input/work fields mirror the source action, its emission has been included
in the summary, and the physical output is unchanged. The live exit is
first reached at that time, whose bound is the selected record's length.

This is the fixed-field part of H1's step block, not yet a full simulated
transition: the coded successor must still be installed by the caller.

**Promotion candidate (host H2, step block; machine-library-design.md §14).**
This component's contract includes Addendum 1's head envelope: at every
intermediate time each mirrored head and the native input head is still at
its start or has made exactly the source action's move, with displacement at
most one. Auxiliary work-head displacement is bounded by `4k + 4`; the table
head advances by `min u (4k + 4)`, and the summary head stays at home. Lookup,
successor installation and table-head restoration remain outside this partial
contract; promotion as a complete step block must wait for those parts.

**Proof sketch.** Project the proved execution equality onto the input,
mirrored bank and summary cell. The summary identity is the inherited
`outputSummary_append`, including when the source action halts. Before the
last step the prefix trace has a reading state, never the live exit. -/
lemma recordTM_effect {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (bits : Fin (recordWidth k) → Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ i : Fin (recordWidth k),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i))
    (hbits : List.ofFn bits = (actionBitsK a).take (recordWidth k))
    (hin : d.inputPos = c.inputPos)
    (htapes : ∀ i : Fin k, d.workTapes (Fin.castAdd 2 i) = c.workTapes i)
    (hheads : ∀ i : Fin k, d.workTapePos (Fin.castAdd 2 i) = c.workTapePos i)
    (hsum : d.workTapeSymbols (summaryTape k) = outputSummary c.output) :
    let e := (recordTM k).runFrom d (recordWidth k + 1)
    e.state = some none ∧ e.inputPos = (a.apply c).inputPos ∧
      (∀ i : Fin k, e.workTapes (Fin.castAdd 2 i) = (a.apply c).workTapes i ∧
        e.workTapePos (Fin.castAdd 2 i) = (a.apply c).workTapePos i) ∧
      e.workTapeSymbols (summaryTape k) = outputSummary (a.apply c).output ∧
      e.output = d.output ∧
      (∀ u < recordWidth k + 1, ((recordTM k).runFrom d u).state ≠ some none) ∧
      recordWidth k + 1 ≤ (actionBitsK a).length ∧
      (∀ u ≤ recordWidth k + 1,
        let f := (recordTM k).runFrom d u
        (∀ i : Fin k, f.workTapePos (Fin.castAdd 2 i) =
          d.workTapePos (Fin.castAdd 2 i) +
            if u = recordWidth k + 1 then ((a.workTapes i).2 : ℤ) else 0) ∧
        f.inputPos = (if u = recordWidth k + 1 then moveInputPos d.inputPos a.inputTape
          else d.inputPos) ∧
        (∀ j : Fin (k + 2), |f.workTapePos j - d.workTapePos j| ≤
          if j.val < k then 1 else (recordWidth k : ℤ)) ∧
        |(f.inputPos.val : ℤ) - d.inputPos.val| ≤ 1 ∧
        f.workTapePos (tableTape k) =
          d.workTapePos (tableTape k) + min u (recordWidth k) ∧
        f.workTapePos (summaryTape k) = d.workTapePos (summaryTape k)) := by
  dsimp only
  have he := recordTM_execute d a bits hq hw hbits
  refine ⟨?_, ?_, ?_, ?_, ?_, ?_, codeActionK_length a,
    recordTM_envelope d a bits hq hw hbits⟩
  · rw [he]
    rfl
  · rw [he]
    exact congrArg (fun p => moveInputPos p a.inputTape) hin
  · intro i
    rw [he]
    simp [applyPayload, readCfg, Fin.addCases_left, mirror_ne_table k i, htapes i, hheads i]
  · rw [he]
    simp [Cfg.workTapeSymbols, applyPayload, readCfg, summaryTape,
      ← hsum, outputSummary_append]
  · rw [he]
    simp [applyPayload, readCfg]
  · intro u hu
    rw [recordTM_read d bits hq hw u (by omega)]
    simp [readCfg]

/-- Every vector of work reads occurs in the serializer's fixed enumeration.
The proof follows its tape-zero-first recursion, independently of coded states. -/
lemma mem_workReadsK (k : ℕ) (w : Fin k → Option Bool) :
    w ∈ workReadsK k := by
  induction k with
  | zero =>
    have hw : w = Fin.elim0 := funext (fun i => Fin.elim0 i)
    simp [workReadsK, hw]
  | succ k ih =>
    apply List.mem_flatMap.mpr
    refine ⟨w 0, ?_, ?_⟩
    · cases w 0 with
      | none => simp
      | some b => cases b <;> simp
    · apply List.mem_map.mpr
      refine ⟨Fin.tail w, ih (Fin.tail w), ?_⟩
      exact Fin.cons_self_tail w

/-- There are exactly three possible reads on each mirrored tape. -/
lemma length_workReadsK (k : ℕ) : (workReadsK k).length = 3 ^ k := by
  induction k with
  | zero => rfl
  | succ k ih => simp [workReadsK, ih, Nat.pow_succ]; omega

/-- Work-read rank is the index in the actual serializer order. -/
private def workRank {k : ℕ} (w : Fin k → Option Bool) : Fin (3 ^ k) :=
  ⟨(workReadsK k).idxOf w, by
    rw [← length_workReadsK]
    exact List.idxOf_lt_length_iff.mpr (mem_workReadsK k w)⟩

/-- Input symbols use the same blank/false/true order as the table writer. -/
private def inputRank : Option Bool → Fin 3
  | none => 0
  | some false => 1
  | some true => 2

/-- The read-vector rank within a state block. Its bound depends only on
`k`, so a physical controller may retain it in finite control. -/
private def readRank {k : ℕ} (inp : Option Bool) (w : Fin k → Option Bool) :
    Fin (3 ^ (k + 1)) :=
  ⟨(inputRank inp).val * 3 ^ k + (workRank w).val, by
    have hi := (inputRank inp).isLt
    have hw := (workRank w).isLt
    have hpos : 0 < 3 ^ k := Nat.pow_pos (by omega)
    have hm := Nat.mul_le_mul_right (3 ^ k) (show (inputRank inp).val ≤ 2 by omega)
    rw [Nat.pow_succ]
    omega⟩

/-- States of the trusted successor-field walker. The halt tag and the
unary successor terminator lead to distinct live exits. -/
inductive SuccPhase where
  | tag
  | unary
  | done (live : Bool)
deriving DecidableEq, Fintype

/-- Walk an already canonized successor field. When `copy` is true, copy
its unary state word to `dst`; when false, only advance the table head.
No catalog leaf walks a bit-terminated, non-blank-delimited word. This
leaf supplies that missing operation; it does not validate the grammar. -/
def succFieldTM {m : ℕ} (src dst : Fin m) (copy : Bool) :
    MultiTapeTM m Bool SuccPhase where
  q₀ := .tag
  tr := fun q _ w => match q with
    | .tag => tapeAction src 0 none .pos
        (if w src = some true then .unary else .done false)
    | .unary =>
        ⟨0, fun j => if j = src then (none, .pos)
          else if copy = true ∧ j = dst then (some (w src), .pos) else (none, 0),
          none, some (if w src = some true then .unary else .done true)⟩
    | .done b => inputMoveAction 0 (.done b)

/-- A framed prefix of the successor copy: the table head advances, and
only an enabled destination receives the consumed prefix. -/
def succFieldCfg {m : ℕ} {x : List Bool} (src dst : Fin m) (copy : Bool)
    (d : Cfg m Bool SuccPhase x) (w : List Bool) (r : ℕ) (q : SuccPhase) :
    Cfg m Bool SuccPhase x :=
  { d with
    state := some q
    workTapes := fun j z =>
      if copy = true ∧ j = dst ∧ d.workTapePos dst ≤ z ∧ z < d.workTapePos dst + r
      then FinTM.bufferTape w (z - d.workTapePos dst) else d.workTapes j z
    workTapePos := fun j => d.workTapePos j +
      if j = src ∨ (copy = true ∧ j = dst) then (r : ℤ) else 0 }

/-- One unary-field transition writes precisely the next copied cell.
The public prefix-write identity handles the destination's arbitrary frame.

**Proof sketch.** The source and destination are distinct. Read the next
source bit from the framed prefix hypothesis. Expanding one machine action
advances the selected heads; on the destination, `catalog_write_take`
extends the overwritten prefix. Every other cell is unchanged. -/
private lemma succFieldTM_unary_step {m : ℕ} {x : List Bool} (src dst : Fin m)
    (hne : src ≠ dst) (copy : Bool) (d : Cfg m Bool SuccPhase x)
    (w : List Bool) (r : ℕ) (hr : r < w.length)
    (hw : d.workTapes src (d.workTapePos src + r) = some w[r]) :
    (succFieldTM src dst copy).step (succFieldCfg src dst copy d w r .unary) =
      succFieldCfg src dst copy d w (r + 1)
        (if w[r] = true then .unary else .done true) := by
  have hs : (succFieldCfg src dst copy d w r .unary).workTapeSymbols src = some w[r] := by
    simpa [succFieldCfg, Cfg.workTapeSymbols, hne] using hw
  simp only [MultiTapeTM.step, succFieldCfg, succFieldTM]
  change (⟨0, _, none, some (if
      (succFieldCfg src dst copy d w r .unary).workTapeSymbols src = some true
      then SuccPhase.unary else .done true)⟩ : Action m Bool SuccPhase).apply _ = _
  rw [hs]
  apply Cfg.ext
  · simp
  · simp
  · funext j
    by_cases hj : j = src
    · subst j
      simp [hne]
    · by_cases hc : copy = true ∧ j = dst
      · obtain ⟨hc, he⟩ := hc
        subst j
        subst copy
        simpa [succFieldCfg, hj, Action.apply, Cfg.workTapeSymbols, hne, hw] using
          catalog_write_take (d.workTapes dst) (d.workTapePos dst) w r hr
      · simp [hj, hc, ← and_assoc, Action.apply]
  · funext j
    by_cases hj : j = src
    · subst j
      simp [hne, Nat.cast_add, add_assoc]
    · by_cases hc : copy = true ∧ j = dst
      · simp [hc, Ne.symm hne, Nat.cast_add, add_assoc]
      · simp [hj, hc]
  · simp

/-- Every prefix of a trusted unary state field has its literal physical
trace, including the terminating false bit. The native input and output
are fixed, and every unrelated head is fixed.

**Proof sketch.** Induct over the consumed cells. Before the last cell the
replicated bit is true; the final false bit takes the live-successor exit.
Use the single prefix-write step for both the copying and skipping modes. -/
private lemma succFieldTM_unary_run {m : ℕ} {x : List Bool} (src dst : Fin m)
    (hne : src ≠ dst) (copy : Bool) (d : Cfg m Bool SuccPhase x)
    (s : ℕ) (hq : d.state = some .unary)
    (hw : ∀ (r : ℕ) (hr : r < s + 1), d.workTapes src (d.workTapePos src + r) =
      some ((List.replicate s true ++ [false])[r]'(by simpa using hr)))
    (r : ℕ) (hr : r ≤ s + 1) :
    (succFieldTM src dst copy).runFrom d r =
      succFieldCfg src dst copy d (List.replicate s true ++ [false]) r
        (if r ≤ s then .unary else .done true) := by
  induction r with
  | zero =>
    apply Cfg.ext <;> simp [succFieldCfg, hq]
    funext j z
    split_ifs <;> first | rfl | omega
  | succ r ih =>
    rw [MultiTapeTM.runFrom_succ_eq_step', ih (by omega), if_pos (by omega)]
    rw [succFieldTM_unary_step src dst hne copy d _ r (by simp; omega) (hw r (by omega))]
    congr 1
    by_cases hlt : r < s
    · rw [List.getElem_append_left (by simpa using hlt)]
      simp [show r + 1 ≤ s by omega]
    · have he : r = s := by omega
      subst r
      simp

/-- Configuration after the successor tag has been consumed. It is a
one-cell table move and changes no other head, cell, input or output. -/
def afterTag {m : ℕ} {x : List Bool} (src : Fin m)
    (d : Cfg m Bool SuccPhase x) : Cfg m Bool SuccPhase x :=
  { d with state := some .unary
           workTapePos := Function.update d.workTapePos src (d.workTapePos src + 1) }

/-- The trusted successor walker consumes its entire variable-width field,
optionally installing the unary successor in a blank destination. Its exit
is live in both cases: `done false` means simulated halt, `done true` means
a next state. This is a record leaf, not a grammar validator.

**Proof sketch.** The first transition reads the tag. A halt tag finishes
immediately. A live tag enters the shared unary-prefix trace for the
successor word, including its terminating false. -/
private lemma succFieldTM_trace {m n : ℕ} {x : List Bool} (src dst : Fin m)
    (hne : src ≠ dst) (copy : Bool) (d : Cfg m Bool SuccPhase x)
    (s : Option (Fin n)) (hq : d.state = some .tag)
    (hw : ∀ (r : ℕ) (hr : r < (optStateBits s).length),
      d.workTapes src (d.workTapePos src + r) = some ((optStateBits s)[r]'hr))
    (t : ℕ) (ht : t ≤ (optStateBits s).length) :
    let word := (s.map unaryFin).getD []
    (succFieldTM src dst copy).runFrom d t =
      if t = 0 then d else succFieldCfg src dst copy (afterTag src d) word (t - 1)
        (if t < (optStateBits s).length then .unary else .done s.isSome) := by
  cases t with
  | zero => rfl
  | succ t =>
    simp only [Nat.succ_ne_zero, if_false, Nat.add_sub_cancel]
    cases s with
    | none =>
      have ht0 : t = 0 := by simpa [optStateBits] using ht
      subst t
      have hs : d.workTapeSymbols src = some false := by
        simpa [Cfg.workTapeSymbols, optStateBits] using hw 0 (by simp [optStateBits])
      change (succFieldTM src dst copy).step d = _
      rw [step_of_tapeAction_move (succFieldTM src dst copy) d
        (q' := SuccPhase.done false) src 0 .pos hq (by simp [succFieldTM, hs])]
      apply Cfg.ext <;> simp [succFieldCfg, afterTag, optStateBits]
      · funext j z
        split_ifs <;> first | rfl | omega
    | some s =>
      have hs : d.workTapeSymbols src = some true := by
        simpa [Cfg.workTapeSymbols, optStateBits] using hw 0 (by simp [optStateBits])
      have first : (succFieldTM src dst copy).step d = afterTag src d := by
        simpa [afterTag] using step_of_tapeAction_move (succFieldTM src dst copy) d
          (q' := SuccPhase.unary) src 0 .pos hq (by simp [succFieldTM, hs])
      have reads : ∀ (r : ℕ) (hr : r < s.val + 1),
          (afterTag src d).workTapes src ((afterTag src d).workTapePos src + r) =
            some ((List.replicate s.val true ++ [false])[r]'(by simpa using hr)) := by
        intro r hr
        simpa [optStateBits, unaryFin, afterTag, Nat.cast_add, add_assoc, add_comm] using
          hw (r + 1) (by simp [optStateBits, unaryFin]; omega)
      have run := succFieldTM_unary_run src dst hne copy (afterTag src d) s.val rfl reads
        t (by simpa [optStateBits, unaryFin] using ht)
      rw [MultiTapeTM.runFrom_succ_eq_step, first]
      have phase : (t ≤ s.val) ↔ (t + 1 < (optStateBits (some s)).length) := by
        simp [optStateBits, unaryFin]; omega
      simpa [unaryFin, phase] using run

/-- Endpoint of the successor-field trace, at exactly the field's bit length. -/
lemma succFieldTM_run {m n : ℕ} {x : List Bool} (src dst : Fin m)
    (hne : src ≠ dst) (copy : Bool) (d : Cfg m Bool SuccPhase x)
    (s : Option (Fin n)) (hq : d.state = some .tag)
    (hw : ∀ (r : ℕ) (hr : r < (optStateBits s).length),
      d.workTapes src (d.workTapePos src + r) = some ((optStateBits s)[r]'hr)) :
    let word := (s.map unaryFin).getD []
    (succFieldTM src dst copy).runFrom d (optStateBits s).length =
      succFieldCfg src dst copy (afterTag src d) word word.length (.done s.isSome) := by
  have trace := succFieldTM_trace src dst hne copy d s hq hw _ le_rfl
  cases s <;> simpa [optStateBits, unaryFin] using trace

/-- Neither successor exit is visited before the field has been consumed. -/
lemma succFieldTM_cut {m n : ℕ} {x : List Bool} (src dst : Fin m)
    (hne : src ≠ dst) (copy : Bool) (d : Cfg m Bool SuccPhase x)
    (s : Option (Fin n)) (hq : d.state = some .tag)
    (hw : ∀ (r : ℕ) (hr : r < (optStateBits s).length),
      d.workTapes src (d.workTapePos src + r) = some ((optStateBits s)[r]'hr)) :
    ∀ t < (optStateBits s).length, ∀ b,
      ((succFieldTM src dst copy).runFrom d t).state ≠ some (.done b) := by
  intro t ht b
  rw [succFieldTM_trace src dst hne copy d s hq hw t (by omega)]
  by_cases hz : t = 0
  · simp [hz, hq]
  · simp [hz, ht, succFieldCfg]

/-- During a successor walk only the table head and an enabled destination
head advance. Every input position and the physical output prefix are fixed.
The exact formulas include both the halt tag and the last unary bit. -/
lemma succFieldTM_heads {m n : ℕ} {x : List Bool} (src dst : Fin m)
    (hne : src ≠ dst) (copy : Bool) (d : Cfg m Bool SuccPhase x)
    (s : Option (Fin n)) (hq : d.state = some .tag)
    (hw : ∀ (r : ℕ) (hr : r < (optStateBits s).length),
      d.workTapes src (d.workTapePos src + r) = some ((optStateBits s)[r]'hr)) :
    ∀ t ≤ (optStateBits s).length,
      let e := (succFieldTM src dst copy).runFrom d t
      e.inputPos = d.inputPos ∧ e.output = d.output ∧
      ∀ j, e.workTapePos j = d.workTapePos j +
        if j = src then (t : ℤ) else if copy = true ∧ j = dst then (t - 1 : ℕ) else 0 := by
  intro t ht
  rw [succFieldTM_trace src dst hne copy d s hq hw t ht]
  by_cases hz : t = 0
  · simp [hz]
  · simp only [if_neg hz]
    refine ⟨rfl, rfl, ?_⟩
    intro j
    by_cases hj : j = src
    · subst j
      simp [succFieldCfg, afterTag, hne]
      omega
    · simp [succFieldCfg, afterTag, hj]

/-- Phase family for a complete record skip. The fixed reader is stopped
before its payload step; the successor walker then consumes the variable tail. -/
def SkipPh (k : ℕ) : Bool → Type
  | false => RecordState k
  | true => SuccPhase

/-- The skipper reuses the inherited fixed reader and the trusted tail leaf. -/
private def skipPart (k : ℕ) : (b : Bool) → MultiTapeTM (k + 2) Bool (SkipPh k b)
  | false => recordTM k
  | true => succFieldTM (tableTape k) (summaryTape k) false

/-- Dispatch at the fixed-width boundary, without executing the action. -/
private def skipRed (k : ℕ) : (b : Bool) → SkipPh k b → Option (Σ b, SkipPh k b)
  | false, some (p, _) => if p.val = recordWidth k then some ⟨true, .tag⟩ else none
  | false, none => none
  | true, _ => none

/-- One whole-record skipper, including its variable successor field.
**Promotion candidate (host H2, record skipper; machine-library-design.md §14).**
Its one stationary dispatch step is explicit in the cost. -/
def recordSkipTM (k : ℕ) : MultiTapeTM (k + 2) Bool (Σ b, SkipPh k b) :=
  phaseHostTM (skipPart k) ⟨false, (recordTM k).q₀⟩ (skipRed k)

/-- The public phase-host theorem transfers the inherited prefix trace and
then takes its sole dispatch. The payload transition is never executed. -/
private lemma recordSkipTM_fixed {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (bits : Fin (recordWidth k) → Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ i : Fin (recordWidth k),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i)) :
    (∀ (u : ℕ) (hu : u ≤ recordWidth k),
      (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) u =
        (readCfg d bits u hu).mapState (Sigma.mk false)) ∧
    (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) (recordWidth k + 1) =
      (readCfg d bits (recordWidth k) le_rfl).mapState
        (fun _ => (⟨true, SuccPhase.tag⟩ : Σ b, SkipPh k b)) := by
  have step := phaseHostTM_run (skipPart k) ⟨false, (recordTM k).q₀⟩ (skipRed k)
    false d (readCfg d bits (recordWidth k) le_rfl) (recordWidth k)
    (recordTM_read d bits hq hw _ le_rfl)
    (some (⟨recordWidth k, by omega⟩, fun i => if i.val < recordWidth k then bits i else false))
    rfl ⟨true, .tag⟩ (by simp [skipRed]) (by
      intro u hu s hs
      change ((recordTM k).runFrom d u).state = some s at hs
      rw [recordTM_read d bits hq hw u (by omega)] at hs
      have he : s = some (⟨u, by omega⟩, fun i => if i.val < u then bits i else false) :=
        (Option.some.inj hs).symm
      subst s
      simp [skipRed, show u ≠ recordWidth k by omega])
  refine ⟨?_, step.2⟩
  intro u hu
  rw [step.1 u hu]
  exact congrArg (Cfg.mapState (Sigma.mk false)) (recordTM_read d bits hq hw u hu)

/-- Once dispatched, the host follows the successor walker literally.
This is an application of public state transport, not another host trace. -/
private lemma recordSkipTM_tail {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool SuccPhase x) (t : ℕ) :
    (recordSkipTM k).runFrom (d.mapState (Sigma.mk true)) t =
      ((succFieldTM (tableTape k) (summaryTape k) false).runFrom d t).mapState
        (Sigma.mk true) := by
  exact MultiTapeTM.runFrom_mapState_of_agreeOn
    (succFieldTM (tableTape k) (summaryTape k) false) (recordSkipTM k)
    ⟨(fun q => (⟨true, q⟩ : Σ b, SkipPh k b)), by
      intro a b h; cases h; rfl⟩ (fun _ => True)
    (fun _ _ _ _ => rfl) d t (fun _ _ _ _ => trivial)

/-- A trusted action record splits at exactly the inherited fixed width.
Only this elementary serialization identity is missing from the public
reader's lower-bound contract; it also fixes the skipper's exact cost. -/
private lemma actionBitsK_split {k n : ℕ} (a : Action k Bool (Fin (n + 1))) :
    ∃ bits : Fin (recordWidth k) → Bool,
      actionBitsK a = List.ofFn bits ++ optStateBits a.state := by
  let p := signBits a.inputTape ++ (List.finRange k).flatMap
    (fun i => optOptBoolBits (a.workTapes i).1 ++ signBits (a.workTapes i).2) ++ optBoolBits
      a.output
  have hl : p.length = recordWidth k := by
    have moveLen : ∀ s : SignType, (signBits s).length = 2 := by decide
    have writeLen : ∀ w : Option (Option Bool), (optOptBoolBits w).length = 2 := by decide
    have outputLen : ∀ w : Option Bool, (optBoolBits w).length = 2 := by decide
    simp [p, List.length_flatMap, moveLen, writeLen, outputLen, recordWidth]
    omega
  refine ⟨fun i => p[i.val]'(by rw [hl]; exact i.isLt), ?_⟩
  change p ++ _ = _
  congr 1
  apply List.ext_getElem
  · simpa using hl
  · intro i hi hi'
    simp only [List.getElem_ofFn]

/-- Split a literal, undelimited word read at an arbitrary physical origin.
This small indexing fact is shared by fixed fields, record lists and state groups. -/
private lemma reads_append (cells : ℤ → Option Bool) (p : ℤ) (a b : List Bool)
    (h : ∀ (r : ℕ) (hr : r < (a ++ b).length), cells (p + r) = some ((a ++ b)[r]'hr)) :
    (∀ (r : ℕ) (hr : r < a.length), cells (p + r) = some (a[r]'hr)) ∧
    (∀ (r : ℕ) (hr : r < b.length), cells (p + a.length + r) = some (b[r]'hr)) := by
  constructor
  · intro r hr
    have left := h r (by simp; omega)
    rw [List.getElem_append_left hr] at left
    exact left
  · intro r hr
    have right := h (a.length + r) (by simp; omega)
    rw [List.getElem_append_right (by omega)] at right
    simpa only [Nat.add_sub_cancel_left, Nat.cast_add, add_assoc] using right

/-- Expose the fixed and variable reads of one trusted record at an arbitrary
physical table position. The tape outside the record is unrestricted. -/
private lemma actionBitsK_reads {k n : ℕ} {S : Type} {x : List Bool}
    (d : Cfg (k + 2) Bool S x) (a : Action k Bool (Fin (n + 1)))
    (hw : ∀ (r : ℕ) (hr : r < (actionBitsK a).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((actionBitsK a)[r]'hr)) :
    ∃ bits : Fin (recordWidth k) → Bool,
      actionBitsK a = List.ofFn bits ++ optStateBits a.state ∧
      (actionBitsK a).length = recordWidth k + (optStateBits a.state).length ∧
      (∀ i : Fin (recordWidth k),
        d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i)) ∧
      (∀ (r : ℕ) (hr : r < (optStateBits a.state).length),
        d.workTapes (tableTape k) (d.workTapePos (tableTape k) + recordWidth k + r) =
          some ((optStateBits a.state)[r]'hr)) := by
  obtain ⟨bits, hb⟩ := actionBitsK_split a
  have hlen : (actionBitsK a).length = recordWidth k + (optStateBits a.state).length := by
    rw [hb, List.length_append, List.length_ofFn]
  have parts := reads_append (d.workTapes (tableTape k)) (d.workTapePos (tableTape k))
    (List.ofFn bits) (optStateBits a.state) (by simpa only [hb] using hw)
  refine ⟨bits, hb, hlen, ?_, ?_⟩
  · intro i
    have cell := parts.1 i.val (by simpa only [List.length_ofFn] using i.isLt)
    rw [List.getElem_ofFn] at cell
    exact cell
  · simpa only [List.length_ofFn] using parts.2

/-- A complete record is skipped silently. The exact elapsed time is its
bit length plus the single phase dispatch. All tape cells are framed, and
only the table head changes; the successor is not copied in skipping mode.

**Proof sketch.** Use the inherited prefix read and public phase dispatch,
then the successor-field endpoint. The two head advances add to the full
record length, including its variable unary tail. -/
lemma recordSkipTM_run {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (a : Action k Bool (Fin (n + 1)))
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ (r : ℕ) (hr : r < (actionBitsK a).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((actionBitsK a)[r]'hr)) :
    (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) ((actionBitsK a).length + 1) =
      { d.mapState (fun _ => (⟨true, SuccPhase.done a.state.isSome⟩ : Σ b, SkipPh k b)) with
        workTapePos := Function.update d.workTapePos (tableTape k)
          (d.workTapePos (tableTape k) + (actionBitsK a).length) } := by
  obtain ⟨bits, hb, hlen, hread, hfield⟩ := actionBitsK_reads d a hw
  let tail : Cfg (k + 2) Bool SuccPhase x :=
    (readCfg d bits (recordWidth k) le_rfl).mapState (fun _ => .tag)
  have htail : ∀ (r : ℕ) (hr : r < (optStateBits a.state).length),
      tail.workTapes (tableTape k) (tail.workTapePos (tableTape k) + r) =
        some ((optStateBits a.state)[r]'hr) := by
    intro r hr
    simpa only [tail, Cfg.mapState, readCfg, Function.update_self] using hfield r hr
  have dispatch := (recordSkipTM_fixed d bits hq hread).2
  have he : (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) (recordWidth k + 1) =
      tail.mapState (Sigma.mk true) := by
    exact dispatch
  rw [show (actionBitsK a).length + 1 = recordWidth k + 1 + (optStateBits a.state).length by
    omega,
    MultiTapeTM.runFrom_add, he, recordSkipTM_tail,
    succFieldTM_run _ _ (Ne.symm (summary_ne_table k)) false tail a.state (by rfl) htail]
  apply Cfg.ext
  · simp [succFieldCfg, tail, Cfg.mapState, readCfg, hq]
  · rfl
  · simp [succFieldCfg, tail, Cfg.mapState, afterTag, readCfg]
  · funext j
    by_cases hj : j = tableTape k
    · subst j
      rw [hlen]
      cases a.state <;>
        simp [succFieldCfg, afterTag, tail, Cfg.mapState, readCfg, optStateBits,
          unaryFin, Nat.cast_add, add_assoc] ; omega
    · simp [succFieldCfg, afterTag, tail, Cfg.mapState, readCfg, hj]
  · rfl

/-- Every prefix of a whole-record skip preserves the complete data frame.
At the sole dispatch the table head pauses; on every other transition it
advances one cell. Neither successor exit can occur before the full field.

**Proof sketch.** Split the horizon at the fixed reader's dispatch. The
first interval is `recordTM_read` transported by the public phase host;
the second is the trusted successor trace transported by public agreement. -/
private lemma recordSkipTM_path {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (bits : Fin (recordWidth k) → Bool) (s : Option (Fin n))
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ i : Fin (recordWidth k),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + i.val) = some (bits i))
    (hs : ∀ (r : ℕ) (hr : r < (optStateBits s).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + recordWidth k + r) =
        some ((optStateBits s)[r]'hr)) :
    ∀ u ≤ recordWidth k + 1 + (optStateBits s).length,
      let f := (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      (∀ j, f.workTapePos j = d.workTapePos j +
        if j = tableTape k then ((min u (recordWidth k) + (u - (recordWidth k + 1)) : ℕ) : ℤ)
        else (0 : ℤ)) ∧
      (u < recordWidth k + 1 + (optStateBits s).length →
        ∀ b, f.state ≠ some ⟨true, .done b⟩) := by
  intro u hu
  dsimp only
  obtain ⟨hprefix, dispatch⟩ := recordSkipTM_fixed d bits hq hw
  by_cases hpre : u ≤ recordWidth k
  · rw [hprefix u hpre]
    refine ⟨rfl, rfl, rfl, ?_, ?_⟩
    · intro j
      simp only [Cfg.mapState, readCfg]
      by_cases hj : j = tableTape k
      · subst j
        simp [min_eq_left hpre, Nat.sub_eq_zero_of_le (by omega : u ≤ recordWidth k + 1)]
      · simp [hj]
    · intro _ b
      simp [Cfg.mapState, readCfg]
  · let tail : Cfg (k + 2) Bool SuccPhase x :=
      (readCfg d bits (recordWidth k) le_rfl).mapState (fun _ => .tag)
    have htail : ∀ (r : ℕ) (hr : r < (optStateBits s).length),
        tail.workTapes (tableTape k) (tail.workTapePos (tableTape k) + r) =
          some ((optStateBits s)[r]'hr) := by
      intro r hr
      simpa only [tail, Cfg.mapState, readCfg, Function.update_self] using hs r hr
    have ht : u - (recordWidth k + 1) ≤ (optStateBits s).length := by omega
    have he : (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) u =
        ((succFieldTM (tableTape k) (summaryTape k) false).runFrom
          tail (u - (recordWidth k + 1))).mapState (Sigma.mk true) := by
      conv_lhs => rw [show u = (recordWidth k + 1) + (u - (recordWidth k + 1)) by omega]
      rw [MultiTapeTM.runFrom_add, dispatch]
      exact recordSkipTM_tail tail _
    rw [he]
    have hhead := succFieldTM_heads _ _ (Ne.symm (summary_ne_table k)) false tail s rfl htail _ ht
    refine ⟨hhead.1, ?_, hhead.2.1, ?_, ?_⟩
    · rw [succFieldTM_trace _ _ (Ne.symm (summary_ne_table k)) false tail s rfl htail _ ht]
      split_ifs <;> rfl
    · intro j
      change ((succFieldTM _ _ false).runFrom tail _).workTapePos j = _
      rw [hhead.2.2 j]
      by_cases hj : j = tableTape k
      · subst j
        simp [tail, Cfg.mapState, readCfg, min_eq_right (by omega : recordWidth k ≤ u),
          Nat.cast_add, add_assoc]
      · simp [tail, Cfg.mapState, readCfg, hj]
    · intro hlt b
      have hc := succFieldTM_cut _ _ (Ne.symm (summary_ne_table k)) false tail s rfl htail
        (u - (recordWidth k + 1)) (by omega) b
      intro bad
      apply hc
      exact Option.map_injective (f := fun q : SuccPhase => (⟨true, q⟩ : Σ b, SkipPh k b))
        (by intro a b h; cases h; rfl) bad

/-- The whole-record skipper's composition contract: a first live exit,
complete framing, exact head trace, and a linear cost in the whole record.
Both the prefix width and the finite controller depend on `k` alone. -/
lemma recordSkipTM_spec {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (a : Action k Bool (Fin (n + 1)))
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ (r : ℕ) (hr : r < (actionBitsK a).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((actionBitsK a)[r]'hr)) :
    (∀ u < (actionBitsK a).length + 1, ∀ b,
      ((recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) u).state ≠ some ⟨true, .done b⟩) ∧
    (∀ u ≤ (actionBitsK a).length + 1,
      let f := (recordSkipTM k).runFrom (d.mapState (Sigma.mk false)) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      (∀ j, f.workTapePos j = d.workTapePos j +
        if j = tableTape k then ((min u (recordWidth k) + (u - (recordWidth k + 1)) : ℕ) : ℤ)
        else 0)) := by
  obtain ⟨bits, _, hlen, hp, ht⟩ := actionBitsK_reads d a hw
  have run := recordSkipTM_path d bits a.state hq hp ht
  constructor
  · intro u hu b
    exact (run u (by omega)).2.2.2.2 (by omega) b
  · intro u hu
    exact ⟨(run u (by omega)).1, (run u (by omega)).2.1,
      (run u (by omega)).2.2.1, (run u (by omega)).2.2.2.1⟩

/-- Each skipper phase is finite, uniformly in the coded state count. -/
instance instFintypeSkipPh (k : ℕ) (b : Bool) : Fintype (SkipPh k b) := by
  cases b <;> dsimp [SkipPh] <;> infer_instance

/-- Equality of the two skipper phase carriers is decidable. -/
instance instDecidableEqSkipPh (k : ℕ) (b : Bool) : DecidableEq (SkipPh k b) := by
  cases b <;> dsimp [SkipPh] <;> infer_instance

/-- The selector's finite router retains the current read rank and a
within-group record countdown. The coded state itself stays on tape. -/
inductive SeekCtrl (k : ℕ) where
  | observe
  | advance (rank : Fin (3 ^ (k + 1)))
  | state (rank : Fin (3 ^ (k + 1)))
  | request (rank rest : Fin (3 ^ (k + 1))) (last : Bool)
  | done
  deriving DecidableEq, Fintype

/-- Selector routing. The unary state begins at cell two of the summary
auxiliary tape: cell zero is the summary and cell one its blank separator.
Observation and one additional move enter that state word; each true skips
one whole state block, and its false terminator selects the read rank. -/
private def seekRouterTM (k : ℕ) : MultiTapeTM (k + 2) Bool (SeekCtrl k) where
  q₀ := .observe
  tr := fun q inp w => match q with
    | .observe => tapeAction (summaryTape k) 0 none .pos
        (.advance (readRank inp (fun i => w (Fin.castAdd 2 i))))
    | .advance rank => tapeAction (summaryTape k) 0 none .pos (.state rank)
    | .state rank =>
        if w (summaryTape k) = some true then
          tapeAction (summaryTape k) 0 none .pos
            (.request rank ⟨3 ^ (k + 1) - 1, by
              have hp : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega); omega⟩ false)
        else inputMoveAction 0 (if hr : rank.val = 0 then .done
          else .request rank ⟨rank.val - 1, by have := rank.isLt; omega⟩ true)
    | .request rank rest last => inputMoveAction 0 (.request rank rest last)
    | .done => inputMoveAction 0 .done

/-- The continuation after one skipped record. `rest` counts records still
to skip after the current one, and `last` distinguishes the rank suffix
from an entire state group. -/
private def seekNext {k : ℕ} (rank rest : Fin (3 ^ (k + 1))) (last : Bool) : SeekCtrl k :=
  if hz : rest.val = 0 then (if last then .done else .state rank)
  else .request rank ⟨rest.val - 1, by have := rest.isLt; omega⟩ last

/-- Phase keys for the selector. `none` is its router; each finite countdown
key tags a call of the same one-record skipper. -/
abbrev SeekKey (k : ℕ) := Option (Fin (3 ^ (k + 1)) × Fin (3 ^ (k + 1)) × Bool)

/-- Finite state carriers of the selector's public phase host. -/
def SeekPh (k : ℕ) : SeekKey k → Type
  | none => SeekCtrl k
  | some _ => Σ b, SkipPh k b

/-- Selector phase machines: one router and the shared record skipper. -/
private def seekPart (k : ℕ) : (key : SeekKey k) → MultiTapeTM (k + 2) Bool (SeekPh k key)
  | none => seekRouterTM k
  | some _ => recordSkipTM k

/-- Only a requested call or a completed whole record dispatches. -/
private def seekRed (k : ℕ) : (key : SeekKey k) → SeekPh k key → Option (Σ key, SeekPh k key)
  | none, .request rank rest last => some ⟨some (rank, rest, last), (recordSkipTM k).q₀⟩
  | some (rank, rest, last), ⟨true, .done _⟩ => some ⟨none, seekNext rank rest last⟩
  | _, _ => none

/-- The selector is one finite physical machine, independent of the number
of coded states. The only loop is through the public phase host and the
proved whole-record skipper. -/
private def seekTM (k : ℕ) : MultiTapeTM (k + 2) Bool (Σ key, SeekPh k key) :=
  phaseHostTM (seekPart k) ⟨none, .observe⟩ (seekRed k)

/-- Reuse the reader's framed configuration as the selector's data, with
only the finite router control renamed. -/
private def seekCfg {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x) (q : SeekCtrl k) :
    Cfg (k + 2) Bool (Σ key, SeekPh k key) x :=
  d.mapState (fun _ => ⟨none, q⟩)

/-- One requested whole-record call, with its two public-host dispatches.
No intermediate point has the selector's final control. All data are framed,
and every non-table head stays fixed throughout this call.

**Proof sketch.** The first dispatch enters the shared skipper. Its first-exit
contract discharges the public phase-host guard, and the host theorem provides
both the complete inner prefix and the final dispatch. -/
private lemma seekTM_request {k n : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (a : Action k Bool (Fin (n + 1))) (rank rest : Fin (3 ^ (k + 1))) (last : Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ (r : ℕ) (hr : r < (actionBitsK a).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((actionBitsK a)[r]'hr)) :
    (seekTM k).runFrom (seekCfg d (.request rank rest last)) ((actionBitsK a).length + 3) =
      seekCfg ({ d with
          workTapePos := Function.update d.workTapePos (tableTape k)
            (d.workTapePos (tableTape k) + (actionBitsK a).length) }) (seekNext rank rest last) ∧
    (∀ u < (actionBitsK a).length + 3,
      ((seekTM k).runFrom (seekCfg d (.request rank rest last)) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ (actionBitsK a).length + 3,
      let f := (seekTM k).runFrom (seekCfg d (.request rank rest last)) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ tableTape k → f.workTapePos j = d.workTapePos j) := by
  let key : SeekKey k := some (rank, rest, last)
  let start := d.mapState (@Sigma.mk Bool (SkipPh k) false)
  let finish : Cfg (k + 2) Bool (Σ b, SkipPh k b) x :=
    { d.mapState (fun _ => ⟨true, SuccPhase.done a.state.isSome⟩) with
      workTapePos := Function.update d.workTapePos (tableTape k)
        (d.workTapePos (tableTape k) + (actionBitsK a).length) }
  have hrun : (seekPart k key).runFrom start ((actionBitsK a).length + 1) = finish :=
    recordSkipTM_run d a hq hw
  have hfirst : (seekTM k).step (seekCfg d (.request rank rest last)) =
      start.mapState (Sigma.mk key) := by
    rw [step_of_inputMoveAction (seekTM k) (seekCfg d (.request rank rest last))
      (q := ⟨none, .request rank rest last⟩) (q' := ⟨key, (recordSkipTM k).q₀⟩) 0
      (by simp [seekCfg, Cfg.mapState, hq]) rfl]
    apply Cfg.ext <;> simp [seekCfg, Cfg.mapState, hq, start, key, recordSkipTM, phaseHostTM]
  have advance (u : ℕ) := MultiTapeTM.runFrom_succ_eq_step (tm := seekTM k)
    (cfg := seekCfg d (.request rank rest last)) (t := u)
  rw [hfirst] at advance
  have hosted := phaseHostTM_run (seekPart k) ⟨none, .observe⟩ (seekRed k)
    key start finish ((actionBitsK a).length + 1) hrun ⟨true, .done a.state.isSome⟩
    (by simp [finish, Cfg.mapState, hq]) ⟨none, seekNext rank rest last⟩ rfl (by
      intro u hu s hs
      have hc := (recordSkipTM_spec d a hq hw).1 u hu
      change ((recordSkipTM k).runFrom start u).state = some s at hs
      obtain ⟨b, q⟩ := s
      cases b with
      | false => rfl
      | true =>
        cases q with
        | tag => rfl
        | unary => rfl
        | done b => exact False.elim (hc b hs))
  have hpre (u : ℕ) (hu : u ≤ (actionBitsK a).length + 1) :
      (seekTM k).runFrom (seekCfg d (.request rank rest last)) (u + 1) =
        ((recordSkipTM k).runFrom start u).mapState (Sigma.mk key) := by
    rw [advance u]
    exact hosted.1 u hu
  have hend : (seekTM k).runFrom (seekCfg d (.request rank rest last))
      ((actionBitsK a).length + 3) =
      seekCfg ({ d with
          workTapePos := Function.update d.workTapePos (tableTape k)
            (d.workTapePos (tableTape k) + (actionBitsK a).length) }) (seekNext rank rest
              last) := by
    rw [show (actionBitsK a).length + 3 = ((actionBitsK a).length + 1 + 1) + 1 by omega,
      advance, hosted.2]
    apply Cfg.ext <;> simp [finish, seekCfg, Cfg.mapState, hq]
  refine ⟨hend, ?_, ?_⟩
  · intro u hu
    cases u with
    | zero => simp [seekCfg, Cfg.mapState, hq]
    | succ u =>
      rw [hpre u (by omega)]
      cases ((recordSkipTM k).runFrom start u).state <;> simp [Cfg.mapState, key]
  · intro u hu
    by_cases he : u = (actionBitsK a).length + 3
    · subst u
      rw [hend]
      exact ⟨rfl, rfl, rfl, fun j hj => by simp [seekCfg, Cfg.mapState, hj]⟩
    · cases u with
      | zero => exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
      | succ u =>
        rw [hpre u (by omega)]
        have hf := (recordSkipTM_spec d a hq hw).2 u (by omega)
        refine ⟨hf.1, hf.2.1, hf.2.2.1, ?_⟩
        intro j hj
        simpa [Cfg.mapState, hj] using hf.2.2.2 j

/-- Cost of skipping a sequence of whole records through the selector:
one skipper dispatch and two selector dispatches per record. -/
private def seekCost {k n : ℕ} (records : List (Action k Bool (Fin (n + 1)))) : ℕ :=
  (records.flatMap actionBitsK).length + 3 * records.length

/-- A finite within-group countdown traverses exactly its list of whole
records. The table contents and the entire non-table data frame are preserved,
and the final router control is first reached only after the last record.

**Proof sketch.** Induct over the nonempty record list. The one-record public
host composition advances by that record's full bit length. The suffix reads
are the right-hand part of the same flat-map word. Add the exact costs; split
any intermediate horizon at the first completed call for the cut and frame. -/
private lemma seekTM_many {k n : ℕ} {x : List Bool}
    (records : List (Action k Bool (Fin (n + 1)))) (hne : records ≠ [])
    (hlen : records.length ≤ 3 ^ (k + 1))
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (rank : Fin (3 ^ (k + 1))) (last : Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hw : ∀ (r : ℕ) (hr : r < (records.flatMap actionBitsK).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((records.flatMap actionBitsK)[r]'hr)) :
    let entry := SeekCtrl.request rank ⟨records.length - 1, by
      have := List.length_pos_iff.mpr hne; omega⟩ last
    let final := if last then SeekCtrl.done else .state rank
    (seekTM k).runFrom (seekCfg d entry) (seekCost records) =
      seekCfg ({d with
        workTapePos := Function.update d.workTapePos (tableTape k)
          (d.workTapePos (tableTape k) + (records.flatMap actionBitsK).length)}) final ∧
    (∀ u < seekCost records,
      ((seekTM k).runFrom (seekCfg d entry) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ seekCost records,
      let f := (seekTM k).runFrom (seekCfg d entry) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ tableTape k → f.workTapePos j = d.workTapePos j) := by
  induction records generalizing d with
  | nil => exact False.elim (hne rfl)
  | cons a as ih =>
    dsimp only
    have parts := reads_append (d.workTapes (tableTape k)) (d.workTapePos (tableTape k))
      (actionBitsK a) (as.flatMap actionBitsK) (by simpa only [List.flatMap_cons] using hw)
    have hfirstReads := parts.1
    have one := seekTM_request d a rank
      ⟨(a :: as).length - 1, by simp only [List.length_cons] at *; omega⟩ last hq hfirstReads
    cases as with
    | nil =>
      simpa only [seekCost, List.flatMap_cons, List.flatMap_nil, List.append_nil,
        List.length_cons, List.length_nil, Nat.zero_add, Nat.add_zero, Nat.mul_one,
        seekNext, Nat.add_sub_cancel, ↓reduceDIte, Nat.sub_zero] using one
    | cons b bs =>
      have bound : bs.length + 2 ≤ 3 ^ (k + 1) := by
        simpa only [List.length_cons, Nat.add_assoc] using hlen
      let ds : Cfg (k + 2) Bool (RecordState k) x :=
        {d with
          workTapePos := Function.update d.workTapePos (tableTape k)
            (d.workTapePos (tableTape k) + (actionBitsK a).length)}
      have htailReads : ∀ (r : ℕ) (hr : r < ((b :: bs).flatMap actionBitsK).length),
          ds.workTapes (tableTape k) (ds.workTapePos (tableTape k) + r) =
            some (((b :: bs).flatMap actionBitsK)[r]'hr) := by
        simpa only [ds, Function.update_self] using parts.2
      have recur := ih (by simp) (by simp only [List.length_cons] at *; omega)
        ds hq htailReads
      have hnext : seekNext rank ⟨(a :: b :: bs).length - 1, by
          simp only [List.length_cons] at *; omega⟩ last =
          .request rank ⟨(b :: bs).length - 1, by simp only [List.length_cons] at *; omega⟩ last
            := by
        simp only [seekNext, List.length_cons, Nat.add_sub_cancel, Nat.add_eq_zero_iff,
          Nat.one_ne_zero, and_false, ↓reduceDIte]
      rw [hnext] at one
      have hcost : seekCost (a :: b :: bs) = (actionBitsK a).length + 3 + seekCost (b :: bs)
        := by
        simp only [seekCost, List.flatMap_cons, List.length_append, List.length_cons]
        omega
      have hend : (seekTM k).runFrom (seekCfg d (.request rank
          ⟨(a :: b :: bs).length - 1, by simp only [List.length_cons] at *; omega⟩ last))
          (seekCost (a :: b :: bs)) =
          seekCfg ({d with
            workTapePos := Function.update d.workTapePos (tableTape k)
              (d.workTapePos (tableTape k) + ((a :: b :: bs).flatMap actionBitsK).length)})
            (if last then .done else .state rank) := by
        rw [hcost, MultiTapeTM.runFrom_add, one.1]
        change (seekTM k).runFrom (seekCfg ds _) _ = _
        rw [recur.1]
        apply Cfg.ext <;> try rfl
        funext j
        by_cases hj : j = tableTape k
        · subst j
          simp [seekCfg, Cfg.mapState, ds, List.length_flatMap, List.flatMap_cons,
            List.length_append, Nat.cast_add, add_assoc]
        · simp [seekCfg, Cfg.mapState, ds, hj]
      refine ⟨hend, ?_, ?_⟩
      · intro u hu
        by_cases hl : u < (actionBitsK a).length + 3
        · exact one.2.1 u hl
        · have he : u = ((actionBitsK a).length + 3) + (u - ((actionBitsK a).length + 3)) := by
            omega
          rw [he, MultiTapeTM.runFrom_add, one.1]
          exact recur.2.1 _ (by omega)
      · intro u hu
        by_cases hl : u ≤ (actionBitsK a).length + 3
        · exact one.2.2 u hl
        · have he : u = ((actionBitsK a).length + 3) + (u - ((actionBitsK a).length + 3)) := by
            omega
          rw [he, MultiTapeTM.runFrom_add, one.1]
          have hf := recur.2.2 (u - ((actionBitsK a).length + 3)) (by omega)
          refine ⟨hf.1, hf.2.1, hf.2.2.1, ?_⟩
          intro j hj
          simpa only [ds, Function.update_of_ne hj] using hf.2.2.2 j hj

/-- A unary-state router step either requests one complete state group and
advances the state head, or requests the finite read-rank suffix. -/
private lemma seekTM_stateStep {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x) (rank : Fin (3 ^ (k + 1))) (b : Bool)
    (hq : d.state = some (recordTM k).q₀)
    (hs : d.workTapeSymbols (summaryTape k) = some b) :
    (seekTM k).step (seekCfg d (.state rank)) =
      if b then
        seekCfg ({d with
          workTapePos := Function.update d.workTapePos (summaryTape k)
            (d.workTapePos (summaryTape k) + 1)})
          (.request rank ⟨3 ^ (k + 1) - 1, by
            have hp : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega); omega⟩ false)
      else seekCfg d (if _hr : rank.val = 0 then .done
        else .request rank ⟨rank.val - 1, by have := rank.isLt; omega⟩ true) := by
  simp only [Cfg.workTapeSymbols] at hs
  cases b <;>
    apply Cfg.ext <;>
    simp [MultiTapeTM.step, seekTM, seekCfg, Cfg.mapState, phaseHostTM,
      seekRed, seekPart, seekRouterTM, hq, Cfg.workTapeSymbols, hs, tapeAction,
        inputMoveAction, Action.mapState]
  all_goals
    funext j
    by_cases hj : j = summaryTape k
    · subst j; simp
    · simp [hj]

/-- At the unary terminator, the selector skips exactly the read rank.
This includes rank zero: the router's single silent step exits immediately. -/
private lemma seekTM_rank {k n : ℕ} {x : List Bool}
    (records : List (Action k Bool (Fin (n + 1))))
    (d : Cfg (k + 2) Bool (RecordState k) x) (rank : Fin (3 ^ (k + 1)))
    (hlen : records.length = rank.val)
    (hq : d.state = some (recordTM k).q₀)
    (hs : d.workTapeSymbols (summaryTape k) = some false)
    (hw : ∀ (r : ℕ) (hr : r < (records.flatMap actionBitsK).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((records.flatMap actionBitsK)[r]'hr)) :
    (seekTM k).runFrom (seekCfg d (.state rank)) (1 + seekCost records) =
      seekCfg ({d with
        workTapePos := Function.update d.workTapePos (tableTape k)
          (d.workTapePos (tableTape k) + (records.flatMap actionBitsK).length)}) .done ∧
    (∀ u < 1 + seekCost records,
      ((seekTM k).runFrom (seekCfg d (.state rank)) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ 1 + seekCost records,
      let f := (seekTM k).runFrom (seekCfg d (.state rank)) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ tableTape k → f.workTapePos j = d.workTapePos j) := by
  have step := seekTM_stateStep d rank false hq hs
  simp only [Bool.false_eq_true, ↓reduceIte] at step
  by_cases hn : records = []
  · subst records
    have hz : rank.val = 0 := hlen.symm
    simp only [hz, ↓reduceDIte] at step
    have he : (seekTM k).runFrom (seekCfg d (.state rank)) 1 = seekCfg d .done := step
    simp only [seekCost, List.flatMap_nil, List.length_nil, Nat.mul_zero,
      Nat.cast_zero, add_zero, Function.update_eq_self]
    refine ⟨he, ?_, ?_⟩
    · intro u hu
      have hu0 : u = 0 := by omega
      simp [hu0, seekCfg, Cfg.mapState, hq]
    · intro u hu
      rcases Nat.le_one_iff_eq_zero_or_eq_one.mp hu with rfl | rfl
      · exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
      · rw [he]
        exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
  · have hp : 0 < rank.val := by rw [← hlen]; exact List.length_pos_iff.mpr hn
    simp only [show rank.val ≠ 0 by omega, ↓reduceDIte] at step
    have spec := seekTM_many records hn (by rw [hlen]; exact Nat.le_of_lt rank.isLt)
      d rank true hq hw
    dsimp only at spec
    simp only [↓reduceIte] at spec
    have hm : (⟨rank.val - 1, by have := rank.isLt; omega⟩ : Fin (3 ^ (k + 1))) =
        ⟨records.length - 1, by have := rank.isLt; omega⟩ := Fin.ext (by dsimp only; omega)
    rw [hm] at step
    have advance (u : ℕ) := MultiTapeTM.runFrom_succ_eq_step (tm := seekTM k)
      (cfg := seekCfg d (.state rank)) (t := u)
    rw [step] at advance
    refine ⟨?_, ?_, ?_⟩
    · rw [Nat.add_comm 1, advance]
      exact spec.1
    · intro u hu
      cases u with
      | zero => simp [seekCfg, Cfg.mapState, hq]
      | succ u =>
        rw [advance u]
        exact spec.2.1 u (by omega)
    · intro u hu
      cases u with
      | zero => exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
      | succ u =>
        rw [advance u]
        exact spec.2.2 u (by omega)

/-- The two auxiliary head advances accumulated by lookup. All mirrored
heads and every tape cell remain in the original arbitrary frame. -/
private def seekShift {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x) (table state : ℕ) :
    Cfg (k + 2) Bool (RecordState k) x :=
  {d with workTapePos := fun j => d.workTapePos j +
    if j = tableTape k then (table : ℤ) else if j = summaryTape k then (state : ℤ) else 0}

/-- Unary-state lookup traverses one complete group per true and then the
read-rank suffix. The result is the selected record boundary, with the state
head on its false terminator. This is a single physical selector for all
state counts and arbitrary source data frames.

**Proof sketch.** Induct on the list of complete state groups. A true takes
one state-head step and invokes the proved finite countdown over that group.
Its returning exit continues the induction. At the false terminator use the
rank contract, including rank zero. Costs are the exact sum of these calls. -/
private lemma seekTM_groups {k n : ℕ} {x : List Bool}
    (groups : List (List (Action k Bool (Fin (n + 1)))))
    (rest : List (Action k Bool (Fin (n + 1))))
    (rank : Fin (3 ^ (k + 1)))
    (hg : ∀ g ∈ groups, g.length = 3 ^ (k + 1)) (hrank : rest.length = rank.val)
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (hq : d.state = some (recordTM k).q₀)
    (hs : ∀ r ≤ groups.length,
      d.workTapes (summaryTape k) (d.workTapePos (summaryTape k) + r) =
        some (decide (r < groups.length)))
    (hw : ∀ (r : ℕ) (hr : r < ((groups.flatten ++ rest).flatMap actionBitsK).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some (((groups.flatten ++ rest).flatMap actionBitsK)[r]'hr)) :
    let records := groups.flatten ++ rest
    let T := groups.length + 1 + seekCost records
    (seekTM k).runFrom (seekCfg d (.state rank)) T =
      seekCfg (seekShift d (records.flatMap actionBitsK).length groups.length) .done ∧
    (∀ u < T, ((seekTM k).runFrom (seekCfg d (.state rank)) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ T,
      let f := (seekTM k).runFrom (seekCfg d (.state rank)) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ tableTape k → j ≠ summaryTape k → f.workTapePos j = d.workTapePos j) := by
  induction groups generalizing d with
  | nil =>
    have hs0 : d.workTapeSymbols (summaryTape k) = some false := by
      simpa [Cfg.workTapeSymbols] using hs 0 (by simp)
    have run := seekTM_rank rest d rank hrank hq hs0 (by simpa using hw)
    dsimp only
    simp only [List.flatten_nil, List.nil_append, List.length_nil, Nat.zero_add]
    refine ⟨run.1.trans ?_, run.2.1, ?_⟩
    · apply Cfg.ext <;> try rfl
      funext j
      by_cases hj : j = tableTape k <;> simp [seekCfg, Cfg.mapState, seekShift, hj]
    · intro u hu
      have frame := run.2.2 u hu
      exact ⟨frame.1, frame.2.1, frame.2.2.1, fun j hj _ => frame.2.2.2 j hj⟩
  | cons g gs ih =>
    have glen := hg g (by simp)
    have gp : g ≠ [] := by
      have pos : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega)
      intro bad; simp [bad] at glen; omega
    have parts := reads_append (d.workTapes (tableTape k)) (d.workTapePos (tableTape k))
      (g.flatMap actionBitsK) ((gs.flatten ++ rest).flatMap actionBitsK)
      (by simpa only [List.flatten_cons, List.flatMap_append, List.append_assoc] using hw)
    let next : Cfg (k + 2) Bool (RecordState k) x :=
      {d with
        workTapePos := Function.update d.workTapePos (summaryTape k)
          (d.workTapePos (summaryTape k) + 1)}
    have hs0 : d.workTapeSymbols (summaryTape k) = some true := by
      simpa [Cfg.workTapeSymbols] using hs 0 (by simp)
    have step := seekTM_stateStep d rank true hq hs0
    simp only [↓reduceIte] at step
    have calls := seekTM_many g gp (by omega) next rank false hq (by
      simpa [next, Function.update_of_ne (Ne.symm (summary_ne_table k))] using parts.1)
    dsimp only at calls
    simp only [Bool.false_eq_true, ↓reduceIte] at calls
    have count : (⟨3 ^ (k + 1) - 1, by
        have p : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega); omega⟩ : Fin (3 ^ (k + 1))) =
        ⟨g.length - 1, by have p := List.length_pos_iff.mpr gp; omega⟩ := by
      apply Fin.ext; dsimp only; omega
    rw [count] at step
    have advance (u : ℕ) := MultiTapeTM.runFrom_succ_eq_step (tm := seekTM k)
      (cfg := seekCfg d (.state rank)) (t := u)
    rw [step] at advance
    let later := seekShift d (g.flatMap actionBitsK).length 1
    have seam : (seekTM k).runFrom (seekCfg d (.state rank)) (1 + seekCost g) =
        seekCfg later (.state rank) := by
      rw [Nat.add_comm 1, advance, calls.1]
      apply Cfg.ext <;> try rfl
      funext j
      by_cases ht : j = tableTape k
      · subst j
        simp [seekCfg, Cfg.mapState, next, later, seekShift, Ne.symm (summary_ne_table k)]
      · by_cases hs : j = summaryTape k
        · subst j
          simp [seekCfg, Cfg.mapState, next, later, seekShift, summary_ne_table k]
        · simp [seekCfg, Cfg.mapState, next, later, seekShift, ht, hs]
    have laterState : ∀ r ≤ gs.length,
        later.workTapes (summaryTape k) (later.workTapePos (summaryTape k) + r) =
          some (decide (r < gs.length)) := by
      intro r hr
      have h := hs (r + 1) (by simp; omega)
      simpa only [later, seekShift, summary_ne_table k, ↓reduceIte,
        List.length_cons, Nat.add_lt_add_iff_right, Nat.cast_add, Nat.cast_one, add_assoc,
        add_comm (r : ℤ) 1] using h
    have laterReads : ∀ (r : ℕ) (hr : r < ((gs.flatten ++ rest).flatMap actionBitsK).length),
        later.workTapes (tableTape k) (later.workTapePos (tableTape k) + r) =
          some (((gs.flatten ++ rest).flatMap actionBitsK)[r]'hr) := by
      simpa only [later, seekShift, ↓reduceIte] using parts.2
    have recur := ih (fun h hh => hg h (by simp [hh])) later hq laterState laterReads
    dsimp only at recur ⊢
    have cost : (g :: gs).length + 1 + seekCost (((g :: gs).flatten ++ rest)) =
        (1 + seekCost g) + (gs.length + 1 + seekCost (gs.flatten ++ rest)) := by
      simp only [seekCost, List.length_cons, List.flatten_cons, List.flatMap_append,
        List.length_append]
      omega
    refine ⟨?_, ?_, ?_⟩
    · rw [cost, MultiTapeTM.runFrom_add, seam, recur.1]
      apply Cfg.ext <;> try rfl
      funext j
      by_cases ht : j = tableTape k
      · subst j
        simp [seekCfg, Cfg.mapState, seekShift, later, List.flatten_cons,
          List.flatMap_append, List.length_append, Nat.cast_add, add_assoc]
      · by_cases hj : j = summaryTape k
        · subst j
          simp [seekCfg, Cfg.mapState, seekShift, later, summary_ne_table k,
            Nat.cast_add, add_assoc, add_comm, List.length_cons]
        · simp [seekCfg, Cfg.mapState, seekShift, later, ht, hj]
    · intro u hu
      by_cases first : u < 1 + seekCost g
      · cases u with
        | zero => simp [seekCfg, Cfg.mapState, hq]
        | succ u =>
          rw [advance u]
          exact calls.2.1 u (by omega)
      · rw [show u = (1 + seekCost g) + (u - (1 + seekCost g)) by omega,
          MultiTapeTM.runFrom_add, seam]
        exact recur.2.1 _ (by omega)
    · intro u hu
      by_cases first : u ≤ 1 + seekCost g
      · cases u with
        | zero => exact ⟨rfl, rfl, rfl, fun _ _ _ => rfl⟩
        | succ u =>
          rw [advance u]
          have frame := calls.2.2 u (by omega)
          refine ⟨frame.1, frame.2.1, frame.2.2.1, ?_⟩
          intro j ht hs
          simpa only [next, Function.update_of_ne hs] using frame.2.2.2 j ht
      · rw [show u = (1 + seekCost g) + (u - (1 + seekCost g)) by omega,
          MultiTapeTM.runFrom_add, seam]
        have frame := recur.2.2 (u - (1 + seekCost g)) (by omega)
        refine ⟨frame.1, frame.2.1, frame.2.2.1, ?_⟩
        intro j ht hs
        simpa only [later, seekShift, if_neg ht, if_neg hs, add_zero] using frame.2.2.2 j ht hs

/-- Capture the finite read rank and enter the unary state word in two
silent moves of the auxiliary head. All mirrored heads stay fixed. -/
private lemma seekTM_observe {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (hq : d.state = some (recordTM k).q₀) :
    let rank := readRank d.inputSymbol (fun i => d.workTapeSymbols (Fin.castAdd 2 i))
    (seekTM k).runFrom (seekCfg d .observe) 2 =
      seekCfg (seekShift d 0 2) (.state rank) ∧
    (∀ u < 2, ((seekTM k).runFrom (seekCfg d .observe) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ 2,
      let f := (seekTM k).runFrom (seekCfg d .observe) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ summaryTape k → f.workTapePos j = d.workTapePos j) := by
  let rank := readRank d.inputSymbol (fun i => d.workTapeSymbols (Fin.castAdd 2 i))
  let next : Cfg (k + 2) Bool (RecordState k) x :=
    {d with
      workTapePos := Function.update d.workTapePos (summaryTape k)
        (d.workTapePos (summaryTape k) + 1)}
  have first : (seekTM k).step (seekCfg d .observe) = seekCfg next (.advance rank) := by
    rw [step_of_tapeAction_move (seekTM k) (seekCfg d .observe)
      (q := ⟨none, .observe⟩) (q' := ⟨none, .advance rank⟩) (summaryTape k) 0 .pos
      (by simp [seekCfg, Cfg.mapState, hq]) (by rfl)]
    apply Cfg.ext <;> simp [seekCfg, Cfg.mapState, next, hq]
  have second : (seekTM k).step (seekCfg next (.advance rank)) =
      seekCfg (seekShift d 0 2) (.state rank) := by
    rw [step_of_tapeAction_move (seekTM k) (seekCfg next (.advance rank))
      (q := ⟨none, .advance rank⟩) (q' := ⟨none, .state rank⟩) (summaryTape k) 0 .pos
      (by simp [seekCfg, Cfg.mapState, next, hq]) (by rfl)]
    apply Cfg.ext <;> try simp [seekCfg, Cfg.mapState, next, hq, seekShift]
    funext j
    by_cases hj : j = summaryTape k
    · subst j
      simp [summary_ne_table k, add_assoc]
    · by_cases ht : j = tableTape k
      · subst j
        simp [Ne.symm (summary_ne_table k)]
      · simp [hj, ht]
  have run : (seekTM k).runFrom (seekCfg d .observe) 2 =
      seekCfg (seekShift d 0 2) (.state rank) := by
    change (seekTM k).step ((seekTM k).step _) = _
    rw [first, second]
  refine ⟨run, ?_, ?_⟩
  · intro u hu
    rcases (show u = 0 ∨ u = 1 by omega) with rfl | rfl
    · simp [seekCfg, Cfg.mapState, hq]
    · change ((seekTM k).step _).state ≠ _
      rw [first]
      simp [seekCfg, Cfg.mapState, next, hq]
  · intro u hu
    rcases (show u = 0 ∨ u = 1 ∨ u = 2 by omega) with rfl | rfl | rfl
    · exact ⟨rfl, rfl, rfl, fun _ _ => rfl⟩
    · rw [show (seekTM k).runFrom (seekCfg d .observe) 1 = seekCfg next (.advance rank)
      from first]
      exact ⟨rfl, rfl, rfl, fun j hj => by simp [seekCfg, Cfg.mapState, next, hj]⟩
    · rw [run]
      refine ⟨rfl, rfl, rfl, ?_⟩
      intro j hj
      simp [seekCfg, Cfg.mapState, seekShift, hj]

/-- One state's ordered action list, before serialization. -/
private def readActions {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (q : Fin (n + 1)) : List (Action k Bool (Fin (n + 1))) :=
  (workReadsK k).map (tr q none) ++
    ((workReadsK k).map (tr q (some false)) ++ (workReadsK k).map (tr q (some true)))

/-- Every state group has the same finite number of read vectors. -/
private lemma readActions_length {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (q : Fin (n + 1)) : (readActions tr q).length = 3 ^ (k + 1) := by
  simp [readActions, length_workReadsK, Nat.pow_succ]
  omega

/-- The finite rank used by the physical router selects the serializer's
actual input/work-read record, including the tape-zero-first work ordering. -/
private lemma readActions_at {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (q : Fin (n + 1)) (inp : Option Bool) (w : Fin k → Option Bool) :
    (readActions tr q)[(readRank inp w).val]'(by
      rw [readActions_length]; exact (readRank inp w).isLt) = tr q inp w := by
  have hidx : (workReadsK k).idxOf w < (workReadsK k).length :=
    List.idxOf_lt_length_iff.mpr (mem_workReadsK k w)
  have hmap (a : Option Bool) :
      ((workReadsK k).map (tr q a))[(workReadsK k).idxOf w]'(by simpa using hidx) = tr q a w := by
    rw [List.getElem_map, List.getElem_idxOf hidx]
  have lens (a : Option Bool) : ((workReadsK k).map (tr q a)).length = 3 ^ k := by
    simp only [List.length_map, length_workReadsK]
  cases inp with
  | none =>
    simp only [readActions,
      readRank, inputRank, workRank, Fin.val_zero, Nat.zero_mul, Nat.zero_add]
    rw [List.getElem_append_left (by simpa only [List.length_map] using hidx)]
    exact hmap none
  | some b =>
    cases b with
    | false =>
      simp only [readActions,
        readRank, inputRank, workRank, Fin.val_one, Nat.one_mul]
      rw [List.getElem_append_right (by rw [lens]; omega)]
      simp only [lens, Nat.add_sub_cancel_left]
      rw [List.getElem_append_left (by simpa only [List.length_map] using hidx)]
      exact hmap (some false)
    | true =>
      simp only [readActions,
        readRank, inputRank, workRank]
      rw [List.getElem_append_right (by rw [lens]; omega)]
      simp only [lens]
      rw [List.getElem_append_right (by rw [lens]; omega)]
      have hi : (2 * 3 ^ k + (workReadsK k).idxOf w - 3 ^ k - 3 ^ k) = (workReadsK k).idxOf w :=
        by omega
      simpa only [lens, show ((2 : Fin 3) : ℕ) = 2 from rfl, hi] using hmap (some true)

/-- The public table writer is the serialization of these state groups. -/
private lemma tableBitsK_groups {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1))) :
    tableBitsK tr = ((List.finRange (n + 1)).flatMap (readActions tr)).flatMap actionBitsK := by
  simp only [tableBitsK, readActions, List.flatMap_assoc, List.flatMap_map,
    List.flatMap_cons, List.flatMap_nil, List.append_nil, List.flatMap_append]

/-- Exact decomposition of the public table around a source transition.
The selector skips `q` complete state groups followed by the captured read
rank. This identifies the next record without any assumption about constant
successor widths. -/
private lemma tableBitsK_cut {k n : ℕ}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (q : Fin (n + 1)) (inp : Option Bool) (w : Fin k → Option Bool) :
    ∃ (groups : List (List (Action k Bool (Fin (n + 1)))))
        (rest : List (Action k Bool (Fin (n + 1)))) (tail : List Bool),
      groups.length = q.val ∧
      (∀ g ∈ groups, g.length = 3 ^ (k + 1)) ∧
      rest.length = (readRank inp w).val ∧
      tableBitsK tr =
        ((groups.flatten ++ rest).flatMap actionBitsK) ++ actionBitsK (tr q inp w) ++ tail := by
  let states := List.finRange (n + 1)
  let r := (readRank inp w).val
  let reads := readActions tr q
  have qs : states = states.take q.val ++ q :: states.drop (q.val + 1) := by
    have h := List.take_append_drop q.val states
    rw [List.drop_eq_getElem_cons (by simp [states])] at h
    simpa only [states, List.getElem_finRange, Fin.eta] using h.symm
  have rlt : r < reads.length := by
    dsimp [r, reads]
    rw [readActions_length]
    exact (readRank inp w).isLt
  have rs : reads = reads.take r ++ tr q inp w :: reads.drop (r + 1) := by
    have h := List.take_append_drop r reads
    rw [List.drop_eq_getElem_cons rlt] at h
    simpa only [reads, r, readActions_at] using h.symm
  refine ⟨(states.take q.val).map (readActions tr), reads.take r,
    (reads.drop (r + 1) ++ (states.drop (q.val + 1)).flatMap (readActions tr)).flatMap
      actionBitsK,
    ?_, ?_, ?_, ?_⟩
  · simp only [List.length_map, List.length_take, states, List.length_finRange]
    exact min_eq_left (by omega)
  · intro g hg
    obtain ⟨a, _, rfl⟩ := List.mem_map.mp hg
    exact readActions_length tr a
  · simp only [List.length_take, min_eq_left (Nat.le_of_lt rlt)]
    rfl
  · rw [tableBitsK_groups]
    change (states.flatMap (readActions tr)).flatMap actionBitsK = _
    conv_lhs => rw [qs]
    simp only [List.flatMap_append, List.flatMap_cons]
    change _ ++ (reads.flatMap actionBitsK ++ _) = _
    conv_lhs => rw [rs]
    simp only [List.flatMap_append, List.flatMap_cons, List.append_assoc, ← List.flatMap_def]

/-- A literal record list has at least as many bits as records. This is a
consequence of the public action-length bound, with no table re-encoding. -/
private lemma records_length {k n : ℕ} (records : List (Action k Bool (Fin (n + 1)))) :
    records.length ≤ (records.flatMap actionBitsK).length := by
  induction records with
  | nil => simp
  | cons a as ih =>
    have hp := codeActionK_length a
    simp only [List.length_cons, List.flatMap_cons, List.length_append]
    omega

/-- A unit work-head bound summed over a run. This is the only general
head-bound argument needed for the auxiliary round envelope; mirrored and
input heads have the stronger exact phase equations.
**Promotion candidate:** a run form of `workTapePos_step_le`. -/
lemma _root_.Turing.MultiTapeTM.abs_workTapePos_runFrom_sub_le {m : ℕ} {S : Type} {x : List Bool}
    (M : MultiTapeTM m Bool S) (d : Cfg m Bool S x) (t : ℕ) (j : Fin m) :
    |(M.runFrom d t).workTapePos j - d.workTapePos j| ≤ (t : ℤ) := by
  let height (u : ℕ) := (M.runFrom d u).workTapePos j
  have unit (u : ℕ) : |height (u + 1) - height u| ≤ 1 := by
    simpa only [height, MultiTapeTM.runFrom_succ_eq_step'] using
      MultiTapeTM.workTapePos_step_le (tm := M) (M.runFrom d u) j
  calc
    |(M.runFrom d t).workTapePos j - d.workTapePos j| =
        |∑ u ∈ Finset.range t, (height (u + 1) - height u)| := by
      rw [Finset.sum_range_sub]; rfl
    _ ≤ ∑ u ∈ Finset.range t, |height (u + 1) - height u| := Finset.abs_sum_le_sum_abs _ _
    _ ≤ ∑ _u ∈ Finset.range t, (1 : ℤ) := Finset.sum_le_sum (fun u _ => unit u)
    _ = (t : ℤ) := by simp

/-- Any machine whose actions keep the native input fixed preserves its
position at every horizon. Public iteration transport supplies the run law. -/
lemma _root_.Turing.MultiTapeTM.inputPos_runFrom_of_inputTape_eq_zero
    {m : ℕ} {S : Type} {x : List Bool}
    (M : MultiTapeTM m Bool S) (h : ∀ q inp work, (M.tr q inp work).inputTape = 0)
    (d : Cfg m Bool S x) (t : ℕ) : (M.runFrom d t).inputPos = d.inputPos := by
  have commute : Function.Semiconj (fun e : Cfg m Bool S x => e.inputPos) M.step id := by
    intro e
    cases hs : e.state with
    | none => simp [MultiTapeTM.step, hs]
    | some q => simp [MultiTapeTM.step, hs, h]
  simpa only [MultiTapeTM.runFrom, Function.iterate_id, id_eq] using commute.iterate_right t d

/-- Exact number of records in a prefix of complete state groups and one
partial read group. This is only a list-length ledger. -/
private lemma groups_length {k n : ℕ}
    (groups : List (List (Action k Bool (Fin (n + 1)))))
    (hg : ∀ g ∈ groups, g.length = 3 ^ (k + 1)) :
    groups.flatten.length = groups.length * 3 ^ (k + 1) := by
  induction groups with
  | nil => simp
  | cons g gs ih =>
    have h := ih (fun a ha => hg a (by simp [ha]))
    simp only [List.flatten_cons, List.length_append, List.length_cons, Nat.add_mul,
      Nat.one_mul, hg g (by simp), h]
    omega

/-- Whole physical lookup, including finite observation and the unary-state
loop. Its exit is first reached at the selected record boundary. The native
input, all tape cells, physical output, and mirrored heads are unchanged. -/
private lemma seekTM_lookup {k n : ℕ} {x : List Bool}
    (groups : List (List (Action k Bool (Fin (n + 1)))))
    (rest : List (Action k Bool (Fin (n + 1))))
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (hg : ∀ g ∈ groups, g.length = 3 ^ (k + 1))
    (hrank : rest.length = (readRank d.inputSymbol
      (fun i => d.workTapeSymbols (Fin.castAdd 2 i))).val)
    (hq : d.state = some (recordTM k).q₀)
    (hs : ∀ r ≤ groups.length,
      d.workTapes (summaryTape k) (d.workTapePos (summaryTape k) + 2 + r) =
        some (decide (r < groups.length)))
    (hw : ∀ (r : ℕ) (hr : r < ((groups.flatten ++ rest).flatMap actionBitsK).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some (((groups.flatten ++ rest).flatMap actionBitsK)[r]'hr)) :
    let records := groups.flatten ++ rest
    let T := groups.length + 3 + seekCost records
    (seekTM k).runFrom (seekCfg d .observe) T =
      seekCfg (seekShift d (records.flatMap actionBitsK).length (groups.length + 2)) .done ∧
    (∀ u < T, ((seekTM k).runFrom (seekCfg d .observe) u).state ≠ some ⟨none, .done⟩) ∧
    (∀ u ≤ T,
      let f := (seekTM k).runFrom (seekCfg d .observe) u
      f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
      ∀ j, j ≠ tableTape k → j ≠ summaryTape k → f.workTapePos j = d.workTapePos j) ∧
    T ≤ 5 * (records.flatMap actionBitsK).length + 3 := by
  let rank := readRank d.inputSymbol (fun i => d.workTapeSymbols (Fin.castAdd 2 i))
  have introRun := seekTM_observe d hq
  have bodyRun := seekTM_groups groups rest rank hg hrank (seekShift d 0 2) hq
    (by simpa [seekShift, summary_ne_table k] using hs)
    (by simpa [seekShift] using hw)
  dsimp only at introRun bodyRun ⊢
  have cost : groups.length + 3 + seekCost (groups.flatten ++ rest) =
      2 + (groups.length + 1 + seekCost (groups.flatten ++ rest)) := by omega
  refine ⟨?_, ?_, ?_, ?_⟩
  · rw [cost, MultiTapeTM.runFrom_add, introRun.1, bodyRun.1]
    apply Cfg.ext <;> try rfl
    funext j
    by_cases ht : j = tableTape k
    · subst j
      simp [seekCfg, Cfg.mapState, seekShift]
    · by_cases hj : j = summaryTape k
      · subst j
        simp [seekCfg, Cfg.mapState, seekShift, summary_ne_table k, Nat.cast_add,
          add_assoc, add_comm]
      · simp [seekCfg, Cfg.mapState, seekShift, ht, hj]
  · intro u hu
    by_cases hi : u < 2
    · exact introRun.2.1 u hi
    · rw [show u = 2 + (u - 2) by omega, MultiTapeTM.runFrom_add, introRun.1]
      exact bodyRun.2.1 _ (by omega)
  · intro u hu
    by_cases hi : u ≤ 2
    · have frame := introRun.2.2 u hi
      exact ⟨frame.1, frame.2.1, frame.2.2.1, fun j _ hj => frame.2.2.2 j hj⟩
    · rw [show u = 2 + (u - 2) by omega, MultiTapeTM.runFrom_add, introRun.1]
      have frame := bodyRun.2.2 (u - 2) (by omega)
      refine ⟨frame.1, frame.2.1, frame.2.2.1, ?_⟩
      intro j ht hj
      simpa only [seekShift, if_neg ht, if_neg hj, add_zero] using frame.2.2.2 j ht hj
  · have hrecord := records_length (groups.flatten ++ rest)
    have hgroup : groups.length ≤ (groups.flatten ++ rest).length := by
      have ht := groups_length groups hg
      have hp : 1 ≤ 3 ^ (k + 1) := by have : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega); omega
      have hm := Nat.mul_le_mul_left groups.length hp
      simp only [Nat.mul_one] at hm
      simp only [List.length_append]
      omega
    unfold seekCost
    omega

/-- Read a unary state word at any valid cell, including its terminating
false. This uses the canonical buffer convention directly. -/
private lemma bufferTape_unary (q r : ℕ) (hr : r ≤ q) :
    FinTM.bufferTape (List.replicate q true ++ [false]) r = some (decide (r < q)) := by
  by_cases hlt : r < q
  · rw [FinTM.bufferTape_nat, List.getElem?_append_left (by simpa only [List.length_replicate]
    using hlt)]
    simp [hlt]
  · have he : r = q := by omega
    subst r
    simp [FinTM.bufferTape_nat]

/-- The summary auxiliary stores its stationary summary at zero, a blank
at one, and the current unary state word starting at two. -/
def summaryCells (summary : Option Bool) (word : List Bool) (z : ℤ) : Option Bool :=
  if z = 0 then summary else FinTM.bufferTape word (z - 2)

/-- The state word and its left delimiter are a framed ordinary buffer,
separated from the summary cell. -/
lemma summaryCells_shift (summary : Option Bool) (word : List Bool) (z : ℤ) (hz : -1 ≤ z) :
    summaryCells summary word (2 + z) = FinTM.bufferTape word z := by
  have hne : 2 + z ≠ 0 := by omega
  simp [summaryCells, hne]

/-- Updating the summary does not disturb the state word. -/
lemma summaryCells_update (summary next : Option Bool) (word : List Bool) :
    Function.update (summaryCells summary word) 0 next = summaryCells next word := by
  funext z
  by_cases hz : z = 0
  · subst z; simp [summaryCells]
  · simp [summaryCells, hz]

/-- The canonical buffer is blank off its finite support. -/
private lemma bufferTape_outside (word : List Bool) (z : ℤ)
    (hz : ¬(0 ≤ z ∧ z < word.length)) : FinTM.bufferTape word z = none := by
  by_cases hn : 0 ≤ z
  · simp only [FinTM.bufferTape, if_pos hn]
    exact List.getElem?_eq_none_iff.mpr (by omega)
  · simp [FinTM.bufferTape, hn]

/-- Copying a complete new state word into the cleared state region yields
exactly the next summary/state layout. -/
lemma summaryCells_copy (summary : Option Bool) (word : List Bool) :
    (fun z : ℤ => if 2 ≤ z ∧ z < 2 + word.length then FinTM.bufferTape word (z - 2)
      else summaryCells summary [] z) = summaryCells summary word := by
  funext z
  by_cases hz : z = 0
  · subst z; simp [summaryCells]
  · by_cases hinside : 2 ≤ z ∧ z < 2 + word.length
    · simp [hinside, summaryCells, hz]
    · have off := bufferTape_outside word (z - 2) (by omega)
      simp [hinside, summaryCells, hz, off]

/-- The catalog's framed clear erases only the state word, preserving the
stationary summary and its blank separator. -/
lemma summaryCells_clear (summary : Option Bool) (word : List Bool) :
    (fun z : ℤ => if 2 ≤ z ∧ z < 2 + word.length then none else summaryCells summary word z) =
      summaryCells summary [] := by
  funext z
  rw [← summaryCells_copy summary word]
  by_cases hi : 2 ≤ z ∧ z < 2 + word.length
  · have hz : z ≠ 0 := by omega
    simp [hi, summaryCells, hz]
  · simp [hi]

/-- Lookup selects the actual source transition in the public table.
The physical input/work reads determine the finite rank. The returned table
prefix contains at least `q` bits, a useful state-head and clearing ledger.

**Proof sketch.** Cut the serializer at the ranked source transition. Feed
its literal prefix and unary state word to the proved physical selector.
The fixed group-size identity and the public action-length bound pay for the
unary state scan, and the exact table cut bounds the whole lookup. -/
private lemma seekTM_select {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (q : Fin (n + 1)) (summary : Option Bool)
    (d : Cfg (k + 2) Bool (RecordState k) x)
    (hq : d.state = some (recordTM k).q₀)
    (ht : d.workTapes (tableTape k) = FinTM.bufferTape (tableBitsK tr))
    (hp : d.workTapePos (tableTape k) = 0)
    (hs : d.workTapes (summaryTape k) = summaryCells summary (unaryFin q))
    (hsp : d.workTapePos (summaryTape k) = 0) :
    let a := tr q d.inputSymbol (fun i => d.workTapeSymbols (Fin.castAdd 2 i))
    ∃ (pre tail : List Bool) (T : ℕ),
      tableBitsK tr = pre ++ actionBitsK a ++ tail ∧ q.val ≤ pre.length ∧
      0 < T ∧ T ≤ 5 * (tableBitsK tr).length + 3 ∧
      (seekTM k).runFrom (seekCfg d .observe) T =
        seekCfg (seekShift d pre.length (q.val + 2)) .done ∧
      (∀ u < T, ((seekTM k).runFrom (seekCfg d .observe) u).state ≠ some ⟨none, .done⟩) ∧
      (∀ u ≤ T,
        let f := (seekTM k).runFrom (seekCfg d .observe) u
        f.inputPos = d.inputPos ∧ f.workTapes = d.workTapes ∧ f.output = d.output ∧
        ∀ j, j ≠ tableTape k → j ≠ summaryTape k → f.workTapePos j = d.workTapePos j) := by
  dsimp only
  obtain ⟨groups, rest, tail, hglen, hgroup, hrank, split⟩ :=
    tableBitsK_cut tr q d.inputSymbol (fun i => d.workTapeSymbols (Fin.castAdd 2 i))
  let records := groups.flatten ++ rest
  let pre := records.flatMap actionBitsK
  have full : ∀ (r : ℕ) (hr : r < (tableBitsK tr).length),
      d.workTapes (tableTape k) (d.workTapePos (tableTape k) + r) =
        some ((tableBitsK tr)[r]'hr) := by
    intro r hr
    simp [ht, hp, FinTM.bufferTape_nat, List.getElem?_eq_getElem hr]
  have words := reads_append (d.workTapes (tableTape k)) (d.workTapePos (tableTape k))
    pre (actionBitsK (tr q d.inputSymbol (fun i => d.workTapeSymbols (Fin.castAdd 2 i))) ++ tail)
    (by simpa only [split, List.append_assoc] using full)
  have stateReads : ∀ r ≤ groups.length,
      d.workTapes (summaryTape k) (d.workTapePos (summaryTape k) + 2 + r) =
        some (decide (r < groups.length)) := by
    intro r hr
    rw [hs, hsp, zero_add, summaryCells_shift _ _ _ (by omega)]
    simpa only [unaryFin, hglen] using bufferTape_unary q.val r (by omega)
  have run := seekTM_lookup groups rest d hgroup hrank hq stateReads words.1
  dsimp only at run
  have gl := groups_length groups hgroup
  have recordsBound := records_length records
  have groupBound : groups.length ≤ records.length := by
    have p : 1 ≤ 3 ^ (k + 1) := by have : 0 < 3 ^ (k + 1) := Nat.pow_pos (by omega); omega
    have h := Nat.mul_le_mul_left groups.length p
    simp only [Nat.mul_one] at h
    dsimp [records]
    simp only [List.length_append]
    omega
  have preBound : pre.length ≤ (tableBitsK tr).length := by
    rw [split]
    simp only [List.length_append]
    dsimp [pre, records]
    omega
  dsimp only [pre, records] at preBound
  refine ⟨pre, tail, groups.length + 3 + seekCost records, split, ?_, by omega,
    run.2.2.2.trans (by omega), ?_, run.2.1, run.2.2.1⟩
  · rw [← hglen]
    exact groupBound.trans recordsBound
  · simpa only [hglen] using run.1

/-- Fixed two-cell routing between the summary cell and its state word.
No parser or variable scan is involved. -/
private def twoMoveTM {m : ℕ} (i : Fin m) (move : SignType) : MultiTapeTM m Bool (Fin 3) where
  q₀ := 0
  tr := fun q _ _ => if h : q.val < 2 then
    tapeAction i 0 none move ⟨q.val + 1, by omega⟩ else inputMoveAction 0 q

/-- Every prefix of the fixed two-cell routing move. The public one-action
movement law is sufficient; all cells, input and output are framed. -/
private lemma twoMoveTM_run {m : ℕ} {x : List Bool} (i : Fin m) (move : SignType)
    (d : Cfg m Bool (Fin 3) x) (hq : d.state = some 0) :
    ∀ (u : ℕ) (hu : u ≤ 2), (twoMoveTM i move).runFrom d u =
      {d with
        state := some ⟨u, by omega⟩
        workTapePos := Function.update d.workTapePos i (d.workTapePos i + (u : ℤ) * move)} := by
  intro u hu
  induction u with
  | zero =>
    apply Cfg.ext <;> simp [hq]
  | succ u ih =>
    rw [MultiTapeTM.runFrom_succ_eq_step', ih (by omega)]
    rw [step_of_tapeAction_move (twoMoveTM i move) _ (q := ⟨u, by omega⟩)
      (q' := ⟨u + 1, by omega⟩) i 0 move rfl (by simp [twoMoveTM, show u < 2 by omega])]
    apply Cfg.ext <;> simp [Function.update_idem, Nat.cast_add, add_mul, add_assoc]

/-- The selector's phase carriers are finite and have decidable equality. -/
instance instFintypeSeekPh (k : ℕ) (key : SeekKey k) : Fintype (SeekPh k key) := by
  cases key <;> dsimp [SeekPh] <;> infer_instance

/-- Equality of selector phase states uses their finite carrier instances. -/
instance instDecidableEqSeekPh (k : ℕ) (key : SeekKey k) : DecidableEq (SeekPh k key) := by
  cases key <;> dsimp [SeekPh] <;> infer_instance

/-- The finite schedule of one generic simulated transition. The live bit
is retained only while restoring the heads after the successor was read. -/
inductive Key where
  | entry | lookup | stateTail | stateHome | clearState | toSummary | record | toState | successor
  | tableTail (live : Bool) | tableHome (live : Bool)
  | stateGap (live : Bool) | stateRestore (live : Bool) | finish (live : Bool)
  | halt
  deriving DecidableEq, Fintype

/-- The round's phase carriers, all independent of the coded state count. -/
abbrev Ph (k : ℕ) : Key → Type
  | .entry | .halt => Unit
  | .lookup => Σ key, SeekPh k key
  | .stateTail | .stateHome | .tableTail _ | .tableHome _ | .stateGap _ | .stateRestore _ =>
      SeekPhase
  | .clearState => SweepPhase
  | .toSummary | .toState | .finish _ => Fin 3
  | .record => RecordState k
  | .successor => SuccPhase

/-- All variable scans use their landed catalog machines. A table rewind
first walks to the right delimiter, then uses `skipBackTM` on the whole word;
this avoids requiring blank separators between encoded records. -/
private def part (k : ℕ) : (key : Key) → MultiTapeTM (k + 2) Bool (Ph k key)
  | .entry | .halt => ⟨(), fun _ _ _ => inputMoveAction 0 ()⟩
  | .lookup => seekTM k
  | .stateTail | .stateGap _ => skipTM (k + 2) (summaryTape k)
  | .stateHome | .stateRestore _ => skipBackTM (k + 2) (summaryTape k)
  | .clearState => clearTM (k + 2) (summaryTape k)
  | .toSummary | .finish _ => twoMoveTM (summaryTape k) .neg
  | .toState => twoMoveTM (summaryTape k) .pos
  | .record => recordTM k
  | .successor => succFieldTM (tableTape k) (summaryTape k) true
  | .tableTail _ => skipTM (k + 2) (tableTape k)
  | .tableHome _ => skipBackTM (k + 2) (tableTape k)

/-- A completed live successor returns to the round anchor; a simulated
halt returns through a separate live exit. -/
def exit (k : ℕ) (live : Bool) : Σ key, Ph k key :=
  if live then ⟨.entry, ()⟩ else ⟨.halt, ()⟩

/-- The round's stationary, silent phase dispatches. A simulated halt goes
to a live physical exit only after the action and all restoration have run. -/
private def red (k : ℕ) : (key : Key) → Ph k key → Option (Σ key, Ph k key)
  | .entry, _ => some ⟨.lookup, (seekTM k).q₀⟩
  | .lookup, q => if q = ⟨none, .done⟩ then some ⟨.stateTail, .run⟩ else none
  | .stateTail, q => if q = .done then some ⟨.stateHome, .run⟩ else none
  | .stateHome, q => if q = .done then some ⟨.clearState, .sweep⟩ else none
  | .clearState, q => if q = .done then some ⟨.toSummary, 0⟩ else none
  | .toSummary, p => if p = 2 then some ⟨.record, (recordTM k).q₀⟩ else none
  | .record, q => if q = none then some ⟨.toState, 0⟩ else none
  | .toState, p => if p = 2 then some ⟨.successor, .tag⟩ else none
  | .successor, .done live => some ⟨.tableTail live, .run⟩
  | .tableTail live, q => if q = .done then some ⟨.tableHome live, .run⟩ else none
  | .tableHome live, q => if q = .done then some ⟨.stateGap live, .run⟩ else none
  | .stateGap live, q => if q = .done then some ⟨.stateRestore live, .run⟩ else none
  | .stateRestore live, q => if q = .done then some ⟨.finish live, 0⟩ else none
  | .finish live, p => if p = 2 then some (exit k live) else none
  | _, _ => none

/-- One generic step block: lookup, clear the old state, execute the fixed
payload, install its successor, and restore both auxiliary heads home.
**Promotion candidate (host H2, step block; machine-library-design.md §14).** -/
def roundTM (k : ℕ) : MultiTapeTM (k + 2) Bool (Σ key, Ph k key) :=
  phaseHostTM (part k) ⟨.entry, ()⟩ (red k)

/-- Finiteness of the round's physical control depends only on `k`. -/
instance instFintypePh (k : ℕ) (key : Key) : Fintype (Ph k key) := by
  cases key <;> dsimp [Ph] <;> infer_instance

/-- The round's entry and live halt exit have decidable equality. -/
instance instDecidableEqPh (k : ℕ) (key : Key) : DecidableEq (Ph k key) := by
  cases key <;> dsimp [Ph] <;> infer_instance

/-- The generic step block as a finite binary machine. -/
def roundFinTM (k : ℕ) : FinTM Bool where
  k := k + 2
  State := Σ key, Ph k key
  tm := roundTM k

/-- Canonical data frame for a round phase. The mirrored bank may contain
arbitrary source tapes and positions, while the two auxiliary words and heads
are explicit. The physical output prefix is independent of simulated output. -/
private def roundData {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (table : List Bool) (summary : Option Bool) (word : List Bool)
    (tablePos statePos : ℤ) (pre : List Bool) : Cfg (k + 2) Bool (RecordState k) x where
  state := some (recordTM k).q₀
  inputPos := c.inputPos
  workTapes := Fin.addCases c.workTapes
    (fun i : Fin 2 => if i = 0 then FinTM.bufferTape table else summaryCells summary word)
  workTapePos := Fin.addCases c.workTapePos (fun i : Fin 2 => if i = 0 then tablePos else statePos)
  output := pre

/-- A mirrored tape is distinct from the summary/state auxiliary. -/
private lemma mirror_ne_summary (k : ℕ) (i : Fin k) : Fin.castAdd 2 i ≠ summaryTape k := by
  have hi := i.isLt
  apply Fin.ne_of_val_ne
  change i.val ≠ k + 1
  omega

/-- Lookup's two head advances preserve the canonical data frame. -/
private lemma roundData_shift {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (table : List Bool) (summary : Option Bool) (word : List Bool)
    (tablePos statePos : ℤ) (pre : List Bool) (a b : ℕ) :
    seekShift (roundData c table summary word tablePos statePos pre) a b =
      roundData c table summary word (tablePos + a) (statePos + b) pre := by
  apply Cfg.ext <;> try rfl
  funext j
  refine Fin.addCases (fun i => ?_) (fun i => ?_) j
  · simp [seekShift, roundData, mirror_ne_table k i, mirror_ne_summary k i]
  · fin_cases i <;> simp [seekShift, roundData, tableTape, summaryTape]

/-- During a round every simulated head is at its start or its single
source-action endpoint. State renaming does not affect this predicate. -/
private def RoundWindow {k n : ℕ} {S : Type} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (d : Cfg (k + 2) Bool S x) : Prop :=
  (d.inputPos = c.inputPos ∨ d.inputPos = (a.apply c).inputPos) ∧
    ∀ i : Fin k, d.workTapePos (Fin.castAdd 2 i) = c.workTapePos i ∨
      d.workTapePos (Fin.castAdd 2 i) = (a.apply c).workTapePos i

/-- A canonical phase frame has the round's start heads or its end heads,
independently of both auxiliary positions and words. -/
private lemma roundData_window {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (table : List Bool) (summary : Option Bool) (word : List Bool)
    (tablePos statePos : ℤ) (pre : List Bool) (after : Bool) :
    RoundWindow c a
      (roundData (if after then a.apply c else c) table summary word tablePos statePos pre) := by
  cases after <;>
    simp [RoundWindow, roundData]

/-- A segment of the already defined round host. Its strict prefix is live,
excludes both round anchors, and satisfies the simulated-head envelope. -/
private def RoundSpan {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (d e : Cfg (k + 2) Bool (Σ key, Ph k key) x) (T : ℕ) : Prop :=
  (roundTM k).runFrom d T = e ∧
    ∀ u < T, ∃ s, ((roundTM k).runFrom d u).state = some s ∧
      s ≠ ⟨.entry, ()⟩ ∧ s ≠ ⟨.halt, ()⟩ ∧
      RoundWindow c a ((roundTM k).runFrom d u)

/-- Concatenate two proved segments by the public run-addition identity.
This lemma only joins their strict-prefix predicates; it defines no host,
loop, or alternative machine trace. -/
private lemma RoundSpan.add {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    {d e f : Cfg (k + 2) Bool (Σ key, Ph k key) x} {T U : ℕ}
    (left : RoundSpan c a d e T) (right : RoundSpan c a e f U) :
    RoundSpan c a d f (T + U) := by
  refine ⟨?_, ?_⟩
  · rw [MultiTapeTM.runFrom_add, left.1, right.1]
  · intro u hu
    by_cases hl : u < T
    · exact left.2 u hl
    · rw [show u = T + (u - T) by omega, MultiTapeTM.runFrom_add, left.1]
      exact right.2 _ (by omega)

/-- Lift one callee contract through the landed phase-host theorem. The
endpoint is one silent dispatch after the callee's first selected exit.
Liveness of every earlier callee state follows from halt absorption and its
proved live endpoint. No machine trace is re-established here. -/
private lemma roundTM_phase {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (key : Key) (hentry : key ≠ .entry) (hhalt : key ≠ .halt)
    (d e : Cfg (k + 2) Bool (Ph k key) x) (T : ℕ)
    (run : (part k key).runFrom d T = e)
    (s : Ph k key) (hs : e.state = some s)
    (next : Σ key, Ph k key) (redirect : red k key s = some next)
    (guard : ∀ u < T, ∀ q, ((part k key).runFrom d u).state = some q → red k key q = none)
    (window : ∀ u ≤ T, RoundWindow c a ((part k key).runFrom d u)) :
    RoundSpan c a (d.mapState (Sigma.mk key)) (e.mapState (fun _ => next)) (T + 1) := by
  have transfer := phaseHostTM_run (part k) ⟨.entry, ()⟩ (red k)
    key d e T run s hs next redirect guard
  refine ⟨transfer.2, ?_⟩
  intro u hu
  rw [transfer.1 u (by omega)]
  have live : ((part k key).runFrom d u).state ≠ none := by
    intro stopped
    have freeze : (part k key).runFrom d T = (part k key).runFrom d u := by
      rw [show T = u + (T - u) by omega, MultiTapeTM.runFrom_add,
        MultiTapeTM.runFrom_of_halt _ stopped]
    have contradiction : e.state = none := by rw [← run, freeze]; exact stopped
    rw [hs] at contradiction
    contradiction
  cases reached : ((part k key).runFrom d u).state with
  | none => exact False.elim (live reached)
  | some q =>
    refine ⟨⟨key, q⟩, by simp [Cfg.mapState, reached], ?_, ?_, window u (by omega)⟩
    · intro bad; exact hentry (congrArg Sigma.fst bad)
    · intro bad; exact hhalt (congrArg Sigma.fst bad)

/-- Enter a round phase with the data frame unchanged. -/
private def atKey {k : ℕ} {x : List Bool}
    (d : Cfg (k + 2) Bool (RecordState k) x) (key : Key) (q : Ph k key) :
    Cfg (k + 2) Bool (Σ key, Ph k key) x :=
  d.mapState (fun _ => ⟨key, q⟩)

/-- Transfer the simulated-head predicate across any data frame equality.
Control types and auxiliary head positions may differ. -/
private lemma roundWindow_frame {k n : ℕ} {S R : Type} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    {d : Cfg (k + 2) Bool S x} {e : Cfg (k + 2) Bool R x}
    (window : RoundWindow c a d) (input : e.inputPos = d.inputPos)
    (heads : ∀ i : Fin k, e.workTapePos (Fin.castAdd 2 i) = d.workTapePos (Fin.castAdd 2 i)) :
    RoundWindow c a e := by
  obtain ⟨hi, hw⟩ := window
  refine ⟨by simpa only [input] using hi, ?_⟩
  intro i
  simpa only [heads i] using hw i

/-- State-word cells of a canonical phase frame, with the left blank included. -/
private lemma roundData_stateCell {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (table : List Bool) (summary : Option Bool) (word : List Bool)
    (tablePos statePos : ℤ) (pre : List Bool) (z : ℤ) (hz : -1 ≤ z) :
    (roundData c table summary word tablePos statePos pre).workTapes (summaryTape k) (2 + z) =
      FinTM.bufferTape word z := by
  simpa only [roundData, summaryTape, Fin.addCases_right, show (1 : Fin 2) ≠ 0 from by decide,
    ↓reduceIte] using
    summaryCells_shift summary word z hz

/-- Clearing preparation restores the old unary-state head, erases its word,
and returns to the summary cell before invoking the selected-record reader.
All mirrored heads remain at their round-start positions throughout.

**Proof sketch.** Apply the public framed skip, reverse skip and clear contracts
in order, followed by the fixed two-cell move. Each is lifted by the public
phase host, and their proved strict prefixes are joined by run addition. -/
private lemma roundTM_prepare {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (q : Fin (n + 1)) (table : List Bool) (summary : Option Bool) (pos : ℤ) (pre : List Bool) :
    RoundSpan c a
      (atKey (roundData c table summary (unaryFin q) pos (q.val + 2) pre) .stateTail .run)
      (atKey (roundData c table summary [] pos 0 pre) .record (recordTM k).q₀)
      (3 * q.val + 16) := by
  let d0 := roundData c table summary (unaryFin q) pos (q.val + 2) pre
  let d1 := roundData c table summary (unaryFin q) pos (q.val + 4) pre
  let d2 := roundData c table summary (unaryFin q) pos 2 pre
  let d3 := roundData c table summary [] pos 2 pre
  let d4 := roundData c table summary [] pos 0 pre
  let s0 := d0.mapState (fun _ => SeekPhase.run)
  let s1 := d1.mapState (fun _ => SeekPhase.run)
  let s2 := d2.mapState (fun _ => SweepPhase.sweep)
  let s3 := d3.mapState (fun _ => (0 : Fin 3))
  have skip := skipTM_run_ofCfg (summaryTape k) [false] s0 rfl (by
    intro z hz hz'
    have cases : z = 0 ∨ z = 1 := by simp only [List.length_cons, List.length_nil] at hz'; omega
    rcases cases with rfl | rfl
    · simp [s0, d0, Cfg.mapState, roundData, summaryTape, summaryCells, unaryFin,
        show (q.val : ℤ) + 2 ≠ 0 by omega, FinTM.bufferTape]
    · have cell := roundData_stateCell c table summary (unaryFin q) pos (q.val + 2) pre
        ((q.val + 1 : ℕ) : ℤ) (by omega)
      have blank : FinTM.bufferTape (unaryFin q) ((q.val + 1 : ℕ) : ℤ) = none := by
        rw [FinTM.bufferTape_nat]
        simp [unaryFin]
      rw [blank] at cell
      convert cell using 1; simp [s0, d0, Cfg.mapState, roundData, summaryTape]; congr 1; omega)
  have skipEnd : (skipTM (k + 2) (summaryTape k)).runFrom s0 2 = d1.mapState (fun _ =>
    SeekPhase.done) := by
    have he := skip.1
    simp only [List.length_cons, List.length_nil, Nat.zero_add] at he
    rw [he]
    apply Cfg.ext <;> try rfl
    funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simp [s0, d0, d1, roundData, Cfg.mapState, mirror_ne_summary k i]
    · fin_cases i <;> simp [s0, d0, d1, roundData, Cfg.mapState, summaryTape, add_assoc]
  have p0 := roundTM_phase c a .stateTail (by decide) (by decide) s0
    (d1.mapState (fun _ => SeekPhase.done)) 2 skipEnd .done rfl ⟨.stateHome, .run⟩
    (by simp [red]) (by
      intro u hu r hr
      have ne : r ≠ .done := fun eq => (skip.2.1 u (by simpa using hu)) (hr.trans (congrArg some
        eq))
      simp [red, ne]) (by
      intro u hu
      apply roundWindow_frame c a
        (roundData_window c a table summary (unaryFin q) pos (q.val + 2) pre false)
      · exact MultiTapeTM.inputPos_runFrom_of_inputTape_eq_zero (skipTM (k + 2) (summaryTape k))
          (by intro r inp work; cases r <;> cases h : work (summaryTape k) <;>
            simp [skipTM, h, tapeAction, inputMoveAction]) s0 u
      · intro i
        simpa only [mirror_ne_summary k i, ↓reduceIte, s0, Cfg.mapState] using skip.2.2
          (Fin.castAdd 2 i) u (by simpa using hu))
  have back := skipBackTM_run_ofCfg (summaryTape k) (unaryFin q) s1 rfl (by
    intro z hz hz'
    have cell := roundData_stateCell c table summary (unaryFin q) pos (q.val + 4) pre z hz
    convert cell using 1; simp [s1, d1, Cfg.mapState, roundData, summaryTape, unaryFin])
  have backEnd : (skipBackTM (k + 2) (summaryTape k)).runFrom s1 ((unaryFin q).length + 3) =
      d2.mapState (fun _ => SeekPhase.done) := by
    rw [back.1]
    apply Cfg.ext <;> try rfl
    funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simp [s1, d1, d2, roundData, Cfg.mapState, mirror_ne_summary k i]
    · fin_cases i <;> simp [s1, d1, d2, roundData, Cfg.mapState, summaryTape, unaryFin]
  have p1 := roundTM_phase c a .stateHome (by decide) (by decide) s1
    (d2.mapState (fun _ => SeekPhase.done)) _ backEnd .done rfl ⟨.clearState, .sweep⟩
    (by simp [red]) (by
      intro u hu r hr
      have ne : r ≠ .done := fun eq => (back.2.1 u hu) (hr.trans (congrArg some eq))
      simp [red, ne]) (by
      intro u hu
      apply roundWindow_frame c a
        (roundData_window c a table summary (unaryFin q) pos (q.val + 4) pre false)
      · exact MultiTapeTM.inputPos_runFrom_of_inputTape_eq_zero (skipBackTM (k + 2) (summaryTape k))
          (by intro r inp work; cases r <;> cases h : work (summaryTape k) <;>
            simp [skipBackTM, h, tapeAction, inputMoveAction]) s1 u
      · intro i
        simpa only [mirror_ne_summary k i, ↓reduceIte, s1, Cfg.mapState] using back.2.2
          (Fin.castAdd 2 i) u hu)
  have clear := clearTM_run_ofCfg (summaryTape k) (unaryFin q) s2 rfl (by
    intro z hz _
    simpa [s2, d2, Cfg.mapState, roundData, summaryTape] using
      roundData_stateCell c table summary (unaryFin q) pos 2 pre z hz)
  have clearEnd : (clearTM (k + 2) (summaryTape k)).runFrom s2 (2 * (unaryFin q).length + 2) =
      d3.mapState (fun _ => SweepPhase.done) := by
    rw [clear.1]
    apply Cfg.ext <;> try rfl
    funext j z
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simp [s2, d2, d3, roundData, Cfg.mapState, mirror_ne_summary k i]
    · fin_cases i
      · simp [s2, d2, d3, roundData, Cfg.mapState, summaryTape]
      · simpa [s2, d2, d3, roundData, Cfg.mapState, summaryTape] using
          congrFun (summaryCells_clear summary (unaryFin q)) z
  have p2 := roundTM_phase c a .clearState (by decide) (by decide) s2
    (d3.mapState (fun _ => SweepPhase.done)) _ clearEnd .done rfl ⟨.toSummary, (0 : Fin 3)⟩
    (by simp [red]) (by
      intro u hu r hr
      have ne : r ≠ .done := fun eq => (clear.2.1 u hu) (hr.trans (congrArg some eq))
      simp [red, ne]) (by
      intro u hu
      apply roundWindow_frame c a (roundData_window c a table summary (unaryFin q) pos 2 pre false)
      · exact MultiTapeTM.inputPos_runFrom_of_inputTape_eq_zero (clearTM (k + 2) (summaryTape k))
          (by intro r inp work; cases r <;> cases h : work (summaryTape k) <;>
            simp [clearTM, h]) s2 u
      · intro i
        simpa only [mirror_ne_summary k i, ↓reduceIte, s2, Cfg.mapState] using clear.2.2
          (Fin.castAdd 2 i) u hu)
  have moveEnd : (twoMoveTM (summaryTape k) .neg).runFrom s3 2 = d4.mapState (fun _ => (2 :
    Fin 3)) := by
    rw [twoMoveTM_run _ _ s3 rfl 2 le_rfl]
    apply Cfg.ext <;> try rfl
    funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simp [s3, d3, d4, roundData, Cfg.mapState, mirror_ne_summary k i]
    · fin_cases i <;> simp [s3, d3, d4, roundData, Cfg.mapState, summaryTape]
  have p3 := roundTM_phase c a .toSummary (by decide) (by decide) s3
    (d4.mapState (fun _ => (2 : Fin 3))) 2 moveEnd 2 rfl ⟨.record, (recordTM k).q₀⟩
    (by simp [red]) (by
      intro u hu r hr
      rw [twoMoveTM_run _ _ s3 rfl u (by omega)] at hr
      have vr : r.val = u := congrArg Fin.val (Option.some.inj hr).symm
      have ne : r ≠ 2 := by intro bad; subst r; omega
      simp [red, ne]) (by
      intro u hu
      rw [twoMoveTM_run _ _ s3 rfl u hu]
      refine ⟨Or.inl rfl, fun i => Or.inl ?_⟩
      simp [s3, d3, roundData, Cfg.mapState, mirror_ne_summary k i])
  have joined := RoundSpan.add c a (RoundSpan.add c a (RoundSpan.add c a p0 p1) p2) p3
  have cost : ((2 + 1) + ((unaryFin q).length + 3 + 1) + (2 * (unaryFin q).length + 2 + 1)) + (2
    + 1) =
      3 * q.val + 16 := by simp [unaryFin]; omega
  simpa only [atKey, s0, d0, d4, Cfg.mapState, roundData, cost] using joined

/-- Literal cells of the middle word of a canonical concatenated buffer.
Both cuts use the same framed append-read lemma. -/
private lemma bufferTape_reads (before word after : List Bool) :
    ∀ (r : ℕ) (hr : r < word.length),
      FinTM.bufferTape (before ++ word ++ after) (before.length + r) = some (word[r]'hr) := by
  have whole : ∀ (r : ℕ) (hr : r < (before ++ word ++ after).length),
      FinTM.bufferTape (before ++ word ++ after) ((r : ℕ) : ℤ) =
        some ((before ++ word ++ after)[r]'hr) := by
    intro r hr
    rw [FinTM.bufferTape_nat, List.getElem?_eq_getElem hr]
  have first := reads_append (FinTM.bufferTape (before ++ word ++ after)) 0 before (word ++ after)
    (by simpa only [zero_add, List.append_assoc] using whole)
  have second := reads_append (FinTM.bufferTape (before ++ word ++ after)) before.length word
    after
    (by simpa only [zero_add] using first.2)
  exact second.1

/-- Execute the inherited fixed payload, return to the cleared state word,
then install the complete successor. The exit still precedes head restoration.
The halting case consumes its false tag and keeps a live physical exit.

**Proof sketch.** Cite `recordTM_effect` for all simulated fields and its head
envelope. Use the exact inherited reader execution only to frame the two
auxiliary tapes. The two-cell move enters the cleared state region, and the
proved successor copier installs its new unary word. Join the three public
phase-host segments; their total is the full record length plus six. -/
private lemma roundTM_payload {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (table before after pre : List Bool) (ht : table = before ++ actionBitsK a ++ after) :
    let word := (a.state.map unaryFin).getD []
    RoundSpan c a
      (atKey (roundData c table (outputSummary c.output) [] before.length 0 pre) .record
        (recordTM k).q₀)
      (atKey (roundData (a.apply c) table (outputSummary (a.apply c).output) word
        (before.length + (actionBitsK a).length) (2 + word.length) pre)
        (.tableTail a.state.isSome) .run)
      ((actionBitsK a).length + 6) := by
  let word := (a.state.map unaryFin).getD []
  let d0 := roundData c table (outputSummary c.output) [] before.length 0 pre
  let d1 := roundData (a.apply c) table (outputSummary (a.apply c).output) []
    (before.length + recordWidth k) 0 pre
  let d2 := roundData (a.apply c) table (outputSummary (a.apply c).output) []
    (before.length + recordWidth k) 2 pre
  let d3 := roundData (a.apply c) table (outputSummary (a.apply c).output) word
    (before.length + (actionBitsK a).length) (2 + word.length) pre
  have reads : ∀ (r : ℕ) (hr : r < (actionBitsK a).length),
      d0.workTapes (tableTape k) (d0.workTapePos (tableTape k) + r) = some ((actionBitsK
        a)[r]'hr) := by
    simpa [d0, roundData, tableTape, ht] using bufferTape_reads before (actionBitsK a) after
  obtain ⟨bits, split, length, fixedReads, fieldReads⟩ := actionBitsK_reads d0 a reads
  have hbits : List.ofFn bits = (actionBitsK a).take (recordWidth k) := by
    rw [split]
    exact (List.take_left' (List.length_ofFn (f := bits))).symm
  have effect := recordTM_effect d0 c a bits rfl fixedReads hbits rfl
    (by intro i; simp [d0, roundData]) (by intro i; simp [d0, roundData])
    (by simp [d0, roundData, Cfg.workTapeSymbols, summaryTape, summaryCells])
  dsimp only at effect
  obtain ⟨recordState, recordInput, recordBank, recordSum, recordOutput, recordCut, _,
    recordHeads⟩ := effect
  have literal := recordTM_execute d0 a bits rfl fixedReads hbits
  have sumStep : summaryEmit (outputSummary c.output) a.output = outputSummary (a.apply c).output :=
    (outputSummary_append c.output a.output).symm
  have recordEnd : (recordTM k).runFrom d0 (recordWidth k + 1) = d1.mapState (fun _ => (none
    : RecordState k)) := by
    apply Cfg.ext
    · exact recordState
    · exact recordInput
    · funext j
      refine Fin.addCases (fun i => ?_) (fun i => ?_) j
      · simpa [d1, roundData, Cfg.mapState] using (recordBank i).1
      · fin_cases i
        · rw [literal]
          simp [applyPayload, readCfg, d0, d1, roundData, Cfg.mapState, tableTape]
        · rw [literal]
          simp [applyPayload, readCfg, d0, d1, roundData, Cfg.mapState, tableTape,
            summaryTape, Cfg.workTapeSymbols,
            show summaryCells (outputSummary c.output) [] 0 = outputSummary c.output from rfl,
            summaryCells_update, sumStep]
    · funext j
      refine Fin.addCases (fun i => ?_) (fun i => ?_) j
      · simpa [d1, roundData, Cfg.mapState] using (recordBank i).2
      · fin_cases i <;> rw [literal] <;>
          simp [applyPayload, readCfg, d0, d1, roundData, Cfg.mapState, tableTape]
    · exact recordOutput
  have p0 := roundTM_phase c a .record (by decide) (by decide) d0
    (d1.mapState (fun _ => (none : RecordState k))) _ recordEnd none rfl ⟨.toState, (0 : Fin 3)⟩
    (by simp [red]) (by
      intro u hu r hr
      have ne : r ≠ none := fun eq => recordCut u hu (hr.trans (congrArg some eq))
      simp [red, ne]) (by
      intro u hu
      have envelope := recordHeads u hu
      by_cases last : u = recordWidth k + 1
      · refine ⟨Or.inr ?_, fun i => Or.inr ?_⟩
        · simpa [last, d0, roundData] using envelope.2.1
        · simpa [last, d0, roundData] using envelope.1 i
      · refine ⟨Or.inl ?_, fun i => Or.inl ?_⟩
        · simpa [last, d0, roundData] using envelope.2.1
        · simpa [last, d0, roundData] using envelope.1 i)
  let moving := d1.mapState (fun _ => (0 : Fin 3))
  have moveEnd : (twoMoveTM (summaryTape k) .pos).runFrom moving 2 = d2.mapState (fun _ => (2
    : Fin 3)) := by
    rw [twoMoveTM_run _ _ moving rfl 2 le_rfl]
    apply Cfg.ext <;> try rfl
    funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simp [moving, d1, d2, roundData, Cfg.mapState, mirror_ne_summary k i]
    · fin_cases i <;> simp [moving, d1, d2, roundData, Cfg.mapState, summaryTape]
  have p1 := roundTM_phase c a .toState (by decide) (by decide) moving
    (d2.mapState (fun _ => (2 : Fin 3))) 2 moveEnd 2 rfl ⟨.successor, .tag⟩
    (by simp [red]) (by
      intro u hu r hr
      rw [twoMoveTM_run _ _ moving rfl u (by omega)] at hr
      have vr : r.val = u := congrArg Fin.val (Option.some.inj hr).symm
      have ne : r ≠ 2 := by intro bad; subst r; omega
      simp [red, ne]) (by
      intro u hu
      rw [twoMoveTM_run _ _ moving rfl u hu]
      refine ⟨Or.inr rfl, fun i => Or.inr ?_⟩
      simp [moving, d1, roundData, Cfg.mapState, mirror_ne_summary k i])
  let copying := d2.mapState (fun _ => SuccPhase.tag)
  have sourceField : ∀ (r : ℕ) (hr : r < (optStateBits a.state).length),
      copying.workTapes (tableTape k) (copying.workTapePos (tableTape k) + r) =
        some ((optStateBits a.state)[r]'hr) := by
    simpa only [copying, d2, d0, roundData, Cfg.mapState, tableTape, Fin.addCases_right,
      ↓reduceIte] using fieldReads
  have tailLength : (optStateBits a.state).length = word.length + 1 := by
    dsimp only [word]
    cases a.state <;> simp [optStateBits, unaryFin]
  have successorEnd : (succFieldTM (tableTape k) (summaryTape k) true).runFrom copying
    (optStateBits a.state).length =
      d3.mapState (fun _ => SuccPhase.done a.state.isSome) := by
    rw [succFieldTM_run _ _ (Ne.symm (summary_ne_table k)) true copying a.state rfl sourceField]
    apply Cfg.ext <;> try rfl
    · funext j z
      refine Fin.addCases (fun i => ?_) (fun i => ?_) j
      · simp [succFieldCfg, afterTag, copying, d2, d3, roundData, Cfg.mapState,
          mirror_ne_summary k i]
      · fin_cases i
        · simp [succFieldCfg, afterTag, copying, d2, d3, roundData, Cfg.mapState,
            tableTape, summaryTape]
        · simpa [succFieldCfg, afterTag, copying, d2, d3, roundData, Cfg.mapState,
            tableTape, summaryTape, word] using
            congrFun (summaryCells_copy (outputSummary (a.apply c).output) word) z
    · funext j
      refine Fin.addCases (fun i => ?_) (fun i => ?_) j
      · simp [succFieldCfg, afterTag, copying, d2, d3, roundData, Cfg.mapState,
          mirror_ne_summary k i, mirror_ne_table k i]
      · fin_cases i
        · simp [succFieldCfg, afterTag, copying, d2, d3, roundData, Cfg.mapState,
            tableTape, summaryTape]
          change _ + _ + 1 + (word.length : ℤ) = _
          rw [length, tailLength]
          push_cast
          ring
        · simp [succFieldCfg, afterTag, copying, d2, d3, roundData, Cfg.mapState,
            tableTape, summaryTape, word]
  have p2 := roundTM_phase c a .successor (by decide) (by decide) copying
    (d3.mapState (fun _ => SuccPhase.done a.state.isSome)) _ successorEnd (.done
      a.state.isSome) rfl
    ⟨.tableTail a.state.isSome, .run⟩ rfl (by
      intro u hu r hr
      cases r with
      | tag => rfl
      | unary => rfl
      | done b => exact False.elim (succFieldTM_cut _ _ (Ne.symm (summary_ne_table k)) true copying
        a.state rfl sourceField u hu b hr)) (by
      intro u hu
      have heads := succFieldTM_heads _ _ (Ne.symm (summary_ne_table k)) true copying a.state rfl
        sourceField u hu
      refine ⟨Or.inr heads.1, fun i => Or.inr ?_⟩
      simpa [copying, d2, roundData, Cfg.mapState, mirror_ne_summary k i, mirror_ne_table k i]
        using heads.2.2 (Fin.castAdd 2 i))
  have combined := RoundSpan.add c a (RoundSpan.add c a p0 p1) p2
  have cost : (recordWidth k + 1 + 1) + (2 + 1) + ((optStateBits a.state).length + 1) =
      (actionBitsK a).length + 6 := by omega
  simpa only [atKey, d0, d3, moving, copying, d1, d2, Cfg.mapState, roundData, cost] using combined

/-- A canonical buffer's suffix, including its right blank and all cells
beyond it. This is the optional-list lookup law at a nonnegative offset. -/
private lemma bufferTape_suffix (before word : List Bool) (z : ℤ) (hz : 0 ≤ z) :
    FinTM.bufferTape (before ++ word) (before.length + z) = FinTM.bufferTape word z := by
  have hp : (before.length : ℤ) + z = ((before.length + z.toNat : ℕ) : ℤ) := by omega
  rw [hp, FinTM.bufferTape_nat, List.getElem?_append_right (Nat.le_add_right _ _),
    Nat.add_sub_cancel_left]
  simp [FinTM.bufferTape, hz]

/-- Restore the table and successor-state heads after payload execution.
The table scan reaches its right blank and the catalog reverse scan returns
it to zero. The copied state word is then rewound to two, and the final fixed
move returns the summary head to zero. The live/halt distinction is retained.

**Proof sketch.** Compose the four landed framed seek contracts and the
fixed two-cell move through the public phase host. All five segments preserve
the source action's final simulated heads and exclude both round anchors until
the last dispatch. The exact cost counts every delimiter and dispatch. -/
private lemma roundTM_restore {k n : ℕ} {x : List Bool}
    (c : Cfg k Bool (Fin (n + 1)) x) (a : Action k Bool (Fin (n + 1)))
    (table before tail word pre : List Bool) (live : Bool) (ht : table = before ++ tail) :
    RoundSpan c a
      (atKey (roundData (a.apply c) table (outputSummary (a.apply c).output) word
        before.length (2 + word.length) pre) (.tableTail live) .run)
      ((roundData (a.apply c) table (outputSummary (a.apply c).output) word 0 0 pre).mapState
        (fun _ => exit k live))
      (tail.length + table.length + word.length + 15) := by
  let data := fun tp sp => roundData (a.apply c) table (outputSummary (a.apply c).output) word tp sp
    pre
  let d0 := data before.length (2 + word.length)
  let d1 := data (table.length + 1) (2 + word.length)
  let d2 := data 0 (2 + word.length)
  let d3 := data 0 (word.length + 3)
  let d4 := data 0 2
  let d5 := data 0 0
  let s0 := d0.mapState (fun _ => SeekPhase.run)
  let s1 := d1.mapState (fun _ => SeekPhase.run)
  let s2 := d2.mapState (fun _ => SeekPhase.run)
  let s3 := d3.mapState (fun _ => SeekPhase.run)
  let s4 := d4.mapState (fun _ => (0 : Fin 3))
  have tableLength : table.length = before.length + tail.length := by rw [ht, List.length_append]
  have tableSkip := skipTM_run_ofCfg (tableTape k) tail s0 rfl (by
    intro z hz _
    simpa [s0, d0, data, roundData, Cfg.mapState, tableTape, ht] using bufferTape_suffix before tail
      z hz)
  have tableSkipEnd : (skipTM (k + 2) (tableTape k)).runFrom s0 (tail.length + 1) =
      d1.mapState (fun _ => SeekPhase.done) := by
    rw [tableSkip.1]
    apply Cfg.ext <;> try rfl
    funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simp [s0, d0, d1, data, roundData, Cfg.mapState, mirror_ne_table k i]
    · fin_cases i <;> simp [s0, d0, d1, data, roundData, Cfg.mapState, tableTape, tableLength,
      Nat.cast_add, add_assoc]
  have p0 := roundTM_phase c a (.tableTail live) (by intro h; cases h) (by intro h; cases h) s0
    (d1.mapState (fun _ => SeekPhase.done)) _ tableSkipEnd .done rfl ⟨.tableHome live, .run⟩
    (by simp [red]) (by
      intro u hu r hr
      have ne : r ≠ .done := fun eq => (tableSkip.2.1 u hu) (hr.trans (congrArg some eq))
      simp [red, ne]) (by
      intro u hu

/-- Exact complete-round execution from a canonical data frame. The cost
is bounded by fifty times the table length plus one, uniformly in the state
count, input, and simulated time. Both continuation and halt use live exits.

**Proof sketch.** Join lookup, preparation, inherited payload execution,
successor copy, and restoration. The one entry dispatch is separate so the
strict interior excludes both round anchors. The serializer cut pays for the
old unary state, selected record, suffix scan, and new unary state; the
remaining constants count dispatches and delimiter cells. -/
private lemma roundTM_canonical {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (q : Fin (n + 1)) (pre : List Bool) :
    let a := tr q c.inputSymbol c.workTapeSymbols
    let start :=
      atKey (roundData c (tableBitsK tr) (outputSummary c.output) (unaryFin q) 0 0 pre) .entry ()
    let finish := (roundData (a.apply c) (tableBitsK tr) (outputSummary (a.apply c).output)
      ((a.state.map unaryFin).getD []) 0 0 pre).mapState (fun _ => exit k a.state.isSome)
    ∃ T, 0 < T ∧ T ≤ 50 * ((tableBitsK tr).length + 1) ∧
      (roundTM k).runFrom start T = finish ∧
      (∀ u, 0 < u → u < T → ∃ s, ((roundTM k).runFrom start u).state = some s ∧
        s ≠ ⟨.entry, ()⟩ ∧ s ≠ ⟨.halt, ()⟩) ∧
      ∀ u ≤ T, RoundWindow c a ((roundTM k).runFrom start u) := by
  let a := tr q c.inputSymbol c.workTapeSymbols
  let d := roundData c (tableBitsK tr) (outputSummary c.output) (unaryFin q) 0 0 pre
  let word := (a.state.map unaryFin).getD []
  obtain ⟨before, tail, L, split, stateSize, lookupBound, lookup⟩ := roundTM_lookup tr c q pre
  change tableBitsK tr = before ++ actionBitsK a ++ tail at split
  have prepare := roundTM_prepare c a q (tableBitsK tr) (outputSummary c.output) before.length pre
  have payload := roundTM_payload c a (tableBitsK tr) before tail pre split
  have restore := roundTM_restore c a (tableBitsK tr) (before ++ actionBitsK a) tail word pre
    a.state.isSome (by simpa only [List.append_assoc] using split)
  simp only [List.length_append, Nat.cast_add] at restore
  have joined := RoundSpan.add c a (RoundSpan.add c a (RoundSpan.add c a lookup prepare)
    payload) restore
  let U := L + (3 * q.val + 16) + ((actionBitsK a).length + 6) +
    (tail.length + (tableBitsK tr).length + word.length + 15)
  have entry : (roundTM k).step (atKey d .entry ()) = atKey d .lookup (seekTM k).q₀ := by
    have one := step_of_inputMoveAction (roundTM k) (atKey d .entry ())
      (q := ⟨.entry, ()⟩) (q' := ⟨.lookup, (seekTM k).q₀⟩) 0 rfl rfl
    simpa only [moveInputPos_zero, atKey, d, roundData, Cfg.mapState] using one
  have advance (u : ℕ) := MultiTapeTM.runFrom_succ_eq_step (tm := roundTM k) (cfg := atKey d
    .entry ()) (t := u)
  simp only [entry] at advance
  have run : (roundTM k).runFrom (atKey d .entry ()) (U + 1) =
      (roundData (a.apply c) (tableBitsK tr) (outputSummary (a.apply c).output) word 0 0
        pre).mapState
        (fun _ => exit k a.state.isSome) := by
    rw [advance]
    exact joined.1
  refine ⟨U + 1, by omega, ?_, run, ?_, ?_⟩
  · have recordSize : (actionBitsK a).length ≤ (tableBitsK tr).length := by
      have hs := congrArg List.length split
      simp only [List.length_append] at hs
      omega
    have prefixSize : before.length ≤ (tableBitsK tr).length := by
      have hs := congrArg List.length split
      simp only [List.length_append] at hs
      omega
    have suffixSize : tail.length ≤ (tableBitsK tr).length := by
      have hs := congrArg List.length split
      simp only [List.length_append] at hs
      omega
    have wordSize : word.length ≤ (actionBitsK a).length := by
      obtain ⟨bits, hs⟩ := actionBitsK_split a
      have hl := congrArg List.length hs
      simp only [List.length_append, List.length_ofFn] at hl
      have tailLen : (optStateBits a.state).length = word.length + 1 := by
        dsimp only [word]
        cases a.state <;> simp [optStateBits, unaryFin]
      omega
    dsimp only [U]
    omega
  · intro u hu ht
    rw [show u = (u - 1) + 1 by omega, advance]
    obtain ⟨s, hs, he, hh, _⟩ := joined.2 (u - 1) (by dsimp only [U] at ht; omega)
    exact ⟨s, hs, he, hh⟩
  · intro u hu
    by_cases hz : u = 0
    · subst u
      exact roundData_window c a (tableBitsK tr) (outputSummary c.output) (unaryFin q) 0 0 pre false
    · by_cases hend : u = U + 1
      · rw [hend, run]
        exact roundData_window c a (tableBitsK tr) (outputSummary (a.apply c).output) word 0 0
          pre true
      · rw [show u = (u - 1) + 1 by omega, advance]
        exact (joined.2 (u - 1) (by dsimp only [U] at hu; omega)).choose_spec.2.2.2

/-- Relation at the generic step block's seam. The first `k` work tapes
mirror the source, and the native input position is its simulated input head.
Auxiliary tape `k` holds the immutable public table; tape `k + 1` holds the
output summary at zero, a blank at one, and the unary state at two. Both
auxiliary heads are home at zero. Simulated halt has an empty state word.
Physical control and the physical output prefix are deliberately independent. -/
def Rep {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x) : Prop :=
  d.inputPos = c.inputPos ∧
  (∀ i : Fin k, d.workTapes (Fin.castAdd 2 i) = c.workTapes i ∧
    d.workTapePos (Fin.castAdd 2 i) = c.workTapePos i) ∧
  d.workTapes (tableTape k) = FinTM.bufferTape (tableBitsK tr) ∧
  d.workTapes (summaryTape k) =
    summaryCells (outputSummary c.output) ((c.state.map unaryFin).getD []) ∧
  d.workTapePos (tableTape k) = 0 ∧ d.workTapePos (summaryTape k) = 0

/-- A related entry configuration is exactly the canonical round frame,
with its arbitrary physical output prefix retained. This only unpacks the
seam relation; it makes no assumption about the mirrored source tapes. -/
private lemma rep_entry {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (q : Fin (n + 1)) (hq : c.state = some q)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x)
    (entry : d.state = some ⟨.entry, ()⟩) (rel : Rep tr c d) :
    d = atKey (roundData c (tableBitsK tr) (outputSummary c.output) (unaryFin q) 0 0 d.output)
      .entry () := by
  obtain ⟨input, bank, table, summary, tableHome, stateHome⟩ := rel
  simp only [hq, Option.map_some, Option.getD_some] at summary
  apply Cfg.ext
  · exact entry
  · exact input
  · funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simpa only [atKey, Cfg.mapState, roundData, Fin.addCases_left] using (bank i).1
    · fin_cases i
      · simpa [atKey, Cfg.mapState, roundData, tableTape] using table
      · simpa [atKey, Cfg.mapState, roundData, summaryTape] using summary
  · funext j
    refine Fin.addCases (fun i => ?_) (fun i => ?_) j
    · simpa only [atKey, Cfg.mapState, roundData, Fin.addCases_left] using (bank i).2
    · fin_cases i
      · simpa [atKey, Cfg.mapState, roundData, tableTape] using tableHome
      · simpa [atKey, Cfg.mapState, roundData, summaryTape] using stateHome
  · rfl

/-- **Promotion candidate (host H2, step block; machine-library-design.md §14).**
One finite physical machine `roundFinTM k`, whose control depends only on
`k`, realizes one transition of any canonized `k`-tape table. `Rep`
fully specifies the seam. The table and unary state are data, never control.
The source action is applied even when its successor is `none`; the block
then reaches the separate live halt exit, after restoring both auxiliaries.

The bound is `50 * (|table| + 1)`. Every simulated head stays at either its
start or its action endpoint, at distance at most one. Every auxiliary head
stays within the same linear bound and returns home. Physical output is
silent at every intermediate time; simulated emission updates the summary.
The strict interior visits neither the continuation anchor nor the halt exit.

**Proof sketch.** Unpack the representation and invoke the complete canonical
round. Output-prefix monotonicity gives silence throughout. The phase window
and the public action head bound give the mirrored envelope. Summing the
public unit step bound gives the auxiliary envelope. The input clamp handles
both boundary blanks. No space ledger or alternative trace is constructed. -/
theorem roundTM_effect {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (q : Fin (n + 1)) (hq : c.state = some q)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x)
    (entry : d.state = some ⟨.entry, ()⟩) (rel : Rep tr c d) :
    let a := tr q c.inputSymbol c.workTapeSymbols
    ∃ T, 0 < T ∧ T ≤ 50 * ((tableBitsK tr).length + 1) ∧
      let e := (roundTM k).runFrom d T
      e.state = some (exit k a.state.isSome) ∧ Rep tr (a.apply c) e ∧
      e.workTapeSymbols (summaryTape k) = summaryEmit (d.workTapeSymbols (summaryTape k)) a.output ∧
      (∀ u, 0 < u → u < T → ∃ s, ((roundTM k).runFrom d u).state = some s ∧
        s ≠ ⟨.entry, ()⟩ ∧ s ≠ ⟨.halt, ()⟩) ∧
      (∀ u ≤ T, ((roundTM k).runFrom d u).output = d.output) ∧
      (∀ u ≤ T,
        let f := (roundTM k).runFrom d u
        (f.inputPos = d.inputPos ∨ f.inputPos = e.inputPos) ∧
        |(f.inputPos.val : ℤ) - d.inputPos.val| ≤ 1 ∧
        (∀ i : Fin k, f.workTapePos (Fin.castAdd 2 i) = d.workTapePos (Fin.castAdd 2 i) ∨
          f.workTapePos (Fin.castAdd 2 i) = e.workTapePos (Fin.castAdd 2 i)) ∧
        ∀ j : Fin (k + 2), |f.workTapePos j - d.workTapePos j| ≤
          if j.val < k then 1 else (50 * ((tableBitsK tr).length + 1) : ℕ)) ∧
      ∀ j : Fin (k + 2), k ≤ j.val → d.workTapePos j = 0 ∧ e.workTapePos j = 0 := by
  let a := tr q c.inputSymbol c.workTapeSymbols
  have canonical := rep_entry tr c q hq d entry rel
  obtain ⟨T, positive, bound, run, cut, window⟩ := roundTM_canonical tr c q d.output
  rw [← canonical] at run cut window
  refine ⟨T, positive, bound, ?_, ?_, ?_, cut, ?_, ?_, ?_⟩
  · rw [run]
    rfl
  · rw [run]
    simp [Rep, roundData, Cfg.mapState, Action.apply, tableTape, summaryTape]
  · have old : d.workTapeSymbols (summaryTape k) = outputSummary c.output := by
      rw [canonical]
      simp [atKey, roundData, Cfg.mapState, Cfg.workTapeSymbols, summaryTape, summaryCells]
    rw [run, old]
    simpa only [roundData, Cfg.mapState, Cfg.workTapeSymbols, summaryTape,
      Fin.addCases_right, show (1 : Fin 2) ≠ 0 from by decide, ↓reduceIte, summaryCells] using
      (outputSummary_append c.output a.output)
  · intro u hu
    have lo := (roundTM k).output_prefix d (Nat.zero_le u)
    have hi := (roundTM k).output_prefix d hu
    rw [run] at hi
    exact hi.eq_of_length_le lo.length_le
  · intro u hu
    have frame := window u hu
    have finalInput : ((roundTM k).runFrom d T).inputPos = (a.apply c).inputPos := by
      rw [run]
      rfl
    have finalHead (i : Fin k) : ((roundTM k).runFrom d T).workTapePos (Fin.castAdd 2 i) =
        (a.apply c).workTapePos i := by
      rw [run]
      simp only [roundData, Cfg.mapState, Fin.addCases_left]
      rfl
    have inputStep : |((a.apply c).inputPos.val : ℤ) - c.inputPos.val| ≤ 1 := by
      change |((moveInputPos c.inputPos a.inputTape).val : ℤ) - c.inputPos.val| ≤ 1
      apply abs_le.mpr
      have hp := c.inputPos.isLt
      cases h : a.inputTape <;> simp only [moveInputPos, SignType.cast] <;>
        split <;> simp <;> omega
    refine ⟨?_, ?_, ?_, ?_⟩
    · simpa only [rel.1, finalInput] using frame.1
    · rw [rel.1]
      rcases frame.1 with start | finish
      · rw [start]; simp
      · rw [finish]; exact inputStep
    · intro i
      simpa only [(rel.2.1 i).2, finalHead i] using frame.2 i
    · intro j
      by_cases hj : j.val < k
      · let i : Fin k := ⟨j.val, hj⟩
        have ji : j = Fin.castAdd 2 i := Fin.ext rfl
        rw [if_pos hj, ji, (rel.2.1 i).2]
        rcases frame.2 i with start | finish
        · rw [start]; simp
        · rw [finish]; exact workTapePos_apply_le a c i
      · rw [if_neg hj]
        exact (MultiTapeTM.abs_workTapePos_runFrom_sub_le (roundTM k) d u j).trans (by omega)
  · intro j hj
    have home (e : Cfg (k + 2) Bool (Σ key, Ph k key) x)
        (ht : e.workTapePos (tableTape k) = 0) (hs : e.workTapePos (summaryTape k) = 0) :
          e.workTapePos j = 0 := by
      have casesAux : j = tableTape k ∨ j = summaryTape k := by
        have limit := j.isLt
        have indices : j.val = k ∨ j.val = k + 1 := by omega
        rcases indices with h | h
        · left; apply Fin.ext; simpa [tableTape] using h
        · right; apply Fin.ext; simpa [summaryTape] using h
      rcases casesAux with rfl | rfl
      · exact ht
      · exact hs
    refine ⟨home d rel.2.2.2.2.1 rel.2.2.2.2.2, ?_⟩
    apply home
    · rw [run]; simp [roundData, Cfg.mapState, tableTape]
    · rw [run]; simp [roundData, Cfg.mapState, summaryTape]

/-- **A blank summary cell records one emission** (additive): updating the empty summary by an
emission gives that emission. Hence a host that blanks the summary cell before a round finds
the round's emission `a.output` there afterwards (the summary clause of `roundTM_effect`), and
can forward it verbatim or append it to a buffer, with the step block unchanged.

**Proof sketch.** Case on the emission. -/
lemma summaryEmit_none (b : Option Bool) : summaryEmit none b = b := by
  rcases b with _ | _ | _ <;> rfl

/-- **Visited cells split at an intermediate time** (additive; generic, proposed as
`Turing.MultiTapeTM.visitedByTapeHead_add_subset`, design note O9): the cells visited by
head `j` in `T₁ + T₂` steps from `d` are those visited in the first `T₁` steps, together with
those visited in `T₂` steps from where the run is at time `T₁`.

**Proof sketch.** A visiting time `u ≤ T₁ + T₂` is either at most `T₁`, or `T₁ + v` with
`v ≤ T₂`, and then run additivity identifies the configuration. -/
lemma _root_.Turing.MultiTapeTM.visitedByTapeHead_add_subset {m : ℕ} {S : Type} {x : List Bool}
    (M : MultiTapeTM m Bool S) (d : Cfg m Bool S x) (T₁ T₂ : ℕ) (j : Fin m) :
    M.visitedByTapeHead d (T₁ + T₂) j ⊆
      M.visitedByTapeHead d T₁ j ∪ M.visitedByTapeHead (M.runFrom d T₁) T₂ j := by
  intro p hp
  simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_union, Finset.mem_image,
    Finset.mem_range] at hp ⊢
  obtain ⟨u, hu, rfl⟩ := hp
  by_cases hle : u ≤ T₁
  · exact Or.inl ⟨u, by omega, rfl⟩
  · refine Or.inr ⟨u - T₁, by omega, ?_⟩
    rw [← MultiTapeTM.runFrom_add]
    congr 2
    omega

/-- **Visited cells after the first step** (additive; generic, proposed as
`Turing.MultiTapeTM.visitedByTapeHead_step_subset`): the cells visited in `t` steps
from the successor of `c`, and `c`'s own head cell, are among those visited in `t + 1` steps
from `c`.

**Proof sketch.** Shift every visiting time by one; time `0` gives `c`'s cell. -/
lemma _root_.Turing.MultiTapeTM.visitedByTapeHead_step_subset {m : ℕ} {S : Type} {x : List Bool}
    (M : MultiTapeTM m Bool S) (c : Cfg m Bool S x) (t : ℕ) (j : Fin m) :
    insert (c.workTapePos j) (M.visitedByTapeHead (M.step c) t j) ⊆
      M.visitedByTapeHead c (t + 1) j := by
  intro p hp
  simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_insert, Finset.mem_image,
    Finset.mem_range] at hp ⊢
  rcases hp with rfl | ⟨v, hv, rfl⟩
  · exact ⟨0, by omega, by rw [MultiTapeTM.runFrom_zero]⟩
  · exact ⟨v + 1, by omega, by rw [MultiTapeTM.runFrom_succ_eq_step]⟩

/-- **The per-round space ledger** (additive): under the contract's hypotheses, with `a` the
action of `tr` at `c` and `B = 50·(|tableBitsK tr| + 1)`, some time `T` with `0 < T ≤ B` ends
the round at `exit k a.state.isSome` in a configuration representing `a.apply c`, with the
contract's cut and silence, and during the round each mirrored head visits only the source
head's cells before and after the step, and each auxiliary head only cells in `[-B, B]`. In
particular a round visits at most `2` cells of a mirrored tape and `2B + 1` of an auxiliary
one.

**Proof sketch.** Take the contract's `T`. The envelope puts each mirrored head at its start
or end position, which the two representations identify with the source's head before and
after the step; each auxiliary head starts at `0` and stays within `B` of it. -/
theorem roundTM_visited {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (q : Fin (n + 1)) (hq : c.state = some q)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x)
    (entry : d.state = some ⟨.entry, ()⟩) (rel : Rep tr c d) :
    let a := tr q c.inputSymbol c.workTapeSymbols
    ∃ T, 0 < T ∧ T ≤ 50 * ((tableBitsK tr).length + 1) ∧
      ((roundTM k).runFrom d T).state = some (exit k a.state.isSome) ∧
      Rep tr (a.apply c) ((roundTM k).runFrom d T) ∧
      (∀ u, 0 < u → u < T → ∃ s, ((roundTM k).runFrom d u).state = some s ∧
        s ≠ ⟨.entry, ()⟩ ∧ s ≠ ⟨.halt, ()⟩) ∧
      (∀ u ≤ T, ((roundTM k).runFrom d u).output = d.output) ∧
      (∀ i : Fin k, (roundTM k).visitedByTapeHead d T (Fin.castAdd 2 i) ⊆
        {c.workTapePos i, (a.apply c).workTapePos i}) ∧
      ∀ j : Fin (k + 2), k ≤ j.val → (roundTM k).visitedByTapeHead d T j ⊆
        Finset.Icc (-((50 * ((tableBitsK tr).length + 1) : ℕ) : ℤ))
          ((50 * ((tableBitsK tr).length + 1) : ℕ) : ℤ) := by
  intro a
  obtain ⟨T, hpos, hT, hexit, hrel, -, hcut, hsil, henv, hhome⟩ :=
    roundTM_effect tr c q hq d entry rel
  refine ⟨T, hpos, hT, hexit, hrel, hcut, hsil, ?_, ?_⟩
  · intro i p hp
    simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_image, Finset.mem_range] at hp
    obtain ⟨u, hu, rfl⟩ := hp
    have h := (henv u (by omega)).2.2.1 i
    rw [(rel.2.1 i).2, (hrel.2.1 i).2] at h
    rcases h with h | h <;> rw [h]
    · exact Finset.mem_insert_self _ _
    · exact Finset.mem_insert_of_mem (Finset.mem_singleton_self _)
  · intro j hj p hp
    simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_image, Finset.mem_range] at hp
    obtain ⟨u, hu, rfl⟩ := hp
    have h := (henv u (by omega)).2.2.2 j
    rw [if_neg (by omega), (hhome j hj).1, sub_zero] at h
    rw [Finset.mem_Icc]
    exact abs_le.mp h

/-- **Rounds compose into a run of the source** (additive, for unclocked hosts: B6's `U₀`,
backlog U-1). Let `M` be a `k`-work-tape binary machine with `n + 1` states and `c` a
configuration of it on `x` that stays live for `t` steps (the `t`-th configuration
included), and let `B = 50·(|tableBitsK M.tr| + 1)`. If the step block is at `entry` in a
configuration `d` representing `c` for `M`'s table, then at some time `T` with
`t ≤ T ≤ t·B` it is back at `entry`, representing `M`'s configuration after `t` steps; up to
`T` it has emitted nothing and never visited `halt`; each mirrored head has visited only cells
that `M`'s head visited in those `t` steps; and each auxiliary head only cells in `[-B, B]`.
[AB09, §1.4.1, proof of Theorem 1.9]

The lower bound `t ≤ T` makes the block's time cofinal with the source's: if the source never
halts, then for every physical horizon `u` the theorem at `t = u` gives `T ≥ u` with no visit
to `halt` up to `T`, so the block never reaches `halt` (round 1 finding 1; the stationary
silent self-loop no longer admits the witness `T = 0`). This is the silent block: the
emitting wrapper that forwards the summary cell is the consumer obligation CO-RF of the design
note, not a consequence of this theorem.

**Proof sketch.** Induct on `t`. At `t + 1`, apply the per-round ledger at the live `c`: the
source's step at `c` is the record's action, and its successor is live, so the round ends at
`entry` representing `M.step c`, after a positive time at most `B`. The induction hypothesis at
`M.step c` continues from there, by run additivity; the positive round time and the inductive
lower bound give `t + 1 ≤ T`. Silence and the absence of `halt` hold round by round: a round's
interior avoids `halt` by the cut, its endpoint is `entry`. The visited cells split at the
round boundary; the first round's are the source's cells at times `0` and `1`, the rest are
covered by induction and the shift by one step. -/
theorem roundTM_iterate {k n : ℕ} {x : List Bool} (M : MultiTapeTM k Bool (Fin (n + 1)))
    (t : ℕ) : ∀ (c : Cfg k Bool (Fin (n + 1)) x),
    (∀ u ≤ t, (M.runFrom c u).state.isSome) →
    ∀ (d : Cfg (k + 2) Bool (Σ key, Ph k key) x),
    d.state = some ⟨.entry, ()⟩ → Rep M.tr c d →
    ∃ T, t ≤ T ∧ T ≤ t * (50 * ((tableBitsK M.tr).length + 1)) ∧
      ((roundTM k).runFrom d T).state = some ⟨.entry, ()⟩ ∧
      Rep M.tr (M.runFrom c t) ((roundTM k).runFrom d T) ∧
      (∀ u ≤ T, ((roundTM k).runFrom d u).output = d.output ∧
        ((roundTM k).runFrom d u).state ≠ some ⟨.halt, ()⟩) ∧
      (∀ i : Fin k, (roundTM k).visitedByTapeHead d T (Fin.castAdd 2 i) ⊆
        M.visitedByTapeHead c t i) ∧
      ∀ j : Fin (k + 2), k ≤ j.val → (roundTM k).visitedByTapeHead d T j ⊆
        Finset.Icc (-((50 * ((tableBitsK M.tr).length + 1) : ℕ) : ℤ))
          ((50 * ((tableBitsK M.tr).length + 1) : ℕ) : ℤ) := by
  have entry_ne_halt : (some ⟨.entry, ()⟩ : Option (Σ key, Ph k key)) ≠
      some ⟨.halt, ()⟩ := by
    intro h
    cases (Sigma.mk.inj (Option.some.inj h)).1
  induction t with
  | zero =>
    intro c _ d hd rel
    refine ⟨0, le_rfl, Nat.zero_le _, hd, rel, ?_, ?_, ?_⟩
    · intro u hu
      have hu0 : u = 0 := by omega
      subst hu0
      rw [MultiTapeTM.runFrom_zero, hd]
      exact ⟨rfl, entry_ne_halt⟩
    · intro i p hp
      simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_image, Finset.mem_range] at hp ⊢
      obtain ⟨u, hu, rfl⟩ := hp
      have hu0 : u = 0 := by omega
      subst hu0
      exact ⟨0, by omega, by rw [MultiTapeTM.runFrom_zero, MultiTapeTM.runFrom_zero,
        (rel.2.1 i).2]⟩
    · intro j hj p hp
      simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_image, Finset.mem_range] at hp
      obtain ⟨u, hu, rfl⟩ := hp
      have hu0 : u = 0 := by omega
      subst hu0
      have h0 : d.workTapePos j = 0 := by
        have hjv : j.val = k ∨ j.val = k + 1 := by omega
        rcases hjv with h | h
        · have : j = tableTape k := Fin.ext (by simpa [tableTape] using h)
          rw [this]; exact rel.2.2.2.2.1
        · have : j = summaryTape k := Fin.ext (by simpa [summaryTape] using h)
          rw [this]; exact rel.2.2.2.2.2
      rw [MultiTapeTM.runFrom_zero, h0, Finset.mem_Icc]
      omega
  | succ t ih =>
    intro c hlive d hd rel
    have hc := hlive 0 (Nat.zero_le _)
    rw [MultiTapeTM.runFrom_zero] at hc
    obtain ⟨q, hq⟩ := Option.isSome_iff_exists.mp hc
    have hstep : M.step c = (M.tr q c.inputSymbol c.workTapeSymbols).apply c := by
      simp only [MultiTapeTM.step, hq]
    obtain ⟨T₁, hpos, hT₁, hexit, hrel, hcut, hsil, hvis, haux⟩ :=
      roundTM_visited M.tr c q hq d hd rel
    have hlive' : ∀ u ≤ t, (M.runFrom (M.step c) u).state.isSome := by
      intro u hu
      rw [← MultiTapeTM.runFrom_succ_eq_step]
      exact hlive (u + 1) (by omega)
    have hsucc : (M.tr q c.inputSymbol c.workTapeSymbols).state.isSome := by
      have h1 := hlive' 0 (Nat.zero_le _)
      rwa [MultiTapeTM.runFrom_zero, hstep] at h1
    rw [hsucc] at hexit
    rw [← hstep] at hrel hvis
    obtain ⟨T₂, hlow, hT₂, hent, hrel₂, hrest, hvis₂, haux₂⟩ :=
      ih (M.step c) hlive' _ hexit hrel
    refine ⟨T₁ + T₂, by omega, ?_, ?_, ?_, ?_, ?_, ?_⟩
    · rw [Nat.succ_mul]; omega
    · rw [MultiTapeTM.runFrom_add]; exact hent
    · rw [MultiTapeTM.runFrom_add (tm := roundTM k) d T₁ T₂,
        MultiTapeTM.runFrom_succ_eq_step (tm := M)]
      exact hrel₂
    · intro u hu
      by_cases hle : u ≤ T₁
      · refine ⟨hsil u hle, ?_⟩
        by_cases h0 : u = 0
        · subst h0
          rw [MultiTapeTM.runFrom_zero, hd]
          exact entry_ne_halt
        · by_cases hT : u = T₁
          · subst hT
            rw [hexit]
            exact entry_ne_halt
          · obtain ⟨s, hs, -, hh⟩ := hcut u (by omega) (by omega)
            rw [hs]
            exact fun h => hh (Option.some.inj h)
      · obtain ⟨hout, hhalt⟩ := hrest (u - T₁) (by omega)
        have hu' : u = T₁ + (u - T₁) := by omega
        rw [hu', MultiTapeTM.runFrom_add]
        refine ⟨?_, hhalt⟩
        rw [hout]
        exact hsil T₁ le_rfl
    · intro i p hp
      have hsplit := MultiTapeTM.visitedByTapeHead_add_subset (roundTM k) d T₁ T₂ (Fin.castAdd 2 i) hp
      have hshift := MultiTapeTM.visitedByTapeHead_step_subset M c t i
      rcases Finset.mem_union.mp hsplit with h | h
      · have h' := hvis i h
        simp only [Finset.mem_insert, Finset.mem_singleton] at h'
        rcases h' with rfl | rfl
        · exact hshift (Finset.mem_insert_self _ _)
        · apply hshift
          apply Finset.mem_insert_of_mem
          simp only [MultiTapeTM.visitedByTapeHead, Finset.mem_image, Finset.mem_range]
          exact ⟨0, by omega, by rw [MultiTapeTM.runFrom_zero]⟩
      · exact hshift (Finset.mem_insert_of_mem (hvis₂ i h))
    · intro j hj p hp
      have hsplit := MultiTapeTM.visitedByTapeHead_add_subset (roundTM k) d T₁ T₂ j hp
      rcases Finset.mem_union.mp hsplit with h | h
      · exact haux j hj h
      · exact haux₂ j hj h

/-- **The round time is the unique first positive exit** (additive). Under the contract's
hypotheses there is exactly one time `T` such that `0 < T`, the block is at `entry` or at
`halt` at time `T`, and it is at neither exit at any time strictly between `0` and `T`. Hence
the existential time of `roundTM_effect` is this `T`, whichever witness is chosen, and a host
may define its round-time function by choice or by `Nat.find` without ambiguity (design note
O7; counter-loop round functions; MIG-0b's round time).

**Proof sketch.** Existence: the contract's time is positive, its endpoint is at
`exit k a.state.isSome`, which is `entry` or `halt`, and its cut keeps every strictly earlier
positive time off both exits. Uniqueness: of two such times, the smaller would lie strictly
between `0` and the larger, where the larger one's cut forbids both exits. -/
theorem roundTM_firstExit_unique {k n : ℕ} {x : List Bool}
    (tr : Fin (n + 1) → Option Bool → (Fin k → Option Bool) → Action k Bool (Fin (n + 1)))
    (c : Cfg k Bool (Fin (n + 1)) x) (q : Fin (n + 1)) (hq : c.state = some q)
    (d : Cfg (k + 2) Bool (Σ key, Ph k key) x)
    (entry : d.state = some ⟨.entry, ()⟩) (rel : Rep tr c d) :
    ∃! T, 0 < T ∧
      (((roundTM k).runFrom d T).state = some ⟨.entry, ()⟩ ∨
        ((roundTM k).runFrom d T).state = some ⟨.halt, ()⟩) ∧
      ∀ u, 0 < u → u < T →
        ((roundTM k).runFrom d u).state ≠ some ⟨.entry, ()⟩ ∧
          ((roundTM k).runFrom d u).state ≠ some ⟨.halt, ()⟩ := by
  obtain ⟨T, hpos, -, hexit, -, -, hcut, -⟩ := roundTM_effect tr c q hq d entry rel
  refine ⟨T, ⟨hpos, ?_, ?_⟩, ?_⟩
  · rw [hexit]
    unfold exit
    split
    · exact Or.inl rfl
    · exact Or.inr rfl
  · intro u hu huT
    obtain ⟨s, hs, hse, hsh⟩ := hcut u hu huT
    rw [hs]
    exact ⟨fun h => hse (Option.some.inj h), fun h => hsh (Option.some.inj h)⟩
  · rintro T' ⟨hpos', hexit', hcut'⟩
    rcases Nat.lt_trichotomy T' T with hlt | heq | hgt
    · obtain ⟨s, hs, hse, hsh⟩ := hcut T' hpos' hlt
      rw [hs] at hexit'
      rcases hexit' with h | h
      · exact absurd (Option.some.inj h) hse
      · exact absurd (Option.some.inj h) hsh
    · exact heq
    · have h := hcut' T hpos hgt
      rw [hexit] at h
      unfold exit at h
      split at h
      · exact absurd rfl h.1
      · exact absurd rfl h.2

end StepBlock
