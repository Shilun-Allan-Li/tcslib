"""H2-M text edits: build the alpha-equivalence probes (probe files, not landing files).

Usage:  python3 -I mkalpha.py EDITED_MOVE.lean UNEDITED_MOVE.lean POS_OUT.lean NEG_OUT.lean
Both moves come from movemanifest.py (with and without --h2m-edits). For every declaration that
the H2-M edits touch, POS_OUT declares, after the edited move, a copy `X_orig` built from the
UNEDITED text (theorem proofs replaced by `sorry`; definitions copied in full) and checks
  example : type_of% @X_orig = type_of% @X := by with_reducible rfl
i.e. the edited and unedited statements are equal without unfolding anything but reducible
constants (alpha-equivalence, proof irrelevance for embedded proofs); for definitions also
  example : @X_orig = @X := by rfl
(the bodies agree). NEG_OUT is the negative control: `seekTM_stateStep` with the unused binder
dropped (`if rank.val = 0 then`, a non-dependent `ite` instead of `dite`) must FAIL the same
reducible check.
"""
import re
import sys

EDITED = ['seekRouterTM', 'seekTM_stateStep', 'tableBitsK_cut',
          'MultiTapeTM.inputPos_runFrom_of_inputTape_eq_zero', 'Ph', 'roundData_window',
          'roundTM_prepare', 'roundTM_payload', 'roundTM_restore', 'roundTM_canonical',
          'rep_entry']
DECL = re.compile(r'^(?:@\[[^\]]*\]\s*)?((?:(?:private|protected|noncomputable)\s+)*)'
                  r'(def|theorem|lemma|abbrev|inductive|structure|instance|class)\s+(\S+)')
STOP = re.compile(r'^(?:/-|end\b|namespace\b|section\b|#|set_option\b|open\b|variable\b)')


def in_comment_at_start(lines):
    out, depth = [], 0
    for l in lines:
        out.append(depth > 0)
        i = 0
        while i < len(l):
            if depth == 0 and l.startswith('--', i):
                break
            if l.startswith('/-', i):
                depth += 1
                i += 2
            elif depth > 0 and l.startswith('-/', i):
                depth -= 1
                i += 2
            else:
                i += 1
    return out


def code_blocks(lines):
    """name -> (kind, code) for every declaration; code excludes the docstring."""
    inc = in_comment_at_start(lines)
    out = {}
    starts = [i for i, l in enumerate(lines) if not inc[i] and DECL.match(l)]
    for i in starts:
        m = DECL.match(lines[i])
        b = i + 1
        while b < len(lines) and (inc[b] or (not DECL.match(lines[b]) and
                                             not STOP.match(lines[b]))):
            b += 1
        while b > i + 1 and not lines[b - 1].strip():
            b -= 1
        name = m.group(3).replace('_root_.Turing.', '')
        out[name] = (m.group(2), '\n'.join(lines[i:b]))
    return out


def split_statement(code):
    """Split at the first top-level `:=` that does not close a top-level `let`."""
    depth, lets = 0, 0
    for i, ch in enumerate(code):
        if ch in '([{⟨':
            depth += 1
        elif ch in ')]}⟩':
            depth -= 1
        elif depth == 0 and code.startswith('let', i) and \
                (i == 0 or not (code[i - 1].isalnum() or code[i - 1] in '_.')) and \
                (i + 3 < len(code) and not (code[i + 3].isalnum() or code[i + 3] == '_')):
            lets += 1
        elif depth == 0 and code.startswith(':=', i):
            if lets:
                lets -= 1
            else:
                return code[:i], code[i:]
    raise ValueError('no top-level := found')


def orig_copy(name, kind, code):
    short = name.split('.')[-1]
    head = code.split('\n')[0]
    new_head = re.sub(r'(\s)((?:_root_\.Turing\.)?(?:MultiTapeTM\.)?)' + re.escape(short) +
                      r'(?=[\s:({\[]|$)', r'\1\2' + short + '_orig', head, count=1)
    assert new_head != head, name
    code = new_head + code[len(head):]
    if kind in ('theorem', 'lemma'):
        stmt, _ = split_statement(code)
        code = stmt.rstrip() + ' := by\n  sorry'
    return code


def main():
    edited_move, unedited_move, pos_out, neg_out = sys.argv[1:5]
    E = open(edited_move, encoding='utf-8').read().rstrip('\n').split('\n')
    U = open(unedited_move, encoding='utf-8').read().rstrip('\n').split('\n')
    assert E[-3:] == ['end StepBlock', '', 'end Turing'], E[-3:]
    ub = code_blocks(U)
    checks = ['', '/-! ## Probe, not a landing file: the H2-M text edits are alpha-equivalent',
              '',
              'Each `X_orig` is the unedited text of an edited declaration `X`. -/', '']
    for n in EDITED:
        kind, code = ub[n]
        checks.append(orig_copy(n, kind, code))
        checks.append('')
        ref = n if not n.startswith('MultiTapeTM.') else n
        checks.append(f'example : type_of% @{ref}_orig = type_of% @{ref} := by with_reducible rfl')
        if kind in ('def', 'abbrev'):
            checks.append(f'example : @{ref}_orig = @{ref} := by rfl')
        checks.append('')
    body = E[:-3]
    with open(pos_out, 'w', encoding='utf-8') as f:
        f.write('\n'.join(body + checks + ['end StepBlock', '', 'end Turing', '']))
    kind, code = ub['seekTM_stateStep']
    old = 'if hr : rank.val = 0 then .done'
    assert code.count(old) == 1
    dropped = orig_copy('seekTM_stateStep', kind, code.replace(old, 'if rank.val = 0 then .done'))
    dropped = dropped.replace('seekTM_stateStep_orig', 'seekTM_stateStep_dropped', 1)
    neg = ['', '/-! ## Negative control: dropping the binder is not alpha-equivalent -/', '',
           dropped, '',
           'example : type_of% @seekTM_stateStep_dropped = type_of% @seekTM_stateStep := by',
           '  with_reducible rfl', '']
    with open(neg_out, 'w', encoding='utf-8') as f:
        f.write('\n'.join(body + neg + ['end StepBlock', '', 'end Turing', '']))
    print(f'{pos_out}: {len(EDITED)} declarations checked')
    print(f'{neg_out}: the dropped-binder control')


if __name__ == '__main__':
    main()
