def comment_mask(text):
    """Per character: True inside a comment (line, block, doc, module doc)."""
    mask = [False] * len(text)
    i, n, depth = 0, len(text), 0
    in_str = False
    while i < n:
        c = text[i]
        if depth == 0 and not in_str:
            if text.startswith('--', i):
                j = text.find('\n', i)
                j = n if j < 0 else j
                for p in range(i, j):
                    mask[p] = True
                i = j
                continue
            if text.startswith('/-', i):
                depth = 1
                mask[i] = mask[i + 1] = True
                i += 2
                continue
            if c == '"':
                in_str = True
            i += 1
            continue
        if in_str:
            if c == '\\':
                i += 2
                continue
            if c == '"':
                in_str = False
            i += 1
            continue
        # inside a block comment
        mask[i] = True
        if text.startswith('/-', i):
            depth += 1
            mask[i + 1] = True
            i += 2
            continue
        if text.startswith('-/', i):
            depth -= 1
            mask[i + 1] = True
            i += 2
            continue
        i += 1
    return mask


def strip_comments(text):
    m = comment_mask(text)
    return ''.join(' ' if m[i] and text[i] != '\n' else text[i] for i in range(len(text)))


class Item:
    def __init__(self, kind, start, end, name=None, dkind=None, mods=''):
        self.kind, self.start, self.end = kind, start, end   # 0-based line indices, end excl.
        self.name, self.dkind, self.mods = name, dkind, mods
        self.doc = None                                       # (start, end) of its docstring


def parse(text):
    """Top-level items: 'decl', 'doc' (a `/--` docstring), 'moddoc', 'comment', 'cmd'.
    A top-level item starts at column 0, outside any comment that began on an earlier line."""
    lines = text.split('\n')
    mask = comment_mask(text)
    offs, o = [], 0
    for ln in lines:
        offs.append(o)
        o += len(ln) + 1
    line_of = []
    for i, ln in enumerate(lines):
        line_of.extend([i] * (len(ln) + 1))
    stripped = strip_comments(text).split('\n')
    starts = []
    pending_attr = False
    for i, ln in enumerate(lines):
        if not ln or ln[0] in ' \t':
            continue
        off = offs[i]
        if off > 0 and mask[off - 1]:
            continue                      # inside a block comment opened on an earlier line
        if ln.startswith('/-') or ln.startswith('--'):
            kind = ('doc' if ln.startswith('/--') else 'moddoc' if ln.startswith('/-!')
                    else 'comment')
            j = off
            while j < len(text) and mask[j]:
                j += 1
            starts.append((i, kind, line_of[j - 1] + 1))
            pending_attr = False
            continue
        s = stripped[i]
        if TOP.match(s):
            if pending_attr:
                pending_attr = False
                continue
            starts.append((i, 'code', None))
            if re.match(r'@\[[^\]]*\]\s*$', s):
                pending_attr = True
    items = []
    for j, (i, kind, cend) in enumerate(starts):
        end = starts[j + 1][0] if j + 1 < len(starts) else len(lines)
        if kind != 'code':
            items.append(Item(kind, i, min(cend, end)))
            continue
        while end > i + 1 and not lines[end - 1].strip():
            end -= 1
        head = stripped[i]
        if re.match(r'@\[[^\]]*\]\s*$', head) and i + 1 < len(lines):
            head = head + ' ' + stripped[i + 1].lstrip()
        m = DECL.match(head)
        if m:
            name = m.group(3)
            if name.startswith('_root_.'):
                name = name[len('_root_.'):]
                if name.startswith('Turing.'):
                    name = name[len('Turing.'):]
            it = Item('decl', i, end, name, m.group(2), m.group(1))
        else:
            it = Item('cmd', i, end)
        items.append(it)
    # attach docstrings to the following declaration
    for j, it in enumerate(items):
        if it.kind == 'doc' and j + 1 < len(items) and items[j + 1].kind == 'decl':
            gap = lines[it.end:items[j + 1].start]
            if all(not g.strip() for g in gap):
                items[j + 1].doc = (it.start, it.end)
                it.kind = 'attached-doc'
    return lines, items


def decl_text(lines, it, with_doc=True):
    a = it.doc[0] if (with_doc and it.doc) else it.start
    return '\n'.join(lines[a:it.end])


def norm_code(t):
    t = strip_comments(t)
    t = ' '.join(t.split())
    t = re.sub(r'^(?:(?:private|protected)\s+)', '', t)
    t = re.sub(r'^(noncomputable\s+)?lemma\b', lambda m: (m.group(1) or '') + 'theorem', t)
    t = t.replace('_root_.Turing.MultiTapeTM.', 'MultiTapeTM.')
    return t


def rename(t):
    for a in sorted(REN, key=len, reverse=True):
        t = re.sub(r'(?<![A-Za-z0-9_.])' + re.escape(a) + r'(?![A-Za-z0-9_])', REN[a], t)
    return t


def idents(code):
    return set(IDENT.findall(code))


def signature(code, dkind):
    """The statement part of a declaration's code: up to the first top-level `:=` that does not
    close a top-level `let` of the statement (or `where` for instances and definitions); the
    whole text for inductive types."""
    if dkind in ('inductive', 'structure', 'class'):
        return code
    depth, lets = 0, 0
    for i, ch in enumerate(code):
        if ch in '([{⟨':
            depth += 1
        elif ch in ')]}⟩':
            depth -= 1
        elif depth == 0 and code.startswith('let', i) and \
                (i == 0 or not (code[i - 1].isalnum() or code[i - 1] in '_.')) and \
                (i + 3 == len(code) or not (code[i + 3].isalnum() or code[i + 3] == '_')):
            lets += 1
        elif depth == 0 and code.startswith(':=', i):
            if lets:
                lets -= 1
            else:
                return code[:i]
        elif depth == 0 and code.startswith(' where', i) and dkind in ('def', 'instance',
                                                                         'abbrev'):
            return code[:i]
    return code


def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()

        v = by_name.get(o, [])
        if len(v) != 1:
            fail(f'moved declaration {o}: found {len(v)} times')
            continue
        moved.append(v[0])
    print(f'# moved declarations found: {len(moved)} of {len(NEW)}')
    moved_set = {it.name for it in moved}
    staying = [it for it in decls if it.name not in moved_set]
    print(f'# staying declarations: {len(staying)}')
    for n in FAMILY:
        if n not in by_name:
            fail(f'summary-clock family member {n} not found')
    newh1 = sorted(it.name for it in staying if re.match(r'h1[A-Z]', it.name.split('.')[-1]))
    known = [n for n in newh1 if n in KNOWN_H1_STAYING]
    unknown = [n for n in newh1 if n not in KNOWN_H1_STAYING]
    print(f'# h1-prefixed declarations outside the inventory, known to stay (H1 assembly, '
          f'fills-8): {len(known)} of {len(KNOWN_H1_STAYING)}: {", ".join(known)}')
    for n in KNOWN_H1_STAYING:
        if n not in by_name:
            print(f'# NOTE: known staying H1 declaration {n} is absent from this base')
    if unknown:
        fail(f'h1-prefixed declarations outside the 121-name inventory and the known staying '
             f'list: {unknown}; decide move (extend the inventory, disclose at the H2-M gate) '
             f'or stay (extend KNOWN_H1_STAYING)')

    # -- groups (maximal runs of moved declarations in file order)
    order = sorted(moved, key=lambda it: it.start)
    groups, cur = [], [order[0]]
    seq = [it for it in decls]
    pos = {id(it): k for k, it in enumerate(seq)}
    for it in order[1:]:
        if pos[id(it)] == pos[id(cur[-1])] + 1:
            cur.append(it)
        else:
            groups.append(cur)
            cur = [it]
    groups.append(cur)
    print(f'\n## Moved regions (located by name): {len(groups)} maximal runs of declarations')
    for g in groups:
        a = (g[0].doc[0] if g[0].doc else g[0].start) + 1
        print(f'  lines {a}-{g[-1].end}: {len(g)} declarations, {g[0].name} .. {g[-1].name}')
    between = []
    for g1, g2 in zip(groups, groups[1:]):
        mid = [it for it in seq[pos[id(g1[-1])] + 1:pos[id(g2[0])]]]
        between.append(mid)
        print(f'  between: {", ".join(it.name for it in mid)}')

    # -- layering: moved code must not use a staying CodeSimulator declaration
    stay_names = {it.name for it in staying}
    stay_short = collections.defaultdict(set)
    for n in stay_names:
        stay_short[n.split('.')[-1]].add(n)
    print('\n## Layering: references from moved code to staying CodeSimulator declarations')
    bad = 0
    for it in moved:
        ids = idents(strip_comments(decl_text(lines, it, False)))
        hit = sorted(x for x in ids if x in stay_names or
                     (x.split('.')[-1] in stay_short and
                      any(x.endswith(s) for s in stay_short[x.split('.')[-1]])))
        if hit:
            bad += 1
            fail(f'moved {it.name} references staying {hit}')
    if not bad:
        print('  none (StepBlock can sit below CodeSimulator)')

    # -- staying references to moved names (code) and mentions (comments)
    print('\n## Staying declarations whose code references moved declarations')
    orig_names = set(NEW)
    req_edits = []
    for it in staying:
        ids = idents(strip_comments(decl_text(lines, it, False)))
        hit = sorted(x for x in ids if x in orig_names or x.split('.')[-1] in orig_names)
        if not hit:
            continue
        rows = []
        for x in hit:
            o = x.split('.')[-1]
            nn = NEW[o]
            vis = ('public' if nn in TIER_A | TIER_B else 'private (tier C)' if nn in TIER_C
                   else 'private (tier D)')
            if o in LAYOUT:
                how = 'resolves to Turing.' + nn + ' (moved, same name and namespace)'
            elif o in SUMMARY:
                how = 'resolves to Turing.StepBlock.' + nn + ' under `open StepBlock`'
            else:
                how = f'TEXT EDIT: rename to {nn} (StepBlock.{nn} under `open StepBlock`)'
                req_edits.append((it.name, o, nn))
            rows.append(f'{x} -> {nn} [{vis}]: {how}')
            if nn not in TIER_A | TIER_B and vis_mode == 'proposed':
                fail(f'staying {it.name} uses {x}, proposed private ({vis}): '
                     f'keep {nn} public or rewrite the reference')
        print(f'  {it.name} (lines {it.start + 1}-{it.end}):')
        for r in rows:
            print(f'    {r}')
    summary_users = sorted({it.name for it in staying
                            if idents(strip_comments(decl_text(lines, it, False)))
                            & set(SUMMARY)})
    print(f'  staying users of the moved summary names: {summary_users}')
    if not set(summary_users) <= set(CLIENTS):
        fail(f'staying declarations outside the `open StepBlock` section (summary-clock family '
             f'and H1 clock) use moved summary names: {sorted(set(summary_users) - set(CLIENTS))}')
    outside = sorted({s0 for s0, _, _ in req_edits} - set(CLIENTS))
    if outside:
        fail(f'renamed references outside the `open StepBlock` section: {outside}')
    if req_edits:
        print('  required text edits to staying code (renamed references):')
        for s, o, nn in req_edits:
            print(f'    {s}: {o} -> {nn}')

    print('\n## Status text: comments and docstrings outside the moved regions naming moved '
          'declarations')
    moved_lines = set()
    for it in moved:
        a = it.doc[0] if it.doc else it.start
        moved_lines.update(range(a, it.end))
    mask = comment_mask(text)
    offs, o = [], 0
    for ln in lines:
        offs.append(o)
        o += len(ln) + 1
    pat = re.compile(r'(?<![A-Za-z0-9_.])(' + '|'.join(sorted(map(re.escape, REN), key=len,
                                                              reverse=True)) + r')(?![A-Za-z0-9_])')
    status = collections.OrderedDict()
    for i, ln in enumerate(lines):
        if i in moved_lines:
            continue
        for m in pat.finditer(ln):
            if mask[offs[i] + m.start()]:
                status.setdefault(i + 1, []).append(m.group(1))
    owner = {}
    for it in items:
        lo = it.doc[0] if getattr(it, 'doc', None) else it.start
        for k in range(lo, it.end):
            owner[k + 1] = it.name or it.kind
    for ln, names in status.items():
        print(f'  line {ln} [{owner.get(ln, "?")}]: {", ".join(names)}')
    if not status:
        print('  none')

    # -- downstream: other library files naming a moved declaration (only the public layout
    #    pair can be named outside the file; the other moved names are private there)
    print('\n## Library files other than CodeSimulator naming a moved declaration '
          '(original names; comments excluded)')
    pat2 = re.compile(r'(?<![A-Za-z0-9_.])(' + '|'.join(sorted(map(re.escape, NEW), key=len,
                                                               reverse=True)) +
                      r')(?![A-Za-z0-9_])')
    nds = 0
    for d0, dirs, fs in os.walk(os.path.join(base, 'TCSlib')):
        dirs[:] = [x for x in dirs if x not in ('.lake', '.git')]
        for f in sorted(fs):
            pth = os.path.join(d0, f)
            if not f.endswith('.lean') or os.path.samefile(pth, path):
                continue
            code = strip_comments(open(pth, encoding='utf-8').read())
            hits = sorted(set(pat2.findall(code)))
            if hits:
                nds += 1
                print(f'  {os.path.relpath(pth, base)}: {", ".join(hits)}')
                bad_hits = [h for h in hits if NEW[h] not in TIER_A | TIER_B]
                if bad_hits:
                    fail(f'{os.path.relpath(pth, base)} names moved declarations that are '
                         f'not public: {bad_hits}')
    if not nds:
        print('  none')

    # -- imports
    print('\n## Imports of CodeSimulator (header)')
    imps = [ln for ln in lines if ln.startswith('import ')]
    for ln in imps:
        print(f'  {ln}')
    print(f'  planned addition: {SB_IMPORT_LINE}' +
          ('  (ALREADY PRESENT)' if SB_IMPORT_LINE in imps else ''))

    # -- the surface check: no public statement names a proposed-private declaration
    print('\n## Surface: moved names in public statements (signatures; bodies for UNFOLDED)')
    newname_of = {it.name: NEW[it.name] for it in moved}
    sig_users = collections.defaultdict(set)
    priv_new = set(newnames) - TIER_A - TIER_B
    for it in moved:
        nn = newname_of[it.name]
        if nn not in TIER_A | TIER_B:
            continue
        code = rename(strip_comments(decl_text(lines, it, False)))
        part = code if nn in UNFOLDED else signature(code, it.dkind)
        for x in idents(part):
            base_x = x
            for cand in newnames:
                if base_x == cand or base_x.endswith('.' + cand) or base_x == 'StepBlock.' + cand:
                    if cand != nn:
                        sig_users[cand].add(nn)
    for cand in sorted(set(newnames)):
        if cand in sig_users:
            tier = ('A' if cand in TIER_A else 'B' if cand in TIER_B else
                    'C' if cand in TIER_C else 'D')
            print(f'  {cand} [tier {tier}] <- {", ".join(sorted(sig_users[cand]))}')
            if cand in priv_new:
                fail(f'proposed-private {cand} occurs in public statement(s) '
                     f'{sorted(sig_users[cand])}')
    unused_c = sorted(c for c in TIER_C if c not in sig_users)
    print(f'  tier C names in no public statement: {len(unused_c)} of {len(TIER_C)}')
    if 'Ph' in sig_users:
        pass
    if '--draft' in opts:
        dl, dits = parse(open(opts['--draft'], encoding='utf-8').read())
        addl = [x for x in opts.get('--additive', '').split(',') if x]
        for it in dits:
            if it.kind == 'decl' and it.name in addl:
                code = strip_comments(decl_text(dl, it, False))
                hits = sorted(c for c in priv_new for x in idents(signature(code, it.dkind))
                              if x == c or x.endswith('.' + c))
                print(f'  draft additive {it.name}: proposed-private names in statement: '
                      f'{hits if hits else "none"}')
                if hits:
                    fail(f'draft additive {it.name} names proposed-private {hits}')

    counts = collections.Counter()
    for it in moved:
        nn = newname_of[it.name]
        pub_r1 = not it.mods.strip().startswith('private') or nn in TIER_A | TIER_B | TIER_C
        pub_r2 = nn in TIER_A | TIER_B
        kind = 'instance' if it.dkind == 'instance' else (
            'theorem' if it.dkind in ('theorem', 'lemma') else 'definition')
        if pub_r1:
            counts['r1', kind] += 1
        if pub_r2:
            counts['r2', kind] += 1
    print('\n## Public surface of the moved declarations (the additive ones excluded)')
    for r in ('r1', 'r2'):
        tot = sum(v for (rr, _), v in counts.items() if rr == r)
        print(f'  {r}: {tot} = {counts[r, "definition"]} definitions/types + '
              f'{counts[r, "instance"]} instances + {counts[r, "theorem"]} theorems')

    # -- emit the move file
    edited = []

    def moved_text(it):
        t = rename(decl_text(lines, it, True))
        nn = newname_of[it.name]
        if nn.startswith('MultiTapeTM.'):
            t = re.sub(r'^((?:private\s+)?(?:lemma|theorem)\s+)MultiTapeTM\.',
                       r'\1_root_.Turing.MultiTapeTM.', t, flags=re.M)
        if vis_mode == 'proposed':
            head = decl_text(lines, it, False)
            pfx_line = rename(head.split('\n')[0])
            want_private = nn not in TIER_A | TIER_B
            new_line = pfx_line
            if want_private and not re.match(r'(?:@\[[^\]]*\]\s*)*private\b', pfx_line):
                new_line = re.sub(r'^((?:@\[[^\]]*\]\s*)*)', r'\1private ', pfx_line, count=1)
            if not want_private:
                new_line = re.sub(r'^((?:@\[[^\]]*\]\s*)*)private\s+', r'\1', pfx_line, count=1)
            if nn.startswith('MultiTapeTM.'):
                new_line = re.sub(r'((?:lemma|theorem)\s+)MultiTapeTM\.',
                                  r'\1_root_.Turing.MultiTapeTM.', new_line)
                pfx_line = re.sub(r'((?:lemma|theorem)\s+)MultiTapeTM\.',
                                  r'\1_root_.Turing.MultiTapeTM.', pfx_line)
            t = t.replace(pfx_line, new_line, 1)
        if edits_on and nn in H2M_EDITS:
            t0 = t
            for old, new in H2M_EDITS[nn]:
                if t.count(old) != 1:
                    fail(f'H2-M edit for {nn}: pattern found {t.count(old)} times: {old[:60]!r}')
                    continue
                t = t.replace(old, new)
            if norm_code(alpha_norm(nn, t)) != norm_code(t0):
                fail(f'H2-M edit for {nn} is not token-identical (up to the ALPHA binder names)')
            else:
                edited.append(nn)
        return t

    def interstitial(g):
        """Comments and module docs strictly between consecutive moved declarations."""
        out = []
        for a, b in zip(g, g[1:]):
            lo = a.end
            hi = b.doc[0] if b.doc else b.start
            for it in items:
                if it.kind in ('moddoc', 'comment') and lo <= it.start < hi:
                    out.append((a.name, '\n'.join(lines[it.start:it.end])))
        return out

    hdr = ('\n'.join('import ' + m for m in SB_IMPORTS) + '\n\n'
           'set_option maxHeartbeats 0\nset_option relaxedAutoImplicit false\n'
           'set_option autoImplicit false\n\nnamespace Turing\n\n')
    lay = [it for it in order if it.name in LAYOUT]
    rest = [it for it in order if it.name not in LAYOUT]
    body = [moved_text(it) for it in lay]
    inter = {}
    for g in groups:
        for after, t in interstitial(g):
            inter.setdefault(after, []).append(rename(t))
    sb = []
    for it in rest:
        sb.append(moved_text(it))
        for t in inter.get(it.name, []):
            sb.append(t)
    move_txt = (hdr + '\n\n'.join(body) + '\n\nnamespace StepBlock\n\n' + '\n\n'.join(sb) +
                '\n\nend StepBlock\n\nend Turing\n')
    left = sorted(set(re.findall(r'\bh1[A-Z][A-Za-z0-9_]*', strip_comments(move_txt))))
    if left:
        fail(f'unrenamed h1 identifiers in moved code: {left}')
    wide = [(k + 1, len(ln)) for k, ln in enumerate(move_txt.split('\n')) if len(ln) > 100]
    print(f'\n## Width of the generated move: {len(wide)} lines over 100 Unicode characters'
          + (f' (H2-M edits applied to {len(edited)} declarations: {", ".join(edited)})'
             if edits_on else ' (no H2-M edits applied)'))
    for k, w in wide:
        print(f'  move line {k}: {w} characters')
    if edits_on:
        missing_e = sorted(set(H2M_EDITS) - set(edited))
        if missing_e:
            fail(f'H2-M edits not applied: {missing_e}')
        if wide:
            fail(f'{len(wide)} lines over 100 characters remain after the H2-M edits')
        else:
            print('  every line within 100 characters; every edit token-identical '
                  '(seekTM_stateStep up to its binder name)')
    if '--emit-move' in opts:
        open(opts['--emit-move'], 'w', encoding='utf-8').write(move_txt)
        print(f'\n# move file written: {opts["--emit-move"]} ({move_txt.count(chr(10))} lines, '
              f'visibility {vis_mode})')

    # -- emit the post-move rehearsal
    if '--emit-postmove' in opts:
        drop = set()
        for it in moved:
            a = it.doc[0] if it.doc else it.start
            drop.update(range(a, it.end))
        for g in groups:
            for a, b in zip(g, g[1:]):
                lo, hi = a.end, (b.doc[0] if b.doc else b.start)
                for it2 in items:
                    if it2.kind in ('moddoc', 'comment') and lo <= it2.start < hi:
                        drop.update(range(it2.start, it2.end))
        cl = sorted((by_name[n][0] for n in CLIENTS if n in by_name), key=lambda it: it.start)
        f0 = cl[0].doc[0] if cl[0].doc else cl[0].start
        f1 = cl[-1].end
        inside = [it.name for it in staying if f0 <= it.start < f1 and it.name not in CLIENTS]
        if inside:
            fail(f'post-move: staying declarations inside the `open StepBlock` section that are '
                 f'not clients: {inside}')
        ren_lines = set()
        for it in cl:
            if it.name in KNOWN_H1_STAYING:
                a = it.doc[0] if it.doc else it.start
                ren_lines.update(range(a, it.end))
        out = []
        cs_imports = [ln for ln in lines if ln.startswith('import ')]
        for k, ln in enumerate(lines):
            if k in drop or ln.startswith('import '):
                continue
            if k == f0:
                out.append('section StepBlockClients\n\nopen StepBlock\n')
            out.append(rename(ln) if k in ren_lines else ln)
            if k == f1 - 1:
                out.append('\nend StepBlockClients')
        # collapse runs of blank lines left by the deletions
        txt = re.sub(r'\n{3,}', '\n\n', '\n'.join(out))
        imports = list(dict.fromkeys(cs_imports + ['import ' + m for m in SB_IMPORTS]))
        mv = move_txt.split('\n')
        mv = [ln for ln in mv if not ln.startswith('import ')]
        post = ('\n'.join(imports) + '\n\n-- ===== Build/StepBlock.lean (generated) =====\n' +
                '\n'.join(mv).lstrip('\n') +
                '\n-- ===== CodeSimulator.lean after H2-M commit 1 (generated) =====\n' + txt)
        open(opts['--emit-postmove'], 'w', encoding='utf-8').write(post)
        print(f'# post-move rehearsal written: {opts["--emit-postmove"]} '
              f'({post.count(chr(10))} lines; base lines {f0 + 1}-{f1} (summary-clock family '
              f'and H1 clock, the step block between them deleted) wrapped in '
              f'`section StepBlockClients` / `open StepBlock`; references renamed in '
              f'{len([c for c in cl if c.name in KNOWN_H1_STAYING])} H1 declarations)')

    # -- compare with a reference move file
    if '--compare-to' in opts:
        ref = opts['--compare-to']
        rl, rits = parse(open(ref, encoding='utf-8').read())
        rdecl = collections.defaultdict(list)
        for it in rits:
            if it.kind == 'decl':
                rdecl[it.name].append(it)
        addl = set(x for x in opts.get('--additive', '').split(',') if x)
        print(f'\n## Comparison with {ref} (sha256 {sha(ref)})')
        nmatch = 0
        body_ok, raw_same, raw_diff = [], 0, []
        allow = set(x for x in opts.get('--allow-body-diff', '').split(',') if x)
        for it in moved:
            nn = newname_of[it.name]
            v = rdecl.get(nn, [])
            if len(v) != 1:
                fail(f'compare: {nn} found {len(v)} times in reference')
                continue
            ra = rename(decl_text(lines, it, False))
            rb = decl_text(rl, v[0], False)
            rawn = lambda t: re.sub(r'^private\s+', '', t).replace('_root_.Turing.MultiTapeTM.',
                                                                     'MultiTapeTM.')
            if rawn(ra) == rawn(rb):
                raw_same += 1
            else:
                raw_diff.append(it.name)
            a = norm_code(alpha_norm(nn, ra))
            b = norm_code(alpha_norm(nn, rb))
            if a == b:
                nmatch += 1
                continue
            sa, sb2 = signature(a, it.dkind), signature(b, it.dkind)
            if sa == sb2 and it.dkind in ('theorem', 'lemma') and it.name in allow:
                body_ok.append(it.name)
                continue
            k = next((j for j in range(min(len(a), len(b))) if a[j] != b[j]),
                     min(len(a), len(b)))
            what = 'statement' if sa != sb2 else 'proof body (not in --allow-body-diff)'
            fail(f'compare: {it.name} -> {nn} differs in its {what} at char {k}:\n'
                 f'    base: ...{a[max(0, k - 60):k + 60]}\n'
                 f'    ref : ...{b[max(0, k - 60):k + 60]}')
        print(f'  token-identical (comments, whitespace, visibility ignored): {nmatch} of '
              f'{len(moved)}; byte-identical code (visibility aside): {raw_same}')
        print(f'  code not byte-identical: {len(raw_diff)}: {", ".join(raw_diff)}')
        if allow:
            if set(raw_diff) != allow:
                fail(f'compare: the declarations whose code changed are not exactly the '
                     f'--allow-body-diff list: extra {sorted(set(raw_diff) - allow)}, '
                     f'unchanged {sorted(allow - set(raw_diff))}')
            else:
                print('  the changed declarations are exactly the --allow-body-diff list')
            print(f'  proof body differs, statement token-identical (allowed): {len(body_ok)}: '
                  f'{", ".join(body_ok)}')
            tok_same_allowed = sorted(allow - set(body_ok))
            print(f'  allowed names whose code is nevertheless token-identical (whitespace or '
                  f'comment changes only): {len(tok_same_allowed)}: {", ".join(tok_same_allowed)}')
            bad_allow = sorted(allow - set(NEW))
            if bad_allow:
                fail(f'--allow-body-diff names outside the inventory: {bad_allow}')
        extra = sorted(n for n in rdecl if n not in set(newnames))
        unexpected = [n for n in extra if n not in addl]
        dupr = sorted(n for n, v in rdecl.items() if len(v) > 1)
        print(f'  moved declarations matching: {nmatch} of {len(moved)}')
        print(f'  reference declarations beyond the 121: {extra if extra else "none"}')
        if unexpected:
            fail(f'compare: unexpected declarations in reference: {unexpected}')
        if dupr:
            fail(f'compare: duplicate names in reference: {dupr}')

    print('\nRESULT:', 'PASS' if not fails else f'FAIL ({len(fails)} failing checks)')
    sys.exit(0 if not fails else 1)


if __name__ == '__main__':
    main()
