-- `runWith_cons`, `runWith_append`: `N` implicit, `O` explicit.

/-! ## Part 1: home `OracleNondeterministic.lean` -/

/-- A1 (route 1; private in the fill): the one `stepWith`-against-`step` comparison,
by `unfold` + `cases`. `rfl` alone is not expected to work (G1 probe: distinct
matchers). -/
theorem probe_toOracleNDTM_stepWith (M : OracleTM k Symbol State) (O : Language Symbol)
    (b : Bool) (cfg : Cfg (k + 1) Symbol State input) :
    M.toOracleNDTM.stepWith O b cfg = M.step O cfg := by
  unfold OracleNDTM.stepWith OracleTM.step
  cases hs : cfg.state <;> rfl

/-- A2 (target 1's prescribed proof shape). -/
theorem probe_toOracleNDTM_runWith (M : OracleTM k Symbol State) (O : Language Symbol)
    (w : List Bool) (cfg : Cfg (k + 1) Symbol State input) :
    M.toOracleNDTM.runWith O w cfg = M.runFrom O cfg w.length := by
  induction w generalizing cfg with
  | nil => rfl
  | cons b w ih =>
    show M.toOracleNDTM.runWith O w (M.toOracleNDTM.stepWith O b cfg) =
      M.runFrom O (M.step O cfg) w.length
    rw [probe_toOracleNDTM_stepWith]
    exact ih (M.step O cfg)

/-- D2 export `OracleNDTM.fixChoice`: `N` with its choice bit fixed to `b`. -/
def probeFixChoice (N : OracleNDTM k Symbol State) (b : Bool) : OracleTM k Symbol State :=
  ⟨N.q₀, N.qQuery, N.qYes, N.qNo, N.tr b⟩

/-- B1, D2 export (the bridge): an ND step is a step of the per-bit machine. It comes
from A1 at `probeFixChoice N b`; both sides are `OracleNDTM.stepWith` at
definitionally equal fields. (Alternative, by the target at `[b]`:
`probe_toOracleNDTM_runWith (probeFixChoice N b) O [b] cfg`.) -/
theorem probe_stepWith_eq_fixChoice_step (N : OracleNDTM k Symbol State)
    (O : Language Symbol) (b : Bool) (cfg : Cfg (k + 1) Symbol State input) :
    N.stepWith O b cfg = (probeFixChoice N b).step O cfg :=
  probe_toOracleNDTM_stepWith (probeFixChoice N b) O b cfg

/-- B2, D2 export (R2): `stepWith` oracle-independence away from `qQuery`, from the
bridge and the deterministic lemma. -/
theorem probe_stepWith_eq_of_ne_qQuery {N : OracleNDTM k Symbol State}
    (O₁ O₂ : Language Symbol) {b : Bool} {cfg : Cfg (k + 1) Symbol State input}
    (h : cfg.state ≠ some N.qQuery) :
    N.stepWith O₁ b cfg = N.stepWith O₂ b cfg := by
  rw [probe_stepWith_eq_fixChoice_step, probe_stepWith_eq_fixChoice_step]
  exact OracleTM.step_eq_of_ne_qQuery (M := probeFixChoice N b) O₁ O₂ h

-- Route 3(b): per-step facts by citation through the bridge.
example (N : OracleNDTM k Symbol State) (O : Language Symbol) (b : Bool)
    (cfg : Cfg (k + 1) Symbol State input) (i : Fin (k + 1)) :
    |(N.stepWith O b cfg).workTapePos i - cfg.workTapePos i| ≤ 1 := by
  rw [probe_stepWith_eq_fixChoice_step]
  exact (probeFixChoice N b).workTapePos_step_le O cfg i

example (N : OracleNDTM k Symbol State) (O : Language Symbol) (b : Bool)

    (hz : z ≠ cfg.workTapePos i) :
    (N.stepWith O b cfg).workTapes i z = cfg.workTapes i z := by

/-- [S] Private: the core induction over choice-word prefixes. -/
theorem probe_runWith_take_eq_of_agree (N : OracleNDTM k Symbol State)
    (O O' : Language Symbol) (x : List Symbol) (w : List Bool)
    (h : ∀ s < w.length,
      (N.runWith O (w.take s) (N.initCfg x)).state = some N.qQuery →
        (OracleTM.queryString (N.runWith O (w.take s) (N.initCfg x)) ∈ O ↔
          OracleTM.queryString (N.runWith O (w.take s) (N.initCfg x)) ∈ O')) :
    ∀ s ≤ w.length,
      N.runWith O (w.take s) (N.initCfg x) = N.runWith O' (w.take s) (N.initCfg x) := by
  sorry

-- The core's successor case: the prefix split and the singleton step.
example (w : List Bool) (s : ℕ) (hs : s < w.length) :
    w.take (s + 1) = w.take s ++ [w[s]] := by
  rw [List.take_succ, List.getElem?_eq_getElem hs, Option.toList_some]

example (N : OracleNDTM k Symbol State) (O : Language Symbol) (u : List Bool) (b : Bool)
    (cfg : Cfg (k + 1) Symbol State input) :
    N.runWith O (u ++ [b]) cfg = N.stepWith O b (N.runWith O u cfg) :=
  (OracleNDTM.runWith_append O u [b] cfg).trans rfl

-- Endpoints of the core: `take 0` and `take length`.
example (w : List Bool) : w.take 0 = [] := List.take_zero
example (w : List Bool) : w.take w.length = w := List.take_length

/-- [T1] Private in the fill: constant-word prefixes are the deterministic run. -/
theorem probe_toOracleNDTM_runWith_take_replicate (M : OracleTM k Symbol State)
    (O : Language Symbol) (x : List Symbol) {s t : ℕ} (hs : s ≤ t) :
    M.toOracleNDTM.runWith O ((List.replicate t false).take s) (M.toOracleNDTM.initCfg x) =
      M.runFrom O (M.initCfg x) s := by
  have h := OracleTM.toOracleNDTM_runWith M O (List.replicate s false) (M.toOracleNDTM.initCfg x)
  rw [List.length_replicate] at h
  rw [List.take_replicate, Nat.min_eq_left hs]
  exact h

-- Membership introduction (targets 7, 9): the `filterMap` function is a lambda, hence
-- `simp only [if_pos hq]`, not `rw`.
example (N : OracleNDTM k Symbol State) (O : Language Symbol) (x : List Symbol)
    (w : List Bool) (s : ℕ) (hs : s < w.length)
    (hq : (N.runWith O (w.take s) (N.initCfg x)).state = some N.qQuery) :
    OracleTM.queryString (N.runWith O (w.take s) (N.initCfg x)) ∈ N.queriesAlong O x w := by
  unfold OracleNDTM.queriesAlong
  exact List.mem_filterMap.mpr ⟨s, List.mem_range.mpr hs, by simp only [if_pos hq]⟩

example (M : OracleTM k Symbol State) (O : Language Symbol) (x : List Symbol)
    (t s : ℕ) (hs : s < t) (hq : (M.runFrom O (M.initCfg x) s).state = some M.qQuery) :
    OracleTM.queryString (M.runFrom O (M.initCfg x) s) ∈ M.queriesWithin O x t := by
  unfold OracleTM.queriesWithin
  exact List.mem_filterMap.mpr ⟨s, List.mem_range.mpr hs, by simp only [if_pos hq]⟩

-- Target 4's one line.
