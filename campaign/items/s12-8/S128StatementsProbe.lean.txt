variable {k : ℕ} {Γ State : Type*}

/-- Decode a rich symbol along `e : Bool ↪ Γ`: `e b ↦ some b`, foreign symbols `↦ none`. -/
def decodeBool [DecidableEq Γ] (e : Bool ↪ Γ) (γ : Γ) : Option Bool :=
  if γ = e true then some true else if γ = e false then some false else none

@[simp] lemma decodeBool_apply [DecidableEq Γ] (e : Bool ↪ Γ) (b : Bool) :
    decodeBool e (e b) = some b := by
  cases b
  · have : e false ≠ e true := fun h => Bool.false_ne_true (e.injective h)
    simp [decodeBool, this]
  · simp [decodeBool]

/-- Lift an action along `e`: written symbols and the emission are mapped by `e`. -/
def Action.liftSymbols (e : Bool ↪ Γ) (a : Action k Bool State) : Action k Γ State where
  inputTape := a.inputTape
  workTapes := fun j => ((a.workTapes j).1.map (Option.map e), (a.workTapes j).2)
  output := a.output.map e
  state := a.state

/-- Lift a configuration along `e`: input, work cells and output mapped by `e`. -/
def Cfg.liftSymbols (e : Bool ↪ Γ) {x : List Bool} (c : Cfg k Bool State x) :
    Cfg k Γ State (x.map e) where
  state := c.state
  inputPos := Fin.cast (by simp) c.inputPos
  workTapes := fun j p => (c.workTapes j p).map e
  workTapePos := c.workTapePos
  output := c.output.map e

/-- Lift a machine along `e`: read through `decodeBool e`, write through `e`. -/
def MultiTapeTM.liftSymbols [DecidableEq Γ] (M : MultiTapeTM k Bool State) (e : Bool ↪ Γ) :
    MultiTapeTM k Γ State where
  q₀ := M.q₀
  tr := fun q inp work =>
    (M.tr q (inp.bind (decodeBool e)) fun j => (work j).bind (decodeBool e)).liftSymbols e

lemma Cfg.liftSymbols_inputSymbol (e : Bool ↪ Γ) {x : List Bool} (c : Cfg k Bool State x) :
    (c.liftSymbols e).inputSymbol = c.inputSymbol.map e := by
  have hv : ((c.liftSymbols e).inputPos : ℕ) = c.inputPos := rfl
  unfold Cfg.inputSymbol
  have e0 : ((c.liftSymbols e).inputPos = 0) ↔ (c.inputPos = 0) := by
    simp only [Fin.ext_iff, hv, Fin.val_zero]
  have e1 : ((c.liftSymbols e).inputPos = ((x.map e).length + 1 : ℕ)) ↔
      (c.inputPos = (x.length + 1 : ℕ)) := by
    simp only [hv, List.length_map]
  by_cases h0 : c.inputPos = 0
  · rw [dif_pos (e0.mpr h0), dif_pos h0]; rfl
  · rw [dif_neg (fun h => h0 (e0.mp h)), dif_neg h0]
    by_cases h1 : c.inputPos = (x.length + 1 : ℕ)
    · rw [dif_pos (e1.mpr h1), dif_pos h1]; rfl
    · rw [dif_neg (fun h => h1 (e1.mp h)), dif_neg h1]
      simp [hv]

lemma moveInputPos_cast {n m : ℕ} (h : n + 2 = m + 2) (p : Fin (n + 2)) (s : SignType) :
    moveInputPos (Fin.cast h p) s = Fin.cast h (moveInputPos p s) := by
  have hnm : n = m := by omega
  subst hnm
  rfl

/-- One step commutes with the lift (private in the proposed file). -/
lemma MultiTapeTM.liftSymbols_step [DecidableEq Γ] (M : MultiTapeTM k Bool State)
    (e : Bool ↪ Γ) {x : List Bool} (c : Cfg k Bool State x) :
    (M.liftSymbols e).step (c.liftSymbols e) = (M.step c).liftSymbols e := by
  cases hs : c.state with
  | none => simp [MultiTapeTM.step, Cfg.liftSymbols, hs]
  | some q =>
    have hin : (c.liftSymbols e).inputSymbol.bind (decodeBool e) = c.inputSymbol := by
      rw [Cfg.liftSymbols_inputSymbol]; cases c.inputSymbol <;> simp
    have hw : (fun j => ((c.liftSymbols e).workTapeSymbols j).bind (decodeBool e)) =
        c.workTapeSymbols := by
      funext j; simp only [Cfg.workTapeSymbols, Cfg.liftSymbols]
      cases c.workTapes j (c.workTapePos j) <;> simp
    simp only [MultiTapeTM.step,
      show (c.liftSymbols e).state = some q by simp [Cfg.liftSymbols, hs], hs]
    change ((M.tr q ((c.liftSymbols e).inputSymbol.bind (decodeBool e))
      fun j => ((c.liftSymbols e).workTapeSymbols j).bind (decodeBool e)).liftSymbols e).apply
        (c.liftSymbols e) = _
    rw [hin, hw]
    apply Cfg.ext
    · rfl
    · exact moveInputPos_cast (by simp) _ _
    · funext j p
      simp only [Action.apply, Action.liftSymbols, Cfg.liftSymbols]
      cases (M.tr q c.inputSymbol c.workTapeSymbols).workTapes j |>.1 with
      | none => simp
      | some v =>
        simp only [Option.map_some]
        by_cases hp : p = c.workTapePos j
        · subst p; simp
        · simp [Function.update_of_ne hp]
    · rfl
    · simp only [Action.apply, Action.liftSymbols, Cfg.liftSymbols, List.map_append]
      cases (M.tr q c.inputSymbol c.workTapeSymbols).output <;> rfl

/-- **The embedding transport** (§12.8 R8.0): from every `Bool` configuration and at
every time, the lifted machine runs as the original, everything mapped along `e`. -/
theorem MultiTapeTM.liftSymbols_runFrom [DecidableEq Γ] (M : MultiTapeTM k Bool State)
    (e : Bool ↪ Γ) {x : List Bool} (c : Cfg k Bool State x) (t : ℕ) :
    (M.liftSymbols e).runFrom (c.liftSymbols e) t = (M.runFrom c t).liftSymbols e := by
  induction t generalizing c with
  | zero => rfl
  | succ t ih =>
    rw [MultiTapeTM.runFrom_succ_eq_step, MultiTapeTM.runFrom_succ_eq_step,
      MultiTapeTM.liftSymbols_step, ih]

    (c.liftSymbols e).state = c.state := rfl

/-- The lifted machine starts at the lift of the original start. -/

/-- The `FinTM` layer of the lift. -/
def FinTM.liftSymbols {Γ : Type} [DecidableEq Γ] (M : FinTM Bool) (e : Bool ↪ Γ) : FinTM Γ :=
  ⟨M.k, M.State, M.tm.liftSymbols e⟩

/-- A `Bool` machine computing `f` computes `f` via `e` when lifted. Its time is unchanged,
which is what `alphabet_reduction` consumes. -/
theorem FinTM.liftSymbols_computesFunInTimeVia {Γ : Type} [DecidableEq Γ] (M : FinTM Bool)
    (e : Bool ↪ Γ) {f : List Bool → List Bool} {T : ℕ → ℕ} (h : M.ComputesFunInTime f T) :
    (M.liftSymbols e).ComputesFunInTimeVia e f T := by
  intro x
  have hx := (FinTM.computesInTime_iff _ _ _ _).mp (h x)
  refine (FinTM.computesInTime_iff _ _ _ _).mpr ?_
  change ((M.tm.liftSymbols e).runFrom ((M.tm.liftSymbols e).initCfg (x.map e)) _).state = none ∧
    ((M.tm.liftSymbols e).runFrom ((M.tm.liftSymbols e).initCfg (x.map e)) _).output = _
  rw [MultiTapeTM.liftSymbols_initCfg, MultiTapeTM.liftSymbols_runFrom]
  exact ⟨hx.1, congrArg (List.map e) hx.2⟩

/-! ## Part C — rows touching no G1-owned file -/

/-- Phases of the constant writer. -/
inductive WritePhase (n : ℕ) where
  | wr (j : Fin (n + 1)) | bk (j : Fin (n + 1)) | done
deriving DecidableEq

/-- **R8.4** Write the fixed word `w₀` from the head and return the head. -/
def writeConstTM {Symbol : Type*} (k : ℕ) (i : Fin k) (w₀ : List Symbol) :
    MultiTapeTM k Symbol (WritePhase w₀.length) where
  q₀ := .wr 0
  tr := fun q _ _ =>
    match q with
    | .wr j =>
      if h : (j : ℕ) < w₀.length then
        ⟨0, fun t => if t = i then (some (some w₀[(j : ℕ)]), SignType.pos) else (none, 0), none,
          some (.wr ⟨j + 1, by omega⟩)⟩
      else ⟨0, fun _ => (none, 0), none, some (.bk ⟨w₀.length, by omega⟩)⟩
    | .bk j =>
      if (j : ℕ) = 0 then ⟨0, fun _ => (none, 0), none, some .done⟩
      else ⟨0, fun t => if t = i then (none, SignType.neg) else (none, 0), none,
          some (.bk ⟨j - 1, by omega⟩)⟩
    | .done => ⟨0, fun _ => (none, 0), none, some .done⟩

/-- **R8.4, framed contract.** Exact time `2|w₀| + 2`; old contents arbitrary. -/
theorem writeConstTM_run_ofCfg {Symbol : Type*} {k : ℕ} {x : List Symbol} (i : Fin k)
    (w₀ : List Symbol) (d : Cfg k Symbol (WritePhase w₀.length) x)
    (hstate : d.state = some (.wr 0)) :
    (writeConstTM k i w₀).runFrom d (2 * w₀.length + 2) =
        { d with
          state := some .done
          workTapes := fun j q =>
            if j = i ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (w₀.length : ℤ)
              then FinTM.bufferTape w₀ (q - d.workTapePos j)
            else d.workTapes j q } ∧
      (∀ t < 2 * w₀.length + 2, ((writeConstTM k i w₀).runFrom d t).state ≠ some .done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * w₀.length + 2 →
        if j = i then
          ((writeConstTM k i w₀).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j) (d.workTapePos j + (w₀.length : ℤ))
        else ((writeConstTM k i w₀).runFrom d t).workTapePos j = d.workTapePos j) := by
  sorry

/-- Phases of the input copier. -/
inductive InCopyPhase where
  | fwd | bwd | done
deriving DecidableEq, Fintype

/-- **R8.5** Copy the native input onto tape `i` from its head, in lockstep, and rewind
both heads together (the destination need not be delimited on the left). -/
def inputCopyTM {Symbol : Type*} (k : ℕ) (i : Fin k) : MultiTapeTM k Symbol InCopyPhase where
  q₀ := .fwd
  tr := fun q inp _ =>
    match q with
    | .fwd =>
      match inp with
      | some s => ⟨1, fun j => if j = i then (some (some s), SignType.pos) else (none, 0), none,
          some .fwd⟩
      | none => ⟨-1, fun j => if j = i then (none, SignType.neg) else (none, 0), none, some .bwd⟩
    | .bwd =>
      match inp with
      | some _ => ⟨-1, fun j => if j = i then (none, SignType.neg) else (none, 0), none, some .bwd⟩
      | none => ⟨1, fun j => if j = i then (none, SignType.pos) else (none, 0), none, some .done⟩
    | .done => ⟨0, fun _ => (none, 0), none, some .done⟩

/-- **R8.5, framed contract.** From input position `1`: exact time `2|x| + 2`. The interval
`[pos, pos+|x|)` of tape `i` holds `x`, and both heads end where they started. -/
theorem inputCopyTM_run_ofCfg {Symbol : Type*} {k : ℕ} {x : List Symbol} (i : Fin k)
    (d : Cfg k Symbol InCopyPhase x) (hstate : d.state = some .fwd) (hin : d.inputPos = 1) :
    (inputCopyTM k i).runFrom d (2 * x.length + 2) =
        { d with
          state := some .done
          workTapes := fun j q =>
            if j = i ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (x.length : ℤ)
              then FinTM.bufferTape x (q - d.workTapePos j)
            else d.workTapes j q } ∧
      (∀ t < 2 * x.length + 2, ((inputCopyTM k i).runFrom d t).state ≠ some .done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * x.length + 2 →
        if j = i then
          ((inputCopyTM k i).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (x.length : ℤ))
        else ((inputCopyTM k i).runFrom d t).workTapePos j = d.workTapePos j) := by
  sorry

/-- **R8.5** Seek the native input head to the binary position held on a dedicated
counter tape (`Bool`; lifted to rich hosts by R8.0). A copy to scratch, a rewind, and a
`counterLoopTM` round "input right" per unit. The counter is consumed and the scratch
restored. -/
theorem exists_inputSeekTM :
    ∃ (S : Type) (_ : Fintype S) (_ : DecidableEq S) (R : MultiTapeTM 2 Bool S) (entry exit : S)
      (C : ℕ),
      ∀ {x : List Bool} (p : ℕ) (w : List Bool), Nat.ofDigits 2 (w.map Bool.toNat) = p →
        p ≤ x.length + 1 → ∀ d : Cfg 2 Bool S x, d.state = some entry →
        (∀ q : ℤ, -1 ≤ q → q ≤ (w.length : ℤ) →
          d.workTapes 0 (d.workTapePos 0 + q) = FinTM.bufferTape w q) →
        (∀ q, d.workTapes 1 q = none) →
        ∃ t ≤ C * (w.length + d.inputPos.val + p + 1),
          (∀ u < t, (R.runFrom d u).state ≠ some exit) ∧
          (R.runFrom d t).state = some exit ∧ ((R.runFrom d t).inputPos : ℕ) = p ∧
          (R.runFrom d t).workTapes = d.workTapes ∧ (R.runFrom d t).workTapePos = d.workTapePos ∧
          (R.runFrom d t).output = d.output := by
  sorry

/-! ## Part D — Catalog rows after the G1 gate (seek; the scan, a G2 statement) -/

/-- Phases of the seek row. -/
inductive SkipPhase where
  | run | done
deriving DecidableEq, Fintype

/-- **R8.3** Seek: move right over a delimited word and one cell beyond its right delimiter. -/
def skipTM {Symbol : Type*} (k : ℕ) (i : Fin k) : MultiTapeTM k Symbol SkipPhase where
  q₀ := .run
  tr := fun q _ w =>
    match q with
    | .run =>
      match w i with
      | some _ => ⟨0, fun j => if j = i then (none, SignType.pos) else (none, 0), none, some .run⟩
      | none => ⟨0, fun j => if j = i then (none, SignType.pos) else (none, 0), none, some .done⟩
    | .done => ⟨0, fun _ => (none, 0), none, some .done⟩

/-- **R8.3, framed contract.** Exact time `|w| + 1`; nothing written. -/
theorem skipTM_run_ofCfg {Symbol : Type*} {k : ℕ} {x : List Symbol} (i : Fin k) (w : List Symbol)
    (d : Cfg k Symbol SkipPhase x) (hstate : d.state = some .run)
    (hw : ∀ p : ℤ, 0 ≤ p → p ≤ (w.length : ℤ) →
      d.workTapes i (d.workTapePos i + p) = FinTM.bufferTape w p) :
    (skipTM k i).runFrom d (w.length + 1) =
        { d with state := some .done,
                 workTapePos := Function.update d.workTapePos i
                   (d.workTapePos i + w.length + 1) } ∧
      (∀ t < w.length + 1, ((skipTM k i).runFrom d t).state ≠ some .done) := by
  sorry

/-- Left-to-right Mealy transduction (pure). -/
def mealyRun {A Γ : Type*} (δ : A → Γ → A × Γ) : A → List Γ → A × List Γ
  | a, [] => (a, [])
  | a, γ :: w => let r := δ a γ; let s := mealyRun δ r.1 w; (s.1, r.2 :: s.2)

/-- The two-pass result: forward by `δf`, then backward by `δb`. -/
def scanResult {A Γ : Type*} (δf δb : A → Γ → A × Γ) (a₀ : A) (w : List Γ) : A × List Γ :=
  let f := mealyRun δf a₀ w
  let b := mealyRun δb f.1 f.2.reverse
  (b.1, b.2.reverse)

/-- Phases of the scan. -/
inductive ScanPhase (A : Type) where
  | fwd (a : A) | bwd (a : A) | done (a : A)
deriving DecidableEq

/-- **R8.2 / G2** The two-pass transducer scan: rewrite a delimited word cell by cell,
left to right by `δf` and back by `δb`, carrying a finite state; the verdict is the final
state. Fold (`δ` returns the cell), map (stateless), mark shifts (delay transducers) and
odometers (carry transducers) are instances. -/
def scanTM {Γ : Type*} {A : Type} (k : ℕ) (i : Fin k) (δf δb : A → Γ → A × Γ) (a₀ : A) :
    MultiTapeTM k Γ (ScanPhase A) where
  q₀ := .fwd a₀
  tr := fun q _ w =>
    match q with
    | .fwd a =>
      match w i with
      | some γ => ⟨0, fun j => if j = i then (some (some (δf a γ).2), SignType.pos) else (none, 0),
          none, some (.fwd (δf a γ).1)⟩
      | none => ⟨0, fun j => if j = i then (none, SignType.neg) else (none, 0), none, some (.bwd a)⟩
    | .bwd a =>
      match w i with
      | some γ => ⟨0, fun j => if j = i then (some (some (δb a γ).2), SignType.neg) else (none, 0),
          none, some (.bwd (δb a γ).1)⟩
      | none => ⟨0, fun j => if j = i then (none, SignType.pos) else (none, 0), none, some (.done a)⟩
    | .done a => ⟨0, fun _ => (none, 0), none, some (.done a)⟩

/-- **R8.2, framed contract.** Exact time `2|w| + 2`. -/
theorem scanTM_run_ofCfg {Γ : Type*} {A : Type} {k : ℕ} {x : List Γ} (i : Fin k)
    (δf δb : A → Γ → A × Γ) (a₀ : A) (w : List Γ) (d : Cfg k Γ (ScanPhase A) x)
    (hstate : d.state = some (.fwd a₀))
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
      d.workTapes i (d.workTapePos i + p) = FinTM.bufferTape w p) :
    (scanTM k i δf δb a₀).runFrom d (2 * w.length + 2) =
        { d with
          state := some (.done (scanResult δf δb a₀ w).1)
          workTapes := fun j q =>
            if j = i ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (w.length : ℤ)
              then FinTM.bufferTape (scanResult δf δb a₀ w).2 (q - d.workTapePos j)
            else d.workTapes j q } ∧
      (∀ t < 2 * w.length + 2, ∀ a, ((scanTM k i δf δb a₀).runFrom d t).state ≠ some (.done a)) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * w.length + 2 →
        if j = i then
          ((scanTM k i δf δb a₀).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (w.length : ℤ))
        else ((scanTM k i δf δb a₀).runFrom d t).workTapePos j = d.workTapePos j) := by
  sorry

/-- **R8.1, framed compare** (generic; the canonical `compareTM_run` re-derives from it). -/
theorem compareTM_run_ofCfg {Symbol : Type*} [DecidableEq Symbol] {k : ℕ} {x : List Symbol}
    (fst snd : Fin k) (hne : fst ≠ snd) (u v : List Symbol) (d : Cfg k Symbol FlagPhase x)
    (hstate : d.state = some FlagPhase.run)
    (hu : ∀ p : ℤ, -1 ≤ p → p ≤ (u.length : ℤ) →
      d.workTapes fst (d.workTapePos fst + p) = FinTM.bufferTape u p)

/-! ## Part E — the alphabet-generic counter loop -/

/-- The last host tape as a one-tape selection. -/
def lastEmb (k : ℕ) : Fin 1 ↪ Fin (k + 1) := ⟨fun _ => Fin.last k, fun a b _ => Subsingleton.elim a b⟩

/-- **R8.0b** `counterLoopTM` over any alphabet: the body runs on the first `k` tapes
(R1, forwarding), and the counter is a dedicated last tape holding digits `e b`, decremented
by the lifted `decrementTM 1 0` through R1. At `e = Function.Embedding.refl Bool` its table
agrees with `counterLoopTM`'s (Z5), which is how the `Bool` contracts are re-derived. -/
def counterLoopTMVia {k : ℕ} {Γ S : Type} [DecidableEq S] [DecidableEq Γ] (e : Bool ↪ Γ)
    (M : MultiTapeTM k Γ S) (a : S) (exit : Option S) :
    MultiTapeTM (k + 1) Γ (CounterLoopState S) where
  q₀ := .dec .run
  tr := fun q inp work =>
    match q with
    | .body s =>
      ((embedEmitTM Fin.castSuccEmb M).tr s inp work).mapState (counterLoopRedirect a exit)
    | .dec f =>
      ((embedEmitTM (lastEmb k) ((decrementTM 1 0).liftSymbols e)).tr f inp work).mapState
        (counterLoopDecExit a)
    | .done => ⟨0, fun _ => (none, 0), none, some .done⟩
    | .escape => ⟨0, fun _ => (none, 0), none, some .escape⟩

/-- The generic host configuration (`counterLoopCfg` with the alphabet generalized in place). -/
def counterLoopCfgVia {k : ℕ} {Γ S : Type} {x : List Γ} (q : Option (CounterLoopState S))
    (c : Cfg k Γ S x) (ct : ℤ → Option Γ) (cp : ℤ) : Cfg (k + 1) Γ (CounterLoopState S) x :=
  { (embedEmitCfg Fin.castSuccEmb (fun _ => ct) (fun _ => cp) [] c).mapState
      CounterLoopState.body with state := q }

/-- The generic orbit. -/
def counterLoopOrbitVia {k : ℕ} {Γ S : Type} {x : List Γ} (M : MultiTapeTM k Γ S)
    (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) (r : ℕ) : Cfg k Γ S x :=
  (fun c => M.runFrom c (τ c))^[r] c

/-- **R8.0b, every round returns** — `counterLoopTM_run_done` over `Γ`, verbatim up to
`Bool ↦ Γ` and the counter word `w.map e`. -/
theorem counterLoopTMVia_run_done {k : ℕ} {Γ S : Type} [DecidableEq S] [DecidableEq Γ]
    (e : Bool ↪ Γ) {x : List Γ} (M : MultiTapeTM k Γ S) (a : S) (exit : Option S)
    (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) (w : List Bool) (ct : ℤ → Option Γ) (cp : ℤ)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) → ct (cp + p) = FinTM.bufferTape (w.map e) p)
    (d : ℕ) (hd : Nat.ofDigits 2 (w.map Bool.toNat) = d)
    (hround : ∀ r < d,
      (counterLoopOrbitVia M τ c r).state = some a ∧
      0 < τ (counterLoopOrbitVia M τ c r) ∧
      (∀ u, 0 < u → u < τ (counterLoopOrbitVia M τ c r) →
        ∃ s, (M.runFrom (counterLoopOrbitVia M τ c r) u).state = some s ∧
          s ≠ a ∧ exit ≠ some s) ∧
      (counterLoopOrbitVia M τ c (r + 1)).state = some a)
    (T : ℕ)
    (hT : T = ∑ r ∈ Finset.range d, τ (counterLoopOrbitVia M τ c r) +
      counterOverhead w (d + 1)) :
    (counterLoopTMVia e M a exit).runFrom (counterLoopCfgVia (some (.dec .run)) c ct cp) T =
        counterLoopCfgVia (some .done) (counterLoopOrbitVia M τ c d)
          (fun q => if cp ≤ q ∧ q < cp + (w.length : ℤ) then some (e true) else ct q) cp ∧
      (∀ t < T,
        ((counterLoopTMVia e M a exit).runFrom (counterLoopCfgVia (some (.dec .run)) c ct cp) t).state
          ≠ some .done ∧
        ((counterLoopTMVia e M a exit).runFrom (counterLoopCfgVia (some (.dec .run)) c ct cp) t).state
          ≠ some .escape) ∧
      (∀ t ≤ T,
        ((counterLoopTMVia e M a exit).runFrom
            (counterLoopCfgVia (some (.dec .run)) c ct cp) t).workTapePos (Fin.last k) ∈
          Finset.Icc (cp - 1) (cp + (w.length : ℤ))) ∧
      (∀ t ≤ T, ∃ r u,
        ((r < d ∧ u ≤ τ (counterLoopOrbitVia M τ c r)) ∨ (r = d ∧ u = 0)) ∧
        ∀ j : Fin k,
          ((counterLoopTMVia e M a exit).runFrom
              (counterLoopCfgVia (some (.dec .run)) c ct cp) t).workTapePos j.castSucc =
            (M.runFrom (counterLoopOrbitVia M τ c r) u).workTapePos j) := by
  sorry

/-- **R8.0b, early exit** — `counterLoopTM_run_escape` over `Γ`. -/
theorem counterLoopTMVia_run_escape {k : ℕ} {Γ S : Type} [DecidableEq S] [DecidableEq Γ]
    (e : Bool ↪ Γ) {x : List Γ} (M : MultiTapeTM k Γ S) (a ex : S) (hea : ex ≠ a)
    (τ : Cfg k Γ S x → ℕ) (c : Cfg k Γ S x) (w : List Bool) (ct : ℤ → Option Γ) (cp : ℤ)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) → ct (cp + p) = FinTM.bufferTape (w.map e) p)
    (r₀ : ℕ) (hr₀ : r₀ < Nat.ofDigits 2 (w.map Bool.toNat))
    (hround : ∀ r ≤ r₀,
      (counterLoopOrbitVia M τ c r).state = some a ∧
      0 < τ (counterLoopOrbitVia M τ c r) ∧
      (∀ u, 0 < u → u < τ (counterLoopOrbitVia M τ c r) →
        ∃ s, (M.runFrom (counterLoopOrbitVia M τ c r) u).state = some s ∧ s ≠ a ∧ s ≠ ex))
    (hexit : (counterLoopOrbitVia M τ c (r₀ + 1)).state = some ex)
    (T : ℕ)
    (hT : T = ∑ r ∈ Finset.range (r₀ + 1), τ (counterLoopOrbitVia M τ c r) +
      counterOverhead w (r₀ + 1)) :
    (counterLoopTMVia e M a (some ex)).runFrom (counterLoopCfgVia (some (.dec .run)) c ct cp) T =
        counterLoopCfgVia (some .escape) (counterLoopOrbitVia M τ c (r₀ + 1))
          (fun q => if cp ≤ q ∧ q < cp + (w.length : ℤ)
            then FinTM.bufferTape ((counterWord w (r₀ + 1)).map e) (q - cp) else ct q) cp ∧
      (∀ t < T,
        ((counterLoopTMVia e M a (some ex)).runFrom
            (counterLoopCfgVia (some (.dec .run)) c ct cp) t).state ≠ some .done ∧
        ((counterLoopTMVia e M a (some ex)).runFrom
            (counterLoopCfgVia (some (.dec .run)) c ct cp) t).state ≠ some .escape) := by

/-! ## Part F — the record carrier (`Build/CfgRecord.lean`) -/

/-- Record cells: digits (the `e` of the lift), simulated work cells with a head-mark
track, the state cell and the tag cell. Fields are separated by blank cells. -/
inductive RecSym (Q T : Type) where
  | digit (b : Bool)
  | cell (content : Option Bool) (mark : Bool)
  | st (q : Option Q)
  | tag (τ : T)
deriving DecidableEq

instance recSymFintype (Q T : Type) [Fintype Q] [Fintype T] : Fintype (RecSym Q T) :=
  derive_fintype% _

/-- The digit embedding. -/
def RecSym.digitEmb (Q T : Type) : Bool ↪ RecSym Q T :=
  ⟨RecSym.digit, fun _ _ h => by cases h; rfl⟩

/-- Fixed-width little-endian binary digits. -/
def binDigits (w v : ℕ) : List Bool := (List.range w).map v.testBit

/-- The input-position field width at input length `n`. -/
def posWidth (n : ℕ) : ℕ := (Nat.bits (n + 1)).length

/-- The record of `c` at window radius `s` with tag `τ`: `k + 3` fields. -/
def cfgRecord {k : ℕ} {Q T : Type} {x : List Bool} (s : ℕ) (c : Cfg k Bool Q x) (τ : T) :
    List (List (RecSym Q T)) :=
  [RecSym.st c.state] ::
    ((binDigits (posWidth x.length) c.inputPos.val).map RecSym.digit) ::
    (((List.finRange k).map fun i => (List.range (2 * s + 1)).map fun j =>
      RecSym.cell (c.workTapes i ((j : ℤ) - s)) (decide (c.workTapePos i = (j : ℤ) - s))) ++
    [[RecSym.tag τ]])

/-- Fields joined by single blanks. -/
def joinFields {Γ : Type*} (fs : List (List Γ)) : List (Option Γ) :=
  List.intercalate [none] (fs.map (List.map some))

/-- The tape of a field list laid from cell `0`. -/
def fieldsTape {Γ : Type*} (fs : List (List Γ)) (z : ℤ) : Option Γ :=
  if 0 ≤ z then ((joinFields fs)[z.toNat]?).join else none

/-- Where field `j` starts. -/
def fieldOffset {Γ : Type*} (fs : List (List Γ)) (j : ℕ) : ℕ :=
  ((fs.take j).map (·.length + 1)).sum

/-- Each field is a framed delimited word of the record tape. -/
theorem fieldsTape_field {Γ : Type*} (fs : List (List Γ)) (j : ℕ) (hj : j < fs.length) :
    ∀ p : ℤ, -1 ≤ p → p ≤ (fs[j].length : ℤ) →
      fieldsTape fs (fieldOffset fs j + p) = FinTM.bufferTape fs[j] p := by
  sorry

/-- The window predicate (verbatim `Turing.Cfg.InWindow` of `ClassPSPACE/TQBF.lean`, which
moves here after C1d; decision 12.8.2). -/
def WindowedP {k : ℕ} {Symbol State : Type} {x : List Symbol} (s : ℕ)
    (c : Cfg k Symbol State x) : Prop :=
  (∀ i, |c.workTapePos i| ≤ (s : ℤ)) ∧ ∀ (i : Fin k) (z : ℤ), (s : ℤ) < |z| → c.workTapes i z = none

/-- **Canonical decoding.** On windowed configurations of one input, records agree exactly
when cores and tags agree. -/
theorem cfgRecord_inj {k : ℕ} {Q T : Type} {x : List Bool} (s : ℕ) {c d : Cfg k Bool Q x}
    {τ τ' : T} (hc : WindowedP s c) (hd : WindowedP s d) :
    cfgRecord s c τ = cfgRecord s d τ' ↔
      (c.state = d.state ∧ c.inputPos = d.inputPos ∧ c.workTapes = d.workTapes ∧
        c.workTapePos = d.workTapePos) ∧ τ = τ' := by
  sorry

theorem cfgRecord_length {k : ℕ} {Q T : Type} {x : List Bool} (s : ℕ) (c : Cfg k Bool Q x)
    (τ : T) : (cfgRecord s c τ).length = k + 3 := by
  simp [cfgRecord]

/-- Validity: the field list is the record of some windowed configuration at input length `n`. -/
def RecordValid (k n s : ℕ) (Q T : Type) (fs : List (List (RecSym Q T))) : Prop :=
  ∃ (x : List Bool) (c : Cfg k Bool Q x) (τ : T), x.length = n ∧ WindowedP s c ∧
    cfgRecord s c τ = fs

/-- The record at the head of tape `r`, delimited. -/
def RecordAt {K : ℕ} {Γ S : Type*} {x : List Γ} (d : Cfg K Γ S x) (r : Fin K)
    (fs : List (List Γ)) : Prop :=
  ∀ p : ℤ, -1 ≤ p → p ≤ ((joinFields fs).length : ℤ) →
    d.workTapes r (d.workTapePos r + p) = fieldsTape fs p

example : (cfgRecord (k := 1) (Q := Unit) (T := Bool) (x := [true]) 1
    (Cfg.init () [true]) false).map List.length = [1, posWidth 1, 3, 1] := by
  simp [cfgRecord, binDigits]

/-! ## Part G — the record rows (existential modules; fills after G2) -/

/-- **R8.7, `recordStepTM` (GAP-4).** For a fixed `Bool` machine `M` and tag update `upd`,
a rich module that, from its entry with the record of a windowed `c` at tape `0` (scratch
tapes blank, native input `x.map e`), reaches `yes` with the record of `M.step c` (tag
`upd τ (M.outputSymbol c)`) when that configuration is windowed, and `no` with everything
unchanged otherwise. First-return contract with a linear bound; the input head, the scratch
tapes and every head are restored. -/
theorem exists_recordStepTM {k : ℕ} {Q T : Type} [Fintype Q] [DecidableEq Q] [Fintype T]
    [DecidableEq T] (M : MultiTapeTM k Bool Q) (upd : T → Option Bool → T)
    (hupd : ∀ τ, upd τ none = τ) :
    ∃ (K : ℕ) (P : Type) (_ : Fintype P) (_ : DecidableEq P)
      (R : MultiTapeTM (K + 1) (RecSym Q T) P) (entry yes no : P) (C : ℕ),
      ∀ {x : List Bool} (s : ℕ) (c : Cfg k Bool Q x) (τ : T), WindowedP s c →
      ∀ d : Cfg (K + 1) (RecSym Q T) P (x.map (RecSym.digitEmb Q T)),
        d.state = some entry → RecordAt d 0 (cfgRecord s c τ) →
        (∀ j : Fin (K + 1), j ≠ 0 → ∀ p, d.workTapes j p = none) →
        ∃ t ≤ C * ((joinFields (cfgRecord s c τ)).length + x.length + 1),
          (∀ u < t, (R.runFrom d u).state ≠ some yes ∧ (R.runFrom d u).state ≠ some no) ∧
          (WindowedP s (M.step c) →
            R.runFrom d t =
              { d with
                state := some yes
                workTapes := fun j q =>
                  if j = 0 ∧ d.workTapePos j ≤ q ∧
                      q < d.workTapePos j + ((joinFields (cfgRecord s c τ)).length : ℤ)
                    then fieldsTape (cfgRecord s (M.step c) (upd τ (M.outputSymbol c)))
                      (q - d.workTapePos j)
                  else d.workTapes j q }) ∧
          (¬ WindowedP s (M.step c) → R.runFrom d t = { d with state := some no }) := by
  sorry

/-- **R8.7, `recordEqTM`.** Two records (tapes `0` and `1`) compared field by field. -/
theorem exists_recordEqTM (k : ℕ) (Q T : Type) [Fintype Q] [DecidableEq Q] [Fintype T]
    [DecidableEq T] :
    ∃ (P : Type) (_ : Fintype P) (_ : DecidableEq P) (R : MultiTapeTM 2 (RecSym Q T) P)
      (entry yes no : P) (C : ℕ),
      ∀ {x : List (RecSym Q T)} (fs gs : List (List (RecSym Q T))),
        fs.length = k + 3 → gs.length = k + 3 →
      ∀ d : Cfg 2 (RecSym Q T) P x, d.state = some entry → RecordAt d 0 fs → RecordAt d 1 gs →
        ∃ t ≤ C * ((joinFields fs).length + 1),
          (∀ u < t, (R.runFrom d u).state ≠ some yes ∧ (R.runFrom d u).state ≠ some no) ∧
          R.runFrom d t = { d with state := some (if fs = gs then yes else no) } := by
  sorry

end Turing
