import TCSlib.Complexity.CookLevin.Hardness
set_option maxHeartbeats 0
set_option autoImplicit false
set_option relaxedAutoImplicit false
open Turing
namespace Complexity
open private clCopyTM from TCSlib.Complexity.CookLevin.Hardness
section
attribute [-simp] Turing.Action.apply
attribute [local irreducible] Turing.MultiTapeTM.step Turing.FinTM.controlAction Turing.Action.apply Turing.moveInputPos
private lemma clCopy_idle {x : List Bool} (c : Cfg 2 Bool (Fin 5) x)
    (hc : c.state = some (4 : Fin 5)) : clCopyTM.tm.step c = c := by
  rw [MultiTapeTM.step_of_some hc]
  change (FinTM.controlAction 0 (some (4 : Fin 5))).apply c = c
  rw [FinTM.controlAction_apply, moveInputPos_zero]
  cases c
  simp_all
end
end Complexity
