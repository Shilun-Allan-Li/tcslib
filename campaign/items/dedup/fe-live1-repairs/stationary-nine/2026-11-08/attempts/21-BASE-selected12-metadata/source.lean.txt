import TCSlib.Complexity.CookLevin.Hardness
set_option maxHeartbeats 0
set_option pp.universes true
open Lean Elab Command Meta in
run_cmd do
  let env ← getEnv
  let owners : List Name := [`TCSlib.Complexity.CookLevin.Hardness]
  let targetUsers : List Name := [`Complexity.clCount_idle, `Complexity.clBank_idle, `Complexity.clCopy_idle, `Complexity.clRow_idle, `Complexity.clCmp_idle, `Complexity.clWipe_idle, `Complexity.clFresh_idle, `Complexity.clLoad_idle, `Complexity.clInput_idle, `Complexity.clPrepare_idle, `Complexity.clPrepared_idle, `Complexity.clRecord_idle]
  for home in owners do
    unless (env.getModuleIdx? home).isSome do throwError "missing owner {home}"
  let entries := env.constants.toList.filterMap fun (n, ci) =>
    match env.getModuleIdxFor? n with
    | some i =>
      let home := env.header.moduleNames[i.toNat]!
      if owners.contains home then some (n, ci, home) else none
    | none => none
  for u in targetUsers do
    let hits := entries.filter fun (n, _, _) => (privateToUserName? n).getD n == u
    unless hits.length == 1 do throwError "changed root {u}: expected one hit, got {hits.length}"
  let st ← liftTermElabM getSimpTheorems
  let isProofPart := fun (s : String) =>
    if s.startsWith "_proof_" then
      let parts := (s.drop 7).splitOn "_"
      !parts.isEmpty && parts.all (fun p => !p.isEmpty && p.toList.all Char.isDigit)
    else false
  let kind := fun (ci : ConstantInfo) => match ci with
    | .axiomInfo _ => "axiom" | .defnInfo _ => "definition" | .thmInfo _ => "theorem"
    | .opaqueInfo _ => "opaque" | .quotInfo _ => "quotient" | .inductInfo _ => "inductive"
    | .ctorInfo _ => "constructor" | .recInfo _ => "recursor"
  let mut selected := 0
  let mut selectedRoots := 0
  let mut excludedAux := 0
  let mut nonprivate := 0
  for (n, ci, home) in entries.toArray.qsort (fun a b => a.1.toString < b.1.toString) do
    let user := (privateToUserName? n).getD n
    let pattern := (n.toString.splitOn ".").any isProofPart
    let propType ← liftTermElabM <| Meta.isProp ci.type
    let confirmedAux := pattern && n.isInternalDetail && propType
    let target := targetUsers.contains user
    if target && !isPrivateName n then throwError "changed root is unexpectedly nonprivate {n}"
    let abi := (!isPrivateName n && !confirmedAux) || target
    if !isPrivateName n then nonprivate := nonprivate + 1
    if confirmedAux then excludedAux := excludedAux + 1
    if target then selectedRoots := selectedRoots + 1
    if abi then selected := selected + 1
    let hints := match ci with
      | .defnInfo di => match di.hints with
        | .abbrev => "abbrev" | .opaque => "opaque" | .regular h => s!"regular:{h}"
      | _ => "not_definition"
    let ip ← liftCoreM <| getInstancePriority? n
    let axs ← liftCoreM <| Lean.collectAxioms n
    let fields := [
      ("name",toJson n.toString),("userName",toJson user.toString),("owner",toJson home.toString),
      ("kind",toJson (kind ci)),("private",toJson (isPrivateName n)),
      ("internal",toJson n.isInternalDetail),("levelParams",toJson (ci.levelParams.map Name.toString)),
      ("valuePresent",toJson ci.value?.isSome),("unsafe",toJson ci.isUnsafe),
      ("partial",toJson ci.isPartial),("kernelHints",toJson hints),
      ("reducibility",toJson (reprStr (getReducibilityStatusCore env n))),
      ("simp",toJson (st.isLemma (.decl n))),("unfold",toJson (st.toUnfold.contains n)),
      ("protected",toJson (isProtected env n)),("instancePriority",toJson ip),
      ("typeIsProp",toJson propType),("generatedProofPattern",toJson pattern),
      ("excludedGeneratedProofAux",toJson confirmedAux),("changedPrivateRoot",toJson target),
      ("abiSelected",toJson abi),("typeStructuralHash64",toJson (toString ci.type.hash)),
      ("axioms",toJson ((axs.qsort Name.lt).toList.map Name.toString))]
    let mut typed := fields
    if abi then
      let pretty ← liftTermElabM <| Meta.ppExpr ci.type
      typed := typed ++ [("typeExpr",toJson (reprStr ci.type)),("typePretty",toJson pretty.pretty)]
    else
      typed := typed ++ [("typeOmissionReason",toJson
        (if confirmedAux then "internal generated-proof-pattern Prop cohort" else "private constant outside twelve selected stationary roots"))]
    logInfo m!"COMPACT|{(Json.mkObj typed).compress}"
  unless selectedRoots == 12 do throwError "changed root count {selectedRoots} !=12"
  let summary := Json.mkObj [("rawOwned",toJson entries.length),("rawNonprivate",toJson nonprivate),
    ("abiSelected",toJson selected),("changedRoots",toJson selectedRoots),
    ("excludedGeneratedProofAux",toJson excludedAux),("owners",toJson (owners.map Name.toString)),
    ("proofValuesRendered",toJson false),("fullAXForEveryOwnedConstant",toJson true)]
  logInfo m!"COMPLETE|{summary.compress}"
