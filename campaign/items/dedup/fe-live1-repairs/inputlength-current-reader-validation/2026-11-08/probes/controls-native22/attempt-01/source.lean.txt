import Lean
import TCSlib.Complexity.TuringMachine.Build.InputLength
set_option maxHeartbeats 0
set_option maxRecDepth 8192
set_option smartUnfolding false
set_option autoImplicit false
set_option relaxedAutoImplicit false
attribute [-simp] Turing.Cfg.init Turing.NDTM.initCfg
open Lean Elab Command Meta in
run_cmd do
  let targets : List Name := [`Turing.Action.mapState, `Turing.Cfg.inputSymbol, `Turing.Cfg.workTapeSymbols, `Turing.FinTM.bufferTape, `Turing.FinTM.controlAction, `Turing.MultiTapeTM.runFrom, `Turing.MultiTapeTM.step, `Turing.inputFoldTM, `Turing.inputMoveAction, `Turing.moveInputPos, `Turing.tapeAction]
  let mut i : Nat := 0
  for name in targets do
    let some (.defnInfo info) := (← getEnv).find? name | throwError "not ordinary definition {name}"
    logInfo m!"CONTROL_DEFINITION|{(Json.mkObj [("name",toJson name.toString),
      ("type_Expr",toJson (reprStr info.type)),("value_Expr",toJson (reprStr info.value)),
      ("universes",toJson (info.levelParams.map Name.toString))]).compress}"
    for mode in ["point", "abstract"] do
      i := i + 1
      let result ← liftTermElabM do
        let levels := info.levelParams.map Level.param
        let head := mkConst name levels
        let solve := fun (xs : Array Expr) (lhs rhs : Expr) => do
          let ty ← mkEq lhs rhs
          let closedType ← mkForallFVars xs ty
          logInfo m!"CONTROL_GOAL|{(Json.mkObj [("id",toJson i),("name",toJson name.toString),
            ("mode",toJson mode),("type_Expr",toJson (reprStr closedType)),
            ("universes",toJson (info.levelParams.map Name.toString))]).compress}"
          let goal ← mkFreshExprSyntheticOpaqueMVar ty
          try
            goal.mvarId!.refl
            pure (some (closedType, ← mkLambdaFVars xs (← instantiateMVars goal)))
          catch _ =>
            logInfo m!"CONTROL|{i}|{name}|{mode}|FAIL"
            logError m!"CONTROL_RFL|{i}|{name}|{mode}|rfl did not close reflected identity"
            pure none
        if mode == "abstract" then solve #[] head info.value
        else forallTelescopeReducing info.type fun xs _ => do
          if xs.isEmpty then
            let unitTy := mkConst ``Unit
            solve #[] (Expr.lam `_u unitTy head .default)
              (Expr.lam `_u unitTy info.value .default)
          else solve xs (mkAppN head xs) ((mkAppN info.value xs).headBeta)
      if let some (ty, value) := result then
        let ty ← liftTermElabM <| instantiateMVars ty
        let value ← liftTermElabM <| instantiateMVars value
        if ty.hasFVar || ty.hasMVar || value.hasFVar || value.hasMVar then
          throwError "unclosed control {i}"
        let kernelName := Name.mkSimple ("ConsumerControl_" ++ toString i)
        liftCoreM <| addDecl (.thmDecl { name := kernelName, levelParams := info.levelParams, type := ty, value := value })
        logInfo m!"CONTROL|{i}|{name}|{mode}|PASS|kernelChecked={kernelName}"
  logInfo m!"CONTROL_COMPLETE|{i}"
