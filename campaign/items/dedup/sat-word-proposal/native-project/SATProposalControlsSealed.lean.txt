set_option maxHeartbeats 0
section ExtraOpacity
attribute [-simp] Turing.Action.apply Turing.Cfg.init Turing.MultiTapeTM.initCfg
attribute [local irreducible] Complexity.PolyTimeComputable Turing.moveInputPos Turing.Action.apply Turing.Cfg.init Turing.Cfg.inputSymbol Turing.Cfg.workTapeSymbols Turing.FinTM.ComputesFunInTime Turing.FinTM.ComputesInTime Turing.MultiTapeTM.ComputesInTimeAndSpace Turing.MultiTapeTM.initCfg Turing.MultiTapeTM.runFrom Turing.MultiTapeTM.spaceUsed Turing.MultiTapeTM.spaceUsedByTape Turing.MultiTapeTM.step Turing.MultiTapeTM.visitedByTapeHead

open Lean Elab Command Meta
run_cmd do
  let names : List Name := [`Complexity.PolyTimeComputable, `Turing.moveInputPos, `Turing.Action.apply, `Turing.Cfg.init, `Turing.Cfg.inputSymbol, `Turing.Cfg.workTapeSymbols, `Turing.FinTM.ComputesFunInTime, `Turing.FinTM.ComputesInTime, `Turing.MultiTapeTM.ComputesInTimeAndSpace, `Turing.MultiTapeTM.initCfg, `Turing.MultiTapeTM.runFrom, `Turing.MultiTapeTM.spaceUsed, `Turing.MultiTapeTM.spaceUsedByTape, `Turing.MultiTapeTM.step, `Turing.MultiTapeTM.visitedByTapeHead]
  let mut index := 0
  for n in names do
    let some (.defnInfo di) := (← getEnv).find? n | throwError "not a definition {n}"
    for mode in ["point", "abstract"] do
      index := index + 1
      let pair ← liftTermElabM do
        let levels := di.levelParams.map Level.param
        let head := mkConst n levels
        let solve := fun (lhs rhs : Expr) => do
          let goalType ← mkEq lhs rhs
          let goal ← mkFreshExprSyntheticOpaqueMVar goalType
          try
            goal.mvarId!.refl
            let value ← instantiateMVars goal
            pure (some (← instantiateMVars goalType, value))
          catch ex =>
            logInfo m!"CONTROL|{n}|{mode}|FAIL"
            logError m!"RFL_CONTROL|{n}|{mode}|{ex.toMessageData}"
            pure none
        if mode == "abstract" then
          solve head di.value
        else
          forallTelescope di.type fun xs resultType => do
            let mut lhs := mkAppN head xs
            let mut rhs := (mkAppN di.value xs).headBeta
            match resultType.getAppFn with
            | .const ``Turing.FinTM _ =>
              lhs := mkProj ``Turing.FinTM 0 lhs
              rhs := mkProj ``Turing.FinTM 0 rhs
            | .const ``Turing.Cfg _ =>
              lhs := mkProj ``Turing.Cfg 0 lhs
              rhs := mkProj ``Turing.Cfg 0 rhs
            | .const ``Turing.MultiTapeTM _ =>
              lhs := mkProj ``Turing.MultiTapeTM 0 lhs
              rhs := mkProj ``Turing.MultiTapeTM 0 rhs
            | .const ``Turing.Action _ =>
              lhs := mkProj ``Turing.Action 0 lhs
              rhs := mkProj ``Turing.Action 0 rhs
            | _ => pure ()
            match ← solve lhs rhs with
            | none => pure none
            | some (type, value) =>
              pure (some (← mkForallFVars xs type, ← mkLambdaFVars xs value))
      match pair with
      | none => pure ()
      | some (type, value) =>
        let declName := Name.mkSimple s!"OvernightSATProposalRfl_{index}_{mode}"
        liftCoreM <| addDecl (.thmDecl {
          name := declName, levelParams := di.levelParams, type := type, value := value })
        logInfo m!"CONTROL|{n}|{mode}|PASS|kernelChecked={declName}"

end ExtraOpacity
