/-! #### Polynomial evaluation -/

/-- **The unary polynomial generator** (landing: `Turing.FinTM.polyUnaryTM`, new in `Primitives`
after the private engine `catalogPolyUnaryTM`; the body is the case split of today's
`computesFunInTime_polyUnary`, `Primitives.lean:2133`). The machine that writes
`C·(|x| + 1)^e` ones: for `e = 0` the constant machine of `List.replicate C true`, and for
`e = d + 1` the private engine with `d + 1` nested unary loop tapes of side `|x| + 1`, installed
by one input scan, which emits `C` ones at each point of the box. -/
def polyUnaryTM (C : ℕ) : ℕ → FinTM Bool
  | 0 => constTM (List.replicate C true)
  | d + 1 => catalogPolyUnaryTM d C

/-- **The unary generator's time** (landing: `Turing.FinTM.polyUnaryTM_computes`, new in
`Primitives` before `computesFunInTime_polyUnary`). The generator writes `C·(|x| + 1)^e` ones
within `(C + 5e + 4)·(n + 1)^e` steps: the sharp degree `e`, not the row's `e + 1`.

**Proof sketch.** For `e = 0`, `constTM_computes` gives `C + 1 ≤ C + 4` steps. For `e = d + 1`,
today's private `catalogPoly_unary_computes d C` (`Primitives.lean:1371`) is exactly the bound
`(C + 5(d + 1) + 4)·(n + 1)^(d + 1)`. This replaces Catalog's `f2_unary_sharp`, whose
`e = 0` case went through a linear allowance only because it reused the constant row's
existential constant. -/
theorem polyUnaryTM_computes (C e : ℕ) :
    (polyUnaryTM C e).ComputesFunInTime
      (fun x => List.replicate (C * (x.length + 1) ^ e) true)
      (fun n => (C + 5 * e + 4) * (n + 1) ^ e) := by
  sorry

/-- **The unary generator's space** (landing: `Turing.FinTM.polyUnaryTM_spaceUsed`, new in
`Primitives` after `polyUnaryTM_computes`; for `e = d + 1` it is Catalog's private
`f2_poly_space`, which moves with `f2_polyHeads`, `f2_polyHeads_bounds` and `f2_poly_step`). At
every time the generator has visited at most `5e·(|x| + 1)` work cells.

**Proof sketch.** For `e = 0` there are no work tapes. For `e = d + 1`, `f2_poly_space`: the
startup lasts `2(|x| + 1)` steps from the origin, and the loop invariant confines every later
head to `[-1, |x| + 1]`, including after the halt; contain the visited images in the larger
interval and sum over the `d + 1` tapes. The number of loop iterations never appears. -/
theorem polyUnaryTM_spaceUsed (C e : ℕ) (x : List Bool) (t : ℕ) :

      5 * e * (x.length + 1) := by
  sorry

/-- **The binary polynomial evaluator** (landing: `Turing.FinTM.polyBitsTM`, new in `Primitives`
after `polyUnaryTM`; the body is the case split of today's
`computesFunInTime_polyBits_spaceUsed`, `Build/Catalog.lean:4377`). The machine computing
`x ↦ Nat.bits (C·(|x| + 1)^e)`: the constant machine of the answer when `C = 0` or `e = 0`, and
otherwise the unary generator buffered into the binary length counter. The split avoids the
generator's linear banks where the answer is a constant (round-1 finding 5 of the space rows). -/
def polyBitsTM (C e : ℕ) : FinTM Bool :=
  if C = 0 then constTM (Nat.bits 0) else
    match e with
    | 0 => constTM (Nat.bits C)
    | d + 1 => bufferedCompTM (polyUnaryTM C (d + 1)) lengthBitsTM

/-- **The binary evaluator's time** (landing: `Turing.FinTM.polyBitsTM_computes`, new in
`Primitives` before `computesFunInTime_polyBits`; proved, from the gated specs). The evaluator
computes `x ↦ Nat.bits (C·(|x| + 1)^e)` within `(7C + 5e + 11)·(n + 1)^e` steps.

**Proof sketch.** If `C = 0` or `e = 0`, the constant machine of the answer takes
`|Nat.bits C| + 1 ≤ C + 1` steps (`Turing.length_bits_le_self`). Otherwise the buffered
composition takes the generator's `(C + 5e + 4)·P` steps, `P = (n + 1)^e`, plus the
intermediate's length `C·P` and `2`, plus the counter's `5·(C·P + 1)`, in all
`(7C + 5e + 4)·P + 7 ≤ (7C + 5e + 11)·P` (`bufferedCompTM_computesInTime`). -/
theorem polyBitsTM_computes (C e : ℕ) :
    (polyBitsTM C e).ComputesFunInTime (fun x => Nat.bits (C * (x.length + 1) ^ e))
      (fun n => (7 * C + 5 * e + 11) * (n + 1) ^ e) := by
  intro x
  have hp : 1 ≤ (x.length + 1) ^ e := Nat.one_le_pow _ _ (Nat.succ_pos _)
  by_cases hC : C = 0
  · subst hC
    simp only [polyBitsTM, if_true, Nat.zero_mul]
    refine (constTM_computes (Nat.bits 0) x).mono ?_
    rw [Nat.zero_bits, List.length_nil]
    exact Nat.mul_pos (by omega) hp
  · cases e with
    | zero =>
      simp only [polyBitsTM, hC, if_false, pow_zero, Nat.mul_one]
      refine (constTM_computes (Nat.bits C) x).mono ?_
      have := Turing.length_bits_le_self C
      omega
    | succ d =>
      simp only [polyBitsTM, hC, if_false]
      have h := bufferedCompTM_computesInTime (polyUnaryTM C (d + 1)) lengthBitsTM
        (polyUnaryTM_computes C (d + 1) x)
        (lengthBitsTM_computes (List.replicate (C * (x.length + 1) ^ (d + 1)) true))
      simp only [List.length_replicate] at h
      refine h.mono ?_
      dsimp only
      generalize (x.length + 1) ^ (d + 1) = P at hp ⊢
      calc (C + 5 * (d + 1) + 4) * P + C * P + 2 + 5 * (C * P + 1)
          = (7 * C + 5 * (d + 1) + 4) * P + 7 := by ring
        _ ≤ (7 * C + 5 * (d + 1) + 4) * P + 7 * P := by omega
        _ = (7 * C + 5 * (d + 1) + 11) * P := by ring

  | e + 1 => loopHostTM (splitBodyTM (catalogPolyUnaryTM e C) (.loop (Fin.last e)))

/-- **The width-parametric split search** (landing: `Turing.FinTM.splitSolveWithTM`, new in
