
/-! ## Consumer probes (evidence only; not landing material)

Two derivations from the assembly lemmas as stated. Probe 1 derives the statement of
`Turing.exists_timed_universal_NDTM` (frozen in `NTimeHierarchy.lean`, not imported here) from the
interpreter core `Turing.exists_ndInterpreterCore` (a gated sorry) and a hypothesis that is the
universal machine's setup obligation. Probe 2 derives the `Complexity.NTIME` membership shape that
the diagonalizer's sketch uses (`D ∈ NTIME (g + 1)` by construction, with `L(D)` read off per
guessed stream). The only sorried declarations they reach are `Turing.exists_ndInterpreterCore`
and `Turing.ndWriterTM_run_block`. -/

namespace Turing

/-- **Consumer probe 1: the universal NDTM.** If every tape bank `K` with a stream tape `jR` has
a setup machine `P` (on some `k ≥ K + 1` tapes, `ι` selecting `K` of them and `cap` outside the
selection) that, on `⟨⟨bits t, α⟩, x⟩`, reaches its exit first within `C_P·(t + 1)`, clean, with
a nonblank block of exactly `t` cells on tape `ι jR` from cell `0`, then the statement of
`Turing.exists_timed_universal_NDTM` holds. The universal machine is
`Turing.guessRunTM P pexit (ι jR) (embedSilentTM ι cap R) entry acc rej` for the core `R` of
`Turing.exists_ndInterpreterCore`, with `C := C_P + C_R + 13`.

**Proof sketch.** `Turing.guessRunTM_silent_spec` with `m = t` and `T = C_R·(t + 1)`: the core's
contract gives, for every stream `s` of length `t`, a first verdict within `C_R·(t + 1)`, which
is `accept` iff the coded branch along `s.take t = s` halts with output `[true]`. The budget
`TP + 2t + C_R·(t + 1) + 11` is at most `C·(t + 1)`, so halting is `HaltsWithin.mono`, and
acceptance is the right side of the silent specification, which is
`Turing.FinNDTM.AcceptsWithin x t` of the coded machine once the first verdict time is unique. -/
theorem probe_universal_of_setup (c : EffectiveNDMachineCode)
    (hsetup : ∀ (K : ℕ) (jR : Fin K), ∃ (k : ℕ) (ι : Fin K ↪ Fin k) (cap : Fin k),
      cap ∉ Set.range ι ∧ ∃ (SP : Type) (_ : Fintype SP) (_ : DecidableEq SP)
        (P : MultiTapeTM k Bool SP) (pexit : SP),
        ∀ α : List Bool, ∃ CP : ℕ, ∀ (x : List Bool) (t : ℕ),
          ∃ TP ≤ CP * (t + 1), ∃ blk : ℤ → Option Bool,
            (∀ p : ℤ, blk p ≠ none ↔ 0 ≤ p ∧ p < t) ∧
            P.runFrom (P.initCfg (pairEncode (pairEncode (Nat.bits t) α) x)) TP =
              ⟨some pexit, 1, Function.update (fun _ _ => none) (ι jR) blk, fun _ => 0, []⟩ ∧
            ∀ u < TP,
              (P.runFrom (P.initCfg (pairEncode (pairEncode (Nat.bits t) α) x)) u).state ≠
                some pexit) :
    ∃ UN : FinNDTM Bool, ∀ α : List Bool, ∃ C : ℕ, ∀ (x : List Bool) (t : ℕ),
      UN.tm.HaltsWithin (pairEncode (pairEncode (Nat.bits t) α) x) (C * (t + 1)) ∧
      (UN.AcceptsWithin (pairEncode (pairEncode (Nat.bits t) α) x) (C * (t + 1)) ↔
        (c.decode α).toFinNDTM.AcceptsWithin x t) := by
  obtain ⟨K, SR, hfin, hdec, R, jR, entry, acc, rej, hne, hR⟩ := exists_ndInterpreterCore c
  obtain ⟨k, ι, cap, hcap, SP, hfinP, hdecP, P, pexit, hP⟩ := hsetup K jR
  refine ⟨⟨k, Σ i, GuessRunPhase.State SP SR i,
    guessRunTM P pexit (ι jR) (embedSilentTM ι cap R) entry acc rej⟩, fun α => ?_⟩
  obtain ⟨CR, hCR⟩ := hR α
  obtain ⟨CP, hCP⟩ := hP α
  refine ⟨CP + CR + 13, fun x t => ?_⟩
  obtain ⟨TP, hTP, blk, hblk, hrun, hfirst⟩ := hCP x t
  -- the core's verdict on every stream of length `t`
  have hcore : ∀ s : List Bool, s.length = t → ∃ T, T ≤ CR * (t + 1) ∧
      (R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s) T).state
        = some (if ((c.decode α).toFinNDTM.tm.runWith s
              ((c.decode α).toFinNDTM.tm.initCfg x)).state = none ∧
            ((c.decode α).toFinNDTM.tm.runWith s
              ((c.decode α).toFinNDTM.tm.initCfg x)).output = [true] then acc else rej) ∧
      ∀ u < T,
        (R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s)
          u).state ≠ some acc ∧
        (R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s)
          u).state ≠ some rej := by
    intro s hs
    obtain ⟨T, -, hT, hst, hno⟩ := hCR x t s (by omega)
    rw [List.take_of_length_le (by omega)] at hst
    exact ⟨T, hT, hst, hno⟩
  have hR' : ∀ s : List Bool, s.length = t → ∃ TR ≤ CR * (t + 1),
      ((R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s)
          TR).state = some acc ∨
        (R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s)
          TR).state = some rej) ∧
      ∀ u < TR,
        (R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s)
          u).state ≠ some acc ∧
        (R.runFrom (streamStartCfg (pairEncode (pairEncode (Nat.bits t) α) x) jR entry s)
          u).state ≠ some rej := by
    intro s hs
    obtain ⟨T, hT, hst, hno⟩ := hcore s hs
    refine ⟨T, hT, ?_, hno⟩
    rw [hst]
    split_ifs
    · exact Or.inl rfl
    · exact Or.inr rfl
  have hspec := guessRunTM_silent_spec ι cap hcap jR R entry acc rej P pexit TP t blk hblk hrun
    hfirst (CR * (t + 1)) hR'
  have hbud : TP + 2 * t + CR * (t + 1) + 11 ≤ (CP + CR + 13) * (t + 1) := by
    have e : (CP + CR + 13) * (t + 1) = CP * (t + 1) + CR * (t + 1) + 13 * (t + 1) := by
      rw [Nat.add_mul, Nat.add_mul]
    rw [e]
    generalize CP * (t + 1) = a at hTP ⊢
    generalize CR * (t + 1) = b
    omega
  refine ⟨hspec.1.mono hbud, (hspec.2 _ hbud).trans ?_⟩
  unfold FinNDTM.AcceptsWithin
  constructor
  · rintro ⟨s, hs, TR, hacc, hf⟩
    obtain ⟨T, -, hst, hno⟩ := hcore s hs
    have hTR : TR = T := by
      by_contra hne'
      rcases Nat.lt_or_gt_of_ne hne' with h | h
      · exact (hno TR h).1 hacc
      · have h' := hf T h
        rw [hst] at h'
        split_ifs at h' with hc
        · exact h'.1 rfl
        · exact h'.2 rfl
    subst hTR
    rw [hst] at hacc
    split_ifs at hacc with hcond
    · exact ⟨s, hs, hcond.1, hcond.2⟩
    · exact absurd (Option.some.inj hacc).symm hne
  · rintro ⟨s, hs, hhalt, hout⟩
    obtain ⟨T, -, hst, hno⟩ := hcore s hs
    exact ⟨s, hs, T, by rw [hst, if_pos ⟨hhalt, hout⟩], hno⟩

/-- **Consumer probe 2: the `NTIME` membership shape of the diagonalizer.** If, on every input
`z`, the setup `P` meets the block hypotheses of `Turing.guessRunTM_runWith_handoff` with stream
length `m z`, the run `Q` reaches a first verdict within `T z` on every stream, and the total
`TP z + 2·m z + T z + 11` is at most `A·(g |z| + 1)`, then the language of inputs with some
stream whose first verdict is `acc` (with empty output) is in `NTIME (g + 1)`, decided by
`Turing.guessRunTM P pexit j Q qentry acc rej` with constant `A`. This is `D ∈ NTIME (g + 1)` by
construction, with `L(D)` characterized per guessed stream, as `Complexity.ntime_hierarchy`'s
sketch uses it.

**Proof sketch.** `Complexity.NTIME` at constant `A` and the bundled host: all-branch halting is
`Turing.guessRunTM_haltsWithin` and `Turing.NDTM.HaltsWithin.mono`; membership is
`Turing.guessRunTM_accepts_iff` at the budget `A·(g |z| + 1)`. -/
theorem probe_ntime_of_guessRun {k : ℕ} {SP SQ : Type} [Fintype SP] [DecidableEq SP]
    [Fintype SQ] [DecidableEq SQ] (P : MultiTapeTM k Bool SP) (pexit : SP) (j : Fin k)
    (Q : MultiTapeTM k Bool SQ) (qentry acc rej : SQ) (g : ℕ → ℕ) (A : ℕ)
    (TP m T : List Bool → ℕ) (cP : (z : List Bool) → Cfg k Bool SP z)
    (hP : ∀ z, P.runFrom (P.initCfg z) (TP z) = cP z ∧ (cP z).state = some pexit ∧
      (∀ u < TP z, (P.runFrom (P.initCfg z) u).state ≠ some pexit) ∧
      (∀ p : ℕ, p < m z → (cP z).workTapes j ((cP z).workTapePos j + p) ≠ none) ∧
      (cP z).workTapes j ((cP z).workTapePos j + m z) = none ∧
      (cP z).workTapes j ((cP z).workTapePos j - 1) = none)
    (hQ : ∀ (z s : List Bool), s.length = m z → ∃ TQ ≤ T z,
      ((Q.runFrom (guessRunCfg (cP z) j s qentry) TQ).state = some acc ∨
        (Q.runFrom (guessRunCfg (cP z) j s qentry) TQ).state = some rej) ∧
      ∀ u < TQ, (Q.runFrom (guessRunCfg (cP z) j s qentry) u).state ≠ some acc ∧
        (Q.runFrom (guessRunCfg (cP z) j s qentry) u).state ≠ some rej)
    (hbud : ∀ z : List Bool, TP z + 2 * m z + T z + 11 ≤ A * (g z.length + 1)) :
    ({z | ∃ s : List Bool, s.length = m z ∧ ∃ TQ : ℕ,
        (Q.runFrom (guessRunCfg (cP z) j s qentry) TQ).state = some acc ∧
        (∀ u < TQ, (Q.runFrom (guessRunCfg (cP z) j s qentry) u).state ≠ some acc ∧
          (Q.runFrom (guessRunCfg (cP z) j s qentry) u).state ≠ some rej) ∧
        (Q.runFrom (guessRunCfg (cP z) j s qentry) TQ).output = []} : Language Bool) ∈
      Complexity.NTIME (fun n => g n + 1) := by
  refine ⟨A, ⟨k, Σ i, GuessRunPhase.State SP SQ i, guessRunTM P pexit j Q qentry acc rej⟩,
    fun z => ?_⟩
  obtain ⟨h1, h2, h3, h4, h5, h6⟩ := hP z
  exact ⟨(guessRunTM_haltsWithin P pexit j Q qentry acc rej (TP z) (cP z) h1 h2 h3 (m z) h4 h5
      h6 (T z) (hQ z)).mono (hbud z),
    (guessRunTM_accepts_iff P pexit j Q qentry acc rej (TP z) (cP z) h1 h2 h3 (m z) h4 h5 h6
      (T z) (hQ z) _ (hbud z)).symm⟩

end Turing
