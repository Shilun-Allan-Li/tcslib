/-- The round-2 contract, verbatim from the patched module. -/
theorem exists_seamLayoutTM (k : ℕ) (src tbl sm : Fin k) (hst : src ≠ tbl) (hss : src ≠ sm)
    (hts : tbl ≠ sm) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM k Bool Q)
      (entry done : Q) (C : ℕ),
      ∀ (a v : List Bool) (q r : ℕ),
        let s := pairEncode a ((List.replicate q true ++ [false]) ++ v)
        r ≤ s.length →
        ∃ T ≤ C * (s.length + 1), ∀ {z : List Bool} (d : Cfg k Bool Q z),
          d.state = some entry →
          d.workTapes src = FinTM.bufferTape s → d.workTapePos src = r →
          d.workTapes tbl = (fun _ => none) → d.workTapePos tbl = 0 →
          d.workTapes sm = (fun _ => none) → d.workTapePos sm = 0 →
          (L.runFrom d T).state = some done ∧
          (L.runFrom d T).inputPos = d.inputPos ∧
          (L.runFrom d T).workTapes =
            Function.update (Function.update (Function.update d.workTapes src (fun _ => none))
              tbl (FinTM.bufferTape v))
              sm (fun p => FinTM.bufferTape (List.replicate q true ++ [false]) (p - 2)) ∧
          (L.runFrom d T).workTapePos =
            Function.update (Function.update (Function.update d.workTapePos src 0) tbl 0) sm 0 ∧
          (∀ u ≤ T, (L.runFrom d u).output = d.output) ∧
          ∀ u < T, (L.runFrom d u).state ≠ some done := by
  sorry


/-- **1. The round-1 form is a corollary**: `T` chosen per configuration, the source head
anywhere in `[0, |s|]`. -/
theorem seamLayout_round1 (k : ℕ) (src tbl sm : Fin k) (hst : src ≠ tbl) (hss : src ≠ sm)
    (hts : tbl ≠ sm) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM k Bool Q)
      (entry done : Q) (C : ℕ),
      ∀ {z : List Bool} (d : Cfg k Bool Q z) (a v : List Bool) (q : ℕ),
        let s := pairEncode a ((List.replicate q true ++ [false]) ++ v)
        d.state = some entry →
        d.workTapes src = FinTM.bufferTape s →
        0 ≤ d.workTapePos src → d.workTapePos src ≤ (s.length : ℤ) →
        d.workTapes tbl = (fun _ => none) → d.workTapePos tbl = 0 →
        d.workTapes sm = (fun _ => none) → d.workTapePos sm = 0 →
        ∃ T ≤ C * (s.length + 1),
          (L.runFrom d T).state = some done ∧
          (L.runFrom d T).inputPos = d.inputPos ∧
          (L.runFrom d T).workTapes =
            Function.update (Function.update (Function.update d.workTapes src (fun _ => none))
              tbl (FinTM.bufferTape v))
              sm (fun p => FinTM.bufferTape (List.replicate q true ++ [false]) (p - 2)) ∧
          (L.runFrom d T).workTapePos =
            Function.update (Function.update (Function.update d.workTapePos src 0) tbl 0) sm 0 ∧
          (∀ u ≤ T, (L.runFrom d u).output = d.output) ∧
          ∀ u < T, (L.runFrom d u).state ≠ some done := by
  obtain ⟨Q, hQ, hQ', L, entry, done, C, h⟩ := exists_seamLayoutTM k src tbl sm hst hss hts
  refine ⟨Q, hQ, hQ', L, entry, done, C, ?_⟩
  intro z d a v q s hstate htape h0 h1 htbl htblp hsm hsmp
  obtain ⟨T, hT, hrun⟩ := h a v q (d.workTapePos src).toNat (by simp only [s] at h1; omega)
  exact ⟨T, hT, hrun d hstate htape (by omega) htbl htblp hsm hsmp⟩

/-- **2. U2-S's use**: for the code `α`'s captured serialization (head one past it), one time `T`
serves every suffix `w` of the native input `pairEncode α w`, and the end is U2-S's tapes `k` and
`k + 1` exactly. -/
theorem seamLayout_u2s {kk K : ℕ} (E : EffectiveCodeK kk) (src tbl sm : Fin K) (hst : src ≠ tbl)
    (hss : src ≠ sm) (hts : tbl ≠ sm) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM K Bool Q)
      (entry done : Q),
      ∀ α : List Bool, ∃ T : ℕ, ∀ (w : List Bool) (d : Cfg K Bool Q (pairEncode α w)),
        d.state = some entry →
        d.workTapes src = FinTM.bufferTape (serializeK (E.machine α)) →
        d.workTapePos src = ((serializeK (E.machine α)).length : ℤ) →
        d.workTapes tbl = (fun _ => none) → d.workTapePos tbl = 0 →
        d.workTapes sm = (fun _ => none) → d.workTapePos sm = 0 →
        (L.runFrom d T).state = some done ∧
        (∀ u < T, (L.runFrom d u).state ≠ some done) ∧
        (∀ u ≤ T, (L.runFrom d u).output = d.output) ∧
        (L.runFrom d T).workTapes tbl = FinTM.bufferTape (tableBitsK (E.machine α).tr) ∧
        (L.runFrom d T).workTapes sm =
          (fun p => FinTM.bufferTape (unaryFin (E.machine α).q₀) (p - 2)) ∧
        (L.runFrom d T).workTapes src = (fun _ => none) ∧
        (L.runFrom d T).workTapePos src = 0 ∧ (L.runFrom d T).workTapePos tbl = 0 ∧
        (L.runFrom d T).workTapePos sm = 0 ∧ (L.runFrom d T).inputPos = d.inputPos := by
  obtain ⟨Q, hQ, hQ', L, entry, done, C, h⟩ := exists_seamLayoutTM K src tbl sm hst hss hts
  refine ⟨Q, hQ, hQ', L, entry, done, fun α => ?_⟩
  have hs : serializeK (E.machine α) =
      pairEncode (E.numStates α).bits
        ((List.replicate ((E.machine α).q₀ : ℕ) true ++ [false]) ++ tableBitsK (E.machine α).tr) := by
    rw [serializeK_eq, unaryFin_eq]
  obtain ⟨T, -, hrun⟩ := h (E.numStates α).bits (tableBitsK (E.machine α).tr) (E.machine α).q₀
    (serializeK (E.machine α)).length (by rw [hs])
  refine ⟨T, fun w d hstate htape hpos htbl htblp hsm hsmp => ?_⟩
  obtain ⟨hdone, hin, htapes, hheads, hout, hfirst⟩ :=
    hrun d hstate (by rw [htape, hs]) (by rw [hpos, hs]) htbl htblp hsm hsmp
  refine ⟨hdone, hfirst, hout, ?_, ?_, ?_, ?_, ?_, ?_, hin⟩
  · rw [htapes, Function.update_of_ne hts, Function.update_self]
  · rw [htapes, Function.update_self, unaryFin_eq]
  · rw [htapes, Function.update_of_ne hss, Function.update_of_ne hst, Function.update_self]
  · rw [hheads, Function.update_of_ne hss, Function.update_of_ne hst, Function.update_self]
  · rw [hheads, Function.update_of_ne hts, Function.update_self]
  · rw [hheads, Function.update_self]

/-- **3. R's use**: at the R1′ capture endpoint with an empty prefix (`Turing.embedSilentCfg_cap`:
the capture tape holds the recorded output, its head one past it), the recorded output being
`serializeK M`, the contract applies, and the end has `StepBlock.Rep`'s table and summary fields
for `M.initCfg x` (empty output, initial state). -/
theorem seamLayout_rep_fields {K m kk n : ℕ} (src tbl sm : Fin K) (hst : src ≠ tbl)
    (hss : src ≠ sm) (hts : tbl ≠ sm) :
    ∃ (Q : Type) (_ : Fintype Q) (_ : DecidableEq Q) (L : MultiTapeTM K Bool Q)
      (entry done : Q) (C : ℕ),
      ∀ (M : MultiTapeTM kk Bool (Fin (n + 1))) (x : List Bool) {S : Type} {z : List Bool}
        (ι : Fin m ↪ Fin K) (_ : src ∉ Set.range ι) (tapes : Fin K → ℤ → Option Bool)
        (heads : Fin K → ℤ) (out₀ : List Bool) (c : Cfg m Bool S z)
        (_ : c.output = serializeK M) (d : Cfg K Bool Q z),
        d.state = some entry →
        d.workTapes src = (embedSilentCfg ι src tapes heads [] out₀ c).workTapes src →
        d.workTapePos src = (embedSilentCfg ι src tapes heads [] out₀ c).workTapePos src →
        d.workTapes tbl = (fun _ => none) → d.workTapePos tbl = 0 →
        d.workTapes sm = (fun _ => none) → d.workTapePos sm = 0 →
        ∃ T ≤ C * ((serializeK M).length + 1),
          (L.runFrom d T).state = some done ∧
          (L.runFrom d T).workTapes tbl = FinTM.bufferTape (tableBitsK M.tr) ∧
          (L.runFrom d T).workTapes sm =
            StepBlock.summaryCells (StepBlock.outputSummary (M.initCfg x).output)
              (((M.initCfg x).state.map unaryFin).getD []) ∧
          (L.runFrom d T).workTapePos tbl = 0 ∧ (L.runFrom d T).workTapePos sm = 0 := by
  obtain ⟨Q, hQ, hQ', L, entry, done, C, h⟩ := exists_seamLayoutTM K src tbl sm hst hss hts
  refine ⟨Q, hQ, hQ', L, entry, done, C, ?_⟩
  intro M x S z ι hcap tapes heads out₀ c hc d hstate htape hpos htbl htblp hsm hsmp
  obtain ⟨hcapT, hcapP⟩ := embedSilentCfg_cap ι src hcap tapes heads [] out₀ c
  rw [hcapT, List.nil_append, hc] at htape
  rw [hcapP, List.nil_append, hc] at hpos
  have hs : serializeK M =
      pairEncode n.bits ((List.replicate (M.q₀ : ℕ) true ++ [false]) ++ tableBitsK M.tr) := by
    rw [serializeK_eq, unaryFin_eq]
  obtain ⟨T, hT, hrun⟩ := h n.bits (tableBitsK M.tr) M.q₀ (serializeK M).length (by rw [hs])
  obtain ⟨hdone, -, htapes, hheads, -, -⟩ :=
    hrun d hstate (by rw [htape, hs]) (by rw [hpos, hs]) htbl htblp hsm hsmp
  refine ⟨T, by rw [hs]; exact hT, hdone, ?_, ?_, ?_, ?_⟩
  · rw [htapes, Function.update_of_ne hts, Function.update_self]
  · rw [htapes, Function.update_self, MultiTapeTM.initCfg_eq, Cfg.init_output,
      Cfg.init_state, Option.map_some, Option.getD_some, StepBlock.outputSummary_nil]
    funext p
    rw [StepBlock.summaryCells_eq, unaryFin_eq]
    split_ifs with hp
    · subst hp
      exact FinTM.bufferTape_of_neg _ (by norm_num)
    · rfl
  · rw [hheads, Function.update_of_ne hts, Function.update_self]
  · rw [hheads, Function.update_self]

/-- **4. The fields assemble `Rep`**: a step-block configuration with those table and summary
fields (and `M.initCfg x`'s input head, bank and heads) represents `M.initCfg x`. -/
example {k n : ℕ} (M : MultiTapeTM k Bool (Fin (n + 1))) (x : List Bool)
    (d : Cfg (k + 2) Bool (Σ key, StepBlock.Ph k key) x)
    (hin : d.inputPos = (M.initCfg x).inputPos)
    (hbank : ∀ i : Fin k, d.workTapes (Fin.castAdd 2 i) = (M.initCfg x).workTapes i ∧
      d.workTapePos (Fin.castAdd 2 i) = (M.initCfg x).workTapePos i)
    (htbl : d.workTapes (StepBlock.tableTape k) = FinTM.bufferTape (tableBitsK M.tr))
    (hsm : d.workTapes (StepBlock.summaryTape k) =
      StepBlock.summaryCells (StepBlock.outputSummary (M.initCfg x).output)
        (((M.initCfg x).state.map unaryFin).getD []))
    (htblp : d.workTapePos (StepBlock.tableTape k) = 0)
    (hsmp : d.workTapePos (StepBlock.summaryTape k) = 0) :
    StepBlock.Rep M.tr (M.initCfg x) d :=
  StepBlock.Rep_iff.mpr ⟨hin, hbank, htbl, hsm, htblp, hsmp⟩

/-- **5. Not vacuous**: for every word, head start and native input there is a start
configuration meeting all seven hypotheses (any state type, distinct tapes). -/
example {k : ℕ} {Q : Type} (entry : Q) (src tbl sm : Fin k) (hst : src ≠ tbl) (hss : src ≠ sm)
    (s z : List Bool) (r : ℕ) :
    ∃ d : Cfg k Bool Q z, d.state = some entry ∧ d.workTapes src = FinTM.bufferTape s ∧
      d.workTapePos src = r ∧ d.workTapes tbl = (fun _ => none) ∧ d.workTapePos tbl = 0 ∧
      d.workTapes sm = (fun _ => none) ∧ d.workTapePos sm = 0 :=
  ⟨⟨some entry, 0, Function.update (fun _ _ => none) src (FinTM.bufferTape s),
      Function.update (fun _ => 0) src (r : ℤ), []⟩,
    rfl, by simp, by simp, by simp [hst.symm],
    by simp [hst.symm], by simp [hss.symm],
    by simp [hss.symm]⟩
