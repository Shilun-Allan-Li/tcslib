variable {k : ℕ}

/-- **Phase alphabet of the lockstep scan**: the forward pass, the return pass carrying the
verdict chosen at the turn, and the live exit anchors, one per verdict. -/
inductive LockScanPhase (V : Type) where
  /-- the forward pass -/
  | fwd
  /-- the return pass, carrying the verdict -/
  | bwd (v : V)
  /-- the live exit anchors, one per verdict -/
  | done (v : V)
deriving DecidableEq

/-- **The lockstep scan.** The lane tapes (`lane j = true`) move together. In the forward pass
the rule reads the whole column under the heads and either writes and steps the lanes right,
or writes, steps them left and turns with a verdict `v`. The return pass steps the lanes left,
writing `bwdWrite`, while the guide tape reads a symbol; at the guide's left blank it steps
right into `done v`. Other tapes never move or change. -/
def lockScanTM {Γ : Type*} {V : Type} (k : ℕ) (lane : Fin k → Bool) (guide : Fin k)
    (fwdRule : (Fin k → Option Γ) → Option V × (Fin k → Option (Option Γ)))
    (bwdWrite : Fin k → Option (Option Γ)) : MultiTapeTM k Γ (LockScanPhase V) where
  q₀ := .fwd
  tr := fun q _ w =>
    match q with
    | .fwd =>
      match (fwdRule w).1 with
      | none => ⟨0, fun j => if lane j then ((fwdRule w).2 j, SignType.pos) else (none, 0),
          none, some .fwd⟩
      | some v => ⟨0, fun j => if lane j then ((fwdRule w).2 j, SignType.neg) else (none, 0),
          none, some (.bwd v)⟩
    | .bwd v =>
      match w guide with
      | some _ => ⟨0, fun j => if lane j then (bwdWrite j, SignType.neg) else (none, 0),
          none, some (.bwd v)⟩
      | none => ⟨0, fun j => if lane j then (none, SignType.pos) else (none, 0),
          none, some (.done v)⟩
    | .done v => ⟨0, fun _ => (none, 0), none, some (.done v)⟩

section
variable {Γ : Type*} {V : Type} {x : List Γ}
variable (lane : Fin k → Bool) (guide : Fin k)
  (fwdRule : (Fin k → Option Γ) → Option V × (Fin k → Option (Option Γ)))
  (bwdWrite : Fin k → Option (Option Γ))

/-- The column the forward pass reads after `r` steps: lane tapes at offset `r`, the others at
their fixed heads. -/
private def lsCol (c : Cfg k Γ (LockScanPhase V) x) (r : ℕ) : Fin k → Option Γ :=
  fun j => c.workTapes j (c.workTapePos j + if lane j then (r : ℤ) else 0)

/-- A lane cell after the forward pass wrote it. -/
private def lsFwdCell (c : Cfg k Γ (LockScanPhase V) x) (j : Fin k) (q : ℤ) : Option Γ :=
  ((fwdRule (lsCol lane c (q - c.workTapePos j).toNat)).2 j).getD (c.workTapes j q)

/-- The forward family: `r` forward steps taken. -/
private def lsF (c : Cfg k Γ (LockScanPhase V) x) (r : ℕ) : Cfg k Γ (LockScanPhase V) x :=
  { c with
    state := some .fwd
    workTapes := fun j q =>
      if lane j ∧ c.workTapePos j ≤ q ∧ q < c.workTapePos j + (r : ℤ)
        then lsFwdCell lane fwdRule c j q else c.workTapes j q
    workTapePos := fun j => c.workTapePos j + if lane j then (r : ℤ) else 0 }

/-- The tapes once the return pass has rewritten the offsets `[r, d)`. -/
private def lsTape (c : Cfg k Γ (LockScanPhase V) x) (d r : ℕ) : Fin k → ℤ → Option Γ :=
  fun j q =>
    if lane j ∧ c.workTapePos j ≤ q ∧ q ≤ c.workTapePos j + (d : ℤ) then
      (if c.workTapePos j + (r : ℤ) ≤ q ∧ q < c.workTapePos j + (d : ℤ)
        then (bwdWrite j).getD (lsFwdCell lane fwdRule c j q) else lsFwdCell lane fwdRule c j q)
    else c.workTapes j q

/-- The return family: the heads one cell left of offset `r`. -/
private def lsR (c : Cfg k Γ (LockScanPhase V) x) (d : ℕ) (v : V) (r : ℕ) :
    Cfg k Γ (LockScanPhase V) x :=
  { c with
    state := some (.bwd v)
    workTapes := lsTape lane fwdRule bwdWrite c d r
    workTapePos := fun j => c.workTapePos j + if lane j then (r : ℤ) - 1 else 0 }

/-- The exit configuration. -/
private def lsD (c : Cfg k Γ (LockScanPhase V) x) (d : ℕ) (v : V) :
    Cfg k Γ (LockScanPhase V) x :=
  { c with state := some (.done v), workTapes := lsTape lane fwdRule bwdWrite c d 0 }

/-- One step of the scan from a live configuration. -/
private lemma ls_step (c' : Cfg k Γ (LockScanPhase V) x) (q : LockScanPhase V)
    (hq : c'.state = some q) :
    (lockScanTM k lane guide fwdRule bwdWrite).step c' =
      ((lockScanTM k lane guide fwdRule bwdWrite).tr q c'.inputSymbol c'.workTapeSymbols).apply c' :=
  MultiTapeTM.step_of_some hq

/-- A lane action applied: the lanes write and move, every other tape is untouched. -/
private lemma ls_apply (c' : Cfg k Γ (LockScanPhase V) x) (wr : Fin k → Option (Option Γ))
    (m : SignType) (q : LockScanPhase V) :
    (⟨0, fun j => if lane j then (wr j, m) else (none, 0), none, some q⟩ :
        Action k Γ (LockScanPhase V)).apply c' =
      { c' with
        state := some q
        workTapes := fun j => if lane j then
            Function.update (c'.workTapes j) (c'.workTapePos j)
              ((wr j).getD (c'.workTapes j (c'.workTapePos j)))
          else c'.workTapes j
        workTapePos := fun j => c'.workTapePos j + if lane j then (m : ℤ) else 0 } := by
  apply Cfg.ext
  · rw [Action.apply_state]
  · rw [Action.apply_inputPos, moveInputPos_zero]
  · funext j
    rw [Action.apply_workTapes, Cfg.workTapeSymbols_eq]
    by_cases h : lane j <;> simp [h]
  · funext j
    rw [Action.apply_workTapePos]
    by_cases h : lane j <;> simp [h]
  · rw [Action.apply_output]; simp

private lemma ls_trace (c : Cfg k Γ (LockScanPhase V) x) (hstate : c.state = some .fwd)
    (d : ℕ) (v : V)
    (hcont : ∀ r < d, (fwdRule (lsCol lane c r)).1 = none)
    (hturn : (fwdRule (lsCol lane c d)).1 = some v)
    (hg : lane guide = true)
    (hguide : ∀ r < d,
      ((fwdRule (lsCol lane c r)).2 guide).getD (lsCol lane c r guide) ≠ none)
    (hleft : c.workTapes guide (c.workTapePos guide - 1) = none) (t : ℕ) :
    (lockScanTM k lane guide fwdRule bwdWrite).runFrom c t =
      catalogTrace (lsF lane fwdRule c) (lsR lane fwdRule bwdWrite c d v)
        (lsD lane fwdRule bwdWrite c d v) d t := by
  have h0 : lsF lane fwdRule c 0 = c := by
    apply Cfg.ext
    · simp [lsF, hstate]
    · rfl
    · funext j q
      simp only [lsF, Nat.cast_zero, add_zero]
      rw [if_neg (by omega)]
    · funext j; simp [lsF]
    · rfl
  conv_lhs => rw [← h0]
  apply catalog_trace_run
  · intro r hr
    have hcol : (lsF lane fwdRule c r).workTapeSymbols = lsCol lane c r := by
      funext j
      rw [Cfg.workTapeSymbols_eq]
      by_cases h : lane j <;> simp [lsF, lsCol, h]
    rw [ls_step lane guide fwdRule bwdWrite _ .fwd (by simp [lsF])]
    simp only [lockScanTM, hcol, hcont r hr]
    rw [ls_apply]
    apply Cfg.ext
    · rfl
    · rfl
    · funext j q
      by_cases h : lane j
      · simp only [lsF, h, if_true, true_and, Function.update_apply]
        by_cases hq : q = c.workTapePos j + r
        · subst hq
          simp [lsFwdCell]
        · simp only [hq, if_false]
          split_ifs <;> first | rfl | omega
      · simp [lsF, h]
    · funext j
      by_cases h : lane j <;> simp [lsF, h] <;> omega
    · rfl
  · have hcol : (lsF lane fwdRule c d).workTapeSymbols = lsCol lane c d := by
      funext j
      rw [Cfg.workTapeSymbols_eq]
      by_cases h : lane j <;> simp [lsF, lsCol, h]
    rw [ls_step lane guide fwdRule bwdWrite _ .fwd (by simp [lsF])]
    simp only [lockScanTM, hcol, hturn]
    rw [ls_apply]
    apply Cfg.ext
    · rfl
    · rfl
    · funext j q
      by_cases h : lane j
      · simp only [lsF, lsR, lsTape, h, if_true, true_and, Function.update_apply]
        by_cases hq : q = c.workTapePos j + d
        · subst hq
          simp [lsFwdCell]
        · simp only [hq, if_false]
          split_ifs <;> first | rfl | omega
      · simp [lsF, lsR, lsTape, h]
    · funext j
      by_cases h : lane j <;> simp [lsF, lsR, h] <;> omega
    · rfl
  · intro r hr
    have hcell : (lsR lane fwdRule bwdWrite c d v (r + 1)).workTapeSymbols guide =
        ((fwdRule (lsCol lane c r)).2 guide).getD (lsCol lane c r guide) := by
      rw [Cfg.workTapeSymbols_eq]
      simp only [lsR, lsTape, hg, if_true, true_and, lsFwdCell, lsCol]
      rw [if_pos (by omega), if_neg (by omega)]
      simp
    obtain ⟨γ, hγ⟩ := Option.ne_none_iff_exists'.mp (hguide r hr)
    rw [ls_step lane guide fwdRule bwdWrite _ (.bwd v) (by simp [lsR])]
    simp only [lockScanTM, hcell, hγ]
    rw [ls_apply]
    apply Cfg.ext
    · rfl
    · rfl
    · funext j q
      by_cases h : lane j
      · simp only [lsR, lsTape, h, if_true, true_and, Function.update_apply]
        by_cases hq : q = c.workTapePos j + (((r + 1 : ℕ) : ℤ) - 1)
        · subst hq
          split_ifs <;> first | rfl | omega
        · simp only [hq, if_false]
          split_ifs <;> first | rfl | omega
      · simp [lsR, lsTape, h]
    · funext j
      by_cases h : lane j <;> simp [lsR, h] <;> omega
    · rfl
  · have hcell : (lsR lane fwdRule bwdWrite c d v 0).workTapeSymbols guide = none := by
      rw [Cfg.workTapeSymbols_eq]
      simp only [lsR, lsTape, hg, if_true, true_and, Nat.cast_zero, zero_sub]
      rw [if_neg (by omega)]
      simpa [sub_eq_add_neg] using hleft
    rw [ls_step lane guide fwdRule bwdWrite _ (.bwd v) (by simp [lsR])]
    simp only [lockScanTM, hcell]
    rw [ls_apply]
    apply Cfg.ext
    · rfl
    · rfl
    · funext j q
      by_cases h : lane j
      · simp only [lsR, lsD, h, if_true, Option.getD_none, Function.update_eq_self]
      · simp [lsR, lsD, h]
    · funext j
      by_cases h : lane j <;> simp [lsR, lsD, h]
    · rfl
  · rw [ls_step lane guide fwdRule bwdWrite _ (.done v) (by simp [lsD])]
    simp only [lockScanTM]
    apply Cfg.ext
    · rw [Action.apply_state]; rfl
    · rw [Action.apply_inputPos, moveInputPos_zero]
    · funext j; rw [Action.apply_workTapes, Cfg.workTapeSymbols_eq]; simp
    · funext j; rw [Action.apply_workTapePos]; simp
    · rw [Action.apply_output]; simp

/-- **The lockstep scan, framed.** Start in the forward phase. Let `col r` be the column the lanes
read at offset `r` (every other tape at its head). Suppose the rule continues on `col r` for
`r < d` and turns on `col d` with verdict `v`, the guide is a lane, the guide's cells at offsets
`r < d` are nonblank once the forward pass has written them, and the guide's left delimiter is
blank. Then the scan is in `done v` at exactly `2d + 2`, with no exit earlier. Each lane's
interval `[pos, pos + d]` holds the forward pass's writes, overwritten by `bwdWrite` on
`[pos, pos + d)`; everything else, the heads, the input position and the output are the
start's. At every time the lane heads stay within `[pos - 1, pos + d]` and the other heads are
fixed.

**Proof sketch.** The shared two-pass trace (`Turing.catalog_trace_run`): `d` forward steps, the
turn, `d` return steps guided by the written guide cells, the entry at the guide's left blank,
and the absorbing exit. Read off its exit configuration and the families' states and heads. -/
theorem lockScanTM_run_ofCfg (c : Cfg k Γ (LockScanPhase V) x) (hstate : c.state = some .fwd)
    (d : ℕ) (v : V)
    (hcont : ∀ r < d, (fwdRule fun j =>
      c.workTapes j (c.workTapePos j + if lane j then (r : ℤ) else 0)).1 = none)
    (hturn : (fwdRule fun j =>
      c.workTapes j (c.workTapePos j + if lane j then (d : ℤ) else 0)).1 = some v)
    (hg : lane guide = true)
    (hguide : ∀ r < d, ((fwdRule fun j =>
      c.workTapes j (c.workTapePos j + if lane j then (r : ℤ) else 0)).2 guide).getD
        (c.workTapes guide (c.workTapePos guide + r)) ≠ none)
    (hleft : c.workTapes guide (c.workTapePos guide - 1) = none) :
    (lockScanTM k lane guide fwdRule bwdWrite).runFrom c (2 * d + 2) =
        { c with
          state := some (.done v)
          workTapes := fun j q =>
            if lane j ∧ c.workTapePos j ≤ q ∧ q ≤ c.workTapePos j + (d : ℤ) then
              (if q < c.workTapePos j + (d : ℤ) then
                (bwdWrite j).getD (((fwdRule fun i => c.workTapes i (c.workTapePos i +
                  if lane i then (((q - c.workTapePos j).toNat : ℕ) : ℤ) else 0)).2 j).getD
                  (c.workTapes j q))
              else ((fwdRule fun i => c.workTapes i (c.workTapePos i +
                  if lane i then (((q - c.workTapePos j).toNat : ℕ) : ℤ) else 0)).2 j).getD
                  (c.workTapes j q))
            else c.workTapes j q } ∧
      (∀ t < 2 * d + 2, ∀ v',
        ((lockScanTM k lane guide fwdRule bwdWrite).runFrom c t).state ≠ some (.done v')) ∧
      (∀ (j : Fin k) (t : ℕ),
        if lane j then
          ((lockScanTM k lane guide fwdRule bwdWrite).runFrom c t).workTapePos j ∈
            Finset.Icc (c.workTapePos j - 1) (c.workTapePos j + (d : ℤ))
        else ((lockScanTM k lane guide fwdRule bwdWrite).runFrom c t).workTapePos j =
          c.workTapePos j) := by
  have htr := ls_trace lane guide fwdRule bwdWrite c hstate d v hcont hturn hg
    (fun r hr => by simpa [lsCol, hg] using hguide r hr) hleft
  refine ⟨?_, ?_, ?_⟩
  · rw [htr]
    simp only [catalogTrace, show ¬2 * d + 2 ≤ d by omega, show ¬2 * d + 2 ≤ 2 * d + 1 by omega,
      if_false]
    apply Cfg.ext
    · rfl
    · rfl
    · funext j q
      simp only [lsD, lsTape, lsFwdCell, Nat.cast_zero, add_zero]
      split_ifs <;> first | rfl | omega
    · rfl
    · rfl
  · intro t ht v' hv
    rw [htr] at hv
    simp only [catalogTrace] at hv
    split_ifs at hv <;> first | (simp [lsF, lsR] at hv) | omega
  · intro j t
    rw [htr]
    simp only [catalogTrace]
    by_cases h : lane j
    · simp only [h, if_true, Finset.mem_Icc]
      split_ifs <;> simp [lsF, lsR, lsD, h] <;> omega
    · simp only [h]
      split_ifs <;> simp [lsF, lsR, lsD, h]

/-- **The lockstep scan's space**: under the framed theorem's hypotheses, at every time each lane
has visited at most `d + 2` cells and every other tape at most one.

**Proof sketch.** The framed theorem's head clause confines a lane head to `[pos - 1, pos + d]`
and fixes the others (`Turing.MultiTapeTM.spaceUsedByTape_le_card_Icc`). -/
theorem lockScanTM_spaceUsedByTape (c : Cfg k Γ (LockScanPhase V) x)
    (hstate : c.state = some .fwd) (d : ℕ) (v : V)
    (hcont : ∀ r < d, (fwdRule fun j =>
      c.workTapes j (c.workTapePos j + if lane j then (r : ℤ) else 0)).1 = none)
    (hturn : (fwdRule fun j =>
      c.workTapes j (c.workTapePos j + if lane j then (d : ℤ) else 0)).1 = some v)
    (hg : lane guide = true)
    (hguide : ∀ r < d, ((fwdRule fun j =>
      c.workTapes j (c.workTapePos j + if lane j then (r : ℤ) else 0)).2 guide).getD
        (c.workTapes guide (c.workTapePos guide + r)) ≠ none)
    (hleft : c.workTapes guide (c.workTapePos guide - 1) = none) (t : ℕ) (j : Fin k) :
    if lane j then (lockScanTM k lane guide fwdRule bwdWrite).spaceUsedByTape c t j ≤ d + 2
    else (lockScanTM k lane guide fwdRule bwdWrite).spaceUsedByTape c t j ≤ 1 := by
  have hh := (lockScanTM_run_ofCfg lane guide fwdRule bwdWrite c hstate d v hcont hturn hg hguide
    hleft).2.2 j
  by_cases h : lane j
  · simp only [h, if_true] at hh ⊢
    have hb := MultiTapeTM.spaceUsedByTape_le_card_Icc _ c t j _ _ fun u _ => hh u
    have he : (c.workTapePos j + (d : ℤ) + 1 - (c.workTapePos j - 1)).toNat = d + 2 := by omega
    exact he ▸ hb
  · simp only [h] at hh ⊢
    have hb := MultiTapeTM.spaceUsedByTape_le_card_Icc _ c t j (c.workTapePos j) (c.workTapePos j)
      fun u _ => by rw [hh u]; exact Finset.mem_Icc.mpr ⟨le_rfl, le_rfl⟩
    simpa using hb

end

/-- A configuration at a live anchor whose action is stationary is fixed by a step. -/
private lemma ls_done_fixed {Γ S : Type*} {x : List Γ} (M : MultiTapeTM k Γ S)
    (c : Cfg k Γ S x) (q : S) (hq : c.state = some q)
    (htr : ∀ inp w, M.tr q inp w = ⟨0, fun _ => (none, 0), none, some q⟩) : M.step c = c := by
  rw [MultiTapeTM.step_of_some hq, htr]
  apply Cfg.ext
  · rw [Action.apply_state, hq]
  · rw [Action.apply_inputPos, moveInputPos_zero]
  · funext j; rw [Action.apply_workTapes, Cfg.workTapeSymbols_eq]; simp
  · funext j; rw [Action.apply_workTapePos]; simp
  · rw [Action.apply_output]; simp

/-- Once the run reaches a fixed configuration at `T`, every later configuration is one at or
before `T`. -/
private lemma ls_after_fixed {Γ S : Type*} {x : List Γ} (M : MultiTapeTM k Γ S)
    (c : Cfg k Γ S x) (T : ℕ) (hfix : M.step (M.runFrom c T) = M.runFrom c T) (t : ℕ) :
    ∃ t' ≤ T, M.runFrom c t = M.runFrom c t' := by
  by_cases ht : t ≤ T
  · exact ⟨t, ht, rfl⟩
  · refine ⟨T, le_rfl, ?_⟩
    obtain ⟨s, rfl⟩ : ∃ s, t = T + s := ⟨t - T, by omega⟩
    clear ht
    induction s with
    | zero => rfl
    | succ s ih =>
      rw [show T + (s + 1) = (T + s) + 1 by omega, MultiTapeTM.runFrom_succ_eq_step', ih, hfix]

/-! ### Clear as a lockstep scan -/

/-- Clear's lockstep scan: one lane, continue on a symbol, turn at the blank, erase on return. -/
private def lsClear {Symbol : Type*} (k : ℕ) (i : Fin k) :
    MultiTapeTM k Symbol (LockScanPhase Unit) :=
  lockScanTM k (fun j => decide (j = i)) i
    (fun w => ((match w i with | some _ => none | none => some ()), fun _ => none))
    (fun _ => some none)

/-- Clear's phases inside the scan's. -/
private def lsClearPhase : LockScanPhase Unit ↪ SweepPhase :=
  ⟨fun | .fwd => .sweep | .bwd _ => .rewind | .done _ => .done, by
    intro a b h; cases a <;> cases b <;> first | rfl | cases h⟩

/-- An action, read through the published renaming equations. -/
private lemma ls_action_mapState {Symbol S S' : Type*} (a : Action k Symbol S) (f : S → S') :
    a.mapState f = ⟨a.inputTape, a.workTapes, a.output, a.state.map f⟩ := by
  rw [show a.mapState f = ⟨(a.mapState f).inputTape, (a.mapState f).workTapes,
      (a.mapState f).output, (a.mapState f).state⟩ from rfl,
    Action.mapState_inputTape,
    show (a.mapState f).workTapes = a.workTapes from funext (Action.mapState_workTapes f a),
    Action.mapState_output, Action.mapState_state]

private lemma ls_clear_agree {Symbol : Type*} (i : Fin k) (q : LockScanPhase Unit)
    (inp : Option Symbol) (work : Fin k → Option Symbol) :
    (clearTM k i).tr (lsClearPhase q) inp work =
      ((lsClear k i).tr q inp work).mapState lsClearPhase := by
  rw [ls_action_mapState]
  cases q with
  | fwd => cases h : work i <;> simp [clearTM, lsClear, lockScanTM, lsClearPhase, h]
  | bwd u => cases h : work i <;> simp [clearTM, lsClear, lockScanTM, lsClearPhase, h]
  | done u => simp [clearTM, lsClear, lockScanTM, lsClearPhase]

/-- A sweep configuration seen as the scan's, in the forward phase. -/
private def lsClearCfg {Symbol : Type*} {x : List Symbol} (d : Cfg k Symbol SweepPhase x) :
    Cfg k Symbol (LockScanPhase Unit) x :=
  ⟨some .fwd, d.inputPos, d.workTapes, d.workTapePos, d.output⟩

private lemma ls_clear_cfg {Symbol : Type*} {x : List Symbol} (d : Cfg k Symbol SweepPhase x)
    (hstate : d.state = some .sweep) : (lsClearCfg d).mapState lsClearPhase = d := by
  apply Cfg.ext
  · rw [Cfg.mapState_state, hstate]; rfl
  · rw [Cfg.mapState_inputPos]; rfl
  · funext j q; rw [Cfg.mapState_workTapes]; rfl
  · funext j; rw [Cfg.mapState_workTapePos]; rfl
  · rw [Cfg.mapState_output]; rfl

/-- Clear runs as its scan, renamed. -/
private lemma ls_clear_run {Symbol : Type*} {x : List Symbol} (i : Fin k)
    (d : Cfg k Symbol SweepPhase x) (hstate : d.state = some .sweep) (t : ℕ) :
    (clearTM k i).runFrom d t = ((lsClear k i).runFrom (lsClearCfg d) t).mapState lsClearPhase := by
  conv_lhs => rw [← ls_clear_cfg d hstate]
  exact MultiTapeTM.runFrom_mapState_of_agreeOn (lsClear k i) (clearTM k i) lsClearPhase
    (fun _ => True) (fun q _ inp work => ls_clear_agree i q inp work) (lsClearCfg d) t
    (fun _ _ _ _ => trivial)

theorem clearTM_run_ofCfg' {Symbol : Type*} {k : ℕ} {x : List Symbol} (i : Fin k) (w : List Symbol)
    (d : Cfg k Symbol SweepPhase x) (hstate : d.state = some SweepPhase.sweep)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
      d.workTapes i (d.workTapePos i + p) = FinTM.bufferTape w p) :
    (clearTM k i).runFrom d (2 * w.length + 2) =
        { d with
          state := some SweepPhase.done
          workTapes := fun j q =>
            if j = i ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (w.length : ℤ)
              then none
            else d.workTapes j q } ∧
      (∀ t < 2 * w.length + 2,
        ((clearTM k i).runFrom d t).state ≠ some SweepPhase.done) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * w.length + 2 →
        if j = i then
          ((clearTM k i).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (w.length : ℤ))
        else ((clearTM k i).runFrom d t).workTapePos j = d.workTapePos j) := by
  have hw' : ∀ (r : ℕ) (hr : r < w.length), d.workTapes i (d.workTapePos i + r) = some (w[r]'hr) := by
    intro r hr
    rw [hw r (by omega) (by omega), FinTM.bufferTape_nat, List.getElem?_eq_getElem hr]
  have hend : d.workTapes i (d.workTapePos i + w.length) = none := by
    rw [hw w.length (by omega) le_rfl, FinTM.bufferTape_nat, List.getElem?_eq_none le_rfl]
  have hg := lockScanTM_run_ofCfg (fun j => decide (j = i)) i
    (fun w => ((match w i with | some _ => none | none => some ()), fun _ => none))
    (fun _ => some none) (lsClearCfg d) rfl w.length ()
    (fun r hr => by
      show (match d.workTapes i (d.workTapePos i + if decide (i = i) then (r : ℤ) else 0) with
        | some _ => none | none => some ()) = none
      rw [if_pos (decide_eq_true rfl), hw' r hr])
    (by
      show (match d.workTapes i (d.workTapePos i + if decide (i = i) then (w.length : ℤ) else 0) with
        | some _ => none | none => some ()) = some ()
      rw [if_pos (decide_eq_true rfl), hend])
    (by simp)
    (fun r hr => by
      show (none : Option (Option Symbol)).getD (d.workTapes i (d.workTapePos i + r)) ≠ none
      rw [Option.getD_none, hw' r hr]; exact Option.some_ne_none _)
    (by simpa [sub_eq_add_neg] using hw (-1) (by omega) (by omega))
  refine ⟨?_, ?_, ?_⟩
  · rw [ls_clear_run i d hstate, hg.1]
    apply Cfg.ext
    · rw [Cfg.mapState_state]; rfl
    · rw [Cfg.mapState_inputPos]; rfl
    · funext j q
      rw [Cfg.mapState_workTapes]
      simp only [lsClearCfg, Option.getD_none, Option.getD_some, decide_eq_true_eq]
      by_cases hj : j = i
      · subst hj
        simp only [true_and]
        split_ifs <;> first | rfl | omega
      · simp [hj]
    · funext j; rw [Cfg.mapState_workTapePos]; rfl
    · rw [Cfg.mapState_output]; rfl
  · intro t ht h
    rw [ls_clear_run i d hstate, Cfg.mapState_state] at h
    cases hs : ((lsClear k i).runFrom (lsClearCfg d) t).state with
    | none => rw [hs] at h; cases h
    | some q =>
      rw [hs] at h
      cases q with
      | fwd => cases h
      | bwd u => cases h
      | done u => exact hg.2.1 t ht u hs
  · intro j t _
    have hh := hg.2.2 j t
    rw [ls_clear_run i d hstate, Cfg.mapState_workTapePos]
    by_cases hj : j = i
    · subst hj; simpa using hh
    · simpa [hj] using hh

/-! ### Increment and compare as lockstep scans -/

/-- The flag rows' phases inside the scan's. -/
private def lsFlagPhase : LockScanPhase Bool ↪ FlagPhase :=
  ⟨fun | .fwd => .run | .bwd b => .rewind b | .done b => .done b, by
    intro a b h; cases a <;> cases b <;> first | rfl | (cases h; rfl) | cases h⟩

/-- A flag configuration seen as the scan's, in the forward phase. -/
private def lsFlagCfg {Symbol : Type*} {x : List Symbol} (d : Cfg k Symbol FlagPhase x) :
    Cfg k Symbol (LockScanPhase Bool) x :=
  ⟨some .fwd, d.inputPos, d.workTapes, d.workTapePos, d.output⟩

private lemma ls_flag_cfg {Symbol : Type*} {x : List Symbol} (d : Cfg k Symbol FlagPhase x)
    (hstate : d.state = some .run) : (lsFlagCfg d).mapState lsFlagPhase = d := by
  apply Cfg.ext
  · rw [Cfg.mapState_state, hstate]; rfl
  · rw [Cfg.mapState_inputPos]; rfl
  · funext j q; rw [Cfg.mapState_workTapes]; rfl
  · funext j; rw [Cfg.mapState_workTapePos]; rfl
  · rw [Cfg.mapState_output]; rfl

/-- Increment's lockstep scan: one lane; flip `true` to `false` and continue, write `true` over
the first `false` and turn with success, turn at the blank with overflow. -/
private def lsInc (k : ℕ) (i : Fin k) : MultiTapeTM k Bool (LockScanPhase Bool) :=
  lockScanTM k (fun j => decide (j = i)) i
    (fun w => match w i with
      | some true => (none, fun _ => some (some false))
      | some false => (some true, fun _ => some (some true))
      | none => (some false, fun _ => none))
    (fun _ => none)

private lemma ls_inc_agree (i : Fin k) (q : LockScanPhase Bool) (inp : Option Bool)
    (work : Fin k → Option Bool) :
    (incrementTM k i).tr (lsFlagPhase q) inp work =
      ((lsInc k i).tr q inp work).mapState lsFlagPhase := by
  rw [ls_action_mapState]
  cases q with
  | fwd =>
    rcases h : work i with _ | _ | _ <;>
      simp [incrementTM, lsInc, lockScanTM, lsFlagPhase, h]
  | bwd u => cases h : work i <;> simp [incrementTM, lsInc, lockScanTM, lsFlagPhase, h]
  | done u => simp [incrementTM, lsInc, lockScanTM, lsFlagPhase]

private lemma ls_inc_run {x : List Bool} (i : Fin k) (d : Cfg k Bool FlagPhase x)
    (hstate : d.state = some .run) (t : ℕ) :
    (incrementTM k i).runFrom d t = ((lsInc k i).runFrom (lsFlagCfg d) t).mapState lsFlagPhase := by
  conv_lhs => rw [← ls_flag_cfg d hstate]
  exact MultiTapeTM.runFrom_mapState_of_agreeOn (lsInc k i) (incrementTM k i) lsFlagPhase
    (fun _ => True) (fun q _ inp work => ls_inc_agree i q inp work) (lsFlagCfg d) t
    (fun _ _ _ _ => trivial)

private theorem catalog_increment_framed' {k : ℕ} {x : List Bool} (i : Fin k)
    (w : List Bool) (d : Cfg k Bool FlagPhase x)
    (hstate : d.state = some FlagPhase.run)
    (hw : ∀ p : ℤ, -1 ≤ p → p ≤ (w.length : ℤ) →
      d.workTapes i (d.workTapePos i + p) = FinTM.bufferTape w p)
    (p : ℕ) (tail : Option (List Bool))
    (hsplit : w = List.replicate p true ++ tail.elim [] (false :: ·)) :
    (incrementTM k i).runFrom d (2 * p + 2) =
        { d with
          state := some (FlagPhase.done tail.isSome)
          workTapes := fun j q =>
            if j = i ∧ d.workTapePos j ≤ q ∧ q < d.workTapePos j + (w.length : ℤ)
              then FinTM.bufferTape (List.replicate p false ++ tail.elim [] (true :: ·))
                (q - d.workTapePos j)
            else d.workTapes j q } ∧
      (∀ t < 2 * p + 2, ∀ b : Bool,
        ((incrementTM k i).runFrom d t).state ≠ some (FlagPhase.done b)) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * p + 2 →
        if j = i then
          ((incrementTM k i).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (p : ℤ))
        else ((incrementTM k i).runFrom d t).workTapePos j = d.workTapePos j) := by
  have hlen : p ≤ w.length := by rw [hsplit]; cases tail <;> simp
  have htrue : ∀ r < p, d.workTapes i (d.workTapePos i + r) = some true := by
    intro r hr
    rw [hw r (by omega) (by omega), FinTM.bufferTape_nat, hsplit,
      List.getElem?_append_left (by simpa using hr), List.getElem?_replicate_of_lt hr]
  have hturn : d.workTapes i (d.workTapePos i + p) = tail.map (fun _ => false) := by
    rw [hw p (by omega) (by omega), FinTM.bufferTape_nat, hsplit,
      List.getElem?_append_right (by simp)]
    cases tail <;> simp
  have hg := lockScanTM_run_ofCfg (fun j => decide (j = i)) i
    (fun w => match w i with
      | some true => (none, fun _ => some (some false))
      | some false => (some true, fun _ => some (some true))
      | none => (some false, fun _ => none))
    (fun _ => none) (lsFlagCfg d) rfl p tail.isSome
    (fun r hr => by simp [lsFlagCfg, htrue r hr])
    (by cases tail <;> simp [lsFlagCfg, hturn])
    (by simp)
    (fun r hr => by simp [lsFlagCfg, htrue r hr])
    (by simpa [lsFlagCfg, sub_eq_add_neg] using hw (-1) (by omega) (by omega))
  refine ⟨?_, ?_, ?_⟩
  · rw [ls_inc_run i d hstate, hg.1]
    apply Cfg.ext
    · rw [Cfg.mapState_state]; rfl
    · rw [Cfg.mapState_inputPos]; rfl
    · funext j q
      rw [Cfg.mapState_workTapes]
      simp only [lsFlagCfg, Option.getD_none, decide_eq_true_eq]
      by_cases hj : j = i
      · subst hj
        by_cases hq : d.workTapePos j ≤ q
        · obtain ⟨m, rfl⟩ : ∃ m : ℕ, q = d.workTapePos j + m :=
            ⟨(q - d.workTapePos j).toNat, by omega⟩
          have hm : (d.workTapePos j + (m : ℤ) - d.workTapePos j).toNat = m := by omega
          simp only [hm, if_true, ite_self, true_and, add_sub_cancel_left, Int.toNat_natCast]
          rcases lt_trichotomy m p with hmp | rfl | hmp
          · rw [htrue m hmp, if_pos (by omega), if_pos (by omega), FinTM.bufferTape_nat,
              List.getElem?_append_left (by simpa using hmp), List.getElem?_replicate_of_lt hmp]
            rfl
          · rw [hturn, if_pos (by omega)]
            cases tail with
            | none =>
              have hl : w.length = m := by rw [hsplit]; simp
              rw [if_neg (by omega)]
              rfl
            | some rest =>
              have hl : w.length = m + 1 + rest.length := by rw [hsplit]; simp; omega
              rw [if_pos (by omega), FinTM.bufferTape_nat, List.getElem?_append_right (by simp)]
              simp
          · rw [if_neg (by omega)]
            split_ifs with h
            · rw [hw m (by omega) (by omega), FinTM.bufferTape_nat, FinTM.bufferTape_nat, hsplit]
              cases tail with
              | none => rw [hsplit] at h; simp at h; omega
              | some rest =>
                obtain ⟨n, rfl⟩ : ∃ n, m = p + n + 1 := ⟨m - p - 1, by omega⟩
                rw [List.getElem?_append_right (by simp; omega),
                  List.getElem?_append_right (by simp; omega)]
                simp only [List.length_replicate, Option.elim]
                rw [show p + n + 1 - p = n + 1 by omega]
                rfl
            · rfl
        · rw [if_neg (by omega), if_neg (by omega)]
      · simp [hj]
    · funext j; rw [Cfg.mapState_workTapePos]; rfl
    · rw [Cfg.mapState_output]; rfl
  · intro t ht b h
    rw [ls_inc_run i d hstate, Cfg.mapState_state] at h
    cases hs : ((lsInc k i).runFrom (lsFlagCfg d) t).state with
    | none => rw [hs] at h; cases h
    | some q =>
      rw [hs] at h
      cases q with
      | fwd => cases h
      | bwd u => cases h
      | done u => exact hg.2.1 t ht u hs
  · intro j t _
    have hh := hg.2.2 j t
    rw [ls_inc_run i d hstate, Cfg.mapState_workTapePos]
    by_cases hj : j = i
    · subst hj; simpa using hh
    · simpa [hj] using hh

/-- Compare's lockstep scan: lanes `fst` and `snd`, continue on equal symbols, turn at the first
mismatch or end with the verdict "both ended", return guided by `fst`. -/
private def lsCmp {Symbol : Type*} [DecidableEq Symbol] (k : ℕ) (fst snd : Fin k) :
    MultiTapeTM k Symbol (LockScanPhase Bool) :=
  lockScanTM k (fun j => decide (j = fst ∨ j = snd)) fst
    (fun w => (match w fst, w snd with
      | some a, some b => if a = b then none else some false
      | none, none => some true
      | _, _ => some false, fun _ => none))
    (fun _ => none)

private lemma ls_cmp_agree {Symbol : Type*} [DecidableEq Symbol] (fst snd : Fin k)
    (q : LockScanPhase Bool) (inp : Option Symbol) (work : Fin k → Option Symbol) :
    (compareTM k fst snd).tr (lsFlagPhase q) inp work =
      ((lsCmp k fst snd).tr q inp work).mapState lsFlagPhase := by
  rw [ls_action_mapState]
  cases q with
  | fwd =>
    rcases h1 : work fst with _ | a <;> rcases h2 : work snd with _ | b <;>
      (try by_cases hab : a = b) <;>
      simp [compareTM, lsCmp, lockScanTM, lsFlagPhase, h1, h2, *]
  | bwd u => cases h : work fst <;> simp [compareTM, lsCmp, lockScanTM, lsFlagPhase, h]
  | done u => simp [compareTM, lsCmp, lockScanTM, lsFlagPhase]

private lemma ls_cmp_run {Symbol : Type*} [DecidableEq Symbol] {x : List Symbol}
    (fst snd : Fin k) (d : Cfg k Symbol FlagPhase x) (hstate : d.state = some .run) (t : ℕ) :
    (compareTM k fst snd).runFrom d t =
      ((lsCmp k fst snd).runFrom (lsFlagCfg d) t).mapState lsFlagPhase := by
  conv_lhs => rw [← ls_flag_cfg d hstate]
  exact MultiTapeTM.runFrom_mapState_of_agreeOn (lsCmp k fst snd) (compareTM k fst snd)
    lsFlagPhase (fun _ => True) (fun q _ inp work => ls_cmp_agree fst snd q inp work)
    (lsFlagCfg d) t (fun _ _ _ _ => trivial)

theorem compareTM_run_stop_ofCfg' {Symbol : Type*} [DecidableEq Symbol] {k : ℕ}
    {x : List Symbol} (fst snd : Fin k) (hne : fst ≠ snd) (u v : List Symbol)
    (d : Cfg k Symbol FlagPhase x) (hstate : d.state = some FlagPhase.run)
    (hu : ∀ p : ℤ, -1 ≤ p → p ≤ (u.length : ℤ) →
      d.workTapes fst (d.workTapePos fst + p) = FinTM.bufferTape u p)
    (hv : ∀ p : ℤ, -1 ≤ p → p ≤ (v.length : ℤ) →
      d.workTapes snd (d.workTapePos snd + p) = FinTM.bufferTape v p) :
    (compareTM k fst snd).runFrom d (2 * compareStop u v + 2) =
        { d with state := some (FlagPhase.done (decide (u = v))) } ∧
      (∀ t < 2 * compareStop u v + 2, ∀ b,
        ((compareTM k fst snd).runFrom d t).state ≠ some (FlagPhase.done b)) ∧
      (∀ (j : Fin k) (t : ℕ), t ≤ 2 * compareStop u v + 2 →
        if j = fst ∨ j = snd then
          ((compareTM k fst snd).runFrom d t).workTapePos j ∈
            Finset.Icc (d.workTapePos j - 1) (d.workTapePos j + (compareStop u v : ℤ))
        else ((compareTM k fst snd).runFrom d t).workTapePos j = d.workTapePos j) := by
  have _ := hne
  have hle := compareStop_le u v
  obtain ⟨hpre, hstop⟩ := compareStop_spec u v
  have hu' : ∀ r : ℕ, r ≤ u.length →
      d.workTapes fst (d.workTapePos fst + r) = u[r]? := fun r hr => by
    rw [hu r (by omega) (by exact_mod_cast hr), FinTM.bufferTape_nat]
  have hv' : ∀ r : ℕ, r ≤ v.length →
      d.workTapes snd (d.workTapePos snd + r) = v[r]? := fun r hr => by
    rw [hv r (by omega) (by exact_mod_cast hr), FinTM.bufferTape_nat]
  have key : u[compareStop u v]? = v[compareStop u v]? ↔ u = v := by
    constructor
    · intro h
      have hnone : u[compareStop u v]? = none := by
        cases hx : u[compareStop u v]? with
        | none => rfl
        | some b => exact absurd ⟨b, hx, h ▸ hx⟩ hstop
      have hu0 : u.length ≤ compareStop u v := List.getElem?_eq_none_iff.mp hnone
      have hv0 : v.length ≤ compareStop u v := List.getElem?_eq_none_iff.mp (h ▸ hnone)
      apply List.ext_getElem?
      intro n
      by_cases hn : n < compareStop u v
      · obtain ⟨b, h1, h2⟩ := hpre n hn
        rw [h1, h2]
      · rw [List.getElem?_eq_none (by omega), List.getElem?_eq_none (by omega)]
    · intro h; rw [h]
  have lf : decide (fst = fst ∨ fst = snd) = true := by simp
  have ls : decide (snd = fst ∨ snd = snd) = true := by simp
  have hg := lockScanTM_run_ofCfg (fun j => decide (j = fst ∨ j = snd)) fst
    (fun w => (match w fst, w snd with
      | some a, some b => if a = b then none else some false
      | none, none => some true
      | _, _ => some false, fun _ => none))
    (fun _ => none) (lsFlagCfg d) rfl (compareStop u v) (decide (u = v))
    (fun r hr => by
      obtain ⟨b, h1, h2⟩ := hpre r hr
      simp only [lsFlagCfg, decide_eq_true_eq, true_or, or_true, if_true]
      rw [hu' r (by omega), hv' r (by omega), h1, h2]
      simp)
    (by
      simp only [lsFlagCfg, decide_eq_true_eq, true_or, or_true, if_true]
      rw [hu' _ (by omega), hv' _ (by omega)]
      cases ha : u[compareStop u v]? with
      | none =>
        cases hb : v[compareStop u v]? with
        | none => simp [key.mp (ha.trans hb.symm)]
        | some b =>
          have : u ≠ v := fun he => by rw [← key] at he; rw [ha, hb] at he; cases he
          simp [this]
      | some a =>
        cases hb : v[compareStop u v]? with
        | none =>
          have : u ≠ v := fun he => by rw [← key] at he; rw [ha, hb] at he; cases he
          simp [this]
        | some b =>
          by_cases hab : a = b
          · subst hab; exact absurd ⟨a, ha, hb⟩ hstop
          · have : u ≠ v := fun he => by
              rw [← key] at he; rw [ha, hb] at he; exact hab (Option.some.inj he)
            simp [hab, this])
    lf
    (fun r hr => by
      obtain ⟨b, h1, _⟩ := hpre r hr
      show (none : Option (Option Symbol)).getD (d.workTapes fst (d.workTapePos fst + r)) ≠ none
      rw [Option.getD_none, hu' r (by omega), h1]; exact Option.some_ne_none _)
    (by simpa [lsFlagCfg, sub_eq_add_neg] using hu (-1) (by omega) (by omega))
  refine ⟨?_, ?_, ?_⟩
  · rw [ls_cmp_run fst snd d hstate, hg.1]
    apply Cfg.ext
    · rw [Cfg.mapState_state]; rfl
    · rw [Cfg.mapState_inputPos]; rfl
    · funext j q
      rw [Cfg.mapState_workTapes]
      simp only [lsFlagCfg, Option.getD_none]
      split_ifs <;> rfl
    · funext j; rw [Cfg.mapState_workTapePos]; rfl
    · rw [Cfg.mapState_output]; rfl
  · intro t ht b h
    rw [ls_cmp_run fst snd d hstate, Cfg.mapState_state] at h
    cases hs : ((lsCmp k fst snd).runFrom (lsFlagCfg d) t).state with
    | none => rw [hs] at h; cases h
    | some q =>
      rw [hs] at h
      cases q with
      | fwd => cases h
      | bwd u => cases h
      | done u => exact hg.2.1 t ht u hs
  · intro j t _
    have hh := hg.2.2 j t
    rw [ls_cmp_run fst snd d hstate, Cfg.mapState_workTapePos]
    by_cases hj : j = fst ∨ j = snd
    · rw [if_pos hj]
      have h2 : decide (j = fst ∨ j = snd) = true := decide_eq_true hj
      simp only [h2, if_true] at hh
      exact hh
    · rw [if_neg hj]
      have h2 : decide (j = fst ∨ j = snd) = false := decide_eq_false hj
      simp only [h2, Bool.false_eq_true, if_false] at hh
      exact hh
