+/-- **Phase alphabet of the lockstep scan**: the forward pass, the return pass carrying the
+verdict chosen at the turn, and the live exit anchors, one per verdict. -/
+inductive LockScanPhase (V : Type) where
+  /-- the forward pass -/
+  | fwd
+  /-- the return pass, carrying the verdict -/
+  | bwd (v : V)
+  /-- the live exit anchors, one per verdict -/
+  | done (v : V)
+deriving DecidableEq
+
+/-- **The lockstep scan.** The lane tapes (`lane j = true`) move together. In the forward pass
+the rule reads the whole column under the heads and either writes and steps the lanes right,
+or writes, steps them left and turns with a verdict `v`. The return pass steps the lanes left,
+writing `bwdWrite`, while the guide tape reads a symbol; at the guide's left blank it steps
+right into `done v`. Other tapes never move or change. -/
+def lockScanTM {Γ : Type*} {V : Type} (k : ℕ) (lane : Fin k → Bool) (guide : Fin k)
+    (fwdRule : (Fin k → Option Γ) → Option V × (Fin k → Option (Option Γ)))
+    (bwdWrite : Fin k → Option (Option Γ)) : MultiTapeTM k Γ (LockScanPhase V) where
+  q₀ := .fwd
+  tr := fun q _ w =>
+    match q with
+    | .fwd =>
+      match (fwdRule w).1 with
+      | none => ⟨0, fun j => if lane j then ((fwdRule w).2 j, SignType.pos) else (none, 0),
+          none, some .fwd⟩
+      | some v => ⟨0, fun j => if lane j then ((fwdRule w).2 j, SignType.neg) else (none, 0),
+          none, some (.bwd v)⟩
+    | .bwd v =>
+      match w guide with
+      | some _ => ⟨0, fun j => if lane j then (bwdWrite j, SignType.neg) else (none, 0),
+          none, some (.bwd v)⟩
+      | none => ⟨0, fun j => if lane j then (none, SignType.pos) else (none, 0),
+          none, some (.done v)⟩
+    | .done v => ⟨0, fun _ => (none, 0), none, some (.done v)⟩

+/-- The forward family: `r` forward steps taken. -/
+private def lsF (c : Cfg k Γ (LockScanPhase V) x) (r : ℕ) : Cfg k Γ (LockScanPhase V) x :=
+  { c with
+    state := some .fwd

+/-- The tapes once the return pass has rewritten the offsets `[r, d)`. -/

+  fun j q =>
